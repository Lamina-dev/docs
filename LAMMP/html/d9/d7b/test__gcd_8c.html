118 {
119 // 覆盖单/双 limb 与低位为 0 的形态组合。强制高位的随机测试几乎不可能
120 // 命中这些分支（单个 limb 为 0 的概率约 2^-64），gcd_22 曾因此在
121 // u0==v0==0 的路径上漏检（漏乘 B），此用例系统性遍历移位组合。
122 // 期望 = gcd(x,y)*2^min(a,b)（x,y 奇）。构造约束：x,y < 2^63 且
123 // s <= 64，保证 x*2^s < 2^127 不溢出 128 位（溢出截断会使期望式失效）
124 u64 seed = 0x51ce1b7ea4df7c21ull;
125 mp_ptr u = alloc_limbs(2), v = alloc_limbs(2), d = alloc_limbs(2), e = alloc_limbs(2);
126 for (int a = 0; a <= 64; ++a) {
127 for (int b = 0; b <= 64; b += 4) {
128 u64 x = (xorshift64(&seed) & (((u64)1 << 63) - 1)) | 1; // 奇且 < 2^63
129 u64 y = (xorshift64(&seed) & (((u64)1 << 63) - 1)) | 1;
130 shifted_pair(x, a, u);
131 shifted_pair(y, b, v);
132 mp_size_t gn = lmmp_gcd_22_(d, u, v);
133 // 期望 = gcd(x,y)*2^min(a,b)（标量 gcd + 移位装配，精确已知）
134 int m = a < b ? a : b;
135 u64 g = gcd_u64(x, y);
136 shifted_pair(g, m, e);
137 mp_size_t en = norm_len(e, 2);
138 TEST_CHECK(mpn_eq(d, gn, e, en));
139 }
