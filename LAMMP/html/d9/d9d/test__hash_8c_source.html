32TEST_CASE("hash/sanity", siphash_xxhash) {
33 mp_limb_t in[8];
34 u64 seed = 0x1234567890abcdefull;
35 for (size_t x_index = 0; x_index < ARRAY_SIZE(in); ++x_index) {
36 in[x_index] = xorshift64(&seed);
37 }
38
39 u64 key2[2] = {0x1111111111111111ull, 0x2222222222222222ull};
40 u64 key1[1] = {0x3333333333333333ull};
41
42 u64 h1 = lmmp_siphash24_(in, 8, key2);
43 u64 h2 = lmmp_siphash24_(in, 8, key2);
44 TEST_CHECK_EQ(h1, h2);
45
46 u64 h3 = lmmp_siphash24_(in, 8, NULL);
47 TEST_CHECK_MSG(h3 != h1, "siphash key changes hash");
48
49 u64 x1 = lmmp_xxhash_(in, 8, key1);
50 u64 x2 = lmmp_xxhash_(in, 8, key1);
51 TEST_CHECK_EQ(x1, x2);
52
53 u64 x3 = lmmp_xxhash_(in, 8, NULL);
54 TEST_CHECK_MSG(x3 != x1, "xxhash key changes hash");
55
56 // 空输入与 NULL key
57 TEST_CHECK_MSG(lmmp_siphash24_(NULL, 0, NULL) != 0, "siphash empty nonzero");
58 TEST_CHECK_MSG(lmmp_xxhash_(NULL, 0, NULL) != 0, "xxhash empty nonzero");
59
60 // 变更输入应改变哈希
61 in[0] ^= 1;
62 TEST_CHECK_MSG(lmmp_xxhash_(in, 8, key1) != x1, "xxhash input change");
63}
