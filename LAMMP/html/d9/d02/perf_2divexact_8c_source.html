28/*
29 精确除法性能：lmmp_divexact_ 公共入口，尺寸全部落在 basecase 档
30 （nn < DIVEXACT_NN_THRESHOLD 且 dn < DIVEXACT_BASECASE_THRESHOLD）。
31
32 输入构造 n = q*d（q、d 顶 limb 置位），确保整除且商规模确定；
33 计时前用 memcmp 校验 dst == q。
34*/
35
36typedef struct {
37 mp_ptr q;
38 mp_ptr d;
39 mp_ptr np;
40 mp_ptr dst;
41 mp_size_t qn;
42 mp_size_t qa; // 实际非零商 limb 数（顶 limb 恒 0）
43} DivexactCtx;
44
45static DivexactCtx make_divexact_ctx(mp_size_t nn, mp_size_t dn, mp_limb_t seed) {
46 DivexactCtx c;
47 c.qn = nn - dn + 1;
48 c.qa = c.qn - 1;
49 c.q = alloc_limbs(c.qa);
50 c.d = alloc_limbs(dn);
51 c.np = alloc_limbs(nn);
52 c.dst = alloc_limbs(c.qn + 1);
53 lmmp_seed_random_(c.q, c.qa, seed, 1);
54 lmmp_seed_random_(c.d, dn, seed * 3 + 1, 1);
55 c.d[0] |= 1;
56 /* lmmp_mul_ 要求 na >= nb：窄窗形态商短于除数，需交换 */
57 if (c.qa >= dn)
58 lmmp_mul_(c.np, c.q, c.qa, c.d, dn);
59 else
60 lmmp_mul_(c.np, c.d, dn, c.q, c.qa);
61 c.dst[c.qn] = 0x5eed5eed5eed5eedull;
62 lmmp_divexact_(c.dst, c.np, nn, c.d, dn);
63 if (memcmp(c.dst, c.q, (size_t)c.qa * sizeof(mp_limb_t)) != 0 || c.dst[c.qa] != 0) {
64 printf("divexact bench: WRONG QUOTIENT at nn=%lld dn=%lld\n", (long long)nn,
65 (long long)dn);
66 }
67 return c;
68}

70static void free_divexact_ctx(const DivexactCtx *c) {
71 lmmp_free(c->q);
72 lmmp_free(c->d);
73 lmmp_free(c->np);
74 lmmp_free(c->dst);
75}

77static double bench_divexact(mp_size_t nn, mp_size_t dn, const bench_options *o) {
78 DivexactCtx c = make_divexact_ctx(nn, dn, 0x9e3779b97f4a7c15ull + (mp_limb_t)nn * 31 + dn);
79 double us;
80 BENCH_MEASURE(us, o, lmmp_divexact_(c.dst, c.np, nn, c.d, dn); bench_sink += c.dst[0];);
81 free_divexact_ctx(&c);
82 return us;
83}

100static void run_wide(void) {
101 for (size_t i = 0; i < ARRAY_SIZE(DIVX_WIDE); ++i) {
102 const mp_size_t * sz = DIVX_WIDE[i];
103 report_ab("divexact wide nn,dn", sz[0], sz[1], bench_divexact(sz[0], sz[1], &bench_default), 0);
