40mp_size_t lmmp_factorial_size_(uint n, mp_bitcnt_t* restrict bits) {
41 mp_size_t rn;
42 if (n < 20) {
43 rn = 64;
44 } else {
45 rn = log2_fac_ceil(n);
46 }
47 rn = (rn + LIMB_BITS - 1) / LIMB_BITS + 2; // more two limbs

60mp_size_t lmmp_factors_mul_(mp_ptr restrict dst, mp_size_t rn, fac_ptr restrict fac, uint nfactors) {
61 lmmp_param_assert(dst != NULL && fac != NULL);
62 lmmp_param_assert(rn > 0 && nfactors > 0);
63 if (nfactors <= FACTORS_MUL_N_THRESHOLD) {
64 // 绝大多数情况下，大的质因数的指数都很小，所以这里只需要考虑小的质因数。
65 dst[0] = 1;
66 rn = 1;
67 mp_limb_t t = 1;
68 for (uint i = 0; i < nfactors; i++) {
69 uint f = fac[i].f;
70 uint j = fac[i].j;
71 lmmp_debug_assert(j != 0 && f <= MP_USHORT_MAX);
72#define MAX_T 0xffffffffffff
73 for (uint e = 0; e < j; e++) {
74 t *= f;
75 if (t > MAX_T) {
76 mul_1(dst, rn, t);
77 t = 1;
78 }
79 }
80 }
81 if (t != 1) {
82 mul_1(dst, rn, t);
83 }
84#undef MAX_T
85 return rn;
86 } else {
87 TEMP_DECL;
88 uint new_nfactors = 0;
89 ulongp restrict limbs = TALLOC_TYPE(nfactors / 2 + 1, ulong);
90 ulong t = 1;
91 mp_size_t limbn = 0;
92 for (uint i = 0; i < nfactors; ++i) {
93 uint f = fac[i].f;
94 uint j = fac[i].j;
95 if (j > 1) {
96 fac[new_nfactors].f = f;
97 fac[new_nfactors++].j = j >> 1;
98 }
99 if (j & 1) {
100 t *= f;
101 if (t > MP_UINT_MAX) {
102 limbs[limbn++] = t;
103 t = 1;
104 }
105 }
106 }
107 if (t != 1) {
108 limbs[limbn++] = t;
109 }
110
111 mp_ptr restrict mp = TALLOC_TYPE(limbn * 2, mp_limb_t);
112 mp_size_t mpn = 0;
113
114 if (new_nfactors > 0) {
115 if (limbn > 0) {
116 mpn = lmmp_elem_mul_ulong_(mp, limbs, limbn, mp + limbn);
117 lmmp_debug_assert(rn >= mpn);
118 mp_size_t tn = ((rn - mpn) >> 1) + 1;
119 // 这里根据mpn的大小估计剩余因子乘积的长度，额外分配两倍的tn，以进行平方。
120 mp_ptr restrict tp = BALLOC_TYPE(3 * tn + 3, mp_limb_t);
121 tn = lmmp_factors_mul_(tp, tn, fac, new_nfactors);
122
123 mp_ptr restrict tp2 = tp + tn + 1;
124 lmmp_sqr_(tp2, tp, tn);
125 tn <<= 1;
126 tn -= tp2[tn - 1] == 0;
127 MUL(dst, tp2, tn, mp, mpn);
128 rn = tn + mpn;
129 rn -= dst[rn - 1] == 0;
130 } else {
131 mp_size_t tn = (rn >> 1) + 1;
132 mp_ptr restrict tp = TALLOC_TYPE(tn, mp_limb_t);
133 tn = lmmp_factors_mul_(tp, tn, fac, new_nfactors);
134 lmmp_sqr_(dst, tp, tn);
135 rn = tn << 1;
136 rn -= dst[rn - 1] == 0;
137 }
138 } else {
139 lmmp_debug_assert(limbn > 0);
140 // 这里不能直接乘入dst，因为dst的大小可能小于limbn，导致溢出
141 if (rn >= limbn) {
142 rn = lmmp_elem_mul_ulong_(dst, limbs, limbn, mp);
143 } else {
144 rn = lmmp_elem_mul_ulong_(mp, limbs, limbn, mp + limbn);
145 lmmp_copy(dst, mp, rn);
146 }
147 }
148 TEMP_FREE;
149 return rn;
150 }
151}

153static inline void count_factors(fac_ptr fac, uint nfactors, uint n, uint p) {
154 uint pn = n;
155 uint e = 0;
156 while (pn > 0) {
157 pn /= p;
158 e += pn;
159 }
160 fac[nfactors].f = p;
161 fac[nfactors].j = e;
162}

164mp_size_t lmmp_odd_factorial_uint_(mp_ptr restrict dst, mp_size_t rn, uint n) {
165 lmmp_param_assert(dst != NULL);
166 lmmp_param_assert(rn > 0);
167 lmmp_param_assert(n > MP_USHORT_MAX);
168
169 lmmp_prime_int_table_init_(n);
170 TEMP_B_DECL;
171 uint nfactors = lmmp_prime_size_(n);
172 fac_ptr restrict fac = BALLOC_TYPE(nfactors, fac_t);
173 nfactors = 0;
174
175 prime_cache_t cache;
176 lmmp_prime_cache_init_(&cache, n);
177 while (cache.is_end == 0) {
178 lmmp_prime_cache_next_(&cache);
179 for (uint i = 0; i < cache.size; i++) {
180 // 对于阶乘n!，对于所有小于等于n的质数，贡献都至少为1
181 count_factors(fac, nfactors++, n, cache.pp[i]);
182 }
183 }
184 lmmp_prime_cache_free_(&cache);
185
186 rn = lmmp_factors_mul_(dst, rn, fac, nfactors);
187
188 TEMP_B_FREE;
189 return rn;
190}

192mp_size_t lmmp_factorial_(mp_ptr restrict dst, mp_bitcnt_t bits, mp_size_t rn, uint n) {
193 lmmp_param_assert(dst != NULL);
194 mp_size_t shw = bits / LIMB_BITS;
195 lmmp_param_assert(rn > shw);
196 bits %= LIMB_BITS;
197 lmmp_zero(dst, shw);
198
199 if (n <= NPR_SHORT_LIMIT)
200 rn = lmmp_odd_nPr_ushort_(dst + shw, rn - shw, n, n);
201 else
202 rn = lmmp_odd_factorial_uint_(dst + shw, rn - shw, n);
203
204 if (bits > 0) {
205 dst[shw + rn] = lmmp_shl_(dst + shw, dst + shw, rn, bits);
206 rn += shw + 1;
207 rn -= dst[rn - 1] == 0;
208 } else {
209 rn += shw;
210 }
211 return rn;
