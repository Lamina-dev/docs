112 {
113 // 回归：顶 limb 为 2 的幂时，归一化后顶 limb 恰为 2^63，曾触发
114 // appr_newton 的严格大于断言失败（发布版为 assume UB）
115 u64 seed = 0x9d1f2e3c4b5a6978ull;
116 const mp_size_t na_values[] = {3, 8, 40};
117 for (size_t na_index = 0; na_index < ARRAY_SIZE(na_values); ++na_index) {
118 mp_size_t na = na_values[na_index];
119 const u64 top_values[] = {1ull, 4ull, 1ull << 32, 1ull << 63};
120 for (size_t top_index = 0; top_index < ARRAY_SIZE(top_values); ++top_index) {
121 u64 top = top_values[top_index];
122 const mp_size_t ni0_values[] = {na, na / 2};
123 for (size_t ni0_index = 0; ni0_index < ARRAY_SIZE(ni0_values); ++ni0_index) {
124 mp_size_t ni0 = ni0_values[ni0_index];
125 mp_size_t ni = ni0 < 1 ? 1 : ni0;
126 mp_ptr a = alloc_limbs(na);
127 mp_ptr q = alloc_limbs(na + ni + 2);
128 random_limbs(a, na, &seed);
129 a[0] |= 1;
130 a[na - 1] = top;
131 lmmp_bninv_(q, a, na, ni);
132
133 // 参照（库内构造）：ref = B^(2*(na+ni)) / (numa * B^ni)，
134 // 文档误差 +[0|1]，放宽到 2
135 mp_size_t dn = na + ni;
136 mp_ptr den = alloc_limbs(dn);
137 lmmp_zero(den, dn);
138 memcpy(den + ni, a, (size_t)na * 8); // numa * B^ni
139 mp_size_t nn = 2 * dn + 1;
140 mp_ptr numer = alloc_limbs(nn);
141 lmmp_zero(numer, nn);
142 numer[2 * dn] = 1; // B^(2*(na+ni))
143 mp_ptr quot = alloc_limbs(nn - dn + 1), rem = alloc_limbs(dn);
144 lmmp_div_(quot, rem, numer, nn, den, dn);
145
146 // |got - ref| <= 2
147 mp_size_t ln = dn + 1;
148 mp_ptr diff = alloc_limbs(ln + 1);
149 mp_limb_t bw = lmmp_sub_(diff, q, ln, quot, ln);
150 bool small = true;
151 if (bw) { // got < ref，反向相减
152 lmmp_sub_(diff, quot, ln, q, ln);
153 }
154 for (mp_size_t i = 1; i < ln; ++i) small = small && (diff[i] == 0);
155 TEST_CHECK_MSG(small && diff[0] <= 2, "bninv close to reference quotient");
156
157 lmmp_free(a); lmmp_free(q); lmmp_free(den); lmmp_free(numer);
158 lmmp_free(quot); lmmp_free(rem); lmmp_free(diff);
