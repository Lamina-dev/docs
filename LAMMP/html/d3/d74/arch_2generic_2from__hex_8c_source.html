27// ASCII -> nibble 值表，非法字符映射为 0（契约要求输入全为合法十六进制数字）
28static const mp_byte_t lmmp_hex_val_tab[256] = {
29 ['0'] = 0, ['1'] = 1, ['2'] = 2, ['3'] = 3, ['4'] = 4,
30 ['5'] = 5, ['6'] = 6, ['7'] = 7, ['8'] = 8, ['9'] = 9,
31 ['A'] = 10, ['B'] = 11, ['C'] = 12, ['D'] = 13, ['E'] = 14, ['F'] = 15,
32 ['a'] = 10, ['b'] = 11, ['c'] = 12, ['d'] = 13, ['e'] = 14, ['f'] = 15,
33};

35mp_size_t lmmp_from_hex_(mp_ptr dst, const char* src, mp_size_t len) {
36 lmmp_param_assert(dst != NULL && src != NULL);
37 do {
38 if (len == 0)
39 return 0;
40 } while (src[--len] == '0');
41 ++len;
42
43 mp_size_t nfull = len >> 4;
44 const char* sp = src;
45 for (mp_size_t i = 0; i < nfull; ++i, sp += 16) {
46 mp_limb_t l = 0;
47 // 组内自高向低读，使得 sp[j] 落在第 j 个 nibble
48 for (int j = 15; j >= 0; --j)
49 l = (l << 4) | lmmp_hex_val_tab[(mp_byte_t)sp[j]];
50 dst[i] = l;
51 }
52
53 int r = (int)(len & 15);
54 if (r) {
55 mp_limb_t l = 0;
56 for (int j = r - 1; j >= 0; --j)
57 l = (l << 4) | lmmp_hex_val_tab[(mp_byte_t)sp[j]];
58 dst[nfull] = l;
59 }
60
61 return nfull + (r != 0);
62}
