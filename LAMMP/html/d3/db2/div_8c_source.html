22mp_limb_t lmmp_div_s_(
23 mp_ptr restrict dstq,
24 mp_ptr restrict numa,
25 mp_size_t na,
26 mp_srcptr restrict numb,
27 mp_size_t nb
28) {
29 lmmp_param_assert(na >= nb);
30 lmmp_param_assert(numa != NULL && numb != NULL && nb > 0);
31 lmmp_param_assert(numb[nb - 1] >= LIMB_B_2);
32
33 TEMP_DECL;
34 mp_limb_t nq = na - nb;
35 mp_limb_t qh;
36 if (nq == 0) {
37 qh = lmmp_cmp_(numa, numb, nb) >= 0;
38 if (qh)
39 lmmp_sub_n_(numa, numa, numb, nb);
40 } else if (nb == 1) {
41 qh = lmmp_div_1_s_(dstq, numa, na, *numb);
42 } else if (nb == 2) {
43 qh = lmmp_div_2_s_(dstq, numa, na, numb);
44 } else if (nq < nb) {
45 qh = lmmp_div_s_(dstq, numa + na - 2 * nq, 2 * nq, numb + nb - nq, nq);
46
47 mp_ptr restrict tp = TALLOC_TYPE(nb, mp_limb_t);
48 if (nq > nb - nq)
49 lmmp_mul_(tp, dstq, nq, numb, nb - nq);
50 else
51 lmmp_mul_(tp, numb, nb - nq, dstq, nq);
52
53 mp_limb_t cy = lmmp_sub_n_(numa, numa, tp, nb);
54 if (qh)
55 cy += lmmp_sub_n_(numa + nq, numa + nq, numb, nb - nq);
56
57 while (cy) {
58 qh -= lmmp_sub_1_(dstq, dstq, nq, 1);
59 cy -= lmmp_add_n_(numa, numa, numb, nb);
60 }
61 } else {
62 mp_limb_t inv21 = lmmp_inv_2_1_(numb[nb - 1], numb[nb - 2]);
63 if (nb < DIV_DIVIDE_THRESHOLD)
64 qh = lmmp_div_basecase_(dstq, numa, na, numb, nb, inv21);
65 else if (nb < DIV_MULINV_L_THRESHOLD || na < 2 * DIV_MULINV_N_THRESHOLD)
66 qh = lmmp_div_divide_(dstq, numa, na, numb, nb, inv21);
67 else {
68 mp_limb_t ni = lmmp_div_inv_size_(nq, nb);
69 mp_ptr restrict invappr = TALLOC_TYPE(ni, mp_limb_t);
70 lmmp_inv_prediv_(invappr, numb, nb, ni);
71 qh = lmmp_div_mulinv_(dstq, numa, na, numb, nb, invappr, ni);
72 }
73 }
74 TEMP_FREE;
75 return qh;
76}

78void lmmp_div_(mp_ptr dstq, mp_ptr dstr, mp_srcptr numa, mp_size_t na, mp_srcptr numb, mp_size_t nb) {
79 lmmp_param_assert(na >= nb);
80 lmmp_param_assert(numa != NULL && numb != NULL && nb > 0);
81 lmmp_param_assert(numb[nb - 1] > 0);
82 if (nb == 1) {
83 mp_limb_t rem = lmmp_div_1_(dstq, numa, na, *numb);
84 if (dstr)
85 *dstr = rem;
86 } else if (nb == 2) {
87 mp_limb_t brem[2];
88 brem[0] = numb[0];
89 brem[1] = numb[1];
90 lmmp_div_2_(dstq, numa, na, brem);
91 if (dstr) {
92 dstr[0] = brem[0];
93 dstr[1] = brem[1];
94 }
95 } else {
96 int adjust = numa[na - 1] >= numb[nb - 1];
97 int cnt = lmmp_leading_zeros_(numb[nb - 1]);
98 mp_size_t nq = na + adjust - nb;
99 if (nq == 0) {
100 if (dstr && dstr != numa)
101 lmmp_copy(dstr, numa, nb);
102 if (dstq)
103 dstq[0] = 0;
104 return;
105 }
106 TEMP_DECL;
107
108 if (!dstq)
109 dstq = TALLOC_TYPE(na - nb + 1, mp_limb_t);
110 dstq[na - nb] = 0;
111
112 if (nq >= nb) {
113 mp_ptr restrict numa2 = TALLOC_TYPE(na + 1, mp_limb_t);
114 mp_ptr numb2;
115 if (cnt) {
116 numa2[na] = lmmp_shl_(numa2, numa, na, cnt);
117 numb2 = TALLOC_TYPE(nb, mp_limb_t);
118 lmmp_shl_(numb2, numb, nb, cnt);
119 } else {
120 numa2[na] = 0;
121 lmmp_copy(numa2, numa, na);
122 numb2 = (mp_ptr)numb;
123 }
124
125 mp_limb_t inv21 = lmmp_inv_2_1_(numb2[nb - 1], numb2[nb - 2]);
126 na += adjust;
127
128 if (nb < DIV_DIVIDE_THRESHOLD)
129 lmmp_div_basecase_(dstq, numa2, na, numb2, nb, inv21);
130 else if (nb < DIV_MULINV_L_THRESHOLD || na < 2 * DIV_MULINV_N_THRESHOLD)
131 lmmp_div_divide_(dstq, numa2, na, numb2, nb, inv21);
132 else {
133 mp_limb_t ni = lmmp_div_inv_size_(nq, nb);
134 mp_ptr invappr = TALLOC_TYPE(ni, mp_limb_t);
135 lmmp_inv_prediv_(invappr, numb2, nb, ni);
136 lmmp_div_mulinv_(dstq, numa2, na, numb2, nb, invappr, ni);
137 }
138
139 if (dstr) {
140 if (cnt)
141 lmmp_shr_(dstr, numa2, nb, cnt);
142 else
143 lmmp_copy(dstr, numa2, nb);
144 }
145 } else {
146 // nq=na-nb+adj<nb
147 //-> na+adj>=2nq+1
148 mp_size_t ni = nb - nq;
149 mp_ptr numa2, numb2;
150 mp_ptr tp = TALLOC_TYPE(nb, mp_limb_t);
151 mp_limb_t cy;
152
153 numa2 = TALLOC_TYPE(nq * 2 + 1, mp_limb_t);
154 if (cnt) {
155 numb2 = TALLOC_TYPE(nq, mp_limb_t);
156 lmmp_shl_(numb2, numb + ni, nq, cnt);
157 numb2[0] |= numb[ni - 1] >> (LIMB_BITS - cnt);
158 cy = lmmp_shl_(numa2, numa + na - 2 * nq, 2 * nq, cnt);
159 if (adjust) {
160 numa2[2 * nq] = cy;
161 ++numa2; // numa2[0] is as significant as numa[ni=na-2nq+adjust]
162 } else
163 numa2[0] |= numa[na - 2 * nq - 1] >> (LIMB_BITS - cnt);
164 } else {
165 numb2 = (mp_ptr)numb + ni;
166 lmmp_copy(numa2, numa + na - 2 * nq, 2 * nq);
167 if (adjust) {
168 numa2[2 * nq] = 0;
169 ++numa2;
170 }
171 }
172
173 // now: 0<=numa2<B^2nq, B^nq/2<=numb2<B^nq, and 0<=numa2/numb2<B^nq
174 // ignored bits could be seen as fraction part of numa and numb
175 // we can prove: Q<=Qh<=Q+2
176 // where Q=floor(numa/numb) is the real quotient
177 // Qh=floor(floor(numa)/floor(numb)) as below
178
179 if (nq == 1) {
180 lmmp_div_1_s_(dstq, numa2, 2, *numb2);
181 } else if (nq == 2) {
182 lmmp_div_2_s_(dstq, numa2, 4, numb2);
183 } else {
184 mp_limb_t inv21 = lmmp_inv_2_1_(numb2[nq - 1], numb2[nq - 2]);
185
186 if (nq < DIV_DIVIDE_THRESHOLD)
187 lmmp_div_basecase_(dstq, numa2, 2 * nq, numb2, nq, inv21);
188 else if (nq < DIV_MULINV_N_THRESHOLD)
189 lmmp_div_divide_(dstq, numa2, 2 * nq, numb2, nq, inv21);
190 else {
191 mp_size_t inv_ni = lmmp_div_inv_size_(nq, nq);
192 mp_ptr invappr = tp;
193 lmmp_inv_prediv_(invappr, numb2, nq, inv_ni);
194 lmmp_div_mulinv_(dstq, numa2, 2 * nq, numb2, nq, invappr, inv_ni);
195 }
196 }
197 /*
198 true remainder = partial remainder - quotient * ignored divisor limbs
199
200 Multiply the first ignored divisor limb by the most significant
201 quotient limb. If that product is > the partial remainder's
202 most significant limb, we know the quotient is too large. This
203 test quickly catches most cases where the quotient is too large;
204 it catches all cases where the quotient is 2 too large.*/
205
206 mp_limb_t x;
207 if (cnt) {
208 mp_limb_t dl;
209 if (ni < 2)
210 dl = 0;
211 else
212 dl = numb[ni - 2];
213 x = (numb[ni - 1] << cnt) | (dl >> (LIMB_BITS - cnt));
214 } else
215 x = numb[ni - 1];
216 mp_limb_t h = (x >> LIMB_BITS / 2) * (dstq[nq - 1] >> LIMB_BITS / 2);
217 mp_limb_t rnb = 0; // remainder[nb]
218 mp_size_t nr = nq; // remainder=rnb:[numa2,nr]:[...,ni]
219
220 if (h > numa2[nq - 1]) {
221 lmmp_dec(dstq);
222 rnb = lmmp_add_n_(numa2, numa2, numb2, nq);
223 }
224
225 // if cnt, recover the shift of partial remainder
226 // and remove the effect of the partial-ignored numa[ni-1] and numb[ni-1]
227 if (cnt) {
228 numa2[nq] = rnb;
229 ++nr;
230 --ni;
231 lmmp_shl_(numa2, numa2, nr, LIMB_BITS - cnt);
232 numa2[0] |= numa[ni] & (LIMB_MAX >> cnt);
233 cy = lmmp_submul_1_(numa2, dstq, nq, numb[ni] & (LIMB_MAX >> cnt));
234 rnb = -(numa2[nq] < cy);
235 numa2[nq] -= cy;
236 }
237
238 if (ni == 0) {
239 if (dstr) {
240 if (rnb)
241 lmmp_add_n_(dstr, numa2, numb, nr);
242 else
243 lmmp_copy(dstr, numa2, nr);
244 }
245 } else {
246 tp[nb - 1] = 0;
247 if (ni < nq)
248 lmmp_mul_(tp, dstq, nq, numb, ni);
249 else
250 lmmp_mul_(tp, numb, ni, dstq, nq);
251
252 if (dstr) {
253 mp_ptr remptr = dstr == numb ? tp : dstr;
254 cy = lmmp_sub_n_(remptr, numa, tp, ni);
255 rnb -= lmmp_sub_nc_(remptr + ni, numa2, tp + ni, nr, cy);
256 if (rnb)
257 lmmp_add_n_(dstr, remptr, numb, nb);
258 else if (dstr != remptr)
259 lmmp_copy(dstr, remptr, nb);
260 } else {
261 int hcmp = lmmp_cmp_(numa2, tp + ni, nr);
262 if (hcmp < 0)
263 --rnb;
264 else if (hcmp == 0)
265 rnb -= (lmmp_cmp_(numa, tp, ni) < 0);
266 }
267 }
268
269 if (rnb)
270 lmmp_dec(dstq);
271 }
272
273 TEMP_FREE;
274 }
