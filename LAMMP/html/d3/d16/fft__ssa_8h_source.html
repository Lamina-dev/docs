78/**
79 * @brief FFT蝶形运算（Butterfly Operation）
80 * (a,b) = (a + b, (a-b) << w ) mod 2^n+1
81 * a=[coef[0],ms->lenw+1], b=[coef[wing],ms->lenw+1], n=ms->lenw * LIMB_BITS
82 * @param ms - 内存栈结构体指针
83 * @param coef - 系数数组指针数组（coef[0]=a, coef[wing]=b）
84 * @param wing - b的索引
85 * @param w - 左移的比特数（0<=w<n）
86 * @warning n = ms->lenw * LIMB_BITS
87 * a,b 均已伪归一化（mod 2^n+1）
88 * ms->temp_coef 有至少 lenw + 1 个字长
89 */
90void lmmp_fft_bfy_(fft_memstack* ms, mp_ptr* coef, mp_size_t wing, mp_size_t w);

96 * @param ms - 内存栈结构体指针
97 * @param coef - 系数数组指针数组（coef[0]=a, coef[wing]=b）
98 * @param wing - b的索引
99 * @param w - 左移的比特数（0<=w<n）
100 * @warning n = ms->lenw * LIMB_BITS

103 */
