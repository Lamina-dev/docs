28/* sqr_toom8 递归可达的最大子问题规模（FFT 接管前） */
29#define SQR_TOOM8_MAX ((SQR_FFT_THRESHOLD + 8 * 2 - 1 + 7) / 8)
30#define MAYBE_sqr_basecase (SQR_TOOM8_THRESHOLD < 8 * SQR_TOOM22_THRESHOLD)
31#define MAYBE_sqr_above_basecase (SQR_TOOM8_MAX >= SQR_TOOM22_THRESHOLD)
32#define MAYBE_sqr_toom2 (SQR_TOOM8_THRESHOLD < 8 * SQR_TOOM33_THRESHOLD)
33#define MAYBE_sqr_above_toom2 (SQR_TOOM8_MAX >= SQR_TOOM33_THRESHOLD)
34#define MAYBE_sqr_toom3 (SQR_TOOM8_THRESHOLD < 8 * SQR_TOOM44_THRESHOLD)
35#define MAYBE_sqr_above_toom3 (SQR_TOOM8_MAX >= SQR_TOOM44_THRESHOLD)
36#define MAYBE_sqr_toom4 (SQR_TOOM8_THRESHOLD < 8 * SQR_TOOM6_THRESHOLD)
37#define MAYBE_sqr_above_toom4 (SQR_TOOM8_MAX >= SQR_TOOM6_THRESHOLD)
38#define MAYBE_sqr_above_toom6 (SQR_TOOM8_MAX >= SQR_TOOM8_THRESHOLD)

40#define lmmp_sqr_(dst, numa, n) \
41 if (MAYBE_sqr_basecase && (!MAYBE_sqr_above_basecase || (n) < SQR_TOOM22_THRESHOLD)) \
42 lmmp_sqr_hard_n_((dst), (numa), (n)); \
43 else if (MAYBE_sqr_toom2 && (!MAYBE_sqr_above_toom2 || (n) < SQR_TOOM33_THRESHOLD)) \
44 lmmp_sqr_toom2_((dst), (numa), (n)); \
45 else if (MAYBE_sqr_toom3 && (!MAYBE_sqr_above_toom3 || (n) < SQR_TOOM44_THRESHOLD)) \
46 lmmp_sqr_toom3_((dst), (numa), (n)); \
47 else if (MAYBE_sqr_toom4 && (!MAYBE_sqr_above_toom4 || (n) < SQR_TOOM6_THRESHOLD)) \
48 lmmp_sqr_toom4_((dst), (numa), (n)); \
49 else if (!MAYBE_sqr_above_toom6 || (n) < SQR_TOOM8_THRESHOLD) \
50 lmmp_sqr_toom6_((dst), (numa), (n)); \
51 else \
52 lmmp_sqr_toom8_((dst), (numa), (n))

54/*
55Evaluate in: +8,-8, +4,-4, +2,-2, +1,-1, +1/2,-1/2, +1/4,-1/4, +1/8,-1/8, 0
56（平方无 inf 点：(-v)^2 = v^2，符号恒正）
57*/
58
59void lmmp_sqr_toom8_(mp_ptr restrict dst, mp_srcptr restrict numa, mp_size_t na) {
60 lmmp_param_assert(na > 0);
61 lmmp_param_assert(dst != NULL);
62 lmmp_param_assert(numa != NULL);
63 lmmp_param_assert(na >= 51); /* 契约排除 s=1 退化点 na=50/57（16 点插值需 2s>3） */
64 lmmp_param_assert(na != 57);
65 mp_size_t n, s;
66
67 n = 1 + ((na - 1) >> 3);
68 s = na - 7 * n;
69
70 lmmp_debug_assert(0 < s && s <= n);
71 /* 契约排除 na=50、57（na>=50 中仅此两点 s=1，不满足 2s>3） */
72 lmmp_debug_assert(s + s > 3);
73
74#define r6 (dst + 3 * n) /* 3n+1 */
75#define r4 (dst + 7 * n) /* 3n+1 */
76#define r2 (dst + 11 * n) /* 3n+1 */
77#define r0 (dst + 15 * n) /* 2s */
78#define r7 (scratch) /* 3n+1 */
79#define r5 (scratch + 3 * n + 1) /* 3n+1 */
80#define r3 (scratch + 6 * n + 2) /* 3n+1 */
81#define r1 (scratch + 9 * n + 3) /* 3n+1 */
82#define v0 (dst + 11 * n) /* n+1 */
83#define v2 (dst + 13 * n + 2) /* n+1 */
84#define wse (scratch + 12 * n + 4) /* 3n+1 */
85
86 /* scratch 总需求 15n+6 limbs；TALLOC 双轨（>32KiB 落池/堆） */
87 TEMP_DECL;
88 mp_ptr restrict scratch = TALLOC_TYPE(15 * n + 6, mp_limb_t);
89
90 /* ±1/8 */
91 lmmp_toom_eval_pm2rexp_(v2, v0, 7, numa, n, s, 3, dst);
92 lmmp_sqr_(dst, v0, n + 1);
93 lmmp_sqr_(r7, v2, n + 1);
94 lmmp_toom_couple_handling_(r7, 2 * n + 1, dst, 0, n, 3, 0);
95
96 /* ±1/4 */
97 lmmp_toom_eval_pm2rexp_(v2, v0, 7, numa, n, s, 2, dst);
98 lmmp_sqr_(dst, v0, n + 1);
99 lmmp_sqr_(r5, v2, n + 1);
100 lmmp_toom_couple_handling_(r5, 2 * n + 1, dst, 0, n, 2, 0);
101
102 /* ±2 */
103 lmmp_toom_eval_pm2_(v2, v0, 7, numa, n, s, dst);
104 lmmp_sqr_(dst, v0, n + 1);
105 lmmp_sqr_(r3, v2, n + 1);
106 lmmp_toom_couple_handling_(r3, 2 * n + 1, dst, 0, n, 1, 2);
107
108 /* ±8 */
109 lmmp_toom_eval_pm2exp_(v2, v0, 7, numa, n, s, 3, dst);
110 lmmp_sqr_(dst, v0, n + 1);
111 lmmp_sqr_(r1, v2, n + 1);
112 lmmp_toom_couple_handling_(r1, 2 * n + 1, dst, 0, n, 3, 6);
113
114 /* ±1/2 */
115 lmmp_toom_eval_pm2rexp_(v2, v0, 7, numa, n, s, 1, dst);
116 lmmp_sqr_(dst, v0, n + 1);
117 lmmp_sqr_(r6, v2, n + 1);
118 lmmp_toom_couple_handling_(r6, 2 * n + 1, dst, 0, n, 1, 0);
119
120 /* ±1 */
121 lmmp_toom_eval_pm1_(v2, v0, 7, numa, n, s, dst);
122 lmmp_sqr_(dst, v0, n + 1);
123 lmmp_sqr_(r4, v2, n + 1);
124 lmmp_toom_couple_handling_(r4, 2 * n + 1, dst, 0, n, 0, 0);
125
126 /* ±4 */
127 lmmp_toom_eval_pm2exp_(v2, v0, 7, numa, n, s, 2, dst);
128 lmmp_sqr_(dst, v0, n + 1);
129 lmmp_sqr_(r2, v2, n + 1);
130 lmmp_toom_couple_handling_(r2, 2 * n + 1, dst, 0, n, 2, 4);
131
132#undef v0
133#undef v2
134
135 /* A(0)^2 */
136 lmmp_sqr_(dst, numa, n);
137
138 lmmp_toom_interp16_(dst, r1, r3, r5, r7, n, 2 * s, 0, wse);
139
140#undef r0
141#undef r1
142#undef r2
143#undef r3
144#undef r4
145#undef r5
146#undef r6
147#undef r7
148
149 TEMP_FREE;
150}
