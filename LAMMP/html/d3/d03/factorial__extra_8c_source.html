40mp_size_t lmmp_2factorial_size_(uint n, mp_bitcnt_t* restrict bits) {
41 mp_size_t rn;
42 uint k;
43 if (n < 30) {
44 rn = 64;
45 k = n / 2;
46 if (n % 2 == 0) {
47 *bits = n - lmmp_limb_popcnt_(k);
48 } else {
49 *bits = 0;
50 }
51 } else {
52 k = n / 2;
53 if (n % 2 == 0) {
54 // n=2k
55 // n! = (2k)! = 2^k * (k!)
56 rn = k + log2_fac_ceil(k);
57 *bits = n - lmmp_limb_popcnt_(k);
58 } else {
59 // n=2k+1
60 // n! = (2k+1)! = (2k+1)! / (2k)!! = (2k+1)! / (2^k * (k!))
61 rn = log2_fac_ceil(n);
62 rn -= k + log2_fac_floor(k);
63 *bits = 0;
64 }
65 }
66 rn = (rn + LIMB_BITS - 1) / LIMB_BITS + 2; // more two limbs
67 return rn;
68}

78mp_size_t lmmp_factors_mul_ushort_(mp_ptr restrict dst, mp_size_t rn, fac_ptr restrict fac, ushort nfactors) {
79 lmmp_param_assert(dst != NULL && fac != NULL);
80 lmmp_param_assert(rn > 0 && nfactors > 0);
81 if (nfactors <= FACTORS_MUL_N_THRESHOLD && rn < 20) {
82 // 此处额外校验rn的大小，因为hyper阶乘和super阶乘在阶数较小时，仅有很少的因子，但其指数很大
83 // 为了避免算法退化，此处校验rn，仅对rn较小时进行朴素乘法。
84 dst[0] = 1;
85 rn = 1;
86 mp_limb_t t = 1;
87 for (ushort i = 0; i < nfactors; i++) {
88 ushort f = fac[i].f;
89 uint j = fac[i].j;
90 lmmp_debug_assert(j != 0);
91#define MAX_T 0xffffffffffff
92 for (uint e = 0; e < j; e++) {
93 t *= f;
94 if (t > MAX_T) {
95 mul_1(dst, rn, t);
96 t = 1;
97 }
98 }
99 }
100 if (t != 1) {
101 mul_1(dst, rn, t);
102 }
103#undef MAX_T
104 return rn;
105 } else {
106 TEMP_DECL;
107 ushort new_nfactors = 0;
108 ulongp restrict limbs = TALLOC_TYPE(nfactors / 4 + 1, ulong);
109 ulong t = 1;
110 mp_size_t limbn = 0;
111 for (ushort i = 0; i < nfactors; ++i) {
112 ushort f = fac[i].f;
113 uint j = fac[i].j;
114 if (j > 1) {
115 fac[new_nfactors].f = f;
116 fac[new_nfactors++].j = j >> 1;
117 }
118 if (j & 1) {
119 t *= f;
120#define MAX_T 0xffffffffffff
121 if (t > MAX_T) {
122 limbs[limbn++] = t;
123 t = 1;
124 }
125#undef MAX_T
126 }
127 }
128 if (t != 1) {
129 limbs[limbn++] = t;
130 }
131
132 mp_ptr restrict mp = TALLOC_TYPE(limbn * 2, mp_limb_t);
133 mp_size_t mpn = 0;
134
135 if (new_nfactors > 0) {
136 if (limbn > 0) {
137 mpn = lmmp_elem_mul_ulong_(mp, limbs, limbn, mp + limbn);
138 lmmp_debug_assert(rn >= mpn);
139 mp_size_t tn = ((rn - mpn) >> 1) + 1;
140 // 这里根据mpn的大小估计剩余因子乘积的长度，额外分配两倍的tn，以进行平方。
141 mp_ptr restrict tp = BALLOC_TYPE(3 * tn + 3, mp_limb_t);
142 tn = lmmp_factors_mul_ushort_(tp, tn, fac, new_nfactors);
143
144 mp_ptr restrict tp2 = tp + tn + 1;
145 lmmp_sqr_(tp2, tp, tn);
146 tn <<= 1;
147 tn -= tp2[tn - 1] == 0;
148 MUL(dst, tp2, tn, mp, mpn);
149 rn = tn + mpn;
150 rn -= dst[rn - 1] == 0;
151 } else {
152 mp_size_t tn = (rn >> 1) + 1;
153 mp_ptr restrict tp = TALLOC_TYPE(tn, mp_limb_t);
154 tn = lmmp_factors_mul_ushort_(tp, tn, fac, new_nfactors);
155 lmmp_sqr_(dst, tp, tn);
156 rn = tn << 1;
157 rn -= dst[rn - 1] == 0;
158 }
159 } else {
160 lmmp_debug_assert(limbn > 0);
161 // 这里不能直接乘入dst，因为dst的大小可能小于limbn，导致溢出
162 if (rn >= limbn) {
163 rn = lmmp_elem_mul_ulong_(dst, limbs, limbn, mp);
164 } else {
165 rn = lmmp_elem_mul_ulong_(mp, limbs, limbn, mp + limbn);
166 lmmp_copy(dst, mp, rn);
167 }
168 }
169 TEMP_FREE;
170 return rn;
171 }
172}

174static inline uint count_2facodd_factors(fac_ptr fac, uint nfactors, uint n, uint k, uint p) {
175 uint e = 0;
176 uint pn = n;
177 ulong inv = MP_ULONG_MAX / p + 1;
178 while (pn > 0) {
179 _udiv32by32_q_preinv(pn, pn, inv);
180 e += pn;
181 }
182 pn = k;
183 while (pn > 0) {
184 _udiv32by32_q_preinv(pn, pn, inv);
185 e -= pn;
186 }
187 if (e > 0) {
188 fac[nfactors].f = p;
189 fac[nfactors++].j = e;
190 }
191 return nfactors;
192}

227static mp_size_t lmmp_odd_2factorial_uint_(mp_ptr restrict dst, mp_size_t rn, uint n, uint k) {
228 lmmp_prime_int_table_init_(n);
229 TEMP_B_DECL;
230 uint nfactors = lmmp_prime_size_(n);
231 fac_ptr restrict fac = BALLOC_TYPE(nfactors, fac_t);
232 nfactors = 0;
233
234 prime_cache_t cache;
235 lmmp_prime_cache_init_(&cache, n);
236 while (cache.is_end == 0) {
237 lmmp_prime_cache_next_(&cache);
238 for (uint i = 0; i < cache.size; i++) {
239 // 对于阶乘n!，对于所有小于等于n的质数，贡献都至少为1
240 nfactors = count_2facodd_factors(fac, nfactors, n, k, cache.pp[i]);
241 }
242 }
243 lmmp_prime_cache_free_(&cache);
244
245 rn = lmmp_factors_mul_(dst, rn, fac, nfactors);
246
247 TEMP_B_FREE;
248 return rn;

251mp_size_t lmmp_2factorial_(mp_ptr restrict dst, mp_bitcnt_t bits, mp_size_t rn, uint n) {
252 static const mp_limb_t odd_2factorial_table[17] = {1, 3, 15, 105, 945, 10395, 135135,
253 2027025, 34459425, 654729075,
254 13749310575, 316234143225,
255 7905853580625, 213458046676875,
256 6190283353629375, 191898783962510625,
257 6332659870762850625};
258
259 lmmp_param_assert(dst != NULL);
260 mp_size_t shw = bits / LIMB_BITS;
261 lmmp_param_assert(rn > shw);
262 bits %= LIMB_BITS;
263 lmmp_zero(dst, shw);
264
265 uint k = n / 2;
266 if (n % 2 == 0) {
267 if (k <= MP_USHORT_MAX)
268 rn = lmmp_odd_nPr_ushort_(dst + shw, rn - shw, k, k);
269 else
270 rn = lmmp_odd_factorial_uint_(dst + shw, rn - shw, k);
271 } else {
272 if (k < 17) {
273 dst[shw] = odd_2factorial_table[k];
274 rn = 1;
275 } else if (n <= MP_USHORT_MAX)
276 rn = lmmp_odd_2factorial_ushort_(dst + shw, rn - shw, n, k);
277 else
278 rn = lmmp_odd_2factorial_uint_(dst + shw, rn - shw, n, k);
279 }
280
281 if (bits > 0) {
282 dst[shw + rn] = lmmp_shl_(dst + shw, dst + shw, rn, bits);
283 rn += shw + 1;
284 rn -= dst[rn - 1] == 0;
285 } else {
286 rn += shw;
287 }
288 return rn;
289}

291static inline uint count_hyperfac_factors(ushort n, ushort p) {
292 uint v = 0;
293 uint npi = n / p;
294 uint pi = p;
295 while (npi > 0) {
296 v += pi * npi * (npi + 1) / 2;
297 npi /= p;
298 pi *= p;
299 }
300 return v;
301}

303static inline uint count_superfac_factors(ushort n, ushort p) {
304 uint v = 0;
305 uint pi = p;
306 uint N = (uint)n + 1;
307 uint npi = n / pi;
308 while (npi > 0) {
309 v += npi * N - pi * npi * (npi + 1) / 2;
310 pi *= p;
311 npi /= p;
312 }
313 return v;

316mp_size_t lmmp_hyperfac_size_(ushort n, mp_bitcnt_t* restrict bits) {
317 if (n == 0 || n == 1) {
318 *bits = 0;
319 return 1;
320 }
321 // A = 1.2824271291006226369 Glaisher–Kinkelin constant
322 const double logA = 0.24875447703378426;
323 const double log2 = 0.69314718055994531; // log(2)
324 double n_sqr = (uint)n * n;
325 double log_n = log(n);
326 double r = 0.5 * n_sqr * log_n - 0.25 * n_sqr + 0.5 * log_n * (double)n + 1.0 / 12.0 * log_n + logA;
327 mp_size_t rn = (mp_size_t)ceil(r / log2);
328 rn = (rn + LIMB_BITS - 1) / LIMB_BITS + 2; // more two limbs
329 *bits = count_hyperfac_factors(n, 2);
330 return rn;
331}

333mp_size_t lmmp_superfac_size_(ushort n, mp_bitcnt_t* restrict bits) {
334 if (n == 0 || n == 1) {
335 *bits = 0;
336 return 1;
337 }
338 // zeta函数在-1处的导数
339 // zeta(-1) = 1/12 - log(A) // A 即 Glaisher–Kinkelin constant
340 const double zeta_diff_neg1 = -0.16542114370045093;
341 const double log2 = 0.69314718055994531; // log(2)
342 const double log_2pi = 1.83787706640934548; // log(2*pi)
343 double z = (double)n + 1;
344 double z_sqr = z * z;
345 double log_z = log(z);
346 double r = 0.5 * z_sqr * log_z - 0.75 * z_sqr + 0.5 * z * log_2pi - 1.0 / 12.0 * log_z + zeta_diff_neg1;
347 mp_size_t rn = (mp_size_t)ceil(r / log2);
348 rn = (rn + LIMB_BITS - 1) / LIMB_BITS + 2; // more two limbs
349 *bits = count_superfac_factors(n, 2);
350 return rn;
351}

446mp_size_t lmmp_primefac_(mp_ptr restrict dst, mp_size_t rn, uint n) {
447 if (n < 2) {
448 dst[0] = 1;
449 return 1;
450 } else if (n <= MP_USHORT_MAX) {
451 TEMP_S_DECL;
452 ushort primen = lmmp_prime_cnt16_(n);
453 ulongp restrict pp = SALLOC_TYPE(primen / 4 + 1, ulong);
454 ulong t = 1;
455 mp_size_t pn = 0;
456 for (ushort i = 0; i < primen; i++) {
457 t *= prime_short_table[i];
458#define MAX_T 0xffffffffffff
459 if (t > MAX_T) {
460 pp[pn++] = t;
461 t = 1;
462 }
463#undef MAX_T
464 }
465 if (t > 1) {
466 pp[pn++] = t;
467 }
468
469 if (rn >= pn) {
470 mp_ptr restrict tp = SALLOC_TYPE(pn, mp_limb_t);
471 pn = lmmp_elem_mul_ulong_(dst, pp, pn, tp);
472 } else {
473 mp_ptr restrict prod = SALLOC_TYPE(pn * 2, mp_limb_t);
474 pn = lmmp_elem_mul_ulong_(prod, pp, pn, prod + pn);
475 lmmp_debug_assert(rn >= pn);
476 lmmp_copy(dst, prod, pn);
477 }
478 TEMP_S_FREE;
479 return pn;
480 } else {
481 TEMP_B_DECL;
482 lmmp_prime_int_table_init_(n);
483 uint pn = lmmp_prime_size_(n);
484 ulongp restrict pp = BALLOC_TYPE(pn / 2 + 1, ulong);
485 pn = 0;
486 ulong t = 2; // 2 is the smallest prime number
487
488 prime_cache_t cache;
489 lmmp_prime_cache_init_(&cache, n);
490 while (cache.is_end == 0) {
491 lmmp_prime_cache_next_(&cache);
492 for (uint i = 0; i < cache.size; i++) {
493 t *= cache.pp[i];
494 if (t > MP_UINT_MAX) {
495 pp[pn++] = t;
496 t = 1;
497 }
498 }
499 }
500 lmmp_prime_cache_free_(&cache);
501
502 if (t > 1) {
503 pp[pn++] = t;
504 }
505
506 if (rn >= pn) {
507 mp_ptr restrict tp = BALLOC_TYPE(pn, mp_limb_t);
508 pn = lmmp_elem_mul_ulong_(dst, pp, pn, tp);
509 } else {
510 mp_ptr restrict prod = BALLOC_TYPE(pn * 2, mp_limb_t);
511 pn = lmmp_elem_mul_ulong_(prod, pp, pn, prod + pn);
512 lmmp_debug_assert(rn >= pn);
513 lmmp_copy(dst, prod, pn);
514 }
515
516 TEMP_B_FREE;
517 return pn;
518 }
