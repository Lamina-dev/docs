37 lmmp_zero(dst, exp / LIMB_BITS);

39 rn = exp / LIMB_BITS + 1;

59static mp_size_t lmmp_3pow_1_(mp_ptr restrict dst, mp_size_t rn, ulong exp) {
60 TEMP_DECL;
61 static const mp_limb_t tab[32] = {
62 1, 3, 9, 27, 81, 243, 729, 2187,
63 6561, 19683, 59049, 177147, 531441, 1594323, 4782969, 14348907,
64 43046721, 129140163, 387420489, 1162261467, 3486784401, 10460353203, 31381059609, 94143178827,
65 282429536481,847288609443,2541865828329,7625597484987,22876792454961,68630377364883,205891132094649,617673396283947,
66 };
67
68 mp_ptr restrict sq = TALLOC_TYPE(rn, mp_limb_t);
69 rn = 1;
70 sq[0] = 1;
71 int i = 12;
72 while ((exp & (0x1full << ((i--) * 5))) == 0);
73 for (++i; i >= 0; --i) {
74 mp_size_t p = (exp & (0x1full << (i * 5))) >> (i * 5);
75 dst[rn] = lmmp_mul_1_(dst, sq, rn, tab[p]);
76 ++rn;
77 rn -= (dst[rn - 1] == 0) ? 1 : 0;
78 if (i > 0) {
79 lmmp_sqr_(sq, dst, rn);
80 rn <<= 1;
81 rn -= (sq[rn - 1] == 0) ? 1 : 0;
82
83 lmmp_sqr_(dst, sq, rn);
84 rn <<= 1;
85 rn -= (dst[rn - 1] == 0) ? 1 : 0;
86
87 lmmp_sqr_(sq, dst, rn);
88 rn <<= 1;
89 rn -= (sq[rn - 1] == 0) ? 1 : 0;
90
91 lmmp_sqr_(dst, sq, rn);
92 rn <<= 1;
93 rn -= (dst[rn - 1] == 0) ? 1 : 0;
94
95 lmmp_sqr_(sq, dst, rn);
96 rn <<= 1;
97 rn -= (sq[rn - 1] == 0) ? 1 : 0;
98 }
99 }
100 TEMP_FREE;
101 return rn;
102}

164static mp_size_t lmmp_6pow_1_(mp_ptr restrict dst, mp_size_t rn, ulong exp) {
165 mp_size_t n = exp / LIMB_BITS;
166 lmmp_zero(dst, n);
167 rn = lmmp_3pow_1_(dst + n, rn - n, exp);
168 dst[n + rn] = lmmp_shl_(dst + n, dst + n, rn, exp % LIMB_BITS);
169 rn += n + 1;
170 rn -= dst[rn - 1] == 0 ? 1 : 0;
171 return rn;
172}

253mp_size_t lmmp_u8_pow_1_(mp_ptr restrict dst, mp_size_t rn, ulong base, ulong exp) {
254 lmmp_param_assert(base >= 2);
255 lmmp_param_assert(base <= MP_UCHAR_MAX);
256 TEMP_DECL;
257 mp_limb_t tab[8];
258 tab[0] = 1;
259 tab[1] = tab[0] * base;
260 tab[2] = tab[1] * base;
261 tab[3] = tab[2] * base;
262 tab[4] = tab[3] * base;
263 tab[5] = tab[4] * base;
264 tab[6] = tab[5] * base;
265 tab[7] = tab[6] * base;
266
267 mp_ptr restrict sq = TALLOC_TYPE(rn, mp_limb_t);
268 sq[0] = 1;
269 rn = 1;
270 int i = 21;
271 while ((exp & (0x7ull << ((i--) * 3))) == 0);
272 for (++i; i >= 0; --i) {
273 mp_size_t p = (exp & (0x7ull << (i * 3))) >> (i * 3);
274 dst[rn] = lmmp_mul_1_(dst, sq, rn, tab[p]);
275 ++rn;
276 rn -= (dst[rn - 1] == 0) ? 1 : 0;
277 if (i > 0) {
278 lmmp_sqr_(sq, dst, rn);
279 rn <<= 1;
280 rn -= (sq[rn - 1] == 0) ? 1 : 0;
281 lmmp_sqr_(dst, sq, rn);
282 rn <<= 1;
283 rn -= (dst[rn - 1] == 0) ? 1 : 0;
284 lmmp_sqr_(sq, dst, rn);
285 rn <<= 1;
286 rn -= (sq[rn - 1] == 0) ? 1 : 0;
287 }
288 }
289 TEMP_FREE;
290 return rn;
291}

293mp_size_t lmmp_u16_pow_1_(mp_ptr restrict dst, mp_size_t rn, ulong base, ulong exp) {
294 lmmp_param_assert(base >= 2);
295 lmmp_param_assert(base <= MP_USHORT_MAX);
296 TEMP_DECL;
297 mp_limb_t tab[8][2];
298 tab[0][0] = 1;
299 tab[0][1] = 0;
300 tab[1][0] = base;
301 tab[1][1] = 0;
302 tab[2][0] = tab[1][0] * base;
303 tab[2][1] = 0;
304 tab[3][0] = tab[2][0] * base;
305 tab[3][1] = 0;
306 lmmp_mullh_(tab[2][0], tab[2][0], tab[4]);
307 lmmp_mullh_(tab[3][0], tab[2][0], tab[5]);
308 lmmp_mullh_(tab[3][0], tab[3][0], tab[6]);
309 lmmp_mullh_(tab[3][0], tab[4][0], tab[7]);
310 tab[7][1] += tab[3][0] * tab[4][1];
311
312 mp_ptr restrict sq = TALLOC_TYPE(rn, mp_limb_t);
313 rn = 1;
314 sq[0] = 1;
315 int i = 21;
316 while ((exp & (0x7ull << ((i--) * 3))) == 0);
317 for (++i; i >= 0; --i) {
318 mp_size_t p = (exp & (0x7ull << (i * 3))) >> (i * 3);
319 mp_srcptr tap = tab[p];
320 mp_size_t tan = (tap[1] != 0) ? 2 : 1;
321 if (rn >= tan)
322 lmmp_mul_basecase_(dst, sq, rn, tap, tan);
323 else
324 lmmp_mul_basecase_(dst, tap, tan, sq, rn);
325 rn += tan;
326 rn -= (dst[rn - 1] == 0) ? 1 : 0;
327
328 if (i > 0) {
329 lmmp_sqr_(sq, dst, rn);
330 rn <<= 1;
331 rn -= (sq[rn - 1] == 0) ? 1 : 0;
332 lmmp_sqr_(dst, sq, rn);
333 rn <<= 1;
334 rn -= (dst[rn - 1] == 0) ? 1 : 0;
335 lmmp_sqr_(sq, dst, rn);
336 rn <<= 1;
337 rn -= (sq[rn - 1] == 0) ? 1 : 0;
338 }
339 }
340 TEMP_FREE;
341 return rn;
342}

352mp_size_t lmmp_u32_pow_1_(mp_ptr restrict dst, mp_size_t rn, ulong base, ulong exp) {
353 lmmp_param_assert(base >= 2);
354 lmmp_param_assert(base <= MP_UINT_MAX);
355 TEMP_DECL;
356 mp_limb_t b1[1] = {base};
357#define b1n 1
358 mp_limb_t b2[1] = {base * base};
359#define b2n 1
360
361 mp_limb_t b3[2];
362 lmmp_mullh_(b2[0], base, b3);
363 mp_size_t b3n = b3[1] != 0 ? 2 : 1;
364
365 mp_limb_t b4[2];
366 lmmp_mullh_(b2[0], b2[0], b4);
367 mp_size_t b4n = b4[1] != 0 ? 2 : 1;
368
369 mp_limb_t b5[3];
370 b5[b4n] = lmmp_mul_1_(b5, b4, b4n, base);
371 mp_size_t b5n = b4n + 1;
372 b5n -= (b5[b5n - 1] == 0) ? 1 : 0;
373
374 mp_limb_t b6[4];
375 lmmp_sqr_hard_n_(b6, b3, b3n);
376 mp_size_t b6n = 2 * b3n;
377 b6n -= (b6[b6n - 1] == 0) ? 1 : 0;
378
379 mp_limb_t b7[4];
380 b7[b5n] = lmmp_mul_1_(b7, b5, b5n, b2[0]);
381 mp_size_t b7n = b5n + 1;
382 b7n -= (b7[b7n - 1] == 0) ? 1 : 0;
383
384 mp_ptr restrict sq = TALLOC_TYPE(rn, mp_limb_t);
385 sq[0] = 1;
386 rn = 1;
387 int i = 21;
388 while ((exp & (0x7ull << ((i--) * 3))) == 0);
389 for (++i; i >= 0; --i) {
390 mp_size_t p = (exp & (0x7ull << (i * 3))) >> (i * 3);
391 switch (p) {
392 case 0:
393 lmmp_copy(dst, sq, rn);
394 break;
395 case 1:
396 mul_b(1);
397 break;
398 case 2:
399 mul_b(2);
400 break;
401 case 3:
402 mul_b(3);
403 break;
404 case 4:
405 mul_b(4);
406 break;
407 case 5:
408 mul_b(5);
409 break;
410 case 6:
411 mul_b(6);
412 break;
413 case 7:
414 mul_b(7);
415 break;
416 }
417 if (i > 0) {
418 lmmp_sqr_(sq, dst, rn);
419 rn <<= 1;
420 rn -= (sq[rn - 1] == 0) ? 1 : 0;
421 lmmp_sqr_(dst, sq, rn);
422 rn <<= 1;
423 rn -= (dst[rn - 1] == 0) ? 1 : 0;
424 lmmp_sqr_(sq, dst, rn);
425 rn <<= 1;
426 rn -= (sq[rn - 1] == 0) ? 1 : 0;
427 }
428 }
429 TEMP_FREE;
430 return rn;
431#undef b1n
432#undef b2n
433}

520/*
521 * base ^ exp
522 *
523 * - base <= 15 : 4-bit lmmp_u4_pow_1_
524 * - base <= 255 : 8-bit lmmp_u8_pow_1_
525 * - base <= 65535 : 16-bit lmmp_u16_pow_1_
526 * - base <= 2^32-1 : 32-bit lmmp_u32_pow_1_
527 * - base <= 2^64-1 : 64-bit lmmp_u64_pow_1_
528 */
529
530mp_size_t lmmp_pow_1_(mp_ptr restrict dst, mp_size_t rn, mp_limb_t base, ulong exp) {
531 lmmp_param_assert(base >= 1);
532 lmmp_param_assert(exp > 0);
533 if (base <= (mp_limb_t)0xf) {
534 return lmmp_u4_pow_1_(dst, rn, base, exp);
535 } else {
536 int tz = lmmp_tailing_zeros_(base);
537 base >>= tz;
538 mp_size_t shw = (exp * tz) / LIMB_BITS;
539 mp_size_t shl = (exp * tz) % LIMB_BITS;
540
541 lmmp_param_assert(rn > shw);
542 lmmp_zero(dst, shw);
543 if (base <= (mp_limb_t)0xf)
544 rn = lmmp_u4_pow_1_(dst + shw, rn - shw, base, exp);
545 else if (base <= (mp_limb_t)MP_UCHAR_MAX)
546 rn = lmmp_u8_pow_1_(dst + shw, rn - shw, base, exp);
547 else if (base <= (mp_limb_t)MP_USHORT_MAX)
548 rn = lmmp_u16_pow_1_(dst + shw, rn - shw, base, exp);
549 else if (base <= (mp_limb_t)MP_UINT_MAX)
