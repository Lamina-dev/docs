41static inline void* lmmp_temp_heap_alloc_(lmmp_alloc_marker* pmarker, size_t size) {
42 /*
43 * pmarker->heap_marker is a head pointer to a linked list of allocated memory blocks.
44 * Each allocated block has a header of size HSIZE, which is used to store the
45 * next pointer of the block. The actual data starts at (mp_byte_t*)p + offset.
46 */
47 // 在经过缓冲池后，size已经对齐至LAMMP_MAX_ALIGN
48 // size = LMMP_ROUND_UP_MULTIPLE(size, LAMMP_MAX_ALIGN);
49#define HSIZE sizeof(void*)
50 const size_t offset = LMMP_ROUND_UP_MULTIPLE(HSIZE, LAMMP_MAX_ALIGN);
51#undef HSIZE
52 void* p = global_heap.alloc(size + offset);
53#if LAMMP_DEBUG_MEMORY_CHECK == 1
54 if (p == NULL) {
55 char msg[128];
56 snprintf(msg, sizeof(msg), "Memory allocation failure (trying to allocate: %zu bytes)", size);
57 lmmp_abort(LAMMP_ERROR_MEMORY_ALLOC_FAILURE, msg, __func__, __LINE__);
58 }
59#endif
60 *(void**)p = pmarker->heap_marker;
61 pmarker->heap_marker = p;
62 return (mp_byte_t*)p + offset;
63}

65static inline void lmmp_temp_heap_free_(lmmp_alloc_marker* pmarker) {
66 /*
67 * Free all allocated memory blocks in the linked list pointed to by pmarker->heap_marker.
68 */
69 while (pmarker->heap_marker != NULL) {
70 void* next = *(void**)(pmarker->heap_marker);
71 global_heap.free(pmarker->heap_marker);
72 pmarker->heap_marker = next;
73 }
74}

76static inline void* lmmp_temp_stack_alloc_(lmmp_alloc_marker* pmarker, size_t size) {
77 /*
78 * On the first call, pmarker->stack_marker is a null pointer.
79 * We will use pmarker->stack_marker to record the stack frame at this time.
80 * When allocating memory subsequently, we will not modify pmarker->stack_marker.
81 * Until all stack memory is finally released, we will move to the initial stack position at once,
82 * which is the position recorded by pmarker->stack_marker.
83 */
84 mp_byte_t* p = (mp_byte_t*)(lmmp_tmpmem_ctx.stack_top);
85 size_t offset = LMMP_ROUND_UP_MULTIPLE(size, LAMMP_MAX_ALIGN);
86 mp_byte_t* new_top = p + offset;
87#if LAMMP_DEBUG_STACK_OVERFLOW_CHECK == 1
88 if (new_top > (mp_byte_t*)(lmmp_tmpmem_ctx.stack_end)) {
89 char msg[128];
90 snprintf(msg, sizeof(msg), "Stack overflow (trying to allocate: %zu bytes, stack remaining: %zu bytes)", offset,
91 (size_t)((mp_byte_t*)lmmp_tmpmem_ctx.stack_end - (mp_byte_t*)lmmp_tmpmem_ctx.stack_top));
92 lmmp_abort(LAMMP_ERROR_MEMORY_ALLOC_FAILURE, msg, __func__, __LINE__);
93 }
94#endif
95 lmmp_tmpmem_ctx.stack_top = new_top;
96 if (pmarker->stack_marker == NULL)
97 pmarker->stack_marker = p;
98 return p;
99}

101static inline void lmmp_temp_stack_free_(lmmp_alloc_marker* pmarker) {
102 // pmarker->stack_marker is not NULL
103 lmmp_tmpmem_ctx.stack_top = pmarker->stack_marker;
104}

106static inline void* lmmp_temp_pool_alloc_(lmmp_alloc_marker* pmarker, size_t size) {
107 mp_byte_t* p = (mp_byte_t*)(lmmp_tmpmem_ctx.pool_top);
108 size_t offset = LMMP_ROUND_UP_MULTIPLE(size, LAMMP_MAX_ALIGN);
109 size_t remaining = lmmp_tmpmem_ctx.remain;
110 if (remaining < 2 * offset) {
111 // 保证pool留有一个offset的空间，使得递归后子问题也尽可能分配到pool中，避免递归时子问题分配到堆上
112 return lmmp_temp_heap_alloc_(pmarker, offset);
113 } else {
114 mp_byte_t* new_top = p + offset;
115 lmmp_tmpmem_ctx.remain -= offset;
116 lmmp_tmpmem_ctx.pool_top = new_top;
117 if (pmarker->pool_marker == NULL)
118 pmarker->pool_marker = p;
119 return p;
120 }
121}

123static inline void lmmp_temp_pool_free_(lmmp_alloc_marker* pmarker) {
124 // pmarker->pool_marker is not NULL
125 size_t freed = (mp_byte_t*)(lmmp_tmpmem_ctx.pool_top) - (mp_byte_t*)(pmarker->pool_marker);
126 lmmp_tmpmem_ctx.remain += freed;
127 lmmp_tmpmem_ctx.pool_top = pmarker->pool_marker;
128}

143#define TEMP_SALLOC_THRESHOLD 0x7f00 // 小内存分配阈值（小于等于该值的内存分配在栈上）

145// 栈内存分配：使用lmmp_temp_stack_alloc_在栈上分配n字节内存（小内存）
146#define TEMP_SALLOC(n) lmmp_temp_stack_alloc_(&__lmmp_marker_, (n))
147// 缓冲池分配：使用lmmp_temp_pool_alloc_优先在pool上分配n字节内存（大内存）
148#define TEMP_BALLOC(n) lmmp_temp_pool_alloc_(&__lmmp_marker_, (n))
149// 临时内存分配：小内存用栈，大内存用堆
150#define TEMP_TALLOC(n) ((n) <= TEMP_SALLOC_THRESHOLD ? TEMP_SALLOC(n) : TEMP_BALLOC(n))
151// 类型化栈内存分配：分配n个type类型的栈内存
152#define SALLOC_TYPE(n, type) ((type*)TEMP_SALLOC((n) * sizeof(type)))
153// 类型化堆内存分配：分配n个type类型的堆内存
154#define BALLOC_TYPE(n, type) ((type*)TEMP_BALLOC((n) * sizeof(type)))
155// 类型化临时内存分配：智能选择栈/堆分配n个type类型内存
156#define TALLOC_TYPE(n, type) ((type*)TEMP_TALLOC((n) * sizeof(type)))
157// 临时内存释放：释放所有通过TEMP_XALLOC系列函数分配的临时内存
158#define TEMP_FREE \
159 do { \
160 (void)__lmmp_temp_decl_guard_; \
161 lmmp_temp_heap_free_(&__lmmp_marker_); \
162 if (__lmmp_marker_.stack_marker != NULL) \
163 lmmp_temp_stack_free_(&__lmmp_marker_); \
164 if (__lmmp_marker_.pool_marker != NULL) \
165 lmmp_temp_pool_free_(&__lmmp_marker_); \
166 } while (0)
167#define TEMP_B_FREE \
168 do { \
169 (void)__lmmp_temp_b_decl_guard_; \
170 lmmp_temp_heap_free_(&__lmmp_marker_); \
171 if (__lmmp_marker_.pool_marker != NULL) \
172 lmmp_temp_pool_free_(&__lmmp_marker_); \
173 } while (0)
174#define TEMP_S_FREE \
175 do { \
176 (void)__lmmp_temp_s_decl_guard_; \
177 if (__lmmp_marker_.stack_marker != NULL) \
178 lmmp_temp_stack_free_(&__lmmp_marker_); \
179 } while (0)
