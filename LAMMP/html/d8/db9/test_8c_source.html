41void lmmp_test_run(const char *name, void (*test)(void)) {
42 unsigned before = failures;
43 ++tests;
44 printf("[ RUN ] %s\n", name);
45 test();
46 if (failures != before)
47 ++failed_tests;
48 printf("[ %s ] %s\n", failures == before ? "PASSED" : "FAILED", name);
49}
