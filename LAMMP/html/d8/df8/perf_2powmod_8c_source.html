27/*
28 奇模数模幂性能：Montgomery REDC + 滑动窗口。
29
30 输入取 RSA 典型形态：指数与模数同规模（ebits ~ n*64），
31 尺寸跨越 REDC_MERSENNE_THRESHOLD(401) 两侧，覆盖
32 全积取高半与梅森折叠两条归约路径。
33*/
34static void fill_odd_mod(mp_ptr m, mp_ptr b, mp_ptr e, size_t n, mp_limb_t seed) {
35 lmmp_seed_random_(m, (mp_size_t)n, seed, 1);
36 m[0] |= 1;
37 m[n - 1] |= (mp_limb_t)1 << 63;
38 lmmp_seed_random_(b, (mp_size_t)n, seed * 3 + 1, 1);
39 lmmp_div_(NULL, b, b, (mp_size_t)n, m, (mp_size_t)n);
40 lmmp_seed_random_(e, (mp_size_t)n, seed * 7 + 2, 1);
41 e[n - 1] |= (mp_limb_t)1 << 63;
42}

44static double bench_powmod_odd(mp_size_t n, const bench_options *o) {
45 mp_ptr m = alloc_limbs(n), b = alloc_limbs(n), e = alloc_limbs(n), dst = alloc_limbs(n);
46 fill_odd_mod(m, b, e, n, 0x51ce77aa1bull + n);
47 double us;
48 BENCH_MEASURE(us, o, lmmp_powmod_odd_(dst, b, e, n, m, n); bench_sink += dst[0];);
49 lmmp_free(m);
50 lmmp_free(b);
51 lmmp_free(e);
52 lmmp_free(dst);
53 return us;
54}

56/*
57 偶模数模幂：CRT 结构 = 奇部 REDC 梯子 + 2^k 部 powlo + 一次合成。
58 m = 2^(64*(nb2-1)+1)*odd：nb2=1 即 m=2*odd（浅移位），nb2=2 即
59 m 含 2^65 因子（深移位）。e ~ 64n 时偶底数结果恒 0（2^k 部短路），
60 故底数强制取奇以衡量典型负载。
61*/
62static double bench_powmod_even(mp_size_t n, uint nb2, const bench_options *o) {
63 if (n < 3 && nb2 > 1) nb2 = 1;
64 if (nb2 < 1 || nb2 >= n) nb2 = 1; /* 奇部至少占顶 1 limb */
65 mp_ptr m = alloc_limbs(n), b = alloc_limbs(n), e = alloc_limbs(n), dst = alloc_limbs(n);
66 lmmp_seed_random_(m, n, 0x51ce77ac0bull + nb2 * 131 + n, 1);
67 if (nb2 == 1) {
68 m[0] = (m[0] << 1) | 2; /* m = 2*odd */
69 } else {
70 lmmp_zero(m, nb2 - 1);
71 m[nb2 - 1] = 2; /* m = 2^(64*(nb2-1)+1)*odd */
72 }
73 m[n - 1] |= (mp_limb_t)1 << 63;
74 lmmp_seed_random_(b, n, 0x51ce77ac1bull + nb2 * 61 + n, 1);
75 lmmp_div_(NULL, b, b, n, m, n);
76 b[0] |= 1;
77 lmmp_seed_random_(e, n, 0x25c0ff350bull + nb2 * 17 + n, 1);
78 e[n - 1] |= (mp_limb_t)1 << 63;
79 double us;
80 BENCH_MEASURE(us, o, lmmp_powmod_(dst, b, e, n, m, n); bench_sink += dst[0];);
81 lmmp_free(m);
82 lmmp_free(b);
83 lmmp_free(e);
84 lmmp_free(dst);
85 return us;
86}

90/*
91 档位含 REDC_MERSENNE_THRESHOLD(401) 两侧的 400/402。指数与模数同规模
92 意味着单 op 成本随 n^3 增长（64L 已 ~9ms、400L 已 ~1s），默认阶梯到
93 阈值上下即止；更大尺寸用 bench_powmod_odd(n, &bench_default) 单点按需测量。
94*/
95static const mp_size_t POWMOD_SZ[] = {1, 8, 64, 200, 400, 402};

99 mp_size_t n = POWMOD_SZ[i];
