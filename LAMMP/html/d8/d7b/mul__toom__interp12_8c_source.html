27#define DO_subrsh(dst, nd, src, ns, s) \
28 do { \
29 mp_limb_t __cy; \
30 lmmp_dec_1((dst), (src)[0] >> (s)); \
31 __cy = lmmp_submul_1_((dst), (src) + 1, (ns) - 1, (mp_limb_t)1 << (LIMB_BITS - (s))); \
32 lmmp_dec_1((dst) + (ns) - 1, __cy); \
33 } while (0)

40void lmmp_toom_interp12_(
41 mp_ptr dst, mp_ptr r1, mp_ptr r3, mp_ptr r5, mp_size_t n, mp_size_t spt, int half, mp_ptr tp
42) {
43 mp_limb_t cy, cy2;
44 mp_size_t n3, n3p1;
45 n3 = 3 * n;
46 n3p1 = n3 + 1;
47
48 lmmp_param_assert(n > 0);
49 lmmp_param_assert(spt > 0);
50 lmmp_param_assert(spt <= 2 * n);
51
52#define r4 (dst + n3) /* 3n+1 */
53#define r2 (dst + 7 * n) /* 3n+1 */
54#define r0 (dst + 11 * n) /* spt <= 2n */
55
56 /* 插值序列（值引用为混合后的和/差半边）：
57 r? <- (r? - r0*x)/d 的消顶部项 + 交叉消元 + 精确除，最后重组 */
58
59 if (half != 0) {
60 cy = lmmp_sub_n_(r3, r3, r0, spt);
61 lmmp_dec_1(r3 + spt, cy);
62
63 cy = lmmp_submul_1_(r2, r0, spt, (mp_limb_t)1 << 10);
64 lmmp_dec_1(r2 + spt, cy);
65 DO_subrsh(r5, n3p1, r0, spt, 2);
66
67 cy = lmmp_submul_1_(r1, r0, spt, (mp_limb_t)1 << 20);
68 lmmp_dec_1(r1 + spt, cy);
69 DO_subrsh(r4, n3p1, r0, spt, 4);
70 }
71
72 r4[n3] -= lmmp_submul_1_(r4 + n, dst, 2 * n, (mp_limb_t)1 << 20);
73 DO_subrsh(r1 + n, 2 * n + 1, dst, 2 * n, 4);
74
75 lmmp_add_n_sub_n_(r1, r4, r4, r1, n3p1);
76
77 r5[n3] -= lmmp_submul_1_(r5 + n, dst, 2 * n, (mp_limb_t)1 << 10);
78 DO_subrsh(r2 + n, 2 * n + 1, dst, 2 * n, 2);
79
80 lmmp_add_n_sub_n_(r2, r5, r5, r2, n3p1);
81
82 r3[n3] -= lmmp_sub_n_(r3 + n, r3 + n, dst, 2 * n);
83
84 lmmp_submul_1_(r4, r5, n3p1, 257);
85 /* 随后除以 2835*4，操作数可能为负（补码） */
86 lmmp_divexact_odd_pi_(r4, r4, n3p1, 2835, BINVERT_2835_, 2);
87 if ((r4[n3] & (LIMB_MAX << (LIMB_BITS - 3))) != 0)
88 r4[n3] |= LIMB_MAX << (LIMB_BITS - 2);
89
90 lmmp_addmul_1_(r5, r4, n3p1, 60);
91 lmmp_divexact_odd_pi_(r5, r5, n3p1, 255, BINVERT_255_, 0);
92
93 cy = lmmp_submul_1_(r2, r3, n3p1, (mp_limb_t)1 << 5);
94 lmmp_debug_assert(cy == 0);
95
96 cy = lmmp_submul_1_(r1, r2, n3p1, 100);
97 lmmp_debug_assert(cy == 0);
98 cy = lmmp_submul_1_(r1, r3, n3p1, (mp_limb_t)1 << 9);
99 lmmp_debug_assert(cy == 0);
100 lmmp_divexact_odd_pi_(r1, r1, n3p1, 42525, BINVERT_42525_, 0);
101
102 cy = lmmp_submul_1_(r2, r1, n3p1, 225);
103 lmmp_debug_assert(cy == 0);
104 lmmp_divexact_odd_pi_(r2, r2, n3p1, 9, BINVERT_9_, 2);
105
106 cy = lmmp_sub_n_(r3, r3, r2, n3p1);
107 lmmp_debug_assert(cy == 0);
108
109 lmmp_shr1sub_n_(r4, r2, r4, n3p1);
110 r4[n3p1 - 1] &= LIMB_MAX >> 1;
111 cy = lmmp_sub_n_(r2, r2, r4, n3p1);
112 lmmp_debug_assert(cy == 0);
113
114 lmmp_shr1add_n_(r5, r5, r1, n3p1);
115 r5[n3p1 - 1] &= LIMB_MAX >> 1;
116
117 /* 最后两步插值（可并入重组） */
118 cy = lmmp_sub_n_(r3, r3, r1, n3p1);
119 lmmp_debug_assert(cy == 0);
120 cy = lmmp_sub_n_(r1, r1, r5, n3p1);
121 lmmp_debug_assert(cy == 0);
122
123 (void)tp; /* 现有原语集（submul/addmul/原生 add_n_sub_n）不消耗临时区 */
124
125 /***************************** 重组 *******************************/
126 /*
127 dst 操作前：
128 |M r0|L r0|___||H r2|M r2|L r2|___||H r4|M r4|L r4|____|H r8|L r8|dst
129 求和方案：
130 |__12|n_11|n_10|n__9|n__8|n__7|n__6|n__5|n__4|n__3|n__2|n___|n___|dst
131 |M r0|L r0|___||H r2|M r2|L r2|___||H r4|M r4|L r4|____|H_r8|L_r8|dst
132 ||H r1|M r1|L r1| ||H r3|M r3|L r3| ||H_r5|M_r5|L_r5|
133 */
134
135 cy = lmmp_add_n_(dst + n, dst + n, r5, n);
136 cy = lmmp_add_1_(dst + 2 * n, r5 + n, n, cy);
137 cy = r5[n3] + lmmp_add_nc_(dst + n3, dst + n3, r5 + 2 * n, n, cy);
138 lmmp_inc_1(dst + n3 + n, cy);
139
140 dst[2 * n3] += lmmp_add_n_(dst + 5 * n, dst + 5 * n, r3, n);
141 cy = lmmp_add_1_(dst + 2 * n3, r3 + n, n, dst[2 * n3]);
142 cy = r3[n3] + lmmp_add_nc_(dst + 7 * n, dst + 7 * n, r3 + 2 * n, n, cy);
143 lmmp_inc_1(dst + 8 * n, cy);
144
145 dst[10 * n] += lmmp_add_n_(dst + 9 * n, dst + 9 * n, r1, n);
146 if (half) {
147 cy = lmmp_add_1_(dst + 10 * n, r1 + n, n, dst[10 * n]);
148 if (spt > n) {
149 cy = r1[n3] + lmmp_add_nc_(dst + 11 * n, dst + 11 * n, r1 + 2 * n, n, cy);
150 lmmp_inc_1(dst + 4 * n3, cy);
151 } else {
152 cy2 = lmmp_add_nc_(dst + 11 * n, dst + 11 * n, r1 + 2 * n, spt, cy);
153 lmmp_debug_assert(cy2 == 0);
154 }
155 } else {
156 cy = lmmp_add_1_(dst + 10 * n, r1 + n, spt, dst[10 * n]);
157 lmmp_debug_assert(cy == 0);
158 }
159
160#undef r4
161#undef r2
162#undef r0
163}
