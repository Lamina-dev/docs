86mp_size_t lmmp_to_hex_(char* dst, mp_srcptr numa, mp_size_t na, bool upper) {
87 lmmp_param_assert(dst != NULL && numa != NULL);
88 do {
89 if (na == 0)
90 return 0;
91 } while (numa[--na] == 0);
92 ++na;
93
94 const uint16_t* ptab = upper ? lmmp_hex_pair_upper : lmmp_hex_pair_lower;
95 mp_limb_t top = numa[na - 1];
96 int bits = lmmp_limb_bits_(top);
97 mp_size_t digits = (mp_size_t)((LIMB_BITS * (na - 1) + bits + 3) >> 2);
98
99 char* wp = dst;
100 for (mp_size_t i = 0; i < na - 1; ++i) {
101 mp_limb_t l = numa[i];
102 for (int j = 0; j < 8; ++j) {
103 uint16_t p = ptab[l & 0xff];
104 wp[0] = (char)p; // 低 nibble 字符在前
105 wp[1] = (char)(p >> 8); // 高 nibble 字符
106 wp += 2;
107 l >>= 8;
108 }
109 }
110 while (top >> 4) { // 剩余至少 2 个 nibble
111 uint16_t p = ptab[top & 0xff];
112 wp[0] = (char)p;
113 wp[1] = (char)(p >> 8);
114 wp += 2;
115 top >>= 8;
116 }
117 if (top)
118 *wp++ = (char)ptab[top];
119
120 return digits;
121}
