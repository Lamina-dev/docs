23void lmmp_inv_prediv_(mp_ptr dst, mp_srcptr numa, mp_size_t na, mp_size_t ni) {
24 lmmp_param_assert(na >= ni);
25 lmmp_param_assert(ni > 0);
26 lmmp_param_assert(numa[na - 1] >= LIMB_B_2);
27 TEMP_DECL;
28 mp_limb_t cy;
29 mp_ptr restrict tp = TALLOC_TYPE(ni + 1, mp_limb_t);
30
31 if (na == ni) {
32 lmmp_copy(tp + 1, numa, ni);
33 tp[0] = 1;
34 cy = 0;
35 } else {
36 cy = lmmp_add_1_(tp, numa + na - (ni + 1), ni + 1, 1);
37 }
38 if (cy)
39 lmmp_zero(dst, ni);
40 else {
41 mp_ptr restrict invappr = TALLOC_TYPE(ni + 1, mp_limb_t);
42 lmmp_invappr_(invappr, tp, ni + 1);
43 lmmp_copy(dst, invappr + 1, ni);
44 }
45 TEMP_FREE;
46}

48mp_limb_t lmmp_div_mulinv_(
49 mp_ptr restrict dstq,
50 mp_ptr restrict numa,
51 mp_size_t na,
52 mp_srcptr restrict numb,
53 mp_size_t nb,
54 mp_srcptr restrict invappr,
55 mp_size_t ni
56) {
57 lmmp_param_assert(na >= nb && nb >= ni);
58 lmmp_param_assert(ni > 0);
59 lmmp_param_assert(numb[nb - 1] >= LIMB_B_2);
60 mp_size_t nq = na - nb, ntp = LMMP_MIN(ni, nq) + nb;
61 mp_limb_t qh;
62 TEMP_DECL;
63 mp_ptr restrict tp = TALLOC_TYPE(ntp, mp_limb_t);
64
65 numa += nq;
66 dstq += nq;
67
68 qh = lmmp_cmp_(numa, numb, nb) >= 0;
69 if (qh) {
70 lmmp_sub_n_(numa, numa, numb, nb);
71 }
72
73 fft_gr_cache mersenne_ctx;
74 int mersenne_flag = 0;
75 fft_cache fft_ctx;
76 int fft_flag = 0;
77 int small_flag = 0;
78
79 while (nq) {
80 if (nq < ni) {
81 invappr += ni - nq;
82 ni = nq;
83 small_flag = 1;
84 }
85 numa -= ni;
86 dstq -= ni;
87 nq -= ni;
88
89 mp_size_t mn, wn;
90 mp_limb_t cy;
91
92 if (small_flag == 1 || ni < MUL_FFT_THRESHOLD)
93 lmmp_mul_n_(tp, numa + nb, invappr, ni);
94 else {
95 if (fft_flag == 0) {
96 mp_size_t hn = lmmp_fft_next_size_((2 * ni + 1) / 2);
97 lmmp_mul_fft_cache_init_(tp, hn, numa + nb, ni, invappr, ni, &fft_ctx);
98 fft_flag = 1;
99 } else {
100 lmmp_mul_fft_cache_(tp, numa + nb, &fft_ctx);
101 }
102 }
103 cy = lmmp_add_n_(dstq, tp + ni, numa + nb, ni);
104 lmmp_assert(cy == 0);
105
106 if (nb < DIV_MULINV_MODM_THRESHOLD || (mn = lmmp_fft_next_size_(nb + 1)) >= nb + ni) {
107 lmmp_mul_(tp, numb, nb, dstq, ni); // nb+ni limbs, high 'ni' cancels
108 } else {
109 // 0<wn<ni<=nb<mn<nb+ni
110 wn = nb + ni - mn;
111
112 // x=b*q
113 // tp=x mod 2^mn-1
114 if (small_flag == 1)
115 lmmp_mul_mersenne_(tp, mn, dstq, ni, numb, nb);
116 else {
117 if (mersenne_flag == 0) {
118 lmmp_mul_mersenne_cache_init_(tp, mn, dstq, ni, numb, nb, &mersenne_ctx);
119 mersenne_flag = 1;
120 } else {
121 lmmp_mul_mersenne_cache_(tp, dstq, &mersenne_ctx);
122 }
123 }
124
125 // tp-=ah:0 mod B^mn-1, if result=0, represent it as B^mn-1
126 cy = lmmp_sub_nc_(tp, tp, numa + mn, wn, 1);
127 if (cy)
128 cy = lmmp_sub_1_(tp + wn, tp + wn, mn - wn, 1);
129 if (!cy)
130 lmmp_inc(tp);
131
132 // if al<<tp,
133 if (lmmp_cmp_(numa + nb, tp + nb, mn - nb) < 0) {
134 // maybe ah=xh+1 and al<<xl,
135 // so we subtracted 1 too much when tp-=ah,
136 // now tp=xl-1 mod B^mn-1, and 0<=al<<xl-1<B^mn-1, so tp=xl-1
137 // or ah=xh and al>=xl,
138 // tp=xl mod B^mn-1, the only possibility is we represented xl=0 as tp=B^mn-1
139 // whatever, just inc and then tp=xl
140 tp[mn] = 0; // set a limit
141 lmmp_inc(tp);
142 }
143 }
144
145 mp_limb_t r = numa[nb] - tp[nb];
146 cy = lmmp_sub_n_(numa, numa, tp, nb);
147
148 while ((r -= cy) || lmmp_cmp_(numa, numb, nb) >= 0) {
149 lmmp_inc(dstq);
150 cy = lmmp_sub_n_(numa, numa, numb, nb);
151 }
152 }
153
154 if (mersenne_flag == 1)
155 lmmp_fft_gr_cache_free_(&mersenne_ctx);
156 if (fft_flag == 1)
157 lmmp_fft_cache_free_(&fft_ctx);
158 TEMP_FREE;
159 return qh;
160}
