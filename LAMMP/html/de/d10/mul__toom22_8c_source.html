29/*
30Evaluate in: -1, 0, +inf
31
32 <-s--><--n-->
33 |-a1-|--a0--|
34 |b1-|--b0--|
35 <-t-><--n-->
36
37v0 = a0 * b0 # A(0)*B(0)
38vm1 = (a0-a1)*(b0-b1) # A(-1)*B(-1)
39vinf= a1 * b1 # A(inf)*B(inf)
40*/

42void lmmp_mul_toom22_(mp_ptr restrict dst, mp_srcptr restrict numa, mp_size_t na, mp_srcptr restrict numb, mp_size_t nb) {
43 lmmp_param_assert(nb >= 5);
44 lmmp_param_assert(na >= nb);
45 lmmp_param_assert(4 * na <= 5 * nb);
46 TEMP_S_DECL;
47 mp_size_t s = na >> 1, n = na - s, t = nb - n;
48 mp_limb_t* restrict vm1 = SALLOC_TYPE(2 * n, mp_limb_t);
49 int vm1_neg = 0;
50 mp_slimb_t cy, cy2;
51
52#define a0 numa
53#define a1 (numa + n)
54#define b0 numb
55#define b1 (numb + n)
56#define asm1 dst
57#define bsm1 (dst + n)
58
59 if (s == n) {
60 if (lmmp_cmp_(a0, a1, n) < 0) {
61 lmmp_sub_n_(asm1, a1, a0, n);
62 vm1_neg = 1;
63 } else
64 lmmp_sub_n_(asm1, a0, a1, n);
65 } else { // s==n-1
66 if (a0[s] == 0 && lmmp_cmp_(a0, a1, s) < 0) {
67 lmmp_sub_n_(asm1, a1, a0, s);
68 asm1[s] = 0;
69 vm1_neg = 1;
70 } else
71 asm1[s] = a0[s] - lmmp_sub_n_(asm1, a0, a1, s);
72 }
73
74 if (t == n) {
75 if (lmmp_cmp_(b0, b1, n) < 0) {
76 lmmp_sub_n_(bsm1, b1, b0, n);
77 vm1_neg ^= 1;
78 } else
79 lmmp_sub_n_(bsm1, b0, b1, n);
80 } else {
81 if (lmmp_zero_q_(b0 + t, n - t) && lmmp_cmp_(b0, b1, t) < 0) {
82 lmmp_sub_n_(bsm1, b1, b0, t);
83 lmmp_zero(bsm1 + t, n - t);
84 vm1_neg ^= 1;
85 } else
86 lmmp_sub_(bsm1, b0, n, b1, t);
87 }
88
89 lmmp_mul_n_(vm1, asm1, bsm1, n);
90
91#undef asm1
92#undef bsm1
93#define v0 dst
94#define vinf (dst + 2 * n)
95
96 lmmp_mul_n_(v0, a0, b0, n);
97
98 lmmp_mul_(vinf, a1, s, b1, t);
99
100 cy = lmmp_add_n_(dst + 2 * n, v0 + n, vinf, n);
101 cy2 = cy + lmmp_add_n_(dst + n, dst + 2 * n, v0, n);
102 cy += lmmp_add_(dst + 2 * n, dst + 2 * n, n, vinf + n, s + t - n);
103
104 if (vm1_neg)
105 cy += lmmp_add_n_(dst + n, dst + n, vm1, 2 * n);
106 else
107 cy -= lmmp_sub_n_(dst + n, dst + n, vm1, 2 * n);
108
109 // no overflow, if s+t>n. proved.
110 lmmp_inc_1(dst + 2 * n, cy2);
111
112 if (cy < 0)
113 lmmp_dec(dst + 3 * n);
114 else
115 lmmp_inc_1(dst + 3 * n, cy);
116 TEMP_S_FREE;
117}
