28static void lmmp_gcd_lehmer_step_(slong u, slong v, mp_gcd_lehmer_t* gcd) {
29#define A (gcd->m11)
30#define B (gcd->m12)
31#define C (gcd->m21)
32#define D (gcd->m22)
33
34 lmmp_debug_assert(u >= 0 && v >= 0);
35 lmmp_debug_assert(u >= v);
36 A = 1; B = 0;
37 C = 0; D = 1;
38
39 while (v != 0) {
40 slong q = u / v;
41 slong t = u % v;
42
43 u = v;
44 v = t;
45
46 t = A - q * C;
47 A = C;
48 C = t;
49 t = B - q * D;
50 B = D;
51 D = t;
52
53 if (v < (slong)LEHMER_MIN_V) break;
54 }
55
56 return;
57#undef A
58#undef B
59#undef C
60#undef D
61}

63static void lmmp_lehmer_extract_(mp_srcptr up, mp_size_t un, mp_srcptr vp, mp_size_t vn, slong* restrict a, slong* restrict b) {
64 lmmp_param_assert(un > 1 && vn > 1);
65 lmmp_param_assert(un >= vn);
66 lmmp_param_assert(up != NULL && vp != NULL);
67 lmmp_param_assert(a != NULL && b != NULL);
68
69 int kz = lmmp_limb_bits_(up[un - 1]);
70 if (kz >= LEHMER_EXACT_BITS) {
71 *a = up[un - 1] >> (kz - LEHMER_EXACT_BITS);
72 if (vn == un)
73 *b = vp[vn - 1] >> (kz - LEHMER_EXACT_BITS);
74 else
75 *b = 0;
76 } else {
77 *a = up[un - 1] << (LEHMER_EXACT_BITS - kz);
78 *a |= up[un - 2] >> (LIMB_BITS - (LEHMER_EXACT_BITS - kz));
79 if (un > vn + 1) {
80 *b = 0;
81 } else if (un == vn + 1) {
82 *b = vp[vn - 1] >> (LIMB_BITS - (LEHMER_EXACT_BITS - kz));
83 } else {
84 *b = vp[vn - 1] << (LEHMER_EXACT_BITS - kz);
85 *b |= vp[vn - 2] >> (LIMB_BITS - (LEHMER_EXACT_BITS - kz));
86 }
87 }
88}

99/**
100 * @brief / a \ = / A B \ * / a \
101 * \ b / \ C D / \ b /
102 * @warning [a,an] > [b,bn]
103 * @note 不保证返回结果 [a,an] > [b,bn]
104 * @return a和b是否有一个为0
105 */
106static bool lmmp_lehmer_mul_(mp_ptr a, mp_size_t* an, mp_ptr b, mp_size_t* bn, mp_gcd_lehmer_t* M, lehmer_stack_t* ms) {
107#define A (M->m11)
108#define B (M->m12)
109#define C (M->m21)
110#define D (M->m22)
111#define an (*an)
112#define bn (*bn)
113 if (A == 0) {
114 /* / 0 1 \ / a \
115 \ 1 -q / \ b / */
116 lmmp_debug_assert(B == 1 && C == 1 && D < 0);
117 mp_limb_t c = lmmp_mul_1_(ms->tp, b, bn, -D);
118 if (c != 0) {
119 ++bn;
120 (ms->tp)[bn - 1] = c;
121 }
122 if (an > bn) {
123 lmmp_sub_(a, a, an, b, bn);
124 } else if (an == bn) {
125 int cmp = lmmp_cmp_(a, b, an);
126 if (cmp >= 0)
127 lmmp_sub_(a, a, an, b, bn);
128 else
129 lmmp_sub_(a, b, bn, a, an);
130 } else {
131 lmmp_sub_(a, b, bn, a, an);
132 an = bn;
133 }
134 while (a[an - 1] == 0 && an > 0) {
135 --an;
136 }
137 // return b = b
138 // a = a - q * b
139 return an == 0;
140 } else {
141 if (A < 0) {
142 A = -A;
143 D = -D;
144 } else {
145 B = -B;
146 C = -C;
147 }
148 // A * a + B * b
149 mp_limb_t ca = lmmp_mul_1_(ms->tp, a, an, A);
150 if (ca == 0)
151 ms->tn = an;
152 else {
153 ms->tn = an + 1;
154 (ms->tp)[ms->tn - 1] = ca;
155 }
156 ca = lmmp_mul_1_(ms->mp, b, bn, B);
157 if (ca == 0)
158 ms->mn = bn;
159 else {
160 ms->mn = bn + 1;
161 (ms->mp)[ms->mn - 1] = ca;
162 }
163
164 if (ms->tn > ms->mn) {
165 lmmp_sub_(ms->np, ms->tp, ms->tn, ms->mp, ms->mn);
166 ms->nn = ms->tn;
167 } else if (ms->mn > ms->tn) {
168 lmmp_sub_(ms->np, ms->mp, ms->mn, ms->tp, ms->tn);
169 ms->nn = ms->mn;
170 } else {
171 int cmp = lmmp_cmp_(ms->tp, ms->mp, ms->tn);
172 if (cmp >= 0) {
173 lmmp_sub_(ms->np, ms->tp, ms->tn, ms->mp, ms->mn);
174 ms->nn = ms->tn;
175 } else {
176 lmmp_sub_(ms->np, ms->mp, ms->mn, ms->tp, ms->tn);
177 ms->nn = ms->mn;
178 }
179 }
180 while (ms->np[ms->nn - 1] == 0 && ms->nn > 0) {
181 --(ms->nn);
182 }
183
184 // C * a + D * b
185 ca = lmmp_mul_1_(ms->tp, a, an, C);
186 if (ca == 0)
187 ms->tn = an;
188 else {
189 ms->tn = an + 1;
190 (ms->tp)[ms->tn - 1] = ca;
191 }
192 ca = lmmp_mul_1_(ms->mp, b, bn, D);
193 if (ca == 0)
194 ms->mn = bn;
195 else {
196 ms->mn = bn + 1;
197 (ms->mp)[ms->mn - 1] = ca;
198 }
199
200 if (ms->tn > ms->mn) {
201 lmmp_sub_(a, ms->tp, ms->tn, ms->mp, ms->mn);
202 an = ms->tn;
203 } else if (ms->mn > ms->tn) {
204 lmmp_sub_(a, ms->mp, ms->mn, ms->tp, ms->tn);
205 an = ms->mn;
206 } else {
207 int cmp = lmmp_cmp_(ms->tp, ms->mp, ms->tn);
208 if (cmp >= 0) {
209 lmmp_sub_(a, ms->tp, ms->tn, ms->mp, ms->mn);
210 an = ms->tn;
211 } else {
212 lmmp_sub_(a, ms->mp, ms->mn, ms->tp, ms->tn);
213 an = ms->mn;
214 }
215 }
216 while (a[an - 1] == 0 && an > 0) {
217 --an;
218 }
219
220 // now a = C * a + D * b
221 // ms->np = A * a + B * b
222 if (ms->nn > 0) {
223 lmmp_copy(b, ms->np, ms->nn);
224 bn = ms->nn;
225 return an == 0;
226 } else {
227 b[0] = 0;
228 bn = 0;
229 return true;
230 }
231 }
232#undef A
233#undef B
234#undef C
235#undef D
236#undef an
237#undef bn
238}

240mp_size_t lmmp_gcd_lehmer_(mp_ptr dst, mp_srcptr up, mp_size_t un, mp_srcptr vp, mp_size_t vn) {
241 lmmp_param_assert(un > 0 && vn > 0);
242 lmmp_param_assert(up != NULL && vp != NULL);
243 lmmp_param_assert(dst != NULL);
244
245 if (un < vn) {
246 LMMP_SWAP(up, vp, mp_srcptr);
247 LMMP_SWAP(un, vn, mp_size_t);
248 } else if (un == vn) {
249 int cmp = lmmp_cmp_(up, vp, un);
250 if (cmp == 0) {
251 lmmp_copy(dst, up, un);
252 return un;
253 } else if (cmp < 0) {
254 LMMP_SWAP(up, vp, mp_srcptr);
255 }
256 }
257 // u > v
258
259 mp_gcd_lehmer_t M;
260 slong x = 0, y = 0;
261
262#define an un
263#define bn vn
264 TEMP_B_DECL;
265 // [a,an+1] [b,bn+1]
266 // A * a_old may overlow
267 mp_ptr a = BALLOC_TYPE(an + 1, mp_limb_t);
268 mp_ptr b = BALLOC_TYPE(bn + 1, mp_limb_t);
269 lehmer_stack_t ms;
270 mp_ptr temp = BALLOC_TYPE((an + 1) * 3, mp_limb_t);
271 ms.tp = temp;
272 ms.mp = temp + (an + 1);
273 ms.np = temp + (an + 1) * 2;
274
275 lmmp_copy(a, up, an);
276 lmmp_copy(b, vp, bn);
277
278 bool bzero = false;
279 while (bzero == false) {
280 if (an > 1 && bn == 1) {
281 dst[0] = lmmp_gcd_1_(a, an, b[0]);
282 return 1;
283 } else if (an == 1 && bn == 1) {
284 dst[0] = lmmp_gcd_11_(a[0], b[0]);
285 return 1;
286 }
287 // a > b
288 lmmp_lehmer_extract_(a, an, b, bn, &x, &y);
289 lmmp_gcd_lehmer_step_(x, y, &M);
290
291 if (M.m21 == 0) {
292 lmmp_div_(NULL, dst, a, an, b, bn);
293 lmmp_copy(a, b, bn);
294 an = bn;
295 while (dst[bn - 1] == 0 && bn > 0) {
296 --bn;
297 }
298 if (bn == 0)
299 bzero = true;
300 else
301 lmmp_copy(b, dst, bn);
302 } else {
303 bzero = lmmp_lehmer_mul_(a, &an, b, &bn, &M, &ms);
304 if ((an < bn) || (an == bn && lmmp_cmp_(a, b, an) < 0)) {
305 LMMP_SWAP(a, b, mp_ptr);
306 LMMP_SWAP(an, bn, mp_size_t);
307 }
308 }
309 }
310 lmmp_copy(dst, a, an);
311 TEMP_B_FREE;
312 return an;
313#undef an
314#undef bn
315}
