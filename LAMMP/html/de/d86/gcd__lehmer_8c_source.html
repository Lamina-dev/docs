39/**
40 * @brief dst = |x - y|（大数减法取绝对值），返回归一化长度
41 * @warning eqsep(dst,[x|y])
42 */
43static mp_size_t lmmp_lehmer_sub_mag_(mp_ptr dst, mp_srcptr x, mp_size_t xn, mp_srcptr y, mp_size_t yn) {
44 mp_size_t rn;
45 if (xn > yn) {
46 lmmp_sub_(dst, x, xn, y, yn);
47 rn = xn;
48 } else if (xn < yn) {
49 lmmp_sub_(dst, y, yn, x, xn);
50 rn = yn;
51 } else {
52 int cmp = lmmp_cmp_(x, y, xn);
53 if (cmp >= 0) {
54 lmmp_sub_(dst, x, xn, y, yn);
55 } else {
56 lmmp_sub_(dst, y, yn, x, xn);
57 }
58 rn = xn;
59 }
60 while (rn > 0 && dst[rn - 1] == 0) {
61 --rn;
62 }
63 return rn;
64}

66/**
67 * @brief / a \ = / A B \ * / a \
68 * \ b / \ C D / \ b /
69 * @warning [a,an] > [b,bn]
70 * @note 不保证返回结果 [a,an] > [b,bn]
71 * @return a和b是否有一个为0；返回 true 时保证 [a,an] 非零（若 a 先归零
72 * 则将 b 移入 a，gcd 即 a）
73 */
74static bool lmmp_lehmer_mul_(mp_ptr a, mp_size_t* an, mp_ptr b, mp_size_t* bn, lmmp_gcd_lehmer_t* M, lehmer_stack_t* ms) {
75#define A (M->m11)
76#define B (M->m12)
77#define C (M->m21)
78#define D (M->m22)
79#define an (*an)
80#define bn (*bn)
81 if (A == 0) {
82 /* / 0 1 \ / a \ = / b \
83 \ 1 -q / \ b / \ a - q*b (new a) / */
84 lmmp_debug_assert(B == 1 && C == 1 && D < 0);
85 mp_limb_t c = lmmp_mul_1_(ms->tp, b, bn, -D);
86 mp_size_t tn = bn;
87 if (c != 0) {
88 ++tn;
89 (ms->tp)[tn - 1] = c;
90 }
91 // a = |a - q*b|, b = b
92 an = lmmp_lehmer_sub_mag_(a, a, an, ms->tp, tn);
93 if (an == 0) {
94 // a - q*b == 0：b | a，gcd = b，移入 a
95 lmmp_copy(a, b, bn);
96 an = bn;
97 b[0] = 0;
98 bn = 0;
99 }
100 return bn == 0;
101 } else {
102 if (A < 0) {
103 A = -A;
104 D = -D;
105 } else {
106 B = -B;
107 C = -C;
108 }
109 // ms->np = |A * a - (-B) * b|（B 取负后系数均为正，差的绝对值即线性组合）
110 mp_limb_t ca = lmmp_mul_1_(ms->tp, a, an, A);
111 ms->tn = an;
112 if (ca != 0)
113 (ms->tp)[ms->tn++] = ca;
114 ca = lmmp_mul_1_(ms->mp, b, bn, B);
115 ms->mn = bn;
116 if (ca != 0)
117 (ms->mp)[ms->mn++] = ca;
118 ms->nn = lmmp_lehmer_sub_mag_(ms->np, ms->tp, ms->tn, ms->mp, ms->mn);
119
120 // a = |C * a - (-D) * b|
121 ca = lmmp_mul_1_(ms->tp, a, an, C);
122 ms->tn = an;
123 if (ca != 0)
124 (ms->tp)[ms->tn++] = ca;
125 ca = lmmp_mul_1_(ms->mp, b, bn, D);
126 ms->mn = bn;
127 if (ca != 0)
128 (ms->mp)[ms->mn++] = ca;
129 an = lmmp_lehmer_sub_mag_(a, ms->tp, ms->tn, ms->mp, ms->mn);
130
131 // now a = C * a + D * b
132 // ms->np = A * a + B * b
133 if (an == 0) {
134 // a 归零：gcd = ms->np，移入 a
135 lmmp_copy(a, ms->np, ms->nn);
136 an = ms->nn;
137 b[0] = 0;
138 bn = 0;
139 return true;
140 }
141 if (ms->nn > 0) {
142 lmmp_copy(b, ms->np, ms->nn);
143 bn = ms->nn;
144 return false;
145 } else {
146 b[0] = 0;
147 bn = 0;
148 return true;
149 }
150 }
151#undef A
152#undef B
153#undef C
154#undef D
155#undef an
156#undef bn
157}

159mp_size_t lmmp_gcd_lehmer_(mp_ptr dst, mp_srcptr up, mp_size_t un, mp_srcptr vp, mp_size_t vn) {
160 lmmp_param_assert(un > 0 && vn > 0);
161 lmmp_param_assert(up != NULL && vp != NULL);
162 lmmp_param_assert(dst != NULL);
163
164 lmmp_param_assert(up[un - 1] != 0);
165 lmmp_param_assert(vp[vn - 1] != 0);
166
167 if (un < vn) {
168 LMMP_SWAP(up, vp, mp_srcptr);
169 LMMP_SWAP(un, vn, mp_size_t);
170 } else if (un == vn) {
171 int cmp = lmmp_cmp_(up, vp, un);
172 if (cmp == 0) {
173 lmmp_copy(dst, up, un);
174 return un;
175 } else if (cmp < 0) {
176 LMMP_SWAP(up, vp, mp_srcptr);
177 }
178 }
179 // u > v
180
181 lmmp_gcd_lehmer_t M;
182 slong x = 0, y = 0;
183
184#define an un
185#define bn vn
186 TEMP_B_DECL;
187 // [a,an+1] [b,bn+1]
188 // A * a_old may overlow
189 mp_ptr a = BALLOC_TYPE(an + 1, mp_limb_t);
190 mp_ptr b = BALLOC_TYPE(bn + 1, mp_limb_t);
191 lehmer_stack_t ms;
192 mp_ptr temp = BALLOC_TYPE((an + 1) * 3, mp_limb_t);
193 ms.tp = temp;
194 ms.mp = temp + (an + 1);
195 ms.np = temp + (an + 1) * 2;
196
197 lmmp_copy(a, up, an);
198 lmmp_copy(b, vp, bn);
199
200 bool bzero = false;
201 while (bzero == false) {
202 if (an > 1 && bn == 1) {
203 dst[0] = lmmp_gcd_1_(a, an, b[0]);
204 TEMP_B_FREE;
205 return 1;
206 } else if (an == 1 && bn == 1) {
207 dst[0] = lmmp_gcd_11_(a[0], b[0]);
208 TEMP_B_FREE;
209 return 1;
210 }
211 // a > b
212 lmmp_gcd_lehmer_extract_(a, an, b, bn, &x, &y);
213 lmmp_gcd_lehmer_step_(x, y, &M);
214
215 if (M.m21 == 0) {
216 lmmp_div_(NULL, dst, a, an, b, bn);
217 lmmp_copy(a, b, bn);
218 an = bn;
219 while (bn > 0 && dst[bn - 1] == 0) {
220 --bn;
221 }
222 if (bn == 0)
223 bzero = true;
224 else
225 lmmp_copy(b, dst, bn);
226 } else {
227 bzero = lmmp_lehmer_mul_(a, &an, b, &bn, &M, &ms);
228 if ((an < bn) || (an == bn && lmmp_cmp_(a, b, an) < 0)) {
229 LMMP_SWAP(a, b, mp_ptr);
230 LMMP_SWAP(an, bn, mp_size_t);
231 }
232 }
233 }
234 lmmp_copy(dst, a, an);
235 TEMP_B_FREE;
236 return an;
237#undef an
238#undef bn
239}
