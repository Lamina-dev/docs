27mp_size_t lmmp_gcd_(mp_ptr dst, mp_srcptr up, mp_size_t un, mp_srcptr vp, mp_size_t vn) {
28 lmmp_param_assert(un > 0 && vn > 0);
29 lmmp_param_assert(up != NULL && vp != NULL && dst != NULL);
30 lmmp_param_assert(up[un - 1] != 0);
31 lmmp_param_assert(vp[vn - 1] != 0);
32
33 if (un < vn) {
34 LMMP_SWAP(up, vp, mp_srcptr);
35 LMMP_SWAP(un, vn, mp_size_t);
36 }
37 // u >= v
38
39 if (vn == 1) {
40 dst[0] = (un == 1) ? lmmp_gcd_11_(up[0], vp[0]) : lmmp_gcd_1_(up, un, vp[0]);
41 return 1;
42 }
43 if (vn == 2) {
44 return (un == 2) ? lmmp_gcd_22_(dst, up, vp) : lmmp_gcd_2_(dst, up, un, vp);
45 }
46 if (un >= GCD_HGCD_THRESHOLD) {
47 return lmmp_gcd_hgcd_(dst, up, un, vp, vn);
48 }
49 return lmmp_gcd_lehmer_(dst, up, un, vp, vn);
50}
