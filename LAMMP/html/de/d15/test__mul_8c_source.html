30/*
31 乘法家族验证网（无 BigInt oracle）。交叉验证铁律：参考算法与被测算法
32 规模必须匹配——
33 1. 小规模（≤ toom 域）：分派器 vs basecase 精确比对（O(n²) 参考在该
34 规模是匹配的独立算法）+ n=1 经 mullh 精确 + 模检查 + 长度不变式
35 2. FFT 域（≥ MUL_FFT_THRESHOLD / SQR_FFT_THRESHOLD）：mul_fft/sqr_fft
36 （分派器所选）vs toom8h/sqr_toom8 直调（纯 Toom 递归，其子问题
37 永不进 FFT：MAYBE_mul_toom8h=0、SQR_TOOM8_MAX<<阈值）——同规模
38 两条独立算法族互比；禁用 basecase（大规模不匹配）
39 3. 平方 == 自乘（sqr vs mul 分派交叉）、mullo == mul 截断
40 4. 模检查：乘积在 32 位素数集下与输入模乘值一致（数学独立）
41 5. 长度不变式：归一化输入的乘积长度 ∈ {na+nb-1, na+nb}
42*/
43
44// 模检查 + 长度不变式
45static void check_prod_mod(const mp_limb_t* a, mp_size_t na, const mp_limb_t* b, mp_size_t nb,
46 const mp_limb_t* dst) {
47 TEST_CHECK_MSG(mod_is_product(dst, na + nb, a, na, b, nb), "modular product check");
48 mp_size_t pl = norm_len(dst, na + nb);
49 TEST_CHECK(pl == na + nb || pl == na + nb - 1);
50}

52// 两条路径精确比对（有效长度 + 逐 limb）
53static bool limbs_same(const mp_limb_t* x, const mp_limb_t* y, mp_size_t n) {
54 return memcmp(x, y, (size_t)n * sizeof(mp_limb_t)) == 0;
55}

57TEST_CASE("mul/basecase", mul_basecase_mod_and_n1_exact) {
58 u64 seed = 0x13579bdf2468ace0ull;
59 for (size_t na = 1; na <= 40; ++na) {
60 for (size_t nb = 1; nb <= na; ++nb) {
61 mp_ptr a = alloc_limbs(na), b = alloc_limbs(nb), dst = alloc_limbs(na + nb + 1);
62 random_limbs(a, na, &seed);
63 random_limbs(b, nb, &seed);
64 lmmp_mul_basecase_(dst, a, na, b, nb);
65 check_prod_mod(a, (mp_size_t)na, b, (mp_size_t)nb, dst);
66
67 // n=1：与 mullh 精确比对
68 if (na == 1 && nb == 1) {
69 mp_limb_t ref[2];
70 lmmp_mullh_(a[0], b[0], ref);
71 TEST_CHECK_EQ(dst[0], ref[0]);
72 TEST_CHECK_EQ(dst[1], ref[1]);
73 }
74 lmmp_free(a); lmmp_free(b); lmmp_free(dst);
75 }
76 }
77 // 写死常数锚点：(2^64-1)^2（低 2 limbs）与 (2^63+1)*(2^64-1)
78 mp_ptr a = alloc_limbs(1), b = alloc_limbs(1), d = alloc_limbs(3);
79 a[0] = ~(u64)0; b[0] = ~(u64)0;
80 lmmp_mul_basecase_(d, a, 1, b, 1);
81 TEST_CHECK_EQ(d[0], 1ull);
82 TEST_CHECK_EQ(d[1], ~(u64)0 - 1);
83 a[0] = ((u64)1 << 63) + 1;
84 lmmp_mul_basecase_(d, a, 1, b, 1);
85 // (2^63+1)*(2^64-1) = 2^127 + 2^63 - 1
86 TEST_CHECK_EQ(d[0], 0x7FFFFFFFFFFFFFFFull);
87 TEST_CHECK_EQ(d[1], 0x8000000000000000ull);
88 lmmp_free(a); lmmp_free(b); lmmp_free(d);
89}

104TEST_CASE("mul/dispatch", mul_dispatch_vs_basecase_cross) {
105 // toom 域内分派器 vs basecase（匹配的独立参考；O(n²) 成本限规模）
106 u64 seed = 0x0123456789abcdefull;
107 const mp_size_t sizes[] = {
108 1, 2, 5, 19, 20, 21, 30, 40, 64, 65, 66, 80, 100, 120,
109 150, 180, 220, 300, 400, 580, 581, 582, 700, 900, 1200
110 };
111 for (size_t na_index = 0; na_index < ARRAY_SIZE(sizes); ++na_index) {
112 mp_size_t na = sizes[na_index];
113 for (int t = 0; t < 3; ++t) {
114 mp_size_t nb = (mp_size_t)((u64)na * (t == 0 ? 1 : (t == 1 ? 2 : 3)) / (t == 0 ? 1 : (t == 1 ? 3 : 4)));
115 if (nb < 1) nb = 1;
116 if (nb > na) nb = na;
117 // basecase 为 O(n^2)，限制总规模。
118 if ((u64)na * nb > 1200000ull) continue;
119
120 mp_ptr a = alloc_limbs(na), b = alloc_limbs(nb);
121 mp_ptr dst = alloc_limbs(na + nb + 2), ref = alloc_limbs(na + nb + 2);
122 random_limbs(a, na, &seed);
123 random_limbs(b, nb, &seed);
124 lmmp_mul_(dst, a, na, b, nb);
125 lmmp_mul_basecase_(ref, a, na, b, nb);
126 TEST_CHECK(limbs_same(dst, ref, na + nb));
127 check_prod_mod(a, na, b, nb, dst);
128 lmmp_free(a); lmmp_free(b); lmmp_free(dst); lmmp_free(ref);
129 }
130 }

133TEST_CASE("mul/fft", mul_fft_vs_toom8h) {
134 // FFT 域（n >= MUL_FFT_THRESHOLD 分派器选 FFT）：与纯 Toom 递归直调
135 // toom8h 互比（同规模独立算法族；basecase 在该规模不匹配，禁用）。
136 // toom8h 契约：nb>=86、na<=4nb（原生比例含平衡与 4:1 不平衡）。
137 u64 seed = 0xf1f2f3f4f5f6f7f8ull;
138 struct operand_sizes { mp_size_t na, nb; } sizes[] = {
139 {2345, 2345}, {2400, 1200}, {2400, 800}, {2400, 601},
140 {3072, 3072}, {4096, 1024}, {5000, 1500}, {6000, 6000},
141 };
142 for (size_t s_index = 0; s_index < ARRAY_SIZE(sizes); ++s_index) {
143 const struct operand_sizes s = sizes[s_index];
144 mp_ptr a = alloc_limbs(s.na), b = alloc_limbs(s.nb);
145 mp_ptr dst = alloc_limbs(s.na + s.nb + 2), ref = alloc_limbs(s.na + s.nb + 2);
146 random_limbs(a, s.na, &seed);
147 random_limbs(b, s.nb, &seed);
148 lmmp_mul_(dst, a, s.na, b, s.nb); // 分派器 → FFT
149 lmmp_mul_toom8h_(ref, a, s.na, b, s.nb); // 纯 Toom 递归
150 TEST_CHECK_MSG(limbs_same(dst, ref, s.na + s.nb), "fft vs toom8h");
151 check_prod_mod(a, s.na, b, s.nb, dst);
152 lmmp_free(a); lmmp_free(b); lmmp_free(dst); lmmp_free(ref);
153 }

202TEST_CASE("mul/hard_direct_all_sizes", mul_hard_direct_n1_to_n19) {
203 /* 硬编码路径全规模直测: n=1..19 经顶层 lmmp_mul_n_/lmmp_sqr_ (阈值下
204 必走硬编码调度) 对照 basecase 与模检查, 覆盖随机/全 1/全 0xFF/最高位等
205 触发进位链极端的输入模式 */
206 u64 seed = 0x0d1c2b3a49586775ull;
207 for (mp_size_t n = 1; n <= LMMP_MUL_HARD_MAX_N; n++) {
208 for (int pat = 0; pat < 6; pat++) {
209 mp_ptr a = alloc_limbs(n), b = alloc_limbs(n);
210 mp_ptr dst = alloc_limbs(2 * n + 2), ref = alloc_limbs(2 * n + 2);
211 for (mp_size_t i = 0; i < n; i++) {
212 switch (pat) {
213 case 0: a[i] = b[i] = xorshift64(&seed); break;
214 case 1: a[i] = b[i] = ~(u64)0; break;
215 case 2: a[i] = b[i] = 1; break;
216 case 3: a[i] = b[i] = (u64)1 << 63; break;
217 case 4: a[i] = b[i] = (~(u64)0) - (u64)(i & 1); break;
218 default: a[i] = b[i] = xorshift64(&seed) >> (i & 63); break;
219 }
220 }
221 lmmp_mul_n_(dst, a, b, n);
222 lmmp_mul_basecase_(ref, a, n, b, n);
223 TEST_CHECK(limbs_same(dst, ref, 2 * n));
224 // 模检查（pattern 输入可能非归一化，用值语义仍成立）
225 TEST_CHECK(mod_is_product(dst, 2 * n, a, n, b, n));
226
227 lmmp_sqr_(dst, a, n);
228 lmmp_mul_basecase_(ref, a, n, a, n);
229 TEST_CHECK(limbs_same(dst, ref, 2 * n));
230 TEST_CHECK(mod_is_product(dst, 2 * n, a, n, a, n));
231 lmmp_free(a); lmmp_free(b); lmmp_free(dst); lmmp_free(ref);
232 }
233 }
