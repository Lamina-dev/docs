264/**
265 * @brief hgcd 变换矩阵
266 * @note / a \ / m00 m01 \ / a' \
267 * | | = | | * | |
268 * \ b / \ m10 m11 / \ b' /
269 * 其中 (a;b) 为 hgcd 入口数对，(a';b') 为归约后数对。矩阵元素均非负，
270 * det(M) = ±1，元素值不超过入口较大分量的规模。
271 * 每个元素显式存储真实长度 n[i][j]（归一化，顶 limb 非零；
272 * n[i][j]==0 表示零元素），不做零填充：元素缓冲 [0, n[i][j]) 之外
273 * 的内容无效，一切访问以长度为准。长度由库内计算过程显式维护
274 * （乘法进位/最高位），不做前导零扫描。
275 * 各元素长度 <= alloc。
276 */
277typedef struct {
278 mp_ptr m[2][2]; /* 元素指针，各指向 alloc limbs 的连续区域 */
279 mp_size_t n[2][2]; /* 各元素真实长度（归一化；0 表示零元素） */
280 mp_size_t alloc; /* 每元素容量（limb） */
281} lmmp_hgcd_matrix_t;

714/**
715 * @brief 大整数素性检验（无试除法，强度分档，大于 128 位）
716 * @param np 待测数指针（nn 个limb）
717 * @param nn 待测数的 limb 长度
718 * @param strength 检测强度 [0,7]。随机基底 MR 分两层："1+n" 搭配指先跑至多
719 * x 轮 1 limb 随机基底（a in [3,B) 均匀，廉价，可尽早拒判）再跑至多 y 轮
720 * 全域（n limb 随机基底，[2,n-2] 均匀）基底。各层均为“至多”轮数，任一轮检出
721 * 合数即提前终止：
722 * | 强度| 基底2 MR | 1 limb 基底 MR | n limb 基底 MR | 强 Lucas |
723 * | --- | ------- | -------------- | -------------- | -------- |
724 * | 0 | √ | 3 | 2 | - |
725 * | 1 | √ | 4 | 3 | - |
726 * | 2 | √ | 5 | 4 | - |
727 * | 3 | √ | 6 | 5 | - |
728 * | 4 | √ | 0 | 0 | √ |
729 * | 5 | √ | 2 | 0 | √ |
730 * | 6 | √ | 3 | 1 | √ |
731 * | 7 | √ | 4 | 2 | √ |
732 * @warning np!=NULL, nn>2, np[nn-1]>0, np[0]%2==1, 0<=strength<=7
733 * @note 1. 不进行试除法，对于含有明显小因子的输入不会早退。建议输入不含小因子或强伪素数、半素数等。
734 * 2. 建议实际使用强度 5 档，能够很好的平衡伪素数和素数的平均耗时，对于真素数不会耗费过长的时间，
735 * 且可以更快的过滤伪素数。该档为 BPSW 型判据（基底 2 MR 叠加强 Lucas）之上增加 2 轮
736 * 1 limb 随机基底 MR；其中 Lucas 半部含 BFW vpsp 与 Q-Euler 两项强化检查
737 * （Baillie-Fiori-Wagstaff 2021，素数必然全过），故接受集为标准 BPSW 接受集的子集
738 * （合数只可能多拒、不可能少拒）。
739 * 3. 强卢卡斯判据本身和 MR 测量相当正交，因此强度档大于4时，增加更多轮次 MR 测试存在边际效应，
740 * 并不能无条件的降低 BPSW 条件下的实际误判率。更多的 MR 轮次，只承担早退的任务，对于部分
741 * 伪素数可以更早的返回，而不是进行卢卡斯检验。但对于可能通过的（伪）素数，总时间反而增加。
742 * @attention 1 limb 基底空间仅 2^64，可能存在精心构造的伪素数，使其可以通过 1 limb 基底空间的所有
743 * MR 测试。此测试主要用于搭配全基底空间快速过滤，不承诺对抗性输入下的强度保证。
744 * @return 0 = 合数；1 = 素数（确定性判据，本契约域不可达）；2 = 极大概率为素数
745 */
746LMMP_API int lmmp_is_prime_n_notrial_(mp_srcptr np, mp_size_t nn, int strength);
