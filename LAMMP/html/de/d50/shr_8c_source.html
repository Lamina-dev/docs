26mp_limb_t lmmp_shr_(mp_ptr dst, mp_srcptr numa, mp_size_t na, mp_size_t shr) {
27 if (shr == 0) {
28 lmmp_copy(dst, numa, na);
29 return 0;
30 }
31 const mp_size_t rshr = LIMB_BITS - shr;
32 mp_limb_t retval = numa[0] << rshr;
33 mp_size_t i = 0;
34 for (; i + 1 < na; i++) dst[i] = (numa[i] >> shr) | (numa[i + 1] << rshr);
35 dst[i] = numa[i] >> shr;
36 return retval;
37}

53mp_limb_t lmmp_shr1add_nc_(mp_ptr dst, mp_srcptr numa, mp_srcptr numb, mp_size_t n, mp_limb_t c) {
54 __uint128_t pr = (__uint128_t)numa[0] + numb[0] + c;
55 mp_limb_t l = (mp_limb_t)pr & 1;
56 for (mp_size_t i = 1; i < n; i++) {
57 __uint128_t cr = (__uint128_t)numa[i] + numb[i] + (mp_limb_t)(pr >> 64);
58 dst[i - 1] = ((mp_limb_t)pr >> 1) | ((mp_limb_t)cr << 63);
59 pr = cr;
60 }
61 dst[n - 1] = ((mp_limb_t)pr >> 1) | ((mp_limb_t)(pr >> 64) << 63);
62 return l;
63}

69mp_limb_t lmmp_shr1sub_nc_(mp_ptr dst, mp_srcptr numa, mp_srcptr numb, mp_size_t n, mp_limb_t c) {
70 mp_limb_t o = numa[0];
71 mp_limb_t u = o - numb[0];
72 mp_limb_t pr = u - c;
73 mp_limb_t d = (u > o) + (pr > u);
74 mp_limb_t l = pr & 1;
75 for (mp_size_t i = 1; i < n; i++) {
76 o = numa[i];
77 u = o - numb[i];
78 mp_limb_t r = u - d;
79 d = (u > o) + (r > u);
80 dst[i - 1] = (pr >> 1) | (r << 63);
81 pr = r;
82 }
83 dst[n - 1] = (pr >> 1) | (d << 63);
84 return l;
85}
