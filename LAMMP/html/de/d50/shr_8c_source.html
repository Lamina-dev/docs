19mp_limb_t lmmp_shr_(mp_ptr dst, mp_srcptr numa, mp_size_t na, mp_size_t shr) {
20 if (shr == 0) {
21 lmmp_copy(dst, numa, na);
22 return 0;
23 } else {
24 mp_limb_t high_limb, low_limb;
25 const mp_size_t rshr = LIMB_BITS - shr;
26 mp_limb_t retval;
27 high_limb = *numa++;
28 retval = (high_limb << rshr);
29 low_limb = high_limb >> shr;
30 while (--na != 0) {
31 high_limb = *numa++;
32 *dst++ = low_limb | (high_limb << rshr);
33 low_limb = high_limb >> shr;
34 }
35 *dst = low_limb;
36 return retval;
37 }
38}
