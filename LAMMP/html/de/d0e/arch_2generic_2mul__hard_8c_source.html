25/* 常量行程循环的完全展开提示: clang 用 #pragma unroll (行程数已知时
26 无视代价模型强制全展开), gcc 用 #pragma GCC unroll 上界 (覆盖最大
27 行程 37); 其余编译器退化为普通循环, 语义不变仅性能差异。 */
28#if defined(__clang__)
29#define LMMP_UNROLL _Pragma("unroll")
30#elif defined(__GNUC__)
31#define LMMP_UNROLL _Pragma("GCC unroll 64")
32#else
33#define LMMP_UNROLL
34#endif

36#define DEF_MUL_HARD(n_) \
37 void lmmp_mul_hard_##n_##_(mp_ptr restrict dst, mp_srcptr restrict numa, mp_srcptr restrict numb) { \
38 mp_limb_t cl = 0; \
39 mp_size_t i, j; \
40 LMMP_UNROLL \
41 for (i = 0; i < n_; i++) { \
42 __uint128_t t = (__uint128_t)numa[i] * numb[0] + cl; \
43 dst[i] = (mp_limb_t)t; \
44 cl = (mp_limb_t)(t >> 64); \
45 } \
46 dst[n_] = cl; \
47 for (j = 1; j < n_; j++) { \
48 cl = 0; \
49 LMMP_UNROLL \
50 for (i = 0; i < n_; i++) { \
51 __uint128_t t = (__uint128_t)numa[i] * numb[j] + dst[i + j] + cl; \
52 dst[i + j] = (mp_limb_t)t; \
53 cl = (mp_limb_t)(t >> 64); \
54 } \
55 dst[n_ + j] = cl; \
56 } \
57 }

59#define DEF_SQR_HARD(n_) \
60 void lmmp_sqr_hard_##n_##_(mp_ptr restrict dst, mp_srcptr restrict numa) { \
61 mp_limb_t cl, x; \
62 mp_size_t i, j; \
63 dst[0] = 0; \
64 x = numa[0]; \
65 cl = 0; \
66 LMMP_UNROLL \
67 for (i = 1; i < n_; i++) { \
68 __uint128_t t = (__uint128_t)numa[i] * x + cl; \
69 dst[i] = (mp_limb_t)t; \
70 cl = (mp_limb_t)(t >> 64); \
71 } \
72 dst[n_] = cl; \
73 /* i: dst[i+1..i+n] += a[i+1..n)*a_i */ \
74 LMMP_UNROLL \
75 for (j = 1; j + 1 < n_; j++) { \
76 x = numa[j]; \
77 cl = 0; \
78 LMMP_UNROLL \
79 for (i = j + 1; i < n_; i++) { \
80 __uint128_t t = (__uint128_t)numa[i] * x + dst[i + j] + cl; \
81 dst[i + j] = (mp_limb_t)t; \
82 cl = (mp_limb_t)(t >> 64); \
83 } \
84 dst[n_ + j] = cl; \
85 } \
86 /* dst[1..2n-2] = 2*dst, 2n-1 = carry(0) */ \
87 cl = 0; \
88 LMMP_UNROLL \
89 for (i = 1; i + 1 < 2 * n_; i++) { \
90 __uint128_t t = (__uint128_t)dst[i] * 2 + cl; \
91 dst[i] = (mp_limb_t)t; \
92 cl = (mp_limb_t)(t >> 64); \
93 } \
94 dst[2 * n_ - 1] = cl; \
95 cl = 0; \
96 LMMP_UNROLL \
97 for (i = 0; i < n_; i++) { \
98 __uint128_t t = (__uint128_t)numa[i] * numa[i] + dst[2 * i] + cl; \
99 dst[2 * i] = (mp_limb_t)t; \
100 cl = (mp_limb_t)(t >> 64); \
101 t = (__uint128_t)dst[2 * i + 1] + cl; \
102 dst[2 * i + 1] = (mp_limb_t)t; \
103 cl = (mp_limb_t)(t >> 64); \
104 } \
105 }
