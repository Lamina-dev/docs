38/**
39 * @brief 初始化哈夫曼乘法树
40 * @param ht 哈夫曼乘法树指针
41 * @param cap 需要插入的元素的最大数量
42 * @warning 0<cap<2^30, ht!=NULL
43 */
44static inline void lmmp_huff_tree_init_(huff_tree* ht, sint cap) {
45 lmmp_param_assert(cap > 0);
46 lmmp_param_assert(cap < (1 << 30));
47 lmmp_param_assert(ht != NULL);
48 ht->root = ALLOC_TYPE(2 * cap - 1, huff_node);
49 ht->cap = cap;
50 ht->size = 0;
51}

58static inline void lmmp_huff_tree_free_(huff_tree* ht) {

66/**
67 * @brief 入队哈夫曼树
68 * @param ht 哈夫曼树指针
69 * @param np 待入队的元素指针
70 * @param nn 元素的 limb 长度
71 * @warning ht!=NULL, ht->size<ht->cap, np!=NULL, nn>0
72 * @note 传入乘法元素必须为非空元素，可以和其他乘法元素指向完全相同
73 */
74static inline void lmmp_huff_tree_push_(huff_tree* ht, mp_srcptr np, mp_size_t nn) {
75 lmmp_param_assert(ht != NULL);
76 lmmp_param_assert(ht->size < ht->cap);
77 lmmp_param_assert(np != NULL);
78 lmmp_param_assert(nn > 0);
79 ht->root[ht->size].np = np;
80 ht->root[ht->size].nn = nn;
81 ht->root[ht->size].left = -1;
82 ht->root[ht->size].right = -1;
83 ht->size++;
84}
