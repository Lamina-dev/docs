45/**
46 * @brief 初始化哈夫曼乘法树
47 * @param ht 哈夫曼乘法树指针
48 * @param cap 需要插入的元素的最大数量
49 * @warning 0<cap<2^30, ht!=NULL
50 */
51static inline void lmmp_huff_tree_init_(huff_tree* ht, sint cap) {
52 lmmp_param_assert(cap > 0);
53 lmmp_param_assert(cap < (1 << 30));
54 lmmp_param_assert(ht != NULL);
55 ht->root = ALLOC_TYPE(2 * cap - 1, huff_node);
56 ht->cap = cap;
57 ht->size = 0;
58}

60/**
61 * @brief 释放哈夫曼树
62 * @param ht 哈夫曼树指针
63 * @warning ht->size==0, ht!=NULL
64 */
65static inline void lmmp_huff_tree_free_(huff_tree* ht) {
66 lmmp_param_assert(ht != NULL);
67 lmmp_free(ht->root);
68 ht->cap = 0;
69 ht->size = 0;
70 ht->root = NULL;
71}

73/**
74 * @brief 入队哈夫曼树
75 * @param ht 哈夫曼树指针
76 * @param np 待入队的元素指针
77 * @param nn 元素的 limb 长度
78 * @warning ht!=NULL, ht->size<ht->cap, np!=NULL, nn>0
79 * @note 传入乘法元素必须为非空元素，可以和其他乘法元素指向完全相同
80 */
81static inline void lmmp_huff_tree_push_(huff_tree* ht, mp_srcptr np, mp_size_t nn) {
82 lmmp_param_assert(ht != NULL);
83 lmmp_param_assert(ht->size < ht->cap);
84 lmmp_param_assert(np != NULL);
85 lmmp_param_assert(nn > 0);
86 ht->root[ht->size].np = np;
87 ht->root[ht->size].nn = nn;
88 ht->root[ht->size].left = -1;
89 ht->root[ht->size].right = -1;
90 ht->size++;
91}

131/**
132 * @brief 计算因子的累乘，并将结果放入dst中
133 * @param dst 结果数组
134 * @param rn 结果数组的长度
135 * @param fac 因子数组（将会被递归覆盖）
136 * @param nfactors 因子数组的长度
137 * @warning 因子必须要单调递增，可以接受重复因子（但不推荐这样做），因子的贡献必须要大于0。
138 * 已假定进入朴素乘法时，因子数组元素都小于等于0xffff，同时假定了，因子数组为小因子大指数形式，
139 * 可以存在大的因子有略大的指数，但整体的趋势应是小因子大指数。
140 * 在组合数以及由阶乘和幂次构成的有理数中，暂未见不满足这些假设的例子。

142 */
