37TEST_CASE("low/addsub", add_sub_deterministic_inplace) {
38 const mp_size_t n = 5;
39 const u64 av[5] = {UINT64_MAX - 1, 0, 0, 1, UINT64_MAX};
40 const u64 bv[5] = {1, 0, UINT64_MAX, UINT64_MAX - 1, 0};
41 mp_ptr a = alloc_limbs(n), b = alloc_limbs(n), d = alloc_limbs(n), r = alloc_limbs(n);
42 for (mp_size_t i = 0; i < n; ++i) { a[i] = av[i]; b[i] = bv[i]; }
43
44 mp_ptr a2 = alloc_limbs(n), b2 = alloc_limbs(n);
45 const u64 add_cy = lmmp_add_n_(r, a, b, n); // 分离缓冲参考：和
46 const u64 sub_bw = lmmp_sub_n_(d, a, b, n); // 分离缓冲参考：差（a >= b）
47 TEST_CHECK_EQ(add_cy, 0u); // 固定输入钉住参考语义：无终位进位/借位
48 TEST_CHECK_EQ(sub_bw, 0u);
49 // eqsep 原地：dst == numa / dst == numb 各两向（返回值与分离缓冲一致）
50 memcpy(a2, a, n * 8);
51 TEST_CHECK_EQ(lmmp_add_n_(a2, a2, b, n), add_cy);
52 TEST_CHECK_EQ(memcmp(a2, r, n * 8), 0);
53 memcpy(b2, b, n * 8);
54 TEST_CHECK_EQ(lmmp_add_n_(b2, a, b2, n), add_cy);
55 TEST_CHECK_EQ(memcmp(b2, r, n * 8), 0);
56 memcpy(a2, a, n * 8);
57 TEST_CHECK_EQ(lmmp_sub_n_(a2, a2, b, n), sub_bw);
58 TEST_CHECK_EQ(memcmp(a2, d, n * 8), 0);
59 memcpy(b2, b, n * 8);
60 TEST_CHECK_EQ(lmmp_sub_n_(b2, a, b2, n), sub_bw);
61 TEST_CHECK_EQ(memcmp(b2, d, n * 8), 0);
62
63 lmmp_free(a); lmmp_free(b); lmmp_free(d); lmmp_free(r);
64 lmmp_free(a2); lmmp_free(b2);
65}

67TEST_CASE("low/shift", shlnot_not_inplace_deterministic) {
68 const mp_size_t n = 6;
69 const u64 vals[6] = {
70 0x0000000000000001ull, 0x8000000000000000ull,
71 0xffffffffffffffffull, 0x123456789abcdef0ull,
72 0x0fedcba987654321ull, 0xaaaaaaaa55555555ull
73 };
74 mp_ptr a = alloc_limbs(n), orig = alloc_limbs(n), r = alloc_limbs(n);
75 for (mp_size_t i = 0; i < n; ++i) a[i] = vals[i];
76 memcpy(orig, a, n * 8);
77
78 lmmp_not_(a, a, n);
79 for (mp_size_t i = 0; i < n; ++i) TEST_CHECK_EQ(a[i], ~orig[i]);
80 lmmp_not_(a, a, n);
81 for (mp_size_t i = 0; i < n; ++i) TEST_CHECK_EQ(a[i], orig[i]);
82
83 const mp_size_t s_values[] = {0, 1, 7, 31, 63};
84 for (size_t s_index = 0; s_index < ARRAY_SIZE(s_values); ++s_index) {
85 mp_size_t s = s_values[s_index];
86 mp_limb_t r1 = lmmp_shlnot_(r, orig, n, s);
87 memcpy(a, orig, n * 8);
88 mp_limb_t r2 = lmmp_shlnot_(a, a, n, s);
89 TEST_CHECK_EQ(r1, r2);
90 TEST_CHECK_EQ(memcmp(a, r, n * 8), 0);
91 }
92 lmmp_free(a); lmmp_free(orig); lmmp_free(r);
93}

95TEST_CASE("low/div", div_1_mod_1_boundary_deterministic) {
96 const mp_size_t ns[] = {1, 2, 3};
97 const u64 fills[] = {0ull, 1ull, ~0ull};
98 const u64 divisors[] = {1ull, 2ull, 0x8000000000000000ull, ~0ull};
99 for (size_t n_index = 0; n_index < ARRAY_SIZE(ns); ++n_index) {
100 mp_size_t n = ns[n_index];
101 for (size_t fill_index = 0; fill_index < ARRAY_SIZE(fills); ++fill_index) {
102 u64 fill = fills[fill_index];
103 mp_ptr a = alloc_limbs(n), q = alloc_limbs(n + 1), t = alloc_limbs(n + 2);
104 for (mp_size_t i = 0; i < n; ++i) a[i] = (i == n - 1) ? fill : (fill >> 1);
105 for (size_t x_index = 0; x_index < ARRAY_SIZE(divisors); ++x_index) {
106 u64 x = divisors[x_index];
107 u64 rem = lmmp_div_1_(q, a, n, x);
108 TEST_CHECK(rem < x);
109 TEST_CHECK_EQ(lmmp_mod_1_(a, n, x), rem);
110 u64 hi = lmmp_mul_1_(t, q, n, x);
111 t[n] = hi;
112 add_small_inplace(t, n + 1, rem);
113 TEST_CHECK(mpn_eq(t, n + 1, a, n)); // q*x + r == a
114 }
115 lmmp_free(a); lmmp_free(q); lmmp_free(t);
116 }
117 }

120TEST_CASE("low/addsub", addshl1_subshl1_boundary_deterministic) {
121 const mp_size_t n = 4;
122 const u64 av[4] = {0, 1, ~(u64)0 - 1, 3};
123 const u64 bv[4] = {(u64)1 << 63, ~(u64)0, 0, 2};
124 mp_ptr a = alloc_limbs(n), b = alloc_limbs(n), d = alloc_limbs(n + 1), t = alloc_limbs(n + 1),
125 b2 = alloc_limbs(n + 1);
126 for (mp_size_t i = 0; i < n; ++i) { a[i] = av[i]; b[i] = bv[i]; }
127
128 u64 c1 = lmmp_add_n_(b2, b, b, n);
129 u64 c2 = lmmp_add_n_(t, a, b2, n);
130 u64 cy = lmmp_addshl1_n_(d, a, b, n);
131 TEST_CHECK_EQ(memcmp(d, t, n * 8), 0);
132 TEST_CHECK_EQ(cy, c1 + c2);
133
134 u64 w1 = lmmp_sub_n_(t, a, b2, n);
135 u64 bw = lmmp_subshl1_n_(d, a, b, n);
136 TEST_CHECK_EQ(memcmp(d, t, n * 8), 0);
137 TEST_CHECK_EQ(bw, w1 + c1);
138
139 lmmp_free(a); lmmp_free(b); lmmp_free(d); lmmp_free(t); lmmp_free(b2);
140}
