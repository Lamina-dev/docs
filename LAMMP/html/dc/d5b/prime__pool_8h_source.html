30/*
31 素性检验的输入池：单次判定成本对输入性质极度敏感（小因子合数
32 试除即拒、素数全基底执行），故以池轮转代替单输入，保证测得
33 目标总体的齐次负载：
34 ODD 随机奇数（合数为主，平均负载）
35 PRIME 素数（全轮完整执行的最坏情形）
36 COPRIME 试除幸存者（无 <=1000 素因子；强 Lucas 对含小因子输入
37 经 Jacobi gcd 提前退出，且 measure 取最小值会被任一
38 快速元素掩盖，须过滤到试除上界之上保证池内齐次）
39*/

47/**
48 * @brief COPRIME 输入池的预筛
49 * @note 与 <=1000 素数之积取 gcd，gcd!=1 等价于存在 <=1000 素因子（池元素
50 * 恒奇，素数 2 无关）
51 */
52static inline bool small_factor_q(mp_srcptr x, mp_size_t n) {
53 static mp_ptr pr = NULL;
54 static mp_size_t prn = 0;
55 if (pr == NULL) {
56 mp_size_t need = lmmp_primefac_size_(1000);
57 pr = alloc_limbs(need);
58 prn = lmmp_primefac_(pr, need, 1000);
59 }
60 mp_ptr g = alloc_limbs(prn);
61 mp_size_t gn = (n >= prn) ? lmmp_gcd_(g, x, n, pr, prn) : lmmp_gcd_(g, pr, prn, x, n);
62 bool has = !(gn == 1 && g[0] == 1);
63 lmmp_free(g);
64 return has;
65}

67/* n limbs 归一化奇数池（元素个数 cnt） */
68static inline mp_ptr pool_n(mp_size_t n, int cnt, mp_limb_t seed, PrimePool kind) {
69 mp_ptr a = alloc_limbs((size_t)n * cnt);
70 for (int i = 0; i < cnt; i++) {
71 mp_ptr cur = a + n * i;
72 lmmp_seed_random_(cur, n, seed + i * 131, 1);
73 cur[n - 1] |= (mp_limb_t)1 << 63;
74 cur[0] |= 1;
75 if (kind == PRIME_POOL_COPRIME) {
76 while (small_factor_q(cur, n)) cur[0] += 2;
77 } else if (kind == PRIME_POOL_PRIME) {
78 while (lmmp_is_prime_n_(cur, n, 4) == 0) cur[0] += 2;
79 }
80 }
81 return a;
82}
