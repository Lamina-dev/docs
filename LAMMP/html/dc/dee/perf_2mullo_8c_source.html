76static void run_mullo(void) {
77 const mp_size_t BC[] = {20, 100};
78 const mp_size_t SZ[] = {50, 500, 2000};
79 for (size_t i = 0; i < ARRAY_SIZE(BC); ++i) {
80 mp_size_t n = BC[i];
81 report_n("mullo_basecase", n, bench_mullo_basecase(n, &bench_default), n * n * 8.0);
82 }
83 for (size_t i = 0; i < ARRAY_SIZE(SZ); ++i) {
84 mp_size_t n = SZ[i];
85 report_n("mullo", n, bench_mullo(n, &bench_default), n * n * 8.0);
86 }
87 for (size_t i = 0; i < ARRAY_SIZE(SZ); ++i) {
88 mp_size_t n = SZ[i];
89 report_n("mullo_dc", n, bench_mullo_dc(n, &bench_default), n * n * 8.0);
90 }
91 for (size_t i = 0; i < ARRAY_SIZE(SZ); ++i) {
92 mp_size_t n = SZ[i];
93 report_n("mullo_fft", n, bench_mullo_fft(n, &bench_default), n * n * 8.0);
94 }
