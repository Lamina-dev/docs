56#define GLOBAL_ENTRIES_NUM (sizeof(lmmp_global_entries) / sizeof(lmmp_global_entries[0]))
57
58void lmmp_global_init(void) {
59 for (size_t i = 0; i < GLOBAL_ENTRIES_NUM; ++i) {
60 if (lmmp_global_entries[i].init != NULL)
61 lmmp_global_entries[i].init();
62 }
63}
64
65void lmmp_global_deinit(void) {
66 for (size_t i = GLOBAL_ENTRIES_NUM; i-- > 0;) {
67 if (lmmp_global_entries[i].deinit != NULL)
68 lmmp_global_entries[i].deinit();
69 }
70}
