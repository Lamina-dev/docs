24mp_limb_t lmmp_gcd_11_(mp_limb_t u, mp_limb_t v) {
25 lmmp_param_assert(u > 0 && v > 0);
26 int count, k;
27 k = lmmp_tailing_zeros_(u | v);
28 u >>= lmmp_tailing_zeros_(u);
29 v >>= lmmp_tailing_zeros_(v);
30 while (u != v) {
31 if (u > v) {
32 u -= v;
33 count = lmmp_tailing_zeros_(u);
34 u >>= count;
35 } else {
36 v -= u;
37 count = lmmp_tailing_zeros_(v);
38 v >>= count;
39 }
40 }
41 return u << k;
42}
