28mp_limb_t lmmp_gcd_11_(mp_limb_t u, mp_limb_t v) {
29 lmmp_param_assert(u > 0 && v > 0);
30 int k = lmmp_tailing_zeros_(u | v); // k = min(tz(u), tz(v))
31 u >>= lmmp_tailing_zeros_(u);
32 v >>= lmmp_tailing_zeros_(v);
33 // u, v 均为奇数；去冗余最低位（隐式最低位表示）
34 u >>= 1;
35 v >>= 1;
36 while (u != v) {
37 mp_limb_t t = u - v;
38 mp_limb_t m = (mp_limb_t)((slong)t >> 63); // u<v -> 1
39 v += m & t; // v = min(u, v)
40 u = (t ^ m) - m; // u = |u - v|
41 u = (u >> 1) >> lmmp_tailing_zeros_(t);
42 }
43 return ((u << 1) + 1) << k;
44}
