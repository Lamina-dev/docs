34static inline int __lmmp_limb_bits_(mp_limb_t x) {
35 int k = 0;
36 while (x) {
37 x >>= 1;
38 k++;
39 }
40 return k;
41}

43static inline int __lmmp_limb_popcnt_(mp_limb_t x) {
44#if defined(__GNUC__) || defined(__clang__)
45 mp_limb_t count;
46#if defined(LMMP_ASM_X64)
47 __asm__ volatile("popcnt %1, %0" : "=r"(count) : "r"(x) : "cc");
48#else
49 count = __builtin_popcountll(x);
50#endif
51 return (int)count;
52#else
53 int k = 0;
54 while (x) {
55 k += x & 1;
56 x >>= 1;
57 }
58 return k;
59#endif
60}

62static inline int __lmmp_leading_zeros_(mp_limb_t x) {
63 if (x == 0) return 64;
64#ifdef __GNUC__
65 return __builtin_clzll(x);
66#else
67 int n = 0;
68 if (x <= 0x00000000FFFFFFFF) { n += 32; x <<= 32; }
69 if (x <= 0x0000FFFFFFFFFFFF) { n += 16; x <<= 16; }
70 if (x <= 0x00FFFFFFFFFFFFFF) { n += 8; x <<= 8; }
71 if (x <= 0x0FFFFFFFFFFFFFFF) { n += 4; x <<= 4; }
72 if (x <= 0x3FFFFFFFFFFFFFFF) { n += 2; x <<= 2; }
73 if (x <= 0x7FFFFFFFFFFFFFFF) { n += 1; x <<= 1; }
74 return n;
75#endif
76}

78static inline int __lmmp_tailing_zeros_(mp_limb_t x) {
79 if (x == 0) return 64;
80#ifdef __GNUC__
81 return __builtin_ctzll(x);
82#else
83 int n = 0;
84 if ((x & 0x00000000FFFFFFFF) == 0) { n += 32; x >>= 32; }
85 if ((x & 0x000000000000FFFF) == 0) { n += 16; x >>= 16; }
86 if ((x & 0x00000000000000FF) == 0) { n += 8; x >>= 8; }
87 if ((x & 0x000000000000000F) == 0) { n += 4; x >>= 4; }
88 if ((x & 0x0000000000000003) == 0) { n += 2; x >>= 2; }
89 if ((x & 0x0000000000000001) == 0) { n += 1; x >>= 1; }
90 return n;
91#endif
92}

94static inline void __lmmp_mullh_(mp_limb_t a, mp_limb_t b, mp_ptr restrict dst) {
95#if (defined(__GNUC__) || defined(__clang__)) && defined(__SIZEOF_INT128__)
96 __uint128_t prod = (__uint128_t)a * b;
97 dst[0] = (mp_limb_t)prod;
98 dst[1] = (mp_limb_t)(prod >> 64);
99#else
100 uint64_t ah = a >> 32, bh = b >> 32;
101 a = (uint32_t)a, b = (uint32_t)b;
102 uint64_t r0 = a * b, r1 = a * bh, r2 = ah * b, r3 = ah * bh;
103 r3 += (r1 >> 32) + (r2 >> 32);
104 r1 = (uint32_t)r1, r2 = (uint32_t)r2;
105 r1 += r2;
106 r1 += (r0 >> 32);
107 dst[1] = r3 + (r1 >> 32);
108 dst[0] = (r1 << 32) | (uint32_t)r0;
109#endif
110}

112static inline mp_limb_t __lmmp_mulh_(mp_limb_t a, mp_limb_t b) {
113#if (defined(__GNUC__) || defined(__clang__)) && defined(__SIZEOF_INT128__)
114 __uint128_t t = (__uint128_t)a * (__uint128_t)b;
115 return (mp_limb_t)(t >> 64);
116#else
117 uint64_t ah = a >> 32, bh = b >> 32;
118 a = (uint32_t)a, b = (uint32_t)b;
119 uint64_t r0 = a * b, r1 = a * bh, r2 = ah * b, r3 = ah * bh;
120 r3 += (r1 >> 32) + (r2 >> 32);
121 r1 = (uint32_t)r1, r2 = (uint32_t)r2;
122 r1 += r2;
123 r1 += (r0 >> 32);
124 return r3 + (r1 >> 32);
125#endif
126}

128static inline void __lmmp_mul_n_(mp_ptr restrict dst, mp_srcptr restrict numa, mp_srcptr restrict numb, mp_size_t n) {
129 if (n < MUL_TOOM22_THRESHOLD)

135 else if (n < MUL_TOOM6H_THRESHOLD)

138 lmmp_mul_toom6h_(dst, numa, n, numb, n);
139 else if (n < MUL_FFT_THRESHOLD)
140 lmmp_mul_toom8h_(dst, numa, n, numb, n);
141 else
142 lmmp_mul_fft_(dst, numa, n, numb, n);
