33// cnt = ctz(x)
34// r = x >> cnt
35// assume x is non-zero
36#define ctz_shr_u64(r, x, cnt) \
37 do { \
38 unsigned long long _x_ = (x); \
39 (cnt) = __builtin_ctzll(_x_); \
40 (r) = _x_ >> (cnt); \
41 } while (0)
42// cnt = clz(x)
43// r = x << cnt
44// assume x is non-zero
45#define clz_shl_u64(r, x, cnt) \
46 do { \
47 unsigned long long _x_ = (x); \
48 (cnt) = __builtin_clzll(_x_); \
49 (r) = (unsigned long long)(_x_ << (cnt)); \
50 } while (0)
51#else
52// cnt = ctz(x)
53// r = x >> cnt
54// assume x is non-zero
55#define ctz_shr_u64(r, x, cnt) \
56 do { \
57 uint64_t _x_ = (x); \
58 int _i_ = 0; \
59 while ((_x_ & 1) == 0) { \
60 _i_++; \
61 _x_ >>= 1; \
62 } \
63 cnt = _i_; \
64 (r) = _x_; \
65 } while (0)
66// cnt = clz(x)
67// r = x << cnt
68// assume x is non-zero
69#define clz_shl_u64(r, x, cnt) \

71 uint64_t _x_ = (x); \
72 int _c_ = 0; \
73 while ((_x_ & 0x8000000000000000ULL) == 0) { \
74 _c_++; \
75 _x_ <<= 1; \
76 } \
77 (cnt) = _c_; \
78 (r) = _x_; \
79 } while (0)

132static inline void _umul64to128_(uint64_t a, uint64_t b, uint64_t *low, uint64_t *high) {
133#if (defined(__GNUC__) || defined(__clang__))
134 __uint128_t prod = (__uint128_t)a * b;
135 *low = (uint64_t)prod;
136 *high = (uint64_t)(prod >> 64);
137#else
138 uint64_t ah = a >> 32, bh = b >> 32;
139 a = (uint32_t)a, b = (uint32_t)b;
140 uint64_t r0 = a * b, r1 = a * bh, r2 = ah * b, r3 = ah * bh;
141 r3 += (r1 >> 32) + (r2 >> 32);
142 r1 = (uint32_t)r1, r2 = (uint32_t)r2;
143 r1 += r2;
144 r1 += (r0 >> 32);
145 *high = r3 + (r1 >> 32);
146 *low = (r1 << 32) | (uint32_t)r0;
147#endif
148}

150static inline uint64_t _umul64to64hi_(uint64_t a, uint64_t b) {
151#if (defined(__GNUC__) || defined(__clang__)) && defined(__SIZEOF_INT128__)
152 __uint128_t t = (__uint128_t)a * (__uint128_t)b;
153 return (uint64_t)(t >> 64);
154#else
155 uint64_t ah = a >> 32, bh = b >> 32;
156 a = (uint32_t)a, b = (uint32_t)b;
157 uint64_t r0 = a * b, r1 = a * bh, r2 = ah * b, r3 = ah * bh;
158 r3 += (r1 >> 32) + (r2 >> 32);
159 r1 = (uint32_t)r1, r2 = (uint32_t)r2;
160 r1 += r2;
161 r1 += (r0 >> 32);
162 return r3 + (r1 >> 32);
163#endif
164}

166static inline void _umul128to256_(uint64_t a_high, uint64_t a_low, uint64_t b_high, uint64_t b_low, uint64_t rr[4]) {
167 /*
168 | res0 | res1 | res2 | res3 |
169 | p0l | p0h | | |
170 | p1l | p1h | |
171 | p2l | p2h | |
172 | | p3l | p3h |
173 */
174#if defined(LMMP_ASM_X64) && defined(__GNUC__) && !defined(__clang__)
175 /* 4×mulx + 运行累加：p1/p2/p3 依次流入 r8/r9 暂存（以 clobber 声明，
176 乱序重命名消除 WAR），每个部分积以 add/adcx/adc 三连加进累加器
177 （进位至多 1，res3 侧经 adc $0 传播）。相对九输出分解式，asm
178 边界同时保活的寄存器由 9 输出 + rdx 压缩为 4 输出 + r8/r9/rdx */
179 uint64_t r0, r1, r2, r3;
180 __asm__(
181 "movq %[al], %%rdx \n\t"
182 "mulxq %[bl], %[r0], %[r1] \n\t" // p0 = al*bl
183 "xorq %[r2], %[r2] \n\t"
184 "xorq %[r3], %[r3] \n\t"
185 "mulxq %[bh], %%r8, %%r9 \n\t" // p1 = al*bh
186 "movq %[ah], %%rdx \n\t"
187 "addq %%r8, %[r1] \n\t" // res1 += p1l
188 "adcxq %%r9, %[r2] \n\t" // res2 += p1h + c
189 "adcq $0, %[r3] \n\t"
190 "mulxq %[bl], %%r8, %%r9 \n\t" // p2 = ah*bl
191 "addq %%r8, %[r1] \n\t"
192 "adcxq %%r9, %[r2] \n\t"
193 "adcq $0, %[r3] \n\t"
194 "mulxq %[bh], %%r8, %%r9 \n\t" // p3 = ah*bh
195 "addq %%r8, %[r2] \n\t" // res2 += p3l
196 "adcxq %%r9, %[r3] \n\t" // res3 += p3h + c
197 : [r0] "=&r"(r0), [r1] "=&r"(r1), [r2] "=&r"(r2), [r3] "=&r"(r3)
198 : [ah] "rm"(a_high), [al] "rm"(a_low), [bh] "rm"(b_high), [bl] "rm"(b_low)
199 : "cc", "rdx", "r8", "r9");
200 rr[0] = r0;
201 rr[1] = r1;
202 rr[2] = r2;
203 rr[3] = r3;
204#elif defined(LMMP_ASM_X64) && defined(__clang__)
205 /* clang-x64 组合式：clang 的 mulhi 识别/进位融合/双移位合并把本
206 形态编成与手写 asm 同构体且无块边界税（asm 块输出强制驻活使
207 栈流量×3、调度冻结，探针 ladder 慢 1.7×）；GCC 无此融合（组合
208 式朴素展开 30+ 条指令），x64-GCC 走上方 asm。位置在 u128
209 typedef 之前，此处直接用 __uint128_t */
210 __uint128_t p00 = (__uint128_t)a_low * b_low;
211 __uint128_t p01 = (__uint128_t)a_low * b_high;
212 __uint128_t p10 = (__uint128_t)a_high * b_low;
213 __uint128_t p11 = (__uint128_t)a_high * b_high;
214 __uint128_t mid = p01 + p10;
215 __uint128_t cm = mid < p01; /* 129 位交叉和进位，属 hi 的第 64 位 */
216 __uint128_t l = (mid << 64) + p00;
217 __uint128_t cl = l < p00; /* 进位属 hi 的第 0 位 */
218 __uint128_t h = p11 + (mid >> 64) + cl + (cm << 64);
219 rr[0] = (uint64_t)l;
220 rr[1] = (uint64_t)(l >> 64);
221 rr[2] = (uint64_t)h;
222 rr[3] = (uint64_t)(h >> 64);
223#elif defined(LMMP_ASM_ARM64) && (defined(__GNUC__) || defined(__clang__))
224 /* 8×(mul|umulh) + adds/adcs 标志链收尾，共 14 条：C 语句间无法保活
225 NZCV 进位，纯 C 形态最少 21 条（进位须 cset 物化）。关键次序：
226 p11h 在 CF=tc（交叉和 129 位顶）后立即 adc xzr 吸收进位（免 cset
227 与额外寄存器），随后 r1/r2/r3 三连以 adds/adcs/adc 单进位链收拢
228 （列和第三项恒 <=1，进位至多 1）。r1/r2 输出寄存器兼作交叉积
229 暂存（写终值前死亡），边界足迹 4 输出 + 4 暂存。 */
230 uint64_t r0, r1, r2, r3, t4, t5, t6, t7;
231 __asm__(
232 "mul %[r0], %[al], %[bl] \n\t" /* p00l -> r0 终值 */
233 "umulh %[r1], %[al], %[bl] \n\t" /* p00h */
234 "mul %[r2], %[al], %[bh] \n\t" /* p01l */
235 "umulh %[r3], %[al], %[bh] \n\t" /* p01h */
236 "mul %[t4], %[ah], %[bl] \n\t" /* p10l */
237 "umulh %[t5], %[ah], %[bl] \n\t" /* p10h */
238 "mul %[t6], %[ah], %[bh] \n\t" /* p11l */
239 "umulh %[t7], %[ah], %[bh] \n\t" /* p11h */
240 "adds %[r2], %[r2], %[t4] \n\t" /* tl = p01l+p10l, CF=tc0 */
241 "adcs %[r3], %[r3], %[t5] \n\t" /* th = p01h+p10h+tc0, CF=tc */
242 "adc %[t7], %[t7], xzr \n\t" /* p11h += tc（立即吃 CF） */
243 "adds %[r1], %[r1], %[r2] \n\t" /* r1 = p00h+tl, CF=c1 */
244 "adcs %[r2], %[t6], %[r3] \n\t" /* r2 = p11l+th+c1, CF=ch */
245 "adc %[r3], %[t7], xzr \n\t" /* r3 = p11h+tc+ch */
246 : [r0] "=&r"(r0), [r1] "=&r"(r1), [r2] "=&r"(r2), [r3] "=&r"(r3),
247 [t4] "=&r"(t4), [t5] "=&r"(t5), [t6] "=&r"(t6), [t7] "=&r"(t7)
248 : [ah] "r"(a_high), [al] "r"(a_low), [bh] "r"(b_high), [bl] "r"(b_low)
249 : "cc");
250 rr[0] = r0;
251 rr[1] = r1;
252 rr[2] = r2;
253 rr[3] = r3;
254#else
255 /* 纯 C 路径：__uint128_t 组合写法在 GCC 下生成代码反而劣于显式
256 limb 进位链（128 位中间值引发寄存器溢出与额外 mov），保留朴素结构。 */
257 uint64_t p1_low, p1_high; // p1 = a_low × b_high
258 uint64_t p2_low, p2_high; // p2 = a_high × b_low
259 _umul64to128_(a_low, b_low, rr, rr + 1);
260 _umul64to128_(a_low, b_high, &p1_low, &p1_high);
261 _umul64to128_(a_high, b_low, &p2_low, &p2_high);
262 _umul64to128_(a_high, b_high, rr + 2, rr + 3);
263 rr[1] += p1_low;
264 uint64_t carry = (rr[1] < p1_low) ? 1 : 0;
265 rr[1] += p2_low;
266 carry += (rr[1] < p2_low) ? 1 : 0;
267
268 rr[2] += carry;
269 carry = (rr[2] < carry) ? 1 : 0;
270 rr[2] += p1_high;
271 carry += (rr[2] < p1_high) ? 1 : 0;
272 rr[2] += p2_high;
273 carry += (rr[2] < p2_high) ? 1 : 0;
274
275 rr[3] += carry;
276#endif
277}

279static inline void _usqr128to256_(uint64_t a_high, uint64_t a_low, uint64_t rr[4]) {
280 /*
281 | res0 | res1 | res2 | res3 |
282 | p0l | p0h | | |
283 | p1l | p1h | |
284 | p1l | p1h | |
285 | | p3l | p3h |
286 */
287#if defined(LMMP_ASM_X64) && defined(__GNUC__) && !defined(__clang__)
288 /* 3×mulx；交叉积 p1 左移一位成为 129 位（v 为溢出位），
289 shld 一条完成高位拼接，adcx/adox 双链吸收 v 与两段进位。
290 res2 的两次加法（+p3l、+c1）各自至多产生一个进位且不会同时
291 产生，分别落在 OF 与 CF，故链尾须以 adox、adcx 各一条收拢，
292 缺一会导致 res2 恰为 2^64-1 再加 c1 时进位丢失（res3 少 1）。 */
293 uint64_t p0l, t_l, t_h, p3l, r1, r2, r3, z;
294 __asm__(
295 "movq %[al], %%rdx \n\t"
296 "mulxq %[al], %[p0l], %[r1] \n\t" // p0 = al*al, r1 = p0h
297 "mulxq %[ah], %[t_l], %[t_h] \n\t" // t = al*ah
298 "movq %[ah], %%rdx \n\t"
299 "mulxq %[ah], %[p3l], %[r3] \n\t" // p3 = ah*ah, r3 = p3h
300 "movq %[t_h], %[r2] \n\t"
301 "shldq $1, %[t_l], %[r2] \n\t" // r2 = t_h<<1 | t_l>>63
302 "shrq $63, %[t_h] \n\t" // t_h = v（129 位溢出，属于 res3 段）
303 "leaq (%[t_l], %[t_l]), %[t_l]\n\t" // t_l = 2*al*ah 低 64 位（lea 不动标志）
304 "xorq %[z], %[z] \n\t" // CF = OF = 0（须在移位类指令之后）
305 "adcxq %[t_l], %[r1] \n\t" // r1 = p0h + t_l, CF = res2 进位
306 "adoxq %[p3l], %[r2] \n\t" // r2 += p3l, OF = res3 进位
307 "adcxq %[z], %[r2] \n\t" // r2 += CF = res2
308 "adoxq %[t_h], %[r3] \n\t" // r3 += v + OF
309 "adcxq %[z], %[r3] \n\t" // r3 += CF = res3（收拢 res2 的 c1 进位）
310 : [p0l] "=&r"(p0l), [t_l] "=&r"(t_l), [t_h] "=&r"(t_h), [p3l] "=&r"(p3l),
311 [r1] "=&r"(r1), [r2] "=&r"(r2), [r3] "=&r"(r3), [z] "=&r"(z)
312 : [ah] "rm"(a_high), [al] "rm"(a_low)
313 : "cc", "rdx");
314 rr[0] = p0l;
315 rr[1] = r1;
316 rr[2] = r2;
317 rr[3] = r3;
318#elif defined(LMMP_ASM_X64) && defined(__clang__)
319 /* clang-x64 组合式（同 _umul128to256_ 分支依据）：交叉积单次
320 乘法后整体加倍，129 位进位经 m >> 127 顶位提取 */
321 __uint128_t p0 = (__uint128_t)a_low * a_low;
322 __uint128_t m = (__uint128_t)a_low * a_high; /* 交叉积，加倍后 129 位 */
323 __uint128_t p1 = (__uint128_t)a_high * a_high;
324 __uint128_t ml = m << 1;
325 __uint128_t mh = m >> 127; /* 加倍进位（0/1），属 hi 的第 64 位 */
326 __uint128_t l = (ml << 64) + p0;
327 __uint128_t cl = l < p0;
328 __uint128_t h = p1 + (ml >> 64) + cl + (mh << 64);
329 rr[0] = (uint64_t)l;
330 rr[1] = (uint64_t)(l >> 64);
331 rr[2] = (uint64_t)h;
332 rr[3] = (uint64_t)(h >> 64);
333#elif defined(LMMP_ASM_ARM64) && (defined(__GNUC__) || defined(__clang__))
334 /* 6×(mul|umulh) + lsr/extr 拆交叉积加倍 + 三连标志链，共 11 条。
335 加倍的 129 位拆法：tc = 交叉 hi >> 63（顶位，属 res3 段），
336 extr 一条取 (hi:lo)>>63 即加倍后的 res2 段，低 limb 以
337 adds 的 lsl #1 移位操作数直接并入 p0h（移出的位已由 extr 收取）。
338 tc 以寄存器进入收尾 adc（CF 链被 r1 的 adds 重置，不可滞留标志）。 */
339 uint64_t r0, r1, r2, r3, t5, t6, t7;
340 __asm__(
341 "mul %[r0], %[al], %[al] \n\t" /* p0l -> r0 终值 */
342 "umulh %[r1], %[al], %[al] \n\t" /* p0h */
343 "mul %[r2], %[al], %[ah] \n\t" /* 交叉 lo */
344 "umulh %[r3], %[al], %[ah] \n\t" /* 交叉 hi */
345 "lsr %[t7], %[r3], 63 \n\t" /* tc = 加倍后第 129 位 */
346 "extr %[r3], %[r3], %[r2], 63 \n\t" /* th2 = (交叉hi:交叉lo)>>63 */
347 "mul %[t5], %[ah], %[ah] \n\t" /* p1l */
348 "umulh %[t6], %[ah], %[ah] \n\t" /* p1h */
349 "adds %[r1], %[r1], %[r2], lsl 1 \n\t" /* r1 = p0h+2*交叉lo, CF=c1 */
350 "adcs %[r2], %[t5], %[r3] \n\t" /* r2 = p1l+th2+c1, CF=ch */
351 "adc %[r3], %[t6], %[t7] \n\t" /* r3 = p1h+tc+ch */
352 : [r0] "=&r"(r0), [r1] "=&r"(r1), [r2] "=&r"(r2), [r3] "=&r"(r3),
353 [t5] "=&r"(t5), [t6] "=&r"(t6), [t7] "=&r"(t7)
354 : [ah] "r"(a_high), [al] "r"(a_low)
355 : "cc");
356 rr[0] = r0;
357 rr[1] = r1;
358 rr[2] = r2;
359 rr[3] = r3;
360#else
361 uint64_t p1_low, p1_high; // p1 = a_low × a_high
362 _umul64to128_(a_low, a_low, rr, rr + 1);
363 _umul64to128_(a_low, a_high, &p1_low, &p1_high);
364 _umul64to128_(a_high, a_high, rr + 2, rr + 3);
365 rr[3] += p1_high >> 63;
366 p1_high = (p1_high << 1) | (p1_low >> 63);
367 p1_low <<= 1;
368 rr[1] += p1_low;
369 uint64_t carry = (rr[1] < p1_low) ? 1 : 0;
370
371 rr[2] += carry;
372 carry = (rr[2] < carry) ? 1 : 0;
373 rr[2] += p1_high;
374 carry += (rr[2] < p1_high) ? 1 : 0;
375
376 rr[3] += carry;
377#endif
378}

380// r = 低 128 位 (a × b)，a、b 为 128 位（高/低 limb 传入）
381static inline __uint128_t _umul128to128_(uint64_t a_high, uint64_t a_low, uint64_t b_high, uint64_t b_low) {
382 // al*bh + ah*bl 的 64 位回绕和恰为交叉贡献的可见部分
383 return (__uint128_t)a_low * b_low + ((__uint128_t)(uint64_t)(a_low * b_high + a_high * b_low) << 64);
384}

386static inline uint64_t _udiv128by64to64_(uint64_t numhi, uint64_t numlo, uint64_t den, uint64_t* r) {
387#if (defined(__GNUC__) || defined(__clang__)) && defined(LMMP_ASM_X64)
388 uint64_t result;
389 __asm__("div %[v]" : "=a"(result), "=d"(*r) : [v] "r"(den), "a"(numlo), "d"(numhi));
390 return result;
391#else
392 const uint64_t b = ((uint64_t)1 << 32);
393
394 uint32_t q1;
395 uint32_t q0;
396
397 uint64_t q;
398
399 int shift;
400
401 uint64_t den10 = den;
402 uint64_t num10 = numlo;
403
404 uint32_t den1;
405 uint32_t den0;
406 uint32_t num1;
407 uint32_t num0;
408
409 uint64_t rem;
410
411 uint64_t qhat;
412 uint64_t rhat;
413
414 uint64_t c1;
415 uint64_t c2;
416
417 if (numhi >= den) {

680// assert d != 0
681static inline _udiv64_t _udiv64_gen_internal_(uint64_t d, int branchfree) {
682 _udiv64_t result;
683 int shift;
684 uint64_t t;
685 clz_shl_u64(t, d, shift);
686 (void)t;
687 uint32_t floor_log_2_d = 63 - shift;
688
689 // Power of 2
690 if ((d & (d - 1)) == 0) {
691 // We need to subtract 1 from the shift value in case of an unsigned
692 // branchfree divider because there is a hardcoded right shift by 1
693 // in its division algorithm. Because of this we also need to add back
694 // 1 in its recovery algorithm.
695 result.magic = 0;
696 result.more = (uint8_t)(floor_log_2_d - (branchfree != 0));
697 } else {
698 uint64_t proposed_m, rem;
699 uint8_t more;
700 // (1 << (64 + floor_log_2_d)) / d
701 proposed_m = _udiv128by64to64_((uint64_t)1 << floor_log_2_d, 0, d, &rem);
702
703 const uint64_t e = d - rem;
704
705 // This power works if e < 2**floor_log_2_d.
706 if (!branchfree && e < ((uint64_t)1 << floor_log_2_d)) {
707 // This power works
708 more = (uint8_t)floor_log_2_d;
709 } else {
710 // We have to use the general 65-bit algorithm. We need to compute
711 // (2**power) / d. However, we already have (2**(power-1))/d and
712 // its remainder. By doubling both, and then correcting the
713 // remainder, we can compute the larger division.
714 // don't care about overflow here - in fact, we expect it
715 proposed_m += proposed_m;
716 const uint64_t twice_rem = rem + rem;
717 if (twice_rem >= d || twice_rem < rem)
718 proposed_m += 1;
719 more = (uint8_t)(floor_log_2_d | _ADD_MARKER);
720 }
721 result.magic = 1 + proposed_m;
722 result.more = more;
723 // result.more's shift should in general be ceil_log_2_d. But if we
724 // used the smaller power, we subtract one from the shift because we're
725 // using the smaller power. If we're using the larger power, we
726 // subtract one from the shift because it's taken care of by the add
727 // indicator. So floor_log_2_d happens to be correct in both cases,
728 // which is why we do it outside of the if statement.
729 }
730 return result;
731}
