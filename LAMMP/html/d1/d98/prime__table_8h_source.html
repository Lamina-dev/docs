58/**
59 * @brief 估计 n 范围内的素数数量
60 * @param n 范围
61 * @note 不会低估素数数量，可能恰好超过 pi(n)，用以估计素数数组需要的空间
62 * @return 素数数量
63 */
64static inline ulong lmmp_prime_size_(ulong n) {
65 if (n < PRIME_SHORT_TABLE_N) {
66 return lmmp_prime_cnt16_(n);
67 } else if (n < 95000) {
68 return (ulong)ceil((double)n / (log(n) - 1.095)) + 1;
69 } else if (n < 355991) {
70 return (ulong)ceil((double)n / (log(n) - 1.0975)) + 1;
71 } else if (n < 1332479531) {
72 // Dusart 2000 估计 - π(x)的上界
73 // lmmp_debug_assert(n >= 355991);
74 double x = (double)n;
75 double lnx = log(x);
76 double lnx2 = lnx * lnx;
77 double r = x / lnx * (1.0 + 1.0 / lnx + 2.51 / lnx2);
78 return (ulong)r;
79 }
80 /*
81 Dusart 2010 估计 - π(x)的严格上界
82 from: Dusart (2010) "Estimates of some functions over primes without R.H."(https://arxiv.org/abs/1002.0442)
83 U(x) = x / [lnx
84 - 1
85 - 1/lnx
86 - 3.35/(lnx)^2
87 - 12.65/(lnx)^3
88 - 71.7/(lnx)^4
89 - 466.1275/(lnx)^5
90 - 3489.8225/(lnx)^6]
91 */
92 // lmmp_debug_assert(n >= 1332479531);
93 double lnx = log(n);
94 double lnx2 = lnx * lnx;
95 double lnx3 = lnx2 * lnx;
96 double lnx4 = lnx3 * lnx;
97 double lnx5 = lnx4 * lnx;
98 double lnx6 = lnx5 * lnx;
99
100 double denom =
101 lnx - 1.0 - 1.0 / lnx - 3.35 / lnx2 - 12.65 / lnx3 - 71.7 / lnx4 - 466.1275 / lnx5 - 3489.8225 / lnx6;
102
103 return (ulong)ceil(n / denom);
104}
