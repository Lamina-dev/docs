24static ushortp lmmp_trialdiv_short_(mp_srcptr restrict np, mp_size_t nn, ushort N, ushort* rn) {
25 if (np == NULL || nn == 0) {
26 *rn = 0;
27 return NULL;
28 }
29 ulong t = 1;
30 ushort idx[20];
31 uint idx_cnt = 0;
32 ushort retn_max = 10;
33 ushortp ret = ALLOC_TYPE(retn_max, ushort);
34 ushort retn = 0;
35 if (!(np[0] & 1)) {
36 ret[retn++] = 2;
37 }
38 for (mp_size_t i = 1;; i++) {
39 ushort p = prime_short_table[i];
40 if (p > N || i >= PRIME_SHORT_TABLE_SIZE - 1) break;
41 t *= p;
42 idx[idx_cnt++] = p;
43 if (t > MAX_T || idx_cnt == 20) {
44 mp_limb_t r = lmmp_mod_1_(np, nn, t);
45 for (uint j = 0; j < idx_cnt; j++) {
46 if (r == 0 || r % idx[j] == 0) {
47 ret[retn++] = idx[j];
48 if (retn == retn_max) {
49 retn_max = retn_max * 12 / 10;
50 ret = REALLOC_TYPE(ret, retn_max, ushort);
51 }
52 }
53 }
54 idx_cnt = 0;
55 t = 1;
56 }
57 }
58 if (idx_cnt > 0) {
59 mp_limb_t r = lmmp_mod_1_(np, nn, t);
60 for (uint j = 0; j < idx_cnt; j++) {
61 if (r == 0 || r % idx[j] == 0) {
62 ret[retn++] = idx[j];
63 if (retn == retn_max) {
64 retn_max += 10;
65 ret = REALLOC_TYPE(ret, retn_max, ushort);
66 }
67 }
68 }
69 }
70 if (retn == 0) {
71 lmmp_free(ret);
72 *rn = 0;
73 return NULL;
74 } else {
75 *rn = retn;
76 return ret;
77 }
78}

80ushortp lmmp_trialdiv_(mp_srcptr restrict np, mp_size_t nn, ushort N, ushort* rn) {
81 ushort primen = lmmp_prime_cnt16_(N);
82 if (primen > 4 * nn) {
83 return lmmp_trialdiv_short_(np, nn, N, rn);
84 } else {
85 /*
86 if [np,nn] is too large, we calculate the product of primes up to N,
87 and divide [np,nn] by the product to get remainder.
88 */
89 TEMP_S_DECL;
90 ulongp restrict pp = SALLOC_TYPE(primen / 4 + 1, ulong);
91 ulong t = 1;
92 mp_size_t pn = 0;
93 for (ushort i = 0; i < primen; i++) {
94 t *= prime_short_table[i];
95 if (t > MAX_T) {
96 pp[pn++] = t;
97 t = 1;
98 }
99 }
100 if (t > 1) {
101 pp[pn++] = t;
102 }
103 mp_ptr restrict prod = SALLOC_TYPE(pn * 2, mp_limb_t);
104 pn = lmmp_elem_mul_ulong_(prod, pp, pn, prod + pn);
105
106 lmmp_div_(NULL, prod, np, nn, prod, pn);
107 while (pn > 0 && prod[pn - 1] == 0) --pn;
108 ushortp restrict ret;
109 if (pn == 0) {
110 ret = ALLOC_TYPE(primen, ushort);
111 for (ushort i = 0; i < primen; i++) {
112 ret[i] = prime_short_table[i];
113 }
114 *rn = primen;
115 } else {
116 ret = lmmp_trialdiv_short_(prod, pn, N, rn);
117 }
118 TEMP_S_FREE;
119 return ret;
120 }
121}
