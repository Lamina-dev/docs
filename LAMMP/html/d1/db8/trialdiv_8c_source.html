32#define MAX_T ((ulong)0x7fffffffffff)

34/* 块数的理论上限：全部奇素数的乘积不到 2^94548（θ(65535) < 65535），而每个
35 * 满块的积都不小于 2^47，所以满块至多 2011 个，再加最后一个乘不满的块，总共
36 * 不会超过 2012 个。容量翻倍时以它封顶，绝不给出超出这个数的空间 */
37#define TRIALDIV_BLK_MAX ((ushort)2012)
38
39/**
40 * 块内字段分三组、各按水位 [0,bn_x) 惰性初始化，纯小 nn 负载不付 chunk
41 * 组预处理、纯大 nn 负载不付 preinv 组预处理：
42 * 打包组 t/p0/pn 建块即得
43 * preinv 组 k/inv 首个 preinv 腿调用覆盖到该块时算
44 * chunk 组 dinv/r2/base/mi 首个 chunk 腿调用覆盖到该块时算
45 */
46typedef struct trialdiv_blk {
47 ulong t; // 块积，也即除数
48 ulong inv; // preinv 组：lmmp_inv_1_(t<<k)
49 ulong dinv; // chunk 组：t*dinv==1 mod B
50 ulong r2; // chunk 组：B^2 mod t
51 ulong base; // chunk 组：B mod t
52 ulong mi; // chunk 组：-dinv mod B
53 ushort p0; // 打包组：块首素数在 prime_short_table 的下标
54 ushort pn; // 打包组：块内素数个数
55 uchar k; // preinv 组：lmmp_leading_zeros_(t)
56} trialdiv_blk;
57
58typedef struct {
59 trialdiv_blk* blks;
60 ushort blkc; // blks 容量
61 ushort bn; // 打包组水位：已闭块数
62 ushort bn_inv; // preinv 组已初始化容量
63 ushort bn_chunk; // chunk 组已初始化容量
64 ushort pi; // 打包游标：下一素数下标
65 ulong t; // 开块部分积（扩容后恒 1：开块总在扩容内闭掉）
66} trialdiv_table;

68static LMMP_THREAD_LOCAL trialdiv_table G = {
69 .blks = NULL, .blkc = 0, .bn = 0, .bn_inv = 0,
70 .bn_chunk = 0, .pi = 1, .t = 1,
71};

73/**
74 * @brief 把开块落盘成 blks[bn] 里的一个新块，随后开块清空
75 * @note 新块的块首下标接在上一块末尾之后，首块从下标 1 开始（素数 2 不打包）。
76 * blks 放不下时容量翻倍，但封顶 TRIALDIV_BLK_MAX；realloc 会原样保留
77 * 旧块内容，不需要重算任何字段
78 */
79static void trialdiv_close(void) {
80 if (G.bn == G.blkc) {
81 G.blkc = G.blkc ? G.blkc * 2 : 32;
82 if (G.blkc > TRIALDIV_BLK_MAX) G.blkc = TRIALDIV_BLK_MAX;
83 G.blks = REALLOC_TYPE(G.blks, G.blkc, trialdiv_blk);
84 }
85 trialdiv_blk* b = &G.blks[G.bn];
86 ushort p0 = G.bn ? (ushort)(G.blks[G.bn - 1].p0 + G.blks[G.bn - 1].pn) : 1;
87 b->t = G.t;
88 b->p0 = p0;
89 b->pn = (ushort)(G.pi - p0);
90 ++G.bn;
91 G.t = 1;
92}

94/**
95 * @brief 打包素数，把下标 [1,cnt) 内的奇素数全部收进闭块
96 * @param cnt <=N 的素数个数，即 lmmp_prime_cnt16_(N) 的返回值（含素数 2）
97 * @note 素数逐个乘进开块，乘积一超过 MAX_T 就闭一块。盖住 cnt 之后，如果
98 * 开块乘积还没满，会继续乘到满块才停。故表中存储的均是完整块，块的
99 * 划分与 N 无关，扩容永远是纯追加，开块也从不跨调用留存。素数表用尽时
100 * 最后一块无法乘满，自动闭合。
101 */
102static void trialdiv_extend_(ushort cnt) {
103 while (G.pi < cnt || G.t > 1) {
104 if (G.pi == PRIME_SHORT_TABLE_SIZE) {
105 trialdiv_close();
106 break;
107 }
108 G.t *= prime_short_table[G.pi++];
109 if (G.t > MAX_T) trialdiv_close();
110 }
111 lmmp_debug_assert(G.t == 1);
112}

114/**
115 * @brief 给 [bn_inv,n) 的块补算 k 和 inv
116 * @param n 要补至的块数，不超过 bn
117 */
118static void trialdiv_init_inv_(ushort n) {
119 for (ushort i = G.bn_inv; i < n; ++i) {
120 trialdiv_blk* b = &G.blks[i];
121 b->k = (uchar)lmmp_leading_zeros_(b->t);
122 b->inv = lmmp_inv_1_(b->t << b->k);
123 }
124 G.bn_inv = n;
125}
126
127/**
128 * @brief 给 [bn_chunk,n) 的块补算折叠腿要用的 dinv、mi、r2、base
129 * @param n 要补到的块数，不超过 bn
130 * @note dinv 是 t 模 2^64 的逆元，mi 就是它的相反数；r2 与 base 共享一次
131 * inv_1_ 未填的块顺手存进 preinv 组字段并推其水位（已填的块直接复用），
132 * 后续 preinv 腿免再算。已经算过的块不会再算
133 */
134static void trialdiv_init_chunk_(ushort n) {
135 for (ushort i = G.bn_chunk; i < n; ++i) {
136 trialdiv_blk* b = &G.blks[i];
137 b->dinv = lmmp_binvert_ulong_(b->t);
138 b->mi = -b->dinv;
139 ulong m = b->t;
140 mp_bitcnt_t k = 0;
141 clz_shl_u64(m, m, k);
142 ulong inv;
143 if (i < G.bn_inv)
144 inv = b->inv;
145 else {
146 b->k = (uchar)k;
147 b->inv = inv = lmmp_inv_1_(m);
148 }
149 b->r2 = _bpow_preinv_(m, k, 2, inv);
150 b->base = _bpow_preinv_(m, k, 1, inv);
151 }
152 G.bn_chunk = n;
153 if (G.bn_inv < n) G.bn_inv = n;
154}

170bool lmmp_trialdiv_(mp_srcptr restrict np, mp_size_t nn, ushort N) {
171 lmmp_param_assert(np != NULL);
172 lmmp_param_assert(nn > 0);
173
174 ushort cnt = lmmp_prime_cnt16_(N);
175 if (cnt <= 1) return false; /* N<3：无奇素数可试除 */
176
177 trialdiv_extend_(cnt);
178
179 /* 用满块前缀：块含 <=N 素数 ⟺ p0 < cnt（p0 升序，首块越界即停） */
180 ushort un = 0;
181 while (un < G.bn && G.blks[un].p0 < cnt) ++un;
182
183 trialdiv_leg_fn leg;
184 if (nn < TRIALDIV_PREINV_THRESHOLD) {
185 if (G.bn_inv < un) trialdiv_init_inv_(un);
186 leg = trialdiv_leg_preinv_;
187 } else if (nn < TRIALDIV_K8_THRESHOLD) {
188 if (G.bn_chunk < un) trialdiv_init_chunk_(un);
189 leg = trialdiv_leg_4_;
190 } else {
191 if (G.bn_chunk < un) trialdiv_init_chunk_(un);
192 leg = trialdiv_leg_8_;
193 }
194
195 for (ushort i = 0; i < un; ++i) {
196 const trialdiv_blk* b = &G.blks[i];
197 mp_limb_t r = leg(np, nn, b);
198 if (r == 0) return true;
199 for (ushort j = b->p0, e = (ushort)(b->p0 + b->pn); j < e; ++j)
200 if (r % prime_short_table[j] == 0) return true;
201 }
202 return false;
203}

214}
