20mp_size_t lmmp_gcd_basecase_(mp_ptr dst, mp_srcptr up, mp_size_t un, mp_srcptr vp, mp_size_t vn) {
21 lmmp_param_assert(un > 0 && vn > 0);
22 if (un < vn) {
23 LMMP_SWAP(up, vp, mp_srcptr);
24 LMMP_SWAP(un, vn, mp_size_t);
25 } else if (un == vn) {
26 int cmp = lmmp_cmp_(up, vp, un);
27 if (cmp < 0) {
28 LMMP_SWAP(up, vp, mp_srcptr);
29 } else if (cmp == 0) {
30 lmmp_copy(dst, up, un);
31 return un;
32 }
33 }
34 TEMP_DECL;
35#define an un
36#define bn vn
37 mp_ptr a = TALLOC_TYPE(an, mp_limb_t);
38 mp_ptr b = TALLOC_TYPE(bn, mp_limb_t);
39 lmmp_copy(a, up, an);
40 lmmp_copy(b, vp, bn);
41 while (bn > 0 || (bn == 1 && b[0] == 0)) {
42 // dst = a % b;
43 lmmp_div_(NULL, dst, a, an, b, bn);
44 lmmp_copy(a, b, bn);
45 an = bn;
46 while (dst[bn - 1] == 0 && bn > 0) {
47 --bn;
48 }
49 lmmp_copy(b, dst, bn);
50 }
51 lmmp_copy(dst, a, an);
52 TEMP_FREE;
53 return an;
54#undef an
55#undef bn
56}
