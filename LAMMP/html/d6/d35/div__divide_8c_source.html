21/**
22 * @brief 分治除法
23 * @param dstq 结果商指针（[dstq,n]=[numa,2*n] div [numb,n]）
24 * @param numa 被除数指针（[numa,n]=[numa,2*n] mod [numb,n]）
25 * @param numb 除数指针
26 * @param n 除数长度
27 * @param inv21 除数高128位的逆
28 * @param tp 临时工作区指针（需要 n 个 limb 的空间）
29 * @warning n>=6, MSB(numb)=1, inv21=(2^192-1)/[numb+n-2,2]-2^64, sep(dstq,numa,numb,tp)
30 * @return 除法结果最高位商
31 */
32static mp_limb_t lmmp_div_divide_n_(
33 mp_ptr restrict dstq,
34 mp_ptr restrict numa,
35 mp_srcptr restrict numb,
36 mp_size_t n,
37 mp_limb_t inv21,
38 mp_ptr restrict tp
39) {
40 lmmp_param_assert(n >= 6);
41 lmmp_param_assert(numb[n - 1] >= LIMB_B_2);
42 mp_size_t lo = n >> 1, hi = n - lo;
43 mp_limb_t cy, qh, ql;
44
45 if (hi < DIV_DIVIDE_THRESHOLD) {
46 qh = lmmp_div_basecase_(dstq + lo, numa + 2 * lo, 2 * hi, numb + lo, hi, inv21);
47 } else {
48 qh = lmmp_div_divide_n_(dstq + lo, numa + 2 * lo, numb + lo, hi, inv21, tp);
49 }
50 lmmp_mul_(tp, dstq + lo, hi, numb, lo);
51
52 cy = lmmp_sub_n_(numa + lo, numa + lo, tp, n);
53 if (qh)
54 cy += lmmp_sub_n_(numa + n, numa + n, numb, lo);
55
56 while (cy) {
57 qh -= lmmp_sub_1_(dstq + lo, dstq + lo, hi, 1);
58 cy -= lmmp_add_n_(numa + lo, numa + lo, numb, n);
59 }
60
61 if (lo < DIV_DIVIDE_THRESHOLD) {
62 ql = lmmp_div_basecase_(dstq, numa + hi, 2 * lo, numb + hi, lo, inv21);
63 } else {
64 ql = lmmp_div_divide_n_(dstq, numa + hi, numb + hi, lo, inv21, tp);
65 }
66 lmmp_mul_(tp, numb, hi, dstq, lo);
67
68 cy = lmmp_sub_n_(numa, numa, tp, n);
69 if (ql)
70 cy += lmmp_sub_n_(numa + lo, numa + lo, numb, hi);
71
72 while (cy) {
73 lmmp_sub_1_(dstq, dstq, lo, 1);
74 cy -= lmmp_add_n_(numa, numa, numb, n);
75 }
76 return qh;
77}

79mp_limb_t lmmp_div_divide_(
80 mp_ptr restrict dstq,
81 mp_ptr restrict numa,
82 mp_size_t na,
83 mp_srcptr restrict numb,
84 mp_size_t nb,
85 mp_limb_t inv21
86) {
87 lmmp_param_assert(na >= 2 * nb);
88 lmmp_param_assert(nb >= 6);
89 lmmp_param_assert(numb[nb - 1] >= LIMB_B_2);
90 mp_size_t nq = na - nb;
91
92 dstq += nq;
93 numa += nq;
94
95 do {
96 nq -= nb;
97 } while (nq >= nb);
98
99 dstq -= nq;
100 numa -= nq;
101
102 /* Perform the typically smaller block first. */
103 mp_limb_t qh = lmmp_div_s_(dstq, numa, nb + nq, numb, nb);
104
105 TEMP_DECL;
106 mp_ptr restrict tp = TALLOC_TYPE(nb, mp_limb_t);
107 nq = na - nb - nq;
108
109 do {
110 dstq -= nb;
111 numa -= nb;
112 lmmp_div_divide_n_(dstq, numa, numb, nb, inv21, tp);
113 nq -= nb;
114 } while (nq > 0);
115
116 TEMP_FREE;
117 return qh;
118}
