28/*
29 模逆元验证（无自建算法 oracle）：
30 1. 标量档：64 位 % 欧几里得判互素 + inv*u mod m == 1（mullh 全积
31 + mod_1 取余）+ inv < m + m=1/不可逆写死语义
32 2. 多 limb：互素性经库 gcd_ 判定；逆元经库乘除重建 inv*u mod m == 1
33 3. dst 与 u 别名（eqsep 契约）稳定性
34*/

36static u64 gcd_u64(u64 a, u64 b) {
37 while (b) {
38 u64 t = a % b;
39 a = b;
40 b = t;
41 }
42 return a;
43}

45TEST_CASE("numth/modinv", modinv_ulong) {
46 // 穷举小值（u*inv mod m 经 mod_1 判模）
47 for (u64 m = 1; m <= 200; ++m) {
48 for (u64 u = 1; u <= 200; ++u) {
49 ulong inv = lmmp_modinv_ulong_(u, m);
50 if (m == 1) {
51 TEST_CHECK_MSG(inv == 0, "m=1 inverse is 0");
52 continue;
53 }
54 u64 r = u % m;
55 ulong g = (r == 0) ? m : gcd_u64(r, m);
56 if (g != 1) {
57 TEST_CHECK_MSG(inv == 0, "non-invertible");
58 } else {
59 mp_limb_t prod[2];
60 lmmp_mullh_(u, inv, prod);
61 TEST_CHECK_MSG(lmmp_mod_1_(prod, 2, m) == 1 && inv < m, "inverse");
62 }
63 }
64 }
65 // 随机 64 位
66 u64 seed = 0x51ce1b7ea4df7c21ull;
67 for (int i = 0; i < 20000; ++i) {
68 u64 m = xorshift64(&seed);
69 u64 u = xorshift64(&seed);
70 if (m == 0) m = 1;
71 if (u == 0) u = 1;
72 ulong inv = lmmp_modinv_ulong_(u, m);
73 u64 r = u % m;
74 ulong g = (r == 0) ? m : gcd_u64(r, m);
75 if (m == 1) {
76 TEST_CHECK_MSG(inv == 0, "m=1");
77 } else if (g != 1) {
78 TEST_CHECK_MSG(inv == 0, "non-invertible random");
79 } else {
80 mp_limb_t prod[2];
81 lmmp_mullh_(u, inv, prod);
82 TEST_CHECK_MSG(lmmp_mod_1_(prod, 2, m) == 1 && inv < m, "inverse random");
83 }
84 }
85}

87TEST_CASE("numth/modinv", modinv_big) {
88 u64 seed = 0x2b44ae9d4d6c8f13ull;
89 const mp_size_t mn_values[] = {2, 3, 5, 9, 17, 33, 60};
90 for (size_t mn_index = 0; mn_index < ARRAY_SIZE(mn_values); ++mn_index) {
91 mp_size_t mn = mn_values[mn_index];
92 for (int iter = 0; iter < 7; ++iter) {
93 mp_ptr m = alloc_limbs(mn);
94 random_limbs(m, mn, &seed);
95 m[0] |= 1;
96 mp_ptr u;
97 mp_size_t un;
98 if (iter == 6) {
99 // un < mn：u 直接取短随机数（覆盖 modinv 短分支）
100 un = mn / 2 > 0 ? mn / 2 : 1;
101 u = alloc_limbs(un);
102 random_limbs(u, un, &seed);
103 u[0] |= 1;
104 } else if (iter % 2 == 0) {
105 // u = m + 1（与 m 互素）
106 un = mn;
107 u = alloc_limbs(mn + 1);
108 lmmp_copy(u, m, mn);
109 lmmp_zero(u + mn, 1);
110 add_small_inplace(u, mn + 1, 1);
111 un = norm_len(u, mn + 1);
112 } else {
113 un = mn;
114 u = alloc_limbs(un);
115 random_limbs(u, un, &seed);
116 u[0] |= 1;
117 }
118 mp_ptr dst = alloc_limbs(mn);
119 mp_size_t rn = lmmp_modinv_(dst, u, un, m, mn);
120 // 互素性与逆元验证（库交叉）：inv*u mod m == 1
121 mp_ptr gr = alloc_limbs(un > mn ? un : mn);
122 mp_size_t gn = lmmp_gcd_(gr, u, un, m, mn);
123 if (gn == 1 && gr[0] == 1) {
124 TEST_CHECK_MSG(rn > 0, "inverse exists");
125 mp_ptr prod = alloc_limbs(mn + un + 1), q = alloc_limbs(mn + un + 1),
126 r = alloc_limbs(mn);
127 if (rn >= un) lmmp_mul_(prod, dst, rn, u, un);
128 else lmmp_mul_(prod, u, un, dst, rn);
129 lmmp_div_(q, r, prod, rn + un, m, mn);
130 TEST_CHECK_MSG(norm_len(r, mn) == 1 && r[0] == 1, "inv*u mod m == 1");
131 TEST_CHECK_MSG(mpn_cmp(dst, rn, m, mn) < 0, "inverse in range");
132 lmmp_free(prod); lmmp_free(q); lmmp_free(r);
133 } else {
134 TEST_CHECK_MSG(rn == 0, "non-invertible reported");
135 }
136 lmmp_free(gr);
137 // dst 与 u 别名（eqsep 契约）
138 mp_ptr u2 = alloc_limbs(un > mn ? un : mn);
139 lmmp_copy(u2, u, un);
140 mp_size_t rn2 = lmmp_modinv_(u2, u2, un, m, mn);
141 TEST_CHECK_MSG(rn2 == rn && (rn == 0 || mpn_cmp(u2, rn2, dst, rn) == 0),
142 "alias stable");
143 lmmp_free(m); lmmp_free(u); lmmp_free(dst); lmmp_free(u2);
144 }
145 }
