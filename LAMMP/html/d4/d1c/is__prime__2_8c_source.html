79static inline void mul_128(u128 a, u128 b, u128 *hi, u128 *lo) {
80#if defined(__clang__) && !defined(LMMP_ASM_ARM64)
81 ulong a0 = _u128low(a), a1 = _u128high(a);
82 ulong b0 = _u128low(b), b1 = _u128high(b);
83 u128 p00 = (u128)a0 * b0;
84 u128 p01 = (u128)a0 * b1;
85 u128 p10 = (u128)a1 * b0;
86 u128 p11 = (u128)a1 * b1;
87 u128 mid = p01 + p10;
88 u128 cm = mid < p01; /* 129 位交叉和进位，属 hi 的第 64 位 */
89 u128 l = (mid << 64) + p00;
90 u128 cl = l < p00; /* 进位属 hi 的第 0 位 */
91 *lo = l;
92 *hi = p11 + (mid >> 64) + cl + (cm << 64);
93#else
94 ulong u[4];
95 _umul128to256_(_u128high(a), _u128low(a), _u128high(b), _u128low(b), u);
96 *lo = ((u128)u[1] << 64) | u[0];
97 *hi = ((u128)u[3] << 64) | u[2];
98#endif
99}

101/* a^2 -> (hi, lo)：交叉积单次乘法后整体加倍（3 乘法平方） */
102static inline void sqr_128(u128 a, u128 *hi, u128 *lo) {
103#if defined(__clang__) && !defined(LMMP_ASM_ARM64)
104 ulong a0 = _u128low(a), a1 = _u128high(a);
105 u128 p0 = (u128)a0 * a0;
106 u128 m = (u128)a0 * a1; /* 交叉积，加倍后 129 位 */
107 u128 p1 = (u128)a1 * a1;
108 u128 ml = m << 1;
109 u128 mh = m >> 127; /* 加倍进位（0/1），属 hi 的第 64 位 */
110 u128 l = (ml << 64) + p0;
111 u128 cl = l < p0;
112 *lo = l;
113 *hi = p1 + (ml >> 64) + cl + (mh << 64);
114#else
115 ulong u[4];
116 _usqr128to256_(_u128high(a), _u128low(a), u);
117 *lo = ((u128)u[1] << 64) | u[0];
118 *hi = ((u128)u[3] << 64) | u[2];
119#endif
120}

122/* REDC：r = (T + q*n)/B^2，T = thi*B^2 + tlo < n*B^2。n 接近 B^2 时和
123 可达 129 位，回绕进位下按模 2^128 回绕减 n 恰为真值且已 < n */
124static inline u128 mont2_redc(u128 thi, u128 tlo, u128 n, u128 ninv) {
125#if defined(LMMP_ASM_ARM64)
126 u128 q = tlo * ninv; /* u128 截断乘即 mod B^2 */
127 u128 qh, ql;
128 mul_128(q, n, &qh, &ql);
129 ulong t0 = _u128low(tlo), t1 = _u128high(tlo);
130 ulong p0 = _u128low(ql), p1 = _u128high(ql);
131 ulong k0 = (t0 + p0) < t0; /* lo 低 limb 进位（lo 值本身无消费者） */
132 ulong l1 = t1 + p1;
133 ulong k1 = l1 < t1;
134 l1 += k0;
135 k1 |= l1 < k0;
136 ulong h0 = _u128low(thi) + _u128low(qh);
137 ulong j0 = h0 < _u128low(qh);
138 h0 += k1;
139 j0 |= h0 < k1;
140 ulong h1 = _u128high(thi) + _u128high(qh);
141 ulong j1 = h1 < _u128high(qh);
142 h1 += j0;
143 j1 |= h1 < j0;
144 ulong n0 = _u128low(n), n1 = _u128high(n);
145 if (j1 || (h1 > n1) || (h1 == n1 && h0 >= n0)) {
146 ulong bw = h0 < n0; /* 借位跨 limb 传播 */
147 h0 -= n0;
148 h1 -= n1 + bw; /* j1 分支下回绕即真值 */
149 }
150 return ((u128)h1 << 64) | h0;
151#else
152 u128 q = tlo * ninv; /* u128 截断乘即 mod B^2 */
153 u128 qh, ql, lo, hi, c, c1, c2;
154 mul_128(q, n, &qh, &ql);
155 lo = tlo + ql;
156 c = lo < tlo;
157 hi = thi + qh;
158 c1 = hi < thi;
159 hi += c;
160 c2 = c && hi == 0;
161 if (c1 | c2)
162 hi -= n;
163 else if (hi >= n)
164 hi -= n;
165 return hi;
166#endif
167}

181/* REDC(a*b)，b 为单 limb：a 为蒙域剩余时结果出域（a_plain*b），
182 专用于转域 */
183static inline u128 mont2_redc_mul_1(u128 a, ulong b, u128 n, u128 ninv) {
184 /* T = a*b < 2^192：h<<64 回绕丢弃的高位入 thi，低位和的进位并入 thi */
185 u128 lo = (u128)_u128low(a) * b;
186 u128 h = (u128)_u128high(a) * b;
187 u128 tlo = lo + (h << 64);
188 return mont2_redc((h >> 64) + (tlo < lo), tlo, n, ninv);
189}

202static inline u128 mont2_dbl(u128 a, u128 n) {
203 u128 s = a + a;
204 if (s < a)
205 s -= n;
206 else if (s >= n)
207 s -= n;

237/* (b|A) 二进制算法：A 奇，b < A（移位/比较/减法，几次迭代） */
238static inline int jacobi_small(ulong b, ulong A) {
239 int sign = 1;
240 if (b == 0) return 0;
241 for (;;) {
242 int tz = lmmp_tailing_zeros_(b);
243 b >>= tz;
244 if ((tz & 1) && ((A & 7) == 3 || (A & 7) == 5)) sign = -sign;
245 if (b == 1) return sign;
246 if (b < A) {
247 ulong s = b;
248 b = A;
249 A = s;
250 if ((b & 3) == 3 && (A & 3) == 3) sign = -sign;
251 }
252 b -= A;
253 if (b == 0) return 0;
254 }
255}

257/*
258 (±Dabs|n)：Dabs 奇（uint 范围），n 128 位奇。n 以双 limb 模乘链折叠
259 （r = (r*(B mod A) + limb) mod A，全程 64 位），再经二次互反律
260 (A|n) = (n|A)*(-1)^(((A-1)/2)((n-1)/2)) 归结到 jacobi_small；
261 D 为负时乘 (-1|n) 因子。gcd(|D|,n)>1 返回 0
262*/
263static inline int jacobi_D(uint Dabs, int Dneg, u128 n) {
264 ulong A = Dabs;
265 ulong Bm = (ulong)-1 % A + 1;
266 ulong b;
267 int sign = 1;
268 if (Bm >= A) Bm -= A; /* Bm = 2^64 mod A */
269 b = _u128high(n) % A;
270 b = (b * Bm + _u128low(n) % A) % A;
271 if ((A & 3) == 3 && (_u128low(n) & 3) == 3) sign = -sign;
272 sign *= jacobi_small(b, A);
273 if (sign == 0) return 0;
274 if (Dneg && (_u128low(n) & 3) == 3) sign = -sign;
275 return sign;
276}

278/* 二次探测第二阶段：y = b^d 已得，检查 y ∈ {1, n-1} 或 y^(2^j) = n-1
279 (0 < j < t)。one/m_1 均为蒙域剩余 */
280static inline int mont2_sprp_stage2(u128 y, slong t, u128 one, u128 m_1, u128 n, u128 ninv) {
281 if (y == one || y == m_1) return 1;
282 for (t--; t > 0; t--) {
283 y = mont2_sqr(y, n, ninv);
284 if (y == m_1) return 1;
285 }
286 return 0;
287}

289/*
290 基底 2 专用幂 y = 2^d（蒙域）：指数最高 k 位（k <= 6）对应的 2^t
291 （t < 2^6，普通域一次移位）经一次 REDC 跳启动，其后逐位平方，
292 逢 1 蒙域加倍。ebits = d 的有效位数
293*/
294static u128 mont2_powmod_2_(u128 d, int ebits, u128 R2, u128 n, u128 ninv) {
295 int k = ebits > 6 ? 6 : ebits;
296 int bit;
297 u128 v = mont2_to_mont1((ulong)1 << (ulong)(d >> (ebits - k)), R2, n, ninv);
298 for (bit = ebits - k - 1; bit >= 0; bit--) {
299 v = mont2_sqr(v, n, ninv);
300 if ((d >> bit) & 1) v = mont2_dbl(v, n);
301 }
302 return v;
303}

305/*
306 模 n in (B, phi13] 的非蒙域 sloppy 算术：minv = floor(B^3/n)，3-limb
307 中间积 x 的近似商
308
309 q' = mulhi_sloppy(floor(x/B), minv) 满足 q - 5 <= q' <= q
310
311 （q = floor(x/n)：截断倒数欠 4 + mulhi 丢交叉低位欠 1；上侧
312 X*minv <= x*B^2/n 保证），故 0 <= x - q'*n < 6n 为非规范剩余：
313 - 双 limb 截断算术对它精确（真值非负且 6n < 2^128，回绕减丢的
314 借位恰好无影响）；q'*n 的高半整体丢弃同理；
315 - bit=1 的小基底乘完全免约简：y*b < 6n*41 = 246n < 2^128 装下，
316 下一平方 (246n)^2 < 2^182 < B^3 仍在 3-limb 窗内
317 （_usqr128to192_ 契约 a < 2^96：246n < 2^91 满足）；
318 - 基底 2 逢 1 双倍：2y < 12n < 2^128 无回绕。
319 链尾经 reduce2_precise 规范化后 stage2 在普通域比较 1 / n-1。
320*/

322/* minv = floor(B^3/n)：2^192 除以 n 的商（lmmp_div_2_ 自带归一化，
323 numb 出参收余数弃用） */
324static void ll_small_preinv(u128 *minv, u128 n) {
325 ulong q[3], nn[4] = {0, 0, 0, 1}, d[2];
326
327 _u128store(d, n);
328 lmmp_div_2_(q, nn, 4, d);
329 *minv = ((u128)q[1] << 64) | q[0];
330}

333static inline void ll_small_sqrmod_sloppy(
334 ulong *r1, ulong *r0, ulong a1, ulong a0,
335 ulong n1, ulong n0, ulong minv1, ulong minv0

343 _sub_ddmmss(*r1, *r0, s[1], s[0], _u128high(t), _u128low(t));

346/* 平方 + 归约到规范剩余 [0,n)（stage2 等值比较用）：sloppy 后条件减
347 收尾，剩余 < 6n 至多 5 次 */
348static inline void ll_small_sqrmod(
349 ulong *r1, ulong *r0, ulong a1, ulong a0,

351) {
352 ll_small_sqrmod_sloppy(r1, r0, a1, a0, n1, n0, minv1, minv0);
353 while (*r1 > n1 || (*r1 == n1 && *r0 >= n0))
354 _sub_ddmmss(*r1, *r0, *r1, *r0, n1, n0);
355}

357/* 2-limb 值规范化到 [0,n)：q0 = hi(x1*minv1) 是 floor(x/n) 的下偏
358 近似（q0 <= q 保证减法非负；欠至多 2 使条件减至多 2 次），
359 q0*n < 246n < 2^128 截断积即真值 */
360static inline void ll_small_reduce2_precise(ulong *r1, ulong *r0, ulong x1, ulong x0, ulong n1, ulong n0, ulong minv1) {
361 ulong t[2];
362
363 _umullo128x64_(n1, n0, _umul64to64hi_(x1, minv1), t);
364 _sub_ddmmss(x1, x0, x1, x0, t[1], t[0]);
365 while (x1 > n1 || (x1 == n1 && x0 >= n0))
366 _sub_ddmmss(x1, x0, x1, x0, n1, n0);
367 *r0 = x0;
368 *r1 = x1;
369}

405/*
406 三基底交错幂 y1 = b1^d, y2 = b2^d, y3 = b3^d（非规范）：指数顶部
407 3 bit 在普通域免归约计算 b^(1 b1 b2)（b <= 41，b^7 < 2^38 单
408 limb），其后逐位 sloppy 平方，逢 1 小基底免约简乘，链尾规范化。
409 ebits < 3（n-1 = 2^s 时 d = 1）退化为直接以 b 起步
410*/
411static void ll_small_powmod_3_(
412 ulong y1[2], ulong y2[2], ulong y3[2],
413 ulong b1, ulong b2, ulong b3, u128 d, int ebits,
414 ulong n1, ulong n0, ulong minv1, ulong minv0
415) {
416 ulong xr1 = 0, xr0, yr1 = 0, yr0, zr1 = 0, zr0;
417 int bit;
418
419 if (ebits >= 3) {
420 ulong bit1 = (ulong)(d >> (ebits - 2)) & 1;
421 ulong bit2 = (ulong)(d >> (ebits - 3)) & 1;
422
423 xr0 = b1 * b1;
424 yr0 = b2 * b2;
425 zr0 = b3 * b3;
426 if (bit1) {
427 xr0 *= b1;
428 yr0 *= b2;
429 zr0 *= b3;
430 }
431 xr0 = xr0 * xr0;
432 yr0 = yr0 * yr0;
433 zr0 = zr0 * zr0;
434 if (bit2) {
435 xr0 *= b1;
436 yr0 *= b2;
437 zr0 *= b3;
438 }
439 bit = ebits - 4;
440 } else {
441 xr0 = b1;
442 yr0 = b2;
443 zr0 = b3;
444 bit = ebits - 2;
445 }
446
447 for (; bit >= 0; bit--) {
448 ll_small_sqrmod_sloppy(&xr1, &xr0, xr1, xr0, n1, n0, minv1, minv0);
449 ll_small_sqrmod_sloppy(&yr1, &yr0, yr1, yr0, n1, n0, minv1, minv0);
450 ll_small_sqrmod_sloppy(&zr1, &zr0, zr1, zr0, n1, n0, minv1, minv0);
451 if ((d >> bit) & 1) {
452 ll_small_mul1_(&xr1, &xr0, xr1, xr0, b1);
453 ll_small_mul1_(&yr1, &yr0, yr1, yr0, b2);
454 ll_small_mul1_(&zr1, &zr0, zr1, zr0, b3);
455 }
456 }
457 ll_small_reduce2_precise(&y1[1], &y1[0], xr1, xr0, n1, n0, minv1);
458 ll_small_reduce2_precise(&y2[1], &y2[0], yr1, yr0, n1, n0, minv1);
459 ll_small_reduce2_precise(&y3[1], &y3[0], zr1, zr0, n1, n0, minv1);
460}

532static inline int is_square_2(u128 n) {
533 ulong hi = _u128high(n), lo = _u128low(n);
534 ulong src[2], rr[2], r;

548/*
549 ============ 强 Lucas-Selfridge 测试（V-only 阶梯） ============
550
551 Selfridge 方法 A：P = 1，Q = (1-D)/4，D 取 5, -7, 9, -11, ... 中
552 首个 (D|n) = -1 者；A* 脆弱区绕行——首候选 D=5 命中时改参数化
553 (P,Q)=(1,1/5)（(5,5) 的 V/P^m 归一化像），使全部输入脱离 Q≡±1
554 脆弱区（见下方 Qm 构造处注释）。V 阶梯递推（蒙域）：
555 V_{2k} = V_k^2 - 2Q^k
556 V_{2k+1} = V_k*V_{k+1} - Q^k
557 V_{2k+2} = V_{k+1}^2 - 2Q^{k+1}
558 Q 链同步阶梯：(Q^k, Q^{k+1}) --bit0--> ((Q^k)^2, (Q^k)^2*Q)，
559 --bit1--> (Q^k*Q^{k+1}, (Q^{k+1})^2)。
560 判据（与标准 U/V 强 Lucas 测试精确一致）：
561 U_d = 0，或 V_{d*2^r} = 0（0 <= r < s），d*2^s = n+1。
562 U_d = 0 经 P=1 恒等式 D*U_d = 2V_{d+1} - V_d 转为 2V_{d+1} = V_d 检出
563 （阶梯末态 Vk1 = V_{d+1}；D 搜索保证 gcd(D,n)=1，等价精确成立，含平方
564 因子的 n 亦然；旧判据 V_d^2 = 4Q^d 仅在 n 无平方因子时等价，对
565 p^2*q 型 n 会多接受）。判据命中后再叠加 BFW 两项强化检查（见尾注），
566 素数接受集不变、合数严格多拒。通过返回 1；判定合数（含 gcd(|D|,n)>1）返回 0
567*/
568static int mont2_lucas_strong_(u128 n, u128 ninv, u128 R2) {
569 u128 one, Qm, Qk, Qk1, Vk, Vk1, d, u, w, r;
570 slong s, i;
571 int j, Dneg;
572 uint Dabs, Qabs;
573
574 /* 寻找 D：非完全平方的 n 其 Jacobi 特征非平凡，序列中必存在
575 (D|n) = -1（全部为 +1 当且仅当 n 为完全平方），搜索必然终止，
576 不设上界；素数的最小 |D| 经验上 < 2*log2(n)^2 量级。完全平方数
577 是唯一不终止情形，经 17 处的延迟检测排除（平方检测走库内
578 sqrt_2_ 快速路径，放在少数次未命中之后，不进常见路径）——固定
579 小上界（如旧版 201）不可行：least |D| 随素数规模按 ~2^-k 几何
580 衰减，区间 [SWbound, 2^128) 内 least |D| > 201 的素数约 2^32 个，
581 上界会把它们误判为合数 */
582 Dabs = 5;
583 Dneg = 0;
584 for (;;) {
585 j = jacobi_D(Dabs, Dneg, n);
586 if (j == 0) return 0;
587 if (j == -1) break;
588 Dabs += 2;
589 Dneg ^= 1;
590 if (Dabs == 17 && is_square_2(n)) return 0;
591 }
592
593 /* Q = (1-D)/4，可为负；d*2^s = n+1（n 奇，n+1 偶） */
594 Qabs = Dneg ? (Dabs + 1) / 4 : (Dabs - 1) / 4;
595 d = factor_2adic(n + 1, &s);
596
597 one = mont2_to_mont1(1, R2, n, ninv);
598 if (Dabs == 5 && !Dneg) {
599 /* A* 脆弱区绕行（归一化形）：首候选 D=5 命中 (5|n)=-1 时
600 Selfridge 参数 (P,Q)=(1,-1) 落 BFW 实证 Q≡±1 脆弱区（伪素数
601 过代表，且 Q-Euler 检查 Q^{(n+1)/2}=±Q 恒真空转），改用等价
602 参数化 (P',Q')=(1,1/5)：V'_m = V_m/P^m 把 A* 的 (5,5) 等距
603 映射到 P=1 形（D' = 1-4Q' = 1/5，(D'|n) = (5|n) = -1 仍
604 合法），阶梯/stage2/末检全部公式零改动。q = 5^{-1} mod n
605 闭式：k*(n mod 5) = 4 (mod 5) 使 k*n+1 被 5 整除、真商 < n，
606 商经 mod 2^128 乘 5 的 2^128 幂逆恢复（2^64 ≡ 1 (mod 5)） */
607 static const uchar kt[5] = {0, 4, 2, 3, 1};
608 static const u128 inv5 =
609 (((u128)0xcccccccccccccccc) << 64) | 0xcccccccccccccccd; /* 5^{-1} mod 2^128 */
610 ulong r5 = (_u128low(n) % 5 + _u128high(n) % 5) % 5;
611 u128 q = (((u128)kt[r5]) * n + 1) * inv5;
612 Qm = mont2_to_mont2(q, R2, n, ninv);
613 } else {
614 Qm = mont2_to_mont1(Qabs, R2, n, ninv);
615 if (!Dneg) Qm = mont2_neg(Qm, n);
616 }
617
618 /* 阶梯初态 k=1：(V_1, V_2, Q^1, Q^2) = (1, 1-2Q, Q, Q^2) */
619 Vk = one;
620 Vk1 = mont2_sub(one, mont2_dbl(Qm, n), n);
621 Qk = Qm;
622 Qk1 = mont2_sqr(Qm, n, ninv);
623
624 for (i = bitlen_2(d) - 2; i >= 0; i--) {
625 if ((d >> i) & 1) {
626 u = mont2_mul_2(Vk, Vk1, n, ninv); /* V_{2k+1} 主项 */
627 w = mont2_sqr(Vk1, n, ninv); /* V_{2k+2} 主项 */
628 r = mont2_dbl(Qk1, n);
629 Vk = mont2_sub(u, Qk, n);
630 Vk1 = mont2_sub(w, r, n);
631 Qk = mont2_mul_2(Qk, Qk1, n, ninv); /* Q^{2k+1} */
632 Qk1 = mont2_sqr(Qk1, n, ninv); /* Q^{2k+2} */
633 } else {
634 u = mont2_sqr(Vk, n, ninv); /* V_{2k} 主项 */
635 w = mont2_mul_2(Vk, Vk1, n, ninv); /* V_{2k+1} 主项 */
636 r = mont2_dbl(Qk, n);
637 Vk = mont2_sub(u, r, n);
638 Vk1 = mont2_sub(w, Qk, n);
639 w = mont2_sqr(Qk, n, ninv); /* Q^{2k} */
640 Qk1 = mont2_mul_2(w, Qm, n, ninv); /* Q^{2k+1} */
641 Qk = w;
642 }
643 }
644
645 /* 判据：V_d = 0（r=0），或 U_d = 0，或 V_{d*2^r} = 0（0 < r < s）。
646 U_d = 0 经 D*U_d = 2V_{d+1} - V_d 检出（末态 Vk1 = V_{d+1}）。
647 BFW 强化（Baillie-Fiori-Wagstaff 2021）：命中不早退，链尾统一
648 追加 vpsp 与 Q-Euler 两检查——素数经 Frobenius（α^{n+1}=αβ=Q）
649 与 Euler 判据必然全过（vpsp@10^15 仅 5 个反例且与 spsp(2)/slpsp
650 全不相交，只拦假想的 BPSW 反例），正常完成路径零额外 REDC，
651 早退路径平均多 ~1-2 次链尾 sqr */
652 int hit = (Vk == 0) | (mont2_dbl(Vk1, n) == Vk);
653 u = mont2_sqr(Vk, n, ninv);
654 for (s--; s > 0; s--) {
655 r = mont2_dbl(Qk, n);
656 Vk = mont2_sub(u, r, n); /* V_{2d} = V_d^2 - 2Q^d（复用 u = V_d^2） */
657 Qk = mont2_sqr(Qk, n, ninv);
658 u = mont2_sqr(Vk, n, ninv);
659 if (Vk == 0) hit = 1;
660 }
661 if (!hit) return 0;
662 /* 末态 u = V_{(n+1)/2}^2，Qk = Q^{(n+1)/2}（s=1 时阶梯末态同型）。
663 vpsp：V_{n+1} = u - 2Qk ≡ 2Q；Q-Euler：Qk = Q*(Q|n) ∈ {Qm,-Qm} */
664 if (mont2_sub(u, mont2_dbl(Qk, n), n) != mont2_dbl(Qm, n)) return 0;
665 if (Qk != Qm && Qk != mont2_neg(Qm, n)) return 0;
666 return 1;
667}
