27/*
28扩展欧几里得（标量核与 n limb x 1 limb 层）。
29
30gcdext_11_ 采用带余除法欧几里得 + 双系数幅度递推。设归约序
31r0 = v, r1 = u mod v, r_{k+1} = r_{k-1} mod r_k，各余数的 u/v 系数为
32(s_k, t_k)：r_k = s_k*u + t_k*v。系数符号严格交替（连分数收敛子性质），
33故幅度递推为纯加法：
34
35 |s_{k+1}| = q_k*|s_k| + |s_{k-1}|, |t_{k+1}| = q_k*|t_k| + |t_{k-1}|
36
37无溢出论证（u64 乘加安全）：
38 - g = gcd(u,v) < v（v|u 已提前返回），g|v 故 v >= 2g；
39 - 终态极小系数界 |s| <= v/(2g) < 2^63、|t| <= u/(2g) < 2^63；
40 - 幅度沿递推单调不减，任意中间 q_k*|s_k| <= |s_{k+1}| <= 终态界，
41 首商 q0 = u/v <= u/(2g) 同界——乘积与和均不越 u64。
42
43gcdext_1_ 将 [up,un] 归约至 r = u mod v 后调标量核：s 对 r 的系数即对 u
44的系数（s*u = s*r + s*q*v，模 v 意义一致，且极小界不变）；t = (g - s*u)/v
45经一次 mul_1 与一次除法精确导出（整除性由恒等式保证）。
46*/

48ulong lmmp_gcdext_11_(ulong u, ulong v, slong* s, slong* t) {
49 lmmp_param_assert(u > 0 && v > 0);
50 lmmp_param_assert(s != NULL);
51
52 int sw = 0;
53 if (u < v) {
54 LMMP_SWAP(u, v, ulong);
55 sw = 1;
56 }
57 // u >= v
58
59 ulong r1 = u % v;
60 if (r1 == 0) {
61 // v | u：g = v，系数在较小者
62 *s = sw ? 1 : 0;
63 if (t != NULL)
64 *t = sw ? 0 : 1;
65 return v;
66 }
67
68 ulong r0 = v;
69 ulong su0 = 0, su1 = 1; // u 系数幅度（符号交替，su1 当前为正）
70 ulong tv0 = 1, tv1 = u / v; // v 系数幅度（与 su 相位相反）
71 int phase = 1; // su1 的符号 [+1|-1]
72 for (;;) {
73 ulong q = r0 / r1;
74 ulong r2 = r0 % r1;
75 if (r2 == 0)
76 break; // g = r1，系数即 (su1, tv1) 当前行
77 ulong su2 = q * su1 + su0;
78 ulong tv2 = q * tv1 + tv0;
79 r0 = r1;
80 r1 = r2;
81 su0 = su1;
82 su1 = su2;
83 tv0 = tv1;
84 tv1 = tv2;
85 phase = -phase;
86 }
87 slong gu = phase > 0 ? (slong)su1 : -(slong)su1;
88 slong gv = phase > 0 ? -(slong)tv1 : (slong)tv1;
89 *s = sw ? gv : gu;
90 if (t != NULL)
91 *t = sw ? gu : gv;
92 return r1;
93}

95mp_size_t lmmp_gcdext_1_(mp_ptr dstg, mp_ptr dsts, mp_ssize_t* sn, mp_ptr dstt, mp_ssize_t* tn,
96 mp_srcptr up, mp_size_t un, ulong v) {
97 lmmp_param_assert(un > 0 && v > 0);
98 lmmp_param_assert(up != NULL && dstg != NULL && dsts != NULL && sn != NULL);
99
100 ulong r = lmmp_mod_1_(up, un, v);
101 if (r == 0) {
102 dstg[0] = v;
103 dsts[0] = 0;
104 *sn = 0;
105 if (dstt != NULL) {
106 dstt[0] = 1;
107 *tn = 1;
108 }
109 return 1;
110 }
111
112 slong sg;
113 ulong g = lmmp_gcdext_11_(r, v, &sg, NULL);
114 lmmp_debug_assert(sg != 0); // r>0 且 v∤r → g<v → σ≠0
115
116 if (dstt == NULL) {
117 dstg[0] = g;
118 dsts[0] = sg < 0 ? (ulong)-sg : (ulong)sg;
119 *sn = sg < 0 ? -1 : 1; /* 带符号长度（sg!=0 已断言） */
120 return 1;
121 }
122
123 /* t = (g - sg*u)/v：X = |sg|*u；N = g - sg*u（符号-幅度，X >= u >= g
124 保证 sg>0 时差非负方向确定）；|t| = N/v。对 up 的读取（X）先于全部
125 输出写入（允许 dstg/dsts/dstt 与 up 别名的契约次序） */
126 TEMP_B_DECL;
127 mp_ptr x = BALLOC_TYPE(un + 2, mp_limb_t);
128 mp_ptr nb = BALLOC_TYPE(un + 2, mp_limb_t);
129 mp_ptr tq = BALLOC_TYPE(un + 2, mp_limb_t);
130 ulong sm = sg < 0 ? (ulong)-sg : (ulong)sg;
131 mp_limb_t c = lmmp_mul_1_(x, up, un, sm);
132 mp_size_t xn = un;
133 if (c != 0)
134 x[xn++] = c;
135 mp_size_t nn;
136 int tneg = 0;
137 if (sg > 0) {
138 // N = g - X = -(X - g)，X >= g（|sg|>=1 且 g|u）
139 lmmp_sub_(nb, x, xn, &g, 1);
140 nn = xn;
141 while (nn > 0 && nb[nn - 1] == 0)
142 --nn;
143 tneg = nn != 0;
144 } else {
145 // N = g + X > 0
146 mp_limb_t cy = lmmp_add_(nb, x, xn, &g, 1);
147 nn = xn;
148 if (cy != 0)
149 nb[nn++] = cy;
150 }
151 // |t| = N/v（恒等式保证整除；N==0 即 t==0 时跳过除法）
152 mp_size_t tqn = 0;
153 if (nn != 0) {
154 mp_limb_t vv[1] = {v};
155 lmmp_div_(tq, NULL, nb, nn, vv, 1);
156 tqn = nn;
157 while (tqn > 0 && tq[tqn - 1] == 0)
158 --tqn;
159 }
160
161 dstg[0] = g;
162 dsts[0] = sm;
163 *sn = sg < 0 ? -1 : 1;
164 lmmp_copy(dstt, tq, tqn);
165 if (tqn == 0)
166 dstt[0] = 0;
167 *tn = tneg ? -(mp_ssize_t)tqn : (mp_ssize_t)tqn;
168 TEMP_B_FREE;
169 return 1;
170}
