26void lmmp_divexact_1_(mp_ptr dst, mp_srcptr np, mp_size_t nn, mp_limb_t d, mp_limb_t dinv) {
27 lmmp_param_assert(d % 2 == 1);
28 lmmp_param_assert(dinv * d == 1);
29 lmmp_param_assert(nn > 0);
30 lmmp_debug_assert(dst != NULL && np != NULL);
31 mp_limb_t c = 0;
32 mp_limb_t l, s, lo, hi, q;
33 mp_size_t i;
34 for (i = 0; i < nn - 1; i++) {
35 s = np[i];
36 l = s - c;
37 c = l > s;
38 q = l * dinv;
39 dst[i] = q;
40 _umul64to128_(q, d, &lo, &hi);
41 c += hi;
42 }
43 s = np[i];
44 l = s - c;
45 c = l > s;
46 q = l * dinv;
47 dst[i] = q;
48 lmmp_debug_assert(c == 0);
49}

51void lmmp_divexact_2_(mp_ptr dst, mp_srcptr np, mp_size_t nn, mp_srcptr restrict dp, mp_srcptr restrict dinv) {
52 lmmp_param_assert(dp[0] % 2 == 1);
53 lmmp_param_assert(nn > 1);
54 lmmp_debug_assert(dst != NULL && np != NULL);
55 mp_limb_t c[2] = {0, 0};
56 mp_limb_t l[2], s[2], q[2], t[4];
57 mp_size_t i;
58 mp_limb_t d0 = dp[0], d1 = dp[1];
59 mp_limb_t ddinv0 = dinv[0], ddinv1 = dinv[1];
60
61 if (nn % 2 == 0) {
62 for (i = 0; i < nn - 2; i += 2) {
63 s[0] = np[i];
64 s[1] = np[i + 1];
65 _u128sub(l, s, c);
66 c[0] = _u128cmp(s, l) ? 1 : 0;
67 c[1] = 0;
68 _umul128to128_(l[1], l[0], ddinv1, ddinv0, q);
69 dst[i] = q[0];
70 dst[i + 1] = q[1];
71 _umul128to256_(q[1], q[0], d1, d0, t);
72 _u128add(c, c, t + 2);
73 }
74 s[0] = np[i];
75 s[1] = np[i + 1];
76 _u128sub(l, s, c);
77 c[0] = _u128cmp(s, l);
78 _umul128to128_(l[1], l[0], ddinv1, ddinv0, q);
79 dst[i] = q[0];
80 // 商最高 limb 为 0 时 c[0] 可能为 1；函数结果仍然正确。
81 } else {
82 i = 0;
83 if (nn >= 5) {
84 for (; i < nn - 4; i += 2) {
85 s[0] = np[i];
86 s[1] = np[i + 1];
87 _u128sub(l, s, c);
88 c[0] = _u128cmp(s, l);
89 c[1] = 0;
90 _umul128to128_(l[1], l[0], ddinv1, ddinv0, q);
91 dst[i] = q[0];
92 dst[i + 1] = q[1];
93 _umul128to256_(q[1], q[0], d1, d0, t);
94 _u128add(c, c, t + 2);
95 }
96 }
97 s[0] = np[i];
98 s[1] = np[i + 1];
99 _u128sub(l, s, c);
100 c[0] = _u128cmp(s, l);
101 _umul128to128_(l[1], l[0], ddinv1, ddinv0, q);
102 dst[i] = q[0];
103 dst[i + 1] = q[1];
104 // 商最高 limb 为 0 时 c[0] 可能为 1；函数结果仍然正确。
105 }

164void lmmp_divexact_unbalanced_(
165 mp_ptr dst,
166 mp_srcptr np,
167 mp_size_t nn,
168 mp_srcptr restrict dp,
169 mp_size_t dn,
170 mp_ptr restrict dinv
171) {
172 lmmp_param_assert(dst != NULL && np != NULL && dp != NULL);
173 lmmp_param_assert(dn > 0);
174 lmmp_param_assert(nn >= dn);
175 lmmp_param_assert((dp[0] & 1) != 0);
176
177 TEMP_DECL;
178 fft_gr_cache mersenne_ctx;
179 fft_cache fft_ctx;
180 int mersenne_flag = 0, fft_flag = 0;
181 mp_ptr restrict tp = TALLOC_TYPE(4 * dn, mp_limb_t);
182 if (dinv == NULL) {
183 dinv = TALLOC_TYPE(dn, mp_limb_t);
184 lmmp_binvert_n_dc_(dinv, dp, dn, tp);
185 }
186
187#define c (tp) // [tp, dn]
188#define l (tp + dn) // [tp + dn, dn]
189#define scratch (tp + 2 * dn) // [tp + 2 * dn, 2*dn]
190
191 mp_size_t i = 0, qn = nn - dn + 1;
192 mp_limb_t ca;
193 lmmp_zero(c, dn);
194 if (qn > dn) {
195 for (; i < qn - dn; i += dn) {
196 lmmp_sub_n_(l, np + i, c, dn);
197 ca = lmmp_cmp_(l, np + i, dn) == 1 ? 1 : 0;
198
199 // lmmp_mullo_n_(dst + i, l, dinv, dn, scratch);
200 if (dn < MULLO_DC_THRESHOLD)
201 lmmp_mullo_dc_(dst + i, l, dinv, scratch, dn);
202 else {
203 if (fft_flag == 0) {
204 mp_size_t hn = lmmp_fft_next_size_((dn * 2 + 1) >> 1);
205 lmmp_mul_fft_cache_init_(scratch, hn, l, dn, dinv, dn, &fft_ctx);
206 fft_flag = 1;
207 } else {
208 lmmp_mul_fft_cache_(scratch, l, &fft_ctx);
209 }
210 lmmp_copy(dst + i, scratch, dn);
211 }
212
213 // lmmp_mulhi_n_(scratch + dn, l, dst + i, dp, dn, scratch);
214 if (dn < MULHI_MERSENNE_THRESHOLD) {
215 lmmp_mul_n_(scratch, dst + i, dp, dn);
216 } else {
217 mp_limb_t cy;
218 mp_size_t m = lmmp_fft_next_size_((dn * 2 + 1) >> 1);
219 if (mersenne_flag == 0) {
220 lmmp_mul_mersenne_cache_init_(scratch, m, dst + i, dn, dp, dn, &mersenne_ctx);
221 mersenne_flag = 1;
222 } else {
223 lmmp_mul_mersenne_cache_(scratch, dst + i, &mersenne_ctx);
224 }
225 cy = lmmp_sub_(scratch, scratch, m, l, dn);
226 if (cy != 0)
227 lmmp_dec(scratch);
228 if (m == dn) {
229 lmmp_copy(scratch + dn, scratch, dn);
230 } else {
231 mp_size_t fn = m - dn;
232 mp_size_t sn = dn - fn;
233 //lmmp_copy(scratch + dn, scratch + dn, fn);
234 lmmp_copy(scratch + dn + fn, scratch, sn);
235 }
236 }
237
238 if (ca) {
239 lmmp_add_1_(c, scratch + dn, dn, 1);
240 } else {
241 lmmp_copy(c, scratch + dn, dn);
242 }
243 }
244 }
245 lmmp_sub_n_(l, np + i, c, dn);
246 lmmp_mullo_n_(dst + i, l, dinv, qn - i, scratch);
247
248 if (mersenne_flag == 1)
249 lmmp_fft_gr_cache_free_(&mersenne_ctx);
250 if (fft_flag == 1)
251 lmmp_fft_cache_free_(&fft_ctx);
252
253 TEMP_FREE;
254#undef c
255#undef l
256#undef scratch
257}

259void lmmp_divexact_basecase_(mp_ptr dst, mp_ptr np, mp_size_t nn, mp_srcptr restrict dp, mp_size_t dn, mp_limb_t dinv) {
260 lmmp_param_assert(dst != NULL && np != NULL && dp != NULL);
261 lmmp_param_assert(dn > 0);
262 lmmp_param_assert(nn >= dn);
263 lmmp_param_assert((dp[0] & 1) != 0);
264 lmmp_param_assert((dp[0] * dinv) == 1);
265 mp_size_t i;
266 mp_limb_t q;
267 mp_limb_t hi;
268 mp_size_t qn = nn - dn + 1;
269 for (i = 0; i < qn; i++) {
270 q = dinv * np[i];
271 hi = lmmp_submul_1_(np + i, dp, dn, q);
272 lmmp_debug_assert(np[i] == 0);
273#if LAMMP_DEBUG_ASSERT_CHECK == 1
274 if (hi && i + dn < nn) {
275 lmmp_dec_1(np + i + dn, hi);
276 } else {
277 lmmp_debug_assert(hi == 0);
278 }
279#else
280 if (hi) {
281 lmmp_dec_1(np + i + dn, hi);
282 }
283#endif
284 dst[i] = q;
285 }

288void lmmp_divexact_divide_(mp_ptr restrict dst, mp_srcptr restrict np, mp_size_t nn, mp_srcptr restrict dp, mp_size_t dn) {
289 lmmp_param_assert(dst != NULL && np != NULL && dp != NULL);
290 lmmp_param_assert(dn > 0);
291 lmmp_param_assert(nn >= dn);
292 lmmp_param_assert((dp[0] & 1) != 0);
293
294 mp_size_t qn = nn - dn + 1;
295 lmmp_param_assert(qn <= dn);
296 mp_size_t rn = qn - qn / 2;
297 qn = qn / 2;
298
299 TEMP_DECL;
300 mp_ptr restrict dinv = TALLOC_TYPE(rn, mp_limb_t);
301 mp_ptr restrict tp = TALLOC_TYPE(3 * rn, mp_limb_t);
302 lmmp_binvert_n_dc_(dinv, dp, rn, tp);
303 lmmp_mullo_n_(dst, np, dinv, rn, tp);
304 if (qn == 0) {
305 TEMP_FREE;
306 return;
307 }
308
309 lmmp_debug_assert(3 * rn >= qn + rn);
310 lmmp_mul_(tp, dp, qn + rn, dst, rn);
311
312 // 即使存在错位，高位的np也一定可以将其约减去，因此无需进行任何处理
313 lmmp_sub_n_(tp, np + rn, tp + rn, qn);
314 lmmp_mullo_n_(dst + rn, tp, dinv, qn, tp + rn);
315 TEMP_FREE;
316}

318void lmmp_divexact_(mp_ptr dst, mp_srcptr np, mp_size_t nn, mp_srcptr restrict dp, mp_size_t dn) {
319 lmmp_param_assert(dst != NULL && np != NULL && dp != NULL);
320 lmmp_param_assert(dn > 0);
321 lmmp_param_assert(nn >= dn);
322 lmmp_param_assert((dp[0] & 1) != 0);
323
324 if (dn == 1) {
325 mp_limb_t dinv = lmmp_binvert_ulong_(dp[0]);
326 lmmp_divexact_1_(dst, np, nn, dp[0], dinv);
327 } else if (dn == 2) {
328 mp_limb_t dinv[2];
329 lmmp_binvert_2_(dinv, dp);
330 lmmp_divexact_2_(dst, np, nn, dp, dinv);
331 } else {
332 mp_size_t qn = nn - dn + 1;
333 if (nn < DIVEXACT_NN_THRESHOLD && dn < DIVEXACT_BASECASE_THRESHOLD) {
334 mp_limb_t dinv = lmmp_binvert_ulong_(dp[0]);
335 TEMP_DECL;
336 mp_ptr restrict tp = TALLOC_TYPE(nn, mp_limb_t);
337 lmmp_copy(tp, np, nn);
338 lmmp_divexact_basecase_(dst, tp, nn, dp, dn, dinv);
339 TEMP_FREE;
340 } else if (qn < dn) {
341 TEMP_DECL;
342 mp_ptr restrict tp;
343 if (dst == np) {
344 tp = TALLOC_TYPE(nn, mp_limb_t);
345 lmmp_copy(tp, np, nn);
346 np = tp;
347 }
348 lmmp_divexact_divide_(dst, np, nn, dp, dn);
349 TEMP_FREE;
350 } else {
351 lmmp_divexact_unbalanced_(dst, np, nn, dp, dn, NULL);
352 }
353 }
