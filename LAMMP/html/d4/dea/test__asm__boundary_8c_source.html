31TEST_CASE("asm/boundary", add_nc_sub_nc_all_ones) {
32 const mp_size_t n = 8;
33 mp_ptr a = alloc_limbs(n);
34 mp_ptr b = alloc_limbs(n);
35 mp_ptr d = alloc_limbs(n);
36 for (mp_size_t i = 0; i < n; ++i) {
37 a[i] = UINT64_MAX;
38 b[i] = 0;
39 }
40 b[0] = 1;
41
42 mp_limb_t cy = lmmp_add_nc_(d, a, b, n, 1);
43 TEST_CHECK_EQ(cy, 1u);
44 for (mp_size_t i = 0; i < n; ++i) TEST_CHECK_EQ(d[i], (i == 0) ? 1u : 0u);
45
46 // a - a - borrow = all ones, borrow 1
47 for (mp_size_t i = 0; i < n; ++i) a[i] = UINT64_MAX;
48 mp_limb_t bw = lmmp_sub_nc_(d, a, a, n, 1);
49 TEST_CHECK_EQ(bw, 1u);
50 for (mp_size_t i = 0; i < n; ++i) TEST_CHECK_EQ(d[i], UINT64_MAX);
51
52 lmmp_free(a); lmmp_free(b); lmmp_free(d);
53}

55/*
56 全 F 极值乘法写死常数：(B^n-1)*(B-1) = (B-1)(B^n-1)，limbs =
57 [1, B-1, ..., B-1, B-2]；addmul/submul 用组合路径复核。

59TEST_CASE("asm/boundary", mul_1_addmul_submul_extremes) {
60 const mp_size_t n = 6;
61 mp_ptr a = alloc_limbs(n), d = alloc_limbs(n + 1), t = alloc_limbs(n + 2),
62 acc = alloc_limbs(n + 1), acc2 = alloc_limbs(n + 1);
63 for (mp_size_t i = 0; i < n; ++i) a[i] = UINT64_MAX;
64 u64 x = UINT64_MAX;
65
66 mp_limb_t cy = lmmp_mul_1_(d, a, n, x);
67 TEST_CHECK_EQ(d[0], 1ull);
68 for (mp_size_t i = 1; i < n; ++i) TEST_CHECK_EQ(d[i], UINT64_MAX);
69 TEST_CHECK_EQ(cy, UINT64_MAX - 1);
70
71 // addmul：acc(F..F) + prod == prod<<.. 组合路径
72 for (mp_size_t i = 0; i < n; ++i) acc[i] = UINT64_MAX;
73 acc[n] = 0;
74 u64 c1 = lmmp_add_n_(t, acc, d, n) + cy; // 进位=加进位+乘进位
75 memcpy(acc2, acc, n * 8);
76 acc2[n] = 0;
77 u64 c2 = lmmp_addmul_1_(acc2, a, n, x);
78 TEST_CHECK_EQ(memcmp(acc2, t, n * 8), 0);
79 TEST_CHECK_EQ(c2, c1);
80
81 // submul 组合路径：borrow = w_sub + hi
82 u64 w = lmmp_sub_n_(t, acc, d, n);
83 memcpy(acc2, acc, n * 8);
84 acc2[n] = 0;
85 u64 c3 = lmmp_submul_1_(acc2, a, n, x);
86 TEST_CHECK_EQ(memcmp(acc2, t, n * 8), 0);
87 TEST_CHECK_EQ(c3, w + cy);
88
89 lmmp_free(a); lmmp_free(d); lmmp_free(t); lmmp_free(acc); lmmp_free(acc2);
90}

92TEST_CASE("asm/boundary", shl_shr_inplace) {
93 u64 seed = 0x1234567890abcdeful;
94 const mp_size_t n = 16;
95 mp_ptr a = alloc_limbs(n), orig = alloc_limbs(n), ref = alloc_limbs(n);
96 for (mp_size_t i = 0; i < n; ++i) a[i] = xorshift64(&seed);
97 memcpy(orig, a, n * 8);
98
99 // 原地调用与分离缓冲调用同结果（库内交叉；每次迭代重置源）
100 const size_t s_values[] = {1, 7, 31, 63};
101 for (size_t s_index = 0; s_index < ARRAY_SIZE(s_values); ++s_index) {
102 size_t s = s_values[s_index];
103 memcpy(a, orig, n * 8);
104 mp_limb_t re = lmmp_shl_(ref, orig, n, s);
105 mp_limb_t r1 = lmmp_shl_(a, a, n, s);
106 TEST_CHECK_EQ(r1, re);
107 TEST_CHECK_EQ(memcmp(a, ref, n * 8), 0);
108
109 memcpy(a, orig, n * 8);
110 re = lmmp_shr_(ref, orig, n, s);
111 r1 = lmmp_shr_(a, a, n, s);
112 TEST_CHECK_EQ(r1, re);
113 TEST_CHECK_EQ(memcmp(a, ref, n * 8), 0);
114 }
115
116 lmmp_free(a); lmmp_free(orig); lmmp_free(ref);
117}

119// 错位重叠族公共骨架：buf 内 [G, G+n) 为 numa，dst=numa+w（w=100 表示分离块
120// sep）。每次调用前重填随机源并快照，调用后校验 [dst,dst+n) 外哨兵不变。
121// 期望值由分离缓冲调用（契约内路径）交叉给出。
122typedef struct {
123 mp_ptr buf, snap, sep, numa, src;
124 size_t cap, g, n;
125} overlap_buf_t;
126
127static overlap_buf_t overlap_buf_create(mp_size_t n, size_t g) {
128 overlap_buf_t ob = {.cap = n + 2 * g, .g = g, .n = n};
129 ob.buf = alloc_limbs(ob.cap);
130 ob.snap = alloc_limbs(ob.cap);
131 ob.sep = alloc_limbs(n);
132 ob.src = alloc_limbs(n);
133 ob.numa = ob.buf + g;
134 return ob;
135}
136
137static void overlap_buf_free(overlap_buf_t *ob) {
138 lmmp_free(ob->buf);
139 lmmp_free(ob->snap);
140 lmmp_free(ob->sep);
141 lmmp_free(ob->src);
142}
143
144static void overlap_buf_set_src(overlap_buf_t *ob) {
145 u64 seed = 0x9e3779b97f4a7c15ull ^ (ob->n * 31 + ob->g);
146 random_limbs(ob->src, ob->n, &seed);
147}
148
149static void overlap_buf_refill(overlap_buf_t *ob, int w) {
150 for (size_t i = 0; i < ob->cap; ++i)
151 ob->buf[i] = 0xA5A5DEADBEEF00F0ull ^ (i * 0x9e37u + (u64)w * 7u);
152 lmmp_copy(ob->numa, ob->src, ob->n);
153 lmmp_copy(ob->snap, ob->buf, ob->cap);
154}
155
156static mp_ptr overlap_buf_dst_of(const overlap_buf_t *ob, int w) {
157 return w == 100 ? ob->sep : ob->numa + w;
158}
159
160static void overlap_buf_check_guard(const overlap_buf_t *ob, int w, const char *tag) {
161 mp_srcptr dst = overlap_buf_dst_of(ob, w);
162 for (size_t i = 0; i < ob->cap; ++i) {
163 if (w != 100 && ob->buf + i >= dst && ob->buf + i < dst + ob->n)
164 continue;
165 if (ob->buf[i] != ob->snap[i]) {
166 TEST_CHECK_MSG(false, tag);
167 return;
168 }
169 }
170}

172TEST_CASE("asm/boundary", shl_shr_overlap_family) {
173 // shl 要求 eqsep，shr 允许 dst <= numa。
174 // 期望值取分离缓冲调用；返回值由输入 limbs 直接推出。
175 for (mp_size_t n = 1; n <= 24; ++n) {
176 overlap_buf_t ob = overlap_buf_create(n, 4);
177 overlap_buf_set_src(&ob);
178
179 for (size_t k = 0; k < 64; ++k) {
180 mp_ptr eshl = alloc_limbs(n), eshr = alloc_limbs(n);
181 mp_limb_t ret_shl = k ? (ob.src[n - 1] >> (64 - k)) : 0;
182 mp_limb_t ret_shr = k ? (ob.src[0] << (64 - k)) : 0;
183 lmmp_shl_(eshl, ob.src, n, k);
184 lmmp_shr_(eshr, ob.src, n, k);
185
186 const int ws_shl[] = {100, 0}; // shl 纯 eqsep：部分重叠族契约外，不做断言
187 for (size_t w_index = 0; w_index < ARRAY_SIZE(ws_shl); ++w_index) {
188 int w = ws_shl[w_index];
189 overlap_buf_refill(&ob, w);
190 mp_limb_t r = lmmp_shl_(overlap_buf_dst_of(&ob, w), ob.numa, n, k);
191 TEST_CHECK_EQ(r, ret_shl);
192 TEST_CHECK_EQ(memcmp(overlap_buf_dst_of(&ob, w), eshl, n * 8), 0);
193 overlap_buf_check_guard(&ob, w, "shl overlap family guard");
194 }
195 const int ws_shr[] = {100, 0, -1, -2, -3};
196 for (size_t w_index = 0; w_index < ARRAY_SIZE(ws_shr); ++w_index) {
197 int w = ws_shr[w_index];
198 overlap_buf_refill(&ob, w);
199 mp_limb_t r = lmmp_shr_(overlap_buf_dst_of(&ob, w), ob.numa, n, k);
200 TEST_CHECK_EQ(r, ret_shr);
201 TEST_CHECK_EQ(memcmp(overlap_buf_dst_of(&ob, w), eshr, n * 8), 0);
202 overlap_buf_check_guard(&ob, w, "shr overlap family guard");
203 }
204 lmmp_free(eshl);
205 lmmp_free(eshr);
206 }
207 overlap_buf_free(&ob);
208 }

211TEST_CASE("asm/boundary", shl_shr_c_overlap_family) {
212 // _c 变体的错位重叠族（c 满足契约：shl_c 的 c 高 64-k 位为 0，
213 // shr_c 的 c 低 64-k 位为 0）；期望 = 分离调用 + 手工并位。
214 u64 seed = 0xc0ffee1234567890ul;
215 for (mp_size_t n = 2; n <= 12; ++n) {
216 overlap_buf_t ob = overlap_buf_create(n, 4);
217 overlap_buf_set_src(&ob);
218
219 const size_t k_values[] = {1, 5, 32, 63};
220 for (size_t k_index = 0; k_index < ARRAY_SIZE(k_values); ++k_index) {
221 size_t k = k_values[k_index];
222 mp_limb_t cshl = xorshift64(&seed) & (((mp_limb_t)1 << k) - 1);
223 mp_limb_t cshr = (xorshift64(&seed) >> (64 - k)) << (64 - k);
224 mp_limb_t ret_shl = ob.src[n - 1] >> (64 - k);
225 mp_limb_t ret_shr = ob.src[0] << (64 - k);
226
227 mp_ptr eshl = alloc_limbs(n), eshr = alloc_limbs(n);
228 lmmp_shl_(eshl, ob.src, n, k);
229 eshl[0] |= cshl;
230 lmmp_shr_(eshr, ob.src, n, k);
231 eshr[n - 1] |= cshr;
232
233 const int ws_shl[] = {100, 0}; // shl_c 纯 eqsep，同 shl_
234 for (size_t w_index = 0; w_index < ARRAY_SIZE(ws_shl); ++w_index) {
235 int w = ws_shl[w_index];
236 overlap_buf_refill(&ob, w);
237 mp_limb_t r = lmmp_shl_c_(overlap_buf_dst_of(&ob, w), ob.numa, n, k, cshl);
238 TEST_CHECK_EQ(r, ret_shl);
239 TEST_CHECK_EQ(memcmp(overlap_buf_dst_of(&ob, w), eshl, n * 8), 0);
240 overlap_buf_check_guard(&ob, w, "shl_c overlap family guard");
241 }
242 const int ws_shr[] = {100, 0, -1, -2, -3};
243 for (size_t w_index = 0; w_index < ARRAY_SIZE(ws_shr); ++w_index) {
244 int w = ws_shr[w_index];
245 overlap_buf_refill(&ob, w);
246 mp_limb_t r = lmmp_shr_c_(overlap_buf_dst_of(&ob, w), ob.numa, n, k, cshr);
247 TEST_CHECK_EQ(r, ret_shr);
248 TEST_CHECK_EQ(memcmp(overlap_buf_dst_of(&ob, w), eshr, n * 8), 0);
249 overlap_buf_check_guard(&ob, w, "shr_c overlap family guard");
250 }
251 lmmp_free(eshl);
252 lmmp_free(eshr);
253 }
254 overlap_buf_free(&ob);
255 }

258TEST_CASE("asm/boundary", div_1_by_one_and_power_of_two) {
259 u64 seed = 0xf00dface12345678ull;
260 const mp_size_t n_values[] = {1, 2, 5, 16};
261 for (size_t n_index = 0; n_index < ARRAY_SIZE(n_values); ++n_index) {
262 mp_size_t n = n_values[n_index];
263 mp_ptr a = alloc_limbs(n), q = alloc_limbs(n + 1), t = alloc_limbs(n + 2);
264 for (mp_size_t i = 0; i < n; ++i) a[i] = xorshift64(&seed);
265
266 // x == 1：商等于原数，余数 0
267 memcpy(q, a, n * 8);
268 u64 r = lmmp_div_1_(q, q, n, 1);
269 TEST_CHECK_EQ(r, 0u);
270 TEST_CHECK_EQ(memcmp(q, a, n * 8), 0);
271
272 // x 为 2 的幂：重建 q*x + r == a
273 u64 x = 1ull << ((seed & 63) ? (seed & 63) : 1);
274 memcpy(q, a, n * 8);
275 r = lmmp_div_1_(q, q, n, x);
276 TEST_CHECK(r < x);
277 u64 hi = lmmp_mul_1_(t, q, n, x);
278 t[n] = hi;
279 u64 ovf = add_small_inplace(t, n + 1, r);
280 (void)ovf;
281 TEST_CHECK(mpn_eq(t, n + 1, a, n));
282
283 lmmp_free(a); lmmp_free(q); lmmp_free(t);
284 }

287/*
288 inv_1 界恒等式（积经 mul_1 双 limb 折算）；mulmod_ulong 与
289 mullh + div_1 路径交叉：qq*mod + r == a*b ⟺ 商余一致。
290*/
291TEST_CASE("asm/boundary", inv_and_mulmod_boundary) {
292 u64 seed = 0x0123456789abcdeful;
293 for (int i = 0; i < 200; ++i) {
294 u64 x = xorshift64(&seed) | LIMB_B_2;
295 u64 inv = lmmp_inv_1_(x);
296 mp_limb_t prod[2];
297 u64 hi = lmmp_mul_1_(prod, &x, 1, inv);
298 prod[1] = hi;
299 u64 p_hi = prod[1] + x; // x*(B+inv) 的高半（无溢出）
300 TEST_CHECK_EQ(p_hi, ~(u64)0);
301 TEST_CHECK(prod[0] >= (u64)(0 - x));
302
303 u64 mod = xorshift64(&seed) | 3;
304 u64 a = xorshift64(&seed) % mod;
305 u64 b = xorshift64(&seed) % mod;
306 u64 qq = 0;
307 u64 r = lmmp_mulmod_ulong_(a, b, mod, &qq);
308 mp_limb_t ab[2], q2[3];
309 lmmp_mullh_(a, b, ab);
310 mp_limb_t abw[2] = {ab[0], ab[1]};
311 u64 r2 = lmmp_div_1_(q2, abw, 2, mod);
312 TEST_CHECK_EQ(r2, r);
313 TEST_CHECK_EQ(q2[0], qq);
314 TEST_CHECK_EQ(norm_len(q2 + 1, 1), (mp_size_t)0);
315 }
