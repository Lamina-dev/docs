26void lmmp_fill(mp_ptr begin, mp_ptr end, mp_limb_t val) {
27 lmmp_param_assert(begin != NULL && end != NULL);
28 for (mp_ptr p = begin; p < end; p++) {
29 *p = val;
30 }
31}

33void lmmp_fill_n(mp_ptr dst, mp_size_t len, mp_limb_t val) {
34 lmmp_param_assert(dst != NULL);
35 for (mp_size_t i = 0; i < len; i++) {
36 dst[i] = val;
37 }
