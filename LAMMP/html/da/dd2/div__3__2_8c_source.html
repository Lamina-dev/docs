27static double bench_div_3_2(const bench_options *o) {
28 mp_ptr np = alloc_limbs(3), npw = alloc_limbs(3), dp = alloc_limbs(2);
29 np[0] = 0x1111111122222222ull;
30 np[1] = 0x3333333344444444ull;
31 np[2] = 0x5555555566666666ull;
32 dp[0] = 0x7777777788888888ull;
33 dp[1] = 0x99999999aaaaaaaall | LIMB_B_2;
34 mp_limb_t inv21 = lmmp_inv_2_1_(dp[1], dp[0]);
35 double us;
36 BENCH_MEASURE(us, o,
37 npw[0] = np[0];
38 npw[1] = np[1];
39 npw[2] = np[2];
40 bench_sink += lmmp_div_3_2_(npw, dp, inv21);
41 );
42 lmmp_free(np);
43 lmmp_free(npw);
44 lmmp_free(dp);
45 return us;
46}
