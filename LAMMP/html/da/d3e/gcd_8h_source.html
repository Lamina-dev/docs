100#define GCD_LEHMER_MIN_V 0x100000000ll /* 停机界：窗口余数将跌破 2^32 时停止 */
101#define GCD_LEHMER_EXACT_BITS 63 /* 顶端对齐窗口位宽 */
102
103typedef struct {
104 slong m11, m12;
105 slong m21, m22;
106} lmmp_gcd_lehmer_t;
107
108/*
109步进矩阵：在窗口 (u,v) 上运行小规模欧几里得。
110消费序 new_a = C*a + D*b、new_b = A*a + B*b（C=m21、D=m22、A=m11、B=m12），
111元素带符号，|A..D| < 2^31（v >= 2^32 停机界：元素 <= x/2^32 < 2^63/2^32）
112*/
113static inline void lmmp_gcd_lehmer_step_(slong u, slong v, lmmp_gcd_lehmer_t* M) {
114#define A (M->m11)
115#define B (M->m12)
116#define C (M->m21)
117#define D (M->m22)
118 A = 1;
119 B = 0;
120 C = 0;
121 D = 1;
122 while (v != 0) {
123 slong q = u / v;
124 slong t = u % v;
125 u = v;
126 v = t;
127 t = A - q * C;
128 A = C;
129 C = t;
130 t = B - q * D;
131 B = D;
132 D = t;
133 if (v < (slong)GCD_LEHMER_MIN_V)
134 break;
135 }
136#undef A
137#undef B
138#undef C
139#undef D
140}

142static inline void lmmp_gcd_lehmer_extract_(mp_srcptr up, mp_size_t un, mp_srcptr vp, mp_size_t vn,
143 slong* restrict a, slong* restrict b) {
144 int kz = lmmp_limb_bits_(up[un - 1]);
145 if (kz >= GCD_LEHMER_EXACT_BITS) {
146 *a = up[un - 1] >> (kz - GCD_LEHMER_EXACT_BITS);
147 if (vn == un)
148 *b = vp[vn - 1] >> (kz - GCD_LEHMER_EXACT_BITS);
149 else
150 *b = 0;
151 } else {
152 *a = up[un - 1] << (GCD_LEHMER_EXACT_BITS - kz);
153 *a |= up[un - 2] >> (LIMB_BITS - (GCD_LEHMER_EXACT_BITS - kz));
154 if (un > vn + 1) {
155 *b = 0;
156 } else if (un == vn + 1) {
157 *b = vp[vn - 1] >> (LIMB_BITS - (GCD_LEHMER_EXACT_BITS - kz));
158 } else {
159 *b = vp[vn - 1] << (GCD_LEHMER_EXACT_BITS - kz);
160 *b |= vp[vn - 2] >> (LIMB_BITS - (GCD_LEHMER_EXACT_BITS - kz));
161 }
162 }
163}

165/*
166dst = (x,xneg) ± (y,yneg)：按符号分派和/差；dst 与 x,y 缓冲分离。
167零向量约定 n==0（neg 无意义）
168*/
169static inline void lmmp_gcdext_smag_(mp_ptr dst, mp_size_t* n, int* neg, mp_srcptr x, mp_size_t xn,
170 int xneg, mp_srcptr y, mp_size_t yn, int yneg) {
171 if (xn == 0) {
172 lmmp_copy(dst, y, yn);
173 *n = yn;
174 *neg = yneg;
175 return;
176 }
177 if (yn == 0) {
178 lmmp_copy(dst, x, xn);
179 *n = xn;
180 *neg = xneg;
181 return;
182 }
183 if (xneg == yneg) {
184 mp_limb_t c;
185 if (xn >= yn) {
186 c = lmmp_add_(dst, x, xn, y, yn);
187 *n = xn;
188 } else {
189 c = lmmp_add_(dst, y, yn, x, xn);
190 *n = yn;
191 }
192 if (c != 0)
193 dst[(*n)++] = c;
194 *neg = xneg;
195 return;
196 }
197 int c = (xn != yn) ? (xn < yn ? -1 : 1) : lmmp_cmp_(x, y, xn);
198 if (c == 0) {
199 *n = 0;
200 *neg = 0;
201 return;
202 }
203 if (c > 0) {
204 lmmp_sub_(dst, x, xn, y, yn);
205 *n = xn;
206 *neg = xneg;
207 } else {
208 lmmp_sub_(dst, y, yn, x, xn);
209 *n = yn;
210 *neg = yneg;
211 }
212 while (*n > 0 && dst[*n - 1] == 0)
213 --*n;
214 if (*n == 0)
215 *neg = 0;
216}
