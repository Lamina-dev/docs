40/* 与 basecase 精确对拍 + 模检查（两条独立实现路径；按契约有效长度） */
41static void check_vs_oracle(const char* tag, mp_ptr a, mp_size_t na, mp_ptr b, mp_size_t nb, void (*fn)(mp_ptr, mp_srcptr, mp_size_t, mp_srcptr, mp_size_t)) {
42 mp_ptr dst = alloc_limbs(na + nb);
43 mp_ptr ref = alloc_limbs(na + nb + 1);
44 fn(dst, a, na, b, nb);
45 lmmp_mul_basecase_(ref, a, na, b, nb);
46 bool ok = memcmp(dst, ref, (size_t)(na + nb) * 8) == 0;
47 if (!ok || !mod_is_product(dst, na + nb, a, na, b, nb)) TEST_CHECK_MSG(false, tag);
48 else TEST_CHECK(true);
49 lmmp_free(dst);
50 lmmp_free(ref);
51}

53/* 内核直调 vs 独立分派路径（尺寸区间落在 toom44 及以下，路径独立） */
54static void check_vs_dispatch(const char* tag, mp_ptr a, mp_size_t na, mp_ptr b, mp_size_t nb, void (*kfn)(mp_ptr, mp_srcptr, mp_size_t, mp_srcptr, mp_size_t)) {
55 mp_ptr r1 = alloc_limbs(na + nb);
56 mp_ptr r2 = alloc_limbs(na + nb);
57 kfn(r1, a, na, b, nb);
58 lmmp_mul_(r2, a, na, b, nb);
59 if (memcmp(r1, r2, (size_t)(na + nb) * sizeof(mp_limb_t)) != 0) TEST_CHECK_MSG(false, tag);
60 else TEST_CHECK(true);
61 lmmp_free(r1);
62 lmmp_free(r2);
63}

65/* toom6h 原生比例范围：an/bn ∈ [1, 17/6)，an*3 < 8*bn 要求 bn < 46 时
66 比例 < 8/3 —— 阶梯各档代表比例 */
67const double kRatios6[] = {1.0, 1.05, 1.15, 1.3, 1.32, 1.45, 1.48, 1.8, 1.88, 2.0, 2.11, 2.4, 2.6, 2.82};
68/* toom8h 原生比例范围：an/bn ∈ [1, 4] */
69const double kRatios8[] = {1.0, 1.1, 1.22, 1.35, 1.5, 1.64, 1.8, 2.0, 2.24, 2.5, 2.85, 3.1, 3.5, 3.99};
70
71static bool in_6h_range(mp_size_t na, mp_size_t nb) {
72 /* 与内核契约一致：比例 < 17/6 且 nb>=46（原 8/3 宽限与 nb 42..45
73 保证域外区间随契约收紧移除） */
74 return na >= nb && nb >= 46 && 6 * na < 17 * nb;
75}

191TEST_CASE("mul/toom68", thresholds_boundary_via_dispatch) {
192 u64 seed = 0x7468726573686f6cull;
193 /* 阈值 ±1 经 lmmp_mul_n_/lmmp_sqr_ 完整分派对拍 oracle */
194 struct threshold_case {
195 mp_size_t base;
196 int is_sqr;
197 } bounds[] = {
198 {MUL_TOOM6H_THRESHOLD, 0},
199 {MUL_TOOM8H_THRESHOLD, 0},
200 {SQR_TOOM6_THRESHOLD, 1},
201 {SQR_TOOM8_THRESHOLD, 1},
202 };
203 for (size_t bd_index = 0; bd_index < ARRAY_SIZE(bounds); ++bd_index) {
204 const struct threshold_case bd = bounds[bd_index];
205 for (int d = -1; d <= 1; ++d) {
206 if (bd.base + (mp_size_t)d < 1) continue;
207 mp_size_t n = bd.base + d;
208 if (bd.is_sqr) {
209 mp_ptr a = alloc_limbs(n);
210 mp_ptr r = alloc_limbs(2 * n), ref = alloc_limbs(2 * n + 1);
211 random_limbs(a, n, &seed);
212 lmmp_sqr_(r, a, n);
213 lmmp_mul_basecase_(ref, a, n, a, n);
214 bool ok = memcmp(r, ref, (size_t)(2 * n) * 8) == 0;
215 if (!ok || !mod_is_product(r, 2 * n, a, n, a, n))
216 TEST_CHECK_MSG(false, "sqr threshold boundary mismatch");
217 else TEST_CHECK(true);
218 lmmp_free(a);
219 lmmp_free(r);
220 lmmp_free(ref);
221 } else {
222 mp_ptr a = alloc_limbs(n), b = alloc_limbs(n);
223 mp_ptr r = alloc_limbs(2 * n), ref = alloc_limbs(2 * n + 1);
224 random_limbs(a, n, &seed);
225 random_limbs(b, n, &seed);
226 lmmp_mul_n_(r, a, b, n);
227 lmmp_mul_basecase_(ref, a, n, b, n);
228 bool ok = memcmp(r, ref, (size_t)(2 * n) * 8) == 0;
229 if (!ok || !mod_is_product(r, 2 * n, a, n, b, n))
230 TEST_CHECK_MSG(false, "mul threshold boundary mismatch");
231 else TEST_CHECK(true);
232 lmmp_free(a);
233 lmmp_free(b);
234 lmmp_free(r);
235 lmmp_free(ref);
236 }
237 }
