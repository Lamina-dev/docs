80/**
81 * @brief 牛顿法求精确逆元（至多产生1的误差）
82 * @note dstq := B^(2*(na+ni)) / ([numa,na] * B^ni) + [0|1]
83 * @warning sep(dstq,numa), dstq!=NULL, numa!=NULL, na>=3, MSB(numa)=1
84 */
85static void lmmp_bninv_appr_newton_(mp_ptr restrict dstq, mp_srcptr restrict numa, mp_size_t na, mp_size_t ni) {
86 lmmp_param_assert(na >= 3);
87 lmmp_param_assert(dstq != NULL && numa != NULL);
88 lmmp_param_assert(numa[na - 1] > LIMB_B_2);
89 TEMP_DECL;
90 if (na < BNINV_NEWTON_THRESHOLD) {
91 mp_ptr restrict bnp = TALLOC_TYPE(2 * na + ni + 1, mp_limb_t);
92 lmmp_zero(bnp, 2 * na + ni + 1);
93 bnp[2 * na + ni] = 1;
94 mp_limb_t inv21 = lmmp_inv_2_1_(numa[na - 1], numa[na - 2]);
95 lmmp_div_basecase_(dstq, bnp, 2 * na + ni + 1, numa, na, inv21);
96 } else {
97 mp_srcptr restrict a_hat;
98
99 mp_size_t N = na + ni;
100 mp_size_t a_hatn = N / 2 + 1;
101 mp_size_t remn = N - a_hatn;
102 mp_size_t ni_hat;
103 if (remn > ni) {
104 ni_hat = 0;
105 a_hat = numa + na - a_hatn;
106 } else {
107 ni_hat = ni - remn;
108 a_hat = numa;
109 a_hatn = na;
110 }
111 mp_size_t q_hatn = a_hatn + ni_hat + 1;
112 mp_size_t qrn = N + 1 - q_hatn;
113 mp_ptr restrict q_hat = dstq + qrn;
114
115 mp_ptr restrict q_hat_sqr = TALLOC_TYPE(2 * q_hatn, mp_limb_t);
116 mp_ptr restrict q_hat_sqr_a = TALLOC_TYPE(2 * q_hatn + na, mp_limb_t);
117
118 lmmp_bninv_appr_newton_(q_hat, a_hat, a_hatn, ni_hat);
119 lmmp_sqr_(q_hat_sqr, q_hat, q_hatn);
120 lmmp_mul_(q_hat_sqr_a, q_hat_sqr, 2 * q_hatn, numa, na);
121 // we can assert q_hat_sqr_a[2*q_hatn+na-1] == 0
122 lmmp_zero(dstq, qrn);
123 if (2 * qrn + ni > 2 * N) {
124 mp_size_t start = 2 * qrn + ni - 2 * N;
125 lmmp_shl_(q_hat, q_hat, q_hatn, 1); // assert no carry
126 lmmp_sub_n_(dstq + start, dstq + start, q_hat_sqr_a, N + 1 - start);
127 } else {
128 mp_size_t start = 2 * N - 2 * qrn - ni;
129 lmmp_shl_(q_hat, q_hat, q_hatn, 1); // assert no carry
130 lmmp_sub_n_(dstq, dstq, q_hat_sqr_a + start, N + 1);
131 }
132 }
133 TEMP_FREE;
134}

136void lmmp_bninv_(mp_ptr restrict dstq, mp_srcptr restrict numa, mp_size_t na, mp_size_t ni) {
137 lmmp_param_assert(na > 0);
138 lmmp_param_assert(dstq != NULL && numa != NULL);
139 lmmp_param_assert(numa[na - 1] > 0);
140 TEMP_DECL;
141 if (na == 1) {
142 lmmp_zero(dstq, 2 + ni);
143 dstq[2 + ni] = 1;
144 lmmp_div_1_(dstq, dstq, 3 + ni, numa[0]);
145 } else if (na == 2) {
146 mp_size_t bn = 2 * 2 + ni + 1;
147 mp_ptr restrict bnp = TALLOC_TYPE(bn, mp_limb_t);
148 lmmp_zero(bnp, bn - 1);
149 bnp[bn - 1] = 1;
150 mp_limb_t d[2] = {numa[0], numa[1]};
151 lmmp_div_2_(dstq, bnp, bn, d);
152 } else if (ni > na) {
153 mp_ptr restrict B = TALLOC_TYPE(2 * na + ni + 1, mp_limb_t);
154 lmmp_zero(B, 2 * na + ni);
155 B[2 * na + ni] = 1;
156 lmmp_div_(dstq, NULL, B, 2 * na + ni + 1, numa, na);
157 } else {
158 int shift = lmmp_leading_zeros_(numa[na - 1]);
159 if (shift > 0) {
160 mp_ptr restrict numa_shift = TALLOC_TYPE(na, mp_limb_t);
161 lmmp_shl_(numa_shift, numa, na, shift);
162 lmmp_bninv_appr_newton_(dstq, numa_shift, na, ni + 1);
163 lmmp_shr_(dstq, dstq, na + ni + 2, LIMB_BITS - shift);
164 } else {
165 lmmp_bninv_appr_newton_(dstq, numa, na, ni + 1);
166 lmmp_copy(dstq, dstq + 1, na + ni + 1);
167 dstq[na + ni + 1] = 0;
168 }
169 }
170 TEMP_FREE;
171 return;
172}
