122void lmmp_abort(lmmp_error_t type, const char* msg, const char* func, int line) {
123 abort_func_lock();
124 lmmp_abort_fn fn = lmmp_abort_func;
125 abort_func_unlock();
126
127 if (fn != NULL) {
128 fn(type, msg, func, line);
129 } else {
130 fprintf(stderr, "LMMP abort at [%s]:%d\n", func, line);
131 fprintf(stderr, "Abort type: %s, abort msg: \n%s\n", type_to_str(type), msg);
132 fflush(stderr);
133 }
134 abort();
135}
