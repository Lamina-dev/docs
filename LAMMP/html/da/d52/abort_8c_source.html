115void lmmp_abort(lmmp_error_t type, const char* msg, const char* func, int line) {
116 abort_func_lock();
117 lmmp_abort_fn fn = lmmp_abort_func;
118 abort_func_unlock();
119
120 if (fn != NULL) {
121 fn(type, msg, func, line);
122 } else {
123 fprintf(stderr, "LAMMP abort at [%s]:%d\n", func, line);
124 fprintf(stderr, "Abort type: %s, abort msg: \n%s\n", type_to_str(type), msg);
125 fflush(stderr);
126 }
127 abort();
128}
