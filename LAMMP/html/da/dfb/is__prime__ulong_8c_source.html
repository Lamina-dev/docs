34mont64 可以用于任意大小的数，但由于考虑了溢出的情况，所以速度理论上会慢一些（并未详细测试）

38static inline ulong mont64_reduce(u128 t, ulong m, ulong m_inv) {
39 // sum = k*m + t < 2^129，第 129 位 c 由 128 位回绕捕获
40 ulong k = _u128low(t) * m_inv;
41 u128 km = (u128)k * m;
42 u128 sum = km + t;
43 uint c = sum < km;
44 ulong hi = _u128high(sum);
45 if (c) return hi - m; // 2^128 + hi - m 的低 128 位高位即结果
46 return hi >= m ? hi - m : hi;
47}

67static inline ulong mont63_reduce(u128 t, ulong m, ulong m_inv) {
68 // m < 2^63 时 k*m + t < 2^127，无 129 位溢出
69 ulong k = _u128low(t) * m_inv;
70 u128 sum = (u128)k * m + t;
71 ulong hi = _u128high(sum);
72 return hi >= m ? hi - m : hi;
73}

87/*
88 强伪素数第二阶段检查：v = b^d（蒙域）已算出。
89 one/m_1 为蒙域的 1 与 m-1。
90*/
91static inline int sprp_check63(ulong v, ulong t, ulong m, ulong m_inv, ulong one, ulong m_1) {
92 if (v == one || v == m_1) return 1;
93 for (t--; t > 0; t--) {
94 v = mont63_mul(v, v, m, m_inv);
95 if (v == m_1) return 1;
96 if (v == one) return 0;
97 }
98 return 0;
99}

101static inline int sprp_check64(ulong v, ulong t, ulong m, ulong m_inv, ulong one, ulong m_1) {
102 if (v == one || v == m_1) return 1;
103 for (t--; t > 0; t--) {
104 v = mont64_mul(v, v, m, m_inv);
105 if (v == m_1) return 1;
106 if (v == one) return 0;
107 }
108 return 0;

111/*
112 基底 2 专用幂（蒙域，LSB 链）：基底以蒙域值起步后，平方链保持
113 base_j = 2^(2^j)*B 不变量，结果恒在蒙域；平方链与 v 乘链相互独立，
114 乱序执行可重叠。
115*/
116static inline ulong powmod2_63(ulong d, ulong R2, ulong m, ulong m_inv) {
117 ulong v = to_mont63(1, R2, m, m_inv);
118 ulong base = to_mont63(2, R2, m, m_inv);
119 while (1) {
120 if (d & 1) v = mont63_mul(v, base, m, m_inv);
121 d >>= 1;
122 if (d == 0) break;
123 base = mont63_mul(base, base, m, m_inv);
124 }
125 return v;
126}

128static inline ulong powmod2_64(ulong d, ulong R2, ulong m, ulong m_inv) {
129 ulong v = to_mont64(1, R2, m, m_inv);
130 ulong base = to_mont64(2, R2, m, m_inv);
131 while (1) {
132 if (d & 1) v = mont64_mul(v, base, m, m_inv);
133 d >>= 1;
134 if (d == 0) break;
135 base = mont64_mul(base, base, m, m_inv);
136 }
137 return v;
138}

140static void powmod_win3_63(ulong *pv, ulong bm, ulong d, int ebits, ulong m, ulong m_inv) {
141 ulong w[8], v = bm;
142 int bit = ebits - 2;
143 w[1] = bm;
144 w[2] = mont63_mul(bm, bm, m, m_inv);
145 w[3] = mont63_mul(w[2], bm, m, m_inv);
146 w[4] = mont63_mul(w[2], w[2], m, m_inv);
147 w[5] = mont63_mul(w[4], bm, m, m_inv);
148 w[6] = mont63_mul(w[3], w[3], m, m_inv);
149 w[7] = mont63_mul(w[6], bm, m, m_inv);
150 while (bit >= 2) {
151 v = mont63_mul(v, v, m, m_inv);
152 v = mont63_mul(v, v, m, m_inv);
153 v = mont63_mul(v, v, m, m_inv);
154 {
155 uint idx = (uint)(d >> (bit - 2)) & 7;
156 if (idx) v = mont63_mul(v, w[idx], m, m_inv);
157 }
158 bit -= 3;
159 }
160 for (; bit >= 0; bit--) {
161 v = mont63_mul(v, v, m, m_inv);
162 if ((d >> bit) & 1) v = mont63_mul(v, bm, m, m_inv);
163 }
164 *pv = v;
165}

257LMMP_NOINLINE bool lmmp_is_prime_ulong_notrial_(ulong n) {
258 lmmp_param_assert(n > 2);
259 if (n <= MP_UINT_MAX) {
260 if (n % 2 == 0)
261 return false;
262 return lmmp_is_prime_uint_notrial_((uint)n);
263 } else if (n < 684630005672341) {
264 ushort bases[2];
265 bases[0] = 2;
266 bases[1] = dj_base49[((0x3AC69A35UL * n) & 0xFFFFFFFFUL) >> 21] + 3;
267 if (n % bases[0] == 0)
268 return false;
269 if (n % bases[1] == 0)
270 return false;
271
272 ulong m_inv = -lmmp_binvert_ulong_(n);
273 ulong R2 = mont63_R2(n);
274 ulong one = to_mont63(1, R2, n, m_inv);
275 ulong m_1 = to_mont63(n - 1, R2, n, m_inv);
276
277 ulong d;
278 ulong t;
279 ctz_shr_u64(d, n - 1, t);
280 int eb = lmmp_limb_bits_(d);
281
282 ulong v2 = powmod2_63(d, R2, n, m_inv);
283 if (!sprp_check63(v2, t, n, m_inv, one, m_1)) return false;
284
285 ulong v;
286 powmod_win3_63(&v, to_mont63(bases[1], R2, n, m_inv), d, eb, n, m_inv);
287 return sprp_check63(v, t, n, m_inv, one, m_1);
288 } else {
289 ushort bases[3];
290 ulong bbmask = dj_base64[((0x3AC69A35UL * n) & 0xFFFFFFFFUL) >> 18];
291 bases[0] = 2;
292 bases[1] = (bbmask & 0x8000) ? 26460 : 9375;
293 bases[2] = (bbmask & 0x7FFF) + 3;
294
295 if (n % bases[0] == 0)
296 return false;
297 if (n % bases[1] == 0)
298 return false;
299 if (n % bases[2] == 0)
300 return n == bases[2];
301
302 ulong one = 1;
303 ulong m_1 = n - 1;
304 ulong m_inv = -lmmp_binvert_ulong_(n);
305
306 ulong d;
307 ulong t;
308 ctz_shr_u64(d, n - 1, t);
309 int eb = lmmp_limb_bits_(d);
310
311 if (n <= MONT63_MAX) {
312 ulong R2 = mont63_R2(n);
313 one = to_mont63(one, R2, n, m_inv);
314 m_1 = to_mont63(m_1, R2, n, m_inv);
315
316 ulong v2 = powmod2_63(d, R2, n, m_inv);
317 if (!sprp_check63(v2, t, n, m_inv, one, m_1)) return false;
318
319 /* 双基底交错 + 3 位窗口 */
320 ulong bm1 = to_mont63(bases[1], R2, n, m_inv);
321 ulong bm2 = to_mont63(bases[2], R2, n, m_inv);
322 ulong v1, v2_;
323 powmod2_win3_63(&v1, &v2_, bm1, bm2, d, eb, n, m_inv);
324 if (!sprp_check63(v1, t, n, m_inv, one, m_1)) return false;
325 return sprp_check63(v2_, t, n, m_inv, one, m_1);
326 } else {
327 ulong R2 = mont64_R2(n);
328 one = to_mont64(one, R2, n, m_inv);
329 m_1 = to_mont64(m_1, R2, n, m_inv);
330
331 ulong v2 = powmod2_64(d, R2, n, m_inv);
332 if (!sprp_check64(v2, t, n, m_inv, one, m_1)) return false;
333
334 /* 双基底交错 + 3 位窗口 */
335 ulong bm1 = to_mont64(bases[1], R2, n, m_inv);
336 ulong bm2 = to_mont64(bases[2], R2, n, m_inv);
337 ulong v1, v2_;
338 powmod2_win3_64(&v1, &v2_, bm1, bm2, d, eb, n, m_inv);
339 if (!sprp_check64(v1, t, n, m_inv, one, m_1)) return false;
340 return sprp_check64(v2_, t, n, m_inv, one, m_1);
341 }
342 }

345bool lmmp_is_prime_ulong_(ulong n) {
346 if (n == 2)
347 return true;
348 if (n % 2 == 0 || n <= 1)
349 return false;
350 if (n <= MP_UINT_MAX) {
351 int judge = lmmp_is_prime_table_(n);
352 if (judge == 0) {
353 return false;
354 } else if (judge == 1) {
355 return true;
356 }
357 }
358 if (trial_div35711(n))
359 return false;
360 return lmmp_is_prime_ulong_notrial_(n);
361}
