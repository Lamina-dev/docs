24mont63 只能用于小于 2^63 的数，否则会溢出导致计算结果不正确
25mont64 可以用于任意大小的数，但由于考虑了溢出的情况，所以速度理论上会慢一些（并未详细测试）

29static inline ulong mont64_reduce(u128 t, ulong m, ulong m_inv) {
30 ulong k = t[0] * m_inv;
31 u192 tmp;
32 _u128mul(tmp, k, m);
33 tmp[0] += t[0];
34 ulong c = tmp[0] < t[0];
35 tmp[1] += c;
36 c = tmp[1] == 0;
37 tmp[1] += t[1];
38 c = tmp[1] < t[1];
39 tmp[2] = c;
40 if (!c) {
41 ulong res = tmp[1] >= m ? tmp[1] - m : tmp[1];
42 return res;
43 } else {
44 _u128sub64(tmp + 1, tmp + 1, m);
45 return tmp[1];
46 }
47}

73static inline ulong mont63_reduce(u128 t, ulong m, ulong m_inv) {
74 ulong k = t[0] * m_inv;
75 u128 tmp;
76 _u128mul(tmp, k, m);
77 _u128add(tmp, tmp, t);
78 ulong res = tmp[1] >= m ? tmp[1] - m : tmp[1];
79 return res;
80}

108uint lmmp_powmod_uint_odd_(uint base, ulong exp, uint mod) {
109 lmmp_param_assert(mod > base);
110 lmmp_param_assert(mod % 2 == 1);
111 lmmp_param_assert(mod > 1);
112 ulong dst = 1;
113 ulong b = base;
114 _udiv64_t binv = _udiv64_gen(mod);
115 ulong q;
116 while (1) {
117 if (exp & 1) {
118 dst *= b;
119 q = _udiv64by64_q_preinv(dst, &binv);
120 dst -= q * mod;
121 }
122 exp >>= 1;
123 if (exp == 0)
124 break;
125 b *= b;
126 q = _udiv64by64_q_preinv(b, &binv);
127 b -= q * mod;
128 }
129 return dst;
130}

132ulong lmmp_powmod_ulong_odd_(ulong base, ulong exp, ulong mod) {
133 lmmp_param_assert(mod > base);
134 lmmp_param_assert(mod % 2 == 1);
135 lmmp_param_assert(mod > 1);
136 if (mod <= MP_UINT_MAX)
137 return lmmp_powmod_uint_odd_(base, exp, mod);
138 else if (mod <= MONT63_MAX) {
139 ulong R2 = mont63_R2(mod);
140 ulong m_inv = lmmp_binvert_ulong_(mod);
141 m_inv = -m_inv;
142 ulong dst = to_mont63(1, R2, mod, m_inv);
143 base = to_mont63(base, R2, mod, m_inv);
144 while (1) {
145 if (exp & 1)
146 dst = mont63_mul(dst, base, mod, m_inv);
147 exp >>= 1;
148 if (exp == 0)
149 break;
150 base = mont63_mul(base, base, mod, m_inv);
151 }
152 return from_mont63(dst, mod, m_inv);
153 } else {
154 ulong R2 = mont64_R2(mod);
155 ulong m_inv = lmmp_binvert_ulong_(mod);
156 m_inv = -m_inv;
157 ulong dst = to_mont64(1, R2, mod, m_inv);
158 base = to_mont64(base, R2, mod, m_inv);
159 while (1) {
160 if (exp & 1)
161 dst = mont64_mul(dst, base, mod, m_inv);
162 exp >>= 1;
163 if (exp == 0)
164 break;
165 base = mont64_mul(base, base, mod, m_inv);
166 }
167 return from_mont64(dst, mod, m_inv);
168 }

171static inline int miller_rabin_32(ulong a, ulong t, ulong u, uint m, _udiv64_t* binv) {
172 ulong v = 1;
173 ulong base = a;
174 ulong q;
175 while (1) {
176 if (u & 1) {
177 v *= base;
178 q = _udiv64by64_q_preinv(v, binv);
179 v -= q * m;
180 }
181 u >>= 1;
182 if (u == 0)
183 break;
184 base *= base;
185 q = _udiv64by64_q_preinv(base, binv);
186 base -= q * m;
187 }
188
189 if (v == 1 || v == m - 1)
190 return 1;
191 for (ulong j = 1; j < t; ++j) {
192 v *= v;
193 q = _udiv64by64_q_preinv(v, binv);
194 v -= q * m;
195 if (v == m - 1)
196 return 1;
197 if (v == 1)
198 return 0;
199 }
200 return 0;
201}

203static inline int miller_rabin_63(ulong a, ulong t, ulong u, ulong m, ulong m_inv, ulong one, ulong m_1) {
204 ulong v = one;
205 ulong base = a;
206 while (1) {
207 if (u & 1)
208 v = mont63_mul(v, base, m, m_inv);
209 u >>= 1;
210 if (u == 0)
211 break;
212 base = mont63_mul(base, base, m, m_inv);
213 }
214 if (v == one || v == m_1)
215 return 1;
216
217 for (ulong j = 1; j < t; ++j) {
218 v = mont63_mul(v, v, m, m_inv);
219 if (v == m_1)
220 return 1;
221 if (v == one)
222 return 0;
223 }
224 return 0;
225}

227static inline int miller_rabin_64(ulong a, ulong t, ulong u, ulong m, ulong m_inv, ulong one, ulong m_1) {
228 ulong v = one;
229 ulong base = a;
230 while (1) {
231 if (u & 1)
232 v = mont64_mul(v, base, m, m_inv);
233 u >>= 1;
234 if (u == 0)
235 break;
236 base = mont64_mul(base, base, m, m_inv);

238 if (v == one || v == m_1)
239 return 1;
240 for (ulong j = 1; j < t; ++j) {
241 v = mont64_mul(v, v, m, m_inv);
242 if (v == m_1)
243 return 1;
244 if (v == one)
245 return 0;
246 }
247 return 0;

250/*******************************************************************************
251 * from http://probableprime.org/download/example-primality.c
252 * Deterministic Miller-Rabin tests for 64-bit.
253 * Hashed 2-bases for n < 684630005672341 (slightly more than 2^49)
254 * Hashed 3-bases for n < 2^64
255 *
256 * Based on Steve Worley's 2^32 example:
257 * http://www.mersenneforum.org/showthread.php?t=12209
258 * With a 3-base encoding idea from Bradley Berg.
259 *
260 * Copyright 2014, Dana Jacobsen <dana@acm.org>
261 *******************************************************************************/

263bool lmmp_is_prime_uint_(uint n) {
264 if (n == 2)
265 return true;
266 if (n % 2 == 0 || n <= 1)
267 return false;
268 int judge = lmmp_is_prime_table_(n);
269 if (judge == 0) {
270 return false;
271 } else if (judge == 1) {
272 return true;
273 }
274
275 if (trial_div35711(n))
276 return false;
277
278 ushort bases[2];
279 bases[0] = 2;
280 bases[1] = dj_base49[((0x3AC69A35UL * n) & 0xFFFFFFFFUL) >> 21] + 3;
281 if (n % bases[0] == 0)
282 return false;
283 if (n % bases[1] == 0)
284 return false;
285
286 ulong u = n - 1, t = 0;
287 while (u % 2 == 0) u /= 2, ++t;
288
289 _udiv64_t binv = _udiv64_gen(n);
290 if (miller_rabin_32(bases[0], t, u, n, &binv))
291 if (miller_rabin_32(bases[1], t, u, n, &binv))
292 return true;
293 else
294 return false;
295 else
296 return false;
297}

299bool lmmp_is_prime_notrial_(ulong n) {
300 lmmp_param_assert(n > 2);
301 if (n < 684630005672341) {
302 ushort bases[2];
303 bases[0] = 2;
304 bases[1] = dj_base49[((0x3AC69A35UL * n) & 0xFFFFFFFFUL) >> 21] + 3;
305 if (n % bases[0] == 0)
306 return false;
307 if (n % bases[1] == 0)
308 return false;
309
310 ulong one = 1;
311 ulong m_1 = n - 1;
312 ulong m_inv = lmmp_binvert_ulong_(n);
313 m_inv = -m_inv;
314 ulong R2 = mont63_R2(n);
315 one = to_mont63(one, R2, n, m_inv);
316 m_1 = to_mont63(m_1, R2, n, m_inv);
317
318 ulong u = n - 1, t = 0;
319 while (u % 2 == 0) u /= 2, ++t;
320
321 if (miller_rabin_63(bases[0], t, u, n, m_inv, one, m_1))
322 if (miller_rabin_63(bases[1], t, u, n, m_inv, one, m_1))
323 return true;
324 else
325 return false;
326 else
327 return false;
328 } else {
329 ushort bases[3];
330 ulong bbmask = dj_base64[((0x3AC69A35UL * n) & 0xFFFFFFFFUL) >> 18];
331 bases[0] = 2;
332 bases[1] = (bbmask & 0x8000) ? 26460 : 9375;
333 bases[2] = (bbmask & 0x7FFF) + 3;
334
335 if (n % bases[0] == 0)
336 return false;
337 if (n % bases[1] == 0)
338 return false;
339 if (n % bases[2] == 0)
340 return false;
341
342 ulong one = 1;
343 ulong m_1 = n - 1;
344 ulong m_inv = lmmp_binvert_ulong_(n);
345 m_inv = -m_inv;
346
347 if (n <= MONT63_MAX) {
348 ulong R2 = mont63_R2(n);
349 one = to_mont63(one, R2, n, m_inv);
350 m_1 = to_mont63(m_1, R2, n, m_inv);
351
352 ulong u = n - 1, t = 0;
353 while (u % 2 == 0) u /= 2, ++t;
354
355 if (miller_rabin_63(bases[0], t, u, n, m_inv, one, m_1))
356 if (miller_rabin_63(bases[1], t, u, n, m_inv, one, m_1))
357 if (miller_rabin_63(bases[2], t, u, n, m_inv, one, m_1))
358 return true;
359 else
360 return false;
361 else
362 return false;
363 else
364 return false;
365 } else {
366 ulong R2 = mont64_R2(n);
367 one = to_mont64(one, R2, n, m_inv);
368 m_1 = to_mont64(m_1, R2, n, m_inv);
369
370 ulong u = n - 1, t = 0;
371 while (u % 2 == 0) u /= 2, ++t;
372
373 if (miller_rabin_64(bases[0], t, u, n, m_inv, one, m_1))
374 if (miller_rabin_64(bases[1], t, u, n, m_inv, one, m_1))
375 if (miller_rabin_64(bases[2], t, u, n, m_inv, one, m_1))
376 return true;
377 else
378 return false;
379 else
380 return false;
381 else
382 return false;
383 }
384 }

465ulong lmmp_next_prime_ulong_(ulong n) {
466 if (n < prime_short_table[PRIME_SHORT_TABLE_SIZE - 1]) {
467 ushort idx = lmmp_prime_cnt16_(n);
468 return prime_short_table[idx];
469 } else if (n >= ULONG_PRIME_MAX) {
470 return MP_ULONG_MAX;
471 } else {
472 n += (n % 2 == 0) ? 1 : 2;
473 while (1) {
474 if (trial_div35711(n)

476 || trial_div17(n)
477 || trial_div19(n)
478 || trial_div23(n)
479 || trial_div29(n)
480 || trial_div31(n)
481 || trial_div37(n)
482 || trial_div41(n)) {
483 n += 2;
484 } else {
485 if (lmmp_is_prime_notrial_(n)) {
486 return n;
487 } else {
488 n += 2;
489 }
490 }
491 }
492 }
