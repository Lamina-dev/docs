27mp_bitcnt_t lmmp_extract_bits_(mp_srcptr restrict num, mp_size_t n, mp_limb_t* restrict ext, int bits) {
28 lmmp_param_assert(bits <= LIMB_BITS && bits > 0);
29 lmmp_param_assert(num != NULL && ext != NULL);
30 lmmp_param_assert(n > 0);
31 if (n == 1) {
32 int lb = lmmp_limb_bits_(num[0]);
33 if (lb <= bits) {
34 *ext = num[0];
35 return 0;
36 } else {
37 *ext = num[0] >> (lb - bits);
38 return lb - bits;
39 }
40 } else {
41 int lb = lmmp_limb_bits_(num[n - 1]);
42 if (lb < bits) {
43 *ext = lb ? (num[n - 1] << (bits - lb)) : 0;
44 *ext |= num[n - 2] >> (LIMB_BITS - bits + lb);
45 return LIMB_BITS * (n - 1) - (bits - lb);
46 } else if (lb == bits) {
47 *ext = num[n - 1];
48 return LIMB_BITS * (n - 1);
49 } else {
50 *ext = num[n - 1] >> (lb - bits);
51 return LIMB_BITS * (n - 1) + (lb - bits);
52 }
53 }
