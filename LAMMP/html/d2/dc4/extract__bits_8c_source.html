20mp_bitcnt_t lmmp_extract_bits_(mp_srcptr restrict num, mp_size_t n, mp_limb_t* restrict ext, int bits) {
21 lmmp_param_assert(bits <= LIMB_BITS && bits > 0);
22 lmmp_param_assert(n > 0);
23 if (n == 1) {
24 int lb = lmmp_limb_bits_(num[0]);
25 if (lb <= bits) {
26 *ext = num[0];
27 return 0;
28 } else {
29 *ext = num[0] >> (lb - bits);
30 return lb - bits;
31 }
32 } else {
33 int lb = lmmp_limb_bits_(num[n - 1]);
34 if (lb < bits) {
35 *ext = num[n - 1] << (bits - lb);
36 *ext |= num[n - 2] >> (LIMB_BITS - bits + lb);
37 return LIMB_BITS * (n - 1) - (bits - lb);
38 } else if (lb == bits) {
39 *ext = num[n - 1];
40 return LIMB_BITS * (n - 1);
41 } else {
42 *ext = num[n - 1] >> (lb - bits);
43 return LIMB_BITS * (n - 1) + (lb - bits);
44 }
45 }
