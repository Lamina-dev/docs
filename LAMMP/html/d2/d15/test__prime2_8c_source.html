234/* notrial 变体：判定集合与全函数精确一致（试除仅承担合数侧廉价早拒，
235 不改变结论）；含小因子合数（MR 侧应同样拒绝）与素数两侧 */
236static void check_prime_2_agreement(mp_limb_t lo, mp_limb_t hi, const char *what) {
237 int full = lmmp_is_prime_2_(lo, hi);
238 int nt = lmmp_is_prime_2_notrial_(lo, hi);
239 TEST_CHECK_MSG(nt == full, what);
240}

242TEST_CASE("numth/prime", is_prime_2_notrial) {
243 /* 边界与分档代表（含偶数拒、psi_12/13 邻域、BPSW 档素数） */
244 check_prime_2_agreement(0x0000000000000000ull, 0x0000000000000001ull, "2^64 even");
245 check_prime_2_agreement(0x000000000000000dull, 0x0000000000000001ull, "2^64+13");
246 check_prime_2_agreement(0xffffffffffffffffull, 0x7fffffffffffffffull, "2^127-1");
247 check_prime_2_agreement(0xffffffffffffff61ull, 0xffffffffffffffffull, "2^128-159");
248 check_prime_2_agreement(0x51adc5b22410a5fdull, 0x000000000002be69ull, "psi_13");
249 check_prime_2_agreement(0x51adc5b22410a68bull, 0x000000000002be69ull, "first prime > SW");
250 /* 小因子合数：3|n、5|n 与 2^64+1（17 整除），notrial 走 MR 同拒 */
251 check_prime_2_agreement(0xffffffffffffffffull, 0xffffffffffffffffull, "2^128-1 (3 divides)");
252 check_prime_2_agreement(0x0000000000000001ull, 0x0000000000000001ull, "2^64+1 Fermat");
253 check_prime_2_agreement(0x0000000000000005ull, 0x0000000000000003ull, "3*2^64+5 (5 divides)");
254 /* 随机奇数流：全量对照（确定性期望经 ref_is_prime64 不适用 128bit，
255 以两变体互证为主，位宽扫 65..128 若干代表点） */
256 u64 seed = 0x6e6f747269616c32ull;
257 const uint HB[] = {1, 3, 17, 63, 64};
258 for (size_t hb_index = 0; hb_index < ARRAY_SIZE(HB); ++hb_index) {
259 uint hb = HB[hb_index];
260 for (int i = 0; i < 200; ++i) {
261 mp_limb_t lo = xorshift64(&seed), hi = xorshift64(&seed);
262 if (hb < 64) hi &= ((mp_limb_t)-1) >> (64 - hb);
263 hi |= (mp_limb_t)1 << (hb - 1);
264 lo |= 1;
265 check_prime_2_agreement(lo, hi, "random odd agree");
266 }
267 }
