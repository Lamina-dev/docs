32mp_size_t lmmp_to_decimal_len_(mp_srcptr numa, mp_size_t na) {
33 int mslbits = 0;
34 if (numa) {
35 do {
36 if (na == 0)
37 return 1;
38 } while (numa[--na] == 0);
39 mslbits = lmmp_limb_bits_(numa[na]);
40 }
41 return lmmp_mulh_(na * LIMB_BITS + mslbits, lmmp_bases_table[DEC_BASE - 2].inv_lg_base) + 1;
42}

53// 初始化第 0 层：10^19 = lmmp_bases_table[DEC_BASE-2].large_base
54static void dec_pow_init_(void) {
55 mp_basepow_t* p0 = &dec_pow_ctx_g.pow[0];
56 p0->p = (mp_ptr)lmmp_alloc(sizeof(mp_limb_t));
57 p0->p[0] = lmmp_bases_table[DEC_BASE - 2].large_base;
58 p0->np = 1;
59 p0->zeros = 0;
60 p0->digits = lmmp_bases_table[DEC_BASE - 2].digits_in_limb;
61 int cnt = lmmp_leading_zeros_(p0->p[0]);
62 if ((p0->norm_cnt = cnt))
63 lmmp_shl_(p0->p, p0->p, 1, cnt);
64 p0->invp = NULL;
65 p0->ni = 0;
66 p0->base = DEC_BASE;
67 dec_pow_ctx_g.top = 0;
68}

70/*
71 * 由第 k-1 层平方构建第 k 层：pow[k] = pow[k-1]^2 = 10^(19*2^k)。
72 * 设第 k-1 层真值 T = v * B^zeros（v 为剥离尾部零 limb 后的奇 limb 数），
73 * v 的尾零 bit 数 = 19*2^(k-1) mod 64 < 64，故 v^2 的尾零 limb 至多 1 个，
74 * 剥离后新 zeros = 2*zeros + 剥离数 = floor(19*2^k / 64)。
75 */
76static void dec_pow_extend_(int k) {
77 mp_basepow_t* pre = &dec_pow_ctx_g.pow[k - 1];
78 mp_size_t np = pre->np;
79 TEMP_B_DECL;
80 // buf 布局：[v: np（非归一化真值）| sq: 2np（平方结果）]
81 mp_ptr buf = BALLOC_TYPE(3 * np + 2, mp_limb_t);
82 mp_ptr v = buf, sq = buf + np;
83
84 lmmp_copy(v, pre->p, np);
85 if (pre->norm_cnt)
86 lmmp_shr_(v, v, np, pre->norm_cnt);
87 lmmp_sqr_(sq, v, np);
88
89 mp_size_t nq = 2 * np;
90 nq -= sq[nq - 1] == 0;
91 mp_size_t zeros = 2 * pre->zeros;
92 while (sq[0] == 0) {
93 ++zeros;
94 ++sq;
95 --nq;
96 }
97
98 mp_basepow_t* cur = &dec_pow_ctx_g.pow[k];
99 cur->p = (mp_ptr)lmmp_alloc((size_t)nq * sizeof(mp_limb_t));
100 lmmp_copy(cur->p, sq, nq);
101 int cnt = lmmp_leading_zeros_(cur->p[nq - 1]);
102 if ((cur->norm_cnt = cnt))
103 lmmp_shl_(cur->p, cur->p, nq, cnt);
104 cur->np = nq;
105 cur->zeros = zeros;
106 cur->digits = (mp_size_t)lmmp_bases_table[DEC_BASE - 2].digits_in_limb << k;
107 cur->base = DEC_BASE;
108 if (nq < DIV_MULINV_L_THRESHOLD) {
109 // 规模较小，使用基础除法，无需预计算逆元
110 cur->invp = NULL;
111 cur->ni = 0;
112 } else {
113 mp_size_t ni = lmmp_div_inv_size_(nq + zeros, nq);
114 cur->invp = (mp_ptr)lmmp_alloc((size_t)ni * sizeof(mp_limb_t));
115 lmmp_inv_prediv_(cur->invp, cur->p, nq, ni);
116 cur->ni = ni;
117 }
118
119 TEMP_B_FREE;
120}

122mp_basepow_t* lmmp_dec_pow_table_(mp_size_t grps, int* ctop) {
123 lmmp_param_assert(grps > 1 && ctop != NULL);
124 if (dec_pow_ctx_g.top < 0)
125 dec_pow_init_();
126 // 最高有效层：2^need <= grps，即 need = floor(log2(grps))
127 int need = lmmp_limb_bits_((mp_limb_t)grps) - 1;
128 while (dec_pow_ctx_g.top < need) {
129 ++dec_pow_ctx_g.top;
130 dec_pow_extend_(dec_pow_ctx_g.top);
131 }
132 *ctop = need;
133 return dec_pow_ctx_g.pow;
134}

136void lmmp_dec_pow_table_free_(void) {
137 if (dec_pow_ctx_g.top < 0)
138 return;
139 for (int i = 0; i <= dec_pow_ctx_g.top; ++i) {
140 lmmp_free(dec_pow_ctx_g.pow[i].p);
141 if (dec_pow_ctx_g.pow[i].invp)
142 lmmp_free(dec_pow_ctx_g.pow[i].invp);
143 }
144 dec_pow_ctx_g.top = -1;
145}

147mp_size_t lmmp_to_decimal_(char* dst, mp_srcptr numa, mp_size_t na) {
148 lmmp_param_assert(dst != NULL && numa != NULL);
149 do {
150 if (na == 0)
151 return 0;
152 } while (numa[--na] == 0);
153 ++na;
154
155 if (na < TO_STR_BASEPOW_THRESHOLD)
156 return lmmp_to_str_basecase_((mp_byte_t*)dst, numa, na, DEC_BASE, '0');
157
158 // 惰性获取全局幂表
159 mp_size_t digits = lmmp_to_str_len_(numa, na, DEC_BASE);
160 mp_size_t grps = (digits - 1) / lmmp_bases_table[DEC_BASE - 2].digits_in_limb + 1;
161 int ctop;
162 mp_basepow_t* pows = lmmp_dec_pow_table_(grps, &ctop);
163
164 TEMP_B_DECL;
165 // numa 拷贝空间（多 1 limb 预留规整移位）+ 各层商空间
166 mp_size_t alloc = na + 1;
167 for (int i = 1; i <= ctop; ++i)
168 alloc += pows[i].np + pows[i].zeros + 1;
169 mp_ptr tp = BALLOC_TYPE(alloc, mp_limb_t);
170
171 lmmp_copy(tp, numa, na);
172 digits = lmmp_to_str_divide_((mp_byte_t*)dst, tp, na, pows + ctop, tp + na + 1, '0');
173
174 TEMP_B_FREE;
175 return digits;
176}
