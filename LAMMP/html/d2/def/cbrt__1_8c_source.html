40ulong lmmp_cbrt_chebyshev_(ulong n) {
41 lmmp_param_assert(n > 0);
42 typedef union {
43 ulong uword_val;
44 double double_val;
45 } uni;
46
47 int rem, mul;
48 double factor, root, dec, dec2;
49 ulong ret, expo, table_index;
50 uni alias;
51
52 /* upper_limit is the max cube root possible for one word */
53
54 const ulong upper_limit = 2642245; /* 2642245 < (2^64)^(1/3) */
55 const ulong expo_mask = 0x7FF0000000000000; /* exponent bits in double */
56 const ulong mantissa_mask = 0x000FFFFFFFFFFFFF; /* mantissa bits in float */
57 const ulong table_mask = 0x000F000000000000; /* first 4 bits of mantissa */
58 const uint mantissa_bits = 52;
59 const ulong bias_hex = 0x3FE0000000000000;
60 const uint bias = 1022;
61 alias.double_val = (double)n;
62
63 expo = alias.uword_val & expo_mask; /* extracting exponent */
64 expo >>= mantissa_bits;
65 expo -= bias; /* Subtracting bias */
66
67 /* extracting first 4 bits of mantissa, this will help select correct poly */
68 /* note mantissa of 0.5 is 0x0000000000000 not 0x1000000000000 */
69
70 table_index = alias.uword_val & table_mask;
71 table_index >>= (mantissa_bits - 4);
72
73 /* extracting decimal part, 0.5 <= dec <= 1 */
74 ret = alias.uword_val & mantissa_mask;
75 ret |= bias_hex;
76 alias.uword_val = ret;
77 dec = alias.double_val;
78
79 rem = expo % 3;
80 expo /= 3;
81 factor = factor_table[rem];
82
83 /* Calculating cube root of dec using chebyshev approximation polynomial */
84
85 dec2 = dec * dec;
86 root = (coeff[table_index][0] + coeff[table_index][1] * dec);
87 root += (coeff[table_index][2] * dec2);
88
89 mul = (ulong)1 << expo; /* mul = 2^expo */
90 root *= mul; /* dec^(1/3) * 2^(expo/3) */
91 root *= factor; /* root*= (expo%3)^(1/3) */
92 ret = root;
93
94 /* In case ret^3 or (ret+1)^3 will cause overflow */
95
96 if (ret >= upper_limit) {
97 if (n >= upper_limit * upper_limit * upper_limit)
98 return upper_limit;
99 ret = upper_limit - 1;
100 }
101 while (ret * ret * ret <= n) {
102 (ret) += 1;
103 if (ret == upper_limit)
104 break;
105 }
106 while (ret * ret * ret > n) (ret) -= 1;
107
108 return ret;
109}

111static inline double lmmp_cbrt_estimate(double a) {
112 typedef union {
113 slong uword_val;
114 double double_val;
115 } uni;
116
117 uni alias;
118 ulong n;
119
120 const ulong mul_factor = (ulong)6148914691236517205;
121 slong s = (slong)4607182418800017408; /* ((1 << 10) - 1) << 52 */
122
123 alias.double_val = a;
124 n = alias.uword_val;
125 n -= s;
126 n = lmmp_mulh_(n, mul_factor);
127 n += s;
128 alias.uword_val = n;
129 return alias.double_val;

132ulong lmmp_cbrt_ulong_(ulong n) {
133 double val, x, xcub, num, den;
134 ulong ret, upper_limit;
135
136 if (n < 125)
137 return (n >= 1) + (n >= 8) + (n >= 27) + (n >= 64);
138 if (n < 1331)
139 return 5 + (n >= 216) + (n >= 343) + (n >= 512) + (n >= 729) + (n >= 1000);
140 if (n < 4913)
141 return 11 + (n >= 1728) + (n >= 2197) + (n >= 2744) + (n >= 3375) + (n >= 4096);
142
143 val = (double)n;
144
145 if (n >= 1ULL << 46)
146 return lmmp_cbrt_chebyshev_(n);
147
148 upper_limit = 2642245; /* 2642245 = floor((2^64)^(1/3)) */
149
150 x = lmmp_cbrt_estimate((double)n);
151
152 /* Kahan's iterations to get cube root */
153
154 xcub = x * x * x;
155 num = (xcub - val) * x;
156 den = (xcub + xcub + val);
157 x -= (num / den);
158 ret = x;
159
160 if (ret >= upper_limit) {
161 if (n >= upper_limit * upper_limit * upper_limit)
162 return upper_limit;
163 ret = upper_limit - 1;
164 }
165 while (ret * ret * ret <= n) {
166 (ret) += 1;
167 if (ret == upper_limit)
168 break;
169 }
170 while (ret * ret * ret > n) (ret) -= 1;
171
172 return ret;
173}
