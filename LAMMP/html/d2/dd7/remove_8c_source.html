22/*
23事实上，我们可以知道，[np,nn]至多为[dp,dn]的2^64次方
242^64 - 1 - 2 - 4 - 8 - ... - 2^63 最为接近2^64
25故，理论上MAX_EXP = 63
26
27但是，实际上，[np,nn]几乎不可能超过[dp,dn]的2^48次方
282^48 - 1 - 2 - 4 - 8 - ... - 2^47 最为接近2^48
29故，我们使用MAX_EXP = 48
30*/
31#define MAX_EXP 48

33/**
34 * qp 为商，rp 为余数，divp 为被除数，divn 为被除数长度，numb 为除数，nb 为除数长度
35 * 如果无法整除，则返回0，否则返回除数 qp 的长度
36 */
37static inline mp_size_t try_div_(mp_ptr qp, mp_ptr rp, mp_srcptr divp, mp_size_t divn, mp_srcptr numb, mp_size_t nb) {
38 if (divn < nb) {
39 return 0;
40 } else {
41 int cmp;
42 if (divn > nb)
43 cmp = 1;
44 else
45 cmp = lmmp_cmp_(divp, numb, nb);
46 if (cmp == 0) {
47 qp[0] = 1;
48 return 1;
49 } else if (cmp < 0) {
50 return 0;
51 } else {
52 // 拷贝原始数，当无法整除时，再进行恢复
53 lmmp_div_(qp, rp, divp, divn, numb, nb);
54 if (lmmp_zero_q_(rp, nb)) {
55 // 整除
56 divn = divn - nb + 1;
57 while (divn > 0 && qp[divn - 1] == 0) --divn;
58 return divn;
59 } else {
60 // 无法整除
61 return 0;
62 }
63 }
64 }
65}

67mp_size_t lmmp_remove_(mp_ptr np, mp_size_t* restrict nn, mp_srcptr dp, mp_size_t dn) {
68 lmmp_param_assert(np != NULL && *nn > 0);
69 lmmp_param_assert(dp != NULL && dn > 0);
70
71 mp_srcptr pd_pow[MAX_EXP];
72 mp_size_t pn_pow[MAX_EXP];
73 // qp as quotient, rp as remainder, divp as divdend
74 mp_ptr qp, rp, divp;
75 mp_ptr prod;
76 mp_size_t divn = *nn, qn;
77 mp_size_t ret = 0;
78 int i, j;
79 pd_pow[0] = dp;
80 pn_pow[0] = dn;
81
82 TEMP_DECL;
83
84 qp = TALLOC_TYPE(*nn, mp_limb_t);
85 rp = TALLOC_TYPE(*nn, mp_limb_t);
86
87 divp = np;
88
89 for (i = 1; i < MAX_EXP; i++) {
90 // if qn == 0, means cannot be divided by pd_pow[i - 1]
91 if ((qn = try_div_(qp, rp, divp, divn, pd_pow[i - 1], pn_pow[i - 1]))) {
92 divn = qn;
93 LMMP_SWAP(qp, divp, mp_ptr);
94
95 ret += (mp_size_t)1 << (i - 1);
96
97 // 先计算下一次试除需要的 d^(2^i) 及其实际 limb 长度，
98 // 再判断是否继续。不能用 2*pn 代替实际长度。
99 prod = TALLOC_TYPE(2 * pn_pow[i - 1], mp_limb_t);
100 lmmp_sqr_(prod, pd_pow[i - 1], pn_pow[i - 1]);
101 pn_pow[i] = 2 * pn_pow[i - 1];
102 pn_pow[i] -= (prod[pn_pow[i] - 1] == 0) ? 1 : 0;
103 pd_pow[i] = prod;
104
105 if (divn < pn_pow[i]) {
106 ++i;
107 break;
108 }
109 } else {
110 break;
111 }
112 }
113 for (j = i - 1; j > 0; --j) {
114 if ((qn = try_div_(qp, rp, divp, divn, pd_pow[j - 1], pn_pow[j - 1]))) {
115 divn = qn;
116 LMMP_SWAP(qp, divp, mp_ptr);
117
118 ret += (mp_size_t)1 << (j - 1);
119 }
120 }
121 if (qn == 0) {
122 // 无法整除：恢复为最后一次试除前的被除数
123 lmmp_copy(np, divp, divn);
124 *nn = divn;
125 } else {
126 // 整除：商在最后一次 swap 后的 divp 中
127 lmmp_copy(np, divp, qn);
128 *nn = qn;
129 }
130
131 TEMP_FREE;
132 return ret;
133}
