23/*
24 算法阈值边界回归（无 BigInt oracle；模式同 test_mul_toom68）：
25 阈值 ±1 两侧的多路径互比 + basecase 精确对拍 + 模检查。
26
27 - mul_n_：toom22/33/44 阈值 ±1 经完整分派，与 mul_basecase_

45// 32 位素数模幂参考
46static u64 modpow32(u64 base, u64 exp, u32 p) {
47 u64 r = 1 % p, a = base % p;
48 while (exp) {
49 if (exp & 1) r = r * a % p;
50 a = a * a % p;
51 exp >>= 1;
52 }
53 return r;
54}

56static void fill_deterministic(mp_ptr p, mp_size_t n, u64 seed) {
57 for (mp_size_t i = 0; i < n; ++i) p[i] = xorshift64(&seed);
58 if (n > 0) p[n - 1] |= (u64)1 << 63;
59}

61// mul_n 阈值边界：与 basecase 逐 limb 精确一致 + 乘积模检查
62static void check_mul_n(mp_size_t n, u64 seed) {
63 mp_ptr a = alloc_limbs(n), b = alloc_limbs(n), d = alloc_limbs(2 * n + 2),
64 ref = alloc_limbs(2 * n + 2);
65 fill_deterministic(a, n, seed);
66 fill_deterministic(b, n, seed ^ 0x9e3779b97f4a7c15ull);
67
68 lmmp_mul_n_(d, a, b, n);
69 lmmp_mul_basecase_(ref, a, n, b, n);
70 bool ok = memcmp(d, ref, (size_t)(2 * n) * 8) == 0;
71 ok = ok && mod_is_product(d, 2 * n, a, n, b, n);
72 TEST_CHECK_MSG(ok, "mul_n at threshold boundary");
73 lmmp_free(a);
74 lmmp_free(b);
75 lmmp_free(d);
76 lmmp_free(ref);
77}

116TEST_CASE("threshold/div", divide_threshold_boundary) {
117 const mp_size_t ns[] = {DIV_DIVIDE_THRESHOLD > 1 ? DIV_DIVIDE_THRESHOLD - 1 : 1, DIV_DIVIDE_THRESHOLD,
118 DIV_DIVIDE_THRESHOLD + 1};
119 u64 seed = 0x0123456789abcdefull;
120 for (size_t n_index = 0; n_index < ARRAY_SIZE(ns); ++n_index) {
121 mp_size_t n = ns[n_index];
122 const mp_size_t na = 2 * n;
123 mp_ptr num = alloc_limbs(na), den = alloc_limbs(n), q = alloc_limbs(n + 2),
124 r = alloc_limbs(n + 2), t = alloc_limbs(2 * n + 4), qd = alloc_limbs(na + 2),
125 sum = alloc_limbs(2 * n + 4);
126 fill_deterministic(num, na, seed);
127 fill_deterministic(den, n, seed ^ 0xfedcba9876543210ull);
128 num[na - 1] &= ~((u64)1 << 63); /* 保证 q 不超过 n+1 limbs */
129
130 lmmp_div_(q, r, num, na, den, n);
131 // q*d + r == num（库乘重建 + 零垫对齐加）
132 lmmp_mul_(qd, q, n + 1, den, n);
133 lmmp_zero(sum, 2 * n + 4);
134 memcpy(sum, qd, (size_t)(2 * n + 1) * 8);
135 lmmp_zero(t, 2 * n + 4);
136 memcpy(t, r, (size_t)n * 8);
137 lmmp_add_n_(sum, sum, t, 2 * n + 4);
138 TEST_CHECK_MSG(mpn_eq(sum, 2 * n + 4, num, na), "div q*d+r rebuild at boundary");
139 // 余数界：r < den
140 TEST_CHECK_MSG(mpn_cmp(r, n, den, n) < 0, "div rem < den at boundary");
141
142 lmmp_free(num);
143 lmmp_free(den);
144 lmmp_free(q);
145 lmmp_free(r);
146 lmmp_free(t);
147 lmmp_free(qd);
148 lmmp_free(sum);
149 }

152TEST_CASE("threshold/divexact", divexact_two_threshold_boundaries) {
153 u64 seed = 0x0fedcba987654321ull;
154 typedef struct {
155 mp_size_t nn, dn;
156 } Case;
157 const Case cases[] = {
158 {DIVEXACT_NN_THRESHOLD > 1 ? DIVEXACT_NN_THRESHOLD - 1 : 1,
159 DIVEXACT_BASECASE_THRESHOLD > 1 ? DIVEXACT_BASECASE_THRESHOLD - 1 : 1},
160 {DIVEXACT_NN_THRESHOLD > 1 ? DIVEXACT_NN_THRESHOLD - 1 : 1, DIVEXACT_BASECASE_THRESHOLD},
161 {DIVEXACT_NN_THRESHOLD, DIVEXACT_BASECASE_THRESHOLD > 1 ? DIVEXACT_BASECASE_THRESHOLD - 1 : 1},
162 {DIVEXACT_NN_THRESHOLD, DIVEXACT_BASECASE_THRESHOLD},
163 {DIVEXACT_NN_THRESHOLD + 1, DIVEXACT_BASECASE_THRESHOLD + 1},
164 };
165 for (size_t c_index = 0; c_index < ARRAY_SIZE(cases); ++c_index) {
166 const Case c = cases[c_index];
167 const mp_size_t qn = c.nn - c.dn;
168 mp_ptr q0 = alloc_limbs(qn), dp = alloc_limbs(c.dn), np = alloc_limbs(c.nn + 2),
169 dst = alloc_limbs(c.nn + 2);
170 fill_deterministic(q0, qn, seed);
171 fill_deterministic(dp, c.dn, seed ^ 0xa5a5a5a55a5a5a5aull);
172 dp[0] |= 1u;
173 dp[c.dn - 1] |= (u64)1 << 63;
174
175 if (qn >= c.dn)
176 lmmp_mul_(np, q0, qn, dp, c.dn);
177 else
178 lmmp_mul_(np, dp, c.dn, q0, qn);
179 np[c.nn] = 0;
180
181 lmmp_divexact_(dst, np, c.nn, dp, c.dn);
182 TEST_CHECK_MSG(mpn_eq(dst, qn, q0, qn),
183 "divexact boundary recovers original quotient");
184
185 lmmp_free(q0);
186 lmmp_free(dp);
187 lmmp_free(np);
188 lmmp_free(dst);
189 }
190}

192TEST_CASE("threshold/pow", pow_1_exp_boundary) {
193 const mp_limb_t base = 3;
194 mp_ptr b = alloc_limbs(1);
195 b[0] = base;
196 const ulong exps[] = {POW_1_EXP_THRESHOLD > 1 ? (ulong)POW_1_EXP_THRESHOLD - 1 : 1,
197 (ulong)POW_1_EXP_THRESHOLD,
198 (ulong)POW_1_EXP_THRESHOLD + 1, 1ul};
199 for (size_t e_index = 0; e_index < ARRAY_SIZE(exps); ++e_index) {
200 ulong e = exps[e_index];
201 const mp_size_t rn = lmmp_pow_size_(b, 1, e);
202 mp_ptr dst = alloc_limbs(rn + 2), dst1 = alloc_limbs(rn + 2);
203 // 多 limb 入口（单 limb 底）与 pow_1_ 精确一致
204 const mp_size_t got = lmmp_pow_(dst, rn, b, 1, e);
205 const mp_size_t got1 = lmmp_pow_1_(dst1, rn, base, e);
206 TEST_CHECK_MSG(got == got1 && memcmp(dst, dst1, (size_t)got * 8) == 0,
207 "pow_ vs pow_1_ at exp boundary");
208 // 模幂参考（32 位素数）
209 bool ok = true;
210 for (int i = 0; i < MOD_P_CNT && ok; ++i)
211 ok = mod32(dst, got, MOD_P[i]) == modpow32(base, e, MOD_P[i]);
212 TEST_CHECK_MSG(ok, "pow boundary modpow check");
213 lmmp_free(dst);
214 lmmp_free(dst1);
215 }
216 lmmp_free(b);
217}
