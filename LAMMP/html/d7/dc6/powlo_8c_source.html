60static inline mp_limb_t getbits(const mp_limb_t* p, mp_bitcnt_t bi, mp_bitcnt_t nbits) {
61 mp_bitcnt_t nbits_in_r;
62 mp_limb_t r;
63 mp_size_t i;
64
65 if (bi <= nbits) {
66 return p[0] & (((mp_limb_t)1 << bi) - 1);
67 } else {
68 bi -= nbits; /* bit index of low bit to extract */
69 i = bi / LIMB_BITS; /* word index of low bit to extract */
70 bi %= LIMB_BITS; /* bit index in low word */
71 r = p[i] >> bi; /* extract (low) bits */
72 nbits_in_r = LIMB_BITS - bi; /* number of bits now in r */
73 if (nbits_in_r < nbits) /* did we get enough bits? */
74 r += p[i + 1] << nbits_in_r; /* prepend bits from higher word */
75 return r & (((mp_limb_t)1 << nbits) - 1);
76 }
77}

83void lmmp_powlo_(mp_ptr restrict dst, mp_srcptr restrict bp, mp_size_t n, mp_srcptr restrict ep, mp_size_t en) {
84 lmmp_param_assert(ep[en - 1] > 0);
85 lmmp_param_assert(n > 0 && en > 0);
86 lmmp_param_assert(dst != NULL && bp != NULL && ep != NULL);
87 mp_bitcnt_t cnt, ebi;
88 unsigned windowsize, this_windowsize;
89 mp_limb_t expbits;
90 mp_limb_t* pp;
91 long i;
92 int flipflop;
93 TEMP_DECL;
94 mp_ptr restrict tp = TALLOC_TYPE(5 * n, mp_limb_t);
95 mp_ptr restrict scratch = tp + 3 * n;
96
97 ebi = count_bits(ep, en);
98
99 windowsize = win_size(ebi);
100 if (windowsize > 1) {
101 mp_limb_t *this_pp, *last_pp;
102 lmmp_debug_assert(windowsize < ebi);
103
104 pp = TALLOC_TYPE((n << (windowsize - 1)), mp_limb_t);
105
106 this_pp = pp;
107
108 lmmp_copy(this_pp, bp, n);
109
110 /* Store b^2 in tp. */
111 lmmp_sqrlo_n_(tp, bp, n, scratch);
112
113 /* Precompute odd powers of b and put them in the temporary area at pp. */
114 i = (1 << (windowsize - 1)) - 1;
115 do {
116 last_pp = this_pp;
117 this_pp += n;
118 lmmp_mullo_n_(this_pp, last_pp, tp, n, scratch);
119 } while (--i != 0);
120
121 expbits = getbits(ep, ebi, windowsize);
122 ebi -= windowsize;
123
124 ctz_shr_u64(expbits, expbits, cnt);
125 ebi += cnt;
126
127 lmmp_copy(dst, pp + n * (expbits >> 1), n);
128 } else {
129 pp = tp + n;
130 lmmp_copy(pp, bp, n);
131 lmmp_copy(dst, bp, n);
132 --ebi;
133 }
134
135 flipflop = 0;
136
137 do {
138 while (getbit(ep, ebi) == 0) {
139 lmmp_sqrlo_n_(tp, dst, n, scratch);
140 LMMP_SWAP(dst, tp, mp_ptr);
141 flipflop = !flipflop;
142 if (--ebi == 0)
143 goto done;
144 }
145
146 /* The next bit of the exponent is 1. Now extract the largest block of
147 bits <= windowsize, and such that the least significant bit is 1. */
148
149 expbits = getbits(ep, ebi, windowsize);
150 this_windowsize = LMMP_MIN(windowsize, ebi);
151
152 ctz_shr_u64(expbits, expbits, cnt);
153 this_windowsize -= cnt;
154 ebi -= this_windowsize;
155
156 while (this_windowsize > 1) {
157 lmmp_sqrlo_n_(tp, dst, n, scratch);
158 lmmp_sqrlo_n_(dst, tp, n, scratch);
159 this_windowsize -= 2;
160 }
161
162 if (this_windowsize != 0)
163 lmmp_sqrlo_n_(tp, dst, n, scratch);
164 else {
165 LMMP_SWAP(dst, tp, mp_ptr);
166 flipflop = !flipflop;
167 }
168 lmmp_mullo_n_(dst, tp, pp + n * (expbits >> 1), n, scratch);
169 } while (ebi != 0);
170
171done:
172 if (flipflop)
173 lmmp_copy(tp, dst, n);
174 TEMP_FREE;
175}
