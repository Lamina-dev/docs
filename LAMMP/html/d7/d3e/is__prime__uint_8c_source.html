32static inline ulong urem32_preinv(ulong t, ulong n, ulong dinv) {
33 ulong q = _umul64to64hi_(t, dinv);
34 ulong r = t - q * n;
35 if (r >= n) r += n;
36 return r;
37}

39static void powmod2b_u32(uint b, uint d, int ebits, uint n, ulong dinv, uint* py2, uint* pyb) {
40 ulong xb = b;
41 if (xb >= n)
42 xb = urem32_preinv(xb, n, dinv);
43
44 if (ebits <= 6) {
45 ulong x2 = urem32_preinv((ulong)1 << d, n, dinv);
46 for (int k = ebits - 2; k >= 0; k--) {
47 xb = urem32_preinv(xb * xb, n, dinv);
48 if ((d >> k) & 1)
49 xb = urem32_preinv(xb * b, n, dinv);
50 }
51 *py2 = (uint)x2;
52 *pyb = (uint)xb;
53 return;
54 }
55
56 ulong x2 = urem32_preinv((ulong)1 << (d >> (ebits - 6)), n, dinv);
57 for (int k = ebits - 2; k >= ebits - 6; k--) {
58 xb = urem32_preinv(xb * xb, n, dinv);
59 if ((d >> k) & 1)
60 xb = urem32_preinv(xb * b, n, dinv);
61 }
62 if (n < (1u << 31)) {
63 for (int k = ebits - 7; k >= 0; k--) {
64 x2 = urem32_preinv(x2 * x2, n, dinv);
65 x2 <<= (d >> k) & 1;
66 xb = urem32_preinv(xb * xb, n, dinv);
67 if ((d >> k) & 1)
68 xb = urem32_preinv(xb * b, n, dinv);
69 }
70 if (x2 >= n) x2 -= n;
71 } else {
72 for (int k = ebits - 7; k >= 0; k--) {
73 x2 = urem32_preinv(x2 * x2, n, dinv);
74 x2 <<= (d >> k) & 1;
75 if (x2 >= n) x2 -= n;
76 xb = urem32_preinv(xb * xb, n, dinv);
77 if ((d >> k) & 1)
78 xb = urem32_preinv(xb * b, n, dinv);
79 }
80 }
81 *py2 = (uint)x2;
82 *pyb = (uint)xb;
83}

85static int sprp_check_u32(uint y, ulong t, uint n, ulong dinv) {
86 if (y == 1 || y == n - 1)
87 return 1;
88 for (t--; t > 0; t--) {
89 ulong v = y;
90 v = urem32_preinv(v * v, n, dinv);
91 y = (uint)v;
92 if (y == n - 1) return 1;
93 if (y == 1) return 0;
94 }
95 return 0;
96}

98LMMP_NOINLINE bool lmmp_is_prime_uint_notrial_(uint n) {
99 lmmp_param_assert(n > 2);
100 lmmp_param_assert(n % 2 == 1);
101
102 uint base = dj_base49[((0x3AC69A35UL * n) & 0xFFFFFFFFUL) >> 21] + 3;
103 if (n % base == 0 || base % n == 0) {
104 if (n > base)
105 return false;
106 for (uint p = 3; p * p <= n; p += 2)
107 if (n % p == 0) return false;
108 return true;
109 }
110
111 ulong dinv = MP_ULONG_MAX / n + 1;
112 ulong d;
113 ulong t;
114 ctz_shr_u64(d, n - 1, t);
115 int ebits = lmmp_limb_bits_(d);
116
117 uint y2, yb;
118 powmod2b_u32(base, (uint)d, ebits, n, dinv, &y2, &yb);
119 if (!sprp_check_u32(y2, t, n, dinv))
120 return false;
121 return sprp_check_u32(yb, t, n, dinv);
122}

124bool lmmp_is_prime_uint_(uint n) {
125 if (n == 2)
126 return true;
127 if (n % 2 == 0 || n <= 1)
128 return false;
129 int judge = lmmp_is_prime_table_(n);
130 if (judge == 0) {
131 return false;
132 } else if (judge == 1) {
133 return true;
134 }
135
136 if (trial_div35711(n))
137 return false;
138
139 return lmmp_is_prime_uint_notrial_(n);
140}
