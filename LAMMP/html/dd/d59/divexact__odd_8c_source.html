27mp_limb_t lmmp_divexact_odd_pi_(mp_ptr dst, mp_srcptr numa, mp_size_t na, mp_limb_t d, mp_limb_t di, unsigned shr) {
28 mp_limb_t c, l, u, un, lo;
29 mp_size_t i;
30
31 lmmp_param_assert(na > 0);
32 lmmp_param_assert((d & 1) != 0);
33 lmmp_param_assert(shr < LIMB_BITS);
34 lmmp_param_assert(d * di == 1);
35
36 if (shr != 0) {
37 c = 0;
38 u = numa[0];
39 for (i = 1; i < na; i++) {
40 un = numa[i];
41 u = (u >> shr) | (un << (LIMB_BITS - shr));
42 l = u - c;
43 c = l > u;
44 l *= di;
45 dst[i - 1] = l;
46 u = un;
47 _umul64to128_(l, d, &lo, &un);
48 c += un;
49 }
50 u >>= shr;
51 l = u - c;
52 c = l > u;
53 l *= di;
54 dst[na - 1] = l;
55 } else {
56 c = 0;
57 i = 0;
58 do {
59 u = numa[i];
60 l = u - c;
61 c = l > u;
62 l *= di;
63 dst[i] = l;
64 _umul64to128_(l, d, &lo, &un);
65 c += un;
66 } while (++i < na);
67 }
68 return c;
69}
