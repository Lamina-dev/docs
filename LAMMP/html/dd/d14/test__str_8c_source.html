23/*
24 字符串转换验证（无 BigInt oracle）。库约定：每字节一个 digit 值
25 （0..base-1），低位在前（little-endian digit order），零值串长 0。
26
27 1. round-trip：to_str → from_str 值复原（多进制 × 多尺寸）
28 2. div_1 重算数字：被除数逐位 div_1 取余重建 digit 串，与 to_str
29 输出精确一致（div_1 由 div 套件独立保证）
30 3. Horner 重建：digit 串经 mul_1+加法逐位求值，与 from_str 输出
31 精确一致
32 4. 写死值锚点（0/1/10/255 等）
33 5. 特化接口：hex（snprintf 按 limb 重建参考串）/ decimal（与
34 to_str+dec_to_chars / from_str 交叉），幂表生命周期（deinit/init）
35*/

45// 参考 digit 串：div_1 逐位取余（低位在前；独立于 to_str_ 的幂表路径）
46static mp_size_t ref_digits(mp_byte_t *out, mp_srcptr x, mp_size_t n, int base) {
47 mp_size_t len = 0;
48 mp_ptr work = alloc_limbs(n + 1), q = alloc_limbs(n + 1);
49 lmmp_zero(work, n + 1);
50 memcpy(work, x, (size_t)n * 8);
51 mp_size_t wn = norm_len(work, n + 1);
52 while (wn > 0) {
53 mp_limb_t r = lmmp_div_1_(q, work, wn, base);
54 out[len++] = (mp_byte_t)r;
55 memcpy(work, q, (size_t)wn * 8);
56 while (wn > 0 && work[wn - 1] == 0) --wn;
57 }
58 lmmp_free(work);
59 lmmp_free(q);
60 return len;
61}

63// 参考 limb 值：Horner val = val*base + d（高位 digit 先；缓冲须已分配
64// 且本函数全程保持零化）
65static mp_size_t ref_value(mp_ptr val, mp_size_t cap, const mp_byte_t* d, mp_size_t len, int base) {
66 lmmp_zero(val, cap);
67 for (mp_size_t i = len; i-- > 0;) {
68 lmmp_mul_1_(val, val, cap, base); // 高位零垫，cap 内无溢出
69 add_small_inplace(val, cap, d[i]);
70 }
71 return norm_len(val, cap);
72}

74// 顶部 limb 的有效位数
75static int limb_bits_of(u64 x) {
76 int k = 0;
77 while (x) {
78 x >>= 1;
79 ++k;
80 }
81 return k;
82}

84// 参考十六进制串：每 limb "%016llx" 自高向低拼接后整体反转，前 digits
85// 个字符即低位在前的约定串（纯 limb 运算，无大整数）
86static mp_size_t ref_hex_string(char *out, mp_srcptr x, mp_size_t n, bool upper) {
87 for (mp_size_t i = 0; i < n; ++i) {
88 char buf[17];
89 snprintf(buf, sizeof(buf), "%016llx", (unsigned long long)x[i]);
90 for (size_t j = 0; j < 16; ++j) {
91 char c = buf[15 - j];
92 out[16 * i + j] = upper ? (char)toupper((unsigned char)c) : c;
93 }
94 }
95 mp_size_t nlen = norm_len(x, n);
96 mp_size_t bits = nlen ? 64 * (nlen - 1) + limb_bits_of(x[nlen - 1]) : 0;
97 return (bits + 3) / 4;
98}

100static void check_hex_one(mp_srcptr x, mp_size_t n) {
101 char *reference = lmmp_alloc(16 * n + 1);
102 mp_size_t digits = ref_hex_string(reference, x, n, false);
103
104 TEST_CHECK_EQ(lmmp_to_hex_len_(x, n), digits);
105 TEST_CHECK_EQ(lmmp_to_hex_len_(NULL, n), (mp_size_t)(n * 16));
106
107 for (int upper = 0; upper <= 1; ++upper) {
108 char* got = (char*)lmmp_alloc((size_t)digits + 8);
109 mp_size_t gotn = lmmp_to_hex_(got, x, n, (bool)upper);
110 TEST_CHECK_EQ(gotn, digits);
111 ref_hex_string(reference, x, n, (bool)upper);
112 TEST_CHECK_MSG(memcmp(got, reference, digits) == 0, "to_hex vs reference");
113 if (norm_len(x, n) > 0) TEST_CHECK_MSG(got[digits - 1] != '0', "to_hex no leading zero");
114
115 // from_hex：大小写混合 + 末尾补 '0'（值前导零）
116 TEST_CHECK_EQ(lmmp_from_hex_len_(got, gotn), norm_len(x, n));
117
118 char *padded = lmmp_alloc(gotn + 5);
119 for (mp_size_t i = 0; i < gotn; ++i)
120 padded[i] =
121 (i & 1) ? (char)toupper((unsigned char)got[(size_t)i]) : got[(size_t)i];
122
123 memset(padded + gotn, '0', 5);
124 mp_size_t need = lmmp_from_hex_len_(padded, gotn + 5);
125 mp_ptr back = alloc_limbs((size_t)need + 2);
126 mp_size_t backn = lmmp_from_hex_(back, padded, gotn + 5);
127 TEST_CHECK_MSG(mpn_eq(back, backn, x, n), "hex roundtrip");
128 lmmp_free(back);
129 lmmp_free(got);
130 lmmp_free(padded);
131 }
132 lmmp_free(reference);
133}

135static void check_decimal_one(mp_srcptr x, mp_size_t n) {
136 // 参考：to_str_(base 10) 数字节 + lmmp_dec_to_chars_（同为低位在前）
137 mp_size_t reflen = lmmp_to_str_len_(x, n, 10);
138 mp_byte_t* digits = (mp_byte_t*)lmmp_alloc((size_t)reflen + 8);
139 mp_size_t refn = lmmp_to_str_(digits, x, n, 10);
140 lmmp_dec_to_chars_(digits, refn);
141
142 mp_size_t dlen = lmmp_to_decimal_len_(x, n);
143 TEST_CHECK_MSG(dlen >= refn, "to_decimal_len upper bound");
144 char* got = (char*)lmmp_alloc((size_t)dlen + 8);
145 mp_size_t gotn = lmmp_to_decimal_(got, x, n);
146 TEST_CHECK_EQ(gotn, refn);
147 if (gotn == refn)
148 TEST_CHECK_MSG(memcmp(got, digits, (size_t)refn) == 0, "to_decimal vs to_str+chars");
149
150 // 独立数字位检查（大尺寸）：div_1 逐位参考不走 to_decimal 的幂表
151 // 路径；got 与 rd 同为低位在前
152 if (n >= 100) {
153 mp_byte_t *rd = lmmp_alloc(64 * n);
154 mp_size_t ref_digits_len = ref_digits(rd, x, n, 10);
155 TEST_CHECK_EQ(ref_digits_len, gotn);
156 for (mp_size_t i = 0; i < gotn && i < ref_digits_len; ++i)
157 if (rd[(size_t)i] != (mp_byte_t)(got[i] - '0')) {
158 TEST_CHECK_MSG(false, "to_decimal digit vs div_1 reference");
159 break;
160 }
161 lmmp_free(rd);
162 }
163
164 // from：对照 from_str_（数字节）
165 mp_byte_t *digit_vals = digits;
166 for (mp_size_t i = 0; i < refn; ++i) digit_vals[(size_t)i] = (mp_byte_t)(digits[i] - '0');
167 mp_size_t wlen = lmmp_from_str_len_(digit_vals, refn, 10);
168 mp_ptr want = alloc_limbs((size_t)wlen + 2);
169 mp_size_t wantn = lmmp_from_str_(want, digit_vals, refn, 10);
170
171 char *padded = lmmp_alloc(gotn + 3);
172 memcpy(padded, got, gotn);
173 memset(padded + gotn, '0', 3);
174 mp_size_t dneed = lmmp_from_decimal_len_(padded, gotn + 3);
175 TEST_CHECK_MSG(dneed >= wantn, "from_decimal_len upper bound");
176 mp_ptr back = alloc_limbs((size_t)dneed + 2);
177 mp_size_t backn = lmmp_from_decimal_(back, padded, gotn + 3);
178 TEST_CHECK_EQ(backn, wantn);
179 if (backn == wantn)
180 TEST_CHECK_MSG(memcmp(back, want, (size_t)wantn * 8) == 0, "from_decimal vs from_str");
181 TEST_CHECK_MSG(mpn_eq(back, backn, x, n), "decimal roundtrip");
182
183 lmmp_free(back);
184 lmmp_free(want);
185 lmmp_free(padded);
186 lmmp_free(got);
187 lmmp_free(digits);
188}

190TEST_CASE("str/roundtrip", to_str_from_str_roundtrip) {
191 u64 seed = 0xe3459cd6b789abeful;
192 int bases[] = {2, 8, 10, 16, 36, 62, 100, 256};
193 for (size_t base_index = 0; base_index < ARRAY_SIZE(bases); ++base_index) {
194 int base = bases[base_index];
195 const mp_size_t n_values[] = {1, 2, 5, 10, 20, 40};
196 for (size_t n_index = 0; n_index < ARRAY_SIZE(n_values); ++n_index) {
197 mp_size_t n = n_values[n_index];
198 for (int iter = 0; iter < 3; ++iter) {
199 mp_ptr x = alloc_limbs(n);
200 random_limbs(x, n, &seed);
201
202 mp_size_t str_len = lmmp_to_str_len_(x, n, base);
203 mp_byte_t* str = (mp_byte_t*)lmmp_alloc((size_t)str_len + 8);
204 mp_size_t actual = lmmp_to_str_(str, x, n, base);
205 TEST_CHECK_MSG(actual <= str_len, "to_str actual <= len");
206
207 mp_size_t need = lmmp_from_str_len_(str, actual, base);
208 mp_ptr back = alloc_limbs((size_t)need + 2);
209 mp_size_t back_n = lmmp_from_str_(back, str, actual, base);
210 TEST_CHECK_MSG(mpn_eq(back, back_n, x, n), "string roundtrip equality");
211
212 // div_1 重算数字：digit 串精确一致
213 mp_byte_t *rd = lmmp_alloc(64 * n);
214 mp_size_t ref_digits_len = ref_digits(rd, x, n, base);
215 bool dok = ref_digits_len == actual &&
216 memcmp(rd, str, (size_t)actual) == 0;
217 TEST_CHECK_MSG(dok, "to_str digits == div_1 recomputation");
218 lmmp_free(rd);
219
220 // Horner 重建：from_str 值精确一致
221 mp_ptr hv = alloc_limbs((size_t)need + 2);
222 mp_size_t hn = ref_value(hv, need + 2, str, actual, base);
223 TEST_CHECK_MSG(mpn_eq(hv, hn, x, n), "from_str value == Horner rebuild");
224
225 lmmp_free(hv);
226 lmmp_free(back);
227 lmmp_free(str);
228 lmmp_free(x);
229 }
230 }
231 // 零值 round-trip（零值串长 0，from_str 空串回零）
232 mp_ptr z = alloc_limbs(2);
233 lmmp_zero(z, 2);
234 {
235 mp_byte_t zbuf[4];
236 mp_size_t zn = lmmp_to_str_(zbuf, z, 2, base);
237 TEST_CHECK_EQ(zn, (mp_size_t)0);
238 mp_ptr back = alloc_limbs(2);
239 mp_size_t bn = lmmp_from_str_(back, zbuf, zn, base);
240 TEST_CHECK_MSG(bn == 0 || (bn == 1 && back[0] == 0), "zero roundtrip");
241 lmmp_free(back);
242 }
243 lmmp_free(z);
244 }
245}

247static void check_value(mp_srcptr x, mp_size_t n, const mp_byte_t *expect,
248 mp_size_t expected_len, int base) {
249 mp_size_t want = lmmp_to_str_len_(x, n, base);
250 mp_byte_t *str = lmmp_alloc(want + 8);
251 mp_size_t actual = lmmp_to_str_(str, x, n, base);
252 TEST_CHECK_EQ(actual, expected_len);
253 if (actual == expected_len)
254 TEST_CHECK_MSG(memcmp(str, expect, actual) == 0, "known digit string match");
255 lmmp_free(str);
256}
