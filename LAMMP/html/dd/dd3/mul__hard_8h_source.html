89/**
90 * @brief 硬编码平衡乘法分派 [dst,2n] = [numa,n] * [numb,n]
91 * @param dst 结果输出指针
92 * @param numa 第一个乘数
93 * @param numb 第二个乘数
94 * @param n 操作数 limb 长度
95 * @warning 0 < n, sep(dst,[numa|numb]), dst!=NULL, numa!=NULL, numb!=NULL;
96 * n <= LMMP_MUL_HARD_MAX_N 时查表直调硬编码函数, 其余回落 mul_basecase
97 * @return 无
98 */
99static inline void __lmmp_mul_hard_n_(mp_ptr dst, mp_srcptr numa, mp_srcptr numb, mp_size_t n) {
100 if (n <= LMMP_MUL_HARD_MAX_N)
101 lmmp_mul_hard_fns_[n](dst, numa, numb);
102 else
103 lmmp_mul_basecase_(dst, numa, n, numb, n);
104}

107 * @brief 硬编码平衡平方分派 [dst,2n] = [numa,n]^2
108 * @param dst 结果输出指针
109 * @param numa 输入操作数
110 * @param n 操作数 limb 长度
111 * @warning 0 < n, sep(dst,numa), dst!=NULL, numa!=NULL;
112 * n <= LMMP_MUL_HARD_MAX_N 时查表直调硬编码函数, 其余回落 sqr_basecase
113 * @return 无
114 */
115static inline void __lmmp_sqr_hard_n_(mp_ptr dst, mp_srcptr numa, mp_size_t n) {
116 if (n <= LMMP_MUL_HARD_MAX_N)
117 lmmp_sqr_hard_fns_[n](dst, numa);
118 else
119 lmmp_sqr_basecase_(dst, numa, n);
120}
