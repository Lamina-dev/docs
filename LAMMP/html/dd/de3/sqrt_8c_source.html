31/*
32 A = Ah * B^(2*lo) + Al
33
34 Ahr = floor(Ah^(1/2))
35 rk = Ah - Ahr^2
36 x_k = Ahr * B^lo
37
38 x_k+1 = (x_k + A / x_k ) / 2
39 = x_k + (A / x_k - x_k) / 2
40 = x_k + (A - x_k^2) / 2 * x_k
41 = Ahr * B^lo + (rk * B^(2*lo) + Al) / 2 * x_k
42 = Ahr * B^lo + Alr
43
44 let Alr = (rk * B^(2*lo) + Al) / 2 * x_k, R = (rk * B^(2*lo) + Al) mod 2 * x_k
45 such that (rk * B^(2*lo) + Al) = Alr * 2 * x_k + R
46 ┌───────────────────────────────────────────────────────────┐
47 = |Alr * 2 * Ahr*B^lo + R = R_correct + (Alr-1) * 2 * Ahr*B^lo|
48 └───────────────────────────────────────────────────┬───────┘
49 |
50 r_k+1 = A - x_k+1^2 |
51 = Ah*B^(2*lo) + Al - Ahr^2*B^(2*lo) - 2*Alr*Ahr*B^lo - Alr^2 |
52 = r_k * B^(2*lo) + Al - 2*Alr*Ahr*B^lo - Alr^2 |
53 = R - Alr^2 |
54 |
55 Alr is either correct or 1 too big, |
56 ┌────────────────────────────────────┐ |
57 r_k+1 = R - (Alr-1)^2 + | 2*Alr*Ahr*B^lo - 2*(Alr-1)*Ahr*B^lo├────────────────────────┘
58 = R - Alr^2 └────────────────────────────────────┘
59(adjust) + 2*Ahr*B^lo + 2*Alr - 1
60*/

62void lmmp_sqrt_divide_(mp_ptr restrict dst, mp_ptr restrict numa, mp_size_t ns, mp_ptr restrict tp, int calr) {
63 lmmp_param_assert(ns > 0);
64 lmmp_param_assert(numa != NULL && dst != NULL && tp != NULL);
65 lmmp_param_assert(numa[2 * ns - 1] >= LIMB_B_4);
66 if (ns == 1) {
67 dst[0] = lmmp_sqrt_2_(numa, numa);
68 } else if (ns == 2) {
69 lmmp_sqrt_4_(dst, calr ? numa : NULL, numa);
70 } else {
71 mp_size_t lo = ns / 2, hi = ns - lo;
72#define Ahr (dst + lo) // [dst+lo, hi]
73#define rk (numa + 2 * lo) // [numa+2*lo, hi+1]
74#define R (numa) // [numa, ns+1]
75#define Alr (tp) // [tp, lo+1]
76#define Alr2 (tp + lo + 1) // [tp + lo+1, 2*lo]
77
78 lmmp_sqrt_divide_(Ahr, rk, hi, tp, 1);
79
80 /*
81 A / (2*x) = A / 2 / x
82 A % (2*x) = 2 * (A / 2 % x) + A % 2
83 */
84 mp_limb_t r = lmmp_shr_(rk - lo, rk - lo, ns + 1, 1);
85 mp_limb_t qh = lmmp_div_s_(Alr, rk - lo, ns + 1, Ahr, hi);
86 lmmp_debug_assert(qh == 0);
87 (rk - lo)[hi] = lmmp_shl_c_(rk - lo, rk - lo, hi, 1, r >> (LIMB_BITS - 1));
88 /*
89 我们根据 sqrt(A/B^2) == floor(sqrt(A)/B) 可以知道，如果Alr正确结果
90 必定被限制在B^lo以内，其至多高估1，因此Alr的最高位必定为0或1，而为1时，
91 即代表此时结果已经高估。
92 */
93 mp_limb_t adj = Alr[lo];
94 lmmp_debug_assert(adj == 0 || adj == 1);
95 if (adj > 0) {
96 lmmp_debug_assert(Alr[0] == 0);
97 lmmp_fill_n(dst, lo, LIMB_MAX);
98 if (calr == 0) return;
99 /*
100 x_k+1 = Ahr * B^lo + Alr
101 = Ahr * B^lo + B^lo - 1
102
103 r_k+1 = R - (B^lo-1)^2 + 2*Alr*Ahr*B^lo - 2*(Alr-adj)*Ahr*B^lo
104 = R - B^(2*lo) + 2*B^lo - 1 + 2*adj*Ahr*B^lo
105 */
106
107 // + 2*adj*Ahr*B^lo
108 mp_limb_t cy = lmmp_addmul_1_(R + lo, Ahr, hi, 2 * adj);
109 (R + ns)[0] += cy;
110 r = (R + ns)[0] < cy;
111
112 // - 1
113 r -= lmmp_sub_1_(R, R, ns + 1, 1);
114
115 // + 2*B^lo
116 r += lmmp_add_1_(R + lo, R + lo, hi + 1, 2);
117
118 // - B^(2*lo)
119 r -= lmmp_sub_1_(R + 2 * lo, R + 2 * lo, hi + 1 - lo, 1);
120 lmmp_debug_assert(r == 0);
121 } else if (calr == 0) {
122 /*
123 calr=0 时无需维护余数，计算 Alr^2 再相减仅仅是为了判定
124 Alr 是否高估 1（即 R 与 Alr^2 的大小），此判定无需完整平方：
125
126 Alr = H*B^(lo-2) + L, H = [Alr+lo-2,2], L < B^(lo-2)
127 Alr^2 = H^2*B^(2*lo-4) + E
128 E = 2*H*L*B^(lo-2) + L^2 < 3*B^(2*lo-2)
129
130 即 Alr^2 的最高两个 limb 必落在 [H^2 的最高两个 limb, +3] 内，
131 于是仅需比较 R 与 H^2 的最高两个 limb（注意 adj=0 时 Alr < B^lo）：
132 R 高于 B^(2*lo) 的 limb 非零 => R > Alr^2
133 [R] > [H^2]+4 （最高两 limb 意义下） => R > Alr^2
134 [R] < [H^2] （最高两 limb 意义下） => R < Alr^2
135 其余情况（最高两 limb 落入宽度 4 的窄带，如完全平方数
136 等构造输入）回退到下方的精确路径，保证正确性。
137 */
138 if (lo >= 4) {
139 mp_size_t i = ns;
140 while (i >= 2 * lo && R[i] == 0) --i;
141 if (i >= 2 * lo) {
142 // R >= B^(2*lo) > Alr^2，Alr 未高估
143 lmmp_copy(dst, Alr, lo);
144 return;
145 }
146 lmmp_sqr_hard_2_(Alr2, Alr + lo - 2); // [Alr2,4] = H^2
147 mp_limb_t h2h = Alr2[3], h2l = Alr2[2]; // H^2 的最高两 limb
148 mp_limb_t rh = R[2 * lo - 1], rl = R[2 * lo - 2];
149 mp_limb_t wl = h2l + 4; // [wh,wl] = [H^2]+4
150 mp_limb_t carry = wl < 4;
151 mp_limb_t wh = h2h + carry;
152 carry = wh < carry;
153 if ((!carry) && (rh > wh || (rh == wh && rl >= wl))) {
154 // [R] > [H^2]+4
155 lmmp_copy(dst, Alr, lo);
156 return;
157 }
158 if (rh < h2h || (rh == h2h && rl < h2l)) {
159 // R < Alr^2，Alr 高估 1
160 lmmp_dec(Alr);
161 lmmp_copy(dst, Alr, lo);
162 return;
163 }
164 }
165 // 窄带或小尺寸回退：精确判定
166 lmmp_sqr_(Alr2, Alr, lo);
167 if (lmmp_sub_(R, R, ns + 1, Alr2, 2 * lo))
168 lmmp_dec(Alr);
169 lmmp_copy(dst, Alr, lo);
170 } else {
171 lmmp_sqr_(Alr2, Alr, lo);
172 mp_limb_t b = lmmp_sub_(R, R, ns + 1, Alr2, 2 * lo);
173 if (b > 0) {
174 // + 2*Ahr*B^lo
175 mp_limb_t cy = lmmp_addshl1_n_(R + lo, R + lo, Ahr, hi);
176 (R + lo)[hi] += cy;
177 b -= (R + lo)[hi] < cy;
178
179 // + 2*Alr
180 cy = lmmp_addshl1_n_(R, R, Alr, lo);
181 b -= lmmp_add_1_(R + lo, R + lo, hi + 1, cy);
182
183 // - 1
184 b += lmmp_sub_1_(R, R, ns + 1, 1);
185
186 lmmp_debug_assert(b == 0);
187 lmmp_dec(Alr);
188 }
189 lmmp_copy(dst, Alr, lo);
190 }
191 }
192#undef Ahr
193#undef rk
194#undef R
195#undef Alr
196#undef Alr2
197}

199/*
200 计算 1/sqrt(a) 的定点下近似（ns >= na，a = [numa,na]，numa[na-1] >= B/4）：
201
202 I = floor(sqrt(B^(2*ns+na) / a)) (= floor(B^(ns+na/2) / sqrt(a)))
203
204 输出 ir = [dstis,ns+1] 的语义（na∈[1,128] x ns∈[3,256] 约 1500 组
205 随机+构造输入实验验证）：
206
207 I - 1 <= ir <= I, ir >= B^ns (即恒有 dstis[ns] = 1)
208
209 ir 是 I 的下近似，且至多低估 1。I 邻近 B^ns 或 2*B^ns 时，
210 末层修正须兼顾 ir <= I 与 dstis[ns] = 1。
211
212 注意 ns < na 时语义不同（lmmp_sqrt_newton_ 的调用恒有 ns >= na，见其
213 @warning）：此时每层仅使用 an 的最高 min(层尺寸,na) 个 limb，最终结果变为
214
215 floor(sqrt(B^(3*ns) / a_top)) - [0|1], a_top = a 的最高 ns 个 limb
216
217 与上面 I 的公式相差 B^((ns-na)/2) 倍（实验 656/656 全部吻合此式）。
218
219 算法：尺寸栈 ns -> (ns>>1)+1 -> ... -> 2，自 nr=2 的基例起逐层将精度
220 翻倍（下一层尺寸 na <= 2*nr-1，牛顿步一轮即可）。当前层近似记
221
222 ir = [dstis-nr, nr+1] = floor(B^(3*nr/2) / sqrt(ar)) - [0|1]
223 ar = an 的最高 nr 个 limb, an = a 的最高 min(na,namax) 个 limb
224
225 基例 (nr=2)：x = a 的最高 2 个 limb，
226
227 i2 = floor((B^5-1) / (floor(sqrt(x*B^4)) + 1))
228
229 分母 +1 使基例系统性略微低估，保证后续每层迭代恒有 ir <= 目标值。
230
231 每层牛顿步（对相对残差 e = 1 - an*ir^2/B^(na+2*nr) 二次收敛 e' ~ (3/2)e^2）：
232
233 d = B^(na+2*nr) - an*ir^2, -4*B^(na+nr) < d < 4*B^(na+nr)
234 i' = ir*B^(na-nr) + ir*d / (2*B^(2*nr))
235
236 ir^2 与 ir^2*an 的高于 B^(naz+nr+1) 的部分对 i' 无影响，故可用梅森变换
237 （mod B^mn-1）截断计算（mn = fft 友好尺寸）；d 的符号由残差最高 limb
238 是否 > 3 判定，d < 0 时通过按位取反 + 借位减实现负数修正。末层的
239 dec_1/inc_1 修正吸收所有截断误差，是产生上述至多 1 ulp 低估的根源。
240*/

242void lmmp_invsqrt_newton_(mp_ptr restrict dstis, mp_size_t ns, mp_srcptr restrict numa, mp_size_t na) {
243 lmmp_param_assert(ns >= 3);
244 lmmp_param_assert(na > 0);
245 lmmp_param_assert(numa != NULL && dstis != NULL);
246 lmmp_param_assert(numa[na - 1] >= LIMB_B_4);
247 mp_size_t nr = ns, namax = na, mn;
248 mp_size_t sizes[LIMB_BITS], *sizp = sizes;
249
250 do {
251 *sizp = nr;
252 nr = (nr >> 1) + 1;
253 ++sizp;
254 } while (nr > 2);
255
256 numa += na;
257 dstis += ns;
258
259 // nr=2
260 // i2=floor((B^5-1)/(1+floor(sqrt(x*B^4))))
261 mp_limb_t numa2[6], sval[3], tp[4];
262 lmmp_zero(numa2, 4);
263 numa2[5] = numa[-1];
264 if (na > 1)
265 numa2[4] = numa[-2];
266 else
267 numa2[4] = 0;
268
269 lmmp_sqrt_divide_(sval, numa2, 3, tp, 0);
270 lmmp_inc(sval);
271
272 for (mp_size_t i = 0; i < 5; ++i) numa2[i] = LIMB_MAX;
273 dstis[0] = lmmp_div_s_(dstis - 2, numa2, 5, sval, 3);
274
275 TEMP_DECL;
276 mp_limb_t alloc_size = na + 2 * ns + 6;
277 mp_ptr restrict xp = TALLOC_TYPE(alloc_size, mp_limb_t);
278 do {
279 na = *--sizp;
280
281 // ar = 0:[numa-nr,nr]
282 // an = 0:[numa-na,na]
283 // ir = 1:[dst-nr,nr] = floor(B^(3*nr/2)/sqrt(ar)) - [0|1]
284 // d = B^(na+2*nr)-an*ir*ir
285 // -4*B^(na+nr) < d < 4*B^(na+nr)
286
287 mp_size_t naz = LMMP_MIN(na, namax);
288 // mp_size_t zeros = na - naz;
289 mp_size_t nsqr, nres = naz + nr + 1;
290 mp_ptr dp = xp + 2 * nr + 1, dip = xp + nr + 1;
291 int cmod; // 1=mod b^mn-1, 0=mod b^(naz+nr+1)
292 int sign; // 1:d<0, 0:d>=0
293 mn = lmmp_mersenne_next_size_(nres);
294
295 // ir^2
296 if (2 * SQRT_NEWTON_MODM_THRESHOLD + mn >= nr * 2 + 1) {
297 cmod = 0;
298 lmmp_sqr_(xp, dstis - nr, nr + 1);
299 nsqr = 2 * nr + 1;
300 } else {
301 cmod = 1;
302 lmmp_sqr_mersenne_(xp, mn, dstis - nr, nr + 1);
303 nsqr = mn;
304 }
305
306 // ir^2*an
307 if (naz < SQRT_NEWTON_MODM_THRESHOLD || naz * 8 < nsqr || mn >= nsqr + naz) {
308 if (cmod == 0)
309 nsqr = LMMP_MIN(nsqr, nres);
310 lmmp_mul_(dp, xp, nsqr, numa - naz, naz);
311 if (cmod == 1) {
312 if (lmmp_add_(dp, dp, mn, dp + mn, naz))
313 lmmp_inc(dp);
314 }
315 } else {
316 if (nsqr > mn) { // cmod==0
317 if (lmmp_add_(xp, xp, mn, xp + mn, nsqr - mn))
318 lmmp_inc(xp);
319 }
320 lmmp_mul_mersenne_(dp, mn, xp, nsqr, numa - naz, naz);
321 cmod = 1;
322 }
323
324 if (cmod == 1) {
325 // naz+nr < mn <= naz+2*nr
326 //[dp,mn] -= B^(naz+2*nr) mod (B^mn-1)
327 dp[mn] = 1;
328 lmmp_dec(dp + naz + 2 * nr - mn);
329 if (dp[mn] == 0)
330 lmmp_dec(dp);
331 }
332
333 if (dp[nres - 1] > 3) { //-d<0
334 if (cmod == 0)
335 lmmp_dec(dp); // for neg to not
336 // else (neg to not) compensate (mod transfer)
337 dp += naz;
338 lmmp_shlnot_(xp, dp + 1, nr, LIMB_BITS - 1);
339 xp[0] ^= dp[0] >> 1;
340 xp[nr] = ~dp[nr] >> 1;
341 sign = 0;
342 } else { //-d>0
343 lmmp_shr_(xp, dp + naz, nr + 1, 1);
344 if ((dp[naz] & 1) || !lmmp_zero_q_(dp, naz))
345 lmmp_inc(xp);
346 sign = 1;
347 }
348
349 lmmp_mul_n_(dip, xp, dstis - nr, nr + 1);
350
351 if (sign) {
352 if (lmmp_zero_q_(dip, 3 * nr - na)) {
353 // a limit for dec
354 dip[2 * nr + 1] = 1;
355 lmmp_dec(dip + 3 * nr - na);
356 }
357 lmmp_not_(dstis - na, dip + 3 * nr - na, na - nr);
358 lmmp_dec_1(dstis - nr, dip[2 * nr] + 1);
359 } else {
360 lmmp_copy(dstis - na, dip + 3 * nr - na, na - nr);
361 lmmp_inc_1(dstis - nr, dip[2 * nr]);
362 }
363
364 nr = na;
365 } while (sizp != sizes);
366 TEMP_FREE;
367}

369/*
370 计算 x = sqrt(a*B^(2nf)) 的舍入近似 r（na 奇偶分别处理，a 归一化使 a2 = a*T^2
371 的最高 limb 最高 2 位非零，T = 2^nsh <= 2^31）：
372
373 ir = [ns+1] limb 逆平方根（多算 1 个 guard limb），Q = a2*ir 为精确整数，
374 偶 na: r = round(Q / 2^(64*(na+1)+nsh))
375 奇 na: r = round(Q / 2^(64*na+nsh+32)) (32 = log2(sqrt(B)))
376
377 记 It = sqrt(B^(2ns+na)/a2)（实值），有 Q/2^e = x*ir/It，故
378
379 r = round(x - eps), eps = x*(It-ir)/It >= 0
380
381 It-ir = frac(It) + [0|1] ∈ [0,2)（实值截断 + 算法至多 1 ulp 低估），而
382 x/It = aT*B^nf/B^(ns+na/2) < B^na/B^(na+1/2)（偶）/ B^na*sqrt(B)/B^(na+1)（奇），
383 因此
384
385 0 <= eps < 2^-63 (na 偶)， 0 <= eps < 2^-31 (na 奇)
386
387 r 不超过 round(x)，
388 仅当 frac(x) ∈ [1/2, 1/2+eps) 时得到 floor(x)（随机输入概率 < 2^-30），
389 此即 [floor|round] 的确切含义。
390*/
391
392void lmmp_sqrt_newton_(mp_ptr dsts, mp_srcptr numa, mp_size_t na, mp_size_t nf) {
393 lmmp_param_assert(na > 0);
394 lmmp_param_assert(nf >= 2);
395 lmmp_param_assert(numa != NULL && dsts != NULL);
396 mp_limb_t high = numa[na - 1];
397 int nsh = lmmp_leading_zeros_(high) / 2;
398 mp_size_t ns = na / 2 + 1 + nf;
399
400 TEMP_DECL;
401 mp_limb_t alloc_size = (nsh ? na : 0) + ns + 1;
402 mp_ptr tp = TALLOC_TYPE(alloc_size, mp_limb_t), numa2;
403 if (nsh) {
404 numa2 = tp;
405 lmmp_shl_(numa2, numa, na, nsh * 2);
406 tp += na;
407 } else
408 numa2 = (mp_ptr)numa;
409
410 lmmp_invsqrt_newton_(tp, ns, numa2, na);
411
412 mp_ptr restrict msqr = TALLOC_TYPE(na + ns + 1, mp_limb_t);
413
414 if (ns + 1 > na)
415 lmmp_mul_(msqr, tp, ns + 1, numa2, na);
416 else
417 lmmp_mul_(msqr, numa2, na, tp, ns + 1);
418
419 mp_limb_t cceil;
420 if (na & 1) {
421 nsh += LIMB_BITS / 2;
422 lmmp_shr_(dsts, msqr + na, ns, nsh);
423 cceil = msqr[na] >> (nsh - 1);
424 } else {
425 if (nsh) {
426 lmmp_shr_(dsts, msqr + na + 1, ns - 1, nsh);
427 cceil = msqr[na + 1] >> (nsh - 1);
428 } else {
429 lmmp_copy(dsts, msqr + na + 1, ns - 1);
430 cceil = msqr[na] >> (LIMB_BITS - 1);
431 }
432 dsts[ns - 1] = 0;
433 }
434
435 if (cceil & 1)
436 lmmp_inc(dsts);
437
438 TEMP_FREE;
439}

441void lmmp_sqrt_(mp_ptr dsts, mp_ptr dstr, mp_srcptr numa, mp_size_t na, mp_size_t nf) {
442 lmmp_debug_assert(na > 0);
443 lmmp_debug_assert(numa[na - 1] > 0);
444 mp_limb_t high = numa[na - 1];
445 int nsh = lmmp_leading_zeros_(high) / 2;
446 mp_size_t nl = na + 2 * nf;
447 if (nl == 1) {
448 mp_limb_t r;
449 mp_limb_t srt = lmmp_sqrt_1_(&r, high << nsh * 2);
450 srt >>= nsh;
451 dsts[0] = srt;
452 if (dstr)
453 dstr[0] = high - srt * srt;
454 } else if (nl <= 4) {
455 mp_limb_t tmp[4], rr[3], s, t, b;
456 mp_ptr src = (mp_ptr)numa;
457
458 if (nf) {
459 lmmp_zero(tmp, 2 * nf);
460 lmmp_shl_(tmp + 2 * nf, numa, na, 2 * nsh); // nsh=0 时函数内即整段拷贝
461 src = tmp;
462 } else if (nsh) {
463 lmmp_shl_(tmp, numa, na, 2 * nsh);
464 src = tmp;
465 }
466
467 if (nl == 2) {
468 s = lmmp_sqrt_2_(rr, src);
469 if (dstr && nsh) {
470 // rr[0..1] += 2*t*s - t^2
471 t = s & (((mp_limb_t)1 << nsh) - 1);
472 rr[1] += lmmp_addmul_1_(rr, &s, 1, 2 * t);
473 b = lmmp_submul_1_(rr, &t, 1, t);
474 lmmp_sub_1_(rr + 1, rr + 1, 1, b);
475 }
476 dsts[0] = s >> nsh;
477 dsts[1] = 0;
478 if (dstr)
479 lmmp_shr_(dstr, rr, 2, 2 * nsh);
480 } else if (nl == 3) {
481 lmmp_sqrt_3_(dsts, dstr ? rr : NULL, src);
482 if (dstr && nsh) {
483 // rr[0..2] += 2*t*[dsts,2] - t^2
484 t = dsts[0] & (((mp_limb_t)1 << nsh) - 1);
