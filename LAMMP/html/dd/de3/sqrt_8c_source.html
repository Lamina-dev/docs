23/*
24 A = Ah * B^(2*lo) + Al
25
26 Ahr = floor(Ah^(1/2))
27 rk = Ah - Ahr^2
28 x_k = Ahr * B^lo
29
30 x_k+1 = (x_k + A / x_k ) / 2
31 = x_k + (A / x_k - x_k) / 2
32 = x_k + (A - x_k^2) / 2 * x_k
33 = Ahr * B^lo + (rk * B^(2*lo) + Al) / 2 * x_k
34 = Ahr * B^lo + Alr
35
36 let Alr = (rk * B^(2*lo) + Al) / 2 * x_k, R = (rk * B^(2*lo) + Al) mod 2 * x_k
37 such that (rk * B^(2*lo) + Al) = Alr * 2 * x_k + R
38 ┌───────────────────────────────────────────────────────────┐
39 = |Alr * 2 * Ahr*B^lo + R = R_correct + (Alr-1) * 2 * Ahr*B^lo|
40 └───────────────────────────────────────────────────┬───────┘
41 |
42 r_k+1 = A - x_k+1^2 |
43 = Ah*B^(2*lo) + Al - Ahr^2*B^(2*lo) - 2*Alr*Ahr*B^lo - Alr^2 |
44 = r_k * B^(2*lo) + Al - 2*Alr*Ahr*B^lo - Alr^2 |
45 = R - Alr^2 |
46 |
47 Alr is either correct or 1 too big, |
48 ┌────────────────────────────────────┐ |
49 r_k+1 = R - (Alr-1)^2 + | 2*Alr*Ahr*B^lo - 2*(Alr-1)*Ahr*B^lo├────────────────────────┘
50 = R - Alr^2 └────────────────────────────────────┘
51(adjust) + 2*Ahr*B^lo + 2*Alr - 1
52*/

54void lmmp_sqrt_divide_(mp_ptr restrict dst, mp_ptr restrict numa, mp_size_t ns, mp_ptr restrict tp, int calr) {
55 lmmp_param_assert(ns > 0);
56 lmmp_param_assert(numa != NULL && dst != NULL && tp != NULL);
57 lmmp_param_assert(numa[2 * ns - 1] >= LIMB_B_4);
58 if (ns == 1) {
59 dst[0] = lmmp_sqrt_2_(numa, numa);
60 } else {
61 mp_size_t lo = ns / 2, hi = ns - lo;
62#define Ahr (dst + lo) // [dst+lo, hi]
63#define rk (numa + 2 * lo) // [numa+2*lo, hi+1]
64#define R (numa) // [numa, ns+1]
65#define Alr (tp) // [tp, lo+1]
66#define Alr2 (tp + lo + 1) // [tp + lo+1, 2*lo]
67
68 lmmp_sqrt_divide_(Ahr, rk, hi, tp, 1);
69
70 /*
71 A / (2*x) = A / 2 / x
72 A % (2*x) = 2 * (A / 2 % x) + A % 2
73 */
74 mp_limb_t r = lmmp_shr_(rk - lo, rk - lo, ns + 1, 1);
75 mp_limb_t qh = lmmp_div_s_(Alr, rk - lo, ns + 1, Ahr, hi);
76 lmmp_debug_assert(qh == 0);
77 (rk - lo)[hi] = lmmp_shl_c_(rk - lo, rk - lo, hi, 1, r >> (LIMB_BITS - 1));
78 /*
79 我们根据 sqrt(A/B^2) == floor(sqrt(A)/B) 可以知道，如果Alr正确结果
80 必定被限制在B^lo以内，其至多高估1，因此Alr的最高位必定为0或1，而为1时，
81 即代表此时结果已经高估。
82 */
83 mp_limb_t adj = Alr[lo];
84 lmmp_debug_assert(adj == 0 || adj == 1);
85 if (adj > 0) {
86 lmmp_debug_assert(Alr[0] == 0);
87 lmmp_fill(dst, 0, lo, LIMB_MAX);
88 if (calr == 0) return;
89 /*
90 x_k+1 = Ahr * B^lo + Alr
91 = Ahr * B^lo + B^lo - 1
92
93 r_k+1 = R - (B^lo-1)^2 + 2*Alr*Ahr*B^lo - 2*(Alr-adj)*Ahr*B^lo
94 = R - B^(2*lo) + 2*B^lo - 1 + 2*adj*Ahr*B^lo
95 */
96
97 // + 2*adj*Ahr*B^lo
98 mp_limb_t cy = lmmp_addmul_1_(R + lo, Ahr, hi, 2 * adj);
99 (R + ns)[0] += cy;
100 r = (R + ns)[0] < cy;
101
102 // - 1
103 r -= lmmp_sub_1_(R, R, ns + 1, 1);
104
105 // + 2*B^lo
106 r += lmmp_add_1_(R + lo, R + lo, hi + 1, 2);
107
108 // - B^(2*lo)
109 r -= lmmp_sub_1_(R + 2 * lo, R + 2 * lo, hi + 1 - lo, 1);
110 lmmp_debug_assert(r == 0);
111 } else {
112 lmmp_sqr_(Alr2, Alr, lo);
113 mp_limb_t b = lmmp_sub_(R, R, ns + 1, Alr2, 2 * lo);
114 if (calr == 0) {
115 if (b > 0)
116 lmmp_dec(Alr);
117 lmmp_copy(dst, Alr, lo);
118 return;
119 }
120 if (b > 0) {
121 // + 2*Ahr*B^lo
122 mp_limb_t cy = lmmp_addshl1_n_(R + lo, R + lo, Ahr, hi);
123 (R + lo)[hi] += cy;
124 b -= (R + lo)[hi] < cy;
125
126 // + 2*Alr
127 cy = lmmp_addshl1_n_(R, R, Alr, lo);
128 b -= lmmp_add_1_(R + lo, R + lo, hi + 1, cy);
129
130 // - 1
131 b += lmmp_sub_1_(R, R, ns + 1, 1);
132
133 lmmp_debug_assert(b == 0);
134 lmmp_dec(Alr);
135 }
136 lmmp_copy(dst, Alr, lo);
137 }
138 }
139#undef Ahr
140#undef rk
141#undef R
142#undef Alr
143#undef Alr2
144}

146void lmmp_invsqrt_newton_(mp_ptr restrict dstis, mp_size_t ns, mp_srcptr restrict numa, mp_size_t na) {
147 lmmp_param_assert(ns >= 3);
148 lmmp_param_assert(na > 0);
149 lmmp_param_assert(numa != NULL && dstis != NULL);
150 lmmp_param_assert(numa[na - 1] >= LIMB_B_4);
151 mp_size_t nr = ns, namax = na, mn;
152 mp_size_t sizes[LIMB_BITS], *sizp = sizes;
153
154 do {
155 *sizp = nr;
156 nr = (nr >> 1) + 1;
157 ++sizp;
158 } while (nr > 2);
159
160 numa += na;
161 dstis += ns;
162
163 // nr=2
164 // i2=floor((B^5-1)/(1+floor(sqrt(x*B^4))))
165 mp_limb_t numa2[6], sval[3], tp[4];
166 lmmp_zero(numa2, 4);
167 numa2[5] = numa[-1];
168 if (na > 1)
169 numa2[4] = numa[-2];
170 else
171 numa2[4] = 0;
172
173 lmmp_sqrt_divide_(sval, numa2, 3, tp, 0);
174 lmmp_inc(sval);
175
176 for (mp_size_t i = 0; i < 5; ++i) numa2[i] = LIMB_MAX;
177 dstis[0] = lmmp_div_s_(dstis - 2, numa2, 5, sval, 3);
178
179 TEMP_DECL;
180 mp_limb_t alloc_size = na + 2 * ns + 6;
181 mp_ptr restrict xp = TALLOC_TYPE(alloc_size, mp_limb_t);
182 do {
183 na = *--sizp;
184
185 // ar = 0:[numa-nr,nr]
186 // an = 0:[numa-na,na]
187 // ir = 1:[dst-nr,nr] = floor(B^(3*nr/2)/sqrt(ar)) - [0|1]
188 // d = B^(na+2*nr)-an*ir*ir
189 // -4*B^(na+nr) < d < 4*B^(na+nr)
190
191 mp_size_t naz = LMMP_MIN(na, namax);
192 // mp_size_t zeros = na - naz;
193 mp_size_t nsqr, nres = naz + nr + 1;
194 mp_ptr dp = xp + 2 * nr + 1, dip = xp + nr + 1;
195 int cmod; // 1=mod b^mn-1, 0=mod b^(naz+nr+1)
196 int sign; // 1:d<0, 0:d>=0
197 mn = lmmp_fft_next_size_(nres);
198
199 // ir^2
200 if (2 * SQRT_NEWTON_MODM_THRESHOLD + mn >= nr * 2 + 1) {
201 cmod = 0;
202 lmmp_sqr_(xp, dstis - nr, nr + 1);
203 nsqr = 2 * nr + 1;
204 } else {
205 cmod = 1;
206 lmmp_sqr_mersenne_(xp, mn, dstis - nr, nr + 1);
207 nsqr = mn;
208 }
209
210 // ir^2*an
211 if (naz < SQRT_NEWTON_MODM_THRESHOLD || naz * 8 < nsqr || mn >= nsqr + naz) {
212 if (cmod == 0)
213 nsqr = LMMP_MIN(nsqr, nres);
214 lmmp_mul_(dp, xp, nsqr, numa - naz, naz);
215 if (cmod == 1) {
216 if (lmmp_add_(dp, dp, mn, dp + mn, naz))
217 lmmp_inc(dp);
218 }
219 } else {
220 if (nsqr > mn) { // cmod==0
221 if (lmmp_add_(xp, xp, mn, xp + mn, nsqr - mn))
222 lmmp_inc(xp);
223 }
224 lmmp_mul_mersenne_(dp, mn, xp, nsqr, numa - naz, naz);
225 cmod = 1;
226 }
227
228 if (cmod == 1) {
229 // naz+nr < mn <= naz+2*nr
230 //[dp,mn] -= B^(naz+2*nr) mod (B^mn-1)
231 dp[mn] = 1;
232 lmmp_dec(dp + naz + 2 * nr - mn);
233 if (dp[mn] == 0)
234 lmmp_dec(dp);
235 }
236
237 if (dp[nres - 1] > 3) { //-d<0
238 if (cmod == 0)
239 lmmp_dec(dp); // for neg to not
240 // else (neg to not) compensate (mod transfer)
241 dp += naz;
242 lmmp_shlnot_(xp, dp + 1, nr, LIMB_BITS - 1);
243 xp[0] ^= dp[0] >> 1;
244 xp[nr] = ~dp[nr] >> 1;
245 sign = 0;
246 } else { //-d>0
247 lmmp_shr_(xp, dp + naz, nr + 1, 1);
248 if ((dp[naz] & 1) || !lmmp_zero_q_(dp, naz))
249 lmmp_inc(xp);
250 sign = 1;
251 }
252
253 lmmp_mul_n_(dip, xp, dstis - nr, nr + 1);
254
255 if (sign) {
256 if (lmmp_zero_q_(dip, 3 * nr - na)) {
257 // a limit for dec
258 dip[2 * nr + 1] = 1;
259 lmmp_dec(dip + 3 * nr - na);
260 }
261 lmmp_not_(dstis - na, dip + 3 * nr - na, na - nr);
262 lmmp_dec_1(dstis - nr, dip[2 * nr] + 1);
263 } else {
264 lmmp_copy(dstis - na, dip + 3 * nr - na, na - nr);
265 lmmp_inc_1(dstis - nr, dip[2 * nr]);
266 }
267
268 nr = na;
269 } while (sizp != sizes);
270 TEMP_FREE;
271}

273void lmmp_sqrt_newton_(mp_ptr dsts, mp_srcptr numa, mp_size_t na, mp_size_t nf) {
274 lmmp_param_assert(na > 0);
275 lmmp_param_assert(nf >= 2);
276 lmmp_param_assert(numa != NULL && dsts != NULL);
277 mp_limb_t high = numa[na - 1];
278 int nsh = lmmp_leading_zeros_(high) / 2;
279 mp_size_t ns = na / 2 + 1 + nf;
280
281 TEMP_DECL;
282 mp_limb_t alloc_size = (nsh ? na : 0) + ns + 1;
283 mp_ptr tp = TALLOC_TYPE(alloc_size, mp_limb_t), numa2;
284 if (nsh) {
285 numa2 = tp;
286 lmmp_shl_(numa2, numa, na, nsh * 2);
287 tp += na;
288 } else
289 numa2 = (mp_ptr)numa;
290
291 lmmp_invsqrt_newton_(tp, ns, numa2, na);
292
293 mp_ptr restrict msqr = TALLOC_TYPE(na + ns + 1, mp_limb_t);
294
295 if (ns + 1 > na)
296 lmmp_mul_(msqr, tp, ns + 1, numa2, na);
297 else
298 lmmp_mul_(msqr, numa2, na, tp, ns + 1);
299
300 mp_limb_t cceil;
301 if (na & 1) {
302 nsh += LIMB_BITS / 2;
303 lmmp_shr_(dsts, msqr + na, ns, nsh);
304 cceil = msqr[na] >> (nsh - 1);
305 } else {
306 if (nsh) {
307 lmmp_shr_(dsts, msqr + na + 1, ns - 1, nsh);
308 cceil = msqr[na + 1] >> (nsh - 1);
309 } else {
310 lmmp_copy(dsts, msqr + na + 1, ns - 1);
311 cceil = msqr[na] >> (LIMB_BITS - 1);
312 }
313 dsts[ns - 1] = 0;
314 }
315
316 if (cceil & 1)
317 lmmp_inc(dsts);
318
319 TEMP_FREE;
320}

322void lmmp_sqrt_(mp_ptr dsts, mp_ptr dstr, mp_srcptr numa, mp_size_t na, mp_size_t nf) {
323 lmmp_debug_assert(na > 0);
324 lmmp_debug_assert(numa[na - 1] > 0);
325 mp_limb_t high = numa[na - 1];
326 int nsh = lmmp_leading_zeros_(high) / 2;
327 mp_size_t nl = na + 2 * nf;
328 if (nl == 1) {
329 mp_limb_t r;
330 mp_limb_t srt = lmmp_sqrt_1_(&r, high << nsh * 2);
331 srt >>= nsh;
332 dsts[0] = srt;
333 if (dstr)
334 dstr[0] = high - srt * srt;
335 } else if (!dstr && nf >= 10 * na + SQRT_INVNEWTON_THRESHOLD) {
336 lmmp_sqrt_newton_(dsts, numa, na, nf);
337 } else {
338 TEMP_DECL;
339 mp_limb_t ns = (nl + 1) / 2;
340 mp_ptr restrict numa2 = TALLOC_TYPE(2 * ns, mp_limb_t);
341 mp_ptr restrict tp = TALLOC_TYPE(3 * ns / 2 + 1, mp_limb_t);
342 lmmp_zero(numa2, 2 * nf);
343 if (nsh)
344 lmmp_shl_(numa2 + 2 * ns - na, numa, na, nsh * 2);
345 else
346 lmmp_copy(numa2 + 2 * ns - na, numa, na);
347 if (nl & 1) {
348 numa2[2 * nf] = 0;
349 nsh += LIMB_BITS / 2;
350 } else {
351 dsts[ns] = 0;
352 }
353
354 lmmp_sqrt_divide_(dsts, numa2, ns, tp, dstr ? 1: 0);
355 if (nsh) {
356 /*
357 let
358 s = sqrt(A*T^2), r = sqrtrem(A*T^2)
359 such that
360 s = T*a + t, where a = sqrt(A) = s // T
361 then
362 sqrtrem(A) = (r + 2*T*a*t + t^2) // T^2
363 = (r + 2*t*(s-t) + t^2) // T^2
364 = (r + 2*t*s - t^2) // T^2
365 */
366 if (dstr) {
367 mp_limb_t t = dsts[0] & (((mp_limb_t)1 << nsh) - 1);
368 numa2[ns] += lmmp_addmul_1_(numa2, dsts, ns, 2 * t);
369 mp_limb_t b = lmmp_submul_1_(numa2, &t, 1, t);
370 lmmp_sub_1_(numa2 + 1, numa2 + 1, ns, b);
371 }
372 lmmp_shr_(dsts, dsts, ns, nsh);
373 }
374 if (dstr) {
375 nsh *= 2;
376 if (nsh >= LIMB_BITS) {
377 nsh -= LIMB_BITS;
378 ++numa2;
379 } else
380 ++ns;
381 if (nsh)
382 lmmp_shr_(dstr, numa2, ns, nsh);
383 else
384 lmmp_copy(dstr, numa2, ns);
385 }
386
387 TEMP_FREE;
388 }
