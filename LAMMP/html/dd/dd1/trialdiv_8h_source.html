80/**
81 * @brief 校验是否能被3,5,7,11整除，能够整除则返回1，否则返回0
82 */
83static inline int trial_div35711(ulong n) {
84 uint rem = n % 1155; // 1155 = 3 * 5 * 7 * 11
85 uint idx = rem / LIMB_BITS;
86 uint bit = rem % LIMB_BITS;
87 return (r35711_mask_map[idx] >> bit) & 1ULL;
88}
