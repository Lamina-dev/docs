69 mp_size_t k = 0;

74mp_size_t lmmp_fft_next_size_(mp_size_t n) {
75 mp_size_t k = lmmp_fft_best_k_(n);
76 lmmp_debug_assert(k >= LOG2_LIMB_BITS);
77 k -= LOG2_LIMB_BITS;
78 n = (((n - 1) >> k) + 1) << k;

100void lmmp_fft_extract_coef_(mp_ptr dst, mp_srcptr numa, mp_size_t bitoffset, mp_size_t bits, mp_size_t lenw) {
101 mp_size_t shr = bitoffset & (LIMB_BITS - 1), offset = bitoffset / LIMB_BITS;
102
103 mp_size_t lena = (bitoffset + bits - 1) / LIMB_BITS - offset + 1, endp = (bits - 1) / LIMB_BITS;
104
105 if (shr)
106 lmmp_shr_(dst, numa + offset, lena, shr);
107 else
108 lmmp_copy(dst, numa + offset, lena);
109
110 dst[endp] &= LIMB_MAX >> (-bits & (LIMB_BITS - 1));
111
112 lmmp_zero(dst + endp + 1, lenw - endp);
113}

115void lmmp_fft_shl_coef_(fft_memstack* ms, mp_ptr* coef, mp_size_t shl) {
116 mp_size_t l = ms->lenw; // 系数的机器字长度
117 mp_size_t w = shl / LIMB_BITS; // 左移的机器字数量
118 shl &= LIMB_BITS - 1; // 剩余的比特偏移（0~LIMB_BITS-1）
119 mp_ptr src = *coef; // 源系数数组
120 mp_ptr dst = ms->temp_coef; // 目标临时数组
121 mp_limb_t cc, rd; // 进位变量（cc=carry, rd=read）
122
123 if (w >= l) {
124 w -= l;
125 if (shl) {
126 lmmp_shl_(dst, src + l - w, w + 1, shl);
127 rd = dst[w];
128 cc = lmmp_shlnot_(dst + w, src, l - w, shl);
129 } else {
130 if (w)
131 lmmp_copy(dst, src + l - w, w);
132 rd = src[l];
133 lmmp_not_(dst + w, src, l - w);
134 cc = 0;
135 }
136 dst[l] = 0;
137 ++cc;
138 lmmp_inc_1(dst, cc);
139
140 if (++rd == 0)
141 lmmp_inc(dst + w + 1);
142 else
143 lmmp_inc_1(dst + w, rd);
144 } else {
145 if (shl) {
146 lmmp_shlnot_(dst, src + l - w, w + 1, shl);
147 rd = ~dst[w];
148 cc = lmmp_shl_(dst + w, src, l - w, shl);
149 } else {
150 if (w)
151 lmmp_not_(dst, src + l - w, w);
152 rd = src[l];
153
154 lmmp_copy(dst + w, src, l - w);
155 cc = 0;
156 }
157 dst[l] = 2;
158 lmmp_inc_1(dst, 3);
159 lmmp_dec_1(dst, cc);
160
161 if (++rd == 0)
162 lmmp_dec(dst + w + 1);
163 else
164 lmmp_dec_1(dst + w, rd);
165
166 cc = dst[l];
167 dst[l] = dst[0] < cc;
168 lmmp_dec_1(dst, cc - dst[l]);
169 }
170
171 ms->temp_coef = src;
172 *coef = dst;
173}

179void lmmp_fft_bfy_(fft_memstack* ms, mp_ptr* coef, mp_size_t wing, mp_size_t w) {
180 mp_ptr numa = coef[0]; // 系数a
181 mp_ptr numb = coef[wing]; // 系数b
182 mp_ptr numc = ms->temp_coef; // 临时数组（存储a-b<<w）
183 mp_size_t shl = w & (LIMB_BITS - 1); // 比特级左移量
184 w /= LIMB_BITS; // 机器字级左移量
185 mp_size_t l = ms->lenw; // 系数长度（机器字）
186
187 mp_slimb_t acyo = 0, scyo = 0, ch;
188 mp_limb_t shlcyo = 0, chp = 0, chn = 0;
189
190 for (mp_size_t off = 0; off < l - w; off += PART_SIZE) {
191 mp_size_t cursize = LMMP_MIN(l - w - off, PART_SIZE);
192 scyo = lmmp_sub_nc_(numc + w + off, numa + off, numb + off, cursize, scyo);
193 acyo = lmmp_add_nc_(numa + off, numa + off, numb + off, cursize, acyo);
194 if (shl)
195 shlcyo = lmmp_shl_c_(numc + w + off, numc + w + off, cursize, shl, shlcyo);
196 }
197
198 ch = shlcyo + (-scyo << shl);
199 if (ch > 0)
200 chp = ch;
201 else
202 chn = -ch;
203
204 scyo = 0;
205 shlcyo = 0;
206
207 for (mp_size_t off = l - w; off < l; off += PART_SIZE) {
208 mp_size_t cursize = LMMP_MIN(l - off, PART_SIZE);
209 scyo = lmmp_sub_nc_(numc + off - (l - w), numb + off, numa + off, cursize, scyo);
210 acyo = lmmp_add_nc_(numa + off, numa + off, numb + off, cursize, acyo);
211 if (shl)
212 shlcyo = lmmp_shl_c_(numc + off - (l - w), numc + off - (l - w), cursize, shl, shlcyo);
213 }
214
215 numc[w] += shlcyo; // 左移进位加到numc[w]
216 scyo = -scyo + numb[l] - numa[l]; // 调整借位（包含最高位）
217 acyo += numa[l] + numb[l]; // 调整进位（包含最高位）
218
219 numa[l] = numa[0] < (mp_limb_t)(acyo);
220 lmmp_dec_1(numa, acyo - numa[l]);
221
222 numc[l] = 1;
223 ++chn;
224 if (scyo > 0)
225 lmmp_inc_1(numc + w, scyo << shl);
226 else if (scyo < 0) {
227 if (scyo == -2 && shl == LIMB_BITS - 1)
228 lmmp_dec(numc + w + 1);
229 else
230 lmmp_dec_1(numc + w, -scyo << shl);
231 }
232 chp += numc[l];
233
234 if (chn >= chp) {
235 numc[l] = 0;
236 lmmp_inc_1(numc, chn - chp);
237 } else {
238 chp -= chn;
239 numc[l] = numc[0] < chp;
240 lmmp_dec_1(numc, chp - numc[l]);
241 }
242
243 coef[wing] = numc;
244 ms->temp_coef = numb;
245}

247void lmmp_ifft_bfy_(fft_memstack* ms, mp_ptr* coef, mp_size_t wing, mp_size_t w) {
248 mp_ptr numa = coef[0];
249 mp_ptr numb = coef[wing];
250 mp_ptr numc = ms->temp_coef;
251 mp_size_t shr = w & (LIMB_BITS - 1);
252 w /= LIMB_BITS;
253 mp_size_t l = ms->lenw;
254
255 mp_slimb_t bcyo = 0, acyo = 0, ah;
256 mp_limb_t shrcyo = shr ? numb[0] << (LIMB_BITS - shr) : 0;
257
258 for (mp_size_t off = l - w; off < l; off += PART_SIZE) {
259 mp_size_t cursize = LMMP_MIN(l - off, PART_SIZE);
260 if (shr)
261 lmmp_shr_c_(numb + off - (l - w), numb + off - (l - w), cursize, shr,
262 numb[off - (l - w) + cursize] << (LIMB_BITS - shr));
263 bcyo = lmmp_add_nc_(numc + off, numa + off, numb + off - (l - w), cursize, bcyo);
264 acyo = lmmp_sub_nc_(numa + off, numa + off, numb + off - (l - w), cursize, acyo);
265 }
266
267 for (mp_size_t off = 0; off < l - w; off += PART_SIZE) {
268 mp_size_t cursize = LMMP_MIN(l - w - off, PART_SIZE);
269 if (shr)
270 lmmp_shr_c_(numb + w + off, numb + w + off, cursize, shr, numb[off + w + cursize] << (LIMB_BITS - shr));
271 bcyo = lmmp_sub_nc_(numc + off, numa + off, numb + w + off, cursize, bcyo);
272 acyo = lmmp_add_nc_(numa + off, numa + off, numb + w + off, cursize, acyo);
273 }
274
275 acyo += numb[l] >> shr;
276 bcyo = -bcyo - (numb[l] >> shr);
277
278 acyo -= numa[l - w - 1] < shrcyo;
279 numa[l - w - 1] -= shrcyo;
280 numc[l - w - 1] += shrcyo;
281 bcyo += numc[l - w - 1] < shrcyo;
282
283 ah = numa[l];
284
285 numa[l] += 1;
286 if (w == 0)
287 numa[l] += acyo;
288 else {
289 if (acyo < 0)
290 lmmp_dec(numa + l - w);
291 else
292 lmmp_inc_1(numa + l - w, acyo);
293 }
294 acyo = numa[l] - 1;
295 if (acyo < 0) {
296 numa[l] = 0;
297 lmmp_inc(numa);
298 } else {
299 numa[l] = numa[0] < (mp_limb_t)acyo;
300 lmmp_dec_1(numa, acyo - numa[l]);
301 }
302
303 numc[l] = ah + 2;
304 if (w == 0)
305 numc[l] += bcyo;
306 else {
307 if (bcyo > 0)
308 lmmp_inc(numc + l - w);
309 else
310 lmmp_dec_1(numc + l - w, -bcyo);
311 }
312 bcyo = numc[l] - 2;
313 if (bcyo <= 0) {
314 numc[l] = 0;
315 lmmp_inc_1(numc, -bcyo);
316 } else {
317 numc[l] = numc[0] < (mp_limb_t)bcyo;
318 lmmp_dec_1(numc, bcyo - numc[l]);
319 }
320
321 coef[wing] = numc;
322 ms->temp_coef = numb;
323}

431void lmmp_mul_fermat_recombine_(
432 fft_memstack* ms,
433 mp_ptr dst,
434 mp_ptr* pfca,
435 mp_size_t K,
436 mp_size_t k,
437 mp_size_t n,
438 mp_size_t M,
439 mp_size_t rn
440) {
441 mp_size_t rhead = 0, nlen = ms->lenw + 1;
442 mp_slimb_t borrow = 0, maxc = 0;
443
444 for (mp_size_t i = 0; i < K; ++i) {
445 lmmp_fft_shr_coef_(ms, pfca + i, (i * n >> k) + k);
446 mp_ptr nums = pfca[i];
447
448 if (nums[nlen - 1]) {
449 lmmp_dec(nums);
450 --nums[nlen - 1];
451 }
452 if (nums[nlen - 1] == 0 && nums[nlen - 2] >> (LIMB_BITS - 1)) {
453 lmmp_dec(nums);
454 --nums[nlen - 1];
455 }
456
457 if (borrow) {
458 mp_size_t brshift = borrow - 1 + n - M;
459 mp_size_t bshl = brshift & (LIMB_BITS - 1);
460 brshift /= LIMB_BITS;
461 --nums[nlen - 1];
462 lmmp_dec_1(nums + brshift, (mp_limb_t)1 << bshl);
463 ++nums[nlen - 1];
464 }
465 borrow = -nums[nlen - 1];
466 nums[nlen - 1] = 0;
467
468 mp_size_t roffset = i * M;
469 mp_size_t shl = roffset & (LIMB_BITS - 1);
470 roffset /= LIMB_BITS;
471
472 if (shl)
473 lmmp_shl_(nums, nums, nlen, shl);
474
475 if (i == 0) {
476 lmmp_copy(dst, nums, nlen);
477 rhead = nlen;
478 } else if (roffset + nlen <= rn) {
479 lmmp_add_(dst + roffset, nums, nlen, dst + roffset, rhead - roffset);
480 rhead = roffset + nlen;
481 } else {
482 maxc += lmmp_add_(dst + roffset, nums, rn - roffset, dst + roffset, rhead - roffset);
483 maxc -= lmmp_sub_(dst, dst, rn, nums + rn - roffset, nlen + roffset - rn);
484 rhead = rn;
485 }
486 }
487
488 if (borrow) {
489 mp_size_t cyshift = borrow - 1 + n - M;
490 mp_size_t cshl = cyshift & (LIMB_BITS - 1);
491 cyshift /= LIMB_BITS;
492 maxc += lmmp_add_1_(dst + cyshift, dst + cyshift, rn - cyshift, (mp_limb_t)1 << cshl);
493 }
494
495 if (maxc > 0) {
496 dst[rn] = dst[0] < (mp_limb_t)maxc;
497 lmmp_dec_1(dst, maxc - dst[rn]);
498 } else {
499 dst[rn] = 0;
500 lmmp_inc_1(dst, -maxc);
501 }
