27// ((mp_size_t)3 << (2 * (n) - 5)) + 1 是预计算的阈值，n是对应的k值
28// 该阈值是 k 层递归安全展开的最小规模，表尾大尺寸段沿用此规则
29#define _FFT_TABLE_ENTRY(n) {((mp_size_t)3 << (2 * (n) - 5)) + 1, (n)}
30#define _FFT_TABLE_ENTRY4(n) \
31 _FFT_TABLE_ENTRY(n), _FFT_TABLE_ENTRY((n) + 1), _FFT_TABLE_ENTRY((n) + 2), _FFT_TABLE_ENTRY((n) + 3)

89mp_size_t lmmp_fft_best_k_(mp_size_t n) {
90 mp_size_t k = 0;
91 while (n >= lmmp_fft_table_[k + 1][0]) ++k;
92 return lmmp_fft_table_[k][1];
93}

95mp_size_t lmmp_fft_next_size_(mp_size_t n) {
96 mp_size_t k = lmmp_fft_best_k_(n);
97 lmmp_debug_assert(k >= LOG2_LIMB_BITS);
98 k -= LOG2_LIMB_BITS;
99 n = (((n - 1) >> k) + 1) << k;
100 return n;
101}

103void* lmmp_fft_memstack_(fft_memstack* ms, mp_size_t size) {
104 if (size) {
105 if (++ms->tempdepth > ms->maxdepth) {
106#if LMMP_DEBUG_MEMORY_CHECK == 1
107 if (ms->maxdepth + 1 >= FFT_MEMSTACK_DEPTH) {
108 lmmp_abort(LMMP_ERROR_OUT_OF_BOUNDS, "fft memstack depth overflow", __func__, __LINE__);
109 }
110#endif
111 ms->mem[++ms->maxdepth] = lmmp_alloc(size);
112 ms->memsize[ms->maxdepth] = size;
113 }
114#if LMMP_DEBUG_MEMORY_CHECK == 1
115 if (ms->memsize[ms->tempdepth] != size) {
116 lmmp_abort(LMMP_ERROR_OUT_OF_BOUNDS, "fft memstack size mismatch at same depth", __func__, __LINE__);
117 }
118#endif
119 return ms->mem[ms->tempdepth];
120 } else {
121 if (--ms->tempdepth < 0) {
122 for (mp_size_t i = 0; i <= (mp_size_t)(ms->maxdepth); ++i) lmmp_free(ms->mem[i]);
123 ms->maxdepth = -1;
124 }
125 return 0;
126 }

129void lmmp_fft_extract_coef_(mp_ptr dst, mp_srcptr numa, mp_size_t bitoffset, mp_size_t bits, mp_size_t lenw) {
130 mp_size_t shr = bitoffset & (LIMB_BITS - 1), offset = bitoffset / LIMB_BITS;
131
132 mp_size_t lena = (bitoffset + bits - 1) / LIMB_BITS - offset + 1, endp = (bits - 1) / LIMB_BITS;
133
134 lmmp_shr_(dst, numa + offset, lena, shr); // shr=0 时函数内即整段拷贝
135
136 dst[endp] &= LIMB_MAX >> (-bits & (LIMB_BITS - 1));
137
138 lmmp_zero(dst + endp + 1, lenw - endp);
139}

141void lmmp_fft_shl_coef_(fft_memstack* ms, mp_ptr* coef, mp_size_t shl) {
142 mp_size_t l = ms->lenw; // 系数的机器字长度
143 mp_size_t w = shl / LIMB_BITS; // 左移的机器字数量
144 shl &= LIMB_BITS - 1; // 剩余的比特偏移（0~LIMB_BITS-1）
145 mp_ptr src = *coef; // 源系数数组
146 mp_ptr dst = ms->temp_coef; // 目标临时数组
147 mp_limb_t cc, rd; // 进位变量（cc=carry, rd=read）
148
149 if (w >= l) {
150 w -= l;
151 if (shl) {
152 lmmp_shl_(dst, src + l - w, w + 1, shl);
153 rd = dst[w];
154 cc = lmmp_shlnot_(dst + w, src, l - w, shl);
155 } else {
156 if (w)
157 lmmp_copy(dst, src + l - w, w);
158 rd = src[l];
159 lmmp_not_(dst + w, src, l - w);
160 cc = 0;
161 }
162 dst[l] = 0;
163 ++cc;
164 lmmp_inc_1(dst, cc);
165
166 if (++rd == 0)
167 lmmp_inc(dst + w + 1);
168 else
169 lmmp_inc_1(dst + w, rd);
170 } else {
171 if (shl) {
172 lmmp_shlnot_(dst, src + l - w, w + 1, shl);
173 rd = ~dst[w];
174 cc = lmmp_shl_(dst + w, src, l - w, shl);
175 } else {
176 if (w)
177 lmmp_not_(dst, src + l - w, w);
178 rd = src[l];
179
180 lmmp_copy(dst + w, src, l - w);
181 cc = 0;
182 }
183 dst[l] = 2;
184 lmmp_inc_1(dst, 3);
185 lmmp_dec_1(dst, cc);
186
187 if (++rd == 0)
188 lmmp_dec(dst + w + 1);
189 else
190 lmmp_dec_1(dst + w, rd);
191
192 cc = dst[l];
193 dst[l] = dst[0] < cc;
194 lmmp_dec_1(dst, cc - dst[l]);
195 }
196
197 ms->temp_coef = src;
198 *coef = dst;
199}

205void lmmp_fft_bfy_(fft_memstack* ms, mp_ptr* coef, mp_size_t wing, mp_size_t w) {
206 mp_ptr numa = coef[0]; // 系数a
207 mp_ptr numb = coef[wing]; // 系数b
208 mp_ptr numc = ms->temp_coef; // 临时数组（存储a-b<<w）
209 mp_size_t shl = w & (LIMB_BITS - 1); // 比特级左移量
210 w /= LIMB_BITS; // 机器字级左移量
211 mp_size_t l = ms->lenw; // 系数长度（机器字）
212
213 mp_slimb_t acyo = 0, scyo = 0;
214 mp_limb_t shlcyo = 0, chp = 0, chn = 0;
215 mp_limb_t out[4]; // fft 融合内核出参 {chp, chn, shlcy2, (acyo<<1)|scyo}
216 if (shl) {
217 lmmp_fft_bfy_fused_(numc, numa, numb, l, w, shl, out);
218 chp = out[0];
219 chn = out[1];
220 shlcyo = out[2];
221 acyo = (mp_slimb_t)(out[3] >> 1);
222 scyo = (mp_slimb_t)(out[3] & 1);
223 } else {
224 for (mp_size_t off = 0; off < l - w; off += PART_SIZE) {
225 mp_size_t cursize = LMMP_MIN(l - w - off, PART_SIZE);
226 scyo = lmmp_sub_nc_(numc + w + off, numa + off, numb + off, cursize, scyo);
227 acyo = lmmp_add_nc_(numa + off, numa + off, numb + off, cursize, acyo);
228 }
229 chn = (mp_limb_t)scyo; // shl=0: ch = 0 - scyo
230 scyo = 0;
231 for (mp_size_t off = l - w; off < l; off += PART_SIZE) {
232 mp_size_t cursize = LMMP_MIN(l - off, PART_SIZE);
233 scyo = lmmp_sub_nc_(numc + off - (l - w), numb + off, numa + off, cursize, scyo);
234 acyo = lmmp_add_nc_(numa + off, numa + off, numb + off, cursize, acyo);
235 }
236 }
237
238 numc[w] += shlcyo; // 左移进位加到numc[w]
239 scyo = -scyo + numb[l] - numa[l]; // 调整借位（包含最高位）
240 acyo += numa[l] + numb[l]; // 调整进位（包含最高位）
241
242 numa[l] = numa[0] < (mp_limb_t)(acyo);
243 lmmp_dec_1(numa, acyo - numa[l]);
244
245 numc[l] = 1;
246 ++chn;
247 if (scyo > 0)
248 lmmp_inc_1(numc + w, (mp_limb_t)scyo << shl);
249 else if (scyo < 0) {
250 if (scyo == -2 && shl == LIMB_BITS - 1)
251 lmmp_dec(numc + w + 1);
252 else
253 lmmp_dec_1(numc + w, (mp_limb_t)(-scyo) << shl);
254 }
255 chp += numc[l];
256
257 if (chn >= chp) {
258 numc[l] = 0;
259 lmmp_inc_1(numc, chn - chp);
260 } else {
261 chp -= chn;
262 numc[l] = numc[0] < chp;
263 lmmp_dec_1(numc, chp - numc[l]);
264 }
265
266 coef[wing] = numc;
267 ms->temp_coef = numb;
268}

270void lmmp_ifft_bfy_(fft_memstack* ms, mp_ptr* coef, mp_size_t wing, mp_size_t w) {
271 mp_ptr numa = coef[0];
272 mp_ptr numb = coef[wing];
273 mp_ptr numc = ms->temp_coef;
274 mp_size_t shr = w & (LIMB_BITS - 1);
275 w /= LIMB_BITS;
276 mp_size_t l = ms->lenw;
277
278 mp_slimb_t bcyo = 0, acyo = 0, ah;
279 mp_limb_t shrcyo = shr ? numb[0] << (LIMB_BITS - shr) : 0;
280 mp_limb_t kc; // ifft 融合内核的打包进位 (acyo<<1)|bcyo
281 if (shr) {
282 kc = lmmp_ifft_bfy_fused_(numc, numa, numb, l, w, shr);
283 acyo = (mp_slimb_t)(kc >> 1);
284 bcyo = (mp_slimb_t)(kc & 1);
285 } else {
286 for (mp_size_t off = l - w; off < l; off += PART_SIZE) {
287 mp_size_t cursize = LMMP_MIN(l - off, PART_SIZE);
288 bcyo = lmmp_add_nc_(numc + off, numa + off, numb + off - (l - w), cursize, bcyo);
289 acyo = lmmp_sub_nc_(numa + off, numa + off, numb + off - (l - w), cursize, acyo);
290 }
291 for (mp_size_t off = 0; off < l - w; off += PART_SIZE) {
292 mp_size_t cursize = LMMP_MIN(l - w - off, PART_SIZE);
293 bcyo = lmmp_sub_nc_(numc + off, numa + off, numb + w + off, cursize, bcyo);
294 acyo = lmmp_add_nc_(numa + off, numa + off, numb + w + off, cursize, acyo);
295 }
296 }
297
298 acyo += numb[l] >> shr;
299 bcyo = -bcyo - (numb[l] >> shr);
300
301 acyo -= numa[l - w - 1] < shrcyo;
302 numa[l - w - 1] -= shrcyo;
303 numc[l - w - 1] += shrcyo;
304 bcyo += numc[l - w - 1] < shrcyo;
305
306 ah = numa[l];
307
308 numa[l] += 1;
309 if (w == 0)
310 numa[l] += acyo;
311 else {
312 if (acyo < 0)
313 lmmp_dec(numa + l - w);
314 else
315 lmmp_inc_1(numa + l - w, acyo);
316 }
317 acyo = numa[l] - 1;
318 if (acyo < 0) {
319 numa[l] = 0;
320 lmmp_inc(numa);
321 } else {
322 numa[l] = numa[0] < (mp_limb_t)acyo;
323 lmmp_dec_1(numa, acyo - numa[l]);
324 }
325
326 numc[l] = ah + 2;
327 if (w == 0)
328 numc[l] += bcyo;
329 else {
330 if (bcyo > 0)
331 lmmp_inc(numc + l - w);
332 else
333 lmmp_dec_1(numc + l - w, -bcyo);
334 }
335 bcyo = numc[l] - 2;
336 if (bcyo <= 0) {
337 numc[l] = 0;
338 lmmp_inc_1(numc, -bcyo);
339 } else {
340 numc[l] = numc[0] < (mp_limb_t)bcyo;
341 lmmp_dec_1(numc, bcyo - numc[l]);
342 }
343
344 coef[wing] = numc;
345 ms->temp_coef = numb;
346}

348static void lmmp_fft_b1_(fft_memstack* ms, mp_ptr* coef, mp_size_t dis, mp_size_t k, mp_size_t w, mp_size_t w0) {
349 if (k == 1)
350 lmmp_fft_bfy_(ms, coef, dis, w0);
351 else {
352 k -= 2;
353 mp_size_t Kq = dis << k;
354 for (mp_size_t i = 0; i < Kq; i += dis) {
355 lmmp_fft_bfy_(ms, coef + i, 2 * Kq, i * w + w0);
356 lmmp_fft_bfy_(ms, coef + i + Kq, 2 * Kq, (i + Kq) * w + w0);
357 lmmp_fft_bfy_(ms, coef + i, Kq, 2 * (i * w + w0));
358 lmmp_fft_bfy_(ms, coef + i + Kq * 2, Kq, 2 * (i * w + w0));
359 }
360 if (k > 0) {
361 lmmp_fft_b1_(ms, coef, dis, k, 4 * w, 4 * w0);
362 lmmp_fft_b1_(ms, coef + Kq, dis, k, 4 * w, 4 * w0);
363 lmmp_fft_b1_(ms, coef + Kq * 2, dis, k, 4 * w, 4 * w0);
364 lmmp_fft_b1_(ms, coef + Kq * 3, dis, k, 4 * w, 4 * w0);
365 }
366 }

369static void lmmp_fft_4_(fft_memstack* ms, mp_ptr* coef, mp_size_t k, mp_size_t w) {
370 if (k == 1)
371 lmmp_fft_bfy_(ms, coef, 1, 0);
372 else {
373 k -= 2;
374 mp_size_t Kq = ((mp_size_t)1) << k;
375 for (mp_size_t i = 0; i < Kq; ++i) {
376 lmmp_fft_bfy_(ms, coef + i, Kq * 2, i * w);
377 lmmp_fft_bfy_(ms, coef + i + Kq, Kq * 2, (i + Kq) * w);
378 lmmp_fft_bfy_(ms, coef + i, Kq, 2 * i * w);
379 lmmp_fft_bfy_(ms, coef + i + 2 * Kq, Kq, 2 * i * w);
380 }
381 if (k > 0) {
382 lmmp_fft_4_(ms, coef, k, w * 4);
383 lmmp_fft_4_(ms, coef + Kq, k, w * 4);
384 lmmp_fft_4_(ms, coef + 2 * Kq, k, w * 4);
385 lmmp_fft_4_(ms, coef + 3 * Kq, k, w * 4);
386 }
387 }

391 mp_size_t k1 = k >> 1;
392 k -= k1;
393 mp_size_t Kp = ((mp_size_t)1) << k;
394 mp_size_t Kq = ((mp_size_t)1) << k1;
395
396 for (mp_size_t i = 0; i < Kp; ++i) lmmp_fft_b1_(ms, coef + i, Kp, k1, w, i * w);
397
398 for (mp_size_t i = 0; i < Kq; ++i) lmmp_fft_4_(ms, coef + Kp * i, k, w * Kq);
399}

454void lmmp_mul_fermat_recombine_(
455 fft_memstack* ms,
456 mp_ptr dst,
457 mp_ptr* pfca,
458 mp_size_t K,
459 mp_size_t k,
460 mp_size_t n,
461 mp_size_t M,
462 mp_size_t rn
463) {
464 mp_size_t rhead = 0, nlen = ms->lenw + 1;
465 mp_slimb_t borrow = 0, maxc = 0;
466
467 for (mp_size_t i = 0; i < K; ++i) {
468 lmmp_fft_shr_coef_(ms, pfca + i, (i * n >> k) + k);
469 mp_ptr nums = pfca[i];
470
471 if (nums[nlen - 1]) {
472 lmmp_dec(nums);
473 --nums[nlen - 1];
474 }
475 if (nums[nlen - 1] == 0 && nums[nlen - 2] >> (LIMB_BITS - 1)) {
476 lmmp_dec(nums);
477 --nums[nlen - 1];
478 }
479
480 if (borrow) {
481 mp_size_t brshift = borrow - 1 + n - M;
482 mp_size_t bshl = brshift & (LIMB_BITS - 1);
483 brshift /= LIMB_BITS;
484 --nums[nlen - 1];
485 lmmp_dec_1(nums + brshift, (mp_limb_t)1 << bshl);
486 ++nums[nlen - 1];
487 }
488 borrow = -nums[nlen - 1];
489 nums[nlen - 1] = 0;
490
491 mp_size_t roffset = i * M;
492 mp_size_t shl = roffset & (LIMB_BITS - 1);
493 roffset /= LIMB_BITS;
494
495 if (shl)
496 lmmp_shl_(nums, nums, nlen, shl);
497
498 if (i == 0) {
499 // 重组先写入 [dst,rn]，最高 limb 在处理进位时写入。
500 rhead = LMMP_MIN(nlen, rn);
501 lmmp_copy(dst, nums, rhead);
502 } else if (roffset + nlen <= rn) {
503 lmmp_add_(dst + roffset, nums, nlen, dst + roffset, rhead - roffset);
504 rhead = roffset + nlen;
505 } else {
506 maxc += lmmp_add_(dst + roffset, nums, rn - roffset, dst + roffset, rhead - roffset);
507 maxc -= lmmp_sub_(dst, dst, rn, nums + rn - roffset, nlen + roffset - rn);
508 rhead = rn;
509 }
