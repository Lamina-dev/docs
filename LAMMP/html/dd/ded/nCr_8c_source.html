33mp_size_t lmmp_nCr_size_(uint n, uint r, mp_bitcnt_t* restrict bits) {
34 lmmp_param_assert(r <= n / 2);
35 lmmp_param_assert(bits != NULL);
36 // nCr = n!/(r!*(n-r)!)，分子ceil、分母floor保证不低估
37 // log2(m!) = lgamma(m+1)/ln2
38 const double ln2 = 0.69314718055994531;
39 mp_size_t rn;
40 if (r < 4 || n < ODD_FACTORIAL_SIZE) {
41 rn = 3;
42 } else if (n == MP_UINT_MAX) {
43 uint mean = n - r / 2 + 1;
44 mp_size_t l1, l2;
45 l1 = lmmp_pow_1_size_(mean, r);
46 l2 = (mp_size_t)floor(lgamma((double)r + 1) / ln2);
47 l2 /= LIMB_BITS;
48 rn = l1 - l2;
49 } else {
50 mp_size_t l1, l2, l3;
51 l1 = (mp_size_t)ceil(lgamma((double)n + 1) / ln2);
52 l2 = (mp_size_t)floor(lgamma((double)r + 1) / ln2);
53 if (n - r < ODD_FACTORIAL_SIZE)
54 l3 = 0;
55 else
56 l3 = (mp_size_t)floor(lgamma((double)(n - r) + 1) / ln2);
57 rn = l1 - l2 - l3;
58 rn = (rn + LIMB_BITS - 1) / LIMB_BITS;
59 }
60 (*bits) = n - lmmp_limb_popcnt_(n);
61 (*bits) -= r - lmmp_limb_popcnt_(r);
62 (*bits) -= n - r - lmmp_limb_popcnt_(n - r);
63 return rn + 2; // more 2 limb
64}

81static inline uint factor_size_int(mp_size_t rn, uint n) {
82 /*
83 我们可以知道，nCr的大小一定不会超过B^rn，因此，B^rn的可以含有的质因数数量即为nCr可以含有的质因数数量的上限。
84 同时，我们这里只计算的是奇数部分，比如我们可以用B^rn可以含有的3的质因数个数来估计nCr的质因数种类数，
85 这是一个绝对上界，同时在不平衡时比pi(n)这个平方上界要紧得多。当然即使是这个上界，实际的质因数个数也可能远远
86 小于这个上界。一个改进想法是，我们使用更大一点的质数，对于n>0xffff，我们选取这个质数为251，
87 而log(B)/log(251)约等于8.02855802854906，我们近似视为8，这也是这里的常数的由来，当然，此估计可能存在低估，
88 但是经过大量的校验，我们未发现任何低估的反例。
89 为了同时处理不平衡与不平衡的情况，我们这里对两个估计进行比较，取较小的一个作为最终结果。不平衡时，approx1要更紧一些。
90 */
91 // 此处假定了LIMB_BITS为64
92 ulong approx1 = rn * 8;
93 lmmp_debug_assert(rn * 8 <= MP_UINT_MAX);
94 ulong approx2 = lmmp_prime_size_(n);
95 return approx1 < approx2 ? approx1 : approx2;
96}

98static inline ushort factor_size_short(mp_size_t rn) {
99 /*
100 经过大量的校验，*8即使是在ushort输入下，也未见低估，但是为了留有冗余，我们还是选择*10，大致相当于质数83。
101 */
102 // 此处假定了LIMB_BITS为64
103 uint approx1 = rn * 10;
104 lmmp_debug_assert(rn * 10 <= MP_USHORT_MAX);
105 return (ushort)approx1;
106}

108static inline uint count_factors(fac_ptr fac, uint nfactors, uint n, uint r, uint nr, uint p) {
109 uint pn = n;
110 uint e = 0;
111 ulong inv = MP_ULONG_MAX / p + 1;
112 while (pn > 0) {
113 _udiv32by32_q_preinv(pn, pn, inv);
114 e += pn;
115 }
116 pn = r;
117 while (pn > 0) {
118 _udiv32by32_q_preinv(pn, pn, inv);
119 e -= pn;
120 }
121 pn = nr;
122 while (pn > 0) {
123 _udiv32by32_q_preinv(pn, pn, inv);
124 e -= pn;
125 }
126 if (e > 0) {
127 fac[nfactors].f = p;
128 fac[nfactors++].j = e;
129 }
130 return nfactors;
131}

136static mp_size_t lmmp_odd_nPr_product_(mp_ptr restrict dst, mp_size_t rn, uint n, uint r) {
137 TEMP_DECL;
138 ulongp restrict limbs = TALLOC_TYPE(r / 2 + 1, ulong);
139 mp_size_t limbn = 0;
140 ulong t = 1;
141 uint v;
142 mp_bitcnt_t cnt = 0;
143 // 需注意 n==UINT_MAX 的情况
144 uint ibase = n - r + 1;
145 for (uint k = 0; k < r; ++k) {
146 ctz_shr_u32(v, ibase + k, cnt);
147 t *= v;
148 if (t > MP_UINT_MAX) {
149 limbs[limbn++] = t;
150 t = 1;
151 }
152 }
153 if (t != 1)
154 limbs[limbn++] = t;
155
156 if (rn >= limbn) {
157 mp_ptr restrict tp = TALLOC_TYPE(limbn, mp_limb_t);
158 rn = lmmp_elem_mul_ulong_(dst, limbs, limbn, tp);
159 } else {
160 mp_ptr restrict tp = TALLOC_TYPE(limbn * 2, mp_limb_t);
161 rn = lmmp_elem_mul_ulong_(tp, limbs, limbn, tp + limbn);
162 lmmp_copy(dst, tp, rn);
163 }
164 TEMP_FREE;
165 return rn;
166}

184static mp_size_t lmmp_odd_nCr_div_(mp_ptr restrict dst, mp_size_t rn, bino_choose_t* restrict ctx) {
185 lmmp_param_assert(rn > 0 && dst != NULL);
186 TEMP_DECL;
187
188 mp_ptr restrict nPr = TALLOC_TYPE(ctx->nPr_n, mp_limb_t);
189 mp_size_t shw1 = ctx->nPr_bits / LIMB_BITS;
190 mp_size_t nPr_n = lmmp_odd_nPr_product_(nPr, ctx->nPr_n - shw1, ctx->n, ctx->r);
191
192 mp_size_t shw2 = ctx->fac_bits / LIMB_BITS;
193 mp_ptr restrict fac = TALLOC_TYPE(ctx->fac_n, mp_limb_t);
194 mp_size_t fac_n = lmmp_odd_factorial_(fac, ctx->fac_n - shw2, ctx->r);
195
196 lmmp_debug_assert(rn >= nPr_n - fac_n + 1);
197 lmmp_divexact_(dst, nPr, nPr_n, fac, fac_n);
198 rn = nPr_n - fac_n + 1;
199 rn -= dst[rn - 1] == 0;
200 TEMP_FREE;
201 return rn;
202}

204mp_size_t lmmp_odd_nCr_ushort_(mp_ptr restrict dst, mp_size_t rn, uint n, uint r) {
205 lmmp_param_assert(n <= MP_USHORT_MAX);
206 lmmp_param_assert(rn > 0 && dst != NULL);
207 lmmp_param_assert(r <= n / 2);
208 if (r < ODD_FACTORIAL_SIZE) {
209 rn = lmmp_odd_nPr_ushort_(dst, rn, n, r);
210 mp_limb_t t = 0;
211 lmmp_odd_nPr_ushort_(&t, 1, r, r);
212 div_1(dst, rn, t);
213 return rn;
214 } else if (rn < BINOMIAL_RN_BASECASE_THRESHOLD) {
215 if (r <= 4 || n > 0xfff) {
216 dst[0] = 1;
217 rn = 1;
218 ulong t, v;
219 mp_bitcnt_t cnt;
220 for (ulong i = 1; i <= r; ++i) {
221 t = n - i + 1;
222 ctz_shr_u64(v, t, cnt);
223 mul_1(dst, rn, v);
224 ctz_shr_u64(v, i, cnt);
225 div_1(dst, rn, v);
226 }
227 return rn;
228 } else {
229 dst[0] = 1;
230 rn = 1;
231 ulong i = 1;
232 ulong t = 1, d = 1, v;
233 mp_bitcnt_t cnt;
234 for (; i <= (ulong)r - 4; i += 4) {
235 // 相邻四个数必含有3这个公共因子
236 d = i * (i + 1) * (i + 2) * (i + 3);
237 d /= 3;
238 t = (n - i + 1) * (n - i) * (n - i - 1) * (n - i - 2);
239 t /= 3;
240 ctz_shr_u64(v, t, cnt);
241 mul_1(dst, rn, v);
242 ctz_shr_u64(v, d, cnt);
243 div_1(dst, rn, v);
244 }
245 for (; i <= r; ++i) {
246 t = n - i + 1;
247 ctz_shr_u64(v, t, cnt);
248 mul_1(dst, rn, v);
249 ctz_shr_u64(v, i, cnt);
250 div_1(dst, rn, v);
251 }
252 return rn;
253 }
254 } else {
255 TEMP_DECL;
256 ushort primen = lmmp_prime_cnt16_(n);
257 ushort nfactors = factor_size_short(rn);
258 nfactors = primen < nfactors ? primen : nfactors;
259 fac_ptr restrict fac = TALLOC_TYPE(nfactors, fac_t);
260 ushort nr = n - r;
261 nfactors = 0;
262 for (ushort i = 1; i < primen; ++i) {
263 ushort p = prime_short_table[i];
264 nfactors = count_factors(fac, nfactors, n, r, nr, p);
265 }
266
267 rn = lmmp_factors_mul_ushort_(dst, rn, fac, nfactors);
268
269 TEMP_FREE;
270 return rn;
271 }

274mp_size_t lmmp_odd_nCr_uint_(mp_ptr restrict dst, mp_size_t rn, uint n, uint r) {
275 lmmp_param_assert(r <= (n / 2));
276 lmmp_param_assert(rn > 0 && dst != NULL);
277 if (r <= 3 || (n > 0xfffffff && rn < BINOMIAL_RN_BASECASE_THRESHOLD)) {
278 dst[0] = 1;
279 rn = 1;
280 ulong t, v;
281 mp_bitcnt_t cnt;
282 for (ulong i = 1; i <= r; ++i) {
283 t = n - i + 1;
284 ctz_shr_u64(v, t, cnt);
285 mul_1(dst, rn, v);
286 ctz_shr_u64(v, i, cnt);
287 div_1(dst, rn, v);
288 }
289 return rn;
290 } else if (rn < BINOMIAL_RN_BASECASE_THRESHOLD) {
291 dst[0] = 1;
292 rn = 1;
293 ulong i = 1;
294 ulong t = 1, d = 1, v;
295 mp_bitcnt_t cnt;
296 for (; i <= (ulong)r - 2; i += 2) {
297 d = i * (i + 1);
298 t = (n - i + 1) * (n - i);
299 ctz_shr_u64(v, t, cnt);
300 mul_1(dst, rn, v);
301 ctz_shr_u64(v, d, cnt);
302 div_1(dst, rn, v);
303 }
304 for (; i <= r; ++i) {
305 t = n - i + 1;
306 ctz_shr_u64(v, t, cnt);
307 mul_1(dst, rn, v);
308 ctz_shr_u64(v, i, cnt);
309 div_1(dst, rn, v);
310 }
311 return rn;
312 } else {
313 bino_choose_t ctx;
314 ctx.nPr_n = lmmp_nPr_size_(n, r, &ctx.nPr_bits);
315 ctx.fac_n = lmmp_factorial_size_(r, &ctx.fac_bits);
316 if (BINOMIAL_DIV_K_THRESHOLD * ctx.nPr_n > BINOMIAL_DIV_B_THRESHOLD * ctx.fac_n) {
317 /* 这个调优值是在近似忽略了质数表的初始化开销，主要瓶颈集中在质数表的遍历的情况下测得的 */
318 /* 因此，当质数表未初始化时，这个调优值将无法代表真实性能边界 */
319 ctx.n = n;
320 ctx.r = r;
321 return lmmp_odd_nCr_div_(dst, rn, &ctx);
322 } else {
323 lmmp_prime_int_table_init_(n);
324 TEMP_B_DECL;
325 uint nfactors = factor_size_int(rn, n);
326 fac_ptr restrict fac = BALLOC_TYPE(nfactors, fac_t);
327 uint nr = n - r;
328
329 nfactors = 0;
330 prime_cache_t cache;
331 lmmp_prime_cache_init_(&cache, n);
332 while (cache.is_end == 0) {
333 lmmp_prime_cache_next_(&cache);
334 for (uint i = 0; i < cache.size; ++i) {
335 nfactors = count_factors(fac, nfactors, n, r, nr, cache.pp[i]);
336 }
337 }
338 lmmp_prime_cache_free_(&cache);
339
340 rn = lmmp_factors_mul_(dst, rn, fac, nfactors);
341
342 TEMP_B_FREE;
343 return rn;
344 }
345 }
346}

348mp_size_t lmmp_nCr_(mp_ptr restrict dst, mp_bitcnt_t bits, mp_size_t rn, uint n, uint r) {
349 lmmp_param_assert(r <= (n / 2));
350 lmmp_param_assert(dst != NULL);
351 mp_size_t shw = bits / LIMB_BITS;
352 lmmp_param_assert(rn > shw);
353
354 bits %= LIMB_BITS;
355 lmmp_zero(dst, shw);
356
357 if (n <= NCR_SHORT_LIMIT)
358 rn = lmmp_odd_nCr_ushort_(dst + shw, rn - shw, n, r);
359 else
360 rn = lmmp_odd_nCr_uint_(dst + shw, rn - shw, n, r);
361
362 if (bits > 0) {
363 dst[shw + rn] = lmmp_shl_(dst + shw, dst + shw, rn, bits);
364 rn += shw + 1;
365 rn -= dst[rn - 1] == 0;
366 } else {
367 rn += shw;
368 }
369 return rn;
