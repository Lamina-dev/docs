22void lmmp_mullo_fft_(mp_ptr dst, mp_srcptr numa, mp_srcptr numb, mp_size_t n, mp_ptr scratch) {
23 lmmp_param_assert(n > 0);
24 mp_size_t hn = lmmp_fft_next_size_((n + n + 1) >> 1);
25 lmmp_debug_assert(n <= hn);
26 mp_ptr tp = ALLOC_TYPE(hn + 1, mp_limb_t);
27
28 if (numa == numb) {
29 lmmp_sqr_mersenne_(scratch, hn, numa, n);
30 lmmp_sqr_fermat_(tp, hn, numa, n);
31 } else {
32 lmmp_mul_mersenne_(scratch, hn, numa, n, numb, n);
33 lmmp_mul_fermat_(tp, hn, numa, n, numb, n);
34 }
35
36 mp_limb_t cy = lmmp_shr1add_nc_(scratch, scratch, tp, hn, tp[hn]);
37 cy <<= LIMB_BITS - 1;
38 scratch[hn - 1] += cy;
39 if (scratch[hn - 1] < cy)
40 lmmp_inc(scratch);
41
42 if (n == hn) {
43 cy = tp[hn] + lmmp_sub_n_(scratch + hn, scratch, tp, hn);
44 // cy==1 means [tp,hn+1]!=0, then [dst,hn]!=0
45 // cy==2 is impossible since [tp,hn+1] is normalized.
46 // so the following dec won't overflow.
47 lmmp_dec_1(scratch, cy);
48 } else {
49 mp_size_t n2 = 2 * n;
50 cy = lmmp_sub_n_(scratch + hn, scratch, tp, n2 - hn);
51 cy = tp[hn] + lmmp_sub_nc_(tp + n2 - hn, scratch + n2 - hn, tp + n2 - hn, 2 * hn - n2, cy);
52 cy = lmmp_sub_1_(scratch, scratch, n2, cy);
53 }
54 lmmp_free(tp);
55 lmmp_copy(dst, scratch, n);
56}

58/*
59 <---t---><---m--->
60 |--a1---|---a0---|
61 |--b1---|---b0---|
62
63 ,
64 |\
65 | \
66 | \
67 +-----,
68 | |
69 | |\
70 | | \
71 | | \
72 +-----+---`
73 ^ m ^ t ^
74
75 此算法是一种不平衡分块的算法，朴素的想法是计算平衡分块，计算一次完整的乘法，然后两次递归的调用此函数计算低位，
76 事实上，我们也可以不平衡的分块，以减少递归深度，具体分析如下：
77 取a和b的低位一定宽度为m，高位宽度为t，则有：
78 计算一次完整的平衡乘法m，然后递归调用计算mullo，长度为t
79 复杂度模型：
80 ML(n) = 2*ML(a*n) + M((1-a)*n)
81 其中ML为mullo的复杂度，M为mul_n的复杂度
82 我们可以假定 M(n)=O(n^e) 即多项式复杂度
83 则有：
84 ML(n) = C(a) * n^e
85 C(a) = a^e / (1-2*(1-a)^e)
86 我们希望C(a)尽可能小，即希望ML(n)尽可能小，则有：
87 a_opt = 1 - 2^(-1/(e-1))
88 e=log(3)/log(2) [Toom-2] -> a ~= 0.694
89 e=log(5)/log(3) [Toom-3] -> a ~= 0.775
90 e=log(7)/log(4) [Toom-4] -> a ~= 0.820
91 e=log(11)/log(6) [Toom-6] -> a ~= 0.871
92 e=log(15)/log(8) [Toom-8] -> a ~= 0.899
93*/

98void lmmp_mullo_dc_(
99 mp_ptr restrict dst,
100 mp_srcptr restrict numa,
101 mp_srcptr restrict numb,
102 mp_ptr restrict tp,
103 mp_size_t n
104) {
105 if (n < MULLO_BASECASE_THRESHOLD) {
106 lmmp_mullo_basecase_(dst, numa, numb, n);
107 return;
108 } else {
109 mp_size_t m, t;
110 if (n < MUL_TOOM33_THRESHOLD) {
111 m = 25 * n / 36;
112 } else if (n < MUL_TOOM44_THRESHOLD) {
113 m = 31 * n / 40;
114 } else if (n < MUL_TOOM66_THRESHOLD) {
115 m = 32 * n / 39;
116 } else if (n < MUL_TOOM88_THRESHOLD) {
117 m = 27 * n / 31;
118 } else {
119 m = 9 * n / 10;
120 }
121 t = n - m;
122 lmmp_debug_assert(2 * n > 4 * t);
123
124#define a0 (numa) // [numa, m]
125#define a1 (numa + m) // [numa+m, t]
126#define b0 (numb) // [numb, m]
127#define b1 (numb + m) // [numb+m, t]
128#define c0 (dst) // [dst, m]
129#define c1 (dst + m) // [dst+m, t]
130#define lo1 (tp) // [tp, t]
131#define lo2 (tp + t) // [tp+t, t]
132#define scratch (tp + 2 * t) // [tp+2*t, 2*t]
133 lmmp_mul_n_(tp, a0, b0, m);
134 lmmp_copy(c0, tp, n);
135 lmmp_mullo_dc_(lo1, a1, b0, scratch, t);
136 lmmp_mullo_dc_(lo2, a0, b1, scratch, t);
137 lmmp_add_n_(c1, c1, lo1, t);
138 lmmp_add_n_(c1, c1, lo2, t);
139 return;
140 }
141#undef a0
142#undef a1
143#undef b0
144#undef b1
145#undef c0
146#undef c1
147#undef lo1
148#undef lo2
149#undef scratch
150}

153 if (n < MULLO_BASECASE_THRESHOLD) {
154 lmmp_mullo_basecase_(dst, numa, numa, n);
155 return;
156 } else {
157 mp_size_t m, t;
158 if (n < MUL_TOOM33_THRESHOLD) {
159 m = 25 * n / 36;
160 } else if (n < MUL_TOOM44_THRESHOLD) {
161 m = 31 * n / 40;
162 } else if (n < MUL_TOOM66_THRESHOLD) {
163 m = 32 * n / 39;
164 } else if (n < MUL_TOOM88_THRESHOLD) {
165 m = 27 * n / 31;
166 } else {
167 m = 9 * n / 10;
168 }
169 t = n - m;
170
171#define a0 (numa)
172#define a1 (numa + m)
173#define c0 (dst)
174#define c1 (dst + m)
175#define lo (tp) // [tp, t]
176#define scratch (tp + t) // [tp+t, 2*t]
177 lmmp_sqr_(tp, a0, m);
178 lmmp_copy(c0, tp, n);
179 lmmp_mullo_dc_(lo, a0, a1, scratch, t);
180 lmmp_addshl1_n_(c1, c1, lo, t);
181 }

183#undef a1
184#undef c0
185#undef c1
186#undef lo
