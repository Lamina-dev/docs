30/**
31 * @brief mullo 双域 CRT 尾部：由梅森域积与费马域积重构低位乘积
32 * [dst,n] = P mod B^n，其中 P = [numa,n]*[numb,n] mod B^(2n)，
33 * [lp,hn] = P mod B^hn-1（梅森域结果），[hp,hn+1] = P mod B^hn+1（费马域结果）
34 * @param dst 输出结果（n 个 limb）
35 * @param lp 输入输出：梅森域结果，兼作重构工作区（需 2*n 个 limb 容量；
36 * n==hn 时写入 [hn,2hn)，n<hn 时写入 [hn,2n)）
37 * @param hp 输入输出：费马域结果（hn+1 个 limb；n<hn 时 [2n-hn,hn) 段被复写）
38 * @param n 低位乘积截断长度
39 * @param hn 变换阶数，hn = lmmp_fft_next_size_(n)
40 * @warning n>0, n<=hn, sep(dst,[lp|hp]), lp 容量 >= 2*n, hp 容量 >= hn+1
41 * @note 先以 shr1add 组合两域残差得 X = ([lp,hn]+[hp,hn+1])/2 的环内表示
42 * （B^hn 在 B^hn-1 意义下 ≡ 1，故 hp 顶 limb 以进位并入），随后按
43 * n==hn（全积重构）或 n<hn（2n 位部分重构）执行分段减法链完成精确 CRT，
44 * 减法链的借位在 [lp,2n) 工作区内闭合，不会越界
45 */
46static void lmmp_mullo_crt_(mp_ptr dst, mp_ptr lp, mp_ptr hp, mp_size_t n, mp_size_t hn) {
47 mp_limb_t cy = lmmp_shr1add_nc_(lp, lp, hp, hn, hp[hn]);
48 cy <<= LIMB_BITS - 1;
49 lp[hn - 1] += cy;
50 if (lp[hn - 1] < cy)
51 lmmp_inc(lp);
52
53 if (n == hn) {
54 cy = hp[hn] + lmmp_sub_n_(lp + hn, lp, hp, hn);
55 // cy==1 means [hp,hn+1]!=0, then [lp,hn]!=0
56 // cy==2 is impossible since [hp,hn+1] is normalized.
57 // so the following dec won't overflow.
58 lmmp_dec_1(lp, cy);
59 } else {
60 mp_size_t n2 = 2 * n;
61 cy = lmmp_sub_n_(lp + hn, lp, hp, n2 - hn);
62 cy = hp[hn] + lmmp_sub_nc_(hp + n2 - hn, lp + n2 - hn, hp + n2 - hn, 2 * hn - n2, cy);
63 cy = lmmp_sub_1_(lp, lp, n2, cy);
64 }
65 lmmp_copy(dst, lp, n);
66}

68void lmmp_mullo_fft_(mp_ptr dst, mp_srcptr numa, mp_srcptr numb, mp_size_t n, mp_ptr scratch) {
69 lmmp_param_assert(n > 0);
70 mp_size_t hn = lmmp_fft_next_size_(n);
71 lmmp_debug_assert(n <= hn);
72 mp_ptr tp = ALLOC_TYPE(hn + 1, mp_limb_t);
73
74 if (numa == numb) {
75 lmmp_sqr_mersenne_fft_(scratch, hn, numa, n);
76 lmmp_sqr_fermat_(tp, hn, numa, n);
77 } else {
78 lmmp_mul_mersenne_fft_(scratch, hn, numa, n, numb, n);
79 lmmp_mul_fermat_(tp, hn, numa, n, numb, n);
80 }
81
82 lmmp_mullo_crt_(dst, scratch, tp, n, hn);
83 lmmp_free(tp);
84}

113/*
114 <---t---><---m--->
115 |--a1---|---a0---|
116 |--b1---|---b0---|
117
118 ,
119 |\
120 | \
121 | \
122 +-----,
123 | |
124 | |\
125 | | \
126 | | \
127 +-----+---`
128 ^ m ^ t ^
129
130 此算法是一种不平衡分块的算法，朴素的想法是计算平衡分块，计算一次完整的乘法，然后两次递归的调用此函数计算低位，
131 事实上，我们也可以不平衡的分块，以减少递归深度，具体分析如下：
132 取a和b的低位一定宽度为m，高位宽度为t，则有：
133 计算一次完整的平衡乘法m，然后递归调用计算mullo，长度为t
134 复杂度模型：
135 ML(n) = 2*ML(a*n) + M((1-a)*n)
136 其中ML为mullo的复杂度，M为mul_n的复杂度
137 我们可以假定 M(n)=O(n^e) 即多项式复杂度
138 则有：
139 ML(n) = C(a) * n^e
140 C(a) = a^e / (1-2*(1-a)^e)
141 我们希望C(a)尽可能小，即希望ML(n)尽可能小，则有：
142 a_opt = 1 - 2^(-1/(e-1))
143 e=log(3)/log(2) [Toom-2] -> a ~= 0.694
144 e=log(5)/log(3) [Toom-3] -> a ~= 0.775
145 e=log(7)/log(4) [Toom-4] -> a ~= 0.820
146 e=log(11)/log(6) [Toom-6] -> a ~= 0.871
147 e=log(15)/log(8) [Toom-8] -> a ~= 0.899
148*/

150void lmmp_mullo_dc_(
151 mp_ptr restrict dst,
152 mp_srcptr restrict numa,
153 mp_srcptr restrict numb,
154 mp_ptr restrict tp,
155 mp_size_t n
156) {
157 if (n < MULLO_BASECASE_THRESHOLD) {
158 lmmp_mullo_basecase_(dst, numa, numb, n);
159 return;
160 } else {
161 mp_size_t m, t;
162 if (n < MUL_TOOM33_THRESHOLD) {
163 m = 25 * n / 36;
164 } else if (n < MUL_TOOM44_THRESHOLD) {
165 m = 31 * n / 40;
166 } else if (n < MUL_TOOM6H_THRESHOLD) {
167 m = 32 * n / 39;
168 } else if (n < MUL_TOOM8H_THRESHOLD) {
169 m = 27 * n / 31;
170 } else {
171 m = 9 * n / 10;
172 }
173 t = n - m;
174 lmmp_debug_assert(2 * n > 4 * t);
175
176#define a0 (numa) // [numa, m]
177#define a1 (numa + m) // [numa+m, t]
178#define b0 (numb) // [numb, m]
179#define b1 (numb + m) // [numb+m, t]
180#define c0 (dst) // [dst, m]
181#define c1 (dst + m) // [dst+m, t]
182#define lo1 (tp) // [tp, t]
183#define lo2 (tp + t) // [tp+t, t]
184#define scratch (tp + 2 * t) // [tp+2*t, 2*t]
185 lmmp_mul_n_(tp, a0, b0, m);
186 lmmp_copy(c0, tp, n);
187 lmmp_mullo_dc_(lo1, a1, b0, scratch, t);
188 lmmp_mullo_dc_(lo2, a0, b1, scratch, t);
189 lmmp_add_n_(c1, c1, lo1, t);
190 lmmp_add_n_(c1, c1, lo2, t);
191 return;
192 }
193#undef a0
194#undef a1
195#undef b0
196#undef b1
197#undef c0
198#undef c1
199#undef lo1
200#undef lo2
201#undef scratch
202}
