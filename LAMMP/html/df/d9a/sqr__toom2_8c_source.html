29/*
30Evaluate in: -1, 0, +inf
31
32 <-s--><--n-->
33 |-a1-|--a0--|
34
35v0 = a0 ^2 # A(0)^2
36vm1 = (a0-a1)^2 # A(-1)^2
37vinf= a1 ^2 # A(inf)^2
38*/

40void lmmp_sqr_toom2_(mp_ptr restrict dst, mp_srcptr restrict numa, mp_size_t na) {
41 lmmp_param_assert(na > 0);
42 lmmp_param_assert(dst != NULL);
43 lmmp_param_assert(numa!= NULL);
44 TEMP_S_DECL;
45 mp_size_t s = na >> 1, n = na - s;
46 mp_limb_t* vm1 = SALLOC_TYPE(2 * n, mp_limb_t);
47 mp_slimb_t cy, cy2;
48
49#define a0 numa
50#define a1 (numa + n)
51#define asm1 dst
52
53 if (s == n) {
54 if (lmmp_cmp_(a0, a1, n) < 0)
55 lmmp_sub_n_(asm1, a1, a0, n);
56 else
57 lmmp_sub_n_(asm1, a0, a1, n);
58 } else { // s==n-1
59 if (a0[s] == 0 && lmmp_cmp_(a0, a1, s) < 0) {
60 lmmp_sub_n_(asm1, a1, a0, s);
61 asm1[s] = 0;
62 } else
63 asm1[s] = a0[s] - lmmp_sub_n_(asm1, a0, a1, s);
64 }
65
66 lmmp_sqr_(vm1, asm1, n);
67
68#undef asm1
69#define v0 dst
70#define vinf (dst + 2 * n)
71
72 lmmp_sqr_(v0, a0, n);
73
74 lmmp_sqr_(vinf, a1, s);
75
76 cy = lmmp_add_n_(dst + 2 * n, v0 + n, vinf, n);
77 cy2 = cy + lmmp_add_n_(dst + n, dst + 2 * n, v0, n);
78 cy += lmmp_add_(dst + 2 * n, dst + 2 * n, n, vinf + n, s + s - n);
79
80 cy -= lmmp_sub_n_(dst + n, dst + n, vm1, 2 * n);
81
82 // no overflow.
83 lmmp_inc_1(dst + 2 * n, cy2);
84
85 if (cy < 0)
86 lmmp_dec(dst + 3 * n);
87 else
88 lmmp_inc_1(dst + 3 * n, cy);
89 TEMP_S_FREE;
90}
