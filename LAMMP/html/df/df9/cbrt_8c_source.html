24static inline void lmmp_cube_3_(mp_ptr restrict dst, mp_limb_t a) {
25 mp_limb_t t[2];
26 lmmp_mullh_(a, a, t);
27 lmmp_mullh_(t[0], a, dst);
28 lmmp_mullh_(t[1], a, t);
29 dst[1] += t[0];
30 dst[2] = t[1] + (dst[1] < t[0] ? 1 : 0);
31}

33mp_limb_t lmmp_cbrtapprox_3_(mp_limb_t a0, mp_limb_t a1, mp_limb_t a2) {
34 lmmp_param_assert(a1 > 0);
35 mp_limb_t x[2];
36 /* exact high 65 bits */
37 mp_limb_t a_hi;
38 mp_bitcnt_t bits;
39 if (a2 == 0) {
40 mp_bitcnt_t a1_bits = lmmp_limb_bits_(a1);
41 bits = LIMB_BITS + a1_bits;
42 a1_bits--;
43 if (a1_bits == 0)
44 a_hi = a0;
45 else
46 a_hi = (a1 << (LIMB_BITS - a1_bits)) | (a0 >> a1_bits);
47 } else {
48 mp_bitcnt_t a2_bits = lmmp_limb_bits_(a2);
49 bits = LIMB_BITS * 2 + a2_bits;
50 a2_bits--;
51 if (a2_bits == 0)
52 a_hi = a1;
53 else
54 a_hi = (a2 << (LIMB_BITS - a2_bits)) | (a1 >> a2_bits);
55 }
56 lmmp_debug_assert(bits >= 65);
57
58 x[1] = bits - 1;
59 x[0] = log2_fixed_64(a_hi);
60
61 mp_limb_t rem = lmmp_div_1_(x, x, 2, 3);
62 if (2 * rem >= 3) // round
63 lmmp_inc(x);
64
65 mp_bitcnt_t shift = x[1];
66 x[0] = exp2_fixed_64(x[0]);
67
68 lmmp_debug_assert(shift <= 64);
69 mp_limb_t r;
70 if (shift == 64)
71 r = LIMB_MAX;
72 else
73 r = (x[0] >> (64 - shift)) | (1ULL << shift);
74
75 // log2/exp2 固定精度近似可能高估 1；高估时回退一档，
76 // 低估 1 仍然满足 floor(cbrt(x))-[0|1] 的接口语义。
77 mp_limb_t t[3], a[3] = {a0, a1, a2};
78 lmmp_cube_3_(t, r);
79 if (lmmp_cmp_(t, a, 3) > 0)
80 --r;
81 return r;
82}

84mp_limb_t lmmp_cbrt_3_(mp_limb_t a0, mp_limb_t a1, mp_limb_t a2) {
85 lmmp_param_assert(a1 > 0);
86
87 mp_limb_t r = lmmp_cbrtapprox_3_(a0, a1, a2);
88 if (r == LIMB_MAX)
89 return LIMB_MAX;
90 mp_limb_t t[3], a[3] = {a0, a1, a2};
91
92 // 近似值可能低估或高估 1（以实测为准，不依赖理论单向偏差）。
93 lmmp_cube_3_(t, r);
94 if (lmmp_cmp_(t, a, 3) > 0)
95 return r - 1;
96
97 lmmp_cube_3_(t, r + 1);
98 if (lmmp_cmp_(t, a, 3) <= 0)
99 return r + 1;
100 return r;
101}

103/*
104 A = Ah * B^(3*lo) + Al
105
106 Ahr = floor(Ah^(1/3))
107 rk = Ah - Ahr^3
108 x_k = Ahr * B^lo
109
110 x_k+1 = (2*x_k + A / x_k^2 ) / 3
111 = x_k + (A / x_k^2 - x_k) / 3
112 = x_k + (A - x_k^3) / 3 * x_k^2
113 = Ahr * B^lo + (rk * B^(3*lo) + Al) / 3 * x_k^2
114 = Ahr * B^lo + Alr
115
116 let Alr = (rk * B^(3*lo) + Al) / 3 * x_k^2, R = (rk * B^(3*lo) + Al) mod 3 * x_k^2
117 such that (rk * B^(3*lo) + Al) = Alr * 3 * x_k^2 + R
118 ┌───────────────────────────────────────────────────────────────────────┐
119 = |Alr * 3 * Ahr^2*B^(2*lo) + R = R_correct + (Alr-1) * 3 * Ahr^2*B^(2*lo)|
120 └───────────────────────────────────────────────────────────────┬───────┘
121 r_k+1 = A - x_k+1^3 |
122 = Ah*B^(3*lo) + Al - Ahr^3*B^(3*lo) - 3*Alr*Ahr^2*B^(2*lo) - 3*Ahr*Alr^2*B^lo - Alr^3 |
123 = r_k * B^(3*lo) + Al - 3*Alr*Ahr^2*B^(2*lo) - 3*Ahr*Alr^2*B^lo - Alr^3 |
124 = R - 3*Ahr*Alr^2*B^lo - Alr^3 |
125 |
126 Alr is either correct or 1 too big. However, in practice, we have found that Alr can be |
127 overestimated by up to 2, and at this point, it can only be Alr = B^lo+1, which is different |
128 from the square root (where the square root Alr is at most B ^ lo) |
129 ┌────────────────────────────────────────────────┐ |
130 r_k+1 = R - 3*Ahr*(Alr-1)^2*B^lo - (Alr-1)^3 + | 3*Alr*Ahr^2*B^(2*lo) - 3*(Alr-1)*Ahr^2*B^(2*lo)├─┘
131 = R - 3*Ahr*Alr^2*B^lo - Alr^3 └────────────────────────────────────────────────┘
132(adjust) + 3*Ahr^2*B^(2*lo) + 6*Ahr*Alr*B^lo - 3*Ahr*B^lo + 3*Alr^2 - 3*Alr + 1
133*/

139void lmmp_cbrt_divide_(mp_ptr restrict dst, mp_ptr restrict numa, mp_size_t ns, mp_ptr restrict tp, int calr) {
140 lmmp_param_assert(ns > 0);
141 lmmp_param_assert(numa != NULL && dst != NULL && tp != NULL);
142 lmmp_param_assert(numa[3 * ns - 1] >= CBRT_DIVIDE_MIN);
143 if (ns == 1) {
144 dst[0] = lmmp_cbrt_3_(numa[0], numa[1], numa[2]);
145 if (calr) {
146 lmmp_cube_3_(tp, dst[0]);
147 lmmp_sub_n_(numa, numa, tp, 3);
148 }
149 } else {
150 mp_size_t lo = ns / 2, hi = ns - lo;
151#define Ahr (dst + lo) // [dst+lo, hi]
152#define rk (numa + 3 * lo) // [numa+3*lo, 2*hi+1]
153#define R (numa) // [numa, 2*ns+1]
154#define Ahr2 (tp) // [tp, 2*hi]
155#define Alr (tp + 2 * hi) // [tp + 2*hi, lo+1]
156#define Alr2 (tp + 2 * hi + lo) // [tp + 2*hi+lo, 2*lo]
157#define scratch (tp + 2 * hi + 3 * lo) // [tp + 2*hi+3*lo, hi+2*lo]
158
159 lmmp_cbrt_divide_(Ahr, rk, hi, tp, 1);
160
161 lmmp_sqr_(Ahr2, Ahr, hi);
162
163 /*
164 A / (3*x^2) = A / 3 / x^2
165 A % (3*x^2) = 3 * (A / 3 % x^2) + A % 3
166 */
167 mp_limb_t r = lmmp_div_1_(rk - lo, rk - lo, hi + 1 + ns, 3);
168 mp_limb_t qh = lmmp_div_s_(Alr, rk - lo, hi + 1 + ns, Ahr2, 2 * hi);
169 lmmp_debug_assert(qh == 0);
170 (rk - lo)[2 * hi] = lmmp_mul_1_(rk - lo, rk - lo, 2 * hi, 3);
171 lmmp_inc_1(rk - lo, r);
172 /*
173 我们根据 cbrt(A/B^3) == floor(cbrt(A)/B) 可以知道，如果Alr正确结果
174 必定被限制在B^lo以内，其至多高估1，因此Alr的最高位必定为0或1，而为1时，
175 即代表此时结果已经高估。
176 */
177 mp_limb_t adj = Alr[lo];
178 lmmp_debug_assert(adj == 0 || adj == 1);
179 if (adj > 0) {
180 // Alr[0] 仅可能为0或1，分别表示高估了1或2
181 lmmp_debug_assert(Alr[0] == 0 || Alr[0] == 1);
182 lmmp_fill(dst, 0, lo, LIMB_MAX);
183 if (calr == 0) return;
184 /*
185 x_k+1 = Ahr * B^lo + Alr
186 = Ahr * B^lo + B^lo - 1
187
188 r_k+1 = R - 3*Ahr*(B^lo-1)^2*B^lo - (B^lo-1)^3 + 3*Alr*Ahr^2*B^(2*lo) - 3*(Alr-adj)*Ahr^2*B^(2*lo)
189 = R - 3*Ahr*B^(3*lo) + 6*Ahr*B^(2*lo) - 3*Ahr*B^lo + 3*adj*Ahr^2*B^(2*lo)
190 - B^(3*lo) + 3*B^(2*lo) - 3*B^lo + 1
191 */
192 adj += Alr[0];
193
194 // - 3*Ahr*B^(3*lo)
195 mp_limb_t cy = lmmp_submul_1_(R + 3 * lo, Ahr, hi, 3);
196 r = lmmp_sub_1_(R + 2 * lo + ns, R + 2 * lo + ns, ns + 1 - 2 * lo, cy);
197
198 // + 6*Ahr*B^(2*lo)
199 cy = lmmp_addmul_1_(R + 2 * lo, Ahr, hi, 6);
200 r -= lmmp_add_1_(R + ns + lo, R + ns + lo, hi + 1, cy);
201
202 // - 3*Ahr*B^lo
203 cy = lmmp_submul_1_(R + lo, Ahr, hi, 3);
204 r += lmmp_sub_1_(R + ns, R + ns, ns + 1, cy);
205
206 // + 3*adj*Ahr^2*B^(2*lo)
207 cy = lmmp_addmul_1_(R + 2 * lo, Ahr2, 2 * hi, 3 * adj);
208 (R + 2 * ns)[0] += cy;
209 r -= (R + 2 * ns)[0] < cy;
210
211 // - B^(3*lo)
212 r -= lmmp_sub_1_(R + 3 * lo, R + 3 * lo, 2 * ns + 1 - 3 * lo, 1);
213
214 // + 3*B^(2*lo)
215 r -= lmmp_add_1_(R + 2 * lo, R + 2 * lo, 2 * hi + 1, 3);
216
217 // - 3*B^lo
218 r += lmmp_sub_1_(R + lo, R + lo, 2 * ns + 1 - lo, 3);
219
220 // + 1
221 r -= lmmp_add_1_(R, R, 2 * ns + 1, 1);
222
223 lmmp_debug_assert(r == 0);
224 } else {
225 lmmp_sqr_(Alr2, Alr, lo);
226 lmmp_mul_(scratch, Alr2, 2 * lo, Alr, lo);
227 r = lmmp_sub_(R, R, 2 * ns + 1, scratch, 3 * lo);
228
229 lmmp_mul_(scratch, Alr2, 2 * lo, Ahr, hi);
230 mp_limb_t b = lmmp_submul_1_(R + lo, scratch, 2 * lo + hi, 3);
231 r += lmmp_sub_1_(R + 2 * lo + ns, R + 2 * lo + ns, ns + 1 - 2 * lo, b);
232
233 if (calr == 0) {
234 if (r > 0)
235 lmmp_dec(Alr);
236 lmmp_copy(dst, Alr, lo);
237 return;
238 }
239 if (r > 0) {
240 // + 3*Alr^2
241 mp_limb_t cy = lmmp_addmul_1_(R, Alr2, 2 * lo, 3);
242 r -= lmmp_add_1_(R + 2 * lo, R + 2 * lo, 2 * hi + 1, cy);
243
244 // - 3*Alr
245 cy = lmmp_submul_1_(R, Alr, lo, 3);
246 r += lmmp_sub_1_(R + lo, R + lo, 2 * ns + 1 - lo, cy);
247
248 // + 6*Ahr*Alr*B^lo
249 lmmp_mul_(scratch, Ahr, hi, Alr, lo);
250 cy = lmmp_addmul_1_(R + lo, scratch, ns, 6);
251 r -= lmmp_add_1_(R + ns + lo, R + ns + lo, ns + 1 - lo, cy);
252
253 // + 3*Ahr^2*B^(2*lo)
254 cy = lmmp_addmul_1_(R + 2 * lo, Ahr2, 2 * hi, 3);
255 // r -= lmmp_add_1_(R + 2 * ns, R + 2 * ns, 1, cy);
256 (R + 2 * ns)[0] += cy;
257 r -= (R + 2 * ns)[0] < cy;
258
259 // + 1
260 r -= lmmp_add_1_(R, R, 2 * ns + 1, 1);
261
262 // - 3*Ahr*B^lo
263 cy = lmmp_submul_1_(R + lo, Ahr, hi, 3);
264 r += lmmp_sub_1_(R + ns, R + ns, ns + 1, cy);
265
266 lmmp_debug_assert(r == 0);
267 lmmp_dec(Alr);
268 }
269 lmmp_copy(dst, Alr, lo);
270 }
271 }

273#undef rk
274#undef R
275#undef Ahr2
276#undef Alr
277#undef Alr2
278#undef scratch
