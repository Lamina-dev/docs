34mp_size_t lmmp_seed_random_(mp_ptr restrict dst, mp_size_t n, mp_limb_t seed, int seed_type) {
35 seed_type %= 2;
36 if (seed_type == 0) {
37 pcg64_128_state rng;
38 lmmp_pcg64_128_srandom(&rng, seed);
39 mp_size_t i = 0;
40 for (; i + 3 < n; i += 4) {
41 dst[i + 0] = lmmp_pcg64_128_random(&rng);
42 dst[i + 1] = lmmp_pcg64_128_random(&rng);
43 dst[i + 2] = lmmp_pcg64_128_random(&rng);
44 dst[i + 3] = lmmp_pcg64_128_random(&rng);
45 }
46 for (; i < n; ++i) {
47 dst[i] = lmmp_pcg64_128_random(&rng);
48 }
49 } else {
50 xoshiro256pp_state rng;
51 lmmp_xoshiro256pp_srandom(&rng, seed);
52 mp_size_t i = 0;
53 for (; i + 3 < n; i += 4) {
54 dst[i + 0] = lmmp_xoshiro256pp_random(&rng);
55 dst[i + 1] = lmmp_xoshiro256pp_random(&rng);
56 dst[i + 2] = lmmp_xoshiro256pp_random(&rng);
57 dst[i + 3] = lmmp_xoshiro256pp_random(&rng);
58 }
59 for (; i < n; ++i) {
60 dst[i] = lmmp_xoshiro256pp_random(&rng);
61 }
62 }
63 while (n > 0 && dst[n - 1] == 0) {
64 --n;
65 }
66 return n;
67}

69mp_size_t lmmp_random_(mp_ptr restrict dst, mp_size_t n) {
70 if (n == 0 || dst == NULL) {
71 return 0;
72 }
73 mp_limb_t seed = lmmp_global_rng.state + n;
74 seed ^= rotl(lmmp_seed_generator(n), 23);
75 lmmp_global_rng.state = lmmp_seed_generator(seed);
76 return lmmp_seed_random_(dst, n, seed, lmmp_global_rng.seed_type);
77}

83lmmp_strong_rng_t* lmmp_strong_rng_init_(mp_size_t k, int seed) {
84 lmmp_param_assert(k > 0);
85
86 lmmp_strong_rng_t* rng = (lmmp_strong_rng_t*)lmmp_alloc(sizeof(lmmp_strong_rng_t));
87 rng->stream.k = k;
88 rng->stream.state = ALLOC_TYPE(k, mp_limb_t);
89
90 mp_limb_t new_seed = lmmp_seed_generator(seed);
91 new_seed = rotl(new_seed, 17) ^ lmmp_seed_generator(k);
92 pcg64_le_seq_init(&rng->stream, 0, new_seed);
93 return rng;
94}

96void lmmp_strong_rng_extern_(lmmp_strong_rng_t* rng, mp_size_t k) {
97 lmmp_param_assert(rng != NULL);
98 lmmp_param_assert(k > 0);
99 if (k <= rng->stream.k) return;
100 rng->stream.state = (mp_limb_t*)lmmp_realloc(rng->stream.state, k * sizeof(mp_limb_t));
101
102 mp_limb_t new_seed = lmmp_seed_generator(rng->stream.k);
103 new_seed = rotl(new_seed, 37) ^ lmmp_seed_generator(k);
104 pcg64_le_seq_init(&rng->stream, rng->stream.k, new_seed);
105 rng->stream.k = k;
106}

118 lmmp_param_assert(dst != NULL);
