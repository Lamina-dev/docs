49mp_size_t lmmp_seed_random_(mp_ptr restrict dst, mp_size_t n, mp_limb_t seed, unsigned seed_type) {
50 seed_type %= 2;
51 if (seed_type == 0) {
52 pcg64_128_state rng;
53 lmmp_pcg64_128_srandom(&rng, seed);
54 mp_size_t i = 0;
55 for (; i + 3 < n; i += 4) {
56 dst[i + 0] = lmmp_pcg64_128_random(&rng);
57 dst[i + 1] = lmmp_pcg64_128_random(&rng);
58 dst[i + 2] = lmmp_pcg64_128_random(&rng);
59 dst[i + 3] = lmmp_pcg64_128_random(&rng);
60 }
61 for (; i < n; ++i) {
62 dst[i] = lmmp_pcg64_128_random(&rng);
63 }
64 } else {
65 xoshiro256pp_state rng;
66 lmmp_xoshiro256pp_srandom(&rng, seed);
67 mp_size_t i = 0;
68 for (; i + 3 < n; i += 4) {
69 dst[i + 0] = lmmp_xoshiro256pp_random(&rng);
70 dst[i + 1] = lmmp_xoshiro256pp_random(&rng);
71 dst[i + 2] = lmmp_xoshiro256pp_random(&rng);
72 dst[i + 3] = lmmp_xoshiro256pp_random(&rng);
73 }
74 for (; i < n; ++i) {
75 dst[i] = lmmp_xoshiro256pp_random(&rng);
76 }
77 }
78 while (n > 0 && dst[n - 1] == 0) {
79 --n;
80 }
81 return n;
82}

84mp_size_t lmmp_random_(mp_ptr restrict dst, mp_size_t n) {
85 if (n == 0 || dst == NULL) {
86 return 0;
87 }
88 mp_limb_t seed = lmmp_global_rng.state + mix64(n);
89 seed ^= lmmp_seed_generator(n);
90 lmmp_global_rng.state = lmmp_seed_generator(seed);
91 return lmmp_seed_random_(dst, n, seed, lmmp_global_rng.seed_type);
92}

99lmmp_strong_rng_t* lmmp_strong_rng_init_(mp_size_t k, unsigned seed) {
100 lmmp_param_assert(k > 0);
101
102 lmmp_strong_rng_t* rng = (lmmp_strong_rng_t*)lmmp_alloc(sizeof(lmmp_strong_rng_t));
103 rng->stream.k = k;
104 rng->stream.state = ALLOC_TYPE(k, mp_limb_t);
105
106 mp_limb_t new_seed = lmmp_seed_generator(seed);
107 new_seed ^= mix64(k);
108 rng->seed = new_seed;
109 pcg64_le_seq_init(&rng->stream, 0, new_seed);
110 return rng;
111}

113void lmmp_strong_rng_extern_(lmmp_strong_rng_t* rng, mp_size_t k) {
114 lmmp_param_assert(rng != NULL);
115 lmmp_param_assert(k > 0);
116 if (k <= rng->stream.k) return;
117 mp_size_t old_k = rng->stream.k;
118 rng->stream.state = (mp_limb_t*)lmmp_realloc(rng->stream.state, k * sizeof(mp_limb_t));
119
120 mp_limb_t new_seed = lmmp_seed_generator(rng->stream.k);
121 new_seed ^= mix64(k + rng->seed);
122 rng->stream.k = k;
123 rng->seed = new_seed;
124 pcg64_le_seq_init(&rng->stream, old_k, new_seed);
125}

127void lmmp_strong_rng_free_(lmmp_strong_rng_t* rng) {
128 if (rng != NULL) {
129 if (rng->stream.state != NULL) {
130 lmmp_free(rng->stream.state);
131 }
132 lmmp_free(rng);
133 }

137 lmmp_param_assert(dst != NULL);
138 lmmp_param_assert(rng != NULL);

144 --n;
145 }
