64/**
65 * @brief 费马数模乘法（缓存版） [dst,rn+1]=[numa,na]*[numb,nb] mod B^rn+1
66 * 第二个操作数将被缓存
67 * @param dst 输出结果缓冲区，长度至少为 rn+1
68 * @param rn 模运算的阶数参数，rn = lmmp_fft_next_size_((na + nb + 1) >> 1)
69 * @param numa 第一个输入操作数，长度为 na
70 * @param na 第一个操作数的 limb 长度
71 * @param numb 第二个输入操作数，长度为 nb
72 * @param nb 第二个操作数的 limb 长度
73 * @param ctx 费马数模乘法缓存上下文
74 * @warning eqsep(dst,[numa|numb]), 0<=[numa,na]<2*B^rn, 0<=[numb,nb]<2*B^rn, rn = lmmp_fft_next_size_((na+nb+1)>>1)
75 * @note [numb,nb]将会被缓存，第二个乘数始终保持不变时，在后续计算中可以调用lmmp_mul_fermat_cache_()函数节省40%的计算时间
76 * numb可以在lmmp_mul_fermat_cache_()函数前被释放，不会影响后续计算
77 * @return 无返回值，结果存储在dst中
78 */
79void lmmp_mul_fermat_cache_init_(mp_ptr dst, mp_size_t rn, mp_srcptr numa, mp_size_t na, mp_srcptr numb, mp_size_t nb,
