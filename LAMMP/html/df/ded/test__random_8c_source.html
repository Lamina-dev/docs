32TEST_CASE("rng/basic", seed_random_reproducible) {
33 mp_size_t n = 16;
34 mp_ptr a = alloc_limbs(n);
35 mp_ptr b = alloc_limbs(n);
36
37 // 相同种子和类型应可复现
38 mp_size_t an = lmmp_seed_random_(a, n, 0x123456789ull, 0);
39 mp_size_t bn = lmmp_seed_random_(b, n, 0x123456789ull, 0);
40 TEST_CHECK_EQ(an, bn);
41 for (mp_size_t i = 0; i < an; ++i) TEST_CHECK_EQ(a[i], b[i]);
42
43 // 不同类型应产生不同序列（概率性，但相同实现下应稳定）
44 mp_size_t cn = lmmp_seed_random_(b, n, 0x123456789ull, 1);
45 bool diff = false;
46 for (mp_size_t i = 0; i < an && i < cn; ++i) if (a[i] != b[i]) diff = true;
47 TEST_CHECK_MSG(diff || an != cn, "seed_type changes sequence");
48
49 lmmp_free(a);
50 lmmp_free(b);
51}

69TEST_CASE("rng/strong", strong_random) {
70 mp_size_t k = 20;
71 mp_size_t n = 12;
72 lmmp_strong_rng_t* rng = lmmp_strong_rng_init_(k, 42);
73 TEST_CHECK_MSG(rng != NULL, "strong rng init");
74
75 mp_ptr a = alloc_limbs(k);
76 mp_ptr b = alloc_limbs(k);
77 mp_size_t an = lmmp_strong_random_(a, n, rng);
78 mp_size_t bn = lmmp_strong_random_(b, n, rng);
79 TEST_CHECK_MSG(an > 0 && bn > 0, "strong random nonzero length");
80 TEST_CHECK_MSG(an <= n && bn <= n, "strong random length bound");
81 bool same = (an == bn);
82 if (same) for (mp_size_t i = 0; i < an; ++i) if (a[i] != b[i]) same = false;
83 TEST_CHECK_MSG(!same, "strong random sequence advances");
84
85 lmmp_strong_rng_free_(rng);
86 lmmp_free(a);
87 lmmp_free(b);
88}

90TEST_CASE("rng/strong", strong_rng_extern_reproducible) {
91 // 回归：扩容时新增状态曾未初始化（realloc 不清零，且初始化循环上界在更新 k 之前取值）
92 const mp_size_t k1 = 8;
93 const mp_size_t k2 = 24;
94
95 lmmp_strong_rng_t* a = lmmp_strong_rng_init_(k1, 7);
96 lmmp_strong_rng_extern_(a, k2);
97 lmmp_strong_rng_t* b = lmmp_strong_rng_init_(k1, 7);
98 lmmp_strong_rng_extern_(b, k2);
99
100 mp_ptr ra = alloc_limbs(k2);
101 mp_ptr rb = alloc_limbs(k2);
102 mp_size_t na = lmmp_strong_random_(ra, k2, a);
103 mp_size_t nb = lmmp_strong_random_(rb, k2, b);
104 TEST_CHECK_EQ(na, nb);
105 TEST_CHECK_MSG(na > 0, "extern grown rng nonzero output length");
106 bool same = true;
107 for (mp_size_t i = 0; i < na; ++i)
108 if (ra[i] != rb[i]) same = false;
109 TEST_CHECK_MSG(same, "extern growth deterministic");
110
111 lmmp_strong_rng_free_(a);
112 lmmp_strong_rng_free_(b);
113 lmmp_free(ra);
114 lmmp_free(rb);
115}
