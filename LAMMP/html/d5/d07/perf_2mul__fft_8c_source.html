26static double bench_mul_fft(mp_size_t a, mp_size_t b, const bench_options *o) {
27 if (a < b) {
28 mp_size_t t = a;
29 a = b;
30 b = t;
31 }
32 mp_ptr pa = alloc_limbs(a), pb = alloc_limbs(b), d = alloc_limbs(a + b + 2);
33 fill_random(pa, a, 0x120e3489bc4ull + a * 31 + b);
34 fill_random(pb, b, 0x92cab5432214ull + b * 17 + a);
35 double us;
36 BENCH_MEASURE(us, o, lmmp_mul_fft_(d, pa, a, pb, b););
37 lmmp_free(pa);
38 lmmp_free(pb);
39 lmmp_free(d);
40 return us;
41}

43static void run_fft(void) {
44 const mp_size_t SZ[] = {100, 500, 2000, 5000};
45 for (size_t i = 0; i < ARRAY_SIZE(SZ); ++i) {
46 mp_size_t n = SZ[i];
47 report_n("mul_fft", n, bench_mul_fft(n, n, &bench_default), n * n * 8.0);
48 }
49}
