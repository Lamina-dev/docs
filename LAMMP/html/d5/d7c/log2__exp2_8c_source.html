26/*
27 记 B = 2^64，x 为 [0,1) 上的小数（64bit 或 128bit 定点），本文件计算
28 log2(1+x) 与 2^x - 1 的定点值。结构为“查表约简 + 短泰勒级数”：
29 log2 查表索引为 5bit（级数 27/14 项），exp2 为 7bit 且 Q 表由
30 16/128 两级表运行时合成（级数 13/7 项）。
31
32 约简恒等式（i 为索引，xl 为剩余低位）：
33
34 log2(1+x) = L[i] + log2(1+u)，u = xl/(2^k*(32+i)) < 2^-5
35 2^x - 1 = Q * (2^t)，Q = 2^(i/128)，t < 2^-7
36
37 u 由倒数表（2^128/(32+i) 的定点值 lg2_R32）一次乘法获得。级数
38 按奇偶次幂分组（w = u^2 或 t^2，均为正系数规避符号分支）：
39
40 log2(1+u) = u*B(w) - w*A(w)
41 2^t - 1 = t*B(w) + w*A(w)
42
43 项数由首个截断项 < 2^-66（64bit 输出）/ 2^-130（128bit 输出）
44 确定：log2 侧 u < 2^-5 故 m 项覆盖 m*5 个 bit（27/14 项），exp2
45 侧 m*7（13/7 项）。lg2 系数按 2^191 尺度截取高 2 limb 存储（c1 =
46 1/ln2 > 1 超出定点范围，链与收尾按半尺度 2^63/2^127 推进、结果
47 左移 1 位恢复；低 limb 为死数据不存）；xp2 系数均 < 1，按满尺度
48 2^192 截取高 2 limb 存储。求值形态：64bit log2 用 Estrin 树
49 （w2/w4 预平方，关键路径 6 级 mul -> 3 级）；128bit 版 B/A 链
50 全展开，非收尾步用 3 乘法舍入 hi128（umul128_tohi128_r，省去
51 a0*b0 全积、以交叉和顶位舍入，|e| <= 1 ulp 且均值近零），每步
52 误差被后续 w^j 衰减（log2 侧 2^-10j、exp2 侧 2^-14j）至预算
53 之下，收尾 1~2 步保留全精度 floor。exp2_64 的 Q 表合成同样用
54 舍入式 hi128（其 1 ulp 误差经 >>64 后 < 2^-63）。
55
56 舍入控制：约简变量 u 的定点化为误差主源（~0.7 ulp），其余各级
57 floor 系统性偏负，故 64bit 版 B/A 链取 floor 而收尾以 +2^63 /
58 舍入位中心化，exp2 的表合成（T16/T8 四舍五入存、T8 运行时再舍入
59 取高位）与 s、Q*s 项亦取舍入值。接口承诺 |e| <= 2 ulp。
60
61 常量表由纯整数高精度脚本离线生成（scripts/lmmp/gen_log2_exp2.py，
62 atanh/exp 级数求值，ln2 取 768bit 定点），每张表首行为 i=0。
63*/

212// hi128(a*b)：128bit 定点积的最高 128bit
213static inline __uint128_t umul128_tohi128(__uint128_t a, __uint128_t b) {
214 uint64_t a0 = (uint64_t)a, a1 = (uint64_t)(a >> 64);
215 uint64_t b0 = (uint64_t)b, b1 = (uint64_t)(b >> 64);
216 __uint128_t p01 = (__uint128_t)a0 * b1;
217 __uint128_t p10 = (__uint128_t)a1 * b0;
218 __uint128_t p11 = (__uint128_t)a1 * b1;
219 __uint128_t w1 = (__uint128_t)(uint64_t)(((__uint128_t)a0 * b0) >> 64) + (uint64_t)p01 + (uint64_t)p10;
220 __uint128_t w2 = (p01 >> 64) + (p10 >> 64) + (uint64_t)p11 + (uint64_t)(w1 >> 64);
221 return ((__uint128_t)((uint64_t)(p11 >> 64) + (uint64_t)(w2 >> 64)) << 64) | (uint64_t)w2;
222}
223
224/*
225 hi128(a*b) 的 3 乘法舍入近似（省去 a0*b0）：精确值 = p11 + cy +
226 (mid>>64) + floor((mid_lo + hi(a0*b0))/2^64)，末项进位 ∈{0,1}；
227 以 round(mid_lo/2^64)（mid_lo 顶位）替代，两者均 ∈{0,1}，故
228 |e| <= 1 ulp、均值近零。用于级数链非收尾步——该误差随后被 w
229 的幂衰减到输出预算之下。
230*/
231static inline __uint128_t umul128_tohi128_r(__uint128_t a, __uint128_t b) {
232 uint64_t a0 = (uint64_t)a, a1 = (uint64_t)(a >> 64);
233 uint64_t b0 = (uint64_t)b, b1 = (uint64_t)(b >> 64);
234 __uint128_t p01 = (__uint128_t)a0 * b1;
235 __uint128_t p10 = (__uint128_t)a1 * b0;
236 __uint128_t mid = p01 + p10;
237 uint64_t cy = (mid < p01);
238 return ((__uint128_t)a1 * b1) + (mid >> 64) + ((__uint128_t)cy << 64) + ((uint64_t)mid >> 63);
239}

242#define add_u128(r, x, y) \
243 do { \
244 (r)[0] = (x)[0] + (y)[0]; \
245 (r)[1] = (x)[1] + (y)[1] + ((r)[0] < (y)[0] ? 1 : 0); \
246 } while (0)

248// r = x - y（128 位；同上）
249#define sub_u128(r, x, y) \
250 do { \
251 uint64_t _b_ = (x)[0] < (y)[0]; \
252 (r)[0] = (x)[0] - (y)[0]; \
253 (r)[1] = (x)[1] - (y)[1] - _b_; \
254 } while (0)

256// i += j（192 位；同上）
257#define add_u192(i, j) \
258 do { \
259 (i)[0] += (j)[0]; \
260 uint64_t _c_ = ((i)[0] < (j)[0]) ? 1 : 0; \
261 (i)[1] += _c_; \
262 _c_ = ((i)[1] < _c_) ? 1 : 0; \
263 (i)[1] += (j)[1]; \
264 _c_ += ((i)[1] < (j)[1]) ? 1 : 0; \
265 (i)[2] += _c_ + (j)[2]; \
266 } while (0)

269#define sub_u192(i, j) \
270 do { \
271 uint64_t _b_ = ((i)[0] < (j)[0]) ? 1 : 0; \
272 (i)[0] -= (j)[0]; \
273 uint64_t _b1_ = ((i)[1] < (j)[1]) ? 1 : 0; \

275 _b1_ += ((i)[1] < _b_) ? 1 : 0; \
276 (i)[1] -= _b_; \
277 (i)[2] = (i)[2] - ((j)[2] + _b1_); \
278 } while (0)

281uint64_t log2_fixed_64(uint64_t x) {
282 unsigned i = (unsigned)(x >> 59);
283 uint64_t xl = x & 0x07FFFFFFFFFFFFFFull;
284
285 /*
286 v = u*2^128 = xl*2^69/(32+i)，u < 2^-5。乘子 iv = floor(2^69/(32+i))
287 由 128bit 倒数表右移导出；i=0 时精确值 2^64 超出单 limb，钳位为
288 2^64-1（v 误差 < xl < 2^59，相对 < 2^-64，可忽略）。xl*iv <
289 2^59*2^64 = 2^123，128bit 全积恰为 v 本身（尺度已对齐）。
290 */
291 uint64_t v[2];
292 uint64_t iv = i ? ((lg2_R32[i][1] << 5) | (lg2_R32[i][0] >> 59)) : UINT64_MAX;
293 _umul64to128_(xl, iv, &v[0], &v[1]);
294
295 // w = u^2*2^64（链乘子），wf = u^2*2^128（收尾乘子）
296 uint64_t rr[4];
297 _usqr128to256_(v[1], v[0], rr);
298 uint64_t w = rr[3];
299
300 /*
301 64bit 输出下 B 误差预算 ~2^-59（经 u 因子放大前），w 每层衰减
302 2^-10，整条链单 limb（2^63 半尺度）即可：链值取 lg2_c27 顶
303 limb。B 为奇次项（m=1..13），A 为偶次项（m=2..14），首个截断
304 项 m=15 < 2^-75 < 2^-66 预算。
305 */
306 /*
307 Estrin 求值树（w2=w^2, w4=w^4）：B/A 各 7/6 次 mulhi 不变，
308 乘法关键路径 6 级 -> 3 级，B/A 两树独立并行。
309 */
310 uint64_t w2 = _umul64to64hi_(w, w);
311 uint64_t w4 = _umul64to64hi_(w2, w2);

345uint64_t exp2_fixed_64(uint64_t x) {

353 uint64_t z = _umul64to64hi_(w, w);

355 // B = (b0+w*b1) + z*(b2+w*b3)，A = a0+w*(a1+w*a2)，单 limb 满尺度链
356 uint64_t p1 = _umul64to64hi_(xp2_c[2][1], w) + xp2_c[0][1];
357 uint64_t p2 = _umul64to64hi_(xp2_c[6][1], w) + xp2_c[4][1];

360 A = _umul64to64hi_(A, w) + xp2_c[1][1];
361
362 // S = s*2^128 = t*B + w*A，sh = round(s*2^64)
363 uint64_t s0[2], s1[2];
364 _umul64to128_(t, B, &s0[0], &s0[1]);
365 _umul64to128_(w, A, &s1[0], &s1[1]);
366 uint64_t sl = s0[0] + s1[0];
367 uint64_t sh = s0[1] + s1[1] + (sl < s0[0]) + (sl >> 63);

369 /*
370 q = (2^(i/128)-1)*2^64：两级表 128bit 全量合成（丢弃表低位分数
371 会被 T16 的大因子放大 ~1ulp，须完整求和），舍入取高 64bit。
372 mm 同样舍入化，全链误差 |e| <= 2。
373 */

377 // qq 仅供 2^128 尺度合成后舍入取高 64bit，hi128 的 1 ulp（2^-128）
378 // 误差经 >>64 后 < 2^-63，用 3 乘法舍入式即可
379 __uint128_t qq = umul128_tohi128_r(((__uint128_t)xp2_T16[k][1] << 64) | xp2_T16[k][0], t8t);
380 uint64_t q0 = xp2_T16[k][0] + (uint64_t)qq;
381 uint64_t cy = (q0 < (uint64_t)qq);
382 q0 += (uint64_t)t8t;
383 cy += (q0 < (uint64_t)t8t);
384 uint64_t q = xp2_T16[k][1] + (uint64_t)(t8t >> 64) + (uint64_t)(qq >> 64) + cy + (q0 >> 63);
385 uint64_t ml, mh;
386 _umul64to128_(q, sh, &ml, &mh);
387 uint64_t r = q + sh + mh + (ml >> 63);
388 if (r < sh) return UINT64_MAX;

393 unsigned i = (unsigned)(high >> 59);

395
396 // xls = xl*2^5（128bit 对齐），v = u*2^128 = hi128(xls*R32[i])
397 uint64_t xls[2] = {low << 5, (h2 << 5) | (low >> 59)};
398 uint64_t rr[4];

400 uint64_t v[2] = {rr[2], rr[3]};

402 // wf = u^2*2^128（链乘子）
403 _usqr128to256_(v[1], v[0], rr);
404 __uint128_t wf = ((__uint128_t)rr[3] << 64) | rr[2];
405
406 /*
407 交错 Horner：B 奇链 c25 -> c1（lg2_c27[24] -> [0]），A 偶链
408 c26 -> c2（[25] -> [1]），单循环双累加器维持 ILP=2。128bit 输出
409 下 B 误差预算 ~2^-122，链取双 limb（2^127 半尺度）；首个截断项
410 m=27 < 2^-135 < 2^-130 预算。链值以 __uint128_t 局部量保活，
411 避免数组形态在循环内产生栈回读写。
412 非收尾步用 3 乘法舍入 hi128（umul128_tohi128_r）：每步 <= 1 ulp
413 误差被后续 wf^j（<= 2^-10j）衰减，j >= 1 即 <= 2^-138，远低于
414 预算；收尾 2 步保留全精度 floor。乘法数 104 -> 84。
415 */
416 __uint128_t B = ((__uint128_t)lg2_c27[24][1] << 64) | lg2_c27[24][0];
417 __uint128_t A = ((__uint128_t)lg2_c27[25][1] << 64) | lg2_c27[25][0];
418 for (int k = 22; k >= 5; k -= 2) {
419 B = umul128_tohi128_r(B, wf) + (((__uint128_t)lg2_c27[k][1] << 64) | lg2_c27[k][0]);
420 A = umul128_tohi128_r(A, wf) + (((__uint128_t)lg2_c27[k + 1][1] << 64) | lg2_c27[k + 1][0]);
421 }
422 B = umul128_tohi128_r(B, wf) + (((__uint128_t)lg2_c27[4][1] << 64) | lg2_c27[4][0]);
423 A = umul128_tohi128_r(A, wf) + (((__uint128_t)lg2_c27[5][1] << 64) | lg2_c27[5][0]);
424 B = umul128_tohi128(B, wf) + (((__uint128_t)lg2_c27[2][1] << 64) | lg2_c27[2][0]);
425 A = umul128_tohi128(A, wf) + (((__uint128_t)lg2_c27[3][1] << 64) | lg2_c27[3][0]);
426 B = umul128_tohi128(B, wf) + (((__uint128_t)lg2_c27[0][1] << 64) | lg2_c27[0][0]);
427 A = umul128_tohi128(A, wf) + (((__uint128_t)lg2_c27[1][1] << 64) | lg2_c27[1][0]);
428
429 // P = 2*(u*B - wf*A)，t1/t2 取 256bit 积的高 192bit（2^191 半尺度）
430 uint64_t t1[3], t2[3];
431 _umul128to256_(v[1], v[0], (uint64_t)(B >> 64), (uint64_t)B, rr);
432 t1[0] = rr[1];
433 t1[1] = rr[2];
434 t1[2] = rr[3];
435 _umul128to256_((uint64_t)(wf >> 64), (uint64_t)wf, (uint64_t)(A >> 64), (uint64_t)A, rr);
436 t2[0] = rr[1];
437 t2[1] = rr[2];
438 t2[2] = rr[3];
439 sub_u192(t1, t2);
440 t1[2] = (t1[2] << 1) | (t1[1] >> 63);
441 t1[1] = (t1[1] << 1) | (t1[0] >> 63);
442 t1[0] <<= 1;
443
444 // result192 = L32[i] + P，加 2^63 使 >>64 截断舍入化，输出高 128bit
445 add_u192(t1, lg2_L32[i]);
446 if (t1[2] < lg2_L32[i][2]) {
447 dst[0] = dst[1] = UINT64_MAX;
448 return;
449 }
450 t1[0] += 0x8000000000000000ull;
451 if (t1[0] < 0x8000000000000000ull) {
452 if (++t1[1] == 0 && ++t1[2] == 0) {

457 dst[0] = t1[1];
