96/**
97 * @brief 计算2x2矩阵和2x2矩阵的乘积
98 * @param dst 结果矩阵。
99 * @param matA 矩阵A
100 * @param matB 矩阵B
101 * @param choose 选择算法，0表示basecase算法，1表示STRASSEN算法
102 * @param tn 缓冲区的limb长度，建议由lmmp_mat22_mul_size_确定
103 * @param maxa matA中最大的元素的limb长度+1，建议由lmmp_mat22_mul_size_确定
104 * @warning dst!=NULL, nonull(dst), [matA|matB]!=NULL, nonull([matA|matB]), sep(dst,[matA|matB]), choose==[0|1]
105 */
106INLINE_ void
107lmmp_mat22_mul_(lmmp_mat22_t* dst, const lmmp_mat22_t* matA, const lmmp_mat22_t* matB, int choose, mp_size_t tn,
108 mp_size_t maxa) {
109 lmmp_param_assert(dst != NULL);
110 lmmp_param_assert(matA != NULL);
111 lmmp_param_assert(matB != NULL);
112 lmmp_param_assert(choose == 0 || choose == 1);
113 if (choose == 0) {
114 lmmp_mat22_mul_basecase_(dst, matA, matB, NULL, tn);
115 } else {
116 lmmp_mat22_mul_strassen_(dst, matA, matB, NULL, tn, maxa);
117 }
118}

124 * @param tn 缓冲区的limb长度，建议由lmmp_mat22_mul_size_确定
125 * @param choose 选择算法，0表示basecase算法，1表示STRASSEN算法
126 * @warning dst!=NULL, nonull(dst), matA!=NULL, nonull(matA), sep(dst,matA), tn>0
127 */
128INLINE_ void
129lmmp_mat22_sqr_(lmmp_mat22_t* dst, const lmmp_mat22_t* mat, int choose, mp_size_t tn) {
130 lmmp_param_assert(dst != NULL);
131 lmmp_param_assert(mat != NULL);
132 lmmp_param_assert(tn > 0);
133 lmmp_param_assert(choose == 0 || choose == 1);
134 if (choose == 0) {
135 lmmp_mat22_sqr_basecase_(dst, mat, NULL, tn);
136 } else {
137 lmmp_mat22_sqr_strassen_(dst, mat, NULL, tn);
138 }
