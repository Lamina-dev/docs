25static double bench_is_strong_lucas(mp_size_t n, PrimePool kind, const bench_options *o) {
26 int cnt = pool_cnt(n);
27 mp_ptr a = pool_n(n, cnt, 0x51ce77ac02ull + n, kind);
28 size_t idx = 0;
29 double us;
30 BENCH_MEASURE(us, o, bench_sink += (mp_limb_t)lmmp_is_strong_lucas_(a + n * (idx++ % cnt), n););
31 lmmp_free(a);
32 return us;
33}
