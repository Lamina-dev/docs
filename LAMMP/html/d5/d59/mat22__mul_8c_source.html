23int lmmp_mat22_mul_size_(
24 lmmp_mat22_t* dst,
25 const lmmp_mat22_t* matA,
26 const lmmp_mat22_t* matB,
27 mp_size_t* tn,
28 mp_size_t* maxa
29) {
30 lmmp_param_assert(matA!= NULL && matB!= NULL && dst!= NULL);
31 lmmp_param_assert(tn != NULL);
32 if (matA == matB) {
33 mp_ssize_t A00 = LMMP_ABS(matA->n00);
34 mp_ssize_t A01 = LMMP_ABS(matA->n01);
35 mp_ssize_t A10 = LMMP_ABS(matA->n10);
36 mp_ssize_t A11 = LMMP_ABS(matA->n11);
37 if (A00 < MAT22_SQR_STRASSEN_THRESHOLD || A01 < MAT22_SQR_STRASSEN_THRESHOLD ||
38 A10 < MAT22_SQR_STRASSEN_THRESHOLD || A11 < MAT22_SQR_STRASSEN_THRESHOLD ) {
39 dst->n00 = LMMP_MAX((A00 + A00), (A01 + A10));
40 dst->n01 = LMMP_MAX((A00 + A01), (A01 + A11));
41 dst->n10 = LMMP_MAX((A10 + A00), (A11 + A10));
42 dst->n11 = LMMP_MAX((A10 + A01), (A11 + A11));
43 *tn = LMMP_MAX(LMMP_MAX(LMMP_MAX(dst->n00, dst->n01), dst->n10), dst->n11) + 1;
44 ++(dst->n00);
45 ++(dst->n01);
46 ++(dst->n10);
47 ++(dst->n11);
48 return 0;
49 } else {
50 *maxa = LMMP_MAX(LMMP_MAX(LMMP_MAX(A00, A01), A10), A11) + 1;
51 *tn = (*maxa << 1) + 1;
52 dst->n00 = *tn;
53 dst->n01 = *tn;
54 dst->n10 = *tn;
55 dst->n11 = *tn;
56 return 1;
57 }
58 } else {
59 mp_ssize_t A00 = LMMP_ABS(matA->n00);
60 mp_ssize_t A01 = LMMP_ABS(matA->n01);
61 mp_ssize_t A10 = LMMP_ABS(matA->n10);
62 mp_ssize_t A11 = LMMP_ABS(matA->n11);
63 mp_ssize_t B00 = LMMP_ABS(matB->n00);
64 mp_ssize_t B01 = LMMP_ABS(matB->n01);
65 mp_ssize_t B10 = LMMP_ABS(matB->n10);
66 mp_ssize_t B11 = LMMP_ABS(matB->n11);
67 if (A00 < MAT22_MUL_STRASSEN_THRESHOLD || A01 < MAT22_MUL_STRASSEN_THRESHOLD ||
68 A10 < MAT22_MUL_STRASSEN_THRESHOLD || A11 < MAT22_MUL_STRASSEN_THRESHOLD ||
69 B00 < MAT22_MUL_STRASSEN_THRESHOLD || B01 < MAT22_MUL_STRASSEN_THRESHOLD ||
70 B10 < MAT22_MUL_STRASSEN_THRESHOLD || B11 < MAT22_MUL_STRASSEN_THRESHOLD) {
71 dst->n00 = LMMP_MAX((A00 + B00), (A01 + B10));
72 dst->n01 = LMMP_MAX((A00 + B01), (A01 + B11));
73 dst->n10 = LMMP_MAX((A10 + B00), (A11 + B10));
74 dst->n11 = LMMP_MAX((A10 + B01), (A11 + B11));
75 *tn = LMMP_MAX(LMMP_MAX(LMMP_MAX(dst->n00, dst->n01), dst->n10), dst->n11);
76 ++(dst->n00);
77 ++(dst->n01);
78 ++(dst->n10);
79 ++(dst->n11);
80 return 0;
81 } else {
82 *maxa = LMMP_MAX(LMMP_MAX(LMMP_MAX(A00, A01), A10), A11) + 1;
83 *tn = *maxa + LMMP_MAX(LMMP_MAX(LMMP_MAX(B00, B01), B10), B11) + 1;
84 dst->n00 = *tn;
85 dst->n01 = *tn;
86 dst->n10 = *tn;
87 dst->n11 = *tn;
88 return 1;
89 }
90 }

93void lmmp_mat22_mul_basecase_(
94 lmmp_mat22_t* dst,
95 const lmmp_mat22_t* matA,
96 const lmmp_mat22_t* matB,
97 mp_ptr tp,
98 mp_size_t tn
99) {
100 lmmp_param_assert(matA != NULL && matB != NULL && dst != NULL);
101 lmmp_param_assert(tn > 0);
102 if (matA == matB) {
103 lmmp_mat22_sqr_basecase_(dst, matA, tp, tn);
104 return;
105 }
106 TEMP_DECL;
107 if (tp == NULL)
108 tp = TALLOC_TYPE(tn * 2, mp_limb_t);
109#define p1 tp
110#define p2 tp + tn
111 mp_ssize_t pn1, pn2;
112 pn1 = lmmp_mul_signed_(p1, matA->a00, matA->n00, matB->a00, matB->n00);
113 pn2 = lmmp_mul_signed_(p2, matA->a01, matA->n01, matB->a10, matB->n10);
114 dst->n00 = lmmp_add_signed_(dst->a00, p1, pn1, p2, pn2);
115 pn1 = lmmp_mul_signed_(p1, matA->a00, matA->n00, matB->a01, matB->n01);
116 pn2 = lmmp_mul_signed_(p2, matA->a01, matA->n01, matB->a11, matB->n11);
117 dst->n01 = lmmp_add_signed_(dst->a01, p1, pn1, p2, pn2);
118 pn1 = lmmp_mul_signed_(p1, matA->a10, matA->n10, matB->a00, matB->n00);
119 pn2 = lmmp_mul_signed_(p2, matA->a11, matA->n11, matB->a10, matB->n10);
120 dst->n10 = lmmp_add_signed_(dst->a10, p1, pn1, p2, pn2);
121 pn1 = lmmp_mul_signed_(p1, matA->a10, matA->n10, matB->a01, matB->n01);
122 pn2 = lmmp_mul_signed_(p2, matA->a11, matA->n11, matB->a11, matB->n11);
123 dst->n11 = lmmp_add_signed_(dst->a11, p1, pn1, p2, pn2);
124#undef p1
125#undef p2
126 TEMP_FREE;
127}

158/*
159 * Strassen 2x2 矩阵乘法的 Winograd 变体
160 *
161 * 输入矩阵：
162 * A = | A11 A12 |
163 * | A21 A22 |
164 * B = | B11 B12 |
165 * | B21 B22 |
166 *
167 * 输出矩阵 C = A * B：
168 * C = | C11 C12 |
169 * | C21 C22 |
170 *
171 *
172 * s1 = A22 + A12
173 * s2 = A22 - A21
174 * s3 = s2 + A12 = A22 - A21 + A12
175 * s4 = s3 - A11 = A22 - A21 + A12 - A11
176 *
177 * t1 = B22 + B12
178 * t2 = B22 - B21
179 * t3 = t2 + B12 = B22 - B21 + B12
180 * t4 = t3 - B11 = B22 - B21 + B12 - B11
181 *
182 * 7 个 Strassen 乘积项
183 * p1 = s1 * t1 = (A22 + A12 ) * (B22 + B12 )
184 * p2 = s2 * t2 = (A22 - A21 ) * (B22 - B21 )
185 * p3 = s3 * t3 = (A22 - A21 + A12) * (B22 - B21 + B12)
186 * p4 = A11 * B11
187 * p5 = A12 * B21
188 * p6 = s4 * B12
189 * p7 = A21 * t4
190 *
191 * U1 = p3 + p5
192 * U2 = p1 - U1
193 * U3 = U1 - p2
194 *
195 * result:
196 * C11 = p4 + p5
197 * C12 = U3 - p6
198 * C21 = U2 - p7
199 * C22 = p2 + U2
200 *
201 * 平方版本（A*A）：所有乘法替换为平方/自身相乘，流程一致。
202 */
