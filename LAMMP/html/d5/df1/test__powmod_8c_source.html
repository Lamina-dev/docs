43// 64 位内任意模数参考模幂：mullh 全积 + mod_1 取余（独立于被测蒙域路径）
44static u64 powmod_ref_u64(u64 b, const u64* ep, mp_size_t en, u64 m) {
45 u64 r = 1 % m;
46 b %= m;
47 mp_limb_t t[2];
48 for (mp_size_t i = en; i-- > 0;) {
49 for (int bit = 63; bit >= 0; --bit) {
50 lmmp_mullh_(r, r, t);
51 r = lmmp_mod_1_(t, 2, m);
52 if ((ep[i] >> bit) & 1) {
53 lmmp_mullh_(r, b, t);
54 r = lmmp_mod_1_(t, 2, m);
55 }
56 }
57 }
58 return r;
59}

64// 奇模数构造：m 随机奇归一化；b 归约到 < m（div_ 一步）
65static void make_odd_mod(mp_ptr m, mp_ptr b, mp_size_t n, u64 *seed) {
66 random_limbs(m, n, seed);
67 m[0] |= 1;
68 mp_ptr raw = alloc_limbs(n), q = alloc_limbs(n);
69 random_limbs(raw, n, seed);
70 raw[n - 1] >>= 1; // raw < 2^(64n-1)，一步归约后必 < m
71 if (raw[n - 1] == 0) raw[n - 1] = 1;
72 lmmp_div_(q, b, raw, n, m, n);
73 lmmp_free(raw);
74 lmmp_free(q);
75}

77// 模乘归约：[a,n]*[b,n] mod [m,n] → r（库乘 + 库除）
78static void mulmod_limbs(mp_ptr r, const mp_limb_t* a, const mp_limb_t* b, const mp_limb_t* m, mp_size_t n,
79 mp_ptr t, mp_ptr q) {
80 lmmp_mul_(t, a, n, b, n);
81 lmmp_div_(q, r, t, 2 * n, m, n);
82}

84// m = k*2^e + form（k <= 2^64，e >= 64）装配到 dst（容量 e/64+3），返回规范化长度
85static mp_size_t sp_shape_limbs(mp_ptr dst, u64 k, int e, int form) {
86 size_t cap = (size_t)e / 64 + 3;
87 lmmp_zero(dst, cap);
88 int s = e % 64;
89 dst[e / 64] = s ? (k << s) : k;
90 if (s) dst[e / 64 + 1] = k >> (64 - s);
91 if (form > 0) dst[0] += 1;
92 else lmmp_dec(dst);
93 return norm_len(dst, (mp_size_t)cap);
94}

96TEST_CASE("numth/powmod", redc_value) {
97 u64 seed = 0x9e3779b97f4a7c15ull;
98 // 覆盖 REDC 两条路径：小尺寸全积取高半、>= REDC_MERSENNE_THRESHOLD 的梅森折叠
99 const mp_size_t n_values[] = {1, 2, 3, 5, 9, 17, 33, 50, 402};
100 for (size_t n_index = 0; n_index < ARRAY_SIZE(n_values); ++n_index) {
101 mp_size_t n = n_values[n_index];
102 for (int iter = 0; iter < 3; ++iter) {
103 mp_ptr m = alloc_limbs(n), a = alloc_limbs(n), b = alloc_limbs(n);
104 make_odd_mod(m, a, n, &seed);
105 make_odd_mod(m, b, n, &seed);
106 mp_ptr t = alloc_limbs(2 * n), dst = alloc_limbs(n + 1), ninv = alloc_limbs(n);
107 lmmp_mul_(t, a, n, b, n);
108 lmmp_binvert_(ninv, m, n, n);
109 // ninv = -m^(-1) mod B^n（两补码取负；逆元为奇不会越顶）
110 u64 carry = 1;
111 for (mp_size_t i = 0; i < n; ++i) {
112 u64 x = ~ninv[i];
113 u64 s = x + carry;
114 carry = (s == 0) ? 1u : 0u;
115 ninv[i] = s;
116 }
117 mp_limb_t cy = lmmp_redc_(dst, t, ninv, m, n);
118 TEST_CHECK_MSG(cy <= 1, "redc carry range");
119 // u = dst + cy*B^n；u < 2m
120 mp_ptr u = alloc_limbs(n + 1);
121 lmmp_zero(u, n + 1);
122 memcpy(u, dst, (size_t)n * 8);
123 if (cy) u[n] = 1;
124 // 2m = m+m（库加）
125 mp_ptr twom2 = alloc_limbs(n + 1);
126 lmmp_zero(twom2, n + 1);
127 lmmp_copy(twom2, m, n);
128 lmmp_add_n_(twom2, twom2, twom2, n + 1);
129 TEST_CHECK_MSG(mpn_cmp(u, n + 1, twom2, n + 1) < 0, "redc < 2m");
130 // u*B^n ≡ t (mod m)：同余仅对模 m 成立，经库除法（Knuth D，独立
131 // 路径）双侧取余精确比对：(u*B^n) mod m == (a*b) mod m
132 {
133 mp_ptr P = alloc_limbs(2 * n + 2), q2 = alloc_limbs(n + 2), rm = alloc_limbs(n),
134 rt = alloc_limbs(n);
135 lmmp_zero(P, 2 * n + 2);
136 memcpy(P + n, u, (size_t)(n + 1) * 8); // P = u*B^n（≤ 2n+1 limbs）
137 lmmp_div_(q2, rm, P, 2 * n + 1, m, n);
138 lmmp_div_(q2, rt, t, 2 * n, m, n);
139 TEST_CHECK_MSG(mpn_eq(rm, n, rt, n), "redc (u*B^n) mod m == t mod m");
140 lmmp_free(P); lmmp_free(q2); lmmp_free(rm); lmmp_free(rt);
141 }
142 lmmp_free(m); lmmp_free(a); lmmp_free(b); lmmp_free(t); lmmp_free(dst);
143 lmmp_free(ninv); lmmp_free(twom2); lmmp_free(u);
144 }
145 }

148TEST_CASE("numth/powmod", powmod_odd_and_dispatch) {
149 u64 seed = 0x1234567890abcdefull;
150 const mp_size_t n_values[] = {1, 2, 3, 4, 7, 12, 33, 70};
151 for (size_t n_index = 0; n_index < ARRAY_SIZE(n_values); ++n_index) {
152 mp_size_t n = n_values[n_index];
153 for (int iter = 0; iter < 3; ++iter) {
154 mp_ptr m = alloc_limbs(n), b = alloc_limbs(n);
155 make_odd_mod(m, b, n, &seed);
156 mp_ptr dst = alloc_limbs(n), dst2 = alloc_limbs(n), dst3 = alloc_limbs(n);
157 // 指数集：短/边界/单双三 limb
158 struct exponent_case { mp_size_t en; u64 w[3]; } exps[] = {
159 {1, {1, 0, 0}}, {1, {2, 0, 0}}, {1, {3, 0, 0}},
160 {1, {17, 0, 0}}, {1, {64, 0, 0}}, {1, {0xdeadbeefull, 0, 0}},
161 {1, {0xffffffffffffffffull, 0, 0}},
162 {2, {0x9e3779b97f4a7c15ull, 0x1234, 0}},
163 {3, {0, 0, 1}}, // e=B^2：偶底数短路形态
164 };
165 for (size_t ex_index = 0; ex_index < ARRAY_SIZE(exps); ++ex_index) {
166 const struct exponent_case ex = exps[ex_index];
167 mp_ptr ep = alloc_limbs(ex.en);
168 memcpy(ep, ex.w, (size_t)ex.en * 8);
169 lmmp_powmod_odd_(dst, b, ep, ex.en, m, n);
170 TEST_CHECK_MSG(mpn_cmp(dst, n, m, n) < 0, "powmod_odd in range");
171 // n==1：u64 简单参考（mullh+mod_1，指数全 limb）
172 if (n == 1) {
173 u64 want = powmod_ref_u64(b[0], ep, ex.en, m[0]);
174 TEST_CHECK_MSG(dst[0] == want, "powmod_odd vs u64 reference");
175 }
176 // 入口一致：powmod_ 分派（奇模数走 odd 路径）
177 lmmp_powmod_(dst2, b, ep, ex.en, m, n);
178 TEST_CHECK_MSG(memcmp(dst, dst2, (size_t)n * 8) == 0, "powmod_ dispatch == odd");
179 // n==1/2 特化入口一致
180 if (n == 1) {
181 lmmp_powmod_1_(dst3, b, ep, ex.en, m[0]);
182 TEST_CHECK_MSG(dst[0] == dst3[0], "powmod_1_ == odd");
183 }
184 if (n == 2) {
185 lmmp_powmod_2_(dst3, b, ep, ex.en, m);
186 TEST_CHECK_MSG(memcmp(dst, dst3, 16) == 0, "powmod_2_ == odd");
187 }
188 lmmp_free(ep);
189 }
190 lmmp_free(m); lmmp_free(b); lmmp_free(dst); lmmp_free(dst2); lmmp_free(dst3);
191 }

194 // 费马小定理：素模 p ∈ MOD_P，b^(p-1) == 1；b^p == b（固定 b）
195 for (int i = 0; i < MOD_P_CNT; ++i) {
196 u32 p = MOD_P[i];
197 u64 bbs[4] = {2, 3, 12345, (u64)p - 1}; // 显式数组：Linux 下 u64 与 ull 字面量异型
198 for (size_t bb_index = 0; bb_index < ARRAY_SIZE(bbs); ++bb_index) {
199 u64 bb = bbs[bb_index];
200 u64 b = bb % p;
201 if (b == 0) continue;
202 mp_limb_t mp[1] = {p}, bp[1] = {b}, ep[1] = {p - 1}, dst[1];
203 lmmp_powmod_odd_(dst, bp, ep, 1, mp, 1);
204 TEST_CHECK_MSG(dst[0] == 1, "fermat b^(p-1) == 1");
205 ep[0] = p;
206 lmmp_powmod_odd_(dst, bp, ep, 1, mp, 1);
207 TEST_CHECK_MSG(dst[0] == b, "fermat b^p == b");
208 }
209 }

304TEST_CASE("numth/powmod", powlo_family) {
305 u64 seed = 0x1b873593f00dbeebull;
306 // powlo vs powmod_(B^n)：b^e mod B^n 的两条独立算法（截断梯子 vs 偶模数 CRT）
307 const mp_size_t n_values[] = {1, 2, 3, 5, 17};
308 for (size_t n_index = 0; n_index < ARRAY_SIZE(n_values); ++n_index) {
309 mp_size_t n = n_values[n_index];
310 for (int iter = 0; iter < 3; ++iter) {
311 mp_ptr b = alloc_limbs(n), ep = alloc_limbs(3), dst = alloc_limbs(n),
312 dst2 = alloc_limbs(n + 1), bn = alloc_limbs(n + 1), bp = alloc_limbs(n + 1);
313 random_limbs(b, n, &seed);
314 lmmp_zero(bn, n + 1);
315 bn[n] = 1; // 模数 B^n（n+1 limb，顶 limb=1）
316 for (mp_size_t i = 0; i < 3; ++i) ep[i] = xorshift64(&seed);
317 ep[2] |= (u64)1 << 63;
318 lmmp_zero(bp, n + 1);
319 memcpy(bp, b, (size_t)n * 8); // b < B^n
320 ulong es[5] = {1, 2, 5, 0xdeadbeef, ep[0]}; // 同上：mp_limb_t 与 ull 异型
321 for (size_t e_index = 0; e_index < ARRAY_SIZE(es); ++e_index) {
322 ulong e = es[e_index];
323 mp_limb_t e1[1] = {e};
324 lmmp_powlo_(dst, b, n, e1, 1);
325 lmmp_powmod_(dst2, bp, e1, 1, bn, n + 1);
326 TEST_CHECK_MSG(memcmp(dst, dst2, (size_t)n * 8) == 0, "powlo == powmod(B^n)");
327 }
328 lmmp_powlo_(dst, b, n, ep, 3);
329 lmmp_powmod_(dst2, bp, ep, 3, bn, n + 1);
330 TEST_CHECK_MSG(memcmp(dst, dst2, (size_t)n * 8) == 0, "powlo 3-limb exp == powmod");
331 // 特化与分派一致
332 if (n == 1) {
333 mp_ptr d1 = alloc_limbs(1);
334 lmmp_powlo_1_(d1, b, ep, 3);
335 TEST_CHECK_MSG(d1[0] == dst[0], "powlo_1_ == powlo_");
336 lmmp_free(d1);
337 }
338 if (n == 2) {
339 mp_ptr d2 = alloc_limbs(2);
340 lmmp_powlo_2_(d2, b, ep, 3);
341 TEST_CHECK_MSG(memcmp(d2, dst, 16) == 0, "powlo_2_ == powlo_");
342 lmmp_free(d2);
343 }
344 // 截断恒等式：powlo(b,e1+e2) == low_n(powlo(b,e1)*powlo(b,e2))
345 u64 e1 = ep[0] >> 1, e2 = ep[0] - e1;
346 mp_limb_t ea[1] = {e1}, eb[1] = {e2};
347 mp_ptr p1 = alloc_limbs(n), p2 = alloc_limbs(n), prod = alloc_limbs(2 * n), dm = alloc_limbs(n);
348 lmmp_powlo_(p1, b, n, ea, 1);
349 lmmp_powlo_(p2, b, n, eb, 1);
350 lmmp_powlo_(dm, b, n, ep, 1);
351 lmmp_mul_(prod, p1, n, p2, n);
352 TEST_CHECK_MSG(memcmp(dm, prod, (size_t)n * 8) == 0, "powlo truncation law");
353 // e=1 快路径 == b
354 mp_limb_t e1limb[1] = {1};
355 lmmp_powlo_(dm, b, n, e1limb, 1);
356 TEST_CHECK_MSG(memcmp(dm, b, (size_t)n * 8) == 0, "powlo e=1 == b");
357 lmmp_free(p1); lmmp_free(p2); lmmp_free(prod); lmmp_free(dm);
358 lmmp_free(b); lmmp_free(ep); lmmp_free(dst); lmmp_free(dst2); lmmp_free(bn);
359 lmmp_free(bp);
360 }

364TEST_CASE("numth/powmod", powmod_even_two_adic) {
365 // 偶模数 m = odd << k：(-1)^e / e=1 / b=0 角点 + 指数律 + 奇分派一致 + CRT 重建
366 u64 seed = 0x243f6a8885a308d3ull;
367 const size_t cap = 12;
368 size_t ks[] = {1, 2, 63, 64, 65, 128, 129, 384, 511, 512};
369 for (size_t k_index = 0; k_index < ARRAY_SIZE(ks); ++k_index) {
370 size_t k = ks[k_index];
371 u64 odds[3] = {1, 3, xorshift64(&seed) | 1};
372 for (size_t odd_index = 0; odd_index < ARRAY_SIZE(odds); ++odd_index) {
373 u64 odd = odds[odd_index];
374 // m = odd << k：跨 limb 由 limb 定位承载，limb 内由 shl 位移承载
375 mp_size_t nb = k / 64 + 2;
376 if (nb > cap) continue;
377 mp_ptr m = alloc_limbs(cap), b = alloc_limbs(cap), dst = alloc_limbs(cap);
378 lmmp_zero(m, cap);
379 int kb = (int)(k % 64);
380 m[k / 64] = kb ? (odd << kb) : odd;
381 if (kb) m[k / 64 + 1] = odd >> (64 - kb);
382 mp_size_t n = norm_len(m, cap);
383 // 归约基 b < m：随机后除
384 u64 s5 = seed ^ (u64)k * 0x9e3779b97f4a7c15ull ^ odd;
385 random_limbs(b, n, &s5);

388 mp_ptr q = alloc_limbs(cap);
389 lmmp_div_(q, b, b, n, m, n);

391 // b=0 → 0；e=1 → b；b=m-1：奇 e → m-1、偶 e → 1
392 mp_ptr z = alloc_limbs(cap);

394 mp_limb_t e2[1] = {2}, eodd[1] = {0x51ed270b9a3c8f11ull}, e1[1] = {1};
395 lmmp_powmod_(dst, z, eodd, 1, m, n);
396 bool zk = norm_len(dst, n) == 0;
397 TEST_CHECK_MSG(zk, "even-mod b=0");
398 lmmp_powmod_(dst, b, e1, 1, m, n);

400 mp_ptr mm1 = alloc_limbs(cap);

402 sub_small_inplace(mm1, n, 1);
403 lmmp_powmod_(dst, mm1, eodd, 1, m, n);
404 TEST_CHECK_MSG(memcmp(dst, mm1, (size_t)n * 8) == 0, "(-1)^odd == m-1");
405 lmmp_powmod_(dst, mm1, e2, 1, m, n);
406 TEST_CHECK_MSG(dst[0] == 1 && norm_len(dst, n) == 1, "(-1)^2 == 1");
407 // 指数律（b^e1 * b^e2 == b^(e1+e2)）
408 {

410 mp_limb_t epa[1] = {ea}, epb[1] = {eb}, eps[1] = {eodd[0]};
411 mp_ptr r1 = alloc_limbs(cap), r2 = alloc_limbs(cap), t = alloc_limbs(2 * cap),
412 qq = alloc_limbs(2 * cap);
413 lmmp_powmod_(r1, b, epa, 1, m, n);
414 lmmp_powmod_(r2, b, epb, 1, m, n);
415 lmmp_powmod_(dst, b, eps, 1, m, n);
416 lmmp_mul_(t, r1, n, r2, n);
417 lmmp_div_(qq, r1, t, 2 * n, m, n);
418 TEST_CHECK_MSG(memcmp(dst, r1, (size_t)n * 8) == 0, "even-mod exponent law");
419 lmmp_free(r1); lmmp_free(r2); lmmp_free(t); lmmp_free(qq);
420 }

422 // （odd==1 时退化为纯 2^k 模数，跳过）
423 if (odd > 1) {
424 mp_limb_t odb[1] = {odd};
425 // r2 = b^e mod odd
426 mp_ptr bmod = alloc_limbs(cap), r2 = alloc_limbs(cap);
427 lmmp_div_(NULL, bmod, b, n, odb, 1);
428 lmmp_powmod_(r2, bmod, eodd, 1, odb, 1);

471 u64 seed = 0x5ec0fefe1e55aa21ull;
472 struct modulus_case { u64 k; int e; int form; } shapes[] = {
473 {1, 64, 1}, {3, 64, 1}, {3, 65, 1}, {0xffffffffull, 127, 1},
474 {0xdeadbeefcafebull, 127, 1}, {0x123456789abcull, 129, 1},

476 {1, 127, -1}, {3, 64, -1}, {0xdeadbeefcafebull, 127, -1},
477 {0x123456789abcull, 191, -1}, {3, 300, -1}, {1, 19937, -1},
478 };
479 for (size_t sh_index = 0; sh_index < ARRAY_SIZE(shapes); ++sh_index) {
480 const struct modulus_case sh = shapes[sh_index];

482 mp_ptr m = alloc_limbs(cap), r = alloc_limbs(cap), t = alloc_limbs(2 * cap);
483 mp_size_t n = sp_shape_limbs(m, sh.k, sh.e, sh.form);
484
485 lmmp_mont_sp_t mc;
486 int form = lmmp_mont_sp_init_(&mc, m, n);
487 TEST_CHECK_MSG(form == sh.form, "sp probe form");
488
489 int samples = n > 100 ? 2 : 4;
490 for (int it = 0; it < samples; ++it) {
491 random_limbs(t, 2 * (size_t)n, &seed);
492 /* 顶半压制到 < m：顶 limb 掩码到 m 位长内，仍越则整体减 m */
493 u64 tb = (u64)top_bits(m[n - 1]);
494 t[2 * (size_t)n - 1] &= (tb == 64) ? ~(u64)0 : (((u64)1 << tb) - 1);
495 if (lmmp_cmp_(t + n, m, n) >= 0) (void)lmmp_sub_n_(t + n, t + n, m, n);
496
497 lmmp_mont_sp_redc_(r, t, &mc);
498 TEST_CHECK_MSG(lmmp_cmp_(r, m, n) < 0, "sp redc < m");
499 // r*B^n ≡ t (mod m)：经库除法双侧取余精确比对（t < m*B^n 已构造）
500 {
501 mp_ptr P = alloc_limbs(2 * n), q = alloc_limbs(n + 2), rm = alloc_limbs(n),
502 rt = alloc_limbs(n);
503 lmmp_zero(P, 2 * n);
504 memcpy(P + n, r, (size_t)n * 8); // P = r*B^n
505 lmmp_div_(q, rm, P, 2 * n, m, n);
506 lmmp_div_(q, rt, t, 2 * n, m, n);
507 TEST_CHECK_MSG(mpn_eq(rm, n, rt, n), "sp redc congruence");
508 lmmp_free(P); lmmp_free(q); lmmp_free(rm); lmmp_free(rt);
509 }
510 }
511 lmmp_free(m); lmmp_free(r); lmmp_free(t);
