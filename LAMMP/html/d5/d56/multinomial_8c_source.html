32static inline mp_size_t fac_size_lower(uint n) {
33 if (n < 20) {
34 return 0;
35 } else {
36 // log2(n!) = lgamma(n+1)/ln2，下界取floor
37 const double ln2 = 0.69314718055994531;
38 return (mp_size_t)floor(lgamma((double)n + 1) / ln2);
39 }
40}

42static inline mp_size_t fac_size_bigger(uint n) {
43 if (n < 20) {
44 return 64;
45 } else {
46 // log2(n!) = lgamma(n+1)/ln2，上界取ceil
47 const double ln2 = 0.69314718055994531;
48 return (mp_size_t)ceil(lgamma((double)n + 1) / ln2);
49 }

52mp_size_t lmmp_multinomial_size_(const uintp r, uint m, ulong* restrict n) {
53 ulong n_ret = 0;
54 uint i = 0;
55 for (; i < m; ++i) n_ret += r[i];
56
57 lmmp_param_assert(n_ret <= MP_UINT_MAX);
58 lmmp_param_assert(n_ret > 0);
59 *n = n_ret;
60 mp_size_t rn = fac_size_bigger(n_ret);
61
62 for (i = 0; i < m; ++i) {
63 rn -= fac_size_lower(r[i]);
64 }
65 rn = (rn + LIMB_BITS - 1) / LIMB_BITS;
66 return rn + 2;
67}

69static inline uint count_factors(fac_ptr fac, uint nfactors, uint n, const uintp r, uint m, uint p) {
70 uint pn = n;
71 uint e = 0;
72 ulong inv = MP_ULONG_MAX / p + 1;
73 while (pn > 0) {
74 _udiv32by32_q_preinv(pn, pn, inv);
75 e += pn;
76 }
77 for (uint i = 0; i < m; ++i) {
78 pn = r[i];
79 while (pn > 0) {
80 _udiv32by32_q_preinv(pn, pn, inv);
81 e -= pn;
82 }
83 }
84 if (e > 0) {
85 fac[nfactors].f = p;
86 fac[nfactors++].j = e;
87 }
88 return nfactors;
89}

91static inline uint factor_size_int(mp_size_t rn, uint n) {
92 /*
93 使用类似组合数的思路来估计缓冲区大小。
94 */
95 ulong approx1 = rn * 8;
96 lmmp_debug_assert(rn * 8 <= MP_UINT_MAX);
97 ulong approx2 = lmmp_prime_size_(n);
98 return approx1 < approx2 ? approx1 : approx2;

107static mp_size_t lmmp_odd_multinomial_ushort_(
108 mp_ptr restrict dst,
109 mp_size_t rn,
110 uint n,
111 const uintp restrict r,
112 uint m
113) {
114 if (n < ODD_FACTORIAL_SIZE) {
115 lmmp_odd_nPr_ushort_(dst, rn, n, n);
116 mp_limb_t t = 0;
117 for (uint i = 0; i < m; ++i) {
118 lmmp_odd_nPr_ushort_(&t, 1, r[i], r[i]);
119 dst[0] /= t;
120 }
121 return 1;
122 } else {
123 TEMP_DECL;
124 ushort primen = lmmp_prime_cnt16_(n);
125 ushort nfactors = factor_size_short(rn);
126 nfactors = primen < nfactors ? primen : nfactors;
127 fac_ptr restrict fac = TALLOC_TYPE(nfactors, fac_t);
128 nfactors = 0;
129 for (ushort i = 1; i < primen; ++i) {
130 ushort p = prime_short_table[i];
131 nfactors = count_factors(fac, nfactors, n, r, m, p);
132 }
133
134 rn = lmmp_factors_mul_ushort_(dst, rn, fac, nfactors);
135
136 TEMP_FREE;
137 return rn;
138 }
139}

141static mp_size_t lmmp_odd_multinomial_uint_(mp_ptr restrict dst, mp_size_t rn, uint n, const uintp restrict r, uint m) {
142 lmmp_prime_int_table_init_(n);
143 TEMP_B_DECL;
144 uint nfactors = factor_size_int(rn, n);
145 fac_ptr restrict fac = BALLOC_TYPE(nfactors, fac_t);
146
147 nfactors = 0;
148 prime_cache_t cache;
149 lmmp_prime_cache_init_(&cache, n);
150 while (cache.is_end == 0) {
151 lmmp_prime_cache_next_(&cache);
152 for (uint i = 0; i < cache.size; ++i) {
153 nfactors = count_factors(fac, nfactors, n, r, m, cache.pp[i]);
154 }
155 }
156 lmmp_prime_cache_free_(&cache);
157
158 rn = lmmp_factors_mul_(dst, rn, fac, nfactors);
159
160 TEMP_B_FREE;
161 return rn;
162}

167mp_size_t lmmp_multinomial_(mp_ptr restrict dst, mp_size_t rn, uint n, const uintp restrict r, uint m) {
168 mp_size_t shl = n - lmmp_limb_popcnt_(n);
169 for (uint j = 0; j < m; ++j) {
170 shl += lmmp_limb_popcnt_(r[j]);
171 shl -= r[j];
172 }
173 mp_size_t shw = shl / LIMB_BITS;
174 shl %= LIMB_BITS;
175 lmmp_zero(dst, shw);
176 if (n <= MULTINOMIAL_SHORT_LIMIT) {
177 rn = lmmp_odd_multinomial_ushort_(dst + shw, rn - shw, n, r, m);
178 } else {
179 rn = lmmp_odd_multinomial_uint_(dst + shw, rn - shw, n, r, m);
180 }
181
182 if (shl > 0) {
183 dst[shw + rn] = lmmp_shl_(dst + shw, dst + shw, rn, shl);
184 rn += shw + 1;
185 rn -= dst[rn - 1] == 0 ? 1 : 0;
186 } else {
187 rn += shw;
188 }
189 return rn;
190}
