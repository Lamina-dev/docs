27 lmmp_param_assert(n > 0);
28 lmmp_param_assert(numa != NULL);
29 lmmp_param_assert(numb != NULL);
30 /* 顶部剥皮 r = 4,1,2,3 个 limb（依 n mod 4，0 取 4）：短行程早退窗口，
31 余段长度恰为 4 的倍数，下方 4-limb 步进无需标量尾与运行时版本化 */
32 mp_size_t r = n & 3;
33 if (r == 0) r = 4;
34 for (mp_size_t k = 0; k < r; k++) {
35 mp_limb_t x = numa[n - 1 - k];
36 mp_limb_t y = numb[n - 1 - k];
37 if (x != y)
38 return (x > y ? 1 : -1);
39 }
40 /* 余段自顶向下按 4 limb 一组：异或折叠判差，有差组内自顶定位 */
41 for (mp_size_t i = n - r; i != 0;) {
42 i -= 4;
43 mp_limb_t d = (numa[i + 0] ^ numb[i + 0]) | (numa[i + 1] ^ numb[i + 1]) |
44 (numa[i + 2] ^ numb[i + 2]) | (numa[i + 3] ^ numb[i + 3]);
45 if (d) {
46 for (mp_size_t j = i + 3; j >= i; j--) {
47 if (numa[j] != numb[j])
48 return (numa[j] > numb[j] ? 1 : -1);
49 }
50 }
51 }
52 return 0;
53}
