23/*
24F[n] 表示 Fibonacci 数列第 n 项（F[0]=0, F[1]=1, F[n]=F[n-1]+F[n-2]），
25并假定了 F[-1] = 1（满足 F[1] = F[0] + F[-1]）。
26
27 F[2k-1] = F[k]^2 + F[k-1]^2
28 F[2k+1] = 4*F[k]^2 - F[k-1]^2 + 2*(-1)^k
29 F[2k] = F[2k+1] - F[2k-1] = F[k] * (F[k] + 2*F[k-1])
30 F[2k+1] = (2*F[k] + F[k-1]) * (2*F[k] - F[k-1]) + 2*(-1)^k
31
32fib2_core_:
33
34设 n 的二进制为 b_{m}...b_0。从查表得到的起点对
35(fp, f1p) = (F[k], F[k-1])（k 即 n 的最高若干位，k <= FIB_TABLE_LIMIT）
36出发，自高位向低位逐位扫描：每处理一位，先由前两组恒等式算出
37(F[2k+1], F[2k-1])（各需一次平方，共 2 次平方），再按当前位是 0/1 用
38F[2k] = F[2k+1] - F[2k-1] 替换掉不需要的那个，把数对推进为
39(F[2k], F[2k-1]) 或 (F[2k+1], F[2k])。终态即 (F[n], F[n-1])。
40
41F[2k+1] 中的 2*(-1)^k 修正不需要进位传播：
42 - k 奇（-2）：fp 此时恰为 F[k-1]^2。平方数 mod 4 in {0,1}，故其
43 bit1 必为 0，fp[0] |= 2 等价于 fp[0] += 2，即给被减数加 2，
44 相当于给结果减 2；
45 - k 偶（+2）：tp 左移 2 位后低 2 位为 0，tp[0] |= 2 等价于
46 tp[0] += 2，直接给 4*F[k]^2 加 2。
47
48lmmp_fibonacci_:
49
50仅求 F[n] 时不必算出整个数对：先以 fibonacci2 计算 k = n/2 处的
51(F[k], F[k-1])，再用第四组恒等式一次乘法合并：
52 - n 偶：F[n] = F[k] * (F[k] + 2*F[k-1])，1 次乘法；
53 - n 奇：F[n] = (2*F[k]+F[k-1]) * (2*F[k]-F[k-1]) + 2*(-1)^(n/2)，
54 1 次乘法。
55合并乘积直接写入 dst，两操作数长度之和 2*size+2（size 为 F[k] 的
56limb 长度）决定了 lmmp_fibonacci_size_ 的取值。
57
58合并处的 ±2 修正在低 limb 上直接完成：
59 - n/2 奇（k 奇，-2）：乘积 = F[n] + 2。F[4t+3] mod 8 in {1,2,5}
60 （F mod 8 以 12 为周期：0,1,1,2,3,5,0,5,5,2,7,1），不会为 6,7，
61 故乘积低 limb >= 2，减 2 绝不借位；
62 - n/2 偶（k 偶，+2）：乘积 = F[n] - 2，低 limb 加回 2。GMP 的经验
63 结论：F[4t+1] 在 n < 3*2^64+1 之内 mod 2^64 恒不为 0,1（已验证
64 至 b=32 且模式规整），故加 2 绝不进位。此处经由 lmmp_add_1_ 做
65 完整进位传播，即使该结论有误也只是多花 O(1) 时间，结果仍然正确。
66
67同时覆盖两个函数的最大写入范围（设 size = limbs(F[n>>1])）：
68 - fibonacci2_ 倍增循环最后一轮：2*size+1（平方写 2*size 个 limb，
69 进位落盘至下标 2*size）；
70 - fibonacci_ 合并乘法：xsize+ysize <= 2*size+2。
71n <= FIB_TABLE_LIMIT 时仅写 1 个 limb。
72*/

112/**
113 * @brief 计算 F[m] 的 limb 长度紧上界
114 * @param m Fibonacci 数列的下标
115 * @warning m>=0
116 * @return limbs(F[m]) <= ret，至多多算 1 个 limb

118static inline mp_size_t fib_limbs_bound_(ulong m) {
119 return (mp_size_t)(lmmp_mulh_(m, FIB_LOG2PHI_FP) / LIMB_BITS) + 1;

122mp_size_t lmmp_fibonacci_size_(ulong n) {
123 if (n <= FIB_TABLE_LIMIT) {
124 return 1;
125 }
126 return 2 * (fib_limbs_bound_(n >> 1) + 1) + 2;

129/**
130 * @brief Fibonacci 数对倍增核心：fp <- F[n]，f1p <- F[n-1]
131 * @param fp F[n] 输出指针
132 * @param f1p F[n-1] 输出指针
133 * @param tp 临时空间（alloc 个limb）
134 * @param alloc fp、f1p、tp 的公共容量（limb数）
135 * @param n 下标
136 * @warning fp!=NULL, f1p!=NULL, tp!=NULL, sep(fp,f1p,tp),
137 * alloc >= 2*(limbs(F[n>>1])+1)+1
138 * @return F[n] 的实际 limb 长度（n==0 时 fp[0]==0）；
139 * F[n-1] 至多多写一个高位0，实际长度为 ret-(f1p[ret-1]==0)
140 */
141static mp_size_t fib2_core_(mp_ptr fp, mp_ptr f1p, mp_ptr tp, mp_size_t alloc, ulong n) {
142 ulong nfirst, mask;
143
144 /* 自顶截去低位直至落入查表范围，mask 对齐到 n 的剩余最高位 */
145 mask = 1;
146 for (nfirst = n; nfirst > FIB_TABLE_LIMIT; nfirst >>= 1) {
147 mask <<= 1;
148 }
149
150 f1p[0] = fib_table[nfirst]; /* F[nfirst-1] */
151 fp[0] = fib_table[nfirst + 1]; /* F[nfirst] */
152 mp_size_t size = 1;
153
154 if (mask == 1) {
155 return size;
156 }
157
158 do {
159 /* 此处 fp == F[k]，f1p == F[k-1]，k 为 n 自 mask 位向上的位串；
160 fp 归一化，f1p 至多一个高位0 */
161 lmmp_debug_assert(alloc >= 2 * size);
162 lmmp_sqr_(tp, fp, size); /* tp = F[k]^2 */
163 lmmp_sqr_(fp, f1p, size); /* fp = F[k-1]^2 */
164 size <<= 1;
165 size -= (tp[size - 1] == 0); /* fp 归一化故 tp 至多一个高位0 */
166
167 /* F[2k-1] = F[k]^2 + F[k-1]^2 */
168 f1p[size] = lmmp_add_n_(f1p, tp, fp, size);
169
170 /* F[2k+1] = 4*F[k]^2 - F[k-1]^2 + 2*(-1)^k，n&mask 即 k 的最低位 */
171 lmmp_debug_assert((fp[0] & 2) == 0); /* fp 为平方数，bit1 必为0 */
172 fp[0] |= (n & mask ? 2 : 0); /* k 奇：给被减数加2即结果减2 */
173 mp_limb_t c = lmmp_shl_(tp, tp, size, 2); /* tp = 4*F[k]^2（溢出位入c）*/
174 tp[0] |= (n & mask ? 0 : 2); /* k 偶：4*F[k]^2 加2 */
175 fp[size] = c - lmmp_sub_n_(fp, tp, fp, size);
176 lmmp_debug_assert(alloc >= size + 1);
177 size += (fp[size] != 0);
178
179 /* 下一位 n&(mask>>1)：为1则推进到 (F[2k+1], F[2k])，
180 为0则推进到 (F[2k], F[2k-1])，F[2k] = F[2k+1] - F[2k-1] */
181 mask >>= 1;
182 if (n & mask) {
183 lmmp_sub_n_(f1p, fp, f1p, size);
184 } else {
185 lmmp_sub_n_(fp, fp, f1p, size);
186 size -= (fp[size - 1] == 0);
187 }
188 } while (mask != 1);
189
190 return size;
191}

193mp_size_t lmmp_fibonacci2_(mp_ptr dst, mp_ptr dst2, mp_size_t rn, ulong n) {
194 lmmp_param_assert(dst != NULL && dst2 != NULL);
195#if LMMP_DEBUG_PARAM_ASSERT_CHECK == 1
196 mp_size_t need = lmmp_fibonacci_size_(n);
197 lmmp_param_assert(rn >= need);
198#endif
199 if (n <= FIB_TABLE_LIMIT) {
200 dst[0] = fib_table[n + 1]; /* F[n] */
201 dst2[0] = fib_table[n]; /* F[n-1]，n==0 时即 F[-1]=1 */
202 return 1;
203 }
204 TEMP_DECL;
205 mp_ptr tp = TALLOC_TYPE(rn, mp_limb_t);
206 mp_size_t size = fib2_core_(dst, dst2, tp, rn, n);
207 TEMP_FREE;
208 return size;
209}

211mp_size_t lmmp_fibonacci_(mp_ptr dst, mp_size_t rn, ulong n) {
212 lmmp_param_assert(dst != NULL);
213#if LMMP_DEBUG_PARAM_ASSERT_CHECK == 1
214 mp_size_t need = lmmp_fibonacci_size_(n);
215 lmmp_param_assert(rn >= need);
216#else
217 (void)rn;
218#endif
219 if (n <= FIB_TABLE_LIMIT) {
220 dst[0] = fib_table[n + 1]; /* F[n]，n==0 时为0 */
221 return (n == 0) ? 0 : 1;
222 }
223
224 /* k = n/2，先算出 (F[k], F[k-1])，再一次乘法合并。
225 limbs_bound(n2)+2 同时覆盖核心循环（2*limbs(F[n2>>1])+1）与
226 合并操作数 xp/yp 的写入（size+2）*/
227 ulong n2 = n >> 1;
228 mp_size_t xalloc = fib_limbs_bound_(n2) + 2;
229 TEMP_DECL;
230 mp_ptr xp = TALLOC_TYPE(xalloc, mp_limb_t);
231 mp_ptr yp = TALLOC_TYPE(xalloc, mp_limb_t);
232 mp_ptr tp = TALLOC_TYPE(xalloc, mp_limb_t);
233 mp_size_t size = fib2_core_(xp, yp, tp, xalloc, n2);
234 mp_size_t xsize, ysize;
235
236 if (n & 1) {
237 /* F[2k+1] = (2F[k]+F[k-1]) * (2F[k]-F[k-1]) + 2*(-1)^k */
238 mp_limb_t c2 = lmmp_shl_(dst, xp, size, 1); /* dst 暂存 2F[k] 低 size 位 */
239 mp_limb_t c = lmmp_add_n_(xp, dst, yp, size);
240 c += c2; /* 2F[k] 的溢出位并入最高 limb，c=[0|1|2] */
241 xp[size] = c;
242 xsize = size + (c != 0);
243 c2 -= lmmp_sub_n_(yp, dst, yp, size); /* yp = 2F[k]-F[k-1]，c2=[0|1] */
244 lmmp_debug_assert(c2 <= 1);
245 yp[size] = c2;
246 ysize = size + c2;
247
248 lmmp_mul_(dst, xp, xsize, yp, ysize);
249 size = xsize + ysize;
250 if (n & 2) {
251 mp_limb_t b = lmmp_sub_1_(dst, dst, size, 2);
252 lmmp_debug_assert(b == 0);
253 } else {
254 mp_limb_t c3 = lmmp_add_1_(dst, dst, size, 2);
255 lmmp_debug_assert(c3 == 0);
256 }
257 } else {
258 /* F[2k] = F[k] * (F[k] + 2F[k-1]) */
259 mp_limb_t c = lmmp_addshl1_n_(yp, xp, yp, size); /* yp = F[k]+2F[k-1] */
260 yp[size] = c;
261 ysize = size + (c != 0);
262
263 lmmp_mul_(dst, yp, ysize, xp, size);
264 size += ysize;
265 }
266
267 size -= (dst[size - 1] == 0);
268 TEMP_FREE;
269 return size;
270}
