46enum { BIG_N = 64, DECIMAL_CAP = 20 * BIG_N + 8, FACTORIAL_N = 65537 };

48static void fill_big(mp_ptr a) {
49 for (mp_size_t i = 0; i < BIG_N; ++i)
50 a[i] = 0x9e3779b97f4a7c15ull * (i + 1) + 0x123456789abcdefull;
51 a[BIG_N - 1] |= 1;
52}

54static mp_size_t big_to_decimal(mp_byte_t *out, mp_srcptr a) {
55 return lmmp_to_str_(out, a, BIG_N, 10);

58static mp_ptr factorial_buffer(mp_size_t n) {
59 mp_ptr buffer = calloc(n + 1, sizeof(*buffer));
60 if (!buffer)
61 abort();
62 return buffer;

65TEST_CASE("global/lifecycle", lifecycle_idempotent) {
66 mp_limb_t a[BIG_N], dst[2 * BIG_N], dst_ref[2 * BIG_N];
67 mp_byte_t before[DECIMAL_CAP], after[DECIMAL_CAP];
68 fill_big(a);
69
70 lmmp_global_init();
71 lmmp_global_init();
72
73 mp_size_t len = big_to_decimal(before, a);
74 TEST_CHECK(len > BIG_N * 15);
75 lmmp_mul_(dst_ref, a, BIG_N, a, BIG_N);
76
77 mp_bitcnt_t bits = 0;
78 mp_size_t rn = lmmp_factorial_size_(FACTORIAL_N, &bits);
79 mp_ptr fac = factorial_buffer(rn);
80 TEST_CHECK(lmmp_factorial_(fac, bits, rn + 1, FACTORIAL_N) > 0);
81
82 lmmp_global_deinit();
83 lmmp_global_deinit();
84 lmmp_global_init();
85
86 TEST_CHECK(big_to_decimal(after, a) == len && memcmp(after, before, len) == 0);
87 lmmp_mul_(dst, a, BIG_N, a, BIG_N);
88 TEST_CHECK(memcmp(dst, dst_ref, sizeof(dst)) == 0);
89
90 mp_bitcnt_t bits2 = 0;
91 mp_size_t rn2 = lmmp_factorial_size_(FACTORIAL_N, &bits2);
92 TEST_CHECK(rn2 == rn && bits2 == bits);
93 mp_ptr fac2 = factorial_buffer(rn2);
94 TEST_CHECK(lmmp_factorial_(fac2, bits2, rn2 + 1, FACTORIAL_N) > 0);
95 TEST_CHECK(rn == rn2 && memcmp(fac, fac2, (rn + 1) * sizeof(*fac)) == 0);
96 free(fac2);
97 free(fac);
98}

100static atomic_long live_allocations;
101
102static void *count_alloc(size_t n) {
103 void *p = malloc(n);
104 if (p)
105 ++live_allocations;
106 return p;
107}
108
109static void count_free(void *p) {
110 if (p)
111 --live_allocations;
112 free(p);
113}
114
115static void *count_realloc(void *p, size_t n) {
116 return p ? realloc(p, n) : count_alloc(n);
117}

119TEST_CASE("global/lifecycle", heap_balance_full_cycle) {
120 const lmmp_heap_allocator_t counting = {count_alloc, count_free, count_realloc};
121 const lmmp_heap_allocator_t libc = {malloc, free, realloc};
122 lmmp_set_heap_allocator(&counting);
123 long baseline = atomic_load(&live_allocations);
124 TEST_CHECK(baseline > 0);
125
126 mp_limb_t a[BIG_N], dst[2 * BIG_N];
127 mp_byte_t decimal[DECIMAL_CAP];
128 fill_big(a);
129 lmmp_mul_(dst, a, BIG_N, a, BIG_N);
130 big_to_decimal(decimal, a);
131
132 mp_bitcnt_t bits = 0;
133 mp_size_t rn = lmmp_factorial_size_(FACTORIAL_N, &bits);
134 mp_ptr fac = factorial_buffer(rn);
135 lmmp_factorial_(fac, bits, rn + 1, FACTORIAL_N);
136 free(fac);
137 TEST_CHECK(atomic_load(&live_allocations) > baseline);
138
139 lmmp_global_deinit();
140 TEST_CHECK(atomic_load(&live_allocations) == 0);
141 lmmp_global_init();
142 TEST_CHECK(atomic_load(&live_allocations) == baseline);
143 lmmp_set_heap_allocator(&libc);
144 TEST_CHECK(atomic_load(&live_allocations) == 0);
145}

147typedef struct {
148 mp_srcptr input;
149 const mp_byte_t *decimal;

155#if defined(LMMP_WINDOWS)
156static unsigned __stdcall thread_worker(void *arg) {
157#else
158static void *thread_worker(void *arg) {
159#endif
160 thread_case *test = arg;
161 lmmp_global_init();
162
163 mp_byte_t decimal[DECIMAL_CAP];
164 bool match = big_to_decimal(decimal, test->input) == test->len &&
165 memcmp(decimal, test->decimal, test->len) == 0;
166 mp_bitcnt_t bits = 0;
167 mp_size_t rn = lmmp_factorial_size_(FACTORIAL_N, &bits);
168 mp_ptr fac = factorial_buffer(rn);
169 test->pass = match && rn == test->rn && bits == test->bits &&
170 lmmp_factorial_(fac, bits, rn + 1, FACTORIAL_N) > 0;
171 free(fac);
172
173 lmmp_global_deinit();
174 return 0;
175}

177TEST_CASE("global/thread", thread_isolation) {
178 enum { WORKERS = 4 };
179 mp_limb_t a[BIG_N];
180 mp_byte_t expected[DECIMAL_CAP], decimal[DECIMAL_CAP];
181 fill_big(a);
182 mp_size_t len = big_to_decimal(expected, a);
183 mp_bitcnt_t bits = 0;
184 mp_size_t rn = lmmp_factorial_size_(FACTORIAL_N, &bits);
185 thread_case cases[WORKERS];
186#if defined(LMMP_WINDOWS)
187 HANDLE workers[WORKERS];
188#else
189 pthread_t workers[WORKERS];
190#endif
191 size_t started = 0;
192 for (; started < WORKERS; ++started) {
193 cases[started] = (thread_case){a, expected, len, rn, bits, false};
194#if defined(LMMP_WINDOWS)
195 workers[started] = (HANDLE)_beginthreadex(NULL, 0, thread_worker, &cases[started], 0, NULL);
196 if (!workers[started])
197 break;
198#else
199 if (pthread_create(&workers[started], NULL, thread_worker, &cases[started]) != 0)
200 break;
201#endif
202 }
203 TEST_CHECK_EQ(started, WORKERS);
204
205 for (int repeat = 0; repeat < 8; ++repeat)
206 TEST_CHECK(big_to_decimal(decimal, a) == len && memcmp(decimal, expected, len) == 0);
207
208 for (size_t i = 0; i < started; ++i) {
209#if defined(LMMP_WINDOWS)
210 if (WaitForSingleObject(workers[i], INFINITE) != WAIT_OBJECT_0)
211 abort();
212 CloseHandle(workers[i]);
213#else
214 if (pthread_join(workers[i], NULL) != 0)
215 abort();
216#endif
217 TEST_CHECK_MSG(cases[i].pass, "worker thread result mismatch");
218 }
219 TEST_CHECK(big_to_decimal(decimal, a) == len && memcmp(decimal, expected, len) == 0);
220}
