29/*
30mont63 只能用于小于 2^63 的数，否则会溢出导致计算结果不正确
31mont64 可以用于任意大小的数，但由于考虑了溢出的情况，所以速度理论上会慢一些（并未详细测试）
32
33（本文件与 is_prime_ulong.c 各持一份同源拷贝，库内惯例勿强行收敛）
34*/
35#define MONT63_MAX ((ulong)(0x7fffffffffffffff))

37static inline ulong mont64_reduce(u128 t, ulong m, ulong m_inv) {
38 // sum = k*m + t < 2^129，第 129 位 c 由 128 位回绕捕获
39 ulong k = _u128low(t) * m_inv;
40 u128 km = (u128)k * m;
41 u128 sum = km + t;
42 uint c = sum < km;
43 ulong hi = _u128high(sum);
44 if (c) return hi - m; // 2^128 + hi - m 的低 128 位高位即结果
45 return hi >= m ? hi - m : hi;
46}

70static inline ulong mont63_reduce(u128 t, ulong m, ulong m_inv) {
71 // m < 2^63 时 k*m + t < 2^127，无 129 位溢出
72 ulong k = _u128low(t) * m_inv;
73 u128 sum = (u128)k * m + t;
74 ulong hi = _u128high(sum);
75 return hi >= m ? hi - m : hi;
76}

94uint lmmp_powmod_uint_odd_(uint base, ulong exp, uint mod) {
95 lmmp_param_assert(mod > base);
96 lmmp_param_assert(mod % 2 == 1);
97 lmmp_param_assert(mod > 1);
98 ulong dst = 1;
99 ulong b = base;
100 _udiv64_t binv = _udiv64_gen(mod);
101 ulong q;
102 while (1) {
103 if (exp & 1) {
104 dst *= b;
105 q = _udiv64by64_q_preinv(dst, &binv);
106 dst -= q * mod;
107 }
108 exp >>= 1;
109 if (exp == 0)
110 break;
111 b *= b;
112 q = _udiv64by64_q_preinv(b, &binv);
113 b -= q * mod;
114 }
115 return dst;
116}

118ulong lmmp_powmod_ulong_odd_(ulong base, ulong exp, ulong mod) {
119 lmmp_param_assert(mod > base);
120 lmmp_param_assert(mod % 2 == 1);
121 lmmp_param_assert(mod > 1);
122 if (mod <= MP_UINT_MAX)
123 return lmmp_powmod_uint_odd_(base, exp, mod);
124 else if (mod <= MONT63_MAX) {
125 ulong R2 = mont63_R2(mod);
126 ulong m_inv = lmmp_binvert_ulong_(mod);
127 m_inv = -m_inv;
128 ulong dst = to_mont63(1, R2, mod, m_inv);
129 base = to_mont63(base, R2, mod, m_inv);
130 while (1) {
131 if (exp & 1)
132 dst = mont63_mul(dst, base, mod, m_inv);
133 exp >>= 1;
134 if (exp == 0)
135 break;
136 base = mont63_mul(base, base, mod, m_inv);
137 }
138 return from_mont63(dst, mod, m_inv);
139 } else {
140 ulong R2 = mont64_R2(mod);
141 ulong m_inv = lmmp_binvert_ulong_(mod);
142 m_inv = -m_inv;
143 ulong dst = to_mont64(1, R2, mod, m_inv);
144 base = to_mont64(base, R2, mod, m_inv);
145 while (1) {
146 if (exp & 1)
147 dst = mont64_mul(dst, base, mod, m_inv);
148 exp >>= 1;
149 if (exp == 0)
150 break;
151 base = mont64_mul(base, base, mod, m_inv);
152 }
153 return from_mont64(dst, mod, m_inv);
154 }
155}
