24/**
25 * @brief 费马变换乘法递归函数（核心乘法逻辑）
26 * @param ms - 内存栈结构体指针
27 * @param pc1 - 第一个数的FFT系数数组指针数组
28 * @param pc2 - 第二个数的FFT系数数组指针数组
29 * @param K0 - FFT块数
30 * @warning K0>0
31 * 所有系数均已伪归一化（mod B^lenw+1）
32 * nsqr=1表示乘法，nsqr=0表示平方
33 */
34static void lmmp_mul_fermat_recurse_(fft_memstack* ms, mp_ptr* pc1, mp_ptr* pc2, mp_size_t K0) {
35 const int nsqr = 1;
36 mp_ptr push_temp_coef = ms->temp_coef;
37 mp_size_t rn = ms->lenw;
38
39 if (rn < MUL_FFT_MODF_THRESHOLD) {
40 mp_ptr temp_mul = (mp_ptr)lmmp_fft_memstack_(ms, (rn + 1) * 2 * LIMB_BYTES);
41 for (mp_size_t i = 0; i < K0; ++i) {
42 lmmp_mul_n_(temp_mul, pc1[i], pc2[i], rn + 1);
43
44 // 模 B^rn+1 归一化：temp_mul - temp_mul[rn ...]
45 mp_limb_t maxc = lmmp_sub_n_(pc1[i], temp_mul, temp_mul + rn, rn) + temp_mul[rn * 2];
46 pc1[i][rn] = 0;
47 lmmp_inc_1(pc1[i], maxc);
48 }
49 lmmp_fft_memstack_(ms, 0);
50 } else {
51 mp_size_t N = rn * LIMB_BITS; // 总比特数
52 mp_size_t k = lmmp_fft_best_k_(rn);
53 mp_size_t K = ((mp_size_t)1) << k; // FFT块数（2^k）
54 lmmp_debug_assert(!(N & (K - 1)));
55 mp_size_t M = N >> k; // 每个块的比特数（N/K）
56 mp_size_t n = 2 * M + k + 2; // 扩展系数长度（保证归一化）
57
58 // 规整n：必须是LIMB_BITS和K的整数倍
59 n = (n + LIMB_BITS - 1) & (-LIMB_BITS);
60 n = (((n - 1) >> k) + 1) << k;
61
62 ms->lenw = n / LIMB_BITS;
63 mp_size_t nlen = ms->lenw + 1;
64
65 ms->temp_coef = (mp_ptr)lmmp_fft_memstack_(ms, (((nlen + 1) << (k + nsqr)) + nlen) * LIMB_BYTES);
66 mp_ptr *pfca = (mp_ptr*)(ms->temp_coef + nlen), *pfcb = pfca;
67 for (mp_size_t i = 0; i < K; ++i) pfca[i] = (mp_ptr)(pfca + K) + i * nlen;
68 pfcb += (nlen + 1) << k;
69 for (mp_size_t i = 0; i < K; ++i) pfcb[i] = (mp_ptr)(pfcb + K) + i * nlen;
70
71 for (mp_size_t j = 0; j < K0; ++j) {
72 mp_ptr numa = pc1[j];
73 mp_ptr numb = pc2[j];
74
75 for (mp_size_t i = 0; i < K; ++i) {
76 lmmp_fft_extract_coef_(pfca[i], numa, M * i, M + (i == K - 1), ms->lenw);
77 if (i > 0)
78 lmmp_fft_shl_coef_(ms, pfca + i, i * n >> k);
79 }
80 lmmp_fft_(ms, pfca, k, n >> (k - 1));
81
82 for (mp_size_t i = 0; i < K; ++i) {
83 lmmp_fft_extract_coef_(pfcb[i], numb, M * i, M + (i == K - 1), ms->lenw);
84 if (i > 0)
85 lmmp_fft_shl_coef_(ms, pfcb + i, i * n >> k);
86 }
87 lmmp_fft_(ms, pfcb, k, n >> (k - 1));
88
89 // dot product
90 lmmp_mul_fermat_recurse_(ms, pfca, pfcb, K);
91
92 lmmp_ifft_(ms, pfca, k, n >> (k - 1));
93
94 lmmp_mul_fermat_recombine_(ms, numa, pfca, K, k, n, M, rn);
95 }
96 lmmp_fft_memstack_(ms, 0);
97 }
98
99 ms->temp_coef = push_temp_coef;
100 ms->lenw = rn;
101}

103void lmmp_mul_fermat_(mp_ptr dst, mp_size_t rn, mp_srcptr numa, mp_size_t na, mp_srcptr numb, mp_size_t nb) {
104 const int nsqr = 1;
105 mp_size_t N = rn * LIMB_BITS; // 结果总比特数
106 mp_size_t k = lmmp_fft_best_k_(rn);
107 mp_size_t K = ((mp_size_t)1) << k; // FFT块数（2^k）
108 lmmp_debug_assert(!(N & (K - 1)));
109 mp_size_t M = N >> k; // 每个块的比特数
110 mp_size_t n = 2 * M + k + 2; // 扩展系数长度
111
112 n = (n + LIMB_BITS - 1) & (-LIMB_BITS);
113 n = (((n - 1) >> k) + 1) << k;
114
115 // 初始化内存栈
116 fft_memstack msr;
117 msr.maxdepth = -1;
118 msr.tempdepth = -1;
119 msr.lenw = n / LIMB_BITS;
120 mp_size_t nlen = msr.lenw + 1;
121
122 msr.temp_coef = (mp_ptr)lmmp_fft_memstack_(&msr, (((nlen + 1) << (k + nsqr)) + nlen) * LIMB_BYTES);
123
124 mp_ptr *pfca = (mp_ptr*)(msr.temp_coef + nlen), *pfcb = pfca;
125 mp_size_t narest = na * LIMB_BITS, nbrest = nb * LIMB_BITS;
126
127 for (mp_size_t i = 0; i < K; ++i) {
128 mp_size_t coeflen;
129 pfca[i] = (mp_ptr)(pfca + K) + i * nlen;
130 if (narest > 0) {
131 coeflen = M + (i == K - 1);
132 coeflen = LMMP_MIN(narest, coeflen);
133 narest -= coeflen;
134 lmmp_fft_extract_coef_(pfca[i], numa, M * i, coeflen, msr.lenw);
135 if (i > 0)
136 lmmp_fft_shl_coef_(&msr, pfca + i, i * n >> k);
137 } else {
138 lmmp_zero(pfca[i], nlen);
139 }
140 }
141 lmmp_fft_(&msr, pfca, k, n >> (k - 1));
142
143 pfcb += (nlen + 1) << k;
144 for (mp_size_t i = 0; i < K; ++i) {
145 mp_size_t coeflen;
146 pfcb[i] = (mp_ptr)(pfcb + K) + i * nlen;
147 if (nbrest > 0) {
148 coeflen = M + (i == K - 1);
149 coeflen = LMMP_MIN(nbrest, coeflen);
150 nbrest -= coeflen;
151 lmmp_fft_extract_coef_(pfcb[i], numb, M * i, coeflen, msr.lenw);
152 if (i > 0)
153 lmmp_fft_shl_coef_(&msr, pfcb + i, i * n >> k);
154 } else {
155 lmmp_zero(pfcb[i], nlen);
156 }
157 }
158 lmmp_fft_(&msr, pfcb, k, n >> (k - 1));
159
160 lmmp_mul_fermat_recurse_(&msr, pfca, pfcb, K);
161
162 lmmp_ifft_(&msr, pfca, k, n >> (k - 1));
163
164 lmmp_mul_fermat_recombine_(&msr, dst, pfca, K, k, n, M, rn);
165
166 // 处理模 B^rn+1 的溢出
167 if (dst[rn] && !lmmp_zero_q_(dst, rn)) {
168 dst[rn] = 0;
169 lmmp_dec(dst);
170 }
171
172 lmmp_fft_memstack_(&msr, 0);
173}

175void lmmp_mul_mersenne_(mp_ptr dst, mp_size_t rn, mp_srcptr numa, mp_size_t na, mp_srcptr numb, mp_size_t nb) {
176 const int nsqr = 1;
177 mp_size_t N = rn * LIMB_BITS; // 结果总比特数
178 mp_size_t k = lmmp_fft_best_k_(rn); // 最优FFT层数
179 mp_size_t K = ((mp_size_t)1) << k; // FFT块数（2^k）
180 // 断言：N必须是K的整数倍
181 lmmp_debug_assert(!(N & (K - 1)));
182 mp_size_t M = N >> k; // 每个块的比特数
183 mp_size_t n = 2 * M + k; // 扩展系数长度（梅森数比费马数少2）
184
185 // 规整n：必须是LIMB_BITS和K/2的整数倍
186 n = (n + LIMB_BITS - 1) & (-LIMB_BITS);
187 n = (((n - 1) >> (k - 1)) + 1) << (k - 1);
188
189 // 初始化内存栈
190 fft_memstack msr;
191 msr.maxdepth = -1;
192 msr.tempdepth = -1;
193 msr.lenw = n / LIMB_BITS;
194 mp_size_t nlen = msr.lenw + 1;
195
196 msr.temp_coef = (mp_ptr)lmmp_fft_memstack_(&msr, (((nlen + 1) << (k + nsqr)) + nlen) * LIMB_BYTES);
197
198 mp_ptr *pfca = (mp_ptr*)(msr.temp_coef + nlen), *pfcb = pfca;
199 mp_size_t narest = na * LIMB_BITS, nbrest = nb * LIMB_BITS;
200
201 for (mp_size_t i = 0; i < K; ++i) {
202 mp_size_t coeflen;
203 pfca[i] = (mp_ptr)(pfca + K) + i * nlen;
204 if (narest > 0) {
205 coeflen = LMMP_MIN(narest, M);
206 narest -= coeflen;
207 lmmp_fft_extract_coef_(pfca[i], numa, M * i, coeflen, msr.lenw);
208 } else {
209 lmmp_zero(pfca[i], nlen);
210 }
211 }
212 lmmp_fft_(&msr, pfca, k, n >> (k - 1));
213
214 pfcb += (nlen + 1) << k;
215 for (mp_size_t i = 0; i < K; ++i) {
216 mp_size_t coeflen;
217 pfcb[i] = (mp_ptr)(pfcb + K) + i * nlen;
218 if (nbrest > 0) {
219 coeflen = LMMP_MIN(nbrest, M);
220 nbrest -= coeflen;
221 lmmp_fft_extract_coef_(pfcb[i], numb, M * i, coeflen, msr.lenw);
222 } else {
223 lmmp_zero(pfcb[i], nlen);
224 }
225 }
226 lmmp_fft_(&msr, pfcb, k, n >> (k - 1));
227
228 lmmp_mul_fermat_recurse_(&msr, pfca, pfcb, K);
229
230 lmmp_ifft_(&msr, pfca, k, n >> (k - 1));
231
232 mp_size_t rhead = 0, maxc = 0;
233 for (mp_size_t i = 0; i < K; ++i) {
234 lmmp_fft_shr_coef_(&msr, pfca + i, k);
235 mp_ptr nums = pfca[i];
236
237 if (nums[nlen - 1]) {
238 lmmp_dec(nums);
239 lmmp_debug_assert(nums[nlen - 1] == 1);
240 nums[nlen - 1] = 0;
241 }
242
243 mp_size_t roffset = i * M;
244 mp_size_t shl = roffset & (LIMB_BITS - 1);
245 roffset /= LIMB_BITS;
246
247 if (shl)
248 lmmp_shl_(nums, nums, nlen, shl);
249
250 if (i == 0) {
251 lmmp_copy(dst, nums, nlen);
252 rhead = nlen;
253 } else if (roffset + nlen <= rn) {
254 lmmp_add_(dst + roffset, nums, nlen, dst + roffset, rhead - roffset);
255 rhead = roffset + nlen;
256 } else {
257 maxc += lmmp_add_(dst + roffset, nums, rn - roffset, dst + roffset, rhead - roffset);
258 maxc += lmmp_add_(dst, dst, rn, nums + rn - roffset, nlen + roffset - rn);
259 rhead = rn;
260 }
261 }
262
263 if (!lmmp_add_1_(dst, dst, rn, 1 + maxc))
264 lmmp_dec(dst);
265
266 lmmp_fft_memstack_(&msr, 0);
267}

482 lmmp_dec(nums);
483 lmmp_debug_assert(nums[ctx->nlen - 1] == 1);
484 nums[ctx->nlen - 1] = 0;
485 }
486
487 mp_size_t roffset = i * ctx->M;
488 mp_size_t shl = roffset & (LIMB_BITS - 1);
489 roffset /= LIMB_BITS;
490
491 if (shl)
492 lmmp_shl_(nums, nums, ctx->nlen, shl);
493
494 if (i == 0) {
495 lmmp_copy(dst, nums, ctx->nlen);
496 rhead = ctx->nlen;
497 } else if (roffset + ctx->nlen <= rn) {
498 lmmp_add_(dst + roffset, nums, ctx->nlen, dst + roffset, rhead - roffset);
499 rhead = roffset + ctx->nlen;
500 } else {
501 maxc += lmmp_add_(dst + roffset, nums, rn - roffset, dst + roffset, rhead - roffset);
502 maxc += lmmp_add_(dst, dst, rn, nums + rn - roffset, ctx->nlen + roffset - rn);
503 rhead = rn;
504 }
505 }
506
507 if (!lmmp_add_1_(dst, dst, rn, 1 + maxc))
508 lmmp_dec(dst);
509
510 lmmp_fft_memstack_(amsr, 0);
511}

513void lmmp_mul_mersenne_cache_(mp_ptr dst, mp_srcptr numa, fft_gr_cache* ctx) {
514 lmmp_param_assert(dst != NULL && numa != NULL);
515 lmmp_param_assert(ctx != NULL);
516 fft_memstack* bmsr = &ctx->bmsr;
517 fft_memstack* amsr = &ctx->amsr;
518
519 amsr->maxdepth = -1;
520 amsr->tempdepth = -1;
521 amsr->lenw = ctx->n / LIMB_BITS;
522 amsr->temp_coef = (mp_ptr)lmmp_fft_memstack_(amsr, ctx->a_size);
523
524 bmsr->temp_coef = ctx->temp_coefb;
525 mp_ptr* pfca = (mp_ptr*)(amsr->temp_coef + ctx->nlen);
526 mp_ptr* pfcb = (mp_ptr*)(bmsr->temp_coef + ctx->nlen);
527
528 mp_size_t narest = ctx->na * LIMB_BITS;
529
530 for (mp_size_t i = 0; i < ctx->K; ++i) {
531 mp_size_t coeflen;
532 pfca[i] = (mp_ptr)(pfca + ctx->K) + i * ctx->nlen;
533 if (narest > 0) {
534 coeflen = ctx->M + (i == ctx->K - 1);
535 coeflen = LMMP_MIN(narest, ctx->M);
536 narest -= coeflen;
537 lmmp_fft_extract_coef_(pfca[i], numa, ctx->M * i, coeflen, amsr->lenw);
538 } else {
539 lmmp_zero(pfca[i], ctx->nlen);
540 }
541 }
542 lmmp_fft_(amsr, pfca, ctx->k, ctx->n >> (ctx->k - 1));
543
544 lmmp_mul_fermat_recurse_(amsr, pfca, pfcb, ctx->K);
545
546 lmmp_ifft_(amsr, pfca, ctx->k, ctx->n >> (ctx->k - 1));
547
548 mp_size_t rhead = 0, maxc = 0;
549 for (mp_size_t i = 0; i < ctx->K; ++i) {
550 lmmp_fft_shr_coef_(amsr, pfca + i, ctx->k);
551 mp_ptr nums = pfca[i];
552
553 if (nums[ctx->nlen - 1]) {
554 lmmp_dec(nums);
555 lmmp_debug_assert(nums[ctx->nlen - 1] == 1);
556 nums[ctx->nlen - 1] = 0;
557 }
558
559 mp_size_t roffset = i * ctx->M;
560 mp_size_t shl = roffset & (LIMB_BITS - 1);
561 roffset /= LIMB_BITS;
562
563 if (shl)
564 lmmp_shl_(nums, nums, ctx->nlen, shl);
565
566 if (i == 0) {
567 lmmp_copy(dst, nums, ctx->nlen);
568 rhead = ctx->nlen;
569 } else if (roffset + ctx->nlen <= ctx->rn) {
570 lmmp_add_(dst + roffset, nums, ctx->nlen, dst + roffset, rhead - roffset);
571 rhead = roffset + ctx->nlen;
572 } else {
573 maxc += lmmp_add_(dst + roffset, nums, ctx->rn - roffset, dst + roffset, rhead - roffset);
574 maxc += lmmp_add_(dst, dst, ctx->rn, nums + ctx->rn - roffset, ctx->nlen + roffset - ctx->rn);
575 rhead = ctx->rn;
576 }
577 }
578
579 if (!lmmp_add_1_(dst, dst, ctx->rn, 1 + maxc))
580 lmmp_dec(dst);
581
582 lmmp_fft_memstack_(amsr, 0);
583}

585void lmmp_mul_fft_(mp_ptr dst, mp_srcptr numa, mp_size_t na, mp_srcptr numb, mp_size_t nb) {
586 lmmp_param_assert(na > 0 && nb > 0);
587 lmmp_param_assert(na >= nb);
588 lmmp_param_assert(dst != NULL && numa != NULL && numb != NULL);
589 mp_size_t hn = lmmp_fft_next_size_((na + nb + 1) >> 1);
590 lmmp_assert(na + nb > hn);
591 mp_ptr tp = ALLOC_TYPE(hn + 1, mp_limb_t);
592
593 mp_srcptr amodm = numa;
594 mp_size_t nam = na;
595 if (na > hn) {
596 /*
597 Z = B^hb - 1
598 amodm = a mod Z
599 */
600 if (lmmp_add_(dst, numa, hn, numa + hn, na - hn))
601 lmmp_inc(dst);
602 amodm = dst;
603 nam = hn;
604 }
605 lmmp_mul_mersenne_(dst, hn, amodm, nam, numb, nb);
606
607 mp_srcptr amodp = numa;
608 mp_size_t nap = na;
609 if (na > hn) {
610 /*
611 Z = B^hp + 1
612 amodp = a mod Z
613 */
614 tp[hn] = 0;
615 if (lmmp_sub_(tp, numa, hn, numa + hn, na - hn))
616 lmmp_inc(tp);
617 amodp = tp;
618 nap = hn + 1;
619 }
620 lmmp_mul_fermat_(tp, hn, amodp, nap, numb, nb);
621
622 mp_limb_t cy = lmmp_shr1add_nc_(dst, dst, tp, hn, tp[hn]);
623 cy <<= LIMB_BITS - 1;
624 dst[hn - 1] += cy;
625 if (dst[hn - 1] < cy)
626 lmmp_inc(dst);
627
628 if (na + nb == 2 * hn) {
629 cy = tp[hn] + lmmp_sub_n_(dst + hn, dst, tp, hn);
630 // cy==1 means [tp,hn+1]!=0, then [dst,hn]!=0
631 // cy==2 is impossible since [tp,hn+1] is normalized.
632 // so the following dec won't overflow.
633 lmmp_dec_1(dst, cy);
634 } else {
635 cy = lmmp_sub_n_(dst + hn, dst, tp, na + nb - hn);
636 cy = tp[hn] + lmmp_sub_nc_(tp + na + nb - hn, dst + na + nb - hn, tp + na + nb - hn, 2 * hn - (na + nb), cy);
637 cy = lmmp_sub_1_(dst, dst, na + nb, cy);
638 }
639 lmmp_free(tp);
640}

701void lmmp_mul_fft_cache_(mp_ptr dst, mp_srcptr numa, fft_cache* ctx) {
702 lmmp_param_assert(dst != NULL && numa != NULL);
703 lmmp_param_assert(ctx != NULL);
704 mp_srcptr amodm = numa;
705 if (ctx->na > ctx->hn) {
706 if (lmmp_add_(dst, numa, ctx->hn, numa + ctx->hn, ctx->na - ctx->hn))
707 lmmp_inc(dst);

709 }
710 lmmp_mul_mersenne_cache_(dst, amodm, &ctx->mersenne);
711
712 mp_srcptr amodp = numa;

714 ctx->tp[ctx->hn] = 0;

720
721 mp_limb_t cy = lmmp_shr1add_nc_(dst, dst, ctx->tp, ctx->hn, ctx->tp[ctx->hn]);
722 cy <<= LIMB_BITS - 1;
723 dst[ctx->hn - 1] += cy;
724 if (dst[ctx->hn - 1] < cy)
725 lmmp_inc(dst);
726
727 if (ctx->na + ctx->nb == 2 * ctx->hn) {
728 cy = ctx->tp[ctx->hn] + lmmp_sub_n_(dst + ctx->hn, dst, ctx->tp, ctx->hn);
729 lmmp_dec_1(dst, cy);
730 } else {
731 cy = lmmp_sub_n_(dst + ctx->hn, dst, ctx->tp, ctx->na + ctx->nb - ctx->hn);
732 cy = ctx->tp[ctx->hn] + lmmp_sub_nc_(ctx->tp + ctx->na + ctx->nb - ctx->hn, dst + ctx->na + ctx->nb - ctx->hn,
733 ctx->tp + ctx->na + ctx->nb - ctx->hn, 2 * ctx->hn - (ctx->na + ctx->nb),
734 cy);
735 cy = lmmp_sub_1_(dst, dst, ctx->na + ctx->nb, cy);
736 }

739void lmmp_mul_fft_unbalance_(
740 mp_ptr restrict dst,
741 mp_srcptr restrict numa,
742 mp_size_t na,
743 mp_srcptr restrict numb,
744 mp_size_t nb
745) {
746 lmmp_param_assert(na >= 3 * nb);
747 mp_ptr restrict ws = ALLOC_TYPE(nb, mp_limb_t);
748 mp_size_t sna = 3 * nb;
749 mp_size_t hn = lmmp_fft_next_size_((sna + nb + 1) >> 1);
750 sna = (hn << 1) - 1 - nb;
751 fft_cache ctx;
752 lmmp_mul_fft_cache_init_(dst, hn, numa, sna, numb, nb, &ctx);
753 dst += sna;
754 numa += sna;
755 na -= sna;
756 lmmp_copy(ws, dst, nb);
757 while (na >= sna) {
758 lmmp_mul_fft_cache_(dst, numa, &ctx);
759 if (lmmp_add_n_(dst, dst, ws, nb))
760 lmmp_inc(dst + nb);
761 dst += sna;
762 numa += sna;
763 na -= sna;
764 lmmp_copy(ws, dst, nb);
765 }
766 lmmp_fft_cache_free_(&ctx);
767 // remaining na < sna
768 if (na >= nb)
769 lmmp_mul_(dst, numa, na, numb, nb);
770 else if (na > 0)
771 lmmp_mul_(dst, numb, nb, numa, na);
772 else // na == 0
773 lmmp_zero(dst, nb);
774 if (lmmp_add_n_(dst, dst, ws, nb))
775 lmmp_inc(dst + nb);
776 lmmp_free(ws);
777}
