27static double bench_gcd_lehmer(mp_size_t n, const bench_options *o) {
28 mp_ptr a = alloc_limbs(n), b = alloc_limbs(n), d = alloc_limbs(n);
29 fill_random(a, n, 0x120bce34ba89eull + n);
30 fill_random(b, n, 0x2cab5423322191ull + n * 3);
31 a[n - 1] |= (mp_limb_t)1 << 63;
32 b[n - 1] >>= 1;
33 if (b[n - 1] == 0) b[n - 1] = 1;
34 a[0] |= 1;
35 b[0] |= 1; /* 契约：输入须归一化（顶 limb 非零）且 a>=b */
36 double us;
37 BENCH_MEASURE(us, o, bench_sink += (mp_limb_t)lmmp_gcd_lehmer_(d, a, n, b, n););
38 lmmp_free(a);
39 lmmp_free(b);
40 lmmp_free(d);
41 return us;
42}
