20mp_limb_t lmmp_div_basecase_(
21 mp_ptr restrict dstq,
22 mp_ptr restrict numa,
23 mp_size_t na,
24 mp_srcptr restrict numb,
25 mp_size_t nb,
26 mp_limb_t inv21
27) {
28 lmmp_param_assert(na >= nb);
29 lmmp_param_assert(nb >= 3);
30 lmmp_param_assert(numb[nb - 1] >= LIMB_B_2);
31 mp_size_t nq = na - nb;
32
33 numa += na;
34
35 mp_limb_t qh = lmmp_cmp_(numa - nb, numb, nb) >= 0;
36 if (qh) {
37 lmmp_sub_n_(numa - nb, numa - nb, numb, nb);
38 }
39
40 nb -= 2;
41 numa -= 2;
42
43 mp_limb_t d1 = numb[nb + 1], d0 = numb[nb + 0];
44
45 while (nq) {
46 mp_limb_t q;
47 --numa;
48 if (numa[2] == d1 && numa[1] == d0) {
49 q = LIMB_MAX;
50 lmmp_submul_1_(numa - nb, numb, nb + 2, q);
51 } else {
52 mp_limb_t cy, cy1;
53 q = lmmp_div_3_2_(numa, numb + nb, inv21);
54 cy = lmmp_submul_1_(numa - nb, numb, nb, q);
55 cy1 = numa[0] < cy;
56 numa[0] -= cy;
57 cy = numa[1] < cy1;
58 numa[1] -= cy1;
59 if (cy) {
60 lmmp_add_n_(numa - nb, numa - nb, numb, nb + 2);
61 --q;
62 }
63 }
64 dstq[--nq] = q;
65 }
66 return qh;
