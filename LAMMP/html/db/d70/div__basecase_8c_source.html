36 lmmp_param_assert(na >= nb);
37 lmmp_param_assert(nb >= 3);
38 lmmp_param_assert(dstq != NULL && numa != NULL && numb != NULL);
39 lmmp_param_assert(numb[nb - 1] >= LIMB_B_2);
40 mp_size_t nq = na - nb;
41
42 numa += na;
43
44 mp_limb_t qh = lmmp_cmp_(numa - nb, numb, nb) >= 0;
45 if (qh) {
46 lmmp_sub_n_(numa - nb, numa - nb, numb, nb);
47 }
48
49 nb -= 2;
50 numa -= 2;
51
52 mp_limb_t d1 = numb[nb + 1], d0 = numb[nb + 0];
53 mp_limb_t rh = numa[1], rl = numa[0];
54
55 while (nq) {
56 mp_limb_t q;
57 --numa;
58 if (rh == d1 && rl == d0) {
59 q = LIMB_MAX;
60 numa[1] = rl;
61 lmmp_submul_1_(numa - nb, numb, nb + 2, q);
62 rh = numa[1];
63 rl = numa[0];
64 } else {
65 mp_limb_t cy, cy1, r0, r1, n0 = numa[0];
66 _udiv_qr_3by2(q, r1, r0, rh, rl, n0, d1, d0, inv21);
67 cy = lmmp_submul_1_(numa - nb, numb, nb, q);
68 cy1 = r0 < cy;
69 r0 -= cy;
70 cy = r1 < cy1;
71 r1 -= cy1;
72 if (cy) {
73 numa[0] = r0;
74 numa[1] = r1;
75 lmmp_add_n_(numa - nb, numa - nb, numb, nb + 2);
76 rh = numa[1];
77 rl = numa[0];
78 --q;
79 } else {
80 rh = r1;
81 rl = r0;
82 }
83 }
84 dstq[--nq] = q;
85 }
86 numa[1] = rh;
87 numa[0] = rl;
88 return qh;
89}
