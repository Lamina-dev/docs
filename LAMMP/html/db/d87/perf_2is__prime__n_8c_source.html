25static double bench_is_prime_n(mp_size_t n, int strength, PrimePool kind, const bench_options *o) {
26 int cnt = pool_cnt(n);
27 mp_ptr a = pool_n(n, cnt, 0x51ce77ac01ull + (mp_limb_t)strength * 131 + n, kind);
28 size_t idx = 0;
29 double us;
30 BENCH_MEASURE(us, o, bench_sink += (mp_limb_t)lmmp_is_prime_n_(a + n * (idx++ % cnt), n, strength););
31 lmmp_free(a);
32 return us;
33}

45/*
46 MR 轮的指数为模数全长，单 op 成本随 n^3 增长（200L 已 ~1s），
47 默认阶梯到 200L 止；更大尺寸用 bench_is_prime_n(n, ...) 单点测量。
48*/
49static void run_is_prime_n(void) {
50 const mp_size_t SZ[] = {3, 8, 16, 32, 64, 200};
51 for (size_t i = 0; i < ARRAY_SIZE(SZ); ++i) {
52 mp_size_t n = SZ[i];
53 report_n("is_prime_n s4 coprime", n, bench_is_prime_n(n, 4, PRIME_POOL_COPRIME, &bench_default), 0);
54 }
55}
