35VectorExprListResult surface_gradient_at_point_checked(
36 const SurfaceSymbolic& surf,
37 const std::vector<std::shared_ptr<SymbolicExpr>>& point,
38 const std::string& operation)
39{
40 std::vector<std::shared_ptr<SymbolicExpr>> gradient;
41 gradient.reserve(surf.vars.size());
42 for (size_t i = 0; i < surf.vars.size(); ++i) {
43 auto derivative = differentiate_surface_checked(surf.F, surf.vars[i], operation);
44 if (!derivative) return VectorExprListResult::failure(derivative.error());
45 auto substituted = derivative.value();
46 for (size_t j = 0; j < surf.vars.size(); ++j) {
47 substituted = substituted->substitute(surf.vars[j], point[j]);
48 if (!substituted || !LMCAS::detail::node(substituted)) {
49 return VectorExprListResult::failure(
50 CasErrc::Inconclusive,
51 "surface derivative substitution is outside the supported domain",
52 operation);
53 }
54 }
55 auto simplified = simplify_checked(
56 substituted, operation,
57 "surface derivative substitution is outside the supported domain");
58 if (!simplified) return VectorExprListResult::failure(simplified.error());
59 gradient.push_back(std::move(simplified.value()));
60 }
61 return VectorExprListResult::success(std::move(gradient));
62}

90PlaneSymbolicResult scaled_tangent_plane(
91 const std::vector<double>& gradient_values,
92 double gradient_scale,
93 const std::vector<double>& point_values,
94 double point_scale,
95 const std::string& operation)
96{
97 double normalized_dot = 0.0;
98 for (size_t i = 0; i < point_values.size(); ++i) {
99 normalized_dot = std::fma(
100 gradient_values[i] / gradient_scale,
101 point_values[i] / point_scale,
102 normalized_dot);
103 }
104 const double plane_constant = normalized_dot * point_scale;
105 if (!std::isfinite(plane_constant)) {
106 return PlaneSymbolicResult::failure(
107 CasErrc::NumericFailure,
108 "tangent plane constant is outside the supported domain",
109 operation);
110 }
111
112 PlaneSymbolic plane;
113 plane.normal.reserve(gradient_values.size());
114 for (double component : gradient_values) {
115 plane.normal.push_back(
116 SymbolicExpr::number(component / gradient_scale));
117 }
118 plane.d = SymbolicExpr::number(plane_constant);
119 return PlaneSymbolicResult::success(std::move(plane));
120}

124VectorExprListResult surface_normal_checked(
125 const SurfaceSymbolic& surf,
126 const std::vector<std::shared_ptr<SymbolicExpr>>& point,
127 ComputationContext& context) {
128 const std::string operation = "surface_normal";
129 auto valid = validate_surface_point(surf, point, context, operation);
130 if (!valid) return VectorExprListResult::failure(valid.error());
131 auto step = context.consume_steps(surf.vars.size() * 4 + 4, operation);
132 if (!step) return VectorExprListResult::failure(step.error());
133
134 auto gradient = surface_gradient_at_point_checked(surf, point, operation);
135 if (!gradient) return gradient;
136 std::vector<double> gradient_values;
137 auto scaled = surface_gradient_scale_checked(
138 gradient.value(), context, operation, gradient_values,
139 "surface normal is undefined at a singular point");
140 if (!scaled) return VectorExprListResult::failure(scaled.error());
141
142 double scaled_norm_sq = 0.0;
143 for (double component : gradient_values) {
144 const double ratio = component / scaled.value();
145 scaled_norm_sq += ratio * ratio;
146 }
147 const double scaled_norm = std::sqrt(scaled_norm_sq);
148
149 std::vector<std::shared_ptr<SymbolicExpr>> result;
150 result.reserve(gradient_values.size());
151 for (double component : gradient_values) {
152 result.push_back(SymbolicExpr::number(
153 (component / scaled.value()) / scaled_norm));
154 }
155 return VectorExprListResult::success(std::move(result));
156}

166PlaneSymbolicResult tangent_plane_checked(
167 const SurfaceSymbolic& surf,
168 const std::vector<std::shared_ptr<SymbolicExpr>>& point,
169 ComputationContext& context) {
170 const std::string operation = "tangent_plane";
171 auto valid = validate_surface_point(surf, point, context, operation);
172 if (!valid) {
173 return PlaneSymbolicResult::failure(valid.error());
174 }
175 auto step = context.consume_steps(surf.vars.size() * 4 + 4, operation);
176 if (!step) {
177 return PlaneSymbolicResult::failure(step.error());
178 }
179
180 auto gradient = surface_gradient_at_point_checked(surf, point, operation);
181 if (!gradient) {
182 return PlaneSymbolicResult::failure(gradient.error());
183 }
184 std::vector<double> gradient_values;
185 auto scaled = surface_gradient_scale_checked(
186 gradient.value(), context, operation, gradient_values,
187 "tangent plane is undefined at a singular point");
188 if (!scaled) {
189 return PlaneSymbolicResult::failure(scaled.error());
190 }
191
192 std::vector<double> point_values;
193 auto point_scale = numeric_vector_scale_checked(
194 point, context, operation, &point_values);
195 if (!point_scale &&
196 (point_scale.error().code == CasErrc::Cancelled ||
197 point_scale.error().code == CasErrc::ResourceLimit)) {
198 return PlaneSymbolicResult::failure(point_scale.error());
199 }
200 if (point_scale && point_scale.value() != 0.0 &&
201 scaled.value() >
202 std::numeric_limits<double>::max() /
203 point_scale.value() /
204 static_cast<double>(point.size())) {
205 return scaled_tangent_plane(
206 gradient_values, scaled.value(), point_values,
207 point_scale.value(), operation);
208 }
209
210 auto dot = vector_dot_checked(gradient.value(), point, context);
211 if (!dot) {
212 return PlaneSymbolicResult::failure(dot.error());
213 }
214 auto d = simplify_checked(
215 dot.value(), operation,
216 "tangent plane constant is outside the supported domain");
217 if (!d) {
218 return PlaneSymbolicResult::failure(d.error());
219 }
220
221 PlaneSymbolic plane;
222 plane.normal = std::move(gradient.value());
223 plane.d = std::move(d.value());
224 return PlaneSymbolicResult::success(std::move(plane));
225}
