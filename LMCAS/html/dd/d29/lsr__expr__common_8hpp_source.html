55template <typename Factory>
56ExprResult make_binary_math_expr(const ExprPtr& lhs,
57 const ExprPtr& rhs,
58 ComputationContext& context,
59 const char* function_name,
60 Factory&& factory) {
61 auto step = context.consume_steps(1, kMathOperation);
62 if (!step) return ExprResult::failure(step.error());
63 if (!lhs || !lamina::detail::node(lhs) ||
64 !rhs || !lamina::detail::node(rhs)) {
65 return expression_failure(CasErrc::InvalidArgument,
66 std::string(function_name) +
67 " arguments cannot be null",
68 kMathOperation);
69 }
70 try {
71 auto result = std::forward<Factory>(factory)(lhs, rhs);
72 if (!result || !lamina::detail::node(result)) {
73 return expression_failure(CasErrc::InternalInvariant,
74 std::string(function_name) +
75 " expression construction failed",
76 kMathOperation);
77 }
78 return ExprResult::success(std::move(result));
79 } catch (const std::bad_alloc&) {
80 return expression_failure(CasErrc::ResourceLimit,
81 std::string(function_name) +
82 " expression allocation failed",
83 kMathOperation);
84 } catch (const std::exception& error) {
85 return expression_failure(CasErrc::InvalidArgument, error.what(),
86 kMathOperation);
