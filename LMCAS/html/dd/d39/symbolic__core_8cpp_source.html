57int SymbolicExpr::get_int() const {
58 if (!is_int()) {
59 throw std::runtime_error("Expression is not an integer");
60 }
61 auto number_node = std::dynamic_pointer_cast<const NumberNode>(impl_->root);
62 if (std::holds_alternative<BigInt>(number_node->value())) {
63 auto value = std::get<BigInt>(number_node->value()).try_to_int64();
64 if (!value || *value < std::numeric_limits<int>::min() ||
65 *value > std::numeric_limits<int>::max()) {
66 throw std::out_of_range("integer value does not fit in int");
67 }
68 return static_cast<int>(*value);
69 }
70 auto value = std::get<Rational>(number_node->value()).to_bigint().try_to_int64();
71 if (!value || *value < std::numeric_limits<int>::min() ||
72 *value > std::numeric_limits<int>::max()) {

154std::shared_ptr<SymbolicExpr> SymbolicExpr::simplify_trig() const {
155 ComputationContext context;
156 detail::RewriteBudget budget(context, context.limits().max_recursion_depth,
157 context.limits().max_ast_nodes, "simplify_trig");
158 auto res = detail::make_expression_ptr(detail::normalize_node(impl_->root, budget));
159 if (!LMCAS::detail::node(res)) return nullptr;
160
161 static const LMCAS::RewriteEngine engine = [] {
162 LMCAS::RewriteEngine configured;
163 using namespace LMCAS;
164 auto x_val = wildcard("x");
165 auto x = LMCAS::detail::make_expression_ptr(x_val);
166
167 auto sinx = SymbolicExpr::sin(x);
168 auto cosx = SymbolicExpr::cos(x);
169 auto n2 = SymbolicExpr::number(2);
170 auto sin2 = SymbolicExpr::power(sinx, n2);
171 auto cos2 = SymbolicExpr::power(cosx, n2);
172
173 auto pat1 = SymbolicExpr::add(sin2, cos2);
174 configured.add_rule(Rule(*pat1, *SymbolicExpr::number(1), {"x"}));
175
176 auto pat2 = SymbolicExpr::add(cos2, sin2);
177 configured.add_rule(Rule(*pat2, *SymbolicExpr::number(1), {"x"}));
178
179 auto two_x = SymbolicExpr::multiply(n2, x);
180 auto sin2x = SymbolicExpr::sin(two_x);
181 auto two_sin_cos = SymbolicExpr::multiply(n2,
182 SymbolicExpr::multiply(sinx, cosx));
183 configured.add_rule(Rule(*sin2x, *two_sin_cos, {"x"}));
184
185 auto cos2x = SymbolicExpr::cos(two_x);
186 auto cos2_sub_sin2 = SymbolicExpr::add(cos2,
187 SymbolicExpr::multiply(sin2, SymbolicExpr::number(-1)));
188 configured.add_rule(Rule(*cos2x, *cos2_sub_sin2, {"x"}));
189 return configured;
190 }();
191
192 auto rewritten = engine.apply_checked(*res, context);
193 if (!rewritten) return res;
194 auto simplified = std::move(rewritten.value());
195 auto result_ptr = LMCAS::detail::make_expression_ptr(simplified);
196 return detail::make_expression_ptr(detail::normalize_node(detail::node(result_ptr), budget));
197}

201SymbolicNodePtr normalize_node(const SymbolicNodePtr& node, RewriteBudget& budget,
202 const FactsQuery& facts, Domain domain) {
203 budget.consume();
204 if (!node) {
205 throw CasError{CasErrc::InvalidArgument, "simplify argument cannot be null",
206 "LMCAS.simplify"};
207 }
208 budget.measure(node);
209 NormalizationVisitor visitor(budget, facts, domain);
210 node->accept(visitor);
211 auto result = visitor.get_result();
212 if (!result) {
213 throw CasError{CasErrc::InternalInvariant, "simplify returned null",
214 "LMCAS.simplify"};
215 }
216 budget.measure(result);
217 return result;

247SymbolicNodePtr substitute_node(const SymbolicNodePtr& node, const std::string& variable,
248 const SymbolicNodePtr& replacement, RewriteBudget& budget,
249 const FactsQuery& facts, Domain domain) {
250 budget.consume();
251 if (!node || !replacement || variable.empty()) {
252 throw CasError{CasErrc::InvalidArgument, "invalid substitution argument",
253 "LMCAS.substitute"};
254 }
255 budget.measure(node);
256 budget.measure(replacement);
257 auto result = substitute_free(node, variable, replacement, &budget);
258 budget.measure(result);
