212std::shared_ptr<SymbolicExpr> SymbolicExpr::matrix(
213 const std::vector<std::vector<std::shared_ptr<SymbolicExpr>>>& elements) {
214 if (elements.empty()) {
215 throw std::invalid_argument("matrix requires at least one row");
216 }
217 const std::size_t column_count = elements.front().size();
218 if (column_count == 0) {
219 throw std::invalid_argument("matrix requires at least one column");
220 }
221
222 std::vector<std::vector<std::shared_ptr<const SymbolicNode>>> node_elements;
223 node_elements.reserve(elements.size());
224 for (const auto& row : elements) {
225 if (row.size() != column_count) {
226 throw std::invalid_argument(
227 "matrix rows must have the same number of columns");
228 }
229 std::vector<std::shared_ptr<const SymbolicNode>> node_row;
230 node_row.reserve(row.size());
231 for (const auto& element : row) {
232 if (!element || !element->impl_->root) {
233 throw std::invalid_argument("matrix elements cannot be null");
234 }
235 node_row.push_back(element->impl_->root);
236 }
237 node_elements.push_back(std::move(node_row));
238 }
239 return lamina::detail::make_expression_ptr(
240 lamina::detail::make_node<MatrixNode>(std::move(node_elements)));
241}

420std::shared_ptr<SymbolicExpr> SymbolicExpr::simplify_trig() const {
421 auto res = simplify();
422 if (!lamina::detail::node(res)) return nullptr;
423
424 static const lamina::RewriteEngine engine = [] {
425 lamina::RewriteEngine configured;
426 using namespace lamina;
427 auto x_val = wildcard("x");
428 auto x = lamina::detail::make_expression_ptr(x_val);
429
430 auto sinx = SymbolicExpr::sin(x);
431 auto cosx = SymbolicExpr::cos(x);
432 auto n2 = SymbolicExpr::number(2);
433 auto sin2 = SymbolicExpr::power(sinx, n2);
434 auto cos2 = SymbolicExpr::power(cosx, n2);
435
436 auto pat1 = SymbolicExpr::add(sin2, cos2);
437 configured.add_rule(Rule(*pat1, *SymbolicExpr::number(1), {"x"}));
438
439 auto pat2 = SymbolicExpr::add(cos2, sin2);
440 configured.add_rule(Rule(*pat2, *SymbolicExpr::number(1), {"x"}));
441
442 auto two_x = SymbolicExpr::multiply(n2, x);
443 auto sin2x = SymbolicExpr::sin(two_x);
444 auto two_sin_cos = SymbolicExpr::multiply(n2,
445 SymbolicExpr::multiply(sinx, cosx));
446 configured.add_rule(Rule(*sin2x, *two_sin_cos, {"x"}));
447
448 auto cos2x = SymbolicExpr::cos(two_x);
449 auto cos2_sub_sin2 = SymbolicExpr::add(cos2,
450 SymbolicExpr::multiply(sin2, SymbolicExpr::number(-1)));
451 configured.add_rule(Rule(*cos2x, *cos2_sub_sin2, {"x"}));
452 return configured;
453 }();
454
455 lamina::ComputationContext context;
456 auto rewritten = engine.apply_checked(*res, context);
457 if (!rewritten) return res;
458 auto simplified = std::move(rewritten.value());
459 auto result_ptr = lamina::detail::make_expression_ptr(simplified);
460 return result_ptr->simplify();
461}

479SymbolicExpr::Type SymbolicExpr::get_type() const {
480 if (!impl_->root) return Type::Number;
481
482 if (std::dynamic_pointer_cast<const NumberNode>(impl_->root)) return Type::Number;
483 if (std::dynamic_pointer_cast<const VariableNode>(impl_->root)) return Type::Variable;
484 if (std::dynamic_pointer_cast<const AddNode>(impl_->root)) return Type::Add;
485 if (std::dynamic_pointer_cast<const MultiplyNode>(impl_->root)) return Type::Multiply;
486 if (std::dynamic_pointer_cast<const PowerNode>(impl_->root)) return Type::Power;
487
488 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(impl_->root)) {
489 switch (func->type()) {
490 case FunctionNode::FuncType::Sin: return Type::Sin;
491 case FunctionNode::FuncType::Cos: return Type::Cos;
492 case FunctionNode::FuncType::Tan: return Type::Tan;
493 case FunctionNode::FuncType::Ln: return Type::Ln;
494 case FunctionNode::FuncType::Log: return Type::Log;
495 case FunctionNode::FuncType::Exp: return Type::Power;
496 case FunctionNode::FuncType::LambertW: return Type::Variable;
497 case FunctionNode::FuncType::Abs: return Type::Abs;
498 case FunctionNode::FuncType::Sqrt: return Type::Sqrt;
499 case FunctionNode::FuncType::Atan2: return Type::Atan2;
500 case FunctionNode::FuncType::ArcSin: return Type::ArcSin;
501 case FunctionNode::FuncType::ArcCos: return Type::ArcCos;
502 case FunctionNode::FuncType::ArcTan: return Type::ArcTan;
503 default: return Type::Variable;
504 }
505 }
506
507 if (auto mat = std::dynamic_pointer_cast<const MatrixNode>(impl_->root)) {
508 if (mat->rows() == 1 || mat->cols() == 1) return Type::Vector;
509 return Type::Matrix;
510 }
511
512 return Type::Number;
513}

515std::vector<std::shared_ptr<SymbolicExpr>> SymbolicExpr::get_operands() const {
516 std::vector<std::shared_ptr<SymbolicExpr>> ops;
517 if (!impl_->root) return ops;
518
519 if (auto add = std::dynamic_pointer_cast<const AddNode>(impl_->root)) {
520 for (const auto& op : add->operands()) ops.push_back(lamina::detail::make_expression_ptr(op));
521 } else if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(impl_->root)) {
522 for (const auto& op : mul->operands()) ops.push_back(lamina::detail::make_expression_ptr(op));
523 } else if (auto pow = std::dynamic_pointer_cast<const PowerNode>(impl_->root)) {
524 ops.push_back(lamina::detail::make_expression_ptr(pow->base()));
525 ops.push_back(lamina::detail::make_expression_ptr(pow->exponent()));
526 } else if (auto func = std::dynamic_pointer_cast<const FunctionNode>(impl_->root)) {
527 for (const auto& arg : func->arguments()) ops.push_back(lamina::detail::make_expression_ptr(arg));
528 }
529 return ops;
