212std::shared_ptr<SymbolicExpr> SymbolicExpr::matrix(
213 const std::vector<std::vector<std::shared_ptr<SymbolicExpr>>>& elements) {
214 if (elements.empty()) {
215 throw std::invalid_argument("matrix requires at least one row");
216 }
217 const std::size_t column_count = elements.front().size();
218 if (column_count == 0) {
219 throw std::invalid_argument("matrix requires at least one column");
220 }
221
222 std::vector<std::vector<std::shared_ptr<const SymbolicNode>>> node_elements;
223 node_elements.reserve(elements.size());
224 for (const auto& row : elements) {
225 if (row.size() != column_count) {
226 throw std::invalid_argument(
227 "matrix rows must have the same number of columns");
228 }
229 std::vector<std::shared_ptr<const SymbolicNode>> node_row;
230 node_row.reserve(row.size());
231 for (const auto& element : row) {
232 if (!element || !element->impl_->root) {
233 throw std::invalid_argument("matrix elements cannot be null");
234 }
235 node_row.push_back(element->impl_->root);
236 }
237 node_elements.push_back(std::move(node_row));
238 }
239 return lamina::detail::make_expression_ptr(
240 lamina::detail::make_node<MatrixNode>(std::move(node_elements)));
241}

269bool SymbolicExpr::is_int() const {
270 auto number_node = std::dynamic_pointer_cast<const NumberNode>(impl_->root);
271 if (!number_node) return false;
272 if (std::holds_alternative<BigInt>(number_node->value())) return true;
273 if (std::holds_alternative<Rational>(number_node->value())) {
274 return std::get<Rational>(number_node->value()).is_integer();
275 }
276
277 const lmmc_real_t value = std::get<lmmc_real_t>(number_node->value());
278 const lmmc_real_t rounded = std::round(value);
279 int equal = 0;
280 lmmc_double_nearly_equal_tol(value, rounded, 1e-12, 1e-12, &equal);
281 return equal != 0;
282}

336class VariablesVisitor : public lamina::detail::SymbolicVisitor {
337 std::set<std::string> bound_vars;
338
339 bool is_bound(const std::string& name) const {
340 return bound_vars.find(name) != bound_vars.end();
341 }
342
343public:
344 std::set<std::string> vars;
345
346 void visit(const NumberNode&) override {}
347 void visit(const VariableNode& node) override {
348 if (!is_bound(node.name())) {
349 vars.insert(node.name());
350 }
351 }
352 void visit(const AddNode& node) override {
353 for(auto& op : node.operands()) op->accept(*this);
354 }
355 void visit(const MultiplyNode& node) override {
356 for(auto& op : node.operands()) op->accept(*this);
357 }
358 void visit(const PowerNode& node) override {
359 node.base()->accept(*this);
360 node.exponent()->accept(*this);
361 }
362 void visit(const FunctionNode& node) override {
363 for(auto& arg : node.arguments()) arg->accept(*this);
364 }
365 void visit(const UninterpretedFunctionNode& node) override {
366 for (const auto& argument : node.arguments()) argument->accept(*this);
367 }
368 void visit(const MatrixNode&) override {
369
370 }
371 void visit(const RelationalNode& node) override {
372 node.left()->accept(*this);
373 node.right()->accept(*this);
374 }
375 void visit(const LogicalNode& node) override {
376 node.left()->accept(*this);
377 if (node.right()) node.right()->accept(*this);
378 }
379 void visit(const PiecewiseNode& node) override {
380 for (const auto& branch : node.branches()) {
381 branch.expression->accept(*this);
382 branch.condition->accept(*this);
383 }
384 if (node.default_expr()) node.default_expr()->accept(*this);
385 }
386 void visit(const SummationNode& node) override {
387 node.lower_bound()->accept(*this);
388 node.upper_bound()->accept(*this);
389 bool inserted = bound_vars.insert(node.index_var()).second;
390 node.body()->accept(*this);
391 if (inserted) bound_vars.erase(node.index_var());
392 }
393 void visit(const ProductNode& node) override {
394 node.lower_bound()->accept(*this);
395 node.upper_bound()->accept(*this);
396 bool inserted = bound_vars.insert(node.index_var()).second;
397 node.body()->accept(*this);
398 if (inserted) bound_vars.erase(node.index_var());
399 }
400 void visit(const TransformNode& node) override {
401 bool inserted = bound_vars.insert(node.source_var()).second;
402 node.body()->accept(*this);
403 if (inserted) bound_vars.erase(node.source_var());
404 if (!is_bound(node.target_var())) vars.insert(node.target_var());
405 }
406 void visit(const QuantifierNode& node) override {
407 node.domain()->accept(*this);
408 bool inserted = bound_vars.insert(node.bound_var()).second;
409 node.predicate()->accept(*this);
410 if (inserted) bound_vars.erase(node.bound_var());
411 }
412 void visit(const SetBuilderNode& node) override {
413 node.domain()->accept(*this);
414 bool inserted = bound_vars.insert(node.element_var()).second;
415 node.predicate()->accept(*this);
416 if (inserted) bound_vars.erase(node.element_var());
417 }
418 void visit(const ComplexNode& node) override {
419 node.real()->accept(*this);
420 node.imag()->accept(*this);
421 }
422 void visit(const FiniteSetNode& node) override {
423 for (const auto& element : node.elements()) element->accept(*this);
424 }
425 void visit(const IntervalNode& node) override {
426 node.lower()->accept(*this);
427 node.upper()->accept(*this);
428 }
429 void visit(const MembershipNode& node) override {
430 node.element()->accept(*this);
431 node.set()->accept(*this);
432 }
433 void visit(const QuantityNode& node) override { node.value()->accept(*this); }
434};

436class SubstituteVisitor : public lamina::detail::SymbolicVisitor {
437 std::string var_name;
438 std::shared_ptr<const SymbolicNode> new_val;
439public:
440 std::shared_ptr<const SymbolicNode> result;
441
442 SubstituteVisitor(std::string v, std::shared_ptr<const SymbolicNode> val)
443 : var_name(std::move(v)), new_val(std::move(val)) {}
444
445 std::shared_ptr<const SymbolicNode> get_result() const { return result; }
446
447 void visit(const NumberNode& node) override {
448 result = node.clone();
449 }
450
451 void visit(const VariableNode& node) override {
452 if (node.name() == var_name) {
453 result = new_val->clone();
454 } else {
455 result = node.clone();
456 }
457 }
458
459 void visit(const AddNode& node) override {
460 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
461 for (const auto& op : node.operands()) {
462 op->accept(*this);
463 new_ops.push_back(result);
464 }
465 result = lamina::detail::make_node<AddNode>(new_ops);
466 }
467
468 void visit(const MultiplyNode& node) override {
469 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
470 for (const auto& op : node.operands()) {
471 op->accept(*this);
472 new_ops.push_back(result);
473 }
474 result = lamina::detail::make_node<MultiplyNode>(new_ops);
475 }
476
477 void visit(const PowerNode& node) override {
478 node.base()->accept(*this);
479 auto new_base = result;
480 node.exponent()->accept(*this);
481 auto new_exp = result;
482 result = lamina::detail::make_node<PowerNode>(new_base, new_exp);
483 }
484
485 void visit(const FunctionNode& node) override {
486 std::vector<std::shared_ptr<const SymbolicNode>> new_args;
487 for (const auto& arg : node.arguments()) {
488 arg->accept(*this);
489 new_args.push_back(result);
490 }
491 result = lamina::detail::make_node<FunctionNode>(node.type(), new_args);
492 }
493
494 void visit(const UninterpretedFunctionNode& node) override {
495 std::vector<std::shared_ptr<const SymbolicNode>> arguments;
496 arguments.reserve(node.arguments().size());
497 for (const auto& argument : node.arguments()) {
498 argument->accept(*this);
499 arguments.push_back(result);
500 }
501 result = lamina::detail::make_node<UninterpretedFunctionNode>(
502 node.name(), std::move(arguments));
503 }
504
505 void visit(const MatrixNode& node) override {
506
507 if (std::holds_alternative<MatrixNode::DenseStorage>(node.storage())) {
508 const auto& dense = std::get<MatrixNode::DenseStorage>(node.storage());
509 MatrixNode::DenseStorage new_dense;
510 for(const auto& e : dense) {
511 e->accept(*this);
512 new_dense.push_back(result);
513 }
514 result = lamina::detail::make_node<MatrixNode>(node.rows(), node.cols(), new_dense);
515 } else {
516 const auto& sparse = std::get<MatrixNode::SparseStorage>(node.storage());
517 MatrixNode::SparseStorage new_sparse;
518 for(const auto& [idx, val] : sparse) {
519 val->accept(*this);
520 new_sparse[idx] = result;
521 }
522 result = lamina::detail::make_node<MatrixNode>(node.rows(), node.cols(), new_sparse);
523 }
524 }
525
526 void visit(const RelationalNode& node) override {
527 node.left()->accept(*this);
528 auto new_left = result;
529 node.right()->accept(*this);
530 auto new_right = result;
531 result = lamina::detail::make_node<RelationalNode>(new_left, new_right, node.op());
532 }
533
534 void visit(const LogicalNode& node) override {
535 node.left()->accept(*this);
536 auto new_left = result;
537 std::shared_ptr<const SymbolicNode> new_right = nullptr;
538 if (node.right()) {
539 node.right()->accept(*this);
540 new_right = result;
541 }
542 result = lamina::detail::make_node<LogicalNode>(new_left, new_right, node.op());
543 }
544
545 void visit(const SummationNode& node) override {
546 /// 指标变量是绑定变量：仅当替换变量不是指标时才替换通项。
547 std::shared_ptr<const SymbolicNode> new_body;
548 if (node.index_var() == var_name) {
549 new_body = node.body()->clone();
550 } else {
551 node.body()->accept(*this);
552 new_body = result;
553 }
554 node.lower_bound()->accept(*this);
555 auto new_lo = result;
556 node.upper_bound()->accept(*this);
557 auto new_hi = result;
558 result = lamina::detail::make_node<SummationNode>(new_body, node.index_var(), new_lo, new_hi);
559 }
560
561 void visit(const ProductNode& node) override {
562 std::shared_ptr<const SymbolicNode> new_body;
563 if (node.index_var() == var_name) {
564 new_body = node.body()->clone();
565 } else {
566 node.body()->accept(*this);
567 new_body = result;
568 }
569 node.lower_bound()->accept(*this);
570 auto new_lo = result;
571 node.upper_bound()->accept(*this);
572 auto new_hi = result;
573 result = lamina::detail::make_node<ProductNode>(new_body, node.index_var(), new_lo, new_hi);
574 }
575
576 void visit(const PiecewiseNode& node) override {
577 std::vector<PiecewiseNode::Branch> new_branches;
578 for (const auto& b : node.branches()) {
579 PiecewiseNode::Branch nb;
580 b.expression->accept(*this);
581 nb.expression = result;
582 b.condition->accept(*this);
583 nb.condition = result;
584 new_branches.push_back(nb);
585 }
586 std::shared_ptr<const SymbolicNode> new_def = nullptr;
587 if (node.default_expr()) {
588 node.default_expr()->accept(*this);
589 new_def = result;
590 }
591 result = lamina::detail::make_node<PiecewiseNode>(std::move(new_branches), new_def);
592 }
593
594 void visit(const ComplexNode& node) override {
595 node.real()->accept(*this);
596 auto new_real = result;
597 node.imag()->accept(*this);
598 auto new_imag = result;
599 result = lamina::detail::make_node<ComplexNode>(new_real, new_imag);
600 }
601
602 void visit(const TransformNode& node) override {
603 std::shared_ptr<const SymbolicNode> new_body;
604 if (node.source_var() == var_name) {
605 new_body = node.body()->clone();
606 } else {
607 node.body()->accept(*this);
608 new_body = result;
609 }
610 result = lamina::detail::make_node<TransformNode>(
611 node.transform_type(), new_body, node.source_var(), node.target_var());
612 }
613
614 void visit(const QuantifierNode& node) override {
615 node.domain()->accept(*this);
616 auto new_domain = result;
617
618 std::shared_ptr<const SymbolicNode> new_predicate;
619 if (node.bound_var() == var_name) {
620 new_predicate = node.predicate()->clone();
621 } else {
622 node.predicate()->accept(*this);
623 new_predicate = result;

636 new_predicate = node.predicate()->clone();

638 node.predicate()->accept(*this);
639 new_predicate = result;
640 }
641
642 result = lamina::detail::make_node<SetBuilderNode>(
643 node.element_var(), new_domain, new_predicate);
644 }

646 void visit(const FiniteSetNode& node) override {

648 elements.reserve(node.elements().size());
649 for (const auto& element : node.elements()) {
650 element->accept(*this);
651 elements.push_back(result);
652 }
653 result = lamina::detail::make_node<FiniteSetNode>(std::move(elements));
654 }

657 node.lower()->accept(*this);

659 node.upper()->accept(*this);
660 result = lamina::detail::make_node<IntervalNode>(
661 lower, result, node.lower_closed(), node.upper_closed());

665 node.element()->accept(*this);

671 void visit(const QuantityNode& node) override {
672 node.value()->accept(*this);

674 result, node.dimension(), node.scale_to_base(), node.display_unit());

761std::shared_ptr<SymbolicExpr> SymbolicExpr::simplify_trig() const {
762 auto res = simplify();
763 if (!lamina::detail::node(res)) return nullptr;
764
765 static const lamina::RewriteEngine engine = [] {
766 lamina::RewriteEngine configured;
767 using namespace lamina;
768 auto x_val = wildcard("x");
769 auto x = lamina::detail::make_expression_ptr(x_val);
770
771 auto sinx = SymbolicExpr::sin(x);
772 auto cosx = SymbolicExpr::cos(x);
773 auto n2 = SymbolicExpr::number(2);
774 auto sin2 = SymbolicExpr::power(sinx, n2);
775 auto cos2 = SymbolicExpr::power(cosx, n2);
776
777 auto pat1 = SymbolicExpr::add(sin2, cos2);
778 configured.add_rule(Rule(*pat1, *SymbolicExpr::number(1), {"x"}));
779
780 auto pat2 = SymbolicExpr::add(cos2, sin2);
781 configured.add_rule(Rule(*pat2, *SymbolicExpr::number(1), {"x"}));
782
783 auto two_x = SymbolicExpr::multiply(n2, x);
784 auto sin2x = SymbolicExpr::sin(two_x);
785 auto two_sin_cos = SymbolicExpr::multiply(n2,
786 SymbolicExpr::multiply(sinx, cosx));
787 configured.add_rule(Rule(*sin2x, *two_sin_cos, {"x"}));
788
789 auto cos2x = SymbolicExpr::cos(two_x);
790 auto cos2_sub_sin2 = SymbolicExpr::add(cos2,
791 SymbolicExpr::multiply(sin2, SymbolicExpr::number(-1)));
792 configured.add_rule(Rule(*cos2x, *cos2_sub_sin2, {"x"}));
793 return configured;
794 }();
795
796 lamina::ComputationContext context;
797 auto rewritten = engine.apply_checked(*res, context);
798 if (!rewritten) return res;
799 auto simplified = std::move(rewritten.value());
800 auto result_ptr = lamina::detail::make_expression_ptr(simplified);
801 return result_ptr->simplify();
802}

820SymbolicExpr::Type SymbolicExpr::get_type() const {
821 if (!impl_->root) return Type::Number;
822
823 if (std::dynamic_pointer_cast<const NumberNode>(impl_->root)) return Type::Number;
824 if (std::dynamic_pointer_cast<const VariableNode>(impl_->root)) return Type::Variable;
825 if (std::dynamic_pointer_cast<const AddNode>(impl_->root)) return Type::Add;
826 if (std::dynamic_pointer_cast<const MultiplyNode>(impl_->root)) return Type::Multiply;
827 if (std::dynamic_pointer_cast<const PowerNode>(impl_->root)) return Type::Power;
828
829 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(impl_->root)) {
830 switch (func->type()) {
831 case FunctionNode::FuncType::Sin: return Type::Sin;
832 case FunctionNode::FuncType::Cos: return Type::Cos;
833 case FunctionNode::FuncType::Tan: return Type::Tan;
834 case FunctionNode::FuncType::Ln: return Type::Ln;
835 case FunctionNode::FuncType::Log: return Type::Log;
836 case FunctionNode::FuncType::Exp: return Type::Power;
837 case FunctionNode::FuncType::LambertW: return Type::Variable;
838 case FunctionNode::FuncType::Abs: return Type::Abs;
839 case FunctionNode::FuncType::Sqrt: return Type::Sqrt;
840 case FunctionNode::FuncType::Atan2: return Type::Atan2;
841 case FunctionNode::FuncType::ArcSin: return Type::ArcSin;
842 case FunctionNode::FuncType::ArcCos: return Type::ArcCos;
843 case FunctionNode::FuncType::ArcTan: return Type::ArcTan;
844 default: return Type::Variable;
845 }
846 }
847
848 if (auto mat = std::dynamic_pointer_cast<const MatrixNode>(impl_->root)) {
849 if (mat->rows() == 1 || mat->cols() == 1) return Type::Vector;
850 return Type::Matrix;
851 }
852
853 return Type::Number;
