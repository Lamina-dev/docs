21Result<double> evaluate_root_function(
22 const std::shared_ptr<SymbolicExpr>& expression,
23 const std::string& variable,
24 lmmc_real_t value,
25 ComputationContext& context,
26 const char* operation) {
27 if (!expression) {
28 return Result<double>::failure(
29 CasErrc::InvalidArgument,
30 "root function expression cannot be null",
31 operation);
32 }
33 if (variable.empty()) {
34 return Result<double>::failure(
35 CasErrc::InvalidArgument,
36 "root variable cannot be empty",
37 operation);
38 }
39 auto evaluated = evaluate_numeric(
40 *expression, NumericBindings{{variable, static_cast<double>(value)}},
41 context);
42 if (!evaluated) return Result<double>::failure(evaluated.error());
43 if (!evaluated.value().is_finite() ||
44 !std::isfinite(evaluated.value().value)) {
45 return Result<double>::failure(
46 CasErrc::NumericFailure,
47 "root function evaluation is not finite",
48 operation);
49 }
50 return Result<double>::success(evaluated.value().value);
51}

84double evaluate_root_callback(
85 const std::shared_ptr<SymbolicExpr>& expression,
86 double x,
87 RootCallbacks& callbacks) {
88 if (callbacks.error) return std::numeric_limits<double>::quiet_NaN();
89 if (callbacks.enforce_bracket &&
90 (x < callbacks.bracket_lower || x > callbacks.bracket_upper)) {
91 callbacks.left_bracket = true;
92 return std::numeric_limits<double>::quiet_NaN();
93 }
94 auto value = evaluate_root_function(
95 expression, *callbacks.variable, x, *callbacks.context,
96 callbacks.operation);
97 if (!value) {
98 callbacks.error = value.error();
99 return std::numeric_limits<double>::quiet_NaN();
100 }
101 return value.value();
102}

145NumericRootResult run_bisection_backend(
146 const std::shared_ptr<SymbolicExpr>& function,
147 const std::string& variable,
148 double lower,
149 double upper,
150 ComputationContext& context,
151 const SolveOptions& options,
152 const char* operation) {
153 if (lower == upper) {
154 auto value = evaluate_root_function(
155 function, variable, lower, context, operation);
156 if (!value) return NumericRootResult::failure(value.error());
157 if (std::abs(value.value()) <= options.tolerance) {
158 return NumericRootResult::success(
159 NumericRoot{lower, std::abs(value.value()), 0});
160 }
161 return NumericRootResult::success(std::nullopt);
162 }
163
164 detail::ensure_lmmc_lifecycle();
165 RootCallbacks callbacks{
166 &function, nullptr, &variable, &context, operation,
167 false, lower, upper, false, std::nullopt};
168 const std::size_t max_iterations =
169 static_cast<std::size_t>(options.max_newton_iterations) * 3;
170 auto config = root_config(options, max_iterations);
171 lmmc_nonlinear_result_t backend{};
172 const auto status = lmmc_bisection_solve(
173 root_function_callback, &callbacks, lower, upper, &config, &backend);
174 if (callbacks.error) return NumericRootResult::failure(*callbacks.error);
175 if (status == LMMC_STATUS_INVALID_ARGUMENT &&
176 backend.failure_reason == LMMC_NONLINEAR_FAILURE_INVALID_BRACKET) {
177 return NumericRootResult::success(std::nullopt);
178 }
179 return backend_root_result(backend, options.tolerance * 100.0);
180}

254NumericRootResult newton_raphson_checked(
255 const std::shared_ptr<SymbolicExpr>& function,
256 const std::shared_ptr<SymbolicExpr>& derivative,
257 const std::string& variable,
258 lmmc_real_t initial,
259 lmmc_real_t bracket_lower,
260 lmmc_real_t bracket_upper,
261 ComputationContext& context,
262 const SolveOptions& options) {
263 auto valid = invalid_root_options(options, kNewtonOperation);
264 if (!valid) return valid;
265 if (!function || !derivative || variable.empty() ||
266 !std::isfinite(initial) || !std::isfinite(bracket_lower) ||
267 !std::isfinite(bracket_upper) || bracket_lower > bracket_upper ||
268 initial < bracket_lower || initial > bracket_upper) {
269 return NumericRootResult::failure(
270 CasErrc::InvalidArgument,
271 "Newton function, derivative, variable, bracket, and initial value are invalid",
272 kNewtonOperation);
273 }
274 auto newton = run_newton_backend(
275 function, derivative, variable, initial, context, options, true,
276 bracket_lower, bracket_upper);
277 if (!newton || newton.value()) return newton;
278 return run_bisection_backend(
279 function, variable, bracket_lower, bracket_upper, context, options,
280 kBisectionOperation);
281}

339NumericRootsResult solve_numeric_checked(
340 const std::shared_ptr<SymbolicExpr>& expr,
341 const std::string& var,
342 ComputationContext& context,
343 const SolveOptions& opts)
344{
345 constexpr const char* operation = "solve_numeric";
346 std::vector<NumericRoot> results;
347 auto options = invalid_root_options(opts, operation);
348 if (!options) return NumericRootsResult::failure(options.error());
349 if (!expr) {
350 return NumericRootsResult::failure(CasErrc::InvalidArgument,
351 "expression cannot be null",
352 operation);
353 }
354 if (var.empty()) {
355 return NumericRootsResult::failure(CasErrc::InvalidArgument,
356 "solve variable cannot be empty",
357 operation);
358 }
359 if (opts.max_roots < -1) {
360 return NumericRootsResult::failure(CasErrc::InvalidArgument,
361 "maximum root count must be -1 or non-negative",
362 operation);
363 }
364 if (opts.max_roots == 0) {
365 return NumericRootsResult::success(std::move(results));
366 }
367
368 auto initial_step = context.consume_steps(1, operation);
369 if (!initial_step) return NumericRootsResult::failure(initial_step.error());
370
371 auto recognized_poly = recognize_rational_polynomial(*expr, var, context);
372 if (!recognized_poly) return NumericRootsResult::failure(recognized_poly.error());
373
374 if (recognized_poly.value() && !recognized_poly.value()->is_zero() &&
375 recognized_poly.value()->degree() >= 1) {
376 const Polynomial<Rational>& poly = *recognized_poly.value();
377 auto df_expr = expr->differentiate(var);
378 auto isolated = isolate_real_roots_checked(poly, context);
379 if (!isolated) return NumericRootsResult::failure(isolated.error());
380 for (const auto& [lo_rat, hi_rat] : isolated.value()) {
381 if (opts.max_roots > 0 &&
382 static_cast<int>(results.size()) >= opts.max_roots) {
383 break;
384 }
385
386 auto interval_step = context.consume_steps(1, operation);
387 if (!interval_step) return NumericRootsResult::failure(interval_step.error());
388
389 lmmc_real_t lo = lo_rat.to_double();
390 lmmc_real_t hi = hi_rat.to_double();
391 lmmc_real_t x0 = (lo + hi) * 0.5;
392 auto root_result = newton_raphson_checked(
393 expr, df_expr, var, x0, lo, hi, context, opts);
394 if (!root_result) return NumericRootsResult::failure(root_result.error());
395 if (!root_result.value()) continue;
396
397 NumericRoot root = *root_result.value();
398 auto verified = evaluate_root_function(
399 expr, var, root.value, context, operation);
400 if (!verified) return NumericRootsResult::failure(verified.error());
401 root.residual = std::abs(verified.value());
402 if (root.residual <= opts.tolerance * 100.0) {
403 results.push_back(root);
404 }
405 }
406 } else {
407 lmmc_real_t x0 = opts.has_initial_guess ? opts.initial_guess : 0.0;
408 auto df_expr = expr->differentiate(var);
409 auto root_result = newton_raphson_checked(
410 expr, df_expr, var, x0, context, opts);
411 if (!root_result) return NumericRootsResult::failure(root_result.error());
412 if (root_result.value()) {
413 NumericRoot root = *root_result.value();
414 auto verified = evaluate_root_function(
415 expr, var, root.value, context, operation);
416 if (!verified) return NumericRootsResult::failure(verified.error());
417 root.residual = std::abs(verified.value());
418 if (root.residual <= opts.tolerance) {
419 results.push_back(root);
420 }
421 }
422 }
423
424 return NumericRootsResult::success(std::move(results));
425}
