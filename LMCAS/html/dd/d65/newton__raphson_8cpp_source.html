58static int count_sign_changes(const std::vector<Rational>& values) {
59 int changes = 0;
60 int last_sign = 0;
61
62 for (const auto& v : values) {
63 if (v.is_zero()) continue;
64
65 int current_sign = (v > Rational(0)) ? 1 : -1;
66 if (last_sign != 0 && current_sign != last_sign) {
67 ++changes;
68 }
69 last_sign = current_sign;
70 }
71 return changes;
72}

98[[maybe_unused]] static int sturm_sign_changes_at_neg_inf(
99 const std::vector<Polynomial<Rational>>& sturm)
100{
101 std::vector<Rational> signs;
102 signs.reserve(sturm.size());
103 for (const auto& p : sturm) {
104 if (p.is_zero()) continue;
105 int deg = p.degree();
106 Rational lc = p.lead_coeff();
107
108 if (deg % 2 == 1) {

110 }
111 signs.push_back(lc);
112 }

116static Rational cauchy_bound(const Polynomial<Rational>& poly) {
117 if (poly.degree() <= 0) return Rational(1);
118
119 Rational lc = poly.lead_coeff();
120 Rational max_ratio(0);
121
122 for (int i = 0; i < poly.degree(); ++i) {
123 Rational coeff = poly.coeffs[i];
124 if (coeff.is_zero()) continue;
125 Rational ratio = coeff.abs() / lc.abs();
126 if (ratio > max_ratio) {
127 max_ratio = ratio;
128 }
129 }
130
131 return Rational(1) + max_ratio;
132}

144std::vector<std::pair<Rational, Rational>> isolate_real_roots(
145 const Polynomial<Rational>& poly)
146{
147 std::vector<std::pair<Rational, Rational>> result;
148
149 if (poly.is_zero() || poly.degree() <= 0) {
150 return result;
151 }
152
153 Polynomial<Rational> sqfree = poly.square_free_part();
154 if (sqfree.is_zero() || sqfree.degree() <= 0) {
155 return result;
156 }
157
158 std::vector<Polynomial<Rational>> sturm;
159 sturm.push_back(sqfree);
160 sturm.push_back(sqfree.differentiate());
161
162 while (true) {
163 size_t n = sturm.size();
164 const auto& prev2 = sturm[n - 2];
165 const auto& prev1 = sturm[n - 1];
166
167 if (prev1.is_zero()) break;
168
169 auto [quotient, remainder] = prev2.div_mod(prev1);
170
171 if (remainder.is_zero()) break;
172
173 Polynomial<Rational> neg_rem(remainder.variable_name);
174 neg_rem.coeffs.resize(remainder.coeffs.size());
175 for (size_t i = 0; i < remainder.coeffs.size(); ++i) {
176 neg_rem.coeffs[i] = -remainder.coeffs[i];
177 }
178 neg_rem.trim();
179
180 sturm.push_back(neg_rem);
181 }
182
183 Rational bound = cauchy_bound(sqfree);
184 Rational lo = -bound;
185 Rational hi = bound;
186
187 int total_roots = roots_in_interval(sturm, lo, hi);
188 if (total_roots <= 0) {
189 return result;
190 }
191
192 struct Interval {
193 Rational lo, hi;
194 int root_count;
195 };
196
197 std::vector<Interval> work_queue;
198 work_queue.push_back({lo, hi, total_roots});
199
200 while (!work_queue.empty()) {
201 Interval current = work_queue.back();
202 work_queue.pop_back();
203
204 if (current.root_count == 0) {
205 continue;
206 }
207
208 if (current.root_count == 1) {
209 result.push_back({current.lo, current.hi});
210 continue;
211 }
212
213 Rational mid = (current.lo + current.hi) / Rational(2);
214
215 int left_count = roots_in_interval(sturm, current.lo, mid);
216 int right_count = roots_in_interval(sturm, mid, current.hi);
217
218 if (left_count + right_count < current.root_count) {
219
220 result.push_back({mid, mid});
221 }
222
223 if (right_count > 0) {
224 work_queue.push_back({mid, current.hi, right_count});
225 }
226 if (left_count > 0) {
227 work_queue.push_back({current.lo, mid, left_count});
228 }
229 }
230
231 std::sort(result.begin(), result.end(),
232 [](const std::pair<Rational, Rational>& a,
233 const std::pair<Rational, Rational>& b) {
234 return a.first < b.first;
235 });
236
237 return result;
238}

240NumericRootResult bisection_checked(
241 const std::shared_ptr<SymbolicExpr>& f,
242 const std::string& var,
243 lmmc_real_t lo,
244 lmmc_real_t hi,
245 ComputationContext& context,
246 const SolveOptions& opts)
247{
248 auto options = invalid_root_options(opts, kBisectionOperation);
249 if (!options) return options;
250 if (!std::isfinite(lo) || !std::isfinite(hi) || lo > hi) {
251 return NumericRootResult::failure(CasErrc::InvalidArgument,
252 "bisection interval must be finite and ordered",
253 kBisectionOperation);
254 }
255
256 auto f_lo_result = evaluate_root_function(f, var, lo, context, kBisectionOperation);
257 if (!f_lo_result) return NumericRootResult::failure(f_lo_result.error());
258 auto f_hi_result = evaluate_root_function(f, var, hi, context, kBisectionOperation);
259 if (!f_hi_result) return NumericRootResult::failure(f_hi_result.error());
260 lmmc_real_t f_lo = f_lo_result.value();
261 lmmc_real_t f_hi = f_hi_result.value();
262
263 if (std::abs(f_lo) < opts.tolerance) {
264 return NumericRootResult::success(NumericRoot{lo, std::abs(f_lo), 0});
265 }
266 if (std::abs(f_hi) < opts.tolerance) {
267 return NumericRootResult::success(NumericRoot{hi, std::abs(f_hi), 0});
268 }
269
270 if (f_lo * f_hi > 0) {
271 return NumericRootResult::success(std::nullopt);
272 }
273
274 lmmc_real_t best_x = std::abs(f_lo) <= std::abs(f_hi) ? lo : hi;
275 lmmc_real_t best_residual = std::min(std::abs(f_lo), std::abs(f_hi));
276 int best_iteration = 0;
277 int max_iter = opts.max_newton_iterations * 3;
278 for (int i = 1; i <= max_iter; ++i) {
279 auto step = context.consume_steps(1, kBisectionOperation);
280 if (!step) return NumericRootResult::failure(step.error());
281 lmmc_real_t mid = (lo + hi) * 0.5;
282 auto f_mid_result = evaluate_root_function(f, var, mid, context, kBisectionOperation);
283 if (!f_mid_result) return NumericRootResult::failure(f_mid_result.error());
284 lmmc_real_t f_mid = f_mid_result.value();
285 lmmc_real_t residual = std::abs(f_mid);
286 if (residual < best_residual) {
287 best_x = mid;
288 best_residual = residual;
289 best_iteration = i;
290 }
291
292 if (residual < opts.tolerance) {
293 return NumericRootResult::success(NumericRoot{mid, residual, i});
294 }
295
296 if (mid == lo || mid == hi) {
297 break;
298 }
299
300 if (f_lo * f_mid < 0) {
301 hi = mid;
302 f_hi = f_mid;
303 } else {
304 lo = mid;
305 f_lo = f_mid;
306 }
307 }
308
309 if (best_residual < opts.tolerance * 100.0) {
310 return NumericRootResult::success(
311 NumericRoot{best_x, best_residual, best_iteration});
312 }
313 return NumericRootResult::success(std::nullopt);
314}

339NumericRootResult newton_raphson_checked(
340 const std::shared_ptr<SymbolicExpr>& f,
341 const std::shared_ptr<SymbolicExpr>& df,
342 const std::string& var,
343 lmmc_real_t x0,
344 lmmc_real_t bracket_lo,
345 lmmc_real_t bracket_hi,
346 ComputationContext& context,
347 const SolveOptions& opts)
348{
349 auto options = invalid_root_options(opts, kNewtonOperation);
350 if (!options) return options;
351 if (!f || !df) {
352 return NumericRootResult::failure(CasErrc::InvalidArgument,
353 "function and derivative cannot be null",
354 kNewtonOperation);
355 }
356 if (!std::isfinite(x0) || !std::isfinite(bracket_lo) ||
357 !std::isfinite(bracket_hi) || bracket_lo > bracket_hi ||
358 x0 < bracket_lo || x0 > bracket_hi) {
359 return NumericRootResult::failure(CasErrc::InvalidArgument,
360 "Newton bracket and initial value must be finite and ordered",
361 kNewtonOperation);
362 }
363 lmmc_real_t x = x0;
364 for (int i = 1; i <= opts.max_newton_iterations; ++i) {
365 auto step = context.consume_steps(1, kNewtonOperation);
366 if (!step) return NumericRootResult::failure(step.error());
367
368 auto fx_result = evaluate_root_function(f, var, x, context, kNewtonOperation);
369 if (!fx_result) return NumericRootResult::failure(fx_result.error());
370 auto dfx_result = evaluate_root_function(df, var, x, context, kNewtonOperation);
371 if (!dfx_result) return NumericRootResult::failure(dfx_result.error());
372 lmmc_real_t fx = fx_result.value();
373 lmmc_real_t dfx = dfx_result.value();
374
375 if (std::abs(fx) < opts.tolerance) {
376 return NumericRootResult::success(NumericRoot{x, std::abs(fx), i});
377 }
378
379 if (std::abs(dfx) < 1e-15) {
380 return bisection_checked(f, var, bracket_lo, bracket_hi, context, opts);
381 }
382
383 lmmc_real_t x_new = x - fx / dfx;
384 if (!std::isfinite(x_new)) {
385 return NumericRootResult::failure(CasErrc::NumericFailure,
386 "Newton update produced a non-finite value",
387 kNewtonOperation);
388 }
389
390 if (i > 1 && std::abs(x_new - x) > 2.0 * std::abs(x - x0)) {
391 x_new = x - 0.5 * fx / dfx;
392 }
393
394 if (x_new < bracket_lo || x_new > bracket_hi) {
395 return bisection_checked(f, var, bracket_lo, bracket_hi, context, opts);
396 }
397
398 x = x_new;
399 }
400
401 return NumericRootResult::success(std::nullopt);
402}
