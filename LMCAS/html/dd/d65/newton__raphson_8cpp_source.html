21double finite_midpoint(double lower, double upper) {
22 if (lower < 0.0 && upper > 0.0) {
23 return lower * 0.5 + upper * 0.5;
24 }
25 return lower + (upper - lower) * 0.5;
26}

28Result<double> evaluate_root_function(
29 const std::shared_ptr<SymbolicExpr>& expression,
30 const std::string& variable,
31 lmmc_real_t value,
32 ComputationContext& context,
33 const char* operation) {
34 if (!expression) {
35 return Result<double>::failure(
36 CasErrc::InvalidArgument,
37 "root function expression cannot be null",
38 operation);
39 }
40 if (variable.empty()) {
41 return Result<double>::failure(
42 CasErrc::InvalidArgument,
43 "root variable cannot be empty",
44 operation);
45 }
46 auto evaluated = evaluate_numeric(
47 *expression, NumericBindings{{variable, static_cast<double>(value)}},
48 context);
49 if (!evaluated) return Result<double>::failure(evaluated.error());
50 if (!evaluated.value().is_finite() ||
51 !std::isfinite(evaluated.value().value)) {
52 return Result<double>::failure(
53 CasErrc::NumericFailure,
54 "root function evaluation is not finite",
55 operation);
56 }
57 return Result<double>::success(evaluated.value().value);
58}

91double evaluate_root_callback(
92 const std::shared_ptr<SymbolicExpr>& expression,
93 double x,
94 RootCallbacks& callbacks) {
95 if (callbacks.error) return std::numeric_limits<double>::quiet_NaN();
96 if (callbacks.enforce_bracket &&
97 (x < callbacks.bracket_lower || x > callbacks.bracket_upper)) {
98 callbacks.left_bracket = true;
99 return std::numeric_limits<double>::quiet_NaN();
100 }
101 auto value = evaluate_root_function(
102 expression, *callbacks.variable, x, *callbacks.context,
103 callbacks.operation);
104 if (!value) {
105 callbacks.error = value.error();
106 return std::numeric_limits<double>::quiet_NaN();
107 }
108 return value.value();
109}

136NumericRootResult backend_root_result(
137 const lmmc_nonlinear_result_t& backend,
138 double residual_limit) {
139 if (!backend.converged || !std::isfinite(backend.root) ||
140 !std::isfinite(backend.residual_norm) ||
141 backend.residual_norm > residual_limit) {
142 return NumericRootResult::success(std::nullopt);
143 }
144 const int iterations =
145 backend.num_iter > static_cast<std::size_t>(std::numeric_limits<int>::max())
146 ? std::numeric_limits<int>::max()

148 return NumericRootResult::success(

152NumericRootResult run_bisection_backend(
153 const std::shared_ptr<SymbolicExpr>& function,
154 const std::string& variable,
155 double lower,
156 double upper,
157 ComputationContext& context,
158 const SolveOptions& options,
159 const char* operation) {
160 if (lower == upper) {
161 auto value = evaluate_root_function(
162 function, variable, lower, context, operation);
163 if (!value) return NumericRootResult::failure(value.error());
164 if (std::abs(value.value()) <= options.tolerance) {
165 return NumericRootResult::success(
166 NumericRoot{lower, std::abs(value.value()), 0});
167 }
168 return NumericRootResult::success(std::nullopt);
169 }
170
171 detail::ensure_lmmc_lifecycle();
172 RootCallbacks callbacks{
173 &function, nullptr, &variable, &context, operation,
174 false, lower, upper, false, std::nullopt};
175 const std::size_t max_iterations =
176 static_cast<std::size_t>(options.max_newton_iterations) * 3;
177 auto config = root_config(options, max_iterations);
178 lmmc_nonlinear_result_t backend{};
179 const auto status = lmmc_bisection_solve(
180 root_function_callback, &callbacks, lower, upper, &config, &backend);
181 if (callbacks.error) return NumericRootResult::failure(*callbacks.error);
182 if (status == LMMC_STATUS_INVALID_ARGUMENT &&
183 backend.failure_reason == LMMC_NONLINEAR_FAILURE_INVALID_BRACKET) {
184 return NumericRootResult::success(std::nullopt);
185 }
186 return backend_root_result(backend, options.tolerance * 100.0);
187}

261NumericRootResult newton_raphson_checked(
262 const std::shared_ptr<SymbolicExpr>& function,
263 const std::shared_ptr<SymbolicExpr>& derivative,
264 const std::string& variable,
265 lmmc_real_t initial,
266 lmmc_real_t bracket_lower,
267 lmmc_real_t bracket_upper,
268 ComputationContext& context,
269 const SolveOptions& options) {
270 auto valid = invalid_root_options(options, kNewtonOperation);
271 if (!valid) return valid;
272 if (!function || !derivative || variable.empty() ||
273 !std::isfinite(initial) || !std::isfinite(bracket_lower) ||
274 !std::isfinite(bracket_upper) || bracket_lower > bracket_upper ||
275 initial < bracket_lower || initial > bracket_upper) {
276 return NumericRootResult::failure(
277 CasErrc::InvalidArgument,
278 "Newton function, derivative, variable, bracket, and initial value are invalid",
279 kNewtonOperation);

281 auto newton = run_newton_backend(
282 function, derivative, variable, initial, context, options, true,
283 bracket_lower, bracket_upper);
284 if (!newton || newton.value()) return newton;
285 return run_bisection_backend(
286 function, variable, bracket_lower, bracket_upper, context, options,
287 kBisectionOperation);
288}

346NumericRootsResult solve_numeric_checked(
347 const std::shared_ptr<SymbolicExpr>& expr,
348 const std::string& var,
349 ComputationContext& context,
350 const SolveOptions& opts)
351{
352 constexpr const char* operation = "solve_numeric";
353 std::vector<NumericRoot> results;
354 auto options = invalid_root_options(opts, operation);
355 if (!options) return NumericRootsResult::failure(options.error());
356 if (!expr) {
357 return NumericRootsResult::failure(CasErrc::InvalidArgument,
358 "expression cannot be null",
359 operation);
360 }
361 if (var.empty()) {
362 return NumericRootsResult::failure(CasErrc::InvalidArgument,
363 "solve variable cannot be empty",
364 operation);
365 }
366 if (opts.max_roots < -1) {
367 return NumericRootsResult::failure(CasErrc::InvalidArgument,
368 "maximum root count must be -1 or non-negative",
369 operation);
370 }
371 if (opts.max_roots == 0) {
372 return NumericRootsResult::success(std::move(results));
373 }
374
375 auto initial_step = context.consume_steps(1, operation);
376 if (!initial_step) return NumericRootsResult::failure(initial_step.error());
377
378 auto recognized_poly = recognize_rational_polynomial(*expr, var, context);
379 if (!recognized_poly) return NumericRootsResult::failure(recognized_poly.error());
380
381 if (recognized_poly.value() && !recognized_poly.value()->is_zero() &&
382 recognized_poly.value()->degree() >= 1) {
383 const Polynomial<Rational>& poly = *recognized_poly.value();
384 auto df_expr = expr->differentiate(var);
385 auto isolated = isolate_real_roots_checked(poly, context);
386 if (!isolated) return NumericRootsResult::failure(isolated.error());
387 for (const auto& [lo_rat, hi_rat] : isolated.value()) {
388 if (opts.max_roots > 0 &&
389 static_cast<int>(results.size()) >= opts.max_roots) {
390 break;
391 }
392
393 auto interval_step = context.consume_steps(1, operation);
394 if (!interval_step) return NumericRootsResult::failure(interval_step.error());
395
396 lmmc_real_t lo = lo_rat.to_double();
397 lmmc_real_t hi = hi_rat.to_double();
398 lmmc_real_t x0 = finite_midpoint(lo, hi);
399 auto root_result = newton_raphson_checked(
400 expr, df_expr, var, x0, lo, hi, context, opts);
401 if (!root_result) return NumericRootsResult::failure(root_result.error());
402 if (!root_result.value()) continue;
403
404 NumericRoot root = *root_result.value();
405 auto verified = evaluate_root_function(
406 expr, var, root.value, context, operation);
407 if (!verified) return NumericRootsResult::failure(verified.error());
408 root.residual = std::abs(verified.value());
409 if (root.residual <= opts.tolerance * 100.0) {
410 results.push_back(root);
411 }
412 }
413 } else {
414 lmmc_real_t x0 = opts.has_initial_guess ? opts.initial_guess : 0.0;
415 auto df_expr = expr->differentiate(var);
416 auto root_result = newton_raphson_checked(
417 expr, df_expr, var, x0, context, opts);
418 if (!root_result) return NumericRootsResult::failure(root_result.error());
419 if (root_result.value()) {
420 NumericRoot root = *root_result.value();
421 auto verified = evaluate_root_function(
422 expr, var, root.value, context, operation);
423 if (!verified) return NumericRootsResult::failure(verified.error());
424 root.residual = std::abs(verified.value());
425 if (root.residual <= opts.tolerance) {
426 results.push_back(root);
427 }
428 }
429 }
430
431 return NumericRootsResult::success(std::move(results));
432}
