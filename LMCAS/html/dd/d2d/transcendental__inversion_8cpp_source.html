39std::vector<std::shared_ptr<SymbolicExpr>> invert_exp_base(
40 const std::shared_ptr<SymbolicExpr>& base,
41 const std::shared_ptr<SymbolicExpr>& c) {
42
43 lmmc_real_t c_val;
44 if (try_evaluate_numeric(c, c_val)) {
45 if (c_val <= 0.0) {
46 return {};
47 }
48 }
49
50 auto ln_c = SymbolicExpr::ln(c);
51 auto ln_a = SymbolicExpr::ln(base);
52 auto sol = SymbolicExpr::divide(ln_c, ln_a)->simplify();
53 return {sol};

74Result<std::optional<detail::AffineForm>> prepare_trigonometric_affine(
75 const ExprPtr& inner, const std::string& var, ComputationContext& context) {
76 auto affine = detail::recognize_affine(*inner, var, context);
77 if (!affine || !affine.value()) { return affine; }
78 /**
79 * @brief 查询值前先规范化尚未求值的多项式系数。
80 * a 的实数性未知时，偏移量 0*a 仍为零；已声明为零的斜率须在除法前识别。
81 */
82 for (auto* coefficient : {&affine.value()->slope, &affine.value()->offset}) {
83 auto normalized = detail::simplify_expression(*coefficient, context);
84 if (!normalized) { return Result<std::optional<detail::AffineForm>>::failure(normalized.error()); }
85 *coefficient = std::move(normalized.value());
86 }
87 return affine;
88}

90Result<bool> real_affine_coefficients(const detail::AffineForm& affine, Tribool nonzero,
91 const FactsQuery& facts, ComputationContext& context) {
92 for (const auto& coefficient : {affine.slope, affine.offset}) {
93 if (coefficient == affine.slope && nonzero == Tribool::False) {
94 continue; /**< 已证有定义的零系数属于实数。 */
95 }
96 auto real_coefficient = detail::query_real_value(detail::node(coefficient), facts, context);
97 if (!real_coefficient) { return Result<bool>::failure(real_coefficient.error()); }
98 auto defined = detail::query_definedness(detail::node(coefficient), facts, Domain::Real, context);
99 if (!defined) { return Result<bool>::failure(defined.error()); }
100 if (real_coefficient.value() != Tribool::True || defined.value() != Tribool::True) {
101 return false;
102 }
103 }
104 return true;
105}

107std::optional<SolveResult> trigonometric_target_range(FunctionNode::FuncType type,
108 const ExprPtr& rhs, const ExprPtr& original, const std::string& var,
109 const FactsQuery& facts, ComputationContext& context, std::vector<ExprPtr>& conditions) {
110 auto real = detail::query_real_value(detail::node(rhs), facts, context);
111 if (!real) { return SolveResult::failure(real.error()); }
112 if (real.value() == Tribool::False) { return SolveResult{SolutionSet{EmptySolutions{}}}; }
113 if (type != FunctionNode::FuncType::Tan) {
114 /**
115 * @brief 构造分支条件前先消除精确端点等式。
116 * 统一 +/-1 分支的恒真条件，使其可合并为完整的半周期解族。
117 */
118 auto lower_difference = detail::simplify_expression(SymbolicExpr::add(rhs, SymbolicExpr::number(1)), context);
119 if (!lower_difference) { return SolveResult::failure(lower_difference.error()); }
120 auto lower = detail::query_nonnegative_value(detail::node(lower_difference.value()), facts, context);
121 if (!lower) { return SolveResult::failure(lower.error()); }
122 auto upper_difference = detail::simplify_expression(subtract(SymbolicExpr::number(1), rhs), context);
123 if (!upper_difference) { return SolveResult::failure(upper_difference.error()); }
124 auto upper = detail::query_nonnegative_value(detail::node(upper_difference.value()), facts, context);
125 if (!upper) { return SolveResult::failure(upper.error()); }
126 if (lower.value() == Tribool::False || upper.value() == Tribool::False) {
127 return SolveResult{SolutionSet{EmptySolutions{}}};
128 }
129 if (lower.value() != Tribool::True) {
130 conditions.push_back(make_expression_ptr(make_node<RelationalNode>(
131 detail::node(rhs), detail::node(SymbolicExpr::number(-1)), RelationalNode::Op::GEQ)));
132 }
133 if (upper.value() != Tribool::True) {
134 conditions.push_back(make_expression_ptr(make_node<RelationalNode>(
135 detail::node(rhs), detail::node(SymbolicExpr::number(1)), RelationalNode::Op::LEQ)));
136 }
137 } else if (real.value() != Tribool::True) {
138 return SolveResult{conditional_preimage(original, var)};
139 }
140 return std::nullopt;
141}

143SolveResult constant_trigonometric_preimage(FunctionNode::FuncType type,
144 const ExprPtr& offset, const ExprPtr& rhs, const ExprPtr& original, const std::string& var,
145 const FactsQuery& facts, ComputationContext& context) {
146 auto constant = make_expression_ptr(make_node<FunctionNode>(
147 type, std::vector<SymbolicNodePtr>{detail::node(offset)}));
148 auto residual = detail::simplify_expression(subtract(constant, rhs), context);
149 if (!residual) { return SolveResult::failure(residual.error()); }
150 auto unequal = detail::query_nonzero_value(detail::node(residual.value()), facts, Domain::Real, context);
151 if (!unequal) { return SolveResult::failure(unequal.error()); }
152 if (unequal.value() == Tribool::False) { return SolutionSet{UniversalSolutions{}}; }
153 if (unequal.value() == Tribool::True) { return SolutionSet{EmptySolutions{}}; }
154 return conditional_preimage(original, var);
155}

157Result<std::string> capture_free_integer_parameter(const ExprPtr& original,
158 const std::string& var, ComputationContext& context) {
159 auto occupied = detail::all_variable_names(detail::node(original));
160 occupied.insert(var);
161 if (context.assumptions()) {
162 /**
163 * @brief 生成的绑定变量须避开全部假设中的名称，包括与方程无关的事实。
164 * 遍历私有作用域栈副本，保持调用方假设栈不变。
165 */
166 AssumptionContext scopes = *context.assumptions();
167 const auto include_expression = [&](const SymbolicExpr& expression) {
168 auto names = detail::all_variable_names(detail::node(expression));
169 occupied.insert(names.begin(), names.end());
170 };
171 for (const auto& conditional : scopes.get_active_conditionals()) {
172 include_expression(conditional.condition);
173 include_expression(conditional.conclusion);
174 }
175 for (;;) {
176 auto names = scopes.current_properties().get_all_symbols();
177 occupied.insert(names.begin(), names.end());
178 for (const auto& relation : scopes.current_relations().get_relations()) {
179 include_expression(relation.lhs);
180 include_expression(relation.rhs);
181 }
182 if (scopes.depth() == 1) { break; }
183 auto popped = scopes.pop();
184 if (!popped) { return Result<std::string>::failure(popped.error()); }
185 }
186 }
187 std::string parameter = "_k";
188 for (std::size_t suffix = 1; occupied.count(parameter); ++suffix) {
189 parameter = "_k" + std::to_string(suffix);
190 }
191 return parameter;
192}

194ExprPtr trigonometric_principal_value(FunctionNode::FuncType type,
195 const ExprPtr& rhs, const ExprPtr& pi) {
196 using Type = FunctionNode::FuncType;
197 auto principal = inverse_trig(type == Type::Sin ? Type::ArcSin : Type::ArcCos, rhs);
198 const auto positive_half = SymbolicExpr::number(Rational(1, 2));
199 const auto negative_half = SymbolicExpr::number(Rational(-1, 2));
200 if (detail::node(rhs)->equals(*detail::node(positive_half))) {
201 principal = SymbolicExpr::multiply(SymbolicExpr::number(
202 type == Type::Sin ? Rational(1, 6) : Rational(1, 3)), pi);
203 } else if (detail::node(rhs)->equals(*detail::node(negative_half))) {
204 principal = SymbolicExpr::multiply(SymbolicExpr::number(
205 type == Type::Sin ? Rational(-1, 6) : Rational(2, 3)), pi);
206 }
207 return principal;
208}

210std::vector<ExprPtr> trigonometric_periodic_values(FunctionNode::FuncType type,
211 const ExprPtr& rhs, const std::string& parameter) {
212 using Type = FunctionNode::FuncType;
213 auto pi = detail::make_expression_ptr(
214 detail::make_node<VariableNode>("pi", true));
215 auto integer_pi = SymbolicExpr::multiply(SymbolicExpr::variable(parameter), pi);
216 auto twice_integer_pi = SymbolicExpr::multiply(SymbolicExpr::number(2), integer_pi);
217 auto half_pi = SymbolicExpr::multiply(SymbolicExpr::number(Rational(1, 2)), pi);
218 auto integer = exact_small_integer_node(detail::node(rhs->simplify()), -1, 1);
219 std::vector<ExprPtr> values;
220 if (type == Type::Tan) {
221 auto principal = integer ?
222 SymbolicExpr::multiply(SymbolicExpr::number(Rational(*integer, 4)), pi) :
223 inverse_trig(Type::ArcTan, rhs);
224 values.push_back(SymbolicExpr::add(principal, integer_pi));
225 } else if (integer && *integer == 0) {
226 values.push_back(type == Type::Sin ? integer_pi :
227 SymbolicExpr::add(half_pi, integer_pi));
228 } else if (integer && (*integer == 1 || *integer == -1)) {
229 auto principal = type == Type::Sin ?
230 SymbolicExpr::multiply(SymbolicExpr::number(*integer), half_pi) :
231 (*integer == 1 ? SymbolicExpr::number(0) : pi);
232 values.push_back(SymbolicExpr::add(principal, twice_integer_pi));
233 } else {
234 auto principal = trigonometric_principal_value(type, rhs, pi);
235 values.push_back(SymbolicExpr::add(principal, twice_integer_pi));
236 values.push_back(SymbolicExpr::add(
237 type == Type::Sin ? subtract(pi, principal) :
238 SymbolicExpr::multiply(SymbolicExpr::number(-1), principal),
239 twice_integer_pi));
240 }
241 return values;
242}

244SolveResult invert_trigonometric(
245 const InversePattern& pattern, const ExprPtr& rhs, const ExprPtr& original,
246 const std::string& var, const FactsQuery& facts, ComputationContext& context) {
247 auto affine = prepare_trigonometric_affine(pattern.inner, var, context);
248 if (!affine) { return SolveResult::failure(affine.error()); }
249 if (!affine.value()) { return conditional_preimage(original, var); }
250 auto nonzero = detail::query_nonzero_value(detail::node(affine.value()->slope), facts, Domain::Real, context);
251 if (!nonzero) { return SolveResult::failure(nonzero.error()); }
252 auto real_coefficients = real_affine_coefficients(*affine.value(), nonzero.value(), facts, context);
253 if (!real_coefficients) { return SolveResult::failure(real_coefficients.error()); }
254 if (!real_coefficients.value()) { return conditional_preimage(original, var); }
255 std::vector<ExprPtr> conditions;
256 if (auto decided = trigonometric_target_range(
257 pattern.func_type, rhs, original, var, facts, context, conditions)) {
258 return std::move(*decided);
259 }
260 if (nonzero.value() == Tribool::False) {
261 return constant_trigonometric_preimage(
262 pattern.func_type, affine.value()->offset, rhs, original, var, facts, context);
263 }
264 if (nonzero.value() != Tribool::True) { return conditional_preimage(original, var); }
265 auto fresh = capture_free_integer_parameter(original, var, context);
266 if (!fresh) { return SolveResult::failure(fresh.error()); }
267 const auto& parameter = fresh.value();
268 auto values = trigonometric_periodic_values(pattern.func_type, rhs, parameter);
269 ParametricSolutions solutions;
270 for (const auto& value : values) {
271 auto mapped = detail::simplify_expression(SymbolicExpr::divide(
272 subtract(value, affine.value()->offset), affine.value()->slope), context);
273 if (!mapped) { return SolveResult::failure(mapped.error()); }
274 solutions.values.push_back(ParametricSolution{
275 std::move(mapped.value()), {parameter}, conditions});
276 }
277 return SolutionSet{std::move(solutions)};
278}

280SolveResult invert_exponential(const ExprPtr& rhs, const ExprPtr& original,
281 const std::string& var, const FactsQuery& facts, ComputationContext& context) {
282 auto real = detail::query_real_value(detail::node(rhs), facts, context);
283 if (!real) { return SolveResult::failure(real.error()); }
284 auto positive = detail::query_positive_value(detail::node(rhs), facts, context);
285 if (!positive) { return SolveResult::failure(positive.error()); }
286 if (real.value() == Tribool::True && positive.value() == Tribool::False) {
287 return SolutionSet{EmptySolutions{}};
288 }
289 if (positive.value() != Tribool::True) { return conditional_preimage(original, var); }
290 auto value = SymbolicExpr::ln(rhs)->simplify();
291 return SolutionSet{FiniteSolutions{{FiniteSolution{std::move(value), 1, {}}}}};
292}

305SolveResult invert_function(
306 const InversePattern& pattern, const ExprPtr& original, const std::string& var,
307 ComputationContext& context, const SolveOptions&) {
308 auto step = context.consume_steps(1, "solve.inverse");
309 if (!step) { return SolveResult::failure(step.error()); }
310 std::optional<detail::AssumptionFacts> assumption_facts;
311 if (context.assumptions()) { assumption_facts.emplace(*context.assumptions()); }
312 const FactsQuery& facts = assumption_facts ? static_cast<const FactsQuery&>(*assumption_facts) :
313 detail::no_facts();
314 auto rhs = pattern.rhs;
315 if (pattern.coefficient) {
316 auto nonzero = detail::query_nonzero_value(detail::node(pattern.coefficient), facts, Domain::Real, context);
317 if (!nonzero) { return SolveResult::failure(nonzero.error()); }
318 if (nonzero.value() != Tribool::True) { return conditional_preimage(original, var); }
319 rhs = SymbolicExpr::divide(rhs, pattern.coefficient)->simplify();
320 }
321 switch (pattern.func_type) {
322 case FunctionNode::FuncType::Sin:
323 case FunctionNode::FuncType::Cos:
324 case FunctionNode::FuncType::Tan:
325 return invert_trigonometric(pattern, rhs, original, var, facts, context);
326 case FunctionNode::FuncType::Exp:
327 return invert_exponential(rhs, original, var, facts, context);
328 case FunctionNode::FuncType::Ln:
329 return invert_logarithm(rhs, original, var, facts, context);
330 default:
331 return SolveResult::failure(CasErrc::Inconclusive,
332 "Inverse requires a complete solution-set representation", "solve.inverse");
333 }
334}
