147/**
148 * @brief 符号表达式 AST 节点基类.
149 *
150 * 所有具体节点(数值,变量,运算符,函数等)均继承此类.
151 * 提供哈希,比较,克隆等通用接口,支持 Visitor 模式遍历.
152 */
153class SymbolicNode {
154protected:
155 mutable std::atomic<std::size_t> cached_hash{0};
156 mutable std::atomic<bool> hash_computed{false};
157
158 SymbolicNode() = default;
159 SymbolicNode(const SymbolicNode&) : cached_hash(0), hash_computed(false) {}
160 SymbolicNode& operator=(const SymbolicNode&) { return *this; }
161
162 /** @brief 计算当前节点的哈希值(由子类实现). */
163 virtual std::size_t compute_hash() const = 0;
164
165 /** @brief 与同类型节点进行比较(由子类实现). */
166 virtual int compare_same_type(const SymbolicNode& other) const = 0;
167
168public:
169 virtual ~SymbolicNode() = default;
170
171 /** @brief 接受 Visitor 访问. */
172 virtual void accept(LMCAS::detail::SymbolicVisitor& visitor) const = 0;
173
174 /** @brief 深拷贝当前节点及其子树. */
175 virtual std::shared_ptr<const SymbolicNode> clone() const = 0;
176
177 /** @brief 返回节点类型的排序优先级,用于规范化排序. */
178 virtual int type_priority() const = 0;
179
180 /**
181 * @brief 获取节点哈希值(带缓存).
182 * @return 哈希值
183 */
184 std::size_t hash() const {
185 if (!hash_computed.load(std::memory_order_acquire)) {
186 std::size_t h = compute_hash();
187 cached_hash.store(h, std::memory_order_relaxed);
188 hash_computed.store(true, std::memory_order_release);
189 return h;
190 }
191 return cached_hash.load(std::memory_order_relaxed);
192 }
193
194 /**
195 * @brief 与另一个节点进行全序比较.
196 * @param other 待比较的节点
197 * @return 小于返回 -1,等于返回 0,大于返回 1
198 */
199 int compare(const SymbolicNode& other) const {
200 if (type_priority() != other.type_priority()) {
201 return type_priority() < other.type_priority() ? -1 : 1;
202 }
203 return compare_same_type(other);
204 }
205
206 /**
207 * @brief 判断两个节点是否结构相等.
208 * @param other 待比较的节点
209 * @return 相等返回 true
210 */
211 bool equals(const SymbolicNode& other) const {
212 if (this == &other) return true;
213 if (hash() != other.hash()) return false;
214 if (type_priority() != other.type_priority()) return false;
215 return compare_same_type(other) == 0;
216 }
217
218 /** @brief 判断节点是否为数值节点. */
219 virtual bool is_number() const { return false; }
220 /** @brief 判断节点数值是否为 1. */
221 virtual bool is_one() const { return false; }
222 /** @brief 判断节点数值是否为 0. */
223 virtual bool is_zero() const { return false; }
224 /** @brief 判断节点是否可证明为严格正数(非数值节点默认返回 false). */
225 virtual bool is_positive() const { return false; }
226};

297struct NodeEqual {
298 bool operator()(const std::shared_ptr<const SymbolicNode>& lhs, const std::shared_ptr<const SymbolicNode>& rhs) const {
299 if (!lhs || !rhs) return lhs == rhs;
300 return lhs->equals(*rhs);
