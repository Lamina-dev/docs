140/**
141 * @brief 符号表达式 AST 节点基类.
142 *
143 * 所有具体节点(数值,变量,运算符,函数等)均继承此类.
144 * 提供哈希,比较,克隆等通用接口,支持 Visitor 模式遍历.
145 */
146class SymbolicNode {
147protected:
148 mutable std::atomic<std::size_t> cached_hash{0};
149 mutable std::atomic<bool> hash_computed{false};
150
151 SymbolicNode() = default;
152 SymbolicNode(const SymbolicNode&) : cached_hash(0), hash_computed(false) {}
153 SymbolicNode& operator=(const SymbolicNode&) { return *this; }
154
155 /** @brief 计算当前节点的哈希值(由子类实现). */
156 virtual std::size_t compute_hash() const = 0;
157
158 /** @brief 与同类型节点进行比较(由子类实现). */
159 virtual int compare_same_type(const SymbolicNode& other) const = 0;
160
161public:
162 virtual ~SymbolicNode() = default;
163
164 /** @brief 接受 Visitor 访问. */
165 virtual void accept(lamina::detail::SymbolicVisitor& visitor) const = 0;
166
167 /** @brief 深拷贝当前节点及其子树. */
168 virtual std::shared_ptr<const SymbolicNode> clone() const = 0;
169
170 /** @brief 返回节点类型的排序优先级,用于规范化排序. */
171 virtual int type_priority() const = 0;
172
173 /**
174 * @brief 获取节点哈希值(带缓存).
175 * @return 哈希值
176 */
177 std::size_t hash() const {
178 if (!hash_computed.load(std::memory_order_acquire)) {
179 std::size_t h = compute_hash();
180 cached_hash.store(h, std::memory_order_relaxed);
181 hash_computed.store(true, std::memory_order_release);
182 return h;
183 }
184 return cached_hash.load(std::memory_order_relaxed);
185 }
186
187 /**
188 * @brief 与另一个节点进行全序比较.
189 * @param other 待比较的节点
190 * @return 小于返回 -1,等于返回 0,大于返回 1
191 */
192 int compare(const SymbolicNode& other) const {
193 if (type_priority() != other.type_priority()) {
194 return type_priority() < other.type_priority() ? -1 : 1;
195 }
196 return compare_same_type(other);
197 }
198
199 /**
200 * @brief 判断两个节点是否结构相等.
201 * @param other 待比较的节点
202 * @return 相等返回 true
203 */
204 bool equals(const SymbolicNode& other) const {
205 if (this == &other) return true;
206 if (hash() != other.hash()) return false;
207 if (type_priority() != other.type_priority()) return false;
208 return compare_same_type(other) == 0;
209 }
210
211 /** @brief 判断节点是否为数值节点. */
212 virtual bool is_number() const { return false; }
213 /** @brief 判断节点数值是否为 1. */
214 virtual bool is_one() const { return false; }
215 /** @brief 判断节点数值是否为 0. */
216 virtual bool is_zero() const { return false; }
217 /** @brief 判断节点是否可证明为严格正数(非数值节点默认返回 false). */
218 virtual bool is_positive() const { return false; }
219};
