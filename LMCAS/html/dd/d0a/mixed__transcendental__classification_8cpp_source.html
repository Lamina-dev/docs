173static BigInt sum_degree(const AddNode& sum, const std::string& var) {
174 BigInt maximum;
175 for (const auto& operand : sum.operands()) {
176 auto degree = degree_in_var(operand, var);
177 if (degree < 0) {
178 return -1;
179 }
180 if (degree > maximum) {
181 maximum = std::move(degree);
182 }
183 }
184 return maximum;
185}

187static BigInt product_degree(const MultiplyNode& product, const std::string& var) {
188 BigInt total;
189 for (const auto& operand : product.operands()) {
190 auto degree = degree_in_var(operand, var);
191 if (degree < 0) {
192 return -1;
193 }
194 total += degree;
195 }
196 return total;

199/**
200 * @internal
201 * @brief 计算节点关于指定变量的多项式次数。
202 * 与 var 无关的子表达式次数为 0。
203 * @param[in] node 符号节点。
204 * @param[in] var 变量名。
205 * @return 关于 var 的次数；超出多项式次数支持域时返回 -1。
206 */
207static BigInt degree_in_var(const std::shared_ptr<const SymbolicNode>& node, const std::string& var) {
208 if (!node) {
209 return 0;
210 }
211
212 if (!expression_depends_on_variable(node, var)) {
213 return 0;
214 }
215
216 if (auto v = std::dynamic_pointer_cast<const VariableNode>(node)) {
217 return (!v->is_constant() && v->name() == var) ? 1 : 0;
218 }
219
220 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
221 return sum_degree(*add, var);
222 }
223
224 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
225 return product_degree(*mul, var);
226 }
227
228 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
229 auto exp_num = std::dynamic_pointer_cast<const NumberNode>(pow->exponent());
230 BigInt e_val;
231 if (!try_get_integer_value(exp_num, e_val) || e_val.is_negative()) {
232 return -1;
233 }
234
235 BigInt base_deg = degree_in_var(pow->base(), var);
236 if (base_deg < 0) {
237 return -1;
238 }
239 return base_deg * e_val;
240 }
241 if (std::dynamic_pointer_cast<const FunctionNode>(node)) {
242 return -1;
243 }
244
245 return 0;
246}

262bool is_polynomial_after_substitution(
263 const TransSubstitutionResult& sub_result)
264{
265 if (sub_result.mappings.empty()) {
266 return false;
267 }
268 if (!sub_result.poly_expr || !LMCAS::detail::node(sub_result.poly_expr)) {
269 return false;
270 }
271
272 const auto& root = LMCAS::detail::node(sub_result.poly_expr);
273
274 for (const auto& m : sub_result.mappings) {
275 if (expression_depends_on_variable(root, m.indeterminate) &&
276 degree_in_var(root, m.indeterminate).is_negative()) {
277 return false;
278 }
279 }
280
281 return true;
282}
