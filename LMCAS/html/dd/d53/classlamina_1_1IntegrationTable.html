54 auto u_is_var = [](const std::string& wc) {
55 return [wc](const MatchMap& m, const std::string& var) -> bool {
56 auto it = m.find(wc);
57 if (it == m.end()) return false;

64 auto u_is_var_a_indep = [](const std::string& u_wc, const std::string& a_wc) {
65 return [u_wc, a_wc](const MatchMap& m, const std::string& var) -> bool {
