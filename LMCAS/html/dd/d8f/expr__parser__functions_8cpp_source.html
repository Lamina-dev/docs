101ExprResult ExprParser::apply_limit(const std::vector<ExprPtr>& arguments) {
102 auto variable = std::dynamic_pointer_cast<const VariableNode>(
103 LMCAS::detail::node(arguments[1]));
104 auto direction = std::dynamic_pointer_cast<const VariableNode>(
105 LMCAS::detail::node(arguments[3]));
106 if (!variable || variable->is_constant() ||
107 !direction || direction->is_constant()) {
108 return fail("limit variable and direction must be symbols");
109 }
110 LimitDirection parsed_direction;
111 if (direction->name() == "both") {
112 parsed_direction = LimitDirection::Both;
113 } else if (direction->name() == "below" ||
114 direction->name() == "left" ||
115 direction->name() == "from_below") {
116 parsed_direction = LimitDirection::FromBelow;
117 } else if (direction->name() == "above" ||
118 direction->name() == "right" ||
119 direction->name() == "from_above") {
120 parsed_direction = LimitDirection::FromAbove;
121 } else {
122 return fail("invalid limit direction");
123 }
124 return ExprResult::success(LMCAS::detail::make_expression_ptr(
125 LMCAS::detail::make_node<LimitNode>(
126 LMCAS::detail::node(arguments[0]), variable->name(),
127 LMCAS::detail::node(arguments[2]), parsed_direction)));
128}

130ExprResult ExprParser::apply_rootof(const std::vector<ExprPtr>& arguments) {
131 auto variable = std::dynamic_pointer_cast<const VariableNode>(
132 LMCAS::detail::node(arguments[1]));
133 auto index = std::dynamic_pointer_cast<const NumberNode>(
134 LMCAS::detail::node(arguments[2]));
135 if (!variable || variable->is_constant() || !index ||
136 std::holds_alternative<lmmc_real_t>(index->value())) {
137 return fail("RootOf variable and index are invalid");
138 }
139 BigInt index_value;
140 if (std::holds_alternative<BigInt>(index->value())) {
141 index_value = std::get<BigInt>(index->value());
142 } else {
143 const auto& rational = std::get<Rational>(index->value());
144 if (!rational.is_integer()) {
145 return fail("RootOf index must be an integer");
146 }
147 index_value = rational.to_bigint();
148 }
149 const auto converted = index_value.try_to_int64();
150 if (!converted || *converted < 0) {
151 return fail("RootOf index is negative or too large");
152 }
153 auto root = make_rootof_checked(
154 arguments[0], variable->name(),
155 static_cast<std::size_t>(*converted));
156 if (!root) {
157 return ExprResult::failure(root.error());
158 }
159 return ExprResult::success(std::move(root.value()));
160}

166ExprResult ExprParser::apply_function(const std::string& name,
167 const std::vector<ExprPtr>& arguments) {
168 if (auto unary = find_unary_function(name)) {
169 if (arguments.size() == 1) {
170 return unary(arguments[0]);
171 }
172 if (name == "log" && arguments.size() == 2) {
173 return apply_log(arguments);
174 }
175 return fail("invalid argument count for function '" + name + "'");
176 }
177 using Constructor = ExprResult (ExprParser::*)(const std::vector<ExprPtr>&);
178 static const struct {
179 const char* name;
180 Constructor construct;
181 std::size_t arity;
182 std::size_t alternative_arity;
183 } functions[] = {
184 {"atan2", &ExprParser::apply_atan2, 2, 2},
185 {"clamp", &ExprParser::apply_clamp, 3, 3},
186 {"Integral", &ExprParser::apply_integral, 2, 4},
187 {"integral", &ExprParser::apply_integral, 2, 4},
188 {"Limit", &ExprParser::apply_limit, 4, 4},
189 {"limit", &ExprParser::apply_limit, 4, 4},
190 {"RootOf", &ExprParser::apply_rootof, 3, 3},
191 {"rootof", &ExprParser::apply_rootof, 3, 3}
192 };
193 for (const auto& entry : functions) {
194 if (name == entry.name) {
195 if (arguments.size() != entry.arity &&
196 arguments.size() != entry.alternative_arity) {
197 return fail("invalid argument count for function '" + name + "'");
198 }
199 return (this->*entry.construct)(arguments);
200 }
201 }
202 if (!arguments.empty() && (name == "max" || name == "min")) {
203 return function_node(name, arguments,
204 name == "max" ? FunctionNode::FuncType::Max : FunctionNode::FuncType::Min);
205 }
206 return uninterpreted_function(name, arguments);
207}

209ExprResult ExprParser::uninterpreted_function(
210 const std::string& name, const std::vector<ExprPtr>& arguments) {
211 try {
212 std::vector<std::shared_ptr<const SymbolicNode>> nodes;
213 nodes.reserve(arguments.size());
214 for (const auto& argument : arguments) {
215 if (!argument || !LMCAS::detail::node(argument)) {
216 return expr_detail::expr_common::expression_failure(CasErrc::InvalidArgument,
217 "function arguments cannot be null",
218 kParseOperation);
219 }
220 nodes.push_back(LMCAS::detail::node(argument));
221 }
222 auto node = LMCAS::detail::make_node<UninterpretedFunctionNode>(
223 name, std::move(nodes));
224 return ExprResult::success(LMCAS::detail::make_expression_ptr(std::move(node)));
225 } catch (const std::bad_alloc&) {
226 return expr_detail::expr_common::expression_failure(CasErrc::ResourceLimit,
227 "function expression allocation failed",
228 kParseOperation);
229 } catch (const std::exception& error) {
230 return expr_detail::expr_common::expression_failure(CasErrc::InternalInvariant, error.what(),
231 kParseOperation);
232 }
233}
