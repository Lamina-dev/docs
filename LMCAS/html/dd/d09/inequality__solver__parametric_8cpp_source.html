28bool satisfies(int sign, InequalityType type) {
29 switch (type) {
30 case InequalityType::GreaterThan: return sign > 0;
31 case InequalityType::GreaterEqual: return sign >= 0;
32 case InequalityType::LessThan: return sign < 0;
33 case InequalityType::LessEqual: return sign <= 0;
34 }
35 return false;
36}

51Result<void> exact_polynomial_nodes(
52 const detail::SymbolicNodePtr& node, const std::string& variable,
53 const std::vector<std::string>& parameters, ComputationContext& context,
54 std::size_t depth) {
55 if (depth >= context.limits().max_recursion_depth) {
56 return Result<void>::failure(CasErrc::ResourceLimit,
57 "parameter expression recursion limit exceeded", kParametricOperation);
58 }
59 auto step = context.consume_steps(1, kParametricOperation);
60 if (!step) return step;
61 if (const auto number = std::dynamic_pointer_cast<const NumberNode>(node)) {
62 if (detail::exact_rational_value(*number)) return Result<void>::success();
63 } else if (const auto symbol = std::dynamic_pointer_cast<const VariableNode>(node)) {
64 if (!symbol->is_constant() &&
65 (symbol->name() == variable ||
66 std::find(parameters.begin(), parameters.end(), symbol->name()) != parameters.end())) {
67 return Result<void>::success();
68 }
69 } else if (const auto sum = std::dynamic_pointer_cast<const AddNode>(node)) {
70 return exact_operands(*sum, variable, parameters, context, depth);
71 } else if (const auto product = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
72 return exact_operands(*product, variable, parameters, context, depth);
73 } else if (const auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
74 if (exact_small_integer_node(
75 power->exponent(), 0, std::numeric_limits<int>::max())) {
76 return exact_polynomial_nodes(
77 power->base(), variable, parameters, context, depth + 1);
78 }
79 }
80 return Result<void>::failure(CasErrc::Inconclusive,
81 "parameter expression is not an exact polynomial", kParametricOperation);
82}

96Result<Interval> linear_interval(
97 const std::vector<std::shared_ptr<SymbolicExpr>>& coeffs,
98 InequalityType type, ComputationContext& context) {
99 const auto slope = detail::exact_rational_value(detail::node(coeffs[1]));
100 if (!slope) {
101 return Result<Interval>::failure(inconclusive().error());
102 }
103 if (*slope == Rational(0)) {
104 return Result<Interval>::failure(inconclusive().error());
105 }
106 auto root = detail::simplify_expression(SymbolicExpr::divide(
107 SymbolicExpr::multiply(SymbolicExpr::number(-1), coeffs[0]),
108 coeffs[1]), context);
109 if (!root) {
110 return Result<Interval>::failure(root.error());
111 }
112
113 const bool greater = satisfies(1, type);
114 const bool strict = type == InequalityType::GreaterThan ||
115 type == InequalityType::LessThan;
116 const bool above = greater != slope->get_numerator().is_negative();
117 const auto endpoint = strict ? Endpoint::open(root.value())
118 : Endpoint::closed(root.value());
119 return above ? Interval{endpoint, Endpoint::pos_inf()}
120 : Interval{Endpoint::neg_inf(), endpoint};
121}

133PiecewiseResult constant_solution(const std::shared_ptr<SymbolicExpr>& coefficient,
134 InequalityType type) {

145PiecewiseResult quadratic_solution(const std::shared_ptr<SymbolicExpr>& coefficient,
146 InequalityType type) {
147 PiecewiseIntervalResult result;
148 for (int sign : {1, 0, -1}) {
149 auto solution = sign == 0 ? zero_polynomial_solution(type)
150 : detail::inequality_support::build_parametric_intervals(
151 {SymbolicExpr::number(0)}, {2}, sign, type);
152 result.cases.push_back({sign_condition(coefficient, sign), std::move(solution)});
153 }
154 return result;
155}
156PiecewiseResult nonlinear_solution(
157 const std::vector<std::shared_ptr<SymbolicExpr>>& coeffs,
158 int degree, InequalityType type) {
159 if (degree == 0) return constant_solution(coeffs[0], type);
160 if (!coeffs[0]->is_zero() || !coeffs[1]->is_zero()) return inconclusive();
161 return quadratic_solution(coeffs[2], type);
162}

181Result<std::vector<std::shared_ptr<SymbolicExpr>>> simplify_coefficients(
182 const Polynomial<SymbolicPolyCoeff>& poly, bool& symbolic,
183 ComputationContext& context) {
184 std::vector<std::shared_ptr<SymbolicExpr>> coeffs;
185 coeffs.reserve(poly.coeffs.size());
186 for (const auto& coefficient : poly.coeffs) {
187 auto simplified = detail::simplify_expression(coefficient.val, context);
188 if (!simplified)
189 return Result<std::vector<std::shared_ptr<SymbolicExpr>>>::failure(simplified.error());
190 auto value = std::move(simplified.value());
191 if (!detail::exact_rational_value(detail::node(value))) symbolic = true;
192 coeffs.push_back(std::move(value));
193 }
194 return coeffs;
195}

198Result<PiecewiseIntervalResult> InequalitySolver::solve_parametric_inequality_checked(
199 const std::shared_ptr<SymbolicExpr>& expr,
200 InequalityType type,
201 const std::string& variable,
202 const std::vector<std::string>& parameters,
203 ComputationContext& context) {
204 auto arguments = check_arguments(expr, variable, parameters);
205 if (!arguments) {
206 return PiecewiseResult::failure(arguments.error());
207 }
208
209 try {
210 if (parameters.empty()) {
211 return solved_inequality(expr, type, variable, context);
212 }
213 auto converted = detail::symbolic_to_poly_checked(*expr, variable, context);
214 if (!converted) {
215 return PiecewiseResult::failure(converted.error());
216 }
217 const auto& poly = converted.value();
218 if (poly.degree() > 2) {
219 return inconclusive();
220 }
221
222 auto exact = exact_polynomial_nodes(
223 detail::node(expr), variable, parameters, context, 0);
224 if (!exact) {
225 return PiecewiseResult::failure(exact.error());
226 }
227
228 if (poly.is_zero()) {
229 PiecewiseIntervalResult result;
230 result.cases.push_back({nullptr, zero_polynomial_solution(type)});
231 return result;
232 }
233
234 bool symbolic = false;
235 auto simplified = simplify_coefficients(poly, symbolic, context);
236 if (!simplified) {
237 return PiecewiseResult::failure(simplified.error());
238 }
239 const auto& coeffs = simplified.value();
240
241 if (!symbolic) {
242 return solved_inequality(expr, type, variable, context);
243 }
244
245 auto nodes = context.reserve_nodes(16, kParametricOperation);
246 if (!nodes) {
247 return PiecewiseResult::failure(nodes.error());
248 }
249 if (poly.degree() != 1) {
250 return nonlinear_solution(coeffs, poly.degree(), type);
251 }
252 auto interval = linear_interval(coeffs, type, context);
253 if (!interval) {
254 return PiecewiseResult::failure(interval.error());
255 }
256 PiecewiseIntervalResult result;
257 result.cases.push_back({nullptr, IntervalUnion::from_checked_normalized(
258 {std::move(interval.value())})});
259 return result;
260 } catch (const CasError& error) {
261 return PiecewiseResult::failure(error);
262 } catch (const std::bad_alloc&) {
263 return PiecewiseResult::failure(
264 CasErrc::ResourceLimit, "parametric inequality allocation failed",
265 kParametricOperation);
266 } catch (const std::exception& error) {
267 return PiecewiseResult::failure(
268 CasErrc::InternalInvariant, error.what(), kParametricOperation);
269 }
270}
