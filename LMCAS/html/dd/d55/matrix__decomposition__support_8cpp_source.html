71std::optional<Rational> exact_rational_expr(const std::shared_ptr<const SymbolicNode>& node) {
72 if (!node) return std::nullopt;
73 if (auto number = std::dynamic_pointer_cast<const NumberNode>(node)) {
74 return LMCAS::detail::exact_rational_value(*number);
75 }
76 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
77 return exact_rational_sum(*add);
78 }
79 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
80 return exact_rational_product(*mul);
81 }
82 if (auto fn = std::dynamic_pointer_cast<const FunctionNode>(node)) {
83 if (fn->type() == FunctionNode::FuncType::Abs && fn->arguments().size() == 1) {
84 auto value = exact_rational_expr(fn->arguments()[0]);
85 if (!value) return std::nullopt;
86 return (*value < Rational(0)) ? (Rational(0) - *value) : *value;
87 }
88 }
89 return std::nullopt;
90}

150std::optional<std::vector<Rational>> exact_rectangular_diagonal_entries(
151 const std::shared_ptr<const MatrixNode>& matrix) {
152 auto values = exact_rational_matrix(matrix);
153 if (!values) return std::nullopt;
154 const size_t rows = values->size();
155 const size_t cols = rows == 0 ? 0 : (*values)[0].size();
156 const size_t diag_count = std::min(rows, cols);
157 std::vector<Rational> diagonal(diag_count, Rational(0));
158 for (size_t row = 0; row < rows; ++row) {
159 for (size_t col = 0; col < cols; ++col) {
160 const bool is_diag = row == col && row < diag_count;
161 if (!is_diag && (*values)[row][col] != Rational(0)) {
162 return std::nullopt;
163 }
164 if (is_diag) {
165 diagonal[row] = (*values)[row][col];
166 }
167 }
168 }
169 return diagonal;
170}

172std::optional<std::vector<Rational>> exact_diagonal_entries(
173 const std::shared_ptr<const MatrixNode>& matrix) {
174 auto values = exact_rational_matrix(matrix);
175 if (!values) return std::nullopt;
176 const size_t rows = values->size();
177 const size_t cols = rows == 0 ? 0 : (*values)[0].size();
178 if (rows != cols) return std::nullopt;
179 std::vector<Rational> diagonal(rows, Rational(0));
180 for (size_t row = 0; row < rows; ++row) {
181 for (size_t col = 0; col < cols; ++col) {
182 if (row == col) {
183 diagonal[row] = (*values)[row][col];
184 } else if ((*values)[row][col] != Rational(0)) {
185 return std::nullopt;
186 }
187 }
188 }
189 return diagonal;
