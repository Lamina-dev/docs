8Result<Rational> rational_determinant_value_impl(
9 std::size_t dimension,
10 std::vector<Rational> values,
11 ComputationContext& context,
12 const std::string& operation) {
13 if (dimension == 0 ||
14 dimension > std::numeric_limits<std::size_t>::max() / dimension ||
15 values.size() != dimension * dimension) {
16 return Result<Rational>::failure(
17 CasErrc::InvalidArgument,
18 "rational determinant storage does not match its dimension",
19 operation);
20 }
21 Rational sign(1);
22 Rational previous(1);
23 if (dimension == 1) return Result<Rational>::success(values[0]);
24 for (std::size_t pivot_column = 0;
25 pivot_column + 1 < dimension; ++pivot_column) {
26 auto budget = context.consume_steps(1, operation);
27 if (!budget) return Result<Rational>::failure(budget.error());
28 std::size_t pivot_row = pivot_column;
29 while (pivot_row < dimension &&
30 values[pivot_row * dimension + pivot_column] == Rational(0)) {
31 ++pivot_row;
32 }
33 if (pivot_row == dimension) {
34 return Result<Rational>::success(Rational(0));
35 }
36 if (pivot_row != pivot_column) {
37 for (std::size_t column = 0; column < dimension; ++column) {
38 std::swap(values[pivot_row * dimension + column],
39 values[pivot_column * dimension + column]);
40 }
41 sign = -sign;
42 }
43 const Rational pivot =
44 values[pivot_column * dimension + pivot_column];
45 for (std::size_t row = pivot_column + 1; row < dimension; ++row) {
46 for (std::size_t column = pivot_column + 1;
47 column < dimension; ++column) {
48 values[row * dimension + column] =
49 (values[row * dimension + column] * pivot -
50 values[row * dimension + pivot_column] *
51 values[pivot_column * dimension + column]) /
52 previous;
53 }
54 values[row * dimension + pivot_column] = Rational(0);
55 }
56 previous = pivot;
57 }
58 return Result<Rational>::success(
59 sign * values[(dimension - 1) * dimension + (dimension - 1)]);
60}
