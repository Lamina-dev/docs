31int root_power_sign(const PowerNode& power) {
32 auto exp_expr = LMCAS::detail::make_expression_ptr(power.exponent());
33 if (auto ev = try_checked_numeric_constant(*exp_expr)) {
34 if (*ev > 0 && *ev < 1.0) {
35 return 1;
36 }
37 }
38 auto base_expr = LMCAS::detail::make_expression_ptr(power.base());
39 auto bv_checked = try_checked_numeric_constant(*base_expr);
40 auto ev_checked = try_checked_numeric_constant(*exp_expr);
41 if (!bv_checked || !ev_checked) {
42 return 0;
43 }
44 double bv = *bv_checked;
45 double ev = *ev_checked;

77int root_expression_sign(const std::shared_ptr<SymbolicExpr>& expr) {
78 if (!expr || !LMCAS::detail::node(expr)) {
79 return 0;
80 }

82 if (direct != 0) {
