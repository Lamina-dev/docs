48[[noreturn]] void throw_property_error(const CasError& error) {
49 if (error.code == CasErrc::InvalidArgument) {
50 throw std::invalid_argument(error.message);
51 }
52 if (error.code == CasErrc::ResourceLimit) {
53 throw std::bad_alloc();
54 }

58template <typename F>
59PropertyStoreResult checked_property_update(PropertyStore& store,
60 const std::string& symbol,
61 const std::string& operation,
62 F&& update) {
63 if (symbol.empty()) {
64 return invalid_empty_symbol(operation);
65 }
66 try {
67 PropertyStore candidate = store;
68 update(candidate);
69 store = std::move(candidate);
70 } catch (const std::bad_alloc&) {
71 return PropertyStoreResult::failure(
72 CasErrc::ResourceLimit, "property-store allocation failed", operation);
73 } catch (const std::invalid_argument& ex) {
74 return PropertyStoreResult::failure(CasErrc::InvalidArgument, ex.what(), operation);
75 } catch (const std::exception& ex) {
76 return PropertyStoreResult::failure(CasErrc::InternalInvariant, ex.what(), operation);
77 }
78 return PropertyStoreResult::success();
79}

102int PropertyStore::domain_specificity(Domain domain) const {
103 switch (domain) {
104 case Domain::Complex: return 0;
105 case Domain::Real: return 1;
106 case Domain::Algebraic: return 2;
107 case Domain::Rational: return 3;
108 case Domain::Integer: return 4;
109 case Domain::Natural: return 5;
110 case Domain::PositiveInt: return 6;
111 }
112 return 0;

115std::vector<Domain> PropertyStore::get_ancestor_domains(Domain domain) const {
116 std::vector<Domain> ancestors;
117 switch (domain) {
118 case Domain::PositiveInt:
119 ancestors.push_back(Domain::Natural);
120 ancestors.push_back(Domain::Integer);
121 ancestors.push_back(Domain::Rational);
122 ancestors.push_back(Domain::Algebraic);
123 ancestors.push_back(Domain::Real);
124 ancestors.push_back(Domain::Complex);
125 break;
126 case Domain::Natural:
127 ancestors.push_back(Domain::Integer);
128 ancestors.push_back(Domain::Rational);
129 ancestors.push_back(Domain::Algebraic);
130 ancestors.push_back(Domain::Real);
131 ancestors.push_back(Domain::Complex);
132 break;
133 case Domain::Integer:
134 ancestors.push_back(Domain::Rational);
135 ancestors.push_back(Domain::Algebraic);
136 ancestors.push_back(Domain::Real);
137 ancestors.push_back(Domain::Complex);
138 break;
139 case Domain::Rational:
140 ancestors.push_back(Domain::Algebraic);
141 ancestors.push_back(Domain::Real);
142 ancestors.push_back(Domain::Complex);
143 break;
144 case Domain::Algebraic:
145 ancestors.push_back(Domain::Real);
146 ancestors.push_back(Domain::Complex);
147 break;
148 case Domain::Real:
149 ancestors.push_back(Domain::Complex);
150 break;
151 case Domain::Complex:
152 break;
153 }
154 return ancestors;
155}

159std::unordered_set<Sign, SignHash> PropertyStore::get_implied_signs(Sign sign) const {
160 std::unordered_set<Sign, SignHash> implied;
161 switch (sign) {
162 case Sign::Positive:
163 implied.insert(Sign::NonNegative);
164 implied.insert(Sign::NonZero);
165 break;
166 case Sign::Negative:
167 implied.insert(Sign::NonPositive);
168 implied.insert(Sign::NonZero);
169 break;
170 case Sign::Zero:
171 implied.insert(Sign::NonNegative);
172 implied.insert(Sign::NonPositive);
173 break;
174 case Sign::NonNegative:
175 case Sign::NonPositive:
176 case Sign::NonZero:
177 break;
178 }
179 return implied;
180}

182void PropertyStore::check_sign_contradiction(const std::string& symbol,
183 const std::unordered_set<Sign, SignHash>& existing,
184 Sign new_sign) {
185 auto contradicts = [](Sign a, Sign b) -> bool {
186 if (a == Sign::Positive && (b == Sign::Negative || b == Sign::Zero || b == Sign::NonPositive))
187 return true;
188 if (a == Sign::Negative && (b == Sign::Positive || b == Sign::Zero || b == Sign::NonNegative))
189 return true;
190 if (a == Sign::NonNegative && b == Sign::Negative)
191 return true;
192 if (a == Sign::NonPositive && b == Sign::Positive)
193 return true;
194 if (a == Sign::Zero && (b == Sign::Positive || b == Sign::Negative || b == Sign::NonZero))
195 return true;
196 if (a == Sign::NonZero && b == Sign::Zero)
197 return true;
198 return false;
199 };
200
201 for (Sign existing_sign : existing) {
202 if (contradicts(new_sign, existing_sign)) {
203 throw std::invalid_argument(
204 "Contradiction for symbol '" + symbol + "': cannot declare sign " +
205 sign_str(new_sign) + " because it conflicts with existing sign " +
206 sign_str(existing_sign));
207 }
208 }
209}

211void PropertyStore::check_domain_sign_consistency(const std::string& symbol,
212 const SymbolProperties& props,
213 Domain new_domain) {
214 // Natural is incompatible with Negative
215 if (new_domain == Domain::Natural || new_domain == Domain::PositiveInt) {
216 if (props.signs.count(Sign::Negative)) {
217 throw std::invalid_argument(
218 "Contradiction for symbol '" + symbol +
219 "': domain " + domain_str(new_domain) +
220 " is incompatible with existing sign Negative");
221 }
222 }
223
224 // PositiveInt is additionally incompatible with Zero and NonPositive
225 if (new_domain == Domain::PositiveInt) {
226 if (props.signs.count(Sign::Zero)) {
227 throw std::invalid_argument(
228 "Contradiction for symbol '" + symbol +
229 "': domain PositiveInt is incompatible with existing sign Zero");
230 }
231 if (props.signs.count(Sign::NonPositive)) {
232 throw std::invalid_argument(
233 "Contradiction for symbol '" + symbol +
234 "': domain PositiveInt is incompatible with existing sign NonPositive");
235 }
236 }
237}

239// declare_domain
240
241void PropertyStore::declare_domain_unchecked(const std::string& symbol, Domain domain) {
242 auto& props = properties_[symbol];
243
244 // Idempotent: same domain already set -> no-op
245 if (props.most_specific_domain == domain) {
246 return;
247 }
248
249 // Specificity preservation: if the new domain is an ancestor (less specific)
250 // of the current most-specific domain, it's a no-op.
251 if (domain_specificity(domain) < domain_specificity(props.most_specific_domain)) {
252 return;
253 }
254
255 // Transcendental guard: transcendental symbols cannot be Algebraic or more specific
256 if (props.transcendental && domain_specificity(domain) > domain_specificity(Domain::Real)) {
257 throw std::invalid_argument(
258 "Contradiction for symbol '" + symbol +
259 "': cannot declare domain " + domain_str(domain) +
260 " because symbol is Transcendental (existing domain: " +
261 domain_str(props.most_specific_domain) + ")");
262 }
263
264 // Check domain-sign cross-constraints before applying
265 check_domain_sign_consistency(symbol, props, domain);
266
267 // Set the new most-specific domain
268 props.most_specific_domain = domain;
269}

289void PropertyStore::declare_sign_unchecked(const std::string& symbol, Sign sign) {
290 auto& props = properties_[symbol];
291
292 // Idempotent: if sign already present, no-op
293 if (props.signs.count(sign)) {
294 return;
295 }
296
297 // Collect all signs to add (the declared sign + its implications)
298 std::unordered_set<Sign, SignHash> to_add;
299 to_add.insert(sign);
300 auto implied = get_implied_signs(sign);
301 to_add.insert(implied.begin(), implied.end());
302
303 // Check each new sign against existing signs for contradictions
304 for (Sign s : to_add) {
305 if (!props.signs.count(s)) {
306 check_sign_contradiction(symbol, props.signs, s);
307 }
308 }
309
310 // Also check domain-sign cross-constraints for the new signs
311 if (props.most_specific_domain == Domain::Natural ||
312 props.most_specific_domain == Domain::PositiveInt) {
313 for (Sign s : to_add) {
314 if (s == Sign::Negative) {
315 throw std::invalid_argument(
316 "Contradiction for symbol '" + symbol +
317 "': cannot declare sign Negative because it conflicts with domain " +
318 domain_str(props.most_specific_domain));
319 }
320 }
321 }
322 if (props.most_specific_domain == Domain::PositiveInt) {
323 for (Sign s : to_add) {
324 if (s == Sign::Zero) {
325 throw std::invalid_argument(
326 "Contradiction for symbol '" + symbol +
327 "': cannot declare sign Zero because it conflicts with domain PositiveInt");
328 }
329 if (s == Sign::NonPositive) {
330 throw std::invalid_argument(
331 "Contradiction for symbol '" + symbol +
332 "': cannot declare sign NonPositive because it conflicts with domain PositiveInt");
333 }
334 }
335 }
336
337 // Store all signs
338 props.signs.insert(to_add.begin(), to_add.end());
339
340 // Zero implies Integer domain
341 if (sign == Sign::Zero) {
342 if (domain_specificity(Domain::Integer) > domain_specificity(props.most_specific_domain)) {
343 props.most_specific_domain = Domain::Integer;
344 }
345 }
346}

366void PropertyStore::declare_parity_unchecked(const std::string& symbol, Parity parity) {
367 auto& props = properties_[symbol];
368
369 // Idempotent: same parity already set -> no-op
370 if (props.parity == parity) {
371 return;
372 }
373
374 // Unknown can always be set
375 if (parity == Parity::Unknown) {
376 props.parity = parity;
377 return;
378 }
379
380 // Contradiction: Even vs Odd
381 if (props.parity != Parity::Unknown && props.parity != parity) {
382 throw std::invalid_argument(
383 "Contradiction for symbol '" + symbol + "': parity conflict (Even vs Odd)");
384 }
385
386 // Auto-promote to Integer domain when Even or Odd is declared
387 if (domain_specificity(Domain::Integer) > domain_specificity(props.most_specific_domain)) {
388 props.most_specific_domain = Domain::Integer;
389 }
390
391 props.parity = parity;
392}

511bool PropertyStore::has_domain(const std::string& symbol, Domain domain) const {
512 auto it = properties_.find(symbol);
513 if (it == properties_.end()) {
514 // Default is Complex; only Complex query returns true
515 return domain == Domain::Complex;
516 }

519 return domain_specificity(it->second.most_specific_domain) >= domain_specificity(domain);

522// Interval overlap detection (private helper)
523
524static bool endpoints_equivalent(const Endpoint& left, const Endpoint& right) {
525 if (left.is_open != right.is_open ||
526 left.is_neg_infinity != right.is_neg_infinity ||
527 left.is_pos_infinity != right.is_pos_infinity) {
528 return false;
529 }
530 if (!left.value || !right.value) {
531 return left.value == right.value;
532 }
533 if (!lamina::detail::node(left.value) || !lamina::detail::node(right.value)) {
534 return lamina::detail::node(left.value) == lamina::detail::node(right.value);
535 }
536 return lamina::detail::node(left.value)->compare(*lamina::detail::node(right.value)) == 0;
537}
538
539static bool intervals_equivalent(const Interval& left, const Interval& right) {
540 return endpoints_equivalent(left.lower, right.lower) &&
541 endpoints_equivalent(left.upper, right.upper);
542}
543
544/**
545 * Helper: returns true if two intervals have a non-empty intersection.
546 * Endpoint ordering is delegated to checked interval operations; unsupported
547 * endpoints are treated conservatively as not proven to overlap.
548 */
549static Result<bool> intervals_overlap_checked_impl(
550 const Interval& a,
551 const Interval& b,
552 ComputationContext& context) {
553 auto left = IntervalUnion::from_intervals_checked({a}, context);
554 if (!left) return Result<bool>::failure(left.error());
555 auto right = IntervalUnion::from_intervals_checked({b}, context);
556 if (!right) return Result<bool>::failure(right.error());
557 auto intersection = left.value().intersect_checked(right.value(), context);
558 if (!intersection) return Result<bool>::failure(intersection.error());
559 return Result<bool>::success(!intersection.value().is_empty());
560}
561
562/**
563 * Helper: returns true if outer interval fully contains inner interval.
564 * outer covers inner iff outer.lower <= inner.lower AND inner.upper <= outer.upper
565 * (with appropriate open/closed boundary handling).
566 */
567static Result<bool> interval_covers_checked_impl(
568 const Interval& outer,
569 const Interval& inner,
570 ComputationContext& context) {
571 auto normalized_outer = IntervalUnion::from_intervals_checked({outer}, context);
572 if (!normalized_outer) {
573 return Result<bool>::failure(normalized_outer.error());
574 }
575 auto normalized_inner = IntervalUnion::from_intervals_checked({inner}, context);
576 if (!normalized_inner) {
577 return Result<bool>::failure(normalized_inner.error());
578 }
579 if (normalized_inner.value().is_empty()) return Result<bool>::success(true);
580 if (normalized_outer.value().is_empty()) return Result<bool>::success(false);
581
582 auto intersection = normalized_outer.value().intersect_checked(
583 normalized_inner.value(), context);
584 if (!intersection) {
585 return Result<bool>::failure(intersection.error());
586 }
587 const auto& intersection_intervals = intersection.value().intervals();
588 const auto& inner_intervals = normalized_inner.value().intervals();
589 if (intersection_intervals.size() != inner_intervals.size()) {
590 return Result<bool>::success(false);
591 }
592 for (std::size_t i = 0; i < inner_intervals.size(); ++i) {
593 if (!intervals_equivalent(intersection_intervals[i], inner_intervals[i])) {
594 return Result<bool>::success(false);
595 }
596 }
597 return Result<bool>::success(true);
598}

609PropertyStoreResult PropertyStore::declare_continuous_checked(
610 const std::string& symbol,
611 const Interval& interval,
612 ComputationContext& context) {
613 constexpr const char* operation = "declare_continuous";
614 if (symbol.empty()) return invalid_empty_symbol(operation);
615 auto valid = validate_property_interval(interval, context, operation);
616 if (!valid) return valid;
617
618 try {
619 PropertyStore candidate = *this;
620 auto& declarations = candidate.properties_[symbol].continuity_decls;
621 for (const auto& declaration : declarations) {
622 if (!declaration.is_differentiable &&
623 intervals_equivalent(declaration.interval, interval)) {
624 return PropertyStoreResult::success();
625 }
626 auto overlap = intervals_overlap_checked_impl(
627 declaration.interval, interval, context);
628 if (!overlap) {
629 return PropertyStoreResult::failure(
630 overlap.error().code, overlap.error().message, operation);
631 }
632 if (overlap.value() && declaration.is_differentiable) {
633 return PropertyStoreResult::failure(
634 CasErrc::InvalidArgument,
635 "Contradiction for symbol '" + symbol +
636 "': continuous-only declaration overlaps an existing "
637 "differentiable declaration",
638 operation);
639 }
640 }
641 declarations.push_back({interval, false});
642 *this = std::move(candidate);
643 } catch (const std::bad_alloc&) {
644 return PropertyStoreResult::failure(
645 CasErrc::ResourceLimit, "property-store allocation failed", operation);
646 } catch (const std::exception& ex) {
647 return PropertyStoreResult::failure(
648 CasErrc::InternalInvariant, ex.what(), operation);
649 }
650 return PropertyStoreResult::success();
651}

709Result<bool> PropertyStore::is_continuous_checked(
710 const std::string& symbol,
711 const Interval& interval,
712 ComputationContext& context) const {
713 constexpr const char* operation = "is_continuous";
714 if (symbol.empty()) {
715 return Result<bool>::failure(
716 CasErrc::InvalidArgument, "symbol name must not be empty", operation);
717 }
718 auto valid = validate_property_interval(interval, context, operation);
719 if (!valid) return Result<bool>::failure(valid.error());
720
721 auto it = properties_.find(symbol);
722 if (it == properties_.end()) {
723 return Result<bool>::success(false);
724 }
725
726 for (const auto& decl : it->second.continuity_decls) {
727 auto covers = interval_covers_checked_impl(decl.interval, interval, context);
728 if (!covers) {
729 return Result<bool>::failure(
730 covers.error().code, covers.error().message, operation);
731 }
732 if (covers.value()) return Result<bool>::success(true);
733 }
734 return Result<bool>::success(false);
735}

888void PropertyStore::declare_transcendental_unchecked(const std::string& symbol) {
889 auto& props = properties_[symbol];
890
891 // Idempotent: already transcendental → no-op
892 if (props.transcendental) {
893 return;
894 }
895
896 // Transcendental numbers are Real but NOT Algebraic (or any sub-domain).
897 // If the symbol already has a domain more specific than Real, that contradicts.
898 if (domain_specificity(props.most_specific_domain) > domain_specificity(Domain::Real)) {
899 throw std::invalid_argument(
900 "Contradiction for symbol '" + symbol +
901 "': cannot declare Transcendental because existing domain is " +
902 domain_str(props.most_specific_domain) +
903 " (Transcendental contradicts Algebraic or more specific domains)");
904 }
905
906 // Set domain to Real (if currently less specific)
907 if (domain_specificity(props.most_specific_domain) < domain_specificity(Domain::Real)) {
908 props.most_specific_domain = Domain::Real;
909 }
910
911 props.transcendental = true;
912}

939void PropertyStore::declare_finiteness_unchecked(const std::string& symbol, Finiteness f) {
940 auto& props = properties_[symbol];
941
942 // Idempotent: same value → no-op
943 if (props.finiteness == f) {
944 return;
945 }
946
947 // Unknown can always be set
948 if (f == Finiteness::Unknown) {
949 props.finiteness = f;
950 return;
951 }
952

954 if (props.finiteness != Finiteness::Unknown && props.finiteness != f) {

962 // Implication: Finite → Bounded
963 if (f == Finiteness::Finite) {
964 declare_bounded_unchecked(symbol, Boundedness::Bounded, std::nullopt);
965 }

992// Matrix definiteness
993
994void PropertyStore::declare_definiteness_unchecked(const std::string& symbol, Definiteness d) {
995 auto& props = properties_[symbol];
996
997 // Idempotent: same value → no-op
998 if (props.definiteness == d) {
999 return;
1000 }
1001
1002 // Unknown can always be set
1003 if (d == Definiteness::Unknown) {
1004 props.definiteness = d;
1005 return;
1006 }
1007
1008 // Check contradictions against existing definiteness
1009 if (props.definiteness != Definiteness::Unknown) {
1010 auto conflicts = [](Definiteness existing, Definiteness incoming) -> bool {
1011 // PositiveDefinite contradicts NegativeDefinite, NegativeSemiDefinite, Indefinite
1012 if (existing == Definiteness::PositiveDefinite &&
1013 (incoming == Definiteness::NegativeDefinite ||
1014 incoming == Definiteness::NegativeSemiDefinite ||
1015 incoming == Definiteness::Indefinite))
1016 return true;
1017 // NegativeDefinite contradicts PositiveDefinite, PositiveSemiDefinite, Indefinite
1018 if (existing == Definiteness::NegativeDefinite &&
1019 (incoming == Definiteness::PositiveDefinite ||
1020 incoming == Definiteness::PositiveSemiDefinite ||
1021 incoming == Definiteness::Indefinite))
1022 return true;
1023 // PositiveSemiDefinite contradicts NegativeDefinite, Indefinite
1024 if (existing == Definiteness::PositiveSemiDefinite &&
1025 (incoming == Definiteness::NegativeDefinite ||
1026 incoming == Definiteness::Indefinite))
1027 return true;
1028 // NegativeSemiDefinite contradicts PositiveDefinite, Indefinite
1029 if (existing == Definiteness::NegativeSemiDefinite &&
1030 (incoming == Definiteness::PositiveDefinite ||
1031 incoming == Definiteness::Indefinite))
1032 return true;
1033 // Indefinite contradicts PositiveDefinite, NegativeDefinite,
1034 // PositiveSemiDefinite, NegativeSemiDefinite
1035 if (existing == Definiteness::Indefinite &&
1036 (incoming == Definiteness::PositiveDefinite ||
1037 incoming == Definiteness::NegativeDefinite ||
1038 incoming == Definiteness::PositiveSemiDefinite ||
1039 incoming == Definiteness::NegativeSemiDefinite))
1040 return true;
1041 return false;
1042 };
1043
1044 if (conflicts(props.definiteness, d)) {
1045 throw std::invalid_argument(
1046 "Contradiction for symbol '" + symbol +
1047 "': definiteness conflict between existing and new declaration");
1048 }
1049 if (props.definiteness == Definiteness::PositiveSemiDefinite &&
1050 d == Definiteness::PositiveDefinite) {
1051 props.definiteness = d;
1052 return;
1053 }
1054 if (props.definiteness == Definiteness::NegativeSemiDefinite &&
1055 d == Definiteness::NegativeDefinite) {
1056 props.definiteness = d;
1057 return;
1058 }
1059 if (props.definiteness == Definiteness::PositiveDefinite &&
1060 d == Definiteness::PositiveSemiDefinite) {
1061 return; // already stronger
1062 }
1063 if (props.definiteness == Definiteness::NegativeDefinite &&
1064 d == Definiteness::NegativeSemiDefinite) {
1065 return; // already stronger
1066 }
1067 }
1068
1069 props.definiteness = d;
1070}
