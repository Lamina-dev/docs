34Result<void> collect_polynomial_variables(
35 const std::shared_ptr<const SymbolicNode>& node,
36 std::set<std::string>& variables,
37 ComputationContext& context) {
38 auto entered = context.enter_recursion(kOperation);
39 if (!entered) return entered;
40 RecursionFrame frame(context);
41
42 if (!node) {
43 return Result<void>::failure(
44 CasErrc::InternalInvariant, "polynomial contains a null node",
45 kOperation);
46 }
47 if (auto number = std::dynamic_pointer_cast<const NumberNode>(node)) {
48 if (std::holds_alternative<lmmc_real_t>(number->value())) {
49 return Result<void>::failure(
50 CasErrc::UnsupportedExpression,
51 "approximate coefficients are not supported by exact GCD",
52 kOperation);
53 }
54 return Result<void>::success();
55 }
56 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(node)) {
57 variables.insert(variable->name());
58 return Result<void>::success();
59 }
60 if (auto addition = std::dynamic_pointer_cast<const AddNode>(node)) {
61 for (const auto& operand : addition->operands()) {
62 auto result = collect_polynomial_variables(operand, variables, context);
63 if (!result) return result;
64 }
65 return Result<void>::success();
66 }
67 if (auto product = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
68 for (const auto& operand : product->operands()) {
69 auto result = collect_polynomial_variables(operand, variables, context);
70 if (!result) return result;
71 }
72 return Result<void>::success();
73 }
74 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
75 auto exponent = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
76 if (!exponent || std::holds_alternative<lmmc_real_t>(exponent->value())) {
77 return Result<void>::failure(
78 CasErrc::UnsupportedExpression,
79 "polynomial exponents must be exact nonnegative integers",
80 kOperation);
81 }
82 return collect_polynomial_variables(power->base(), variables, context);
83 }
84 return Result<void>::failure(
85 CasErrc::UnsupportedExpression,
86 "expression is outside the exact rational polynomial domain",
87 kOperation);
88}

90Result<std::size_t> polynomial_exponent(
91 const std::shared_ptr<const SymbolicNode>& node) {
92 auto number = std::dynamic_pointer_cast<const NumberNode>(node);
93 if (!number || std::holds_alternative<lmmc_real_t>(number->value())) {
94 return Result<std::size_t>::failure(
95 CasErrc::UnsupportedExpression,
96 "polynomial exponents must be exact nonnegative integers",
97 kOperation);
98 }
99
100 BigInt integer;
101 if (std::holds_alternative<BigInt>(number->value())) {
102 integer = std::get<BigInt>(number->value());
103 } else {
104 const Rational& rational = std::get<Rational>(number->value());
105 if (!rational.is_integer()) {
106 return Result<std::size_t>::failure(
107 CasErrc::UnsupportedExpression,
108 "polynomial exponents must be integers", kOperation);
109 }
110 integer = rational.to_BigInt();
111 }
112
113 const auto value = integer.try_to_int64();
114 if (!value || *value < 0) {
115 return Result<std::size_t>::failure(
116 CasErrc::UnsupportedExpression,
117 "polynomial exponents must be representable nonnegative integers",
118 kOperation);
119 }
120 return Result<std::size_t>::success(static_cast<std::size_t>(*value));
121}

123Result<void> check_term_budget(std::size_t lhs_terms,
124 std::size_t rhs_terms,
125 ComputationContext& context) {
126 const std::size_t limit = context.limits().max_expansion_terms;
127 if (lhs_terms != 0 && rhs_terms > limit / lhs_terms) {
128 return Result<void>::failure(
129 CasErrc::ResourceLimit,
130 "polynomial expansion term budget exhausted", kOperation);
131 }
132 return context.consume_steps(lhs_terms * rhs_terms + 1, kOperation);
133}

135Result<MultiPoly> multiply_checked(const MultiPoly& lhs,
136 const MultiPoly& rhs,
137 ComputationContext& context) {
138 auto budget = check_term_budget(
139 static_cast<std::size_t>(lhs.num_terms()),
140 static_cast<std::size_t>(rhs.num_terms()), context);
141 if (!budget) return Result<MultiPoly>::failure(budget.error());
142 MultiPoly result = lhs * rhs;
143 if (static_cast<std::size_t>(result.num_terms()) >
144 context.limits().max_expansion_terms) {
145 return Result<MultiPoly>::failure(
146 CasErrc::ResourceLimit,
147 "polynomial expansion term budget exhausted", kOperation);
148 }
149 return Result<MultiPoly>::success(std::move(result));

152Result<MultiPoly> node_to_multivariate_polynomial(
153 const std::shared_ptr<const SymbolicNode>& node,
154 const std::vector<std::string>& variables,
155 ComputationContext& context) {
156 auto entered = context.enter_recursion(kOperation);
157 if (!entered) return Result<MultiPoly>::failure(entered.error());
158 RecursionFrame frame(context);
159
160 if (auto number = std::dynamic_pointer_cast<const NumberNode>(node)) {
161 if (std::holds_alternative<BigInt>(number->value())) {
162 return Result<MultiPoly>::success(MultiPoly(
163 Rational(std::get<BigInt>(number->value())), variables));
164 }
165 if (std::holds_alternative<Rational>(number->value())) {
166 return Result<MultiPoly>::success(MultiPoly(
167 std::get<Rational>(number->value()), variables));
168 }
169 return Result<MultiPoly>::failure(
170 CasErrc::UnsupportedExpression,
171 "approximate coefficients are not supported by exact GCD",
172 kOperation);
173 }
174 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(node)) {
175 Monomial monomial(variables.size(), 0);
176 auto position = std::find(variables.begin(), variables.end(),
177 variable->name());
178 if (position == variables.end()) {
179 return Result<MultiPoly>::failure(
180 CasErrc::InternalInvariant,
181 "polynomial variable collection is inconsistent", kOperation);
182 }
183 monomial[static_cast<std::size_t>(position - variables.begin())] = 1;
184 return Result<MultiPoly>::success(MultiPoly(
185 {{std::move(monomial), Rational(1)}}, variables));
186 }
187 if (auto addition = std::dynamic_pointer_cast<const AddNode>(node)) {
188 MultiPoly result(Rational(0), variables);
189 for (const auto& operand : addition->operands()) {
190 auto converted = node_to_multivariate_polynomial(
191 operand, variables, context);
192 if (!converted) return converted;
193 auto step = context.consume_steps(
194 static_cast<std::size_t>(result.num_terms() +
195 converted.value().num_terms()) + 1,
196 kOperation);
197 if (!step) return Result<MultiPoly>::failure(step.error());
198 result = result + converted.value();
199 if (static_cast<std::size_t>(result.num_terms()) >
200 context.limits().max_expansion_terms) {
201 return Result<MultiPoly>::failure(
202 CasErrc::ResourceLimit,
203 "polynomial expansion term budget exhausted", kOperation);
204 }
205 }
206 return Result<MultiPoly>::success(std::move(result));
207 }
208 if (auto product = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
209 MultiPoly result(Rational(1), variables);
210 for (const auto& operand : product->operands()) {
211 auto converted = node_to_multivariate_polynomial(
212 operand, variables, context);
213 if (!converted) return converted;
214 auto multiplied = multiply_checked(result, converted.value(), context);
215 if (!multiplied) return multiplied;
216 result = std::move(multiplied.value());
217 }
218 return Result<MultiPoly>::success(std::move(result));
219 }
220 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
221 auto exponent = polynomial_exponent(power->exponent());
222 if (!exponent) return Result<MultiPoly>::failure(exponent.error());
223 auto base = node_to_multivariate_polynomial(
224 power->base(), variables, context);
225 if (!base) return base;
226
227 MultiPoly result(Rational(1), variables);
228 MultiPoly factor = std::move(base.value());
229 std::size_t remaining = exponent.value();
230 while (remaining != 0) {
231 if ((remaining & 1U) != 0) {
232 auto multiplied = multiply_checked(result, factor, context);
233 if (!multiplied) return multiplied;
234 result = std::move(multiplied.value());
235 }
236 remaining >>= 1U;
237 if (remaining != 0) {
238 auto squared = multiply_checked(factor, factor, context);
239 if (!squared) return squared;
240 factor = std::move(squared.value());
241 }
242 }
243 return Result<MultiPoly>::success(std::move(result));
244 }
245 return Result<MultiPoly>::failure(
246 CasErrc::UnsupportedExpression,
247 "expression is outside the exact rational polynomial domain",
248 kOperation);
249}

251MultiPoly make_monic(const MultiPoly& polynomial) {
252 if (polynomial.is_zero()) return polynomial;
253 const Rational leading = polynomial.terms().front().second;
254 return polynomial * (Rational(1) / leading);

257Result<std::shared_ptr<SymbolicExpr>> multivariate_polynomial_to_expression(
258 const MultiPoly& polynomial,
259 ComputationContext& context) {
260 if (polynomial.is_zero()) {
261 auto nodes = context.reserve_nodes(1, kOperation);
262 if (!nodes) return Result<std::shared_ptr<SymbolicExpr>>::failure(nodes.error());
263 return Result<std::shared_ptr<SymbolicExpr>>::success(
264 SymbolicExpr::number(0));
265 }
266
267 std::vector<std::shared_ptr<SymbolicExpr>> terms;
268 terms.reserve(polynomial.terms().size());
269 for (const auto& [monomial, coefficient] : polynomial.terms()) {
270 auto step = context.consume_steps(1, kOperation);
271 if (!step) return Result<std::shared_ptr<SymbolicExpr>>::failure(step.error());
272
273 std::vector<std::shared_ptr<SymbolicExpr>> factors;
274 const bool constant_term = total_degree(monomial) == 0;
275 if (coefficient != Rational(1) || constant_term) {
276 factors.push_back(SymbolicExpr::number(coefficient));
277 }
278 for (std::size_t index = 0;
279 index < polynomial.variables().size() && index < monomial.size();
280 ++index) {
281 if (monomial[index] == 0) continue;
282 auto variable = SymbolicExpr::variable(polynomial.variables()[index]);
283 factors.push_back(monomial[index] == 1
284 ? variable
285 : SymbolicExpr::power(
286 variable, SymbolicExpr::number(monomial[index])));
287 }
288
289 auto reserve = context.reserve_nodes(factors.size() * 2 + 1, kOperation);
290 if (!reserve) {
291 return Result<std::shared_ptr<SymbolicExpr>>::failure(reserve.error());
292 }
293 std::shared_ptr<SymbolicExpr> term = factors.empty()
294 ? SymbolicExpr::number(1)
295 : factors.front();
296 for (std::size_t index = 1; index < factors.size(); ++index) {
297 term = SymbolicExpr::multiply(term, factors[index]);
298 }
299 terms.push_back(std::move(term));
300 }
301
302 auto reserve = context.reserve_nodes(terms.size(), kOperation);
303 if (!reserve) {
304 return Result<std::shared_ptr<SymbolicExpr>>::failure(reserve.error());
305 }
306 auto expression = terms.front();
307 for (std::size_t index = 1; index < terms.size(); ++index) {
308 expression = SymbolicExpr::add(expression, terms[index]);
309 }
310 return Result<std::shared_ptr<SymbolicExpr>>::success(std::move(expression));
311}

315SymbolicGcdResult symbolic_polynomial_gcd(
316 const SymbolicExpr& lhs,
317 const SymbolicExpr& rhs,
318 ComputationContext& context) {
319 try {
320 std::set<std::string> variable_set;
321 auto lhs_variables = collect_polynomial_variables(
322 detail::node(lhs), variable_set, context);
323 if (!lhs_variables) return SymbolicGcdResult::failure(lhs_variables.error());
324 auto rhs_variables = collect_polynomial_variables(
325 detail::node(rhs), variable_set, context);
326 if (!rhs_variables) return SymbolicGcdResult::failure(rhs_variables.error());
327 std::vector<std::string> variables(variable_set.begin(), variable_set.end());
328
329 auto lhs_polynomial = node_to_multivariate_polynomial(
330 detail::node(lhs), variables, context);
331 if (!lhs_polynomial) return SymbolicGcdResult::failure(lhs_polynomial.error());
332 auto rhs_polynomial = node_to_multivariate_polynomial(
333 detail::node(rhs), variables, context);
334 if (!rhs_polynomial) return SymbolicGcdResult::failure(rhs_polynomial.error());
335
336 auto step = context.consume_steps(
337 static_cast<std::size_t>(lhs_polynomial.value().num_terms() +
338 rhs_polynomial.value().num_terms()) + 1,
339 kOperation);
340 if (!step) return SymbolicGcdResult::failure(step.error());
341
342 MultiPoly gcd = make_monic(multivariate_gcd(
343 lhs_polynomial.value(), rhs_polynomial.value()));
344 if (!gcd.is_zero()) {
345 try {
346 MultiPoly lhs_quotient = lhs_polynomial.value().exact_div(gcd);
347 MultiPoly rhs_quotient = rhs_polynomial.value().exact_div(gcd);
348 MultiPoly residual = make_monic(multivariate_gcd(
349 lhs_quotient, rhs_quotient));
350 if (!residual.is_constant()) {
351 return SymbolicGcdResult::failure(
352 CasErrc::Inconclusive,
353 "GCD candidate is not maximal", kOperation);
354 }
355 } catch (const std::exception&) {
356 return SymbolicGcdResult::failure(
357 CasErrc::Inconclusive,
358 "GCD candidate failed exact divisibility verification",
359 kOperation);
360 }
361 }
362 return multivariate_polynomial_to_expression(gcd, context);
363 } catch (const std::bad_alloc&) {
364 return SymbolicGcdResult::failure(
365 CasErrc::ResourceLimit, "symbolic GCD allocation failed", kOperation);
366 } catch (const std::exception& error) {
367 return SymbolicGcdResult::failure(
368 CasErrc::Inconclusive, error.what(), kOperation);
369 }
370}
