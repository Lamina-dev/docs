59 auto f_prime_sq = SymbolicExpr::power(f_prime, SymbolicExpr::number(2));

61 auto arc_factor = SymbolicExpr::sqrt(one_plus_fp_sq);
62 auto abs_f = calculus_utils_make_abs(f);
63 if (!abs_f || !LMCAS::detail::node(abs_f)) {
64 return ExpressionResult::failure(
65 CasErrc::InternalInvariant,
66 "absolute value node construction failed",
67 operation);
68 }
69 auto integrand = SymbolicExpr::multiply(abs_f, arc_factor);
70
71 auto integral = calculus_utils_try_symbolic_definite(
72 integrand, var, a, b, context);
73 if (!integral) {
74 return integral;
75 }
76 if (!integral.value() ||
77 !LMCAS::detail::node(integral.value())) {
78 return ExpressionResult::failure(
79 CasErrc::Inconclusive,
80 "surface area integral could not be evaluated exactly",
81 operation);
82 }

85 SymbolicExpr::number(2), LMCAS::detail::make_expression_ptr(
86 LMCAS::detail::make_node<VariableNode>("pi", true)));

88 two_pi, std::move(integral.value()));
