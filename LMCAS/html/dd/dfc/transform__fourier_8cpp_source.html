21static bool te_is_abs_variable(const std::shared_ptr<const SymbolicNode>& node,
22 const std::string& variable) {
23 auto function = std::dynamic_pointer_cast<const FunctionNode>(node);
24 if (!function || function->type() != FunctionNode::FuncType::Abs ||
25 function->arguments().size() != 1) return false;
26 auto argument = std::dynamic_pointer_cast<const VariableNode>(function->arguments()[0]);
27 return argument && !argument->is_constant() && argument->name() == variable;

29static std::shared_ptr<SymbolicExpr> te_abs_decay(
30 const std::shared_ptr<SymbolicExpr>& f, const std::string& t) {
31 auto function = std::dynamic_pointer_cast<const FunctionNode>(detail::node(f));
32 if (!function || function->type() != FunctionNode::FuncType::Exp ||
33 function->arguments().size() != 1) return nullptr;
34 auto argument = detail::make_expression_ptr(function->arguments()[0]);
35 auto parts = te_split_coeff(argument, t);

37 return detail::make_expression_ptr(SymbolicFactory::create_multiply({
38 SymbolicFactory::create_number(BigInt(-1)), detail::node(parts.first)}))->simplify();

41static std::shared_ptr<SymbolicExpr> te_decay_condition(
42 const std::shared_ptr<SymbolicExpr>& rate) {
43 auto product = std::dynamic_pointer_cast<const MultiplyNode>(detail::node(rate));
44 if (!product) return te_gt_condition(rate, SymbolicExpr::number(0));
45 std::vector<std::shared_ptr<const SymbolicNode>> factors;
46 for (const auto& factor : product->operands()) {
47 auto number = std::dynamic_pointer_cast<const NumberNode>(factor);
48 if (!number || !number->is_positive()) factors.push_back(factor);
49 }
50 auto lhs = detail::make_expression_ptr(SymbolicFactory::create_multiply(std::move(factors)));
51 return te_gt_condition(lhs, SymbolicExpr::number(0));
52}

54static bool te_accept_decay(const std::shared_ptr<SymbolicExpr>& rate,
55 const AssumptionContext& assumptions,
56 std::vector<std::shared_ptr<SymbolicExpr>>& conditions,
57 ComputationContext& context) {
58 InferenceEngine inference(assumptions);
59 auto real = inference.query_real_checked(*rate, context);
60 if (!real) {
61 if (real.error().code == CasErrc::DomainError) return false;
62 throw real.error();
63 }
64 if (real.value() != Tribool::True) {
65 return false;
66 }
67 auto positive = inference.query_positive_checked(*rate, context);
68 if (!positive) {
69 throw positive.error();
70 }
71 if (positive.value() == Tribool::True) {
72 return true;
73 }
74 auto nonpositive = inference.query_nonpositive_checked(*rate, context);
75 if (!nonpositive) {
76 throw nonpositive.error();
77 }
78 if (positive.value() == Tribool::False || nonpositive.value() == Tribool::True) {
79 return false;
80 }
81 conditions.push_back(te_decay_condition(rate));
82 return true;
83}

85static std::shared_ptr<SymbolicExpr> te_abs_fourier(
86 const std::shared_ptr<SymbolicExpr>& rate, const std::string& omega) {
87 auto two = SymbolicFactory::create_number(BigInt(2));
88 auto denominator = SymbolicFactory::create_add({
89 SymbolicFactory::create_power(detail::node(rate), two),
90 SymbolicFactory::create_power(SymbolicFactory::create_variable(omega), two)});
91 auto reciprocal = SymbolicFactory::create_power(
92 denominator, SymbolicFactory::create_number(BigInt(-1)));
93 return detail::make_expression_ptr(SymbolicFactory::create_multiply({
94 two, detail::node(rate), reciprocal}))->simplify();
95}

126static std::shared_ptr<SymbolicExpr> fourier_transform_core(
127 const std::shared_ptr<SymbolicExpr>& f, const std::string& t,
128 const std::string& omega, ComputationContext& context,
129 const AssumptionContext& assumptions,
130 std::vector<std::shared_ptr<SymbolicExpr>>& conditions) {
131 auto step = context.consume_steps(1, "fourier_transform.recursive");
132 if (!step) {
133 return nullptr;
134 }
135 if (!f || !detail::node(f)) {
136 return nullptr;
137 }
138 if (!te_depends_on(f, t)) {
139 return te_fourier_unevaluated(f, t, omega);
140 }
141 auto addition = std::dynamic_pointer_cast<const AddNode>(detail::node(f));
142 if (addition) {
143 return te_fourier_sum(*addition, t, omega, context, assumptions, conditions);
144 }
145 auto [coefficient, body] = te_split_coeff(f, t);
146 if (!coefficient->is_one()) {
147 auto transformed = fourier_transform_core(body, t, omega, context, assumptions, conditions);
148 if (transformed) {
149 return SymbolicExpr::multiply(coefficient, transformed);
150 }
151 }
152 if (const auto gaussian_rate = te_match_pure_gaussian(f, t)) {
153 return te_gaussian_fourier(*gaussian_rate, omega);
154 }
155 auto decay = te_abs_decay(f, t);
156 if (decay) {
157 if (te_accept_decay(decay, assumptions, conditions, context)) {
158 return te_abs_fourier(decay, omega);
159 }
160 return te_fourier_unevaluated(f, t, omega);
161 }
162 return te_fourier_unevaluated(f, t, omega);
163}

207static std::shared_ptr<SymbolicExpr> inverse_fourier_transform_core(
208 const std::shared_ptr<SymbolicExpr>& F, const std::string& omega,
209 const std::string& t, ComputationContext& context) {
210 auto step = context.consume_steps(1, "inverse_fourier_transform.recursive");
211 if (!step) return nullptr;
212 if (!F || !LMCAS::detail::node(F)) return nullptr;
213 auto tv = SymbolicExpr::variable(t);
214 if (!te_depends_on(F, omega)) {
215 return LMCAS::detail::make_expression_ptr(LMCAS::detail::make_node<TransformNode>(
216 TransformNode::TransformType::InverseFourier, LMCAS::detail::node(F)->clone(), omega,
217 SymbolicFactory::create_variable(t)));
218 }
219 auto add_node = std::dynamic_pointer_cast<const AddNode>(LMCAS::detail::node(F));
220 if (add_node) {
221 std::shared_ptr<SymbolicExpr> result;
222 for (auto& op : add_node->operands()) {
223 auto ift = inverse_fourier_transform_core(
224 LMCAS::detail::make_expression_ptr(op), omega, t, context);
225 if (!ift) {
226 return LMCAS::detail::make_expression_ptr(LMCAS::detail::make_node<TransformNode>(
227 TransformNode::TransformType::InverseFourier, LMCAS::detail::node(F)->clone(), omega,
228 SymbolicFactory::create_variable(t)));
229 }
230 result = result ? SymbolicExpr::add(result, ift) : ift;
231 }
232 return result;
233 }
234
235 if (!contains_inexact_number(detail::node(F))) {
236 auto canonical = te_abs_fourier(SymbolicExpr::number(1), omega);
237 if (detail::node(F->simplify())->equals(*detail::node(canonical))) {
238 auto abs_t = detail::make_expression_ptr(detail::make_node<FunctionNode>(
239 FunctionNode::FuncType::Abs,
240 std::vector<std::shared_ptr<const SymbolicNode>>{detail::node(tv)}));
241 return SymbolicExpr::exp(SymbolicExpr::multiply(
242 SymbolicExpr::number(-1), abs_t));
243 }
244 }
245
246 if (const auto rate = te_match_pure_gaussian(F, omega)) {
247 const double coefficient =
248 1.0 / std::sqrt(4.0 * 3.14159265358979323846 * *rate);
249 auto exponent = SymbolicExpr::multiply(
250 SymbolicExpr::number(-1.0 / (4.0 * *rate)),
251 SymbolicExpr::power(tv, SymbolicExpr::number(2)));
252 return SymbolicExpr::multiply(
253 SymbolicExpr::number(coefficient), SymbolicExpr::exp(exponent));
254 }
255 return LMCAS::detail::make_expression_ptr(LMCAS::detail::make_node<TransformNode>(
256 TransformNode::TransformType::InverseFourier, LMCAS::detail::node(F)->clone(), omega,
257 SymbolicFactory::create_variable(t)));
258}

260TransformEngineResult inverse_fourier_transform_checked(
261 const std::shared_ptr<SymbolicExpr>& F,
262 const std::string& omega,
263 const std::string& t,
264 ComputationContext& context) {
265 const std::string operation = "inverse_fourier_transform";
266 auto valid = te_validate_expr_vars(F, omega, t, context, operation);
267 if (!valid) return TransformEngineResult::failure(valid.error());
268 auto step = context.consume_steps(8, operation);
269 if (!step) return TransformEngineResult::failure(step.error());
270 try {
271 auto expression = inverse_fourier_transform_core(
272 F, omega, t, context);
273 auto final_access = context.consume_steps(0, operation);
274 if (!final_access) {
275 return TransformEngineResult::failure(final_access.error());
276 }
277 if (contains_inexact_number(detail::node(expression))) {
278 return TransformEngineResult::failure(
279 CasErrc::Inconclusive,
280 "inverse Fourier round trip requires an exact candidate",
281 operation);
282 }
283 auto result = te_wrap_transform_result(expression, operation);
284 if (!result) return result;
285 AssumptionContext empty_assumptions;
286 const auto& assumptions = context.assumptions() ? *context.assumptions() : empty_assumptions;
287 std::vector<std::shared_ptr<SymbolicExpr>> conditions;
288 auto round_trip = fourier_transform_core(
289 expression, t, omega, context, assumptions, conditions);
290 if (!round_trip || !conditions.empty() ||
291 te_contains_transform(LMCAS::detail::node(round_trip))) {
292 return TransformEngineResult::failure(
293 CasErrc::Inconclusive,
294 "inverse Fourier round trip is not proved", operation);
295 }
296 auto verified = check_equivalent(round_trip, F, context);
297 if (!verified) return TransformEngineResult::failure(verified.error());
298 if (!std::holds_alternative<ProvedZeroResidual>(verified.value())) {
299 return TransformEngineResult::failure(
300 CasErrc::Inconclusive,
301 "inverse Fourier round trip is not proved", operation);
302 }
303 result.value().certificate =
304 ExactRoundTripProof{SymbolicExpr::number(0)};
305 return result;
306 } catch (const CasError& error) {
307 return TransformEngineResult::failure(error);
308 } catch (const std::bad_alloc&) {
309 return TransformEngineResult::failure(CasErrc::ResourceLimit,
310 "transform allocation failed",
311 operation);
312 } catch (const std::exception& e) {
313 return TransformEngineResult::failure(CasErrc::InternalInvariant,
314 e.what(), operation);
315 }
316}
