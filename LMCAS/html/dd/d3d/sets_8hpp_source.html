142class FiniteSetNode : public SymbolicNode {
143private:
144 LMCAS_AST_NODE_FACTORY_FRIEND;
145 const std::vector<std::shared_ptr<const SymbolicNode>> elements_;
146
147 explicit FiniteSetNode(std::vector<std::shared_ptr<const SymbolicNode>> elements)
148 : elements_(canonicalize(std::move(elements))) {}
149
150 static std::vector<std::shared_ptr<const SymbolicNode>> canonicalize(
151 std::vector<std::shared_ptr<const SymbolicNode>> elements) {
152 for (const auto& element : elements) {
153 if (!element) throw std::invalid_argument("finite set elements cannot be null");
154 }
155 std::sort(elements.begin(), elements.end(),
156 [](const auto& lhs, const auto& rhs) { return lhs->compare(*rhs) < 0; });
157 elements.erase(std::unique(elements.begin(), elements.end(),
158 [](const auto& lhs, const auto& rhs) {
159 return lhs->equals(*rhs);
160 }),
161 elements.end());
162 return elements;
163 }
164
165public:
166 const std::vector<std::shared_ptr<const SymbolicNode>>& elements() const noexcept {
167 return elements_;
168 }
169 bool contains(const SymbolicNode& element) const {
170 return std::any_of(elements_.begin(), elements_.end(),
171 [&](const auto& candidate) { return candidate->equals(element); });
172 }
173 int type_priority() const override { return 104; }
174
175protected:
176 std::size_t compute_hash() const override {
177 std::size_t seed = static_cast<std::size_t>(type_priority());
178 for (const auto& element : elements_) hash_combine(seed, element->hash());
179 return seed;
180 }
181 int compare_same_type(const SymbolicNode& other) const override {
182 const auto& set = static_cast<const FiniteSetNode&>(other);
183 const auto count = std::min(elements_.size(), set.elements_.size());
184 for (std::size_t index = 0; index < count; ++index) {
185 const int comparison = elements_[index]->compare(*set.elements_[index]);
186 if (comparison != 0) return comparison;
187 }
188 if (elements_.size() == set.elements_.size()) return 0;
189 return elements_.size() < set.elements_.size() ? -1 : 1;
190 }
191
192public:
193 void accept(LMCAS::detail::SymbolicVisitor& visitor) const override {
194 LMCAS::detail::SymbolicVisitor::DepthGuard guard(visitor);
195 visitor.visit(*this);
196 }
197 std::shared_ptr<const SymbolicNode> clone() const override {
198 std::vector<std::shared_ptr<const SymbolicNode>> elements;
199 elements.reserve(elements_.size());
200 for (const auto& element : elements_) elements.push_back(element->clone());
201 return LMCAS::detail::make_node<FiniteSetNode>(std::move(elements));
202 }
