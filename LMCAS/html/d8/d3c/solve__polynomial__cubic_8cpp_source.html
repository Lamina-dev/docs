20 if (!b->simplify()->get_number_value_is_zero()) {

36static std::vector<std::shared_ptr<SymbolicExpr>> symbolic_cubic_roots(
37 const std::shared_ptr<SymbolicExpr>& a, const std::shared_ptr<SymbolicExpr>& b,
38 const std::shared_ptr<SymbolicExpr>& c, const std::shared_ptr<SymbolicExpr>& d) {
39 auto a2 = SymbolicExpr::power(a, num(2));
40 auto a3 = SymbolicExpr::power(a, num(3));
41 auto b2 = SymbolicExpr::power(b, num(2));
42 auto b3 = SymbolicExpr::power(b, num(3));
43 auto three_ac = SymbolicExpr::multiply(
44 num(3), SymbolicExpr::multiply(a, c));
45 auto p_num = sub(three_ac, b2);
46 auto p_den = SymbolicExpr::multiply(num(3), a2);
47 auto p = SymbolicExpr::divide(p_num, p_den)->simplify();
48 auto two_b3 = SymbolicExpr::multiply(num(2), b3);
49 auto nine_abc = SymbolicExpr::multiply(
50 num(9), SymbolicExpr::multiply(
51 a, SymbolicExpr::multiply(b, c)));
52 auto twentyseven_a2d = SymbolicExpr::multiply(
53 num(27), SymbolicExpr::multiply(a2, d));
54 auto q_num = SymbolicExpr::add(
55 sub(two_b3, nine_abc), twentyseven_a2d);
56 auto q_den = SymbolicExpr::multiply(num(27), a3);
57 auto q = SymbolicExpr::divide(q_num, q_den)->simplify();
58 auto shift = SymbolicExpr::divide(
59 b, SymbolicExpr::multiply(num(3), a))->simplify();
60
61
62 auto q_half = SymbolicExpr::divide(q, num(2));
63 auto p_third = SymbolicExpr::divide(p, num(3));
64 auto D_expr = SymbolicExpr::add(
65 SymbolicExpr::power(q_half, num(2)),
66 SymbolicExpr::power(p_third, num(3)))->simplify();
67
68 auto neg_q_half = negate(q_half)->simplify();
69 auto sqrt_D = SymbolicExpr::sqrt(D_expr);
70
71 auto u = cbrt_expr(SymbolicExpr::add(neg_q_half, sqrt_D))->simplify();
72 auto v = cbrt_expr(sub(neg_q_half, sqrt_D))->simplify();
73
74 auto t1 = SymbolicExpr::add(u, v);
75 auto x1 = sub(t1, shift)->simplify();
76
77 auto neg_half_sum = SymbolicExpr::divide(negate(SymbolicExpr::add(u, v)), num(2));
78 auto diff_uv = sub(u, v);
79 auto sqrt3_half = SymbolicExpr::divide(SymbolicExpr::sqrt(num(3)), num(2));
80 auto imag_coeff = SymbolicExpr::multiply(sqrt3_half, diff_uv);
81
82 auto i_unit = SymbolicExpr::sqrt(num(-1));
83 auto x2 = sub(SymbolicExpr::add(neg_half_sum, SymbolicExpr::multiply(i_unit, imag_coeff)), shift)->simplify();
84 auto x3 = sub(sub(neg_half_sum, SymbolicExpr::multiply(i_unit, imag_coeff)), shift)->simplify();
85
86 return {x1, x2, x3};
87}

89bool polynomial_solver_detail::bounded_rational_root_search(
90 const Polynomial<Rational>& polynomial) {
91 if (polynomial.degree() <= 1) return true;
92 if (polynomial.lead_coeff() != Rational(1)) {
93 return bounded_rational_root_search(polynomial.make_monic());
94 }
95 BigInt denominator_lcm(1);
96 for (const auto& coefficient : polynomial.coeffs) {
97 denominator_lcm = BigInt::lcm(
98 denominator_lcm, coefficient.get_denominator());
99 if (denominator_lcm.bit_length() > 32) return false;
100 }
101 /**
102 * @brief find_rational_roots 在枚举因子前移除零根。
103 * 大小限制针对本原整数多项式的实际首尾系数；
104 * 各分母虽小，其最小公倍数仍可能很大。
105 */
106 for (const auto& coefficient : polynomial.coeffs) {
107 if (coefficient != Rational(0)) {
108 return (coefficient * Rational(denominator_lcm))
109 .get_numerator().bit_length() <= 32;
110 }
111 }
112 return true;
113}

115static std::vector<std::shared_ptr<SymbolicExpr>> exact_factor_roots(
116 Polynomial<Rational> polynomial) {
117 std::vector<std::shared_ptr<SymbolicExpr>> roots;
118 if (polynomial.degree() == 3 && polynomial.coeffs[0] == Rational(0)) {
119 roots.push_back(SymbolicExpr::number(0));
120 polynomial.coeffs.erase(polynomial.coeffs.begin());
121 }
122 if (polynomial.degree() == 3) {
123 if (bounded_rational_root_search(polynomial)) {
124 for (const auto& root : find_rational_roots(polynomial)) {
125 Polynomial<Rational> linear(
126 {-root, Rational(1)}, polynomial.variable_name);
127 auto [quotient, remainder] = polynomial.div_mod(linear);
128 if (!remainder.is_zero()) {
129 throw std::logic_error("solve_cubic: inexact rational root division");
130 }
131 roots.push_back(SymbolicExpr::number(root));
132 polynomial = std::move(quotient);
133 }
134 }
135 }
136 if (polynomial.degree() == 1) {
137 roots.push_back(SymbolicExpr::number(
138 -polynomial.coeffs[0] / polynomial.coeffs[1]));
139 } else if (polynomial.degree() == 2) {
140 const auto& a = polynomial.coeffs[2];
141 const auto& b = polynomial.coeffs[1];
142 const auto& c = polynomial.coeffs[0];
143 auto radical = SymbolicExpr::sqrt(
144 SymbolicExpr::number(b * b - Rational(4) * a * c));
145 auto negative_b = SymbolicExpr::number(-b);
146 auto denominator = SymbolicExpr::number(Rational(2) * a);
147 roots.push_back(SymbolicExpr::divide(
148 SymbolicExpr::add(negative_b, radical), denominator)->simplify());
149 roots.push_back(SymbolicExpr::divide(
150 sub(negative_b, radical), denominator)->simplify());
151 } else if (polynomial.degree() == 3) {
152 auto expression = poly_to_symbolic(polynomial);
153 for (int index = 0; index < polynomial.degree(); ++index) {
154 roots.push_back(SymbolicExpr::root_of(
155 expression, polynomial.variable_name, index));
156 }
157 }
158 return roots;
159}

161static std::vector<std::shared_ptr<SymbolicExpr>> exact_cubic_roots(
162 const Polynomial<Rational>& polynomial) {
163 if (polynomial.is_zero()) {
164 throw std::invalid_argument(
165 "solve_cubic: identically zero equation has no finite root list");
166 }
167 std::vector<std::shared_ptr<SymbolicExpr>> roots;
168 for (const auto& [factor, multiplicity] :
169 square_free_factorization(polynomial)) {
170 auto factor_roots = exact_factor_roots(factor);
171 for (int repeat = 0; repeat < multiplicity; ++repeat) {
172 roots.insert(roots.end(), factor_roots.begin(), factor_roots.end());
173 }
174 }
175 return roots;
176}

178std::vector<std::shared_ptr<SymbolicExpr>> solve_cubic(
179 const std::shared_ptr<SymbolicExpr>& a,
180 const std::shared_ptr<SymbolicExpr>& b,
181 const std::shared_ptr<SymbolicExpr>& c,
182 const std::shared_ptr<SymbolicExpr>& d,
183 const std::string& variable) {
184 Polynomial<SymbolicPolyCoeff> symbolic(
185 {SymbolicPolyCoeff(d), SymbolicPolyCoeff(c),
186 SymbolicPolyCoeff(b), SymbolicPolyCoeff(a)}, variable);
187 Polynomial<Rational> exact;
188 if (convert_to_rational_poly(symbolic, exact)) {
189 return exact_cubic_roots(exact);
190 }
191 if (a->simplify()->get_number_value_is_zero()) {
192 return solve_degenerate_cubic(b, c, d);
193 }
194 auto depression = numeric_cubic_depression(a, b, c, d);
195 if (depression) return numeric_cubic_roots(*depression);
196 return symbolic_cubic_roots(a, b, c, d);
197}
