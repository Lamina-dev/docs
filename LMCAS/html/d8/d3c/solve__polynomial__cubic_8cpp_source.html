95bool polynomial_solver_detail::bounded_rational_root_search(
96 const Polynomial<Rational>& polynomial) {
97 if (polynomial.degree() <= 1) return true;
98 if (polynomial.lead_coeff() != Rational(1)) {
99 return bounded_rational_root_search(polynomial.make_monic());
100 }
101 BigInt denominator_lcm(1);
102 for (const auto& coefficient : polynomial.coeffs) {
103 denominator_lcm = BigInt::lcm(
104 denominator_lcm, coefficient.get_denominator());
105 if (denominator_lcm.bit_length() > 32) return false;
106 }
107 /**
108 * @brief find_rational_roots 在枚举因子前移除零根。
109 * 大小限制针对本原整数多项式的实际首尾系数；
110 * 各分母虽小，其最小公倍数仍可能很大。
111 */
112 for (const auto& coefficient : polynomial.coeffs) {
113 if (coefficient != Rational(0)) {
114 return (coefficient * Rational(denominator_lcm))
115 .get_numerator().bit_length() <= 32;
116 }
117 }
118 return true;
119}

121static std::vector<std::shared_ptr<SymbolicExpr>> exact_factor_roots(
122 Polynomial<Rational> polynomial, ComputationContext& context) {
123
124 std::vector<std::shared_ptr<SymbolicExpr>> roots;
125 while (polynomial.degree() >= 3 && polynomial.coeffs[0] == Rational(0)) {
126 roots.push_back(SymbolicExpr::number(0));
127 polynomial.coeffs.erase(polynomial.coeffs.begin());
128 }
129 if (polynomial.degree() >= 3) {
130 if (bounded_rational_root_search(polynomial)) {
131 for (const auto& root : find_rational_roots(polynomial)) {
132 Polynomial<Rational> linear(
133 {-root, Rational(1)}, polynomial.variable_name);
134 auto [quotient, remainder] = polynomial.div_mod(linear);
135 if (!remainder.is_zero()) {
136 throw std::logic_error("polynomial solver: inexact rational root division");
137 }
138 roots.push_back(SymbolicExpr::number(root));
139 polynomial = std::move(quotient);
140 }
141 }
142 }
143 if (polynomial.degree() == 1) {
144 roots.push_back(SymbolicExpr::number(
145 -polynomial.coeffs[0] / polynomial.coeffs[1]));
146 } else if (polynomial.degree() == 2) {
147 auto quadratic = solve_quadratic_internal(
148 SymbolicExpr::number(polynomial.coeffs[2]),
149 SymbolicExpr::number(polynomial.coeffs[1]),
150 SymbolicExpr::number(polynomial.coeffs[0]), context);
151 roots.insert(roots.end(), quadratic.begin(), quadratic.end());
152 } else if (polynomial.degree() == 4 &&
153 polynomial.coeffs[3] == Rational(0) &&
154 polynomial.coeffs[1] == Rational(0)) {
155 auto biquadratic = solve_biquadratic_impl(
156 SymbolicExpr::number(polynomial.coeffs[4]),
157 SymbolicExpr::number(polynomial.coeffs[2]),
158 SymbolicExpr::number(polynomial.coeffs[0]),
159 polynomial.variable_name, context);
160 roots.insert(roots.end(), biquadratic.begin(), biquadratic.end());
161 } else if (polynomial.degree() >= 3) {
162 auto expression = poly_to_symbolic(polynomial);
163 for (int index = 0; index < polynomial.degree(); ++index) {
164 auto root = make_rootof_checked(expression, polynomial.variable_name,
165 static_cast<std::size_t>(index), context);
166 if (!root) throw root.error();
167 roots.push_back(std::move(root.value()));
168 }
169 }
170 return roots;
171}

173std::vector<std::shared_ptr<SymbolicExpr>> polynomial_solver_detail::exact_polynomial_roots(
174 const Polynomial<Rational>& polynomial, ComputationContext& context) {
175 if (polynomial.is_zero()) {
176 throw std::invalid_argument(
177 "polynomial solver: identically zero equation has no finite root list");
178 }
179 std::vector<std::shared_ptr<SymbolicExpr>> roots;
180 for (const auto& [factor, multiplicity] :
181 square_free_factorization(polynomial)) {
182 auto factor_roots = exact_factor_roots(factor, context);
183 for (int repeat = 0; repeat < multiplicity; ++repeat) {
184 roots.insert(roots.end(), factor_roots.begin(), factor_roots.end());
185 }
186 }
187 return roots;
188}
