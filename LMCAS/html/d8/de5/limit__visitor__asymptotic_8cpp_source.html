33std::shared_ptr<const SymbolicNode> LimitVisitor::rational_leading_limit(
34 const std::shared_ptr<const SymbolicNode>& leading_numerator,
35 const std::shared_ptr<const SymbolicNode>& leading_denominator, int degree_difference) {
36 if (degree_difference == 0) {
37 auto ratio = detail::make_node<MultiplyNode>(std::vector<std::shared_ptr<const SymbolicNode>>{
38 leading_numerator, detail::make_node<PowerNode>(leading_denominator, detail::make_node<NumberNode>(BigInt(-1)))});
39 NormalizationVisitor norm(*context_, detail::no_facts(), Domain::Real, rewrite_budget());
40 ratio->accept(norm);
41 return norm.get_result();
42 }
43 auto numerator_sign = get_node_sign(leading_numerator);
44 auto denominator_sign = get_node_sign(leading_denominator);
45 if (!denominator_sign || !*denominator_sign) { return nullptr; }
46 if (!numerator_sign || !*numerator_sign) { return nullptr; }
47 int sign = *numerator_sign * *denominator_sign;
48 if (is_limit_at_neg_infinity() && degree_difference % 2) { sign = -sign; }
49 return signed_infinity(sign);
50}

52std::optional<Rational> LimitVisitor::algebraic_sum_growth(const AddNode& sum) const {
53 std::optional<Rational> maximum;
54 std::optional<int> leading_sign;
55 std::size_t leading_count = 0;
56 bool common_sign = true;
57 LimitVisitor witness(var, point, direction, assumption_ctx_, context_);
58 for (const auto& term : sum.operands()) {
59 if (term->is_zero()) { continue; }
60 auto degree = algebraic_growth_degree(term);
61 if (!degree) { return std::nullopt; }
62 if (!maximum || *degree > *maximum) {
63 maximum = *degree;
64 leading_count = 1;
65 leading_sign = witness.determine_sign_near_point(term, direction);
66 common_sign = true;
67 } else if (*degree == *maximum) {
68 ++leading_count;
69 auto sign = witness.determine_sign_near_point(term, direction);
70 common_sign = common_sign && compatible_leading_sign(sign, leading_sign);
71 }
72 }
73 if (maximum && (leading_count == 1 || common_sign)) { return maximum; }
74 return std::nullopt;
75}

121bool LimitVisitor::exponential_dominates(
122 const std::shared_ptr<const SymbolicNode>& exponential,
123 const std::shared_ptr<const SymbolicNode>& other) {
124 auto f = std::dynamic_pointer_cast<const FunctionNode>(exponential);

187/**
188 * @brief 通过多项式次数比较计算有理函数在无穷处的极限.
189 *
190 * 对于 P(x)/Q(x):
191 * - deg(P) < deg(Q) -> 0
192 * - deg(P) = deg(Q) -> 首项系数之比
193 * - deg(P) > deg(Q) -> +/-infinity，符号由首项系数及负无穷处次数差的奇偶性决定
194 *
195 * @param[in] num 分子 AST 节点
196 * @param[in] den 分母 AST 节点
197 * @return 极限结果,非有理函数时返回 nullptr
198 */
199std::shared_ptr<const SymbolicNode> LimitVisitor::limit_rational_at_infinity(
200 const std::shared_ptr<const SymbolicNode>& num,
201 const std::shared_ptr<const SymbolicNode>& den) {
202 charge();
203 auto numerator = symbolic_to_poly<SymbolicPolyCoeff>(detail::make_expression_ptr(num), var);
204 auto denominator = symbolic_to_poly<SymbolicPolyCoeff>(detail::make_expression_ptr(den), var);
205 if (!numerator && numerator.error().code != CasErrc::UnsupportedExpression) { throw numerator.error(); }
206 if (!denominator && denominator.error().code != CasErrc::UnsupportedExpression) { throw denominator.error(); }
207 if (!numerator || !denominator || denominator.value().is_zero()) { return nullptr; }
208 auto normalized_denominator = detail::simplify_expression(
209 denominator.value().lead_coeff().val, *context_);
210 if (!normalized_denominator) { throw normalized_denominator.error(); }
211 auto leading_denominator = detail::node(normalized_denominator.value());
212 if (!rational_denominator_supported(leading_denominator, num, den)) { return nullptr; }
213 if (numerator.value().is_zero() || numerator.value().degree() < denominator.value().degree())
214 { return detail::make_node<NumberNode>(BigInt(0)); }
215 auto normalized_numerator = detail::simplify_expression(
216 numerator.value().lead_coeff().val, *context_);
217 if (!normalized_numerator) { throw normalized_numerator.error(); }
218 auto leading_numerator = detail::node(normalized_numerator.value());
219 return rational_leading_limit(leading_numerator, leading_denominator,
220 numerator.value().degree() - denominator.value().degree());
221}

231LimitVisitor::GrowthClass LimitVisitor::classify_growth(const std::shared_ptr<const SymbolicNode>& node) const {
232 if (!node) {
233 return GrowthClass::Unknown;
234 }
235
236 if (auto num = std::dynamic_pointer_cast<const NumberNode>(node)) {
237 return GrowthClass::Constant;
238 }
239
240 if (auto v = std::dynamic_pointer_cast<const VariableNode>(node)) {
241 return (!v->is_constant() && v->name() == var)
242 ? GrowthClass::Polynomial : GrowthClass::Constant;
243 }
244
245 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(node)) {
246 return function_growth(*func);
247 }
248
249 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
250 return power_growth(*pow);
251 }
252
253 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
254 return operands_growth(mul->operands());
255 }
256
257 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
258 return operands_growth(add->operands());
259 }
260
261 return GrowthClass::Unknown;
262}

279LimitVisitor::GrowthClass LimitVisitor::power_growth(const PowerNode& node) const {
280 if (is_limit_variable(node.base())) {
281 return GrowthClass::Polynomial;

312/**
313 * @brief 求代数增长次数，供指数、多项式、对数增长率比较使用。
314 * 乘积次数相加，幂次数相乘；和式须排除最高次项相消。
315 * 非整数幂要求底数在邻域内有定义且为正。
316 * @param[in] expr 表达式节点
317 * @return 有理增长次数；无法证明时返回 std::nullopt。
318 */
319std::optional<Rational> LimitVisitor::algebraic_growth_degree(
320 const std::shared_ptr<const SymbolicNode>& expr) const {
321 charge();
322 if (!expr) { return std::nullopt; }
323 auto polynomial = symbolic_to_poly<Rational>(detail::make_expression_ptr(expr), var);
324 if (!polynomial && polynomial.error().code != CasErrc::UnsupportedExpression) { throw polynomial.error(); }
325 if (polynomial && !polynomial.value().is_zero())
326 { return Rational(polynomial.value().degree()); }
327 if (auto sum = std::dynamic_pointer_cast<const AddNode>(expr)) {
328 auto cancelled = detail::make_expression_ptr(expr)->cancel();
329 if (cancelled && detail::node(cancelled)->compare(*expr) != 0 && defined_near_point(expr))
330 { return algebraic_growth_degree(detail::node(cancelled)); }
331 return algebraic_sum_growth(*sum);
332 }
333 if (auto product = std::dynamic_pointer_cast<const MultiplyNode>(expr)) {
334 return algebraic_product_growth(*product);
335 }
336 if (auto power = std::dynamic_pointer_cast<const PowerNode>(expr)) {
337 return algebraic_power_growth(*power);
338 }
339 return std::nullopt;
340}
341bool LimitVisitor::polynomial_logarithmic_bound(
342 const std::shared_ptr<const SymbolicNode>& expr) const {
343 auto polynomial = symbolic_to_poly<SymbolicPolyCoeff>(detail::make_expression_ptr(expr), var);
344 if (!polynomial && polynomial.error().code != CasErrc::UnsupportedExpression) { throw polynomial.error(); }
345 if (polynomial && defined_near_point(expr)) { return true; }
346 if (algebraic_growth_degree(expr)) { return true; }
347 if (auto product = std::dynamic_pointer_cast<const MultiplyNode>(expr)) {
348 return polynomial_logarithmic_operands(product->operands());
349 }
350 if (auto sum = std::dynamic_pointer_cast<const AddNode>(expr)) {
351 return polynomial_logarithmic_operands(sum->operands());
352 }
353 if (auto power = std::dynamic_pointer_cast<const PowerNode>(expr)) {
354 BigInt integer;
355 auto n = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
356 return n && try_get_integer_value(n, integer) && integer > BigInt(0) &&
357 polynomial_logarithmic_bound(power->base());
358 }
359 return logarithmic_algebraic_bound(expr);
360}
361std::shared_ptr<const SymbolicNode> LimitVisitor::limit_by_growth_comparison(
362 const std::shared_ptr<const SymbolicNode>& num,
363 const std::shared_ptr<const SymbolicNode>& den) {
364 if (exponential_dominates(den, num)) { return detail::make_node<NumberNode>(BigInt(0)); }
365 if (exponential_dominates(num, den)) {
366 auto denominator = eval_limit(den);
367 auto sign = get_node_sign(denominator);
368 if (sign && *sign) { return signed_infinity(*sign); }
369 }
370 auto log = std::dynamic_pointer_cast<const FunctionNode>(num);
371 auto denominator_degree = algebraic_growth_degree(den);
372 if (log && log->type() == FunctionNode::FuncType::Ln &&
373 polynomial_logarithmic_bound(num) && denominator_degree && *denominator_degree > Rational(0))
374 { return detail::make_node<NumberNode>(BigInt(0)); }
375 return nullptr;
376}

436 * @param[in] node AST 节点

439 */
440std::shared_ptr<const SymbolicNode> LimitVisitor::substitute_neg_t(
441 const std::shared_ptr<const SymbolicNode>& node, const std::string& t_var) const {
442
443 if (!node) {
444 return nullptr;
445 }
446
447 if (auto num = std::dynamic_pointer_cast<const NumberNode>(node)) {
448 return node->clone();
449 }
450
451 if (auto v = std::dynamic_pointer_cast<const VariableNode>(node)) {
452 return substitute_negative_variable(*v, var, t_var);
453 }
454
455 auto add = std::dynamic_pointer_cast<const AddNode>(node);
456 auto mul = add ? std::shared_ptr<const MultiplyNode>{}
457 : std::dynamic_pointer_cast<const MultiplyNode>(node);
458 if (add || mul) {
459 const auto& operands = add ? add->operands() : mul->operands();
460 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
461 for (const auto& op : operands) {
462 auto sub = substitute_neg_t(op, t_var);
463 if (!sub) {
464 return nullptr;
465 }
466 new_ops.push_back(sub);
467 }
468 if (add) {
469 return LMCAS::detail::make_node<AddNode>(new_ops);
470 }
471 return LMCAS::detail::make_node<MultiplyNode>(new_ops);
472 }
473
474 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
475 auto new_base = substitute_neg_t(pow->base(), t_var);
476 auto new_exp = substitute_neg_t(pow->exponent(), t_var);
477 if (!new_base || !new_exp) {
478 return nullptr;
479 }
480 return LMCAS::detail::make_node<PowerNode>(new_base, new_exp);
481 }
482
483 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(node)) {
484 return substitute_function_neg_t(*func, t_var);
485 }
486 return node->clone();
487}

489std::shared_ptr<const SymbolicNode> LimitVisitor::substitute_function_neg_t(
490 const FunctionNode& node, const std::string& t_var) const {
491 if (node.type() == FunctionNode::FuncType::Infinity) {
492 return node.clone();
