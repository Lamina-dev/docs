103 std::shared_ptr<const SymbolicNode> polynomial() const {
104 const auto variable_node =
105 SymbolicFactory::create_variable(variable());
106 std::vector<std::shared_ptr<const SymbolicNode>> terms;
107 for (std::size_t degree = 0;
108 degree < id_.polynomial.coeffs.size(); ++degree) {
109 const auto& coefficient = id_.polynomial.coeffs[degree];
110 if (coefficient == Rational(0)) continue;
111 auto coefficient_node =
112 SymbolicFactory::create_number(coefficient);
113 if (degree == 0) {
114 terms.push_back(coefficient_node);
115 continue;
116 }
117 auto variable_power = degree == 1
118 ? variable_node
119 : SymbolicFactory::create_power(
120 variable_node,
121 SymbolicFactory::create_number(BigInt(degree)));
122 terms.push_back(
123 coefficient == Rational(1)
124 ? variable_power
125 : SymbolicFactory::create_multiply(
126 {coefficient_node, variable_power}));
127 }
128 return SymbolicFactory::create_add(std::move(terms));
129 }
