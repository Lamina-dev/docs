103 std::size_t index() const noexcept { return id_.index; }

105 return display_variable_;
106 }
107 std::shared_ptr<const SymbolicNode> polynomial() const {
108 const auto variable_node =
109 SymbolicFactory::create_variable(variable());
110 std::vector<std::shared_ptr<const SymbolicNode>> terms;
111 for (std::size_t degree = 0;
112 degree < id_.polynomial.coeffs.size(); ++degree) {
113 const auto& coefficient = id_.polynomial.coeffs[degree];
114 if (coefficient == Rational(0)) continue;
115 auto coefficient_node =
116 SymbolicFactory::create_number(coefficient);
117 if (degree == 0) {
118 terms.push_back(coefficient_node);
119 continue;
120 }
121 auto variable_power = degree == 1
122 ? variable_node
123 : SymbolicFactory::create_power(
124 variable_node,
125 SymbolicFactory::create_number(
126 BigInt(static_cast<long long>(degree))));
127 terms.push_back(
128 coefficient == Rational(1)
129 ? variable_power
130 : SymbolicFactory::create_multiply(
131 {coefficient_node, variable_power}));
132 }
133 return SymbolicFactory::create_add(std::move(terms));
134 }
135 int type_priority() const override { return 110; }
136
137protected:
138 std::size_t compute_hash() const override {
139 std::size_t seed = static_cast<std::size_t>(type_priority());
140 hash_combine(seed, id_.hash());
141 return seed;
142 }
143
144 int compare_same_type(const SymbolicNode& other) const override {
145 return id_.compare(static_cast<const RootOfNode&>(other).id_);
146 }
147
148public:
149 void accept(LMCAS::detail::SymbolicVisitor& visitor) const override {
150 LMCAS::detail::SymbolicVisitor::DepthGuard guard(visitor);
151 visitor.visit(*this);
152 }
153

155 return LMCAS::detail::make_node<RootOfNode>(
156 id_, display_variable_);
157 }
