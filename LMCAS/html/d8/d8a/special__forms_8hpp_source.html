105 std::shared_ptr<const SymbolicNode> polynomial() const {
106 const auto variable_node =
107 SymbolicFactory::create_variable(variable());
108 std::vector<std::shared_ptr<const SymbolicNode>> terms;
109 for (std::size_t degree = 0;
110 degree < id_.polynomial.coeffs.size(); ++degree) {
111 const auto& coefficient = id_.polynomial.coeffs[degree];
112 if (coefficient == Rational(0)) continue;
113 auto coefficient_node =
114 SymbolicFactory::create_number(coefficient);
115 if (degree == 0) {
116 terms.push_back(coefficient_node);
117 continue;
118 }
119 auto variable_power = degree == 1
120 ? variable_node
121 : SymbolicFactory::create_power(
122 variable_node,
123 SymbolicFactory::create_number(
124 BigInt(static_cast<long long>(degree))));
125 terms.push_back(
126 coefficient == Rational(1)
127 ? variable_power
128 : SymbolicFactory::create_multiply(
129 {coefficient_node, variable_power}));
130 }
131 return SymbolicFactory::create_add(std::move(terms));
132 }
