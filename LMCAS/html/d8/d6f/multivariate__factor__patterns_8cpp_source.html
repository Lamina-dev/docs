26/**
27 * @brief 检测差平方模式 a² - b²
28 *
29 * 判断多项式是否恰好有两项，一正一负，且两项的单项式各分量指数均为偶数。
30 * 若匹配，返回 (a, b)，其中 a² 为正项，b² 为负项。
31 *
32 * @param[in] poly 输入多项式
33 * @return 若匹配则返回 (a, b)，否则返回 nullopt
34 */
35std::optional<std::pair<MultiPoly, MultiPoly>>
36detect_difference_of_squares(const MultiPoly& poly)
37{
38 if (poly.num_terms() != 2) return std::nullopt;
39
40 const auto& terms = poly.terms();
41 const auto& vars = poly.variables();
42
43 /// 确定哪项为正、哪项为负
44 int pos_idx = -1, neg_idx = -1;
45 if (terms[0].second > Rational(0) && terms[1].second < Rational(0)) {
46 pos_idx = 0; neg_idx = 1;
47 } else if (terms[1].second > Rational(0) && terms[0].second < Rational(0)) {
48 pos_idx = 1; neg_idx = 0;
49 } else {
50 return std::nullopt;
51 }
52
53 /// 检查系数绝对值相等（都应为完全平方的系数）
54 /// 对于差平方 a² - b²，正项系数和负项系数的绝对值不必相等
55 /// 但两项的系数本身必须是完全平方数
56 Rational pos_coeff = terms[pos_idx].second;
57 Rational neg_coeff = -terms[neg_idx].second; // 取绝对值
58
59 /// 检查系数是否为完全平方有理数
60 /// 分子和分母都必须是完全平方数
61 BigInt pos_num = pos_coeff.get_numerator();
62 BigInt pos_den = pos_coeff.get_denominator();
63 BigInt neg_num = neg_coeff.get_numerator();
64 BigInt neg_den = neg_coeff.get_denominator();
65
66 if (pos_num < BigInt(0) || neg_num < BigInt(0)) return std::nullopt;
67
68 /// 检查是否为完全平方
69 /// 注意：不能依赖 BigInt::sqrt()（该实现有 bug，对小整数返回错误值），
70 /// 改用 double 估计取整后用精确乘法验证。
71 auto is_perfect_square = [](const BigInt& n) -> std::pair<bool, BigInt> {
72 if (n == BigInt(0)) return {true, BigInt(0)};
73 if (n == BigInt(1)) return {true, BigInt(1)};
74 if (n < BigInt(0)) return {false, BigInt(0)};
75 double dn = n.to_double();
76 long long est = (long long)std::llround(std::sqrt(dn));
77 /// 在估计值附近搜索，抵消浮点误差
78 for (long long cand = (est > 2 ? est - 2 : 0); cand <= est + 2; ++cand) {
79 BigInt r(cand);
80 if (r * r == n) return {true, r};
81 }
82 return {false, BigInt(0)};
83 };
84
85 auto [pos_num_sq, pos_num_root] = is_perfect_square(pos_num);
86 if (!pos_num_sq) return std::nullopt;
87 auto [pos_den_sq, pos_den_root] = is_perfect_square(pos_den);
88 if (!pos_den_sq) return std::nullopt;
89 auto [neg_num_sq, neg_num_root] = is_perfect_square(neg_num);
90 if (!neg_num_sq) return std::nullopt;
91 auto [neg_den_sq, neg_den_root] = is_perfect_square(neg_den);
92 if (!neg_den_sq) return std::nullopt;
93
94 /// 检查两项的单项式各分量指数均为偶数
95 const Monomial& pos_mono = terms[pos_idx].first;
96 const Monomial& neg_mono = terms[neg_idx].first;
97
98 for (size_t i = 0; i < pos_mono.size(); ++i) {
99 if (pos_mono[i] % 2 != 0) return std::nullopt;
100 }
101 for (size_t i = 0; i < neg_mono.size(); ++i) {
102 if (neg_mono[i] % 2 != 0) return std::nullopt;
103 }
104
105 /// 构造 a：正项单项式指数减半，系数为 sqrt(pos_coeff)
106 Monomial a_mono(vars.size(), 0);
107 for (size_t i = 0; i < pos_mono.size(); ++i) {
108 a_mono[i] = pos_mono[i] / 2;
109 }
110 Rational a_coeff(pos_num_root, pos_den_root);
111 std::vector<MultiPoly::Term> a_terms = {{a_mono, a_coeff}};
112 MultiPoly a(std::move(a_terms), vars);
113
114 /// 构造 b：负项单项式指数减半，系数为 sqrt(neg_coeff)
115 Monomial b_mono(vars.size(), 0);
116 for (size_t i = 0; i < neg_mono.size(); ++i) {
117 b_mono[i] = neg_mono[i] / 2;
118 }
119 Rational b_coeff(neg_num_root, neg_den_root);
120 std::vector<MultiPoly::Term> b_terms = {{b_mono, b_coeff}};
121 MultiPoly b(std::move(b_terms), vars);
122
123 return std::make_pair(a, b);
124}

126/**
127 * @brief 检测二项式幂模式 xⁿ - yⁿ
128 *
129 * 判断多项式是否为两个不同变量的纯幂之差，系数分别为 +1 和 -1，
130 * 且幂次相同。若匹配，返回 (变量1, 变量2, 幂次)。
131 *
132 * @param[in] poly 输入多项式
133 * @return 若匹配则返回 (var1, var2, n)，否则返回 nullopt
134 */
135std::optional<std::tuple<std::string, std::string, int>>
136detect_binomial_power(const MultiPoly& poly)
137{
138 if (poly.num_terms() != 2) return std::nullopt;
139
140 const auto& terms = poly.terms();
141 const auto& vars = poly.variables();
142
143 /// 确定正项和负项
144 int pos_idx = -1, neg_idx = -1;
145 if (terms[0].second == Rational(1) && terms[1].second == Rational(-1)) {
146 pos_idx = 0; neg_idx = 1;
147 } else if (terms[1].second == Rational(1) && terms[0].second == Rational(-1)) {
148 pos_idx = 1; neg_idx = 0;
149 } else {
150 return std::nullopt;
151 }
152
153 /// 检查每项的单项式是否为单变量的纯幂（恰好一个非零指数）
154 auto get_single_var_power = [&](const Monomial& mono)
155 -> std::optional<std::pair<int, int>> {
156 int nonzero_count = 0;
157 int var_index = -1;
158 int exponent = 0;
159 for (size_t i = 0; i < mono.size(); ++i) {
160 if (mono[i] != 0) {
161 nonzero_count++;
162 var_index = static_cast<int>(i);
163 exponent = mono[i];
164 }
165 }
166 if (nonzero_count != 1 || exponent <= 1) return std::nullopt;
167 return std::make_pair(var_index, exponent);
168 };
169
170 auto pos_info = get_single_var_power(terms[pos_idx].first);
171 if (!pos_info) return std::nullopt;
172 auto neg_info = get_single_var_power(terms[neg_idx].first);
173 if (!neg_info) return std::nullopt;
174
175 auto [pos_var_idx, pos_exp] = *pos_info;
176 auto [neg_var_idx, neg_exp] = *neg_info;
177
178 /// 幂次必须相同，变量必须不同
179 if (pos_exp != neg_exp) return std::nullopt;
180 if (pos_var_idx == neg_var_idx) return std::nullopt;
181
182 return std::make_tuple(vars[pos_var_idx], vars[neg_var_idx], pos_exp);
183}

185/**
186 * @brief 提取公因子单项式
187 *
188 * 计算所有项单项式的逐分量最小值（GCD 单项式），将其从每项中减去。
189 *
190 * @param[in] poly 输入多项式
191 * @return pair(公因子单项式, 提取后的多项式)
192 */
193std::pair<Monomial, MultiPoly>
194extract_common_monomial(const MultiPoly& poly)
195{
196 const auto& vars = poly.variables();
197 Monomial identity(vars.size(), 0);
198
199 if (poly.is_zero() || poly.num_terms() == 0) {
200 return {identity, poly};
201 }
202
203 const auto& terms = poly.terms();
204
205 /// 计算逐分量最小值
206 Monomial gcd_mono = terms[0].first;
207 for (size_t t = 1; t < terms.size(); ++t) {
208 const Monomial& mono = terms[t].first;
209 for (size_t i = 0; i < gcd_mono.size(); ++i) {
210 int val = (i < mono.size()) ? mono[i] : 0;
211 if (val < gcd_mono[i]) gcd_mono[i] = val;
212 }
213 }
214
215 /// 检查是否为平凡（全零）
216 bool is_trivial = true;
217 for (size_t i = 0; i < gcd_mono.size(); ++i) {
218 if (gcd_mono[i] != 0) { is_trivial = false; break; }
219 }
220 if (is_trivial) return {identity, poly};
221
222 /// 从每项中减去 GCD 单项式
223 std::vector<MultiPoly::Term> quotient_terms;
224 quotient_terms.reserve(terms.size());
225 for (const auto& term : terms) {
226 Monomial new_mono(vars.size(), 0);
227 for (size_t i = 0; i < vars.size(); ++i) {
228 int orig = (i < term.first.size()) ? term.first[i] : 0;
229 new_mono[i] = orig - gcd_mono[i];
230 }
231 quotient_terms.emplace_back(std::move(new_mono), term.second);
232 }
233
234 MultiPoly quotient(std::move(quotient_terms), vars);
235 return {gcd_mono, quotient};
236}

238/**
239 * @brief 齐次二元多项式分解
240 *
241 * 对齐次二元多项式 f(x, y)，令 y=1 得到一元多项式 f(x, 1)，
242 * 对其进行一元因式分解，然后将每个因子重新齐次化。
243 *
244 * @param[in] poly 输入多项式（须为齐次二元）
245 * @return 若适用则返回分解结果，否则返回 nullopt
246 */
247std::optional<MultiFactorResult>
248factor_homogeneous_bivariate(const MultiPoly& poly)
249{
250 if (!poly.is_homogeneous() || poly.num_vars() != 2) return std::nullopt;
251
252 const auto& vars = poly.variables();
253 std::string var_x = vars[0];
254 std::string var_y = vars[1];
255
256 /// 去齐次化：令 y = 1，得到关于 x 的一元多项式
257 MultiPoly dehomogenized = poly.eval(var_y, Rational(1));
258
259 /// 转换为一元多项式
260 Polynomial<Rational> uni_poly;
261 try {
262 uni_poly = dehomogenized.to_univariate();
263 } catch (...) {
264 return std::nullopt;
265 }
266
267 /// 一元因式分解
268 std::vector<Polynomial<Rational>> uni_factors = factor_univariate_bridge(uni_poly);
269 if (uni_factors.size() <= 1) return std::nullopt;
270
271 /// 重新齐次化每个因子
272 MultiFactorResult result;
273 result.constant = poly.numeric_content();
274
275 for (const auto& uf : uni_factors) {
276 int d = uf.degree();
277 /// 对一元因子的每项 c*x^k，齐次化为 c*x^k*y^(d-k)
278 std::vector<MultiPoly::Term> homo_terms;
279 for (int k = 0; k <= d; ++k) {
280 Rational coeff = (k < static_cast<int>(uf.coeffs.size())) ? uf.coeffs[k] : Rational(0);
281 if (coeff.is_zero()) continue;
282 Monomial mono(2, 0);
283 mono[0] = k; // x 的指数
284 mono[1] = d - k; // y 的指数
285 homo_terms.emplace_back(std::move(mono), coeff);
286 }
287 MultiPoly homo_factor(std::move(homo_terms), vars);
288 homo_factor = homo_factor.make_primitive();
289 result.factors.push_back(std::move(homo_factor));
290 result.multiplicities.push_back(1);
291 }
292
293 /// 调整常数因子使乘积等于原多项式
294 /// 计算所有因子的乘积
295 MultiPoly product(Rational(1), vars);
296 for (const auto& f : result.factors) {
297 product = product * f;
298 }
299 if (!product.is_zero() && !poly.is_zero()) {
300 /// constant = poly / product 的比例系数
301 try {
302 MultiPoly ratio = poly.exact_div(product);
303 if (ratio.is_constant()) {
304 result.constant = ratio.numeric_content();
305 if (!ratio.terms().empty() && ratio.terms()[0].second < Rational(0)) {
306 result.constant = -result.constant;
307 }
308 }
309 } catch (...) {
310 /// 若除法失败，使用首项系数比
311 if (!product.terms().empty() && !poly.terms().empty()) {
312 result.constant = poly.terms()[0].second / product.terms()[0].second;
313 }
314 }
315 }
316
317 return result;
318}
