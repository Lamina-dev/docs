26/**
27 * @brief 检测差平方模式 a^2 - b^2
28 *
29 * 判断多项式是否恰好有两项,一正一负,且两项的单项式各分量指数均为偶数.
30 * 若匹配,返回 (a, b),其中 a^2 为正项,b^2 为负项.
31 *
32 * @param[in] poly 输入多项式
33 * @return 若匹配则返回 (a, b),否则返回 nullopt
34 */
35std::optional<std::pair<MultiPoly, MultiPoly>>
36detect_difference_of_squares(const MultiPoly& poly)
37{
38 if (poly.num_terms() != 2) return std::nullopt;
39
40 const auto& terms = poly.terms();
41 const auto& vars = poly.variables();
42
43 /// 确定哪项为正,哪项为负
44 int pos_idx = -1, neg_idx = -1;
45 if (terms[0].second > Rational(0) && terms[1].second < Rational(0)) {
46 pos_idx = 0; neg_idx = 1;
47 } else if (terms[1].second > Rational(0) && terms[0].second < Rational(0)) {
48 pos_idx = 1; neg_idx = 0;
49 } else {
50 return std::nullopt;
51 }
52
53 /// 检查系数绝对值相等(都应为完全平方的系数)
54 /// 对于差平方 a^2 - b^2,正项系数和负项系数的绝对值不必相等
55 /// 但两项的系数本身必须是完全平方数
56 Rational pos_coeff = terms[pos_idx].second;
57 Rational neg_coeff = -terms[neg_idx].second; // 取绝对值
58
59 /// 检查系数是否为完全平方有理数
60 /// 分子和分母都必须是完全平方数
61 BigInt pos_num = pos_coeff.get_numerator();
62 BigInt pos_den = pos_coeff.get_denominator();
63 BigInt neg_num = neg_coeff.get_numerator();
64 BigInt neg_den = neg_coeff.get_denominator();
65
66 if (pos_num < BigInt(0) || neg_num < BigInt(0)) return std::nullopt;
67
68 /// 完全平方判定先以 double 估计候选整数,再用 BigInt 乘法精确验证,
69 /// 使小整数结果独立于 BigInt::sqrt() 的当前实现.
70 auto is_perfect_square = [](const BigInt& n) -> std::pair<bool, BigInt> {
71 if (n == BigInt(0)) return {true, BigInt(0)};
72 if (n == BigInt(1)) return {true, BigInt(1)};
73 if (n < BigInt(0)) return {false, BigInt(0)};
74 double dn = n.to_double();
75 long long est = (long long)std::llround(std::sqrt(dn));
76 /// 在估计值附近搜索,抵消浮点误差
77 for (long long cand = (est > 2 ? est - 2 : 0); cand <= est + 2; ++cand) {
78 BigInt r(cand);
79 if (r * r == n) return {true, r};
80 }
81 return {false, BigInt(0)};
82 };
83
84 auto [pos_num_sq, pos_num_root] = is_perfect_square(pos_num);
85 if (!pos_num_sq) return std::nullopt;
86 auto [pos_den_sq, pos_den_root] = is_perfect_square(pos_den);
87 if (!pos_den_sq) return std::nullopt;
88 auto [neg_num_sq, neg_num_root] = is_perfect_square(neg_num);
89 if (!neg_num_sq) return std::nullopt;
90 auto [neg_den_sq, neg_den_root] = is_perfect_square(neg_den);
91 if (!neg_den_sq) return std::nullopt;
92
93 /// 检查两项的单项式各分量指数均为偶数
94 const Monomial& pos_mono = terms[pos_idx].first;
95 const Monomial& neg_mono = terms[neg_idx].first;
96
97 for (size_t i = 0; i < pos_mono.size(); ++i) {
98 if (pos_mono[i] % 2 != 0) return std::nullopt;
99 }
100 for (size_t i = 0; i < neg_mono.size(); ++i) {
101 if (neg_mono[i] % 2 != 0) return std::nullopt;
102 }
103
104 /// 构造 a:正项单项式指数减半,系数为 sqrt(pos_coeff)
105 Monomial a_mono(vars.size(), 0);
106 for (size_t i = 0; i < pos_mono.size(); ++i) {
107 a_mono[i] = pos_mono[i] / 2;
108 }
109 Rational a_coeff(pos_num_root, pos_den_root);
110 std::vector<MultiPoly::Term> a_terms = {{a_mono, a_coeff}};
111 MultiPoly a(std::move(a_terms), vars);
112
113 /// 构造 b:负项单项式指数减半,系数为 sqrt(neg_coeff)
114 Monomial b_mono(vars.size(), 0);
115 for (size_t i = 0; i < neg_mono.size(); ++i) {
116 b_mono[i] = neg_mono[i] / 2;
117 }
118 Rational b_coeff(neg_num_root, neg_den_root);
119 std::vector<MultiPoly::Term> b_terms = {{b_mono, b_coeff}};
120 MultiPoly b(std::move(b_terms), vars);
121
122 return std::make_pair(a, b);
123}

125/**
126 * @brief 检测二项式幂模式 xⁿ - yⁿ
127 *
128 * 判断多项式是否为两个不同变量的纯幂之差,系数分别为 +1 和 -1,
129 * 且幂次相同.若匹配,返回 (变量1, 变量2, 幂次).
130 *
131 * @param[in] poly 输入多项式
132 * @return 若匹配则返回 (var1, var2, n),否则返回 nullopt
133 */
134std::optional<std::tuple<std::string, std::string, int>>
135detect_binomial_power(const MultiPoly& poly)
136{
137 if (poly.num_terms() != 2) return std::nullopt;
138
139 const auto& terms = poly.terms();
140 const auto& vars = poly.variables();
141
142 /// 确定正项和负项
143 int pos_idx = -1, neg_idx = -1;
144 if (terms[0].second == Rational(1) && terms[1].second == Rational(-1)) {
145 pos_idx = 0; neg_idx = 1;
146 } else if (terms[1].second == Rational(1) && terms[0].second == Rational(-1)) {
147 pos_idx = 1; neg_idx = 0;
148 } else {
149 return std::nullopt;
150 }
151
152 /// 检查每项的单项式是否为单变量的纯幂(恰好一个非零指数)
153 auto get_single_var_power = [&](const Monomial& mono)
154 -> std::optional<std::pair<int, int>> {
155 int nonzero_count = 0;
156 int var_index = -1;
157 int exponent = 0;
158 for (size_t i = 0; i < mono.size(); ++i) {
159 if (mono[i] != 0) {
160 nonzero_count++;
161 var_index = static_cast<int>(i);
162 exponent = mono[i];
163 }
164 }
165 if (nonzero_count != 1 || exponent <= 1) return std::nullopt;
166 return std::make_pair(var_index, exponent);
167 };
168
169 auto pos_info = get_single_var_power(terms[pos_idx].first);
170 if (!pos_info) return std::nullopt;
171 auto neg_info = get_single_var_power(terms[neg_idx].first);
172 if (!neg_info) return std::nullopt;
173
174 auto [pos_var_idx, pos_exp] = *pos_info;
175 auto [neg_var_idx, neg_exp] = *neg_info;
176
177 /// 幂次必须相同,变量必须不同
178 if (pos_exp != neg_exp) return std::nullopt;
179 if (pos_var_idx == neg_var_idx) return std::nullopt;
180
181 return std::make_tuple(vars[pos_var_idx], vars[neg_var_idx], pos_exp);
182}

184/**
185 * @brief 提取公因子单项式
186 *
187 * 计算所有项单项式的逐分量最小值(GCD 单项式),将其从每项中减去.
188 *
189 * @param[in] poly 输入多项式
190 * @return pair(公因子单项式, 提取后的多项式)
191 */
192std::pair<Monomial, MultiPoly>
193extract_common_monomial(const MultiPoly& poly)
194{
195 const auto& vars = poly.variables();
196 Monomial identity(vars.size(), 0);
197
198 if (poly.is_zero() || poly.num_terms() == 0) {
199 return {identity, poly};
200 }
201
202 const auto& terms = poly.terms();
203
204 /// 计算逐分量最小值
205 Monomial gcd_mono = terms[0].first;
206 for (size_t t = 1; t < terms.size(); ++t) {
207 const Monomial& mono = terms[t].first;
208 for (size_t i = 0; i < gcd_mono.size(); ++i) {
209 int val = (i < mono.size()) ? mono[i] : 0;
210 if (val < gcd_mono[i]) gcd_mono[i] = val;
211 }
212 }
213
214 /// 检查是否为平凡(全零)
215 bool is_trivial = true;
216 for (size_t i = 0; i < gcd_mono.size(); ++i) {
217 if (gcd_mono[i] != 0) { is_trivial = false; break; }
218 }
219 if (is_trivial) return {identity, poly};
220
221 /// 从每项中减去 GCD 单项式
222 std::vector<MultiPoly::Term> quotient_terms;
223 quotient_terms.reserve(terms.size());
224 for (const auto& term : terms) {
225 Monomial new_mono(vars.size(), 0);
226 for (size_t i = 0; i < vars.size(); ++i) {
227 int orig = (i < term.first.size()) ? term.first[i] : 0;
228 new_mono[i] = orig - gcd_mono[i];
229 }
230 quotient_terms.emplace_back(std::move(new_mono), term.second);
231 }
232
233 MultiPoly quotient(std::move(quotient_terms), vars);
234 return {gcd_mono, quotient};
235}

237/**
238 * @brief 齐次二元多项式分解
239 *
240 * 对齐次二元多项式 f(x, y),令 y=1 得到一元多项式 f(x, 1),
241 * 对其进行一元因式分解,然后将每个因子重新齐次化.
242 *
243 * @param[in] poly 输入多项式(须为齐次二元)
244 * @return 若适用则返回分解结果,否则返回 nullopt
245 */
246std::optional<MultiFactorResult>
247factor_homogeneous_bivariate(const MultiPoly& poly)
248{
249 if (!poly.is_homogeneous() || poly.num_vars() != 2) return std::nullopt;
250
251 const auto& vars = poly.variables();
252 std::string var_x = vars[0];
253 std::string var_y = vars[1];
254
255 /// 去齐次化:令 y = 1,得到关于 x 的一元多项式
256 MultiPoly dehomogenized = poly.eval(var_y, Rational(1));
257
258 /// 转换为一元多项式
259 Polynomial<Rational> uni_poly;
260 try {
261 uni_poly = dehomogenized.to_univariate();
262 } catch (const std::invalid_argument&) {
263 return std::nullopt;
264 }
265
266 /// 一元因式分解
267 ComputationContext context;
268 auto checked_factors = factor_univariate_bridge_checked(uni_poly, context);
269 if (!checked_factors ||
270 checked_factors.value().completeness != Completeness::Complete) {
271 return std::nullopt;
272 }
273 std::vector<Polynomial<Rational>> uni_factors =
274 std::move(checked_factors.value().value);
275 if (uni_factors.size() <= 1) return std::nullopt;
276
277 /// 重新齐次化每个因子
278 MultiFactorResult result;
279 result.constant = poly.numeric_content();
280
281 for (const auto& uf : uni_factors) {
282 int d = uf.degree();
283 /// 对一元因子的每项 c*x^k,齐次化为 c*x^k*y^(d-k)
284 std::vector<MultiPoly::Term> homo_terms;
285 for (int k = 0; k <= d; ++k) {
286 Rational coeff = (k < static_cast<int>(uf.coeffs.size())) ? uf.coeffs[k] : Rational(0);
287 if (coeff.is_zero()) continue;
288 Monomial mono(2, 0);
289 mono[0] = k; // x 的指数
290 mono[1] = d - k; // y 的指数
291 homo_terms.emplace_back(std::move(mono), coeff);
292 }
293 MultiPoly homo_factor(std::move(homo_terms), vars);
294 homo_factor = homo_factor.make_primitive();
295 result.factors.push_back(std::move(homo_factor));
296 result.multiplicities.push_back(1);
297 }
298
299 /// 调整常数因子使乘积等于原多项式
300 /// 计算所有因子的乘积
301 MultiPoly product(Rational(1), vars);
302 for (const auto& f : result.factors) {
303 product = product * f;
304 }
305 if (!product.is_zero() && !poly.is_zero()) {
306 /// constant = poly / product 的比例系数
307 try {
308 MultiPoly ratio = poly.exact_div(product);
309 if (ratio.is_constant()) {
310 result.constant = ratio.numeric_content();
311 if (!ratio.terms().empty() && ratio.terms()[0].second < Rational(0)) {
312 result.constant = -result.constant;
313 }
314 }
315 } catch (const std::runtime_error&) {
316 /// 若除法失败,使用首项系数比
317 if (!product.terms().empty() && !poly.terms().empty()) {
318 result.constant = poly.terms()[0].second / product.terms()[0].second;
319 }
320 }
321 }
322
323 return result;
324}
