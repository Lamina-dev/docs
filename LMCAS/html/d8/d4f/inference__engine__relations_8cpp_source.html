25Result<void> InferenceEngine::Impl::accumulate_sum_relation_sign(
26 const std::shared_ptr<const SymbolicNode>& operand, const SymbolicExpr& zero_expr,
27 bool& all_gt_zero, bool& all_geq_zero, ComputationContext& context) const {
28 auto step = context.consume_steps(1, "inference.relations");

33 auto gt = ctx.has_relation_checked(

35 if (!gt) {
36 return Result<void>::failure(gt.error());

39 op_expr, zero_expr, RelationalNode::Op::GEQ, context);

43 bool op_gt = gt.value();
44 bool op_geq = geq.value();
45 if (!op_gt) {

53 }
54 if (!op_gt && !op_geq) {
55 all_geq_zero = false;
56 }

126Result<bool> InferenceEngine::Impl::relation_rhs_nonnegative(const Relation& rel, const SymbolicExpr& zero_expr, ComputationContext& context) const {
127 auto step = context.consume_steps(1, "inference.relations");
128 if (!step) return Result<bool>::failure(step.error());
129 const auto& rel_store = ctx;
130 if (auto rhs_num = std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(rel.rhs))) {
131 if (rhs_num->is_zero() || rhs_num->is_positive()) {
132 return true;
133 }
134 }
135
136 if (auto rhs_var = std::dynamic_pointer_cast<const VariableNode>(LMCAS::detail::node(rel.rhs))) {
137 if (rhs_var->is_constant() ||
138 ctx.has_sign(rhs_var->name(), Sign::NonNegative)) {
139 return true;
140 }
141 }
142
143 auto geq = rel_store.has_relation_checked(
144 rel.rhs, zero_expr, RelationalNode::Op::GEQ, context);
145 if (!geq || geq.value()) return geq;
146 return rel_store.has_relation_checked(
147 rel.rhs, zero_expr, RelationalNode::Op::GT, context);
148}

150InferenceTriboolResult InferenceEngine::Impl::positive_relation_chain(
151 const SymbolicExpr& expr,
152 const SymbolicExpr& zero_expr, ComputationContext& context) const {
153 auto relations = ctx.get_visible_relations_checked(context);
154 if (!relations) return InferenceTriboolResult::failure(relations.error());
155 detail::RewriteBudget budget(context, context.limits().max_recursion_depth,
156 std::numeric_limits<std::size_t>::max(), "inference.relations");
157 budget.measure(detail::node(expr));
158 for (const auto& rel : relations.value()) {
159 auto step = context.consume_steps(1, "inference.relations");
160 if (!step) throw step.error();
161 if (rel.op != RelationalNode::Op::GT) {
162 continue;
163 }
164
165 if (!LMCAS::detail::node(rel.lhs) || !LMCAS::detail::node(expr)) {
166 continue;
167 }
168 budget.measure(detail::node(rel.lhs));
169 if (!LMCAS::detail::node(rel.lhs)->equals(*LMCAS::detail::node(expr))) {
170 continue;
171 }
172 if (!LMCAS::detail::node(rel.rhs)) {
173 continue;
174 }
175
176 auto nonnegative = relation_rhs_nonnegative(rel, zero_expr, context);
177 if (!nonnegative) return InferenceTriboolResult::failure(nonnegative.error());
178 if (nonnegative.value()) {
179 return InferenceTriboolResult::success(Tribool::True);
180 }
181 }
182 return InferenceTriboolResult::success(Tribool::Unknown);
183}
