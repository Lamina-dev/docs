11int sign_changes(const std::vector<int>& signs) {
12 int changes = 0;
13 int previous = 0;
14 for (const int current : signs) {
15 if (current == 0) continue;
16 if (previous != 0 && current != previous) ++changes;
17 previous = current;
18 }
19 return changes;
20}

22int polynomial_sign_at(
23 const Polynomial<Rational>& polynomial,
24 const Rational& point) {
25 if (polynomial.is_zero()) return 0;
26 const int degree = polynomial.degree();
27 const BigInt numerator = point.get_numerator();
28 const BigInt denominator = point.get_denominator();
29 BigInt coefficient_denominator(1);
30 for (const auto& coefficient : polynomial.coeffs) {
31 coefficient_denominator *= coefficient.get_denominator();
32 }
33 std::vector<BigInt> numerator_powers(
34 static_cast<std::size_t>(degree) + 1, BigInt(1));
35 std::vector<BigInt> denominator_powers(
36 static_cast<std::size_t>(degree) + 1, BigInt(1));
37 for (int exponent = 1; exponent <= degree; ++exponent) {
38 numerator_powers[static_cast<std::size_t>(exponent)] =
39 numerator_powers[static_cast<std::size_t>(exponent - 1)] *
40 numerator;
41 denominator_powers[static_cast<std::size_t>(exponent)] =
42 denominator_powers[static_cast<std::size_t>(exponent - 1)] *
43 denominator;
44 }
45 BigInt value(0);
46 for (int exponent = 0; exponent <= degree; ++exponent) {
47 const auto& coefficient =
48 polynomial.coeffs[static_cast<std::size_t>(exponent)];
49 if (coefficient == Rational(0)) continue;
50 value += coefficient.get_numerator() *
51 (coefficient_denominator / coefficient.get_denominator()) *
52 numerator_powers[static_cast<std::size_t>(exponent)] *
53 denominator_powers[
54 static_cast<std::size_t>(degree - exponent)];
55 }
56 if (value == BigInt(0)) return 0;
57 return value < BigInt(0) ? -1 : 1;
58}

60Result<SturmSequence> make_sturm_sequence(
61 Polynomial<Rational> polynomial,
62 ComputationContext& context,
63 const std::string& operation) {
64 auto step = context.consume_steps(1, operation);
65 if (!step) return Result<SturmSequence>::failure(step.error());
66
67 polynomial.trim();
68 if (polynomial.degree() <= 0) {
69 return Result<SturmSequence>::success({});
70 }
71 polynomial = polynomial.square_free_part().make_monic();
72
73 SturmSequence sequence;
74 sequence.push_back(polynomial);
75 sequence.push_back(polynomial.differentiate());
76 while (!sequence.back().is_zero()) {
77 step = context.consume_steps(1, operation);
78 if (!step) return Result<SturmSequence>::failure(step.error());
79 const auto& dividend = sequence[sequence.size() - 2];
80 const auto& divisor = sequence.back();
81 auto division = dividend.div_mod(divisor);
82 auto remainder = std::move(division.second);
83 if (remainder.is_zero()) break;
84 for (auto& coefficient : remainder.coeffs) coefficient = -coefficient;
85 remainder.trim();
86 sequence.push_back(std::move(remainder));
87 }
88 if (!sequence.empty() && sequence.back().is_zero()) sequence.pop_back();
89 return Result<SturmSequence>::success(std::move(sequence));
90}

92Result<int> variations_at(
93 const SturmSequence& sequence,
94 const Rational& point,
95 ComputationContext& context,
96 const std::string& operation) {
97 auto step = context.consume_steps(sequence.size() + 1, operation);
98 if (!step) return Result<int>::failure(step.error());
99 std::vector<int> signs;
100 signs.reserve(sequence.size());
101 for (const auto& member : sequence) {
102 signs.push_back(polynomial_sign_at(member, point));
103 }
104 return Result<int>::success(sign_changes(signs));
105}

107Result<std::size_t> count_open_roots(
108 const Polynomial<Rational>& square_free,
109 const SturmSequence& sequence,
110 const Rational& lower,
111 const Rational& upper,
112 ComputationContext& context,
113 const std::string& operation) {
114 if (!(lower < upper) || sequence.empty()) {
115 return Result<std::size_t>::success(0);
116 }
117 auto lower_variations = variations_at(sequence, lower, context, operation);
118 if (!lower_variations) {
119 return Result<std::size_t>::failure(lower_variations.error());
120 }
121 auto upper_variations = variations_at(sequence, upper, context, operation);
122 if (!upper_variations) {
123 return Result<std::size_t>::failure(upper_variations.error());
124 }
125 int count = lower_variations.value() - upper_variations.value();
126 if (polynomial_sign_at(square_free, upper) == 0) --count;
127 if (count < 0) {
128 return Result<std::size_t>::failure(
129 CasErrc::InternalInvariant,
130 "Sturm variation produced a negative open-interval count",
131 operation);
132 }
133 return Result<std::size_t>::success(static_cast<std::size_t>(count));
134}

136Rational strict_cauchy_bound(const Polynomial<Rational>& polynomial) {
137 Rational maximum(0);
138 const Rational leading = polynomial.lead_coeff().abs();
139 for (int degree = 0; degree < polynomial.degree(); ++degree) {
140 const Rational ratio = polynomial.coeffs[degree].abs() / leading;
141 if (maximum < ratio) maximum = ratio;
142 }
143 return Rational(2) + maximum;
144}

148Result<std::size_t> count_real_roots_exact(
149 const Polynomial<Rational>& polynomial,
150 const Rational& lower,
151 const Rational& upper,
152 ComputationContext& context,
153 const std::string& operation) {
154 if (upper < lower) {
155 return Result<std::size_t>::failure(
156 CasErrc::InvalidArgument,
157 "real-root interval bounds must be ordered",
158 operation);
159 }
160 try {
161 auto step = context.consume_steps(1, operation);
162 if (!step) return Result<std::size_t>::failure(step.error());
163 if (polynomial.degree() <= 0) {
164 return Result<std::size_t>::success(0);
165 }
166 auto square_free = polynomial.square_free_part().make_monic();
167 if (lower == upper) {
168 return Result<std::size_t>::success(
169 polynomial_sign_at(square_free, lower) == 0 ? 1U : 0U);
170 }
171 auto sequence = make_sturm_sequence(square_free, context, operation);
172 if (!sequence) return Result<std::size_t>::failure(sequence.error());
173 auto interior = count_open_roots(
174 square_free, sequence.value(), lower, upper, context, operation);
175 if (!interior) return interior;
176 std::size_t count = interior.value();
177 if (polynomial_sign_at(square_free, lower) == 0) ++count;
178 if (polynomial_sign_at(square_free, upper) == 0) ++count;
179 return Result<std::size_t>::success(count);
180 } catch (const std::bad_alloc&) {
181 return Result<std::size_t>::failure(
182 CasErrc::ResourceLimit,
183 "Sturm root counting allocation failed",
184 operation);
185 } catch (const std::exception& error) {
186 return Result<std::size_t>::failure(
187 CasErrc::InternalInvariant, error.what(), operation);
188 }
189}

191Result<std::vector<RationalInterval>> isolate_real_roots_exact(
192 const Polynomial<Rational>& polynomial,
193 ComputationContext& context,
194 const std::string& operation) {
195 try {
196 auto step = context.consume_steps(1, operation);
197 if (!step) {
198 return Result<std::vector<RationalInterval>>::failure(step.error());
199 }
200 if (polynomial.degree() <= 0) {
201 return Result<std::vector<RationalInterval>>::success({});
202 }
203 if (polynomial.degree() == 1) {
204 const Rational root =
205 -polynomial.coeffs[0] / polynomial.coeffs[1];
206 return Result<std::vector<RationalInterval>>::success(
207 {{root, root}});
208 }
209 auto square_free = polynomial.square_free_part().make_monic();
210 auto sequence = make_sturm_sequence(square_free, context, operation);
211 if (!sequence) {
212 return Result<std::vector<RationalInterval>>::failure(
213 sequence.error());
214 }
215
216 const Rational bound = strict_cauchy_bound(square_free);
217 struct Pending {
218 Rational lower;
219 Rational upper;
220 std::size_t count;
221 };
222 auto total = count_open_roots(
223 square_free, sequence.value(), -bound, bound, context, operation);
224 if (!total) {
225 return Result<std::vector<RationalInterval>>::failure(total.error());
226 }
227
228 std::vector<Pending> pending;
229 std::vector<RationalInterval> isolated;
230 if (total.value() != 0) pending.push_back({-bound, bound, total.value()});
231 isolated.reserve(total.value());
232
233 while (!pending.empty()) {
234 step = context.consume_steps(1, operation);
235 if (!step) {
236 return Result<std::vector<RationalInterval>>::failure(step.error());
237 }
238 Pending current = std::move(pending.back());
239 pending.pop_back();
240 if (current.count == 1) {
241 while (polynomial_sign_at(square_free, current.lower) == 0 ||
242 polynomial_sign_at(square_free, current.upper) == 0) {
243 const Rational midpoint =
244 (current.lower + current.upper) / Rational(2);
245 if (polynomial_sign_at(square_free, midpoint) == 0) {
246 current.lower = midpoint;
247 current.upper = midpoint;
248 break;
249 }
250 if (polynomial_sign_at(square_free, current.lower) == 0) {
251 auto right = count_open_roots(
252 square_free, sequence.value(), midpoint,
253 current.upper, context, operation);
254 if (!right) {
255 return Result<std::vector<RationalInterval>>::failure(
256 right.error());
257 }
258 if (right.value() == 1) {
259 current.lower = midpoint;
260 } else {
261 current.upper = midpoint;
262 }
263 } else {
264 auto left = count_open_roots(
265 square_free, sequence.value(), current.lower,
266 midpoint, context, operation);
267 if (!left) {
268 return Result<std::vector<RationalInterval>>::failure(
269 left.error());
270 }
271 if (left.value() == 1) {
272 current.upper = midpoint;
273 } else {
274 current.lower = midpoint;
275 }
276 }
277 }
278 isolated.emplace_back(current.lower, current.upper);
279 continue;
280 }
281
282 const Rational midpoint =
283 (current.lower + current.upper) / Rational(2);
284 const bool midpoint_is_root =
285 polynomial_sign_at(square_free, midpoint) == 0;
286 auto left = count_open_roots(
287 square_free, sequence.value(), current.lower, midpoint,
288 context, operation);
289 if (!left) {
290 return Result<std::vector<RationalInterval>>::failure(left.error());
291 }
292 auto right = count_open_roots(
293 square_free, sequence.value(), midpoint, current.upper,
294 context, operation);
295 if (!right) {
296 return Result<std::vector<RationalInterval>>::failure(right.error());
297 }
298 const std::size_t accounted = left.value() + right.value() +
299 (midpoint_is_root ? 1U : 0U);
300 if (accounted != current.count) {
301 return Result<std::vector<RationalInterval>>::failure(
302 CasErrc::InternalInvariant,
303 "Sturm subdivision did not preserve the root count",
304 operation);
305 }
306 if (right.value() != 0) {
307 pending.push_back({midpoint, current.upper, right.value()});
308 }
309 if (midpoint_is_root) isolated.emplace_back(midpoint, midpoint);
310 if (left.value() != 0) {
311 pending.push_back({current.lower, midpoint, left.value()});
312 }
313 }
314
315 std::sort(isolated.begin(), isolated.end(),
316 [](const RationalInterval& left,
317 const RationalInterval& right) {
318 if (left.first != right.first) {
319 return left.first < right.first;
320 }
321 return left.second < right.second;
322 });
323 if (isolated.size() != total.value()) {
324 return Result<std::vector<RationalInterval>>::failure(
325 CasErrc::InternalInvariant,
326 "Sturm isolation lost a distinct real root",
327 operation);
328 }
329 return Result<std::vector<RationalInterval>>::success(
330 std::move(isolated));
331 } catch (const std::bad_alloc&) {
332 return Result<std::vector<RationalInterval>>::failure(
333 CasErrc::ResourceLimit,
334 "Sturm root isolation allocation failed",
335 operation);
336 } catch (const std::exception& error) {
337 return Result<std::vector<RationalInterval>>::failure(
338 CasErrc::InternalInvariant, error.what(), operation);
339 }
340}
