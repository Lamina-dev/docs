139RewriteBudget::Shape RewriteBudget::measure_shape(
140 const SymbolicNodePtr& node, std::size_t path_depth, std::size_t active_depth) {
141 check_step_access();
142 auto step = context_.consume_steps(1, operation_);
143 if (!step) { throw step.error(); }
144 if (!node) { return {}; }
145 {
146 const auto found = shapes_.find(node.get());
147 if (found != shapes_.end()) {
148 const auto owner = found->second.owner.lock();
149 if (owner && owner.get() == node.get()) {
150 const auto shape = found->second.shape;
151 require_shape(shape, path_depth, active_depth);
152 return shape;
153 }
154 shapes_.erase(found);
155 }
156 }
157 if (path_depth >= max_depth_) {
158 throw CasError{CasErrc::ResourceLimit, kDepthMessage, operation_};
159 }
160 Shape shape;
161 {
162 MeasurementRecursion recursion(context_, operation_);
163 Measurement measurement(*this, path_depth, active_depth);
164 node->accept(measurement);
165 shape = measurement.shape;
166 }
167 require_shape(shape, path_depth, active_depth);
168 if (max_nodes_ != 0) {
169 if (shapes_.size() >= max_nodes_) {
170 auto cleared = context_.consume_steps(shapes_.size(), operation_);
171 if (!cleared) { throw cleared.error(); }
172 shapes_.clear();
173 }
174 shapes_.emplace(node.get(), Entry{node, shape});
175 }
176 return shape;
177}

188std::size_t RewriteBudget::append_size(std::size_t current, std::size_t child) {
189 auto access = context_.consume_steps(0, operation_);
190 if (!access) { throw access.error(); }
191 if (child != 0 && current == 0) { consume(); }
192 if (current > max_nodes_ || child > max_nodes_ - current) {
193 throw CasError{CasErrc::ResourceLimit, kGrowthMessage, operation_};
194 }
195 return current + child;
196}

198std::size_t RewriteBudget::input_nodes(
199 const std::shared_ptr<const SymbolicNode>& lhs,
200 const std::shared_ptr<const SymbolicNode>& rhs,
201 ComputationContext& context, std::size_t max_depth, const char* operation) {
202 RewriteBudget budget(context, max_depth, context.limits().max_ast_nodes, operation);
203 const auto left = budget.measure(lhs);
204 const auto right = budget.measure(rhs);
205 if (right > std::numeric_limits<std::size_t>::max() - left) {
206 throw CasError{CasErrc::ResourceLimit, kGrowthMessage, operation};
207 }
208 return left + right;
209}
