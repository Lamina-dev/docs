7/**
8 * @brief 分段函数节点,表示条件分支表达式.
9 *
10 * 存储有序的 (表达式, 条件) 对列表和可选的默认表达式.
11 * 条件应为 RelationalNode 或 LogicalNode 表达式.
12 */
13class PiecewiseNode : public SymbolicNode {
14public:
15 /** @brief 分支结构,包含表达式和对应条件. */
16 struct Branch {
17 std::shared_ptr<const SymbolicNode> expression; ///< 分支值
18 std::shared_ptr<const SymbolicNode> condition; ///< 条件(RelationalNode 或 LogicalNode)
19 };
20
21private:
22 LMCAS_AST_NODE_FACTORY_FRIEND;
23
24 const std::vector<Branch> branches_;
25 const std::shared_ptr<const SymbolicNode> default_expr_;
26
27 /**
28 * @brief 构造分段函数节点.
29 * @param br 分支列表
30 * @param def 默认表达式(可为 nullptr)
31 */
32 PiecewiseNode(std::vector<Branch> br, std::shared_ptr<const SymbolicNode> def = nullptr)
33 : branches_(std::move(br)), default_expr_(std::move(def)) {
34 if (branches_.empty()) {
35 throw std::invalid_argument("PiecewiseNode requires at least one branch");
36 }
37 for (const auto& branch : branches_) {
38 if (!branch.expression || !branch.condition) {
39 throw std::invalid_argument("PiecewiseNode branch fields cannot be null");
40 }
41 }
42 }
43
44public:
45 const std::vector<Branch>& branches() const noexcept { return branches_; }
46 const std::shared_ptr<const SymbolicNode>& default_expr() const noexcept {
47 return default_expr_;
48 }
49
50 int type_priority() const override { return 7; }
51
52protected:
53 std::size_t compute_hash() const override {
54 std::size_t seed = 0;
55 hash_combine(seed, type_priority());
56 for (const auto& b : branches_) {
57 hash_combine(seed, b.expression->hash());
58 hash_combine(seed, b.condition->hash());
59 }
60 if (default_expr_) {
61 hash_combine(seed, default_expr_->hash());
62 }
63 return seed;
64 }
65
66 int compare_same_type(const SymbolicNode& other) const override {
67 const auto& o = static_cast<const PiecewiseNode&>(other);
68 if (branches_.size() != o.branches_.size()) {
69 return branches_.size() < o.branches_.size() ? -1 : 1;
70 }
71 for (size_t i = 0; i < branches_.size(); ++i) {
72 int cmp = branches_[i].expression->compare(*o.branches_[i].expression);
73 if (cmp != 0) return cmp;
74 cmp = branches_[i].condition->compare(*o.branches_[i].condition);
75 if (cmp != 0) return cmp;
76 }
77 bool has_def = (default_expr_ != nullptr);
78 bool o_has_def = (o.default_expr_ != nullptr);
79 if (has_def != o_has_def) return has_def ? 1 : -1;
80 if (has_def && o_has_def) {
81 return default_expr_->compare(*o.default_expr_);
82 }
83 return 0;
84 }
85
86public:
87 void accept(LMCAS::detail::SymbolicVisitor& visitor) const override { LMCAS::detail::SymbolicVisitor::DepthGuard guard(visitor); visitor.visit(*this); }
88
89 std::shared_ptr<const SymbolicNode> clone() const override {
90 std::vector<Branch> new_branches;
91 new_branches.reserve(branches_.size());
92 for (const auto& b : branches_) {
93 new_branches.push_back({b.expression->clone(), b.condition->clone()});
94 }
95 auto new_def = default_expr_ ? default_expr_->clone() : nullptr;
96 return LMCAS::detail::make_node<PiecewiseNode>(std::move(new_branches), std::move(new_def));
97 }
98};

481/** A finite, unordered set of structurally distinct expressions. */
482class FiniteSetNode : public SymbolicNode {
483private:
484 LMCAS_AST_NODE_FACTORY_FRIEND;
485 const std::vector<std::shared_ptr<const SymbolicNode>> elements_;
486
487 explicit FiniteSetNode(std::vector<std::shared_ptr<const SymbolicNode>> elements)
488 : elements_(canonicalize(std::move(elements))) {}
489
490 static std::vector<std::shared_ptr<const SymbolicNode>> canonicalize(
491 std::vector<std::shared_ptr<const SymbolicNode>> elements) {
492 for (const auto& element : elements) {
493 if (!element) throw std::invalid_argument("finite set elements cannot be null");
494 }
495 std::sort(elements.begin(), elements.end(),
496 [](const auto& lhs, const auto& rhs) { return lhs->compare(*rhs) < 0; });
497 elements.erase(std::unique(elements.begin(), elements.end(),
498 [](const auto& lhs, const auto& rhs) {
499 return lhs->equals(*rhs);
500 }),
501 elements.end());
502 return elements;
503 }
504
505public:
506 const std::vector<std::shared_ptr<const SymbolicNode>>& elements() const noexcept {
507 return elements_;
508 }
509 bool contains(const SymbolicNode& element) const {
510 return std::any_of(elements_.begin(), elements_.end(),
511 [&](const auto& candidate) { return candidate->equals(element); });
512 }
513 int type_priority() const override { return 104; }
514
515protected:
516 std::size_t compute_hash() const override {
517 std::size_t seed = static_cast<std::size_t>(type_priority());
518 for (const auto& element : elements_) hash_combine(seed, element->hash());
519 return seed;
520 }
521 int compare_same_type(const SymbolicNode& other) const override {
522 const auto& set = static_cast<const FiniteSetNode&>(other);
523 const auto count = std::min(elements_.size(), set.elements_.size());
524 for (std::size_t index = 0; index < count; ++index) {
525 const int comparison = elements_[index]->compare(*set.elements_[index]);
526 if (comparison != 0) return comparison;
527 }
528 if (elements_.size() == set.elements_.size()) return 0;
529 return elements_.size() < set.elements_.size() ? -1 : 1;
530 }
531
532public:
533 void accept(LMCAS::detail::SymbolicVisitor& visitor) const override {
534 LMCAS::detail::SymbolicVisitor::DepthGuard guard(visitor);
535 visitor.visit(*this);
536 }
537 std::shared_ptr<const SymbolicNode> clone() const override {
538 std::vector<std::shared_ptr<const SymbolicNode>> elements;
539 elements.reserve(elements_.size());
540 for (const auto& element : elements_) elements.push_back(element->clone());
541 return LMCAS::detail::make_node<FiniteSetNode>(std::move(elements));
542 }
543};

620 hash_combine(seed, element_->hash());
621 hash_combine(seed, set_->hash());
622 return seed;
623 }

640/** A symbolic value paired with its canonical dimension and display scale. */
641class QuantityNode : public SymbolicNode {
642private:
643 LMCAS_AST_NODE_FACTORY_FRIEND;
644 const std::shared_ptr<const SymbolicNode> value_;
645 const LMCAS::DimensionSignature dimension_;
646 const Rational scale_to_base_;
647 const std::string display_unit_;
648
649 QuantityNode(std::shared_ptr<const SymbolicNode> value,
650 LMCAS::DimensionSignature dimension,
651 Rational scale_to_base, std::string display_unit)
652 : value_(std::move(value)), dimension_(std::move(dimension)),
653 scale_to_base_(std::move(scale_to_base)),
654 display_unit_(std::move(display_unit)) {
655 if (!value_) throw std::invalid_argument("quantity value cannot be null");
656 if (scale_to_base_.is_zero()) throw std::invalid_argument("quantity scale cannot be zero");
657 }
658
659public:
660 const auto& value() const noexcept { return value_; }
661 const auto& dimension() const noexcept { return dimension_; }
662 const Rational& scale_to_base() const noexcept { return scale_to_base_; }
663 const std::string& display_unit() const noexcept { return display_unit_; }
664 int type_priority() const override { return 107; }
665
666protected:
667 std::size_t compute_hash() const override {
668 std::size_t seed = static_cast<std::size_t>(type_priority());
669 hash_combine(seed, value_->hash());
670 hash_combine(seed, std::hash<std::string>{}(dimension_.to_string()));
671 hash_combine(seed, scale_to_base_.hash());
672 hash_combine(seed, std::hash<std::string>{}(display_unit_));
673 return seed;
674 }
675 int compare_same_type(const SymbolicNode& other) const override {
676 const auto& quantity = static_cast<const QuantityNode&>(other);
677 int comparison = dimension_.to_string().compare(quantity.dimension_.to_string());
678 if (comparison != 0) return comparison;
679 if (scale_to_base_ < quantity.scale_to_base_) return -1;
680 if (quantity.scale_to_base_ < scale_to_base_) return 1;
681 comparison = display_unit_.compare(quantity.display_unit_);
682 return comparison != 0 ? comparison : value_->compare(*quantity.value_);
683 }
684
685public:
686 void accept(LMCAS::detail::SymbolicVisitor& visitor) const override {
687 LMCAS::detail::SymbolicVisitor::DepthGuard guard(visitor);
688 visitor.visit(*this);
689 }
690 std::shared_ptr<const SymbolicNode> clone() const override {
691 return LMCAS::detail::make_node<QuantityNode>(
692 value_->clone(), dimension_, scale_to_base_, display_unit_);
693 }
694};

703inline std::shared_ptr<const SymbolicNode> SymbolicFactory::create_add(std::vector<std::shared_ptr<const SymbolicNode>> ops) {
704 if (ops.empty()) return create_number(::LMCAS::BigInt(0));
705 for (const auto& op : ops) {
706 if (!op) {
707 throw std::invalid_argument("create_add operand cannot be null");
708 }
709 }
710 if (ops.size() == 1) return ops[0];
711
712 std::vector<std::shared_ptr<const SymbolicNode>> flat_ops;
713 flat_ops.reserve(ops.size());
714 for (const auto& op : ops) {
715 if (op->is_zero()) continue;
716 if (auto add = std::dynamic_pointer_cast<const AddNode>(op)) {
717 flat_ops.insert(flat_ops.end(), add->operands().begin(), add->operands().end());
718 } else {
719 flat_ops.push_back(op);
720 }
721 }
722
723 if (flat_ops.empty()) return create_number(::LMCAS::BigInt(0));
724 if (flat_ops.size() == 1) return flat_ops[0];
725 return LMCAS::detail::make_node<AddNode>(std::move(flat_ops));
726}

728inline std::shared_ptr<const SymbolicNode> SymbolicFactory::create_multiply(std::vector<std::shared_ptr<const SymbolicNode>> ops) {
729 if (ops.empty()) return create_number(::LMCAS::BigInt(1));
730
731 for (const auto& op : ops) {
732 if (!op) {
733 throw std::invalid_argument("create_multiply operand cannot be null");
734 }
735 if (op->is_zero()) return op;
736 }
737
738 if (ops.size() == 1) return ops[0];
739
740 std::vector<std::shared_ptr<const SymbolicNode>> flat_ops;
741 flat_ops.reserve(ops.size());
742 for (const auto& op : ops) {
743 if (op->is_one()) continue;
744 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(op)) {
745 flat_ops.insert(flat_ops.end(), mul->operands().begin(), mul->operands().end());
746 } else {
747 flat_ops.push_back(op);
748 }
749 }
750
751 if (flat_ops.empty()) return create_number(::LMCAS::BigInt(1));
752 if (flat_ops.size() == 1) return flat_ops[0];
753 return LMCAS::detail::make_node<MultiplyNode>(std::move(flat_ops));
754}

756inline std::shared_ptr<const SymbolicNode> SymbolicFactory::create_power(std::shared_ptr<const SymbolicNode> base, std::shared_ptr<const SymbolicNode> exponent) {
757 if (!base || !exponent) {
758 throw std::invalid_argument("create_power operands cannot be null");
759 }
760 if (exponent->is_zero()) {
761 /// x^0 的化简需要 base 的定义域证明;base=0 时保留 PowerNode.
762 if (base->is_zero()) {
763 return LMCAS::detail::make_node<PowerNode>(std::move(base), std::move(exponent));
764 }
765 if (auto num = std::dynamic_pointer_cast<const NumberNode>(base)) {
766 if (!num->is_zero()) return create_number(::LMCAS::BigInt(1));
767 }
768 if (auto complex = std::dynamic_pointer_cast<const ComplexNode>(base)) {
769 auto real_num = std::dynamic_pointer_cast<const NumberNode>(complex->real());
770 auto imag_num = std::dynamic_pointer_cast<const NumberNode>(complex->imag());
771 if ((real_num && !real_num->is_zero()) || (imag_num && !imag_num->is_zero())) {
772 return create_number(::LMCAS::BigInt(1));
773 }
774 }
775 return LMCAS::detail::make_node<PowerNode>(std::move(base), std::move(exponent));
776 }
777 if (exponent->is_one()) return base;
778 if (base->is_zero()) {
779 /// 指数严格为正时 0^x 化简为 0,其余指数保留 PowerNode 及其定义域条件.
780 if (exponent->is_positive()) return create_number(::LMCAS::BigInt(0));
781 return LMCAS::detail::make_node<PowerNode>(std::move(base), std::move(exponent));
782 }
783 if (base->is_one()) return create_number(::LMCAS::BigInt(1));
784 return LMCAS::detail::make_node<PowerNode>(std::move(base), std::move(exponent));
785}

787inline std::shared_ptr<const SymbolicNode> SymbolicFactory::create_complex(std::shared_ptr<const SymbolicNode> real, std::shared_ptr<const SymbolicNode> imag) {
788 if (!real || !imag) {
789 throw std::invalid_argument("create_complex operands cannot be null");
790 }
791 if (imag->is_zero()) {
792 return real;
793 }
794 return LMCAS::detail::make_node<ComplexNode>(std::move(real), std::move(imag));
795}
