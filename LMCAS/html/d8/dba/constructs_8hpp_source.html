5/**
6 * @brief 分段函数节点，表示条件分支表达式。
7 *
8 * 存储有序的 (表达式, 条件) 对列表和可选的默认表达式。
9 * 条件应为 RelationalNode 或 LogicalNode 表达式。
10 */
11class PiecewiseNode : public SymbolicNode {
12public:
13 /** @brief 分支结构，包含表达式和对应条件。 */
14 struct Branch {
15 std::shared_ptr<const SymbolicNode> expression; ///< 分支值
16 std::shared_ptr<const SymbolicNode> condition; ///< 条件（RelationalNode 或 LogicalNode）
17 };
18
19private:
20 LAMINA_AST_NODE_FACTORY_FRIEND;
21
22 const std::vector<Branch> branches_;
23 const std::shared_ptr<const SymbolicNode> default_expr_;
24
25 /**
26 * @brief 构造分段函数节点。
27 * @param br 分支列表
28 * @param def 默认表达式（可为 nullptr）
29 */
30 PiecewiseNode(std::vector<Branch> br, std::shared_ptr<const SymbolicNode> def = nullptr)
31 : branches_(std::move(br)), default_expr_(std::move(def)) {
32 if (branches_.empty()) {
33 throw std::invalid_argument("PiecewiseNode requires at least one branch");
34 }
35 for (const auto& branch : branches_) {
36 if (!branch.expression || !branch.condition) {
37 throw std::invalid_argument("PiecewiseNode branch fields cannot be null");
38 }
39 }
40 }
41
42public:
43 const std::vector<Branch>& branches() const noexcept { return branches_; }
44 const std::shared_ptr<const SymbolicNode>& default_expr() const noexcept {
45 return default_expr_;
46 }
47
48 int type_priority() const override { return 7; }
49
50protected:
51 std::size_t compute_hash() const override {
52 std::size_t seed = 0;
53 hash_combine(seed, type_priority());
54 for (const auto& b : branches_) {
55 hash_combine(seed, b.expression->hash());
56 hash_combine(seed, b.condition->hash());
57 }
58 if (default_expr_) {
59 hash_combine(seed, default_expr_->hash());
60 }
61 return seed;
62 }
63
64 int compare_same_type(const SymbolicNode& other) const override {
65 const auto& o = static_cast<const PiecewiseNode&>(other);
66 if (branches_.size() != o.branches_.size()) {
67 return branches_.size() < o.branches_.size() ? -1 : 1;
68 }
69 for (size_t i = 0; i < branches_.size(); ++i) {
70 int cmp = branches_[i].expression->compare(*o.branches_[i].expression);
71 if (cmp != 0) return cmp;
72 cmp = branches_[i].condition->compare(*o.branches_[i].condition);
73 if (cmp != 0) return cmp;
74 }
75 bool has_def = (default_expr_ != nullptr);
76 bool o_has_def = (o.default_expr_ != nullptr);
77 if (has_def != o_has_def) return has_def ? 1 : -1;
78 if (has_def && o_has_def) {
79 return default_expr_->compare(*o.default_expr_);
80 }
81 return 0;
82 }
83
84public:
85 void accept(lamina::detail::SymbolicVisitor& visitor) const override { lamina::detail::SymbolicVisitor::DepthGuard guard(visitor); visitor.visit(*this); }
86
87 std::shared_ptr<const SymbolicNode> clone() const override {
88 std::vector<Branch> new_branches;
89 new_branches.reserve(branches_.size());
90 for (const auto& b : branches_) {
91 new_branches.push_back({b.expression->clone(), b.condition->clone()});
92 }
93 auto new_def = default_expr_ ? default_expr_->clone() : nullptr;
94 return lamina::detail::make_node<PiecewiseNode>(std::move(new_branches), std::move(new_def));
95 }
96};

479/** A finite, unordered set of structurally distinct expressions. */
480class FiniteSetNode : public SymbolicNode {
481private:
482 LAMINA_AST_NODE_FACTORY_FRIEND;
483 const std::vector<std::shared_ptr<const SymbolicNode>> elements_;
484
485 explicit FiniteSetNode(std::vector<std::shared_ptr<const SymbolicNode>> elements)
486 : elements_(canonicalize(std::move(elements))) {}
487
488 static std::vector<std::shared_ptr<const SymbolicNode>> canonicalize(
489 std::vector<std::shared_ptr<const SymbolicNode>> elements) {
490 for (const auto& element : elements) {
491 if (!element) throw std::invalid_argument("finite set elements cannot be null");
492 }
493 std::sort(elements.begin(), elements.end(),
494 [](const auto& lhs, const auto& rhs) { return lhs->compare(*rhs) < 0; });
495 elements.erase(std::unique(elements.begin(), elements.end(),
496 [](const auto& lhs, const auto& rhs) {
497 return lhs->equals(*rhs);
498 }),
499 elements.end());
500 return elements;
501 }
502
503public:
504 const std::vector<std::shared_ptr<const SymbolicNode>>& elements() const noexcept {
505 return elements_;
506 }
507 bool contains(const SymbolicNode& element) const {
508 return std::any_of(elements_.begin(), elements_.end(),
509 [&](const auto& candidate) { return candidate->equals(element); });
510 }
511 int type_priority() const override { return 104; }
512
513protected:
514 std::size_t compute_hash() const override {
515 std::size_t seed = static_cast<std::size_t>(type_priority());
516 for (const auto& element : elements_) hash_combine(seed, element->hash());
517 return seed;
518 }
519 int compare_same_type(const SymbolicNode& other) const override {
520 const auto& set = static_cast<const FiniteSetNode&>(other);
521 const auto count = std::min(elements_.size(), set.elements_.size());
522 for (std::size_t index = 0; index < count; ++index) {
523 const int comparison = elements_[index]->compare(*set.elements_[index]);
524 if (comparison != 0) return comparison;
525 }
526 if (elements_.size() == set.elements_.size()) return 0;
527 return elements_.size() < set.elements_.size() ? -1 : 1;
528 }
529
530public:
531 void accept(lamina::detail::SymbolicVisitor& visitor) const override {
532 lamina::detail::SymbolicVisitor::DepthGuard guard(visitor);
533 visitor.visit(*this);
534 }
535 std::shared_ptr<const SymbolicNode> clone() const override {
536 std::vector<std::shared_ptr<const SymbolicNode>> elements;
537 elements.reserve(elements_.size());
538 for (const auto& element : elements_) elements.push_back(element->clone());
539 return lamina::detail::make_node<FiniteSetNode>(std::move(elements));
540 }

639class QuantityNode : public SymbolicNode {
640private:
641 LAMINA_AST_NODE_FACTORY_FRIEND;

643 const lamina::DimensionSignature dimension_;

647 QuantityNode(std::shared_ptr<const SymbolicNode> value,
648 lamina::DimensionSignature dimension,
649 Rational scale_to_base, std::string display_unit)
650 : value_(std::move(value)), dimension_(std::move(dimension)),
651 scale_to_base_(std::move(scale_to_base)),
652 display_unit_(std::move(display_unit)) {
653 if (!value_) throw std::invalid_argument("quantity value cannot be null");
654 if (scale_to_base_.is_zero()) throw std::invalid_argument("quantity scale cannot be zero");
655 }
656
657public:
658 const auto& value() const noexcept { return value_; }
659 const auto& dimension() const noexcept { return dimension_; }
660 const Rational& scale_to_base() const noexcept { return scale_to_base_; }

663
664protected:
665 std::size_t compute_hash() const override {
666 std::size_t seed = static_cast<std::size_t>(type_priority());
667 hash_combine(seed, value_->hash());
668 hash_combine(seed, std::hash<std::string>{}(dimension_.to_string()));
669 hash_combine(seed, scale_to_base_.hash());
670 hash_combine(seed, std::hash<std::string>{}(display_unit_));
671 return seed;
672 }
673 int compare_same_type(const SymbolicNode& other) const override {
674 const auto& quantity = static_cast<const QuantityNode&>(other);
675 int comparison = dimension_.to_string().compare(quantity.dimension_.to_string());
676 if (comparison != 0) return comparison;
677 if (scale_to_base_ < quantity.scale_to_base_) return -1;
678 if (quantity.scale_to_base_ < scale_to_base_) return 1;
679 comparison = display_unit_.compare(quantity.display_unit_);
680 return comparison != 0 ? comparison : value_->compare(*quantity.value_);
681 }
682
683public:
684 void accept(lamina::detail::SymbolicVisitor& visitor) const override {
685 lamina::detail::SymbolicVisitor::DepthGuard guard(visitor);
686 visitor.visit(*this);
687 }
688 std::shared_ptr<const SymbolicNode> clone() const override {
689 return lamina::detail::make_node<QuantityNode>(
690 value_->clone(), dimension_, scale_to_base_, display_unit_);
691 }
692};

701inline std::shared_ptr<const SymbolicNode> SymbolicFactory::create_add(std::vector<std::shared_ptr<const SymbolicNode>> ops) {
702 if (ops.empty()) return create_number(::BigInt(0));
703 for (const auto& op : ops) {
704 if (!op) {
705 throw std::invalid_argument("create_add operand cannot be null");
706 }
707 }
708 if (ops.size() == 1) return ops[0];
709
710 std::vector<std::shared_ptr<const SymbolicNode>> flat_ops;
711 flat_ops.reserve(ops.size());
712 for (const auto& op : ops) {
713 if (op->is_zero()) continue;
714 if (auto add = std::dynamic_pointer_cast<const AddNode>(op)) {
715 flat_ops.insert(flat_ops.end(), add->operands().begin(), add->operands().end());
716 } else {
717 flat_ops.push_back(op);
718 }
719 }
720
721 if (flat_ops.empty()) return create_number(::BigInt(0));
722 if (flat_ops.size() == 1) return flat_ops[0];
723 return lamina::detail::make_node<AddNode>(std::move(flat_ops));
724}

726inline std::shared_ptr<const SymbolicNode> SymbolicFactory::create_multiply(std::vector<std::shared_ptr<const SymbolicNode>> ops) {
727 if (ops.empty()) return create_number(::BigInt(1));
728
729 for (const auto& op : ops) {
730 if (!op) {
731 throw std::invalid_argument("create_multiply operand cannot be null");

733 if (op->is_zero()) return op;
734 }

736 if (ops.size() == 1) return ops[0];
737
738 std::vector<std::shared_ptr<const SymbolicNode>> flat_ops;

754inline std::shared_ptr<const SymbolicNode> SymbolicFactory::create_power(std::shared_ptr<const SymbolicNode> base, std::shared_ptr<const SymbolicNode> exponent) {
755 if (!base || !exponent) {
756 throw std::invalid_argument("create_power operands cannot be null");
757 }
758 if (exponent->is_zero()) {
759 // 不能盲目把 x^0 折叠成 1：base 可能为 0 时原表达式有定义域条件。
760 if (base->is_zero()) {
761 return lamina::detail::make_node<PowerNode>(std::move(base), std::move(exponent));
762 }
763 if (auto num = std::dynamic_pointer_cast<const NumberNode>(base)) {
764 if (!num->is_zero()) return create_number(::BigInt(1));
765 }
766 if (auto complex = std::dynamic_pointer_cast<const ComplexNode>(base)) {
767 auto real_num = std::dynamic_pointer_cast<const NumberNode>(complex->real());
768 auto imag_num = std::dynamic_pointer_cast<const NumberNode>(complex->imag());
769 if ((real_num && !real_num->is_zero()) || (imag_num && !imag_num->is_zero())) {
770 return create_number(::BigInt(1));
771 }
772 }
773 return lamina::detail::make_node<PowerNode>(std::move(base), std::move(exponent));
774 }
775 if (exponent->is_one()) return base;
776 if (base->is_zero()) {
777 // 0^x = 0 仅当指数严格为正时才安全；x<=0 或符号未知时保留 PowerNode。
778 if (exponent->is_positive()) return create_number(::BigInt(0));
779 return lamina::detail::make_node<PowerNode>(std::move(base), std::move(exponent));
780 }
781 if (base->is_one()) return create_number(::BigInt(1));
782 return lamina::detail::make_node<PowerNode>(std::move(base), std::move(exponent));
783}
