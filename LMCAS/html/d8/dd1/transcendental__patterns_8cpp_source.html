52std::optional<ScaledFunction> scaled_function(
53 const std::shared_ptr<const SymbolicNode>& node, const std::string& var) {
54 auto function = std::dynamic_pointer_cast<const FunctionNode>(node);
55 if (function && invertible_function(*function, var) &&
56 expression_depends_on_variable(node, var)) {
57 return ScaledFunction{function, make_node<NumberNode>(BigInt(1))};
58 }
59 auto product = std::dynamic_pointer_cast<const MultiplyNode>(node);
60 if (!product) {
61 return std::nullopt;
62 }
63 std::vector<std::shared_ptr<const SymbolicNode>> coefficients;
64 std::shared_ptr<const FunctionNode> found;
65 for (const auto& operand : product->operands()) {
66 auto candidate = std::dynamic_pointer_cast<const FunctionNode>(operand);
67 if (candidate && invertible_function(*candidate, var) && !found &&
68 expression_depends_on_variable(operand, var)) {
69 found = candidate;
70 } else {
71 coefficients.push_back(operand);
72 }
73 }
74 if (!found) {
75 return std::nullopt;
76 }
77 auto coefficient = SymbolicFactory::create_multiply(std::move(coefficients));
78 if (expression_depends_on_variable(coefficient, var)) {
79 return std::nullopt;
80 }
81 return ScaledFunction{found, coefficient};
82}

84std::optional<InversePattern> additive_function_pattern(
85 const AddNode& sum, const std::string& var) {
86 std::optional<ScaledFunction> function;
87 std::vector<std::shared_ptr<const SymbolicNode>> constants;
88 for (const auto& operand : sum.operands()) {
89 auto candidate = scaled_function(operand, var);
90 if (candidate && !function) {
91 function = std::move(candidate);
92 continue;
93 }
94 if (expression_depends_on_variable(operand, var)) {
95 return std::nullopt;
96 }
97 constants.push_back(operand);
98 }
99 if (!function) {
100 return std::nullopt;
101 }
102 auto negative = SymbolicExpr::multiply(
103 constant_sum(std::move(constants)), SymbolicExpr::number(-1));
104 return InversePattern{function->function->type(),
105 make_expression_ptr(function->function->arguments()[0]), negative,
106 make_expression_ptr(function->coefficient)};
107}

123std::optional<std::shared_ptr<SymbolicExpr>> lambert_inner(
124 const MultiplyNode& product, const std::string& var) {
125 for (std::size_t i = 0; i < product.operands().size(); ++i) {
126 auto function = std::dynamic_pointer_cast<const FunctionNode>(product.operands()[i]);
127 if (!function || function->type() != FunctionNode::FuncType::Exp ||
128 function->arguments().size() != 1) {
129 continue;
130 }
131 const auto& argument = function->arguments()[0];
132 auto remainder = product_without(product, i);
133 const bool equal = !remainder || !argument
134 ? remainder == argument : remainder->equals(*argument);
135 if (equal && expression_depends_on_variable(remainder, var)) {
136 return make_expression_ptr(remainder);
137 }
138 }
139 return std::nullopt;
140}

142std::optional<LambertWPattern> opposite_exponential_lambert_pattern(
143 const AddNode& sum, const std::string& var) {
144 if (sum.operands().size() != 2) {
145 return std::nullopt;
146 }
147 for (std::size_t index = 0; index < 2; ++index) {
148 auto exponential = std::dynamic_pointer_cast<const FunctionNode>(
149 sum.operands()[index]);
150 if (!exponential ||
151 exponential->type() != FunctionNode::FuncType::Exp ||
152 exponential->arguments().size() != 1) {
153 continue;
154 }
155 const auto& argument = exponential->arguments()[0];
156 const auto& other = sum.operands()[1 - index];
157 if (!argument || !other || !argument->equals(*other) ||
158 !expression_depends_on_variable(argument, var)) {
159 continue;
160 }
161 auto inner = SymbolicExpr::multiply(
162 SymbolicExpr::number(-1), make_expression_ptr(argument))->simplify();
163 return LambertWPattern{std::move(inner), SymbolicExpr::number(1)};
164 }
165 return std::nullopt;

168std::optional<LambertWPattern> additive_lambert_pattern(
169 const AddNode& sum, const std::string& var) {
170 if (auto direct = opposite_exponential_lambert_pattern(sum, var)) {
171 return direct;
172 }
173 std::optional<std::shared_ptr<SymbolicExpr>> found;
174 std::vector<std::shared_ptr<const SymbolicNode>> constants;
175 bool other_variables = false;
176 for (const auto& operand : sum.operands()) {
177 auto product = std::dynamic_pointer_cast<const MultiplyNode>(operand);
178 if (product) {
179 auto candidate = lambert_inner(*product, var);
180 if (candidate && !found) {
181 found = candidate;
182 continue;
183 }
184 }
185 if (!expression_depends_on_variable(operand, var)) {
186 constants.push_back(operand);
187 } else {
188 other_variables = true;
189 }
190 }
191 if (!found || other_variables) {
192 return std::nullopt;
193 }
194 auto rhs = SymbolicExpr::multiply(
195 constant_sum(std::move(constants)), SymbolicExpr::number(-1))->simplify();
196 return LambertWPattern{*found, rhs};

199std::optional<ExpBasePattern> power_pattern(const PowerNode& power, const std::string& var) {
200 if (expression_depends_on_variable(power.base(), var) ||
201 !expression_depends_on_variable(power.exponent(), var)) {
202 return std::nullopt;
203 }
204 auto base = make_expression_ptr(power.base());
205 auto exponent = make_expression_ptr(power.exponent());
206 lmmc_real_t value;
207 if (try_evaluate_numeric(base, value) && value > 0 &&
208 std::abs(value - 1.0) > LMMC_REAL_EPSILON) {
209 return ExpBasePattern{base, exponent, nullptr};
210 }
211 return std::nullopt;
