48inline double polynomial_root_backward_error(
49 const RealPolynomial& coefficients, const Complex& root) {
50 Complex value{coefficients.front(), 0.0};
51 double scale = std::abs(coefficients.front());
52 const double root_magnitude = root.magnitude();
53 for (size_t i = 1; i < coefficients.size(); ++i) {
54 value = value * root + Complex{coefficients[i], 0.0};
55 scale = scale * root_magnitude + std::abs(coefficients[i]);
56 }
57 if (!std::isfinite(value.re) || !std::isfinite(value.im) ||
58 !std::isfinite(scale))
59 return INFINITY;
60 if (scale == 0.0) {
61 // At an exact zero root with zero constant term, both the residual
62 // and its coefficientwise bound vanish: no perturbation is needed.
63 // A nonzero root can instead underflow the bound, which is not proof.
64 return root.re == 0.0 && root.im == 0.0 &&
65 coefficients.back() == 0.0
66 ? 0.0
67 : INFINITY;
68 }
69 return value.magnitude() / scale;
70}

80inline void trim_polynomial(RealPolynomial& polynomial) {
81 const double scale = polynomial_scale(polynomial);
82 const double tolerance = std::max(1.0, scale) * 1e-10;
83 auto first = polynomial.begin();
84 while (first + 1 != polynomial.end() && std::abs(*first) <= tolerance) {
85 ++first;
86 }
87 polynomial.erase(polynomial.begin(), first);
88 if (polynomial.empty()) polynomial.push_back(0.0);
89}

91inline std::pair<RealPolynomial, RealPolynomial> divide_polynomials(
92 const RealPolynomial& dividend, const RealPolynomial& divisor) {
93 if (divisor.empty() || divisor.front() == 0.0 ||
94 dividend.size() < divisor.size()) {
95 return {{0.0}, dividend};
96 }
97
98 RealPolynomial remainder = dividend;
99 RealPolynomial quotient(
100 dividend.size() - divisor.size() + 1, 0.0);
101 for (size_t i = 0; i < quotient.size(); ++i) {
102 const double factor = remainder[i] / divisor.front();
103 quotient[i] = factor;
104 for (size_t j = 0; j < divisor.size(); ++j) {
105 remainder[i + j] -= factor * divisor[j];
106 }
107 }
108 remainder.erase(
109 remainder.begin(), remainder.begin() + quotient.size());
110 if (remainder.empty()) remainder.push_back(0.0);
111 trim_polynomial(quotient);
112 trim_polynomial(remainder);
113 return {std::move(quotient), std::move(remainder)};
114}

116inline RealPolynomial monic_polynomial(RealPolynomial polynomial) {
117 trim_polynomial(polynomial);
118 if (polynomial.empty() || polynomial.front() == 0.0) return {0.0};
119 const double leading = polynomial.front();
120 for (double& coefficient : polynomial) coefficient /= leading;
121 return polynomial;
122}

124inline RealPolynomial polynomial_derivative(
125 const RealPolynomial& polynomial) {
126 if (polynomial.size() <= 1) return {0.0};
127 RealPolynomial derivative;
128 derivative.reserve(polynomial.size() - 1);
129 const size_t degree = polynomial.size() - 1;
130 for (size_t i = 0; i < degree; ++i) {
131 derivative.push_back(
132 polynomial[i] * static_cast<double>(degree - i));
133 }
134 return derivative;
135}

137inline RealPolynomial approximate_polynomial_gcd(
138 RealPolynomial lhs, RealPolynomial rhs) {
139 lhs = monic_polynomial(std::move(lhs));
140 rhs = monic_polynomial(std::move(rhs));
141 for (int iteration = 0; iteration < 8 && rhs.size() > 1; ++iteration) {
142 auto division = divide_polynomials(lhs, rhs);
143 RealPolynomial remainder = std::move(division.second);
144 if (remainder.size() == 1 && remainder.front() == 0.0) {
145 return rhs;
146 }
147 lhs = std::move(rhs);
148 rhs = monic_polynomial(std::move(remainder));
149 }
150 return {1.0};
151}
