95Result<double> numeric_vector_scale_checked(
96 const std::vector<std::shared_ptr<SymbolicExpr>>& expressions,
97 ComputationContext& context,
98 const std::string& operation,
99 std::vector<double>* values)
100{
101 double max_abs = 0.0;
102 if (values) {
103 values->clear();
104 values->reserve(expressions.size());
105 }
106 for (const auto& expression : expressions) {
107 if (!expression || !LMCAS::detail::node(expression)) {
108 return Result<double>::failure(
109 CasErrc::InvalidArgument,
110 "numeric vector contains a null expression",
111 operation);
112 }
113 auto evaluated = evaluate_numeric(
114 *expression, NumericBindings{}, context);
115 if (!evaluated) {
116 if (evaluated.error().code == CasErrc::Cancelled ||
117 evaluated.error().code == CasErrc::ResourceLimit) {
118 return Result<double>::failure(evaluated.error());
119 }
120 return Result<double>::failure(
121 CasErrc::NumericFailure,
122 "vector is not finite numeric in the supported domain",
123 operation);
124 }
125 if (!evaluated.value().is_finite() ||
126 !std::isfinite(evaluated.value().value)) {
127 return Result<double>::failure(
128 CasErrc::NumericFailure,
129 "vector is not finite numeric in the supported domain",
130 operation);
131 }
132 max_abs = std::max(max_abs, std::abs(evaluated.value().value));
133 if (values) {
134 values->push_back(evaluated.value().value);
135 }
136 }
137 return Result<double>::success(max_abs);
138}

164Result<void> checked_nonzero_numeric_or_exact(
165 const std::shared_ptr<SymbolicExpr>& expr,
166 ComputationContext& context,
167 const std::string& operation,
168 const std::string& domain_message,
169 const std::string& inconclusive_message)
170{
171 if (!expr || !LMCAS::detail::node(expr)) {
172 return Result<void>::failure(CasErrc::Inconclusive,
173 inconclusive_message, operation);
174 }
175 auto simplified = expr->simplify();
176 if (!simplified || !LMCAS::detail::node(simplified)) {
177 return Result<void>::failure(CasErrc::Inconclusive,
178 inconclusive_message, operation);
179 }
180 if (simplified->is_zero()) {
181 return Result<void>::failure(CasErrc::DomainError,
182 domain_message, operation);
183 }
184 auto numeric = evaluate_numeric(*simplified, NumericBindings{}, context);
185 if (!numeric) {
186 if (numeric.error().code == CasErrc::Cancelled ||
187 numeric.error().code == CasErrc::ResourceLimit) {
188 return Result<void>::failure(numeric.error());
189 }
190 return Result<void>::failure(CasErrc::Inconclusive,
191 inconclusive_message, operation);
192 }
193 if (!numeric.value().is_finite() ||
194 !std::isfinite(numeric.value().value)) {
195 return Result<void>::failure(CasErrc::NumericFailure,
196 inconclusive_message, operation);
197 }
198 if (numeric.value().value == 0.0) {
199 return Result<void>::failure(CasErrc::DomainError,
200 domain_message, operation);
201 }
202 return Result<void>::success();
203}
