126ExprResult abs(const ExprPtr& expression, ComputationContext& context) {
127 auto step = context.consume_steps(1, kAbsOperation);
128 if (!step) {
129 return ExprResult::failure(step.error());
130 }
131 auto re = real(expression, context);
132 if (!re) {
133 return re;
134 }
135 auto im = imag(expression, context);
136 if (!im) {
137 return im;
138 }
139 try {
140 auto re_squared = SymbolicExpr::power(re.value(), SymbolicExpr::number(2));
141 auto im_squared = SymbolicExpr::power(im.value(), SymbolicExpr::number(2));
142 auto sum = SymbolicExpr::add(re_squared, im_squared);
143 auto result = SymbolicExpr::sqrt(sum)->simplify();
144 if (!result || !LMCAS::detail::node(result)) {
145 return expression_failure(CasErrc::InternalInvariant,
146 "complex absolute value construction failed",
147 kAbsOperation);
148 }
149 return ExprResult::success(std::move(result));
150 } catch (const std::bad_alloc&) {
151 return expression_failure(CasErrc::ResourceLimit,
152 "complex absolute value allocation failed",
153 kAbsOperation);
154 } catch (const std::exception& error) {
155 return expression_failure(CasErrc::InternalInvariant, error.what(),
156 kAbsOperation);
157 }
158}
