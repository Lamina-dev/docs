21static int detect_parity(const std::shared_ptr<SymbolicExpr>& f, const std::string& var) {
22 if (!f) {
23 return 0;
24 }
25 auto neg_x = SymbolicExpr::multiply(SymbolicExpr::number(-1), SymbolicExpr::variable(var));
26 auto f_neg = f->substitute(var, neg_x);
27 if (!f_neg) {
28 return 0;
29 }
30 f_neg = f_neg->simplify();
31 auto f_s = f->simplify();
32 auto diff_even = SymbolicExpr::add(f_neg, SymbolicExpr::multiply(SymbolicExpr::number(-1), f_s));
33 if (diff_even) { diff_even = diff_even->simplify(); if (diff_even->is_zero()) {
34 return 1;
35 } }
36 auto diff_odd = SymbolicExpr::add(f_neg, f_s);
37 if (diff_odd) { diff_odd = diff_odd->simplify(); if (diff_odd->is_zero()) {
38 return -1;
39 } }
40 return 0;
41}

80static std::optional<int> essential_exponent(
81 const std::shared_ptr<SymbolicExpr>& f,
82 const std::shared_ptr<SymbolicExpr>& shift) {
83 auto fn = std::dynamic_pointer_cast<const FunctionNode>(detail::node(f));
84 if (!fn || fn->type() != FunctionNode::FuncType::Exp ||
85 fn->arguments().size() != 1) {
86 return std::nullopt;
87 }
88 auto argument = detail::make_expression_ptr(fn->arguments()[0])->simplify();
89 auto power = std::dynamic_pointer_cast<const PowerNode>(detail::node(argument));
90 if (!power) {
91 return std::nullopt;
92 }
93 auto exponent = exact_small_integer_node(
94 power->exponent(), -std::numeric_limits<int>::max(), std::numeric_limits<int>::max());
95 if (!exponent) {
96 return std::nullopt;
97 }
98 auto base = detail::make_expression_ptr(power->base())->simplify();
99 auto delta = shift->simplify();
100 if (base->compare(delta) == 0 && *exponent < 0) {
101 return -*exponent;
102 }
103 auto nested = std::dynamic_pointer_cast<const PowerNode>(detail::node(base));
104 if (nested && *exponent == -1) {
105 auto m = exact_small_integer_node(
106 nested->exponent(), 1, std::numeric_limits<int>::max());
107 auto nested_base = detail::make_expression_ptr(nested->base())->simplify();
108 if (m && nested_base->compare(delta) == 0) {
109 return m;
110 }
111 }
112 return std::nullopt;
113}

115static LaurentSeriesResult essential_laurent(
116 const std::shared_ptr<SymbolicExpr>& shift, int m, int order_neg,
117 ComputationContext& context) {
118 constexpr const char* operation = "laurent_series_full";
119 auto series = SymbolicExpr::number(1);
120 BigInt factorial(1);
121 for (int k = 1; k <= order_neg / m; ++k) {
122 auto step = context.consume_steps(1, operation);
123 if (!step) {
124 return LaurentSeriesResult::failure(step.error());
125 }
126 factorial *= BigInt(k);
127 auto term = SymbolicExpr::divide(
128 SymbolicExpr::number(1),
129 SymbolicExpr::multiply(
130 SymbolicExpr::number(factorial),
131 SymbolicExpr::power(shift, SymbolicExpr::number(m * k))));
132 series = SymbolicExpr::add(series, term);
133 }
134 return LaurentResult{
135 series->simplify(), SingularityType::Essential, 0,
136 SymbolicExpr::number(m == 1 ? 1 : 0)};
137}

143static Result<void> add_fourier_harmonic(
144 Integrator& integrator, const std::shared_ptr<SymbolicExpr>& f,
145 const std::string& var, const std::shared_ptr<SymbolicExpr>& half_lo,
146 const std::shared_ptr<SymbolicExpr>& half_hi,
147 const std::shared_ptr<SymbolicExpr>& L,
148 const std::shared_ptr<SymbolicExpr>& arg, bool cosine,
149 std::shared_ptr<SymbolicExpr>& result, ComputationContext& context) {
150 auto integrand = SymbolicExpr::multiply(
151 f, cosine ? SymbolicExpr::cos(arg) : SymbolicExpr::sin(arg));
152 auto integrated = integrator.integrate_def_checked(
153 *integrand, var, *half_lo, *half_hi, context);
154 if (!integrated) {
155 return Result<void>::failure(integrated.error());
156 }
157 auto coefficient = SymbolicExpr::divide(
158 detail::make_expression_ptr(integrated.value()), L)->simplify();
159 if (!(detail::node(coefficient) && detail::node(coefficient)->is_zero())) {
160 result = SymbolicExpr::add(
161 result, SymbolicExpr::multiply(
162 coefficient, cosine ? SymbolicExpr::cos(arg) : SymbolicExpr::sin(arg)));
163 }
164 return Result<void>::success();
165}

175static Result<FourierPeriod> prepare_fourier_period(
176 const std::shared_ptr<SymbolicExpr>& period, const std::string& var,
177 ComputationContext& context) {
178 /**
179 * @brief 系数恒等式按输入 binary64 值精确运算，倒数与乘积也保持精确。
180 * pi 的 binary64 近似值仍按该数值处理。
181 */
182 class ExactPeriod final : public detail::SymbolicRewriter {
183 public:
184 void visit(const NumberNode& node) override {
185 const auto* value = std::get_if<lmmc_real_t>(&node.value());
186 set_result(value && std::isfinite(*value)
187 ? detail::make_node<NumberNode>(Rational::from_double(*value)) : current());
188 }
189 } exact_period;
190 auto T = detail::make_expression_ptr(exact_period.rewrite(detail::node(period)));
191 std::optional<detail::AssumptionFacts> assumed;
192 if (context.assumptions()) { assumed.emplace(*context.assumptions()); }
193 const auto& facts = assumed ? static_cast<const FactsQuery&>(*assumed) : detail::no_facts();
194 auto positive_period = detail::query_positive_value(detail::node(T), facts, context);
195 if (!positive_period) { return Result<FourierPeriod>::failure(positive_period.error()); }
196 if (expression_depends_on_variable(detail::node(T), var) ||
197 positive_period.value() != Tribool::True) {
198 return Result<FourierPeriod>::failure(
199 positive_period.value() == Tribool::False ? CasErrc::InvalidArgument : CasErrc::Inconclusive,
200 "Fourier period must be a proved positive constant", "fourier_series");
201 }
202 auto pi = LMCAS::detail::make_expression_ptr(
203 LMCAS::detail::make_node<VariableNode>("pi", true));
204 auto two = SymbolicExpr::number(2);
205 auto L = SymbolicExpr::divide(T, two);
206 auto half_lo = SymbolicExpr::multiply(SymbolicExpr::number(-1), L);
207 auto half_hi = L;
208 auto w = SymbolicExpr::divide(SymbolicExpr::multiply(two, pi), T);
209 return FourierPeriod{std::move(two), std::move(L), std::move(half_lo),
210 std::move(half_hi), std::move(w)};
211}

213static ExpressionResult fourier_constant_coefficient(Integrator& integrator,
214 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
215 const FourierPeriod& period, ComputationContext& context) {
216 auto a0_integral = integrator.integrate_def_checked(
217 *f, var, *period.lower, *period.upper, context);
218 if (!a0_integral) { return ExpressionResult::failure(a0_integral.error()); }
219 auto a0 = SymbolicExpr::divide(
220 LMCAS::detail::make_expression_ptr(a0_integral.value()), period.half_length)->simplify();
221 return SymbolicExpr::divide(a0, period.two);
222}

224static Result<void> fourier_harmonics(Integrator& integrator,
225 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
226 const FourierPeriod& period, const std::shared_ptr<SymbolicExpr>& x,
227 int n_terms, int parity, std::shared_ptr<SymbolicExpr>& result, ComputationContext& context) {
228 for (int k = 1; k <= n_terms; ++k) {
229 auto kw = SymbolicExpr::multiply(SymbolicExpr::number(k), period.frequency);
230 auto arg = SymbolicExpr::multiply(kw, x);
231 if (parity != -1) {
232 auto added = add_fourier_harmonic(integrator, f, var, period.lower,
233 period.upper, period.half_length, arg, true, result, context);
234 if (!added) { return added; }
235 }
236 if (parity != 1) {
237 auto added = add_fourier_harmonic(integrator, f, var, period.lower,
238 period.upper, period.half_length, arg, false, result, context);
239 if (!added) { return added; }
240 }
241 }
242 return Result<void>::success();
243}

245ExpressionResult fourier_series_checked(
246 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
247 const std::shared_ptr<SymbolicExpr>& period, int n_terms,
248 ComputationContext& context) {
249 constexpr const char* operation = "fourier_series";
250 if (!f || !period || var.empty() || n_terms < 0) {
251 return ExpressionResult::failure(
252 CasErrc::InvalidArgument,
253 "Fourier expansion requires an expression, variable, positive period, and non-negative term count",
254 operation);
255 }
256 const std::size_t expansion_terms =
257 1 + 2 * static_cast<std::size_t>(n_terms);
258 if (expansion_terms > context.limits().max_expansion_terms) {
259 return ExpressionResult::failure(
260 CasErrc::ResourceLimit,
261 "Fourier term count exceeds the expansion budget", operation);
262 }
263
264 try {
265 auto access = context.consume_steps(1, operation);
266 if (!access) {
267 return ExpressionResult::failure(access.error());
268 }
269
270 auto prepared = prepare_fourier_period(period, var, context);
271 if (!prepared) { return ExpressionResult::failure(prepared.error()); }
272
273 int parity = detect_parity(f, var);
274 Integrator integrator;
275 auto x = SymbolicExpr::variable(var);
276
277 auto constant = fourier_constant_coefficient(integrator, f, var, prepared.value(), context);
278 if (!constant) { return constant; }
279 auto result = std::move(constant.value());
280 auto harmonics = fourier_harmonics(
281 integrator, f, var, prepared.value(), x, n_terms, parity, result, context);
282 if (!harmonics) { return ExpressionResult::failure(harmonics.error()); }
283 return ExpressionResult::success(result->simplify());
284 } catch (const std::bad_alloc&) {
285 return ExpressionResult::failure(
286 CasErrc::ResourceLimit,
287 "Fourier expansion allocation failed", operation);
288 } catch (const std::exception& error) {
289 return ExpressionResult::failure(
290 CasErrc::InternalInvariant, error.what(), operation);
291 }
292}

339 const std::shared_ptr<SymbolicExpr>& center, const std::string& operation) {
340 if (!result.series || !LMCAS::detail::node(result.series)) {
341 return Result<void>::failure(

344 operation);

457static Result<int> extract_laurent_regular_part(
458 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
459 const std::shared_ptr<SymbolicExpr>& center,
460 const std::shared_ptr<SymbolicExpr>& shift,
461 std::shared_ptr<SymbolicExpr>& regular, ComputationContext& context) {
462 constexpr const char* operation = "laurent_series_full";
463 auto center_number = std::dynamic_pointer_cast<const NumberNode>(detail::node(center));
464 if (!center_number || std::holds_alternative<lmmc_real_t>(center_number->value())) {
465 return Result<int>::failure(CasErrc::Inconclusive,
466 "Laurent center must be an exact number", operation);
467 }
468 Rational point = std::holds_alternative<BigInt>(center_number->value())
469 ? Rational(std::get<BigInt>(center_number->value()))
470 : std::get<Rational>(center_number->value());
471 RationalDecompositionStrategy rational;
472 Polynomial<Rational> numerator(var), denominator(var);
473 auto recognized = rational.extract_rational(*f, var, numerator, denominator);
474 if (!recognized) {
475 return Result<int>::failure(recognized.error());
476 }
477 if (!recognized.value()) {
478 return recognize_regular_laurent(f, var, context);
479 }
480 if (numerator.is_zero()) {
481 return 0;
482 }
483 const Polynomial<Rational> factor({Rational(0) - point, Rational(1)}, var);
484 int pole_order = 0;
485 while (denominator.eval(point) == Rational(0)) {
486 auto step = context.consume_steps(1, operation);
487 if (!step) {
488 return Result<int>::failure(step.error());
489 }
490 if (pole_order >= 64) {
491 return Result<int>::failure(CasErrc::Inconclusive,
492 "Laurent pole order exceeds supported range", operation);
493 }
494 denominator = denominator.div_mod(factor).first;
495 ++pole_order;
496 }
497 if (pole_order > 0) {
498 regular = SymbolicExpr::multiply(
499 SymbolicExpr::power(shift, SymbolicExpr::number(pole_order)), f)->cancel()->simplify();
500 }
501 return pole_order;
502}

504static LaurentSeriesResult laurent_series_full_impl(
505 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
506 const std::shared_ptr<SymbolicExpr>& center, int order_neg, int order_pos,
507 ComputationContext& context) {
508 LaurentResult res{nullptr, SingularityType::Removable, 0, SymbolicExpr::number(0)};
509 if (!f || !center) {
510 return res;
511 }
512
513 auto x = SymbolicExpr::variable(var);
514 auto shift = SymbolicExpr::add(x, SymbolicExpr::multiply(SymbolicExpr::number(-1), center));
515 if (!expression_depends_on_variable(detail::node(center), var)) {
516 if (auto m = essential_exponent(f, shift)) {
517 return essential_laurent(shift, *m, order_neg, context);
518 }
519 }
520
521 std::shared_ptr<SymbolicExpr> regular = f;
522 auto extracted = extract_laurent_regular_part(
523 f, var, center, shift, regular, context);
524 if (!extracted) {
525 return LaurentSeriesResult::failure(extracted.error());
526 }
527 const int pole_order = extracted.value();
528
529 /// 对 regular(解析部分)做 Taylor 展开
530 int taylor_order = order_pos + pole_order + 1;
531 if (taylor_order < 1) {
532 taylor_order = 1;
533 }
534 auto taylor = regular->series(var, center, taylor_order);
535 if (!taylor) {
536 return res;
537 }
538
539 /// Laurent = taylor / (x-c)^pole_order
540 std::shared_ptr<SymbolicExpr> laurent = taylor;
541 if (pole_order > 0) {
542 auto powm = SymbolicExpr::power(shift, SymbolicExpr::number(pole_order));
543 laurent = SymbolicExpr::divide(taylor, powm)->simplify();
544 }
545
546 res.series = laurent->simplify();
547 res.pole_order = pole_order;
548 if (pole_order == 0) {
549 res.singularity = SingularityType::Removable;
550 } else {
551 res.singularity = SingularityType::Pole;
552 }
553
554 /// 留数 = Taylor 展开中 (x-c)^(pole_order-1) 的系数
555 if (pole_order >= 1) {
556 /// a_{m-1} = (1/(m-1)!) lim_{x->c} d^{m-1}/dx^{m-1} [regular]
557 auto deriv = regular;
558 for (int i = 0; i < pole_order - 1; ++i) deriv = deriv->differentiate(var);
559 /// 用极限求值,稳健处理 0/0 形式(如 z/(z(z+1)) 在 z=0).
560 auto limited = limit_expression_checked(
561 deriv, var, center, LimitDirection::Both, context);
562 if (!limited) {
563 return LaurentSeriesResult::failure(limited.error());
564 }
565 auto val = std::move(limited.value());
566 if (!val) {
567 val = deriv->substitute(var, center);
568 }
569 auto fact = BigInt::factorial_checked(BigInt(pole_order - 1), context);
570 if (!fact) {
571 return LaurentSeriesResult::failure(fact.error());
572 }
573 res.residue = SymbolicExpr::divide(val, SymbolicExpr::number(fact.value()))->simplify();
574 }
575
576 return res;
577}

579std::shared_ptr<SymbolicExpr> asymptotic_expand(
580 const std::shared_ptr<SymbolicExpr>& f, const std::string& var, int order) {
581 if (!f || order < 0) {
582 return nullptr;
583 }
584 /// 通过 x = 1/t 替换,在 t=0 处做 Taylor 展开,再回代 t = 1/x,
585 /// 得到按 x 递减幂次的渐近展开.
586 auto x = SymbolicExpr::variable(var);
587 std::string tname = var + "__asym_t";
588 auto t = SymbolicExpr::variable(tname);
589 auto inv_t = SymbolicExpr::divide(SymbolicExpr::number(1), t);
590 auto g = f->substitute(var, inv_t);
591 if (!g) {
592 return nullptr;
593 }
594 g = g->simplify();
595 auto taylor_t = g->series(tname, SymbolicExpr::number(0), order + 1);
596 if (!taylor_t) {
597 return nullptr;
598 }
599 /// 回代 t = 1/x
600 auto inv_x = SymbolicExpr::divide(SymbolicExpr::number(1), x);
601 auto back = taylor_t->substitute(tname, inv_x);
602 if (!back) {
603 return nullptr;
604 }
605 return back->simplify();
606}
