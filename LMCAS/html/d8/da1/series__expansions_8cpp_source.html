18static int detect_parity(const std::shared_ptr<SymbolicExpr>& f, const std::string& var) {
19 if (!f) return 0;
20 auto neg_x = SymbolicExpr::multiply(SymbolicExpr::number(-1), SymbolicExpr::variable(var));
21 auto f_neg = f->substitute(var, neg_x);
22 if (!f_neg) return 0;
23 f_neg = f_neg->simplify();
24 auto f_s = f->simplify();
25 auto diff_even = SymbolicExpr::add(f_neg, SymbolicExpr::multiply(SymbolicExpr::number(-1), f_s));
26 if (diff_even) { diff_even = diff_even->simplify(); if (diff_even->is_zero()) return 1; }
27 auto diff_odd = SymbolicExpr::add(f_neg, f_s);
28 if (diff_odd) { diff_odd = diff_odd->simplify(); if (diff_odd->is_zero()) return -1; }
29 return 0;
30}

71ExpressionResult fourier_series_checked(
72 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
73 const std::shared_ptr<SymbolicExpr>& period, int n_terms,
74 ComputationContext& context) {
75 constexpr const char* operation = "fourier_series";
76 if (!f || !period || var.empty() || n_terms < 0) {
77 return ExpressionResult::failure(
78 CasErrc::InvalidArgument,
79 "Fourier expansion requires an expression, variable, positive period, and non-negative term count",
80 operation);
81 }
82 const std::size_t expansion_terms =
83 1 + 2 * static_cast<std::size_t>(n_terms);
84 if (expansion_terms > context.limits().max_expansion_terms) {
85 return ExpressionResult::failure(
86 CasErrc::ResourceLimit,
87 "Fourier term count exceeds the expansion budget", operation);
88 }
89
90 try {
91 auto access = context.consume_steps(1, operation);
92 if (!access) return ExpressionResult::failure(access.error());
93
94 auto T = period;
95 auto pi = LMCAS::detail::make_expression_ptr(
96 LMCAS::detail::make_node<VariableNode>("pi"));
97 auto two = SymbolicExpr::number(2);
98 auto L = SymbolicExpr::divide(T, two);
99 auto half_lo = SymbolicExpr::multiply(SymbolicExpr::number(-1), L);
100 auto half_hi = L;
101 auto w = SymbolicExpr::divide(
102 SymbolicExpr::multiply(two, pi), T);
103
104 int parity = detect_parity(f, var);
105 Integrator integrator;
106 auto x = SymbolicExpr::variable(var);
107
108 auto a0_integral = integrator.integrate_def_checked(
109 *f, var, *half_lo, *half_hi, context);
110 if (!a0_integral) {
111 return ExpressionResult::failure(a0_integral.error());
112 }
113 auto a0 = SymbolicExpr::divide(
114 LMCAS::detail::make_expression_ptr(a0_integral.value()), L)->simplify();
115 auto result = SymbolicExpr::divide(a0, two);
116
117 for (int k = 1; k <= n_terms; ++k) {
118 auto kw = SymbolicExpr::multiply(SymbolicExpr::number(k), w);
119 auto arg = SymbolicExpr::multiply(kw, x);
120
121 if (parity != -1) {
122 auto integrand = SymbolicExpr::multiply(
123 f, SymbolicExpr::cos(arg));
124 auto integrated = integrator.integrate_def_checked(
125 *integrand, var, *half_lo, *half_hi, context);
126 if (!integrated) {
127 return ExpressionResult::failure(integrated.error());
128 }
129 auto coefficient = SymbolicExpr::divide(
130 LMCAS::detail::make_expression_ptr(integrated.value()), L)->simplify();
131 if (!(LMCAS::detail::node(coefficient) &&
132 LMCAS::detail::node(coefficient)->is_zero())) {
133 result = SymbolicExpr::add(
134 result,
135 SymbolicExpr::multiply(
136 coefficient, SymbolicExpr::cos(arg)));
137 }
138 }
139 if (parity != 1) {
140 auto integrand = SymbolicExpr::multiply(
141 f, SymbolicExpr::sin(arg));
142 auto integrated = integrator.integrate_def_checked(
143 *integrand, var, *half_lo, *half_hi, context);
144 if (!integrated) {
145 return ExpressionResult::failure(integrated.error());
146 }
147 auto coefficient = SymbolicExpr::divide(
148 LMCAS::detail::make_expression_ptr(integrated.value()), L)->simplify();
149 if (!(LMCAS::detail::node(coefficient) &&
150 LMCAS::detail::node(coefficient)->is_zero())) {
151 result = SymbolicExpr::add(
152 result,
153 SymbolicExpr::multiply(
154 coefficient, SymbolicExpr::sin(arg)));
155 }
156 }
157 }
158 return ExpressionResult::success(result->simplify());
159 } catch (const std::bad_alloc&) {
160 return ExpressionResult::failure(
161 CasErrc::ResourceLimit,
162 "Fourier expansion allocation failed", operation);
163 } catch (const std::exception& error) {
164 return ExpressionResult::failure(
165 CasErrc::InternalInvariant, error.what(), operation);
166 }
167}

210LaurentSeriesResult laurent_series_full_checked(
211 const std::shared_ptr<SymbolicExpr>& f,
212 const std::string& var,
213 const std::shared_ptr<SymbolicExpr>& center,
214 int order_neg,
215 int order_pos,
216 ComputationContext& context)
217{
218 const std::string operation = "laurent_series_full";
219 auto input = validate_series_expr_point(f, center, var, context, operation);
220 if (!input) return LaurentSeriesResult::failure(input.error());
221 auto order_check = validate_laurent_orders(order_neg, order_pos, operation);
222 if (!order_check) return LaurentSeriesResult::failure(order_check.error());
223 auto budget = context.consume_steps(static_cast<std::size_t>(order_neg + order_pos + 1) * 16 + 16,
224 operation);
225 if (!budget) return LaurentSeriesResult::failure(budget.error());
226 auto simplified_input = f->simplify();
227 if (!simplified_input || !LMCAS::detail::node(simplified_input)) {
228 return LaurentSeriesResult::failure(
229 CasErrc::Inconclusive,
230 "checked Laurent input could not be simplified in the supported domain",
231 operation);
232 }
233 auto supported_power = supported_laurent_integer_power(
234 LMCAS::detail::node(simplified_input), var);
235 try {
236 auto built = laurent_series_full_impl(
237 f, var, center, order_neg, order_pos, context);
238 if (!built) return built;
239 auto result = std::move(built.value());
240 if (!result.series || !LMCAS::detail::node(result.series)) {
241 return LaurentSeriesResult::failure(
242 CasErrc::Inconclusive,
243 "Laurent series could not be constructed in the supported domain",
244 operation);
245 }
246 if (!result.residue || !LMCAS::detail::node(result.residue)) {
247 return LaurentSeriesResult::failure(
248 CasErrc::InternalInvariant,
249 "Laurent series produced a null residue expression",
250 operation);
251 }
252 if (supported_power && center->is_zero()) {
253 int expected_pole_order =
254 *supported_power < 0 ? -*supported_power : 0;
255 if (result.pole_order != expected_pole_order) {
256 return LaurentSeriesResult::failure(
257 CasErrc::Inconclusive,
258 "Laurent pole order could not be verified",
259 operation);
260 }
261 }
262 return LaurentSeriesResult::success(std::move(result));
263 } catch (const std::bad_alloc&) {
264 return LaurentSeriesResult::failure(CasErrc::ResourceLimit,
265 "allocation failed while calculating Laurent series",
266 operation);
267 } catch (const std::exception& ex) {
268 return LaurentSeriesResult::failure(CasErrc::InternalInvariant,
269 ex.what(),
270 operation);
271 }

285static LaurentSeriesResult laurent_series_full_impl(
286 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
287 const std::shared_ptr<SymbolicExpr>& center, int order_neg, int order_pos,
288 ComputationContext& context) {
289 LaurentResult res{nullptr, SingularityType::Removable, 0, SymbolicExpr::number(0)};
290 if (!f || !center) return res;
291
292 auto x = SymbolicExpr::variable(var);
293 /// (x - center)
294 auto shift = SymbolicExpr::add(x, SymbolicExpr::multiply(SymbolicExpr::number(-1), center));
295
296 /// 寻找极点阶数 m:使 (x-c)^m * f 在 c 处解析(有限).
297 int pole_order = 0;
298 std::shared_ptr<SymbolicExpr> regular = f;
299 {
300 auto test = f;
301 const int MAX_M = (order_neg > 0 ? order_neg : 5);
302 /// 先判断 f 在 center 是否已解析
303 auto f_at = f->substitute(var, center);
304 if (f_at) f_at = f_at->simplify();
305 bool finite = f_at && f_at->is_number();
306 if (!finite) {
307 for (int m = 1; m <= MAX_M; ++m) {
308 auto powm = SymbolicExpr::power(shift, SymbolicExpr::number(m));
309 auto g = SymbolicExpr::multiply(powm, f)->simplify();
310 auto g_at = g->substitute(var, center);
311 if (g_at) g_at = g_at->simplify();
312 if (g_at && g_at->is_number()) {
313 pole_order = m;
314 regular = g; // (x-c)^m f(x),在 c 处解析
315 break;
316 }
317 }
318 }
319 }
320
321 /// 对 regular(解析部分)做 Taylor 展开
322 int taylor_order = order_pos + pole_order + 1;
323 if (taylor_order < 1) taylor_order = 1;
324 auto taylor = regular->series(var, center, taylor_order);
325 if (!taylor) return res;
326
327 /// Laurent = taylor / (x-c)^pole_order
328 std::shared_ptr<SymbolicExpr> laurent = taylor;
329 if (pole_order > 0) {
330 auto powm = SymbolicExpr::power(shift, SymbolicExpr::number(pole_order));
331 laurent = SymbolicExpr::divide(taylor, powm)->simplify();
332 }
333
334 res.series = laurent->simplify();
335 res.pole_order = pole_order;
336 if (pole_order == 0) res.singularity = SingularityType::Removable;
337 else res.singularity = SingularityType::Pole;
338
339 /// 留数 = Taylor 展开中 (x-c)^(pole_order-1) 的系数
340 if (pole_order >= 1) {
341 /// a_{m-1} = (1/(m-1)!) lim_{x->c} d^{m-1}/dx^{m-1} [regular]
342 auto deriv = regular;
343 for (int i = 0; i < pole_order - 1; ++i) deriv = deriv->differentiate(var);
344 /// 用极限求值,稳健处理 0/0 形式(如 z/(z(z+1)) 在 z=0).
345 auto limited = limit_expression_checked(
346 deriv, var, center, LimitDirection::Both, context);
347 if (!limited) {
348 return LaurentSeriesResult::failure(limited.error());
349 }
350 auto val = std::move(limited.value());
351 if (!val) val = deriv->substitute(var, center);
352 long long fact = 1;
353 for (int i = 2; i <= pole_order - 1; ++i) fact *= i;
354 res.residue = SymbolicExpr::divide(val, SymbolicExpr::number((int)fact))->simplify();
355 }
356
357 return res;
358}

360std::shared_ptr<SymbolicExpr> asymptotic_expand(
361 const std::shared_ptr<SymbolicExpr>& f, const std::string& var, int order) {
362 if (!f || order < 0) return nullptr;
363 /// 通过 x = 1/t 替换,在 t=0 处做 Taylor 展开,再回代 t = 1/x,
364 /// 得到按 x 递减幂次的渐近展开.
365 auto x = SymbolicExpr::variable(var);
366 std::string tname = var + "__asym_t";
367 auto t = SymbolicExpr::variable(tname);
368 auto inv_t = SymbolicExpr::divide(SymbolicExpr::number(1), t);
369 auto g = f->substitute(var, inv_t);
370 if (!g) return nullptr;
371 g = g->simplify();
372 auto taylor_t = g->series(tname, SymbolicExpr::number(0), order + 1);
373 if (!taylor_t) return nullptr;
374 /// 回代 t = 1/x
375 auto inv_x = SymbolicExpr::divide(SymbolicExpr::number(1), x);
376 auto back = taylor_t->substitute(tname, inv_x);
377 if (!back) return nullptr;
378 return back->simplify();
379}
