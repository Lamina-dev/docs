134ExprMatchResult expr_match(const ExprPtr& pattern,
135 const ExprPtr& target,
136 const std::vector<std::string>& wildcards,
137 ComputationContext& context) {
138 auto step = context.consume_steps(1, kExprMatchOperation);
139 if (!step) return ExprMatchResult::failure(step.error());
140 if (!pattern || !LMCAS::detail::node(pattern)) {
141 return expr_match_failure(CasErrc::InvalidArgument,
142 "match pattern cannot be null",
143 kExprMatchOperation);
144 }
145 if (!target || !LMCAS::detail::node(target)) {
146 return expr_match_failure(CasErrc::InvalidArgument,
147 "match target cannot be null",
148 kExprMatchOperation);
149 }
150
151 std::unordered_set<std::string> wildcard_set;
152 wildcard_set.reserve(wildcards.size());
153 for (const auto& wildcard : wildcards) {
154 if (wildcard.empty()) {
155 return expr_match_failure(CasErrc::InvalidArgument,
156 "wildcard names cannot be empty",
157 kExprMatchOperation);
158 }
159 if (!wildcard_set.insert(wildcard).second) {
160 return expr_match_failure(CasErrc::InvalidArgument,
161 "wildcard names must be unique",
162 kExprMatchOperation);
163 }
164 }
165
166 try {
167 MatchMap raw_bindings;
168 const bool matched = Matcher::match(
169 *pattern, *target, wildcard_set, raw_bindings);
170 if (!matched) {
171 return ExprMatchResult::success(ExprMatch{false, {}});
172 }
173
174 std::vector<std::string> names;
175 names.reserve(raw_bindings.size());
176 for (const auto& binding : raw_bindings) {
177 names.push_back(binding.first);
178 }
179 std::sort(names.begin(), names.end());
180
181 ExprMatch result;
182 result.matched = true;
183 result.bindings.reserve(names.size());
184 for (const auto& name : names) {
185 const auto& value = raw_bindings.at(name);
186 auto node = LMCAS::detail::node(value);
187 if (!node) {
188 return expr_match_failure(CasErrc::InternalInvariant,
189 "matcher produced a null binding",
190 kExprMatchOperation);
191 }
192 result.bindings.push_back(
193 ExprMatchBinding{name,
194 LMCAS::detail::make_expression_ptr(node)});
195 }
196 return ExprMatchResult::success(std::move(result));
197 } catch (const std::bad_alloc&) {
198 return expr_match_failure(CasErrc::ResourceLimit,
199 "expression matching allocation failed",
200 kExprMatchOperation);
201 } catch (const std::exception& error) {
202 return expr_match_failure(CasErrc::InternalInvariant, error.what(),
203 kExprMatchOperation);
204 }
205}
