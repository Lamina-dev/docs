362bool vector_calculus_expr_zero_after_substitution(
363 const std::shared_ptr<SymbolicExpr>& expr,
364 const std::map<std::string, std::shared_ptr<SymbolicExpr>>& point)
365{
366 if (!expr || !LMCAS::detail::node(expr)) return false;
367 auto substituted = expr;
368 for (const auto& [var, value] : point) {
369 if (!value || !LMCAS::detail::node(value)) return false;
370 substituted = substituted->substitute(var, value);
371 if (!substituted || !LMCAS::detail::node(substituted)) return false;
372 }
373 substituted = substituted->simplify();
374 if (!substituted || !LMCAS::detail::node(substituted)) return false;
375 if (substituted->is_zero()) return true;
376 double value = 0.0;
377 return vector_calculus_checked_finite_numeric(substituted, value) &&
378 std::abs(value) < 1e-8;
379}
