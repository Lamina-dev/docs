29template <typename Transform>
30ExpressionResult checked_transform_expr(const ExprPtr& expression,
31 ComputationContext& context,
32 const char* operation,
33 const char* transform_name,
34 Transform&& transform) {
35 auto step = context.consume_steps(1, operation);
36 if (!step) return ExpressionResult::failure(step.error());
37 if (!expression || !LMCAS::detail::node(expression)) {
38 return ExpressionResult::failure(CasErrc::InvalidArgument,
39 std::string(transform_name) +
40 " argument cannot be null",
41 operation);
42 }
43 try {
44 auto result = std::forward<Transform>(transform)(*expression);
45 if (!result || !LMCAS::detail::node(result)) {
46 return ExpressionResult::failure(CasErrc::InternalInvariant,
47 std::string(transform_name) +
48 " returned null",
49 operation);
50 }
51 return ExpressionResult::success(std::move(result));
52 } catch (const CasError& error) {
53 return ExpressionResult::failure(error);
54 } catch (const std::bad_alloc&) {
55 return ExpressionResult::failure(CasErrc::ResourceLimit,
56 std::string(transform_name) +
57 " allocation failed",
58 operation);
59 } catch (const std::length_error& error) {
60 return ExpressionResult::failure(CasErrc::ResourceLimit, error.what(),
61 operation);
62 } catch (const std::exception& error) {
63 return ExpressionResult::failure(CasErrc::InternalInvariant, error.what(),
64 operation);
65 }
