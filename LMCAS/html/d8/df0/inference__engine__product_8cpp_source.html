13std::uint8_t sign_set(Sign target) {
14 switch (target) {
15 case Sign::Positive: return positive_sign;
16 case Sign::Negative: return negative_sign;
17 case Sign::Zero: return zero_sign;
18 case Sign::NonNegative: return zero_sign | positive_sign;
19 case Sign::NonPositive: return negative_sign | zero_sign;
20 case Sign::NonZero: return negative_sign | positive_sign;
21 default: return all_signs;
22 }
23}

25std::uint8_t multiply_sign_sets(std::uint8_t left, std::uint8_t right) {
26 std::uint8_t result = 0;
27 for (std::uint8_t a : {negative_sign, zero_sign, positive_sign}) {
28 for (std::uint8_t b : {negative_sign, zero_sign, positive_sign}) {
29 if (!(left & a) || !(right & b)) continue;
30 result |= a == zero_sign || b == zero_sign ? zero_sign :
31 a == b ? positive_sign : negative_sign;
32 }
33 }
34 return result;
35}

54InferenceTriboolResult InferenceEngine::Impl::product_parity_sign(
55 std::uint8_t possible, Sign target) {
56 if (!possible) return Tribool::Unknown;
57 const auto wanted = sign_set(target);
58 if ((possible & wanted) == possible) return Tribool::True;
59 if (!(possible & wanted)) return Tribool::False;
60 return Tribool::Unknown;
61}

64 const InferenceEngine& engine, const MultiplyNode& node, Sign target, ComputationContext& context) const {
65 auto defined = real_definedness(engine, node, context);
66 if (!defined || defined.value() != Tribool::True) { return defined; }
67 const detail::AssumptionFacts facts(engine);
68 bool all_real = true;
69 for (const auto& operand : node.operands()) {
70 auto real = detail::query_real_value(operand, facts, context);
71 if (!real) { return real; }
72 all_real = all_real && real.value() == Tribool::True;
73 }
74 if (!all_real) { return Tribool::Unknown; }
75 std::uint8_t possible = positive_sign;
76 for (const auto& operand : node.operands()) {
77 auto result = accumulate_product_sign(engine, operand, possible, context);
78 if (!result) { return result; }

83InferenceTriboolResult InferenceEngine::Impl::quotient_sign(
84 const InferenceEngine& engine, const SymbolicExpr& numerator,
85 const SymbolicExpr& denominator, Sign target, ComputationContext& context) const {
86 const detail::AssumptionFacts facts(engine);
87 bool all_defined = true;
88 bool all_real = true;
89 for (const auto* expression : {&numerator, &denominator}) {
90 auto defined = real_definedness(engine, *detail::node(*expression), context);
91 if (!defined) { return defined; }
92 all_defined = all_defined && defined.value() == Tribool::True;
93 auto real = detail::query_real_value(detail::node(*expression), facts, context);
94 if (!real) { return real; }
95 all_real = all_real && real.value() == Tribool::True;
96 }
97 auto nonzero = detail::query_nonzero_value(detail::node(denominator), facts, Domain::Real, context);
98 if (!nonzero) { return nonzero; }
99 if (nonzero.value() == Tribool::False) {
100 return InferenceTriboolResult::failure(
101 CasErrc::DomainError, "division denominator is zero", "inference.division");
102 }
103 if (!all_defined || !all_real || nonzero.value() != Tribool::True)
104 { return Tribool::Unknown; }
105 std::uint8_t possible = positive_sign;
106 auto first = accumulate_product_sign(engine, detail::node(numerator), possible, context);
107 if (!first) { return first; }
108 auto second = accumulate_product_sign(engine, detail::node(denominator), possible, context);
109 if (!second) { return second; }
110 return product_parity_sign(possible, target);
111}

115 try {
116 if (node.operands().size() != 2) {

118 }
119 std::shared_ptr<const SymbolicNode> numerator;
120 std::shared_ptr<const SymbolicNode> denominator;
121 for (const auto& operand : node.operands()) {
122 auto step = context.consume_steps(1, "inference.product");
123 if (!step) {
124 throw step.error();
125 }
126 const auto power = std::dynamic_pointer_cast<const PowerNode>(operand);
127 const auto exponent = power
128 ? std::dynamic_pointer_cast<const NumberNode>(
129 power->exponent())
130 : nullptr;

132 denominator = power->base();
