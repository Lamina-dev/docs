248static ExpressionResult calculate_residue_impl(
249 const std::shared_ptr<SymbolicExpr>& f,
250 const std::string& z,
251 const std::shared_ptr<SymbolicExpr>& z0,
252 int order,
253 ComputationContext& context) {
254 if (order < 1) return SymbolicExpr::number(0);
255
256 auto var_z = SymbolicExpr::variable(z);
257 auto z_minus_z0 = SymbolicExpr::add(var_z, SymbolicExpr::multiply(z0, SymbolicExpr::number(-1)));
258 auto pow_term = SymbolicExpr::power(z_minus_z0, SymbolicExpr::number(order));
259
260 auto F = SymbolicExpr::multiply(pow_term, f);
261
262 for (int i = 0; i < order - 1; i++) {
263 F = F->differentiate(z);
264 }
265
266 auto fact = BigInt::factorial_checked(BigInt(order - 1), context);
267 if (!fact) return ExpressionResult::failure(fact.error());
268
269 F = SymbolicExpr::divide(F, SymbolicExpr::number(fact.value()));
270
271 return limit_expression_checked(
272 F, z, z0, LimitDirection::Both, context, Domain::Complex);
273}

321static ExpressionResult cauchy_integral_impl(
322 const std::shared_ptr<SymbolicExpr>& f,
323 const std::string& z,
324 const std::shared_ptr<SymbolicExpr>& z0,
325 int n, ComputationContext& context) {
326
327 /// n is the power in the denominator: \oint f(z)/(z-z0)^n dz
328 if (n < 1) return SymbolicExpr::number(0);
329
330 auto deriv = f;
331 for (int i = 0; i < n - 1; i++) {
332 deriv = deriv->differentiate(z);
333 }
334
335 auto f_n_minus_1_z0 = deriv->substitute(z, z0);
336
337 auto fact = BigInt::factorial_checked(BigInt(n - 1), context);
338 if (!fact) return ExpressionResult::failure(fact.error());
339
340 auto term = SymbolicExpr::divide(f_n_minus_1_z0, SymbolicExpr::number(fact.value()));
341
342 auto pi_node = LMCAS::detail::make_node<VariableNode>("pi", true);
343 auto i_node = SymbolicFactory::create_complex(
344 LMCAS::detail::node(SymbolicExpr::number(0)),
345 LMCAS::detail::node(SymbolicExpr::number(1)));
346
347 auto pi_expr = LMCAS::detail::make_expression_ptr(pi_node);
348 auto i_expr = LMCAS::detail::make_expression_ptr(i_node);
349
350 auto two_pi_i = SymbolicExpr::multiply(SymbolicExpr::number(2), SymbolicExpr::multiply(pi_expr, i_expr));
351
352 return SymbolicExpr::multiply(two_pi_i, term)->simplify();
353}

367std::optional<int> bounded_complex_power(
368 const std::shared_ptr<const SymbolicNode>& node) {
369 auto number = std::dynamic_pointer_cast<const NumberNode>(node);
370 if (!number) return std::nullopt;
371 BigInt exponent;
372 if (const auto* integer = std::get_if<BigInt>(&number->value())) {
373 exponent = *integer;
374 } else if (const auto* rational = std::get_if<Rational>(&number->value())) {
375 if (!rational->is_integer()) return std::nullopt;
376 exponent = rational->get_numerator();
377 } else {
378 const double value = std::get<lmmc_real_t>(number->value());
379 if (!std::isfinite(value) || value != std::floor(value) ||
380 value < 0 || value > 16) return std::nullopt;
381 return static_cast<int>(value);
382 }
383 if (exponent < BigInt(0) || exponent > BigInt(16)) return std::nullopt;
384 return static_cast<int>(*exponent.try_to_int64());
385}

387/// 递归地将表达式分解为 (实部, 虚部)，把 ComplexNode 视为 a+bi。
388/// 仅处理加法、乘法、数值与 ComplexNode 组合；其余子表达式视为实值。
389void split_real_imag(const std::shared_ptr<const SymbolicNode>& node,
390 std::shared_ptr<SymbolicExpr>& re,
391 std::shared_ptr<SymbolicExpr>& im) {
392 re = SymbolicExpr::number(0);
393 im = SymbolicExpr::number(0);
394 if (!node) return;
395
396
397 if (auto cn = std::dynamic_pointer_cast<const ComplexNode>(node)) {
398 re = LMCAS::detail::make_expression_ptr(cn->real());
399 im = LMCAS::detail::make_expression_ptr(cn->imag());
400 return;
401 }
402 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
403 for (const auto& op : add->operands()) {
404 std::shared_ptr<SymbolicExpr> r2, i2;
405 split_real_imag(op, r2, i2);
406 re = SymbolicExpr::add(re, r2);
407 im = SymbolicExpr::add(im, i2);
408 }
409 re = re->simplify();
410 im = im->simplify();
411 return;
412 }
413 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
414 /// 累乘：(a+bi)(c+di) = (ac-bd) + (ad+bc)i
415 std::shared_ptr<SymbolicExpr> accR = SymbolicExpr::number(1);
416 std::shared_ptr<SymbolicExpr> accI = SymbolicExpr::number(0);
417 for (const auto& op : mul->operands()) {
418 std::shared_ptr<SymbolicExpr> r2, i2;
419 split_real_imag(op, r2, i2);
420 auto ac = SymbolicExpr::multiply(accR, r2);
421 auto bd = SymbolicExpr::multiply(accI, i2);
422 auto ad = SymbolicExpr::multiply(accR, i2);
423 auto bc = SymbolicExpr::multiply(accI, r2);
424 accR = SymbolicExpr::add(ac, SymbolicExpr::multiply(SymbolicExpr::number(-1), bd))->simplify();
425 accI = SymbolicExpr::add(ad, bc)->simplify();
426 }
427 re = accR;
428 im = accI;
429 return;
430 }
431 if (auto pw = std::dynamic_pointer_cast<const PowerNode>(node)) {
432 /// 对整数次幂，展开为重复乘法以分离实/虚部。
433 auto exponent = bounded_complex_power(pw->exponent());
434 std::shared_ptr<SymbolicExpr> baseR, baseI;
435 split_real_imag(pw->base(), baseR, baseI);
436 bool base_real = LMCAS::detail::node(baseI) && LMCAS::detail::node(baseI)->is_zero();
437 if (exponent && !base_real) {
438 /// (a+bi)^e via repeated complex multiplication
439 std::shared_ptr<SymbolicExpr> accR = SymbolicExpr::number(1);
440 std::shared_ptr<SymbolicExpr> accI = SymbolicExpr::number(0);
441 for (int k = 0; k < *exponent; ++k) {
442 auto ac = SymbolicExpr::multiply(accR, baseR);
443 auto bd = SymbolicExpr::multiply(accI, baseI);
444 auto ad = SymbolicExpr::multiply(accR, baseI);
445 auto bc = SymbolicExpr::multiply(accI, baseR);
446 accR = SymbolicExpr::add(ac, SymbolicExpr::multiply(SymbolicExpr::number(-1), bd))->simplify();
447 accI = SymbolicExpr::add(ad, bc)->simplify();
448 }
449 re = accR; im = accI;
450 return;
451 }
452 /// 实底数或非整数指数：视为实值
453 if (base_real) {
454 re = LMCAS::detail::make_expression_ptr(node);
455 im = SymbolicExpr::number(0);
456 return;
457 }
458 /// 退化情形：原样返回为实部
459 re = LMCAS::detail::make_expression_ptr(node);
460 im = SymbolicExpr::number(0);
461 return;
462 }
463 /// 默认：视为实值表达式
464 re = LMCAS::detail::make_expression_ptr(node);
465 im = SymbolicExpr::number(0);
466}

529 ComputationContext& context) {
530 const std::string operation = "conjugate";
531 auto input = validate_complex_expr_input(expr, context, operation);
532 if (!input) return ExpressionResult::failure(input.error());
533 if (has_function_of_explicit_complex(LMCAS::detail::node(expr))) {

540 std::shared_ptr<SymbolicExpr> re, im;
541 split_real_imag(LMCAS::detail::node(expr), re, im);
542 /// conj(a+bi) = a - bi
543 auto neg_im = SymbolicExpr::multiply(SymbolicExpr::number(-1), im)->simplify();
544 if (!re || !LMCAS::detail::node(re) || !neg_im || !LMCAS::detail::node(neg_im)) {
545 return ExpressionResult::failure(CasErrc::InternalInvariant,
546 "conjugate construction failed",
547 operation);
548 }
549 if (LMCAS::detail::node(neg_im) && LMCAS::detail::node(neg_im)->is_zero()) {
550 auto simplified_re = re->simplify();
551 if (!simplified_re || !LMCAS::detail::node(simplified_re)) {
552 return ExpressionResult::failure(CasErrc::InternalInvariant,
553 "conjugate construction failed",
554 operation);
555 }
556 return ExpressionResult::success(simplified_re);
557 }
558 auto cn = SymbolicFactory::create_complex(
559 LMCAS::detail::node(re->simplify()), LMCAS::detail::node(neg_im));
560 if (!cn) {
561 return ExpressionResult::failure(CasErrc::InternalInvariant,
562 "conjugate construction failed",
563 operation);
564 }
565 return ExpressionResult::success(LMCAS::detail::make_expression_ptr(cn));

625static bool is_analytic_impl(const std::shared_ptr<SymbolicExpr>& f, const std::string& z) {
626 if (!f) return false;
627 /// 将 z 替换为 (z_re + i·z_im)，分离 u、v，检验 Cauchy-Riemann。
628 std::string xr = z + "_re";
629 std::string xi = z + "_im";
630 auto zr = SymbolicExpr::variable(xr);
631 auto zi = SymbolicExpr::variable(xi);
632 auto i_unit = LMCAS::detail::make_expression_ptr(
633 SymbolicFactory::create_complex(
634 LMCAS::detail::node(SymbolicExpr::number(0)),
635 LMCAS::detail::node(SymbolicExpr::number(1))));
636 auto z_sub = SymbolicExpr::add(zr, SymbolicExpr::multiply(i_unit, zi));
637
638 auto fz = f->substitute(z, z_sub);
639 if (!fz) return false;
640 fz = fz->simplify();
641
642 std::shared_ptr<SymbolicExpr> u, v;
643 split_real_imag(LMCAS::detail::node(fz), u, v);
644
645 auto ux = differentiate_if_dependent(u, xr);
646 auto uy = differentiate_if_dependent(u, xi);
647 auto vx = differentiate_if_dependent(v, xr);
648 auto vy = differentiate_if_dependent(v, xi);
649
