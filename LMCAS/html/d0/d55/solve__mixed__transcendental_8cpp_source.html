23std::optional<CasError> validate_mixed_input(
24 const std::shared_ptr<SymbolicExpr>& expr,
25 const std::string& var, const SolveOptions& opts) {
26 if (!expr || !detail::node(expr)) {
27 return CasError{
28 CasErrc::InvalidArgument, "待求解表达式不能为空", mixed_operation};
29 }
30 if (var.empty()) {
31 return CasError{
32 CasErrc::InvalidArgument, "求解变量不能为空", mixed_operation};
33 }
34 if (!std::isfinite(opts.tolerance) || opts.tolerance <= 0.0 ||
35 opts.max_newton_iterations <= 0 || opts.max_roots == 0 || opts.max_roots < -1) {
36 return CasError{
37 CasErrc::InvalidArgument, "求解选项包含无效界限", mixed_operation};
38 }
39 if (opts.has_search_interval &&
40 (!std::isfinite(opts.search_lo) || !std::isfinite(opts.search_hi) ||
41 opts.search_lo >= opts.search_hi)) {
42 return CasError{
43 CasErrc::InvalidArgument, "搜索区间必须为有限递增区间", mixed_operation};
44 }
45 return std::nullopt;
46}

48MixedTranscendentalResult classify_constant_equation(
49 const std::shared_ptr<SymbolicExpr>& expr,
50 bool unresolved_parameters, const SolveOptions& opts, ComputationContext& context) {
51 using Roots = MathResult<std::vector<std::shared_ptr<SymbolicExpr>>>;
52 if (unresolved_parameters) {
53 return MixedTranscendentalResult::success(Roots{
54 {}, Completeness::Inconclusive,
55 "equation depends on unresolved parameters but not the solve variable"});
56 }
57 auto evaluated = evaluate_numeric(*expr, {}, context);
58 if (!evaluated &&
59 (evaluated.error().code == CasErrc::ResourceLimit ||
60 evaluated.error().code == CasErrc::Cancelled ||
61 evaluated.error().code == CasErrc::InternalInvariant)) {
62 return MixedTranscendentalResult::failure(evaluated.error());
63 }
64 if (!evaluated || !std::isfinite(evaluated.value().value)) {
65 return MixedTranscendentalResult::success(Roots{
66 {}, Completeness::Inconclusive,
67 "constant equation could not be evaluated in the real domain"});
68 }
69 if (std::abs(evaluated.value().value) <= opts.tolerance) {
70 return MixedTranscendentalResult::success(Roots{
71 {}, Completeness::Inconclusive,
72 "equation is identically zero; every real value satisfies it"});
73 }
74 return MixedTranscendentalResult::success(Roots{{}, Completeness::Complete, {}});
75}

90 std::optional<CasError> solve(ComputationContext& context) {
91 auto factored = factor_transcendental(expr_, var_, context);
92 if (!factored) return factored.error();
93 auto factors = std::move(factored.value());
94 if (factors.empty()) {
95 factors.push_back(expr_);
96 complete_ = false;
97 }
98 for (const auto& factor : factors) {
99 auto step = context.consume_steps(1, mixed_operation);
100 if (!step) {
101 return step.error();
102 }
103 auto failure = solve_factor(factor, context);
104 if (failure) {
105 return failure;
106 }

112 MixedTranscendentalResult finish() {
113 struct RepresentedRoot {
114 lmmc_real_t value;
115 std::shared_ptr<SymbolicExpr> expression;
116 bool symbolic;
117 };
118 auto numeric_values = deduplicate_roots(roots_, opts_.tolerance, -1);
119 std::vector<RepresentedRoot> values;
120 values.reserve(symbolic_roots_.size() + numeric_values.size());
121 for (auto& root : symbolic_roots_) {
122 values.push_back({root.value, std::move(root.expression), true});
123 }
124 for (const auto value : numeric_values) {
125 values.push_back(
126 {value, SymbolicExpr::number(static_cast<double>(value)), false});
127 }
128 std::sort(values.begin(), values.end(),
129 [](const RepresentedRoot& left, const RepresentedRoot& right) {
130 if (left.value != right.value) { return left.value < right.value; }
131 return left.symbolic && !right.symbolic;
132 });
133 const auto threshold = 10.0 * opts_.tolerance;
134 std::vector<RepresentedRoot> unique;
135 unique.reserve(values.size());
136 for (auto& root : values) {
137 if (!unique.empty() &&
138 std::fabs(root.value - unique.back().value) < threshold) {
139 if (root.symbolic && !unique.back().symbolic) {
140 unique.back() = std::move(root);
141 }
142 continue;
143 }
144 unique.push_back(std::move(root));
145 }
146 if (opts_.max_roots > 0 &&
147 unique.size() > static_cast<std::size_t>(opts_.max_roots)) {
148 unique.resize(static_cast<std::size_t>(opts_.max_roots));
149 complete_ = false;
150 }
151 std::vector<std::shared_ptr<SymbolicExpr>> results;
152 results.reserve(unique.size());
153 for (auto& root : unique) {
154 results.push_back(std::move(root.expression));
155 }
156 return MixedTranscendentalResult::success(
157 MathResult<std::vector<std::shared_ptr<SymbolicExpr>>>{
158 std::move(results),
159 complete_ ? Completeness::Complete : Completeness::Inconclusive,
160 complete_ ? std::string{} : "有界数值隔离未能证明排除全部未检测根"});
161 }

164 void polish_value(lmmc_real_t& value, lmmc_real_t& residual, int& iterations) {
165 auto polish = opts_;
166 polish.has_initial_guess = true;
167 polish.initial_guess = value;
168 auto refined = solve_numeric_checked(expr_, var_, context_, polish);
169 if (!refined) {
170 if (is_mixed_terminal_error(refined.error().code)) {
171 failure_ = refined.error();
172 }
173 return;
174 }
175 for (const auto& candidate : refined.value()) {
176 if (candidate.value < interval_.lo || candidate.value > interval_.hi) { continue; }
177 const auto candidate_residual = std::fabs(
178 detail::mixed_evaluate_at(expr_, var_, candidate.value));
179 if (std::isfinite(candidate_residual) && candidate_residual < residual) {
180 value = candidate.value;
181 residual = candidate_residual;
182 iterations += candidate.iterations;
183 }
184 }
185 }

187 bool accept_value(lmmc_real_t value, int iterations) {
188 if (!std::isfinite(value)) { complete_ = false; return false; }
189 if (value < interval_.lo || value > interval_.hi) return true;
190 auto residual = std::fabs(detail::mixed_evaluate_at(expr_, var_, value));
191 if (std::isfinite(residual) && residual > opts_.tolerance) {
192 polish_value(value, residual, iterations);
193 }
194 if (!std::isfinite(residual) || residual > opts_.tolerance) {
195 complete_ = false;
196 return false;
197 }
198 roots_.push_back({value, residual, iterations});
199 return true;

213 std::optional<bool> accept_proved_symbolic(
214 std::shared_ptr<SymbolicExpr> simplified,
215 lmmc_real_t value) {
216 auto bound = detail::substitute_raw(expr_, var_, simplified);
217 std::optional<detail::AssumptionFacts> assumptions;
218 if (context_.assumptions()) {
219 assumptions.emplace(*context_.assumptions());
220 }
221 const FactsQuery& facts = assumptions
222 ? static_cast<const FactsQuery&>(*assumptions)
223 : detail::no_facts();
224 auto defined = detail::query_definedness(
225 detail::node(bound), facts, Domain::Real, context_);
226 if (!defined) {
227 if (is_mixed_terminal_error(defined.error().code)) {
228 failure_ = defined.error();
229 return false;
230 }
231 return std::nullopt;
232 }
233 if (defined.value() == Tribool::False) {
234 return true;
235 }
236 if (defined.value() != Tribool::True) {
237 return std::nullopt;
238 }
239
240 auto residual = check_zero_residual(bound, context_);
241 if (!residual) {
242 if (is_mixed_terminal_error(residual.error().code)) {
243 failure_ = residual.error();
244 return false;
245 }
246 return std::nullopt;
247 }
248 if (!std::holds_alternative<ProvedZeroResidual>(
249 residual.value())) {
250 return std::nullopt;
251 }
252 symbolic_roots_.push_back(
253 SymbolicRoot{value, std::move(simplified)});
254 return true;
255 }

257 bool accept_symbolic(const std::shared_ptr<SymbolicExpr>& root) {
258 if (!root) {
259 complete_ = false;
260 return false;
261 }
262 auto simplified = root->simplify();
263 if (!simplified || !detail::node(simplified)) {
264 complete_ = false;
265 return false;
266 }
267 auto evaluated = evaluate_numeric(*simplified, {}, context_);
268 if (!evaluated) {
269 return handle_symbolic_evaluation_error(evaluated.error());
270 }
271 const auto value = evaluated.value().value;
272 if (!std::isfinite(value)) {
273 complete_ = false;
274 return false;
275 }
276 if (value < interval_.lo || value > interval_.hi) {
277 return true;
278 }
279 auto accepted =
280 accept_proved_symbolic(std::move(simplified), value);
281 if (accepted) {
282 return *accepted;
283 }
284 return accept_value(value, 0);
285 }

361 std::optional<CasError> solve_factor(
362 const std::shared_ptr<SymbolicExpr>& factor,
363 ComputationContext& context) {
364 if (!factor || !detail::node(factor)) {
365 complete_ = false;
366 return std::nullopt;
367 }
368 if (!expression_depends_on_variable(detail::node(factor), var_)) {
369 return check_constant_factor(factor);
370 }
371 auto polynomial = solve_polynomial_factor(factor, context);
372 if (!polynomial) {
373 return polynomial.error();
374 }
375 if (polynomial.value()) {
376 return std::nullopt;
377 }
378 auto symbolic = solve_transcendental_factor(factor, context);
379 if (!symbolic) {
380 return symbolic.error();
381 }
382 if (symbolic.value()) {
383 return std::nullopt;
384 }
385
386 auto numeric = detail::mixed_numerical_path(
387 factor, var_, interval_, opts_, context, complete_);
388 if (!numeric) {
389 return std::move(numeric.error());
390 }
391 for (const auto& root : numeric.value()) {
392 accept_value(root.value, root.iterations);
393 }
394 return std::nullopt;
395 }

413MixedTranscendentalResult solve_mixed_equation(
414 const std::shared_ptr<SymbolicExpr>& expr, const std::string& var,
415 const SolveOptions& opts, ComputationContext& context) {
416 auto step = context.consume_steps(1, mixed_operation);
417 if (!step) {
418 return MixedTranscendentalResult::failure(step.error());
419 }
420 const auto variables = free_variables(detail::node(expr));
421 if (variables.find(var) == variables.end()) {
422 return classify_constant_equation(expr, !variables.empty(), opts, context);
423 }
424 auto interval = determine_search_interval(expr, var, opts);
425 if (!interval) {
426 return MixedTranscendentalResult::failure(
427 CasErrc::InvalidArgument, "无法确定有效搜索区间", mixed_operation);
428 }
429 if (!std::isfinite(interval->lo) || !std::isfinite(interval->hi) ||
430 interval->lo >= interval->hi) {
431 return MixedTranscendentalResult::failure(
432 CasErrc::InvalidArgument, "搜索区间必须为有限递增区间", mixed_operation);
433 }
434 MixedRootCollection collection(expr, var, *interval, opts, context);
435 auto failure = collection.solve(context);
436 if (failure) {
437 return MixedTranscendentalResult::failure(std::move(*failure));
438 }
439 return collection.finish();
440}
