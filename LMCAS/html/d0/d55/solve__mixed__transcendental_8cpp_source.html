149/**
150 * @internal
151 * @brief 计算节点关于指定变量的多项式次数。
152 *
153 * 对不依赖 var 的子表达式返回 0；对无法确定次数的情形（非多项式结构）返回 -1。
154 *
155 * @param[in] node 符号节点
156 * @param[in] var 变量名
157 * @return 关于 var 的次数；-1 表示非多项式结构
158 */
159static int degree_in_var(const std::shared_ptr<const SymbolicNode>& node, const std::string& var) {
160 if (!node) return 0;
161
162 if (!expression_depends_on_variable(node, var)) return 0;
163
164 if (auto v = std::dynamic_pointer_cast<const VariableNode>(node)) {
165 return (v->name() == var) ? 1 : 0;
166 }
167
168 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
169 int max_deg = 0;
170 for (const auto& op : add->operands()) {
171 int d = degree_in_var(op, var);
172 if (d < 0) return -1;
173 max_deg = std::max(max_deg, d);
174 }
175 return max_deg;
176 }
177
178 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
179 int total_deg = 0;
180 for (const auto& op : mul->operands()) {
181 int d = degree_in_var(op, var);
182 if (d < 0) return -1;
183 total_deg += d;
184 }
185 return total_deg;
186 }
187
188 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
189 auto exp_num = std::dynamic_pointer_cast<const NumberNode>(pow->exponent());
190 if (!exp_num) return -1;
191
192 int e_val = -1;
193 if (std::holds_alternative<BigInt>(exp_num->value())) {
194 const auto& bi = std::get<BigInt>(exp_num->value());
195 if (!bi.IsNegative()) e_val = bi.to_int();
196 } else if (std::holds_alternative<Rational>(exp_num->value())) {
197 const auto& r = std::get<Rational>(exp_num->value());
198 if (r.is_integer()) {
199 BigInt bi = r.to_BigInt();
200 if (!bi.IsNegative()) e_val = bi.to_int();
201 }
202 } else if (std::holds_alternative<lmmc_real_t>(exp_num->value())) {
203 lmmc_real_t d = std::get<lmmc_real_t>(exp_num->value());
204 if (std::isfinite(d) && d >= 0 && d == std::floor(d) && d < 1000.0) {
205 e_val = static_cast<int>(d);
206 }
207 }
208
209 if (e_val < 0) return -1;
210
211 int base_deg = degree_in_var(pow->base(), var);
212 if (base_deg < 0) return -1;
213 return base_deg * e_val;
214 }
215
216 /// FunctionNode 依赖 var 意味着非多项式结构
217 if (std::dynamic_pointer_cast<const FunctionNode>(node)) {
218 return -1;
219 }
220
221 return 0;
222}

224bool is_polynomial_after_substitution(
225 const TransSubstitutionResult& sub_result)
226{
227 if (sub_result.mappings.empty()) return false;
228 if (!sub_result.poly_expr || !lamina::detail::node(sub_result.poly_expr)) return false;
229
230 const auto& root = lamina::detail::node(sub_result.poly_expr);
231
232 for (const auto& m : sub_result.mappings) {
233 if (expression_depends_on_variable(root, m.indeterminate)) {

235 if (deg < 0) return false;
236 }
237 }
238
239 return true;

266/**
267 * @internal
268 * @brief 尝试从表达式中提取关于变量的线性系数 k（即 k*x + c 形式中的 k）。
269 *
270 * 支持的模式：
271 * - 变量本身 x → k = 1
272 * - 乘法 k*x（数值 * 变量）→ 提取数值
273 * - 加法 k*x + c → 在加法项中查找含变量的项并提取系数
274 * - 非线性参数（x^2, sin(x) 等）→ 返回 NaN
275 *
276 * @param[in] node 参数表达式节点
277 * @param[in] var 变量名
278 * @return 线性系数 k；非线性时返回 NaN
279 */
280static lmmc_real_t extract_linear_coefficient(
281 const std::shared_ptr<const SymbolicNode>& node,
282 const std::string& var)
283{
284 if (!node || !expression_depends_on_variable(node, var)) return 0.0;
285
286 /// 模式 1: 变量本身 → k = 1
287 if (auto v = std::dynamic_pointer_cast<const VariableNode>(node)) {
288 if (v->name() == var) return 1.0;
289 return std::numeric_limits<lmmc_real_t>::quiet_NaN();
290 }
291
292 /// 模式 2: 乘法节点 k*x
293 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
294 /// 检查是否恰好是 数值 * 变量 的形式
295 if (mul->operands().size() == 2) {
296 std::shared_ptr<const SymbolicNode> num_part = nullptr;
297 std::shared_ptr<const SymbolicNode> var_part = nullptr;
298
299 for (const auto& op : mul->operands()) {
300 if (op->is_number()) {
301 num_part = op;
302 } else if (auto vn = std::dynamic_pointer_cast<const VariableNode>(op)) {
303 if (vn->name() == var) var_part = op;
304 }
305 }
306
307 if (num_part && var_part) {
308 return extract_real_value(num_part);
309 }
310 }
311 /// 非简单 k*x 形式 → 非线性
312 return std::numeric_limits<lmmc_real_t>::quiet_NaN();
313 }
314
315 /// 模式 3: 加法节点 k*x + c
316 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
317 lmmc_real_t coeff = std::numeric_limits<lmmc_real_t>::quiet_NaN();
318 bool found_var_term = false;
319
320 for (const auto& op : add->operands()) {
321 if (!expression_depends_on_variable(op, var)) {
322 /// 常数项，跳过
323 continue;
324 }
325 /// 含变量的项 — 只允许一个
326 if (found_var_term) {
327 /// 多个含变量的项 → 非线性
328 return std::numeric_limits<lmmc_real_t>::quiet_NaN();
329 }
330 found_var_term = true;
331 coeff = extract_linear_coefficient(op, var);
332 }
333
334 return found_var_term ? coeff : std::numeric_limits<lmmc_real_t>::quiet_NaN();
335 }
336
337 /// 其他节点类型（PowerNode, FunctionNode 等）→ 非线性
338 return std::numeric_limits<lmmc_real_t>::quiet_NaN();
339}

437
438 lmmc_real_t k = extract_linear_coefficient(n.arguments()[0], target_var);
439 if (std::isnan(k) || k == 0.0) {

442 } else {

445 period = LMMC_CONST_PI / std::fabs(k);
446 } else {
447 /// sin/cos
448 period = 2.0 * LMMC_CONST_PI / std::fabs(k);
449 }
450 if (period > max_period) {
451 max_period = period;
452 }
453 }
454 }
455 }
456
457 /// 递归检查参数子树
458 for (auto& arg : n.arguments()) {
459 arg->accept(*this);
460 }
461 }

464std::optional<SearchInterval> determine_search_interval(
465 const std::shared_ptr<SymbolicExpr>& expr,
466 const std::string& var,
467 const SolveOptions& opts)
468{
469 /// 优先级 1: 用户指定区间
470 if (opts.has_search_interval) {
471 if (opts.search_lo >= opts.search_hi) return std::nullopt;
472 if ((opts.search_hi - opts.search_lo) <= opts.tolerance) return std::nullopt;
473 return SearchInterval{opts.search_lo, opts.search_hi};
474 }
475
476 /// 优先级 2: 周期扩展
477 lmmc_real_t lo = -10.0;
478 lmmc_real_t hi = 10.0;
479
480 if (expr && lamina::detail::node(expr)) {
481 PeriodicCollector collector(var);
482 lamina::detail::node(expr)->accept(collector);
483
484 if (collector.found_periodic && !collector.has_nonlinear_periodic && collector.max_period > 0.0) {
485 /// 扩展区间覆盖 2 个完整周期（对称于 0）
486 lmmc_real_t half_span = collector.max_period; // 2 periods / 2 = period (symmetric)
487 if (half_span > lo * -1.0 || half_span > hi) {
488 lo = -half_span;
489 hi = half_span;
490 }
491 /// 钳制到 [-100, 100]
492 if (lo < -100.0) lo = -100.0;
493 if (hi > 100.0) hi = 100.0;
494 }
495 }
496
497 /// 最终验证
498 if (lo >= hi) return std::nullopt;
499 if ((hi - lo) <= opts.tolerance) return std::nullopt;
500
501 return SearchInterval{lo, hi};
502}

506 * @internal
507 * @brief 在指定点对表达式进行数值求值。
508 *
509 * 将变量替换为数值后调用 to_numeric() 获取浮点结果。
510 * 若求值产生 NaN 或无穷大，返回 NaN。
511 *
512 * @param[in] expr 待求值表达式
513 * @param[in] var 变量名
514 * @param[in] x 求值点
515 * @return 数值结果；异常时返回 NaN
516 */
517/**
518 * @internal
519 * @brief 递归数值求值：遍历 AST 计算表达式的浮点值。
520 *
521 * 处理 NumberNode、FunctionNode、AddNode、MultiplyNode、PowerNode。
522 * 若遇到无法求值的节点（如 VariableNode），返回 NaN。
523 */
524static lmmc_real_t recursive_eval(const std::shared_ptr<const SymbolicNode>& node) {
525 if (!node) return 0.0;
526
527 if (auto num = std::dynamic_pointer_cast<const NumberNode>(node)) {
528 if (std::holds_alternative<lmmc_real_t>(num->value()))
529 return std::get<lmmc_real_t>(num->value());
530 if (std::holds_alternative<BigInt>(num->value()))
531 return static_cast<lmmc_real_t>(std::get<BigInt>(num->value()).to_double());
532 if (std::holds_alternative<Rational>(num->value()))
533 return static_cast<lmmc_real_t>(std::get<Rational>(num->value()).to_double());
534 return 0.0;
535 }
536
537 if (std::dynamic_pointer_cast<const VariableNode>(node)) {
538 return std::numeric_limits<lmmc_real_t>::quiet_NaN();
539 }
540
541 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
542 lmmc_real_t sum = 0.0;
543 for (const auto& op : add->operands()) {
544 lmmc_real_t v = recursive_eval(op);
545 if (std::isnan(v)) return v;
546 sum += v;
547 }
548 return sum;
549 }
550
551 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
552 lmmc_real_t product = 1.0;
553 for (const auto& op : mul->operands()) {
554 lmmc_real_t v = recursive_eval(op);
555 if (std::isnan(v)) return v;
556 product *= v;
557 }
558 return product;
559 }
560
561 if (auto pw = std::dynamic_pointer_cast<const PowerNode>(node)) {
562 lmmc_real_t base = recursive_eval(pw->base());
563 lmmc_real_t exp = recursive_eval(pw->exponent());
564 if (std::isnan(base) || std::isnan(exp))
565 return std::numeric_limits<lmmc_real_t>::quiet_NaN();
566 if (base == 0.0 && exp < 0.0)
567 return std::numeric_limits<lmmc_real_t>::quiet_NaN();
568 return std::pow(base, exp);
569 }
570
571 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(node)) {
572 if (func->arguments().size() == 1) {
573 lmmc_real_t arg = recursive_eval(func->arguments()[0]);
574 if (std::isnan(arg)) return arg;
575 switch (func->type()) {
576 case FunctionNode::FuncType::Sin: return std::sin(arg);
577 case FunctionNode::FuncType::Cos: return std::cos(arg);
578 case FunctionNode::FuncType::Tan: return std::tan(arg);
579 case FunctionNode::FuncType::Exp: return std::exp(arg);
580 case FunctionNode::FuncType::Ln: return std::log(arg);
581 case FunctionNode::FuncType::Sqrt: return std::sqrt(arg);
582 case FunctionNode::FuncType::Abs: return std::abs(arg);
583 case FunctionNode::FuncType::ArcSin:
584 if (arg < -1.0 || arg > 1.0) return std::numeric_limits<lmmc_real_t>::quiet_NaN();
585 return std::asin(arg);
586 case FunctionNode::FuncType::ArcCos:
587 if (arg < -1.0 || arg > 1.0) return std::numeric_limits<lmmc_real_t>::quiet_NaN();
588 return std::acos(arg);
589 case FunctionNode::FuncType::ArcTan: return std::atan(arg);
590 case FunctionNode::FuncType::Sinh: return std::sinh(arg);
591 case FunctionNode::FuncType::Cosh: return std::cosh(arg);
592 case FunctionNode::FuncType::Tanh: return std::tanh(arg);
593 case FunctionNode::FuncType::Sec: {
594 lmmc_real_t c = std::cos(arg);
595 if (std::fabs(c) < 1e-15) return std::numeric_limits<lmmc_real_t>::quiet_NaN();
596 return 1.0 / c;
597 }
598 case FunctionNode::FuncType::Csc: {
599 lmmc_real_t s = std::sin(arg);
600 if (std::fabs(s) < 1e-15) return std::numeric_limits<lmmc_real_t>::quiet_NaN();
601 return 1.0 / s;
602 }
603 case FunctionNode::FuncType::Cot: {
604 lmmc_real_t s = std::sin(arg);
605 if (std::fabs(s) < 1e-15) return std::numeric_limits<lmmc_real_t>::quiet_NaN();
606 return std::cos(arg) / s;
607 }
608 default: break;
609 }
610 }
611 if (func->arguments().size() == 2 && func->type() == FunctionNode::FuncType::Atan2) {
612 lmmc_real_t y = recursive_eval(func->arguments()[0]);
613 lmmc_real_t x_val = recursive_eval(func->arguments()[1]);
614 if (std::isnan(y) || std::isnan(x_val)) return std::numeric_limits<lmmc_real_t>::quiet_NaN();
615 return std::atan2(y, x_val);
616 }
617 }
618
619 return std::numeric_limits<lmmc_real_t>::quiet_NaN();
620}

626{
627 try {
628 auto substituted = expr->substitute(var, SymbolicExpr::number(static_cast<double>(x)));
629 if (!substituted || !lamina::detail::node(substituted))
630 return std::numeric_limits<lmmc_real_t>::quiet_NaN();
631 lmmc_real_t val = recursive_eval(lamina::detail::node(substituted));
632 if (!std::isfinite(val)) return std::numeric_limits<lmmc_real_t>::quiet_NaN();
633 return val;
634 } catch (...) {
635 return std::numeric_limits<lmmc_real_t>::quiet_NaN();
636 }
637}

639/**
640 * @internal
641 * @brief 在指定点求值，若结果为 NaN 则尝试在附近重新采样。
642 *
643 * 最多重试 max_retries 次，每次将偏移量减半并向两侧尝试。
644 *
645 * @param[in] expr 表达式
646 * @param[in] var 变量名
647 * @param[in] x 原始采样点
648 * @param[in] half_width 初始偏移量（子区间宽度的一半）
649 * @param[in] max_retries 最大重试次数
650 * @return 有效数值结果；所有重试失败时返回 NaN
651 */
652static lmmc_real_t evaluate_with_retry(
653 const std::shared_ptr<SymbolicExpr>& expr,
654 const std::string& var,
655 lmmc_real_t x,
656 lmmc_real_t half_width,
657 int max_retries = 3)
658{
659 lmmc_real_t val = evaluate_at(expr, var, x);
660 if (!std::isnan(val)) return val;
661
662 lmmc_real_t offset = half_width;
663 for (int i = 0; i < max_retries; ++i) {
664 offset *= 0.5;
665 /// 尝试右侧
666 val = evaluate_at(expr, var, x + offset);
667 if (!std::isnan(val)) return val;
668 /// 尝试左侧
669 val = evaluate_at(expr, var, x - offset);
670 if (!std::isnan(val)) return val;
671 }
672 return std::numeric_limits<lmmc_real_t>::quiet_NaN();
673}

675std::vector<IsolatedInterval> isolate_roots(
676 const std::shared_ptr<SymbolicExpr>& expr,
677 const std::shared_ptr<SymbolicExpr>& derivative,
678 const std::string& var,
679 const SearchInterval& interval,
680 const SolveOptions& opts)
681{
682 std::vector<IsolatedInterval> result;
683
684 /// 无效区间检查
685 if (interval.lo >= interval.hi) return result;
686
687 int max_roots_limit = opts.max_roots;
688 constexpr lmmc_real_t MIN_WIDTH = 1e-6;
689
690 /// 使用栈进行自适应细分（避免深递归）
691 struct SubInterval {
692 lmmc_real_t lo;
693 lmmc_real_t hi;
694 lmmc_real_t f_lo;
695 lmmc_real_t f_hi;
696 };
697
698 /// 初始均匀划分为 N 个子区间
699 constexpr int INITIAL_DIVISIONS = 64;
700 lmmc_real_t total_width = interval.hi - interval.lo;
701 lmmc_real_t step = total_width / INITIAL_DIVISIONS;
702
703 /// 预计算初始采样点的函数值
704 std::vector<lmmc_real_t> sample_vals(INITIAL_DIVISIONS + 1);
705 for (int i = 0; i <= INITIAL_DIVISIONS; ++i) {
706 lmmc_real_t x = interval.lo + i * step;
707 sample_vals[i] = evaluate_with_retry(expr, var, x, step * 0.5);
708 }
709
710 /// 收集需要进一步处理的子区间（含符号变化或需要细分的）
711 std::vector<SubInterval> work_stack;
712 for (int i = 0; i < INITIAL_DIVISIONS; ++i) {
713 lmmc_real_t a = interval.lo + i * step;
714 lmmc_real_t b = interval.lo + (i + 1) * step;
715 lmmc_real_t fa = sample_vals[i];
716 lmmc_real_t fb = sample_vals[i + 1];
717
718 /// 跳过两端都是 NaN 的子区间
719 if (std::isnan(fa) && std::isnan(fb)) continue;
720
721 /// 若一端为 NaN，尝试缩小区间
722 if (std::isnan(fa) || std::isnan(fb)) {
723 /// 将此子区间加入工作栈进行进一步细分
724 work_stack.push_back({a, b, fa, fb});
725 continue;
726 }
727
728 /// 检测符号变化
729 if (fa * fb < 0.0) {
730 work_stack.push_back({a, b, fa, fb});
731 }
732 }
733
734 /// 自适应细分处理工作栈
735 while (!work_stack.empty()) {
736 /// 检查 max_roots 限制
737 if (max_roots_limit > 0 && static_cast<int>(result.size()) >= max_roots_limit) {
738 break;
739 }
740
741 SubInterval current = work_stack.back();
742 work_stack.pop_back();
743
744 lmmc_real_t width = current.hi - current.lo;
745 lmmc_real_t fa = current.f_lo;
746 lmmc_real_t fb = current.f_hi;
747
748 /// 处理 NaN 端点：尝试重新采样
749 if (std::isnan(fa)) {
750 fa = evaluate_with_retry(expr, var, current.lo, width * 0.25);
751 if (std::isnan(fa)) {
752 /// 若宽度足够，细分后重试
753 if (width > MIN_WIDTH * 2.0) {
754 lmmc_real_t mid = (current.lo + current.hi) * 0.5;
755 lmmc_real_t fm = evaluate_with_retry(expr, var, mid, width * 0.125);
756 work_stack.push_back({mid, current.hi, fm, fb});
757 }
758 continue;
759 }
760 }
761 if (std::isnan(fb)) {
762 fb = evaluate_with_retry(expr, var, current.hi, width * 0.25);
763 if (std::isnan(fb)) {
764 if (width > MIN_WIDTH * 2.0) {
765 lmmc_real_t mid = (current.lo + current.hi) * 0.5;
766 lmmc_real_t fm = evaluate_with_retry(expr, var, mid, width * 0.125);
767 work_stack.push_back({current.lo, mid, fa, fm});
768 }
769 continue;
770 }
771 }
772
773 /// 无符号变化 → 跳过
774 if (fa * fb >= 0.0) continue;
775
776 /// 符号变化确认 — 检查是否需要进一步细分
777 /// 若宽度已达最小限制，直接接受
778 if (width <= MIN_WIDTH) {
779 bool confirmed = false;
780 if (derivative) {
781 lmmc_real_t da = evaluate_at(derivative, var, current.lo);
782 lmmc_real_t db = evaluate_at(derivative, var, current.hi);
783 if (!std::isnan(da) && !std::isnan(db) && da * db > 0.0) {
784 confirmed = true;
785 }
786 }
787 result.push_back({current.lo, current.hi, confirmed});
788 continue;
789 }
790
791 /// 单调性确认：若导数在区间内不变号，则恰含一个根
792 if (derivative) {
793 lmmc_real_t da = evaluate_at(derivative, var, current.lo);
794 lmmc_real_t db = evaluate_at(derivative, var, current.hi);
795
796 if (!std::isnan(da) && !std::isnan(db) && da * db > 0.0) {
797 /// 导数不变号 → 单调 → 恰含一个根
798 result.push_back({current.lo, current.hi, true});
799 continue;
800 }
801
802 /// 导数变号 → 可能含多个根，需要细分
803 if (!std::isnan(da) && !std::isnan(db) && da * db < 0.0) {
804 /// 仅在宽度允许时细分
805 if (width > MIN_WIDTH * 2.0) {
806 lmmc_real_t mid = (current.lo + current.hi) * 0.5;
807 lmmc_real_t fm = evaluate_with_retry(expr, var, mid, width * 0.125);
808
809 if (std::isnan(fm)) {
810 /// 中点求值失败，接受当前区间
811 result.push_back({current.lo, current.hi, false});
812 } else {
813 /// 将两半加入工作栈
814 if (fa * fm < 0.0) {
815 work_stack.push_back({current.lo, mid, fa, fm});
816 }
817 if (fm * fb < 0.0) {
818 work_stack.push_back({mid, current.hi, fm, fb});
819 }
820 /// 若两半都无符号变化但原区间有，说明根恰在中点附近
821 if (fa * fm >= 0.0 && fm * fb >= 0.0) {
822 result.push_back({current.lo, current.hi, false});
823 }
824 }
825 continue;
826 }
827 }
828
829 /// 导数有 NaN → 无法确认单调性，接受区间但标记未确认
830 result.push_back({current.lo, current.hi, false});
831 } else {
832 /// 无导数可用 — 尝试细分以缩小区间
833 if (width > MIN_WIDTH * 4.0) {
834 lmmc_real_t mid = (current.lo + current.hi) * 0.5;
835 lmmc_real_t fm = evaluate_with_retry(expr, var, mid, width * 0.125);
836
837 if (std::isnan(fm)) {
838 /// 中点求值失败，接受当前区间
839 result.push_back({current.lo, current.hi, false});
840 } else {
841 if (fa * fm < 0.0) {
842 work_stack.push_back({current.lo, mid, fa, fm});
843 }
844 if (fm * fb < 0.0) {
845 work_stack.push_back({mid, current.hi, fm, fb});
846 }
847 if (fa * fm >= 0.0 && fm * fb >= 0.0) {
848 result.push_back({current.lo, current.hi, false});
849 }
850 }
851 } else {
852 /// 宽度已足够小，接受
853 result.push_back({current.lo, current.hi, false});
854 }
855 }
856 }
857
858 /// 按区间下界排序
859 std::sort(result.begin(), result.end(),
860 [](const IsolatedInterval& a, const IsolatedInterval& b) {
861 return a.lo < b.lo;
862 });
863
864 /// 应用 max_roots 限制
865 if (max_roots_limit > 0 && static_cast<int>(result.size()) > max_roots_limit) {
866 result.resize(static_cast<size_t>(max_roots_limit));
867 }
868
869 return result;
870}

873std::vector<lmmc_real_t> deduplicate_roots(
874 std::vector<NumericRoot>& roots,
875 lmmc_real_t tolerance,
876 int max_roots)
877{
878 if (roots.empty()) return {};
879
880 /// 按根值升序排序
881 std::sort(roots.begin(), roots.end(),
882 [](const NumericRoot& a, const NumericRoot& b) {
883 return a.value < b.value;
884 });
885
886 /// 去重：相邻比较，差值 < 10 * tolerance 视为重复，保留残差较小者
887 lmmc_real_t threshold = 10.0 * tolerance;
888 std::vector<NumericRoot> survivors;
889 survivors.reserve(roots.size());
890 survivors.push_back(roots[0]);
891
892 for (size_t i = 1; i < roots.size(); ++i) {
893 NumericRoot& last = survivors.back();
894 if (std::fabs(roots[i].value - last.value) < threshold) {
895 /// 重复：保留残差更小者（相等时保留先出现的，即 last）
896 if (roots[i].residual < last.residual) {
897 last = roots[i];
898 }
899 } else {
900 survivors.push_back(roots[i]);
901 }
902 }
903
904 /// 收集存活根的值（已按升序排列）
905 std::vector<lmmc_real_t> result;
906 result.reserve(survivors.size());
907 for (const auto& r : survivors) {
908 result.push_back(r.value);
909 }
910
911 /// 应用 max_roots 限制
912 if (max_roots > 0 && static_cast<int>(result.size()) > max_roots) {
913 result.resize(static_cast<size_t>(max_roots));
914 }
915
916 return result;
917}

920/**
921 * @internal
922 * @brief 对隔离区间执行根精化，返回高精度数值根。
923 *
924 * 算法流程：
925 * 1. 以区间中点为初始猜测；
926 * 2. 若导数可用，执行带区间约束的 Newton-Raphson 迭代：
927 * - 每步计算 x_new = x - f(x)/f'(x)；
928 * - 若 x_new 超出 [lo, hi] 或残差增大，丢弃该迭代并执行一步二分法；
929 * - 若 f'(x) == 0，执行一步二分法；
930 * 3. 若导数不可用（为 nullptr），执行纯二分法；
931 * 4. 收敛条件：|f(x)| < tolerance；
932 * 5. 达到最大迭代次数后，返回残差最小的迭代结果（仅当残差 < tolerance）。
933 *
934 * @param[in] expr 待求解表达式
935 * @param[in] derivative 表达式的导数（可为 nullptr，此时使用纯二分法）
936 * @param[in] var 求解变量名
937 * @param[in] interval 隔离子区间
938 * @param[in] opts 求解选项（tolerance, max_newton_iterations）
939 * @return 精化后的数值根；未收敛或残差超限时返回 nullopt
940 */
941std::optional<NumericRoot> refine_root(
942 const std::shared_ptr<SymbolicExpr>& expr,
943 const std::shared_ptr<SymbolicExpr>& derivative,
944 const std::string& var,
945 const IsolatedInterval& interval,
946 const SolveOptions& opts)
947{
948 lmmc_real_t lo = interval.lo;
949 lmmc_real_t hi = interval.hi;
950 lmmc_real_t x = (lo + hi) * 0.5;
951
952 lmmc_real_t f_lo = evaluate_at(expr, var, lo);
953 lmmc_real_t f_hi = evaluate_at(expr, var, hi);
954
955 /// 端点已是根的情形
956 if (!std::isnan(f_lo) && std::fabs(f_lo) < opts.tolerance) {
957 return NumericRoot{lo, std::fabs(f_lo), 0};
958 }
959 if (!std::isnan(f_hi) && std::fabs(f_hi) < opts.tolerance) {
960 return NumericRoot{hi, std::fabs(f_hi), 0};
961 }
962
963 /// 跟踪最佳迭代结果（残差最小）
964 lmmc_real_t best_x = x;
965 lmmc_real_t best_residual = std::numeric_limits<lmmc_real_t>::max();
966 int best_iter = 0;
967
968 int max_iter = opts.max_newton_iterations;
969
970 /// 纯二分法路径：导数不可用
971 if (!derivative) {
972 /// 确保端点有效且异号
973 if (std::isnan(f_lo) || std::isnan(f_hi) || f_lo * f_hi > 0.0) {
974 /// 尝试中点求值
975 lmmc_real_t fx = evaluate_at(expr, var, x);
976 if (!std::isnan(fx) && std::fabs(fx) < opts.tolerance) {
977 return NumericRoot{x, std::fabs(fx), 1};
978 }
979 return std::nullopt;
980 }
981
982 for (int i = 1; i <= max_iter; ++i) {
983 lmmc_real_t mid = (lo + hi) * 0.5;
984 lmmc_real_t f_mid = evaluate_at(expr, var, mid);
985
986 if (std::isnan(f_mid)) {
987 /// NaN 时缩小区间继续
988 hi = mid;
989 continue;
990 }
991
992 lmmc_real_t residual = std::fabs(f_mid);
993 if (residual < best_residual) {
994 best_x = mid;
995 best_residual = residual;
996 best_iter = i;
997 }
998
999 if (residual < opts.tolerance) {
1000 return NumericRoot{mid, residual, i};
1001 }
1002
1003 /// 保持符号变化的半区间
1004 if (f_lo * f_mid < 0.0) {
1005 hi = mid;
1006 f_hi = f_mid;
1007 } else {
1008 lo = mid;
1009 f_lo = f_mid;
1010 }
1011
1012 /// 区间宽度收敛
1013 if ((hi - lo) < opts.tolerance) {
1014 lmmc_real_t final_x = (lo + hi) * 0.5;
1015 lmmc_real_t final_res = std::fabs(evaluate_at(expr, var, final_x));
1016 if (!std::isnan(final_res) && final_res < best_residual) {
1017 best_x = final_x;
1018 best_residual = final_res;
1019 best_iter = i;
1020 }
1021 break;
1022 }
1023 }
1024
1025 /// 返回最佳结果（仅当残差 < tolerance）
1026 if (best_residual < opts.tolerance) {
1027 return NumericRoot{best_x, best_residual, best_iter};
1028 }
1029 return std::nullopt;
1030 }
1031
1032 /// 带区间约束的 Newton-Raphson 路径
1033 /// 确保端点函数值有效
1034 if (std::isnan(f_lo) || std::isnan(f_hi)) {
1035 /// 尝试重新采样端点
1036 if (std::isnan(f_lo)) {
1037 f_lo = evaluate_at(expr, var, lo + (hi - lo) * 0.01);
1038 lo = lo + (hi - lo) * 0.01;
1039 }
1040 if (std::isnan(f_hi)) {
1041 f_hi = evaluate_at(expr, var, hi - (hi - lo) * 0.01);
1042 hi = hi - (hi - lo) * 0.01;
1043 }
1044 if (std::isnan(f_lo) || std::isnan(f_hi)) {
1045 return std::nullopt;
1046 }
1047 }
1048
1049 for (int i = 1; i <= max_iter; ++i) {
1050 lmmc_real_t fx = evaluate_at(expr, var, x);
1051
1052 if (std::isnan(fx)) {
1053 /// NaN 时执行二分步
1054 x = (lo + hi) * 0.5;
1055 continue;
1056 }
1057
1058 lmmc_real_t residual = std::fabs(fx);
1059
1060 /// 更新最佳结果
1061 if (residual < best_residual) {
1062 best_x = x;
1063 best_residual = residual;
1064 best_iter = i;
1065 }
1066
1067 /// 收敛判定
1068 if (residual < opts.tolerance) {
1069 return NumericRoot{x, residual, i};
1070 }
1071
1072 /// 计算导数值
1073 lmmc_real_t dfx = evaluate_at(derivative, var, x);
1074
1075 /// 导数为零或 NaN → 执行二分步
1076 if (std::isnan(dfx) || std::fabs(dfx) < 1e-15) {
1077 /// 二分步：保持符号变化
1078 lmmc_real_t mid = (lo + hi) * 0.5;
1079 lmmc_real_t f_mid = evaluate_at(expr, var, mid);
1080
1081 if (!std::isnan(f_mid)) {
1082 if (!std::isnan(f_lo) && f_lo * f_mid < 0.0) {
1083 hi = mid;
1084 f_hi = f_mid;
1085 } else {
1086 lo = mid;
1087 f_lo = f_mid;
1088 }
1089 }
1090 x = (lo + hi) * 0.5;
1091 continue;
1092 }
1093
1094 /// Newton 更新
1095 lmmc_real_t x_new = x - fx / dfx;
1096
1097 /// 检查 Newton 迭代是否有效：
1098 /// 1. x_new 必须在 [lo, hi] 内
1099 /// 2. 残差不应增大
1100 bool do_bisection = false;
1101
1102 if (x_new < lo || x_new > hi) {
1103 do_bisection = true;
1104 } else {
1105 /// 检查新点的残差是否增大
1106 lmmc_real_t f_new = evaluate_at(expr, var, x_new);
1107 if (std::isnan(f_new) || std::fabs(f_new) > residual) {
1108 do_bisection = true;
1109 } else {
1110 /// Newton 步有效，更新状态
1111 /// 缩小区间：根据 fx 的符号更新端点
1112 if (!std::isnan(f_lo) && fx * f_lo < 0.0) {
1113 hi = x;
1114 f_hi = fx;
1115 } else if (!std::isnan(f_hi) && fx * f_hi < 0.0) {
1116 lo = x;
1117 f_lo = fx;
1118 }
1119 x = x_new;
1120 continue;
1121 }
1122 }
1123
1124 if (do_bisection) {
1125 /// 丢弃 Newton 迭代，执行一步二分法
1126 lmmc_real_t mid = (lo + hi) * 0.5;
1127 lmmc_real_t f_mid = evaluate_at(expr, var, mid);
1128
1129 if (!std::isnan(f_mid)) {
1130 if (!std::isnan(f_lo) && f_lo * f_mid < 0.0) {
1131 hi = mid;
1132 f_hi = f_mid;
1133 } else {
1134 lo = mid;
1135 f_lo = f_mid;
1136 }
1137 }
1138 x = (lo + hi) * 0.5;
1139 }
1140
1141 /// 区间宽度收敛
1142 if ((hi - lo) < opts.tolerance) {
1143 lmmc_real_t final_x = (lo + hi) * 0.5;
1144 lmmc_real_t final_fx = evaluate_at(expr, var, final_x);
1145 if (!std::isnan(final_fx) && std::fabs(final_fx) < opts.tolerance) {
1146 return NumericRoot{final_x, std::fabs(final_fx), i};
1147 }
1148 break;
1149 }
1150 }
1151
1152 /// 达到最大迭代次数：返回最佳结果（仅当残差 < tolerance）
1153 if (best_residual < opts.tolerance) {
1154 return NumericRoot{best_x, best_residual, best_iter};
1155 }
1156 return std::nullopt;
1157}

1186/**
1187 * @internal
1188 * @brief 对单个因子执行数值路径：根隔离 + 精化。
1189 *
1190 * 计算因子的导数，调用 isolate_roots 隔离根，再逐一调用 refine_root 精化。
1191 *
1192 * @param[in] factor 待求解因子表达式
1193 * @param[in] var 变量名
1194 * @param[in] interval 搜索区间
1195 * @param[in] opts 求解选项
1196 * @return 精化后的数值根列表
1197 */
1198static std::vector<NumericRoot> numerical_path(
1199 const std::shared_ptr<SymbolicExpr>& factor,
1200 const std::string& var,
1201 const SearchInterval& interval,
1202 const SolveOptions& opts)
1203{
1204 std::vector<NumericRoot> roots;
1205
1206 /// 计算导数

1236 const std::shared_ptr<SymbolicExpr>& expr,
1237 const std::string& var,
1238 const SolveOptions& opts)
1239{
1240 try {
1241 /// 1. 早期退出：表达式不依赖变量
1242 if (!expr || !lamina::detail::node(expr) || !expression_depends_on_variable(lamina::detail::node(expr), var)) {
1243 return {};
1244 }
1245
1246 /// 2. 确定搜索区间
1247 auto interval_opt = determine_search_interval(expr, var, opts);
1248 if (!interval_opt.has_value()) {
1249 return {};
1250 }
1251 SearchInterval interval = interval_opt.value();
1252
1253 /// 3. 符号预处理：因式分解
1254 auto factors = factor_transcendental(expr, var);
1255
1256 /// 收集所有数值根
1257 std::vector<NumericRoot> all_roots;
1258
1259 /// 4. 多因子处理
1260 bool is_single_factor_same_as_input = false;
1261 if (factors.size() == 1) {
1262 /// 检查唯一因子是否结构等于原表达式
1263 if (factors[0] && factors[0]->to_string() == expr->to_string()) {
1264 is_single_factor_same_as_input = true;
1265 }
1266 }
1267
1268 if (!is_single_factor_same_as_input && factors.size() > 1) {
1269 /// 多因子：对每个因子独立求解
1270 for (const auto& factor : factors) {
1271 if (!factor || !lamina::detail::node(factor)) continue;
1272
1273 /// 跳过不依赖变量的因子（常数因子）
1274 if (!expression_depends_on_variable(lamina::detail::node(factor), var)) continue;
1275
1276 /// (a) 尝试转换为多项式
1277 bool solved_as_poly = false;
1278 try {
1279 auto poly = symbolic_to_poly<SymbolicPolyCoeff>(factor, var);
1280 int deg = poly.degree();
1281 if (deg >= 1 && deg <= 4) {
1282 /// 委托给多项式求解器
1283 auto poly_roots = solve_by_factoring(poly, var);
1284 for (const auto& root_expr : poly_roots) {
1285 if (!root_expr) continue;
1286 /// 提取数值
1287 auto simplified = root_expr->simplify();
1288 if (!simplified || !lamina::detail::node(simplified)) continue;
1289 lmmc_real_t val = recursive_eval(lamina::detail::node(simplified));
1290 if (std::isfinite(val)) {
1291 /// 验证根在搜索区间内
1292 if (val >= interval.lo && val <= interval.hi) {
1293 lmmc_real_t residual = std::fabs(evaluate_at(expr, var, val));
1294 if (!std::isnan(residual)) {
1295 all_roots.push_back({val, residual, 0});
1296 }
1297 }
1298 }
1299 }
1300 solved_as_poly = true;
1301 }
1302 } catch (...) {
1303 /// 转换失败，不是多项式
1304 }
1305
1306 if (solved_as_poly) continue;
1307
1308 /// (b) 尝试 solve_transcendental（收集符号解的数值）
1309 try {
1310 auto trans_roots = solve_transcendental(factor, var);
1311 if (!trans_roots.empty()) {
1312 for (const auto& root_expr : trans_roots) {
1313 if (!root_expr) continue;
1314 auto simplified = root_expr->simplify();
1315 if (!simplified || !lamina::detail::node(simplified)) continue;
1316 lmmc_real_t val = recursive_eval(lamina::detail::node(simplified));
1317 if (std::isfinite(val)) {
1318 if (val >= interval.lo && val <= interval.hi) {
1319 lmmc_real_t residual = std::fabs(evaluate_at(expr, var, val));
1320 if (!std::isnan(residual)) {
1321 all_roots.push_back({val, residual, 0});
1322 }
1323 }
1324 }
1325 }
1326 }
1327 } catch (...) {
1328 /// solve_transcendental 失败
1329 }
1330
1331 /// (c) 数值路径：对该因子执行根隔离 + 精化
1332 /// 始终执行数值路径以捕获区间内所有周期根
1333 auto factor_roots = numerical_path(factor, var, interval, opts);
1334 all_roots.insert(all_roots.end(), factor_roots.begin(), factor_roots.end());
1335 }
1336 } else {
1337 /// 5. 单因子等于输入 或 分解返回空/单因子：直接数值路径
1338 auto numeric_roots = numerical_path(expr, var, interval, opts);
1339 all_roots.insert(all_roots.end(), numeric_roots.begin(), numeric_roots.end());
1340 }
1341
1342 /// 6. 结果组装：去重、排序、转换
1343 auto deduplicated = deduplicate_roots(all_roots, opts.tolerance, opts.max_roots);
1344 return assemble_results(deduplicated);
1345
1346 } catch (...) {
1347 return {};
1348 }
1349}
