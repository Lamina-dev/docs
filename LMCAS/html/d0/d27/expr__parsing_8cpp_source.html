245ExprResult ExprParser::parse_primary() {
246 skip_space();
247 if (match('(')) {
248 auto inner = parse_logical_or();
249 if (!inner) { return inner; }
250 if (match(',')) {
251 auto rhs = parse_logical_or();
252 if (!rhs) { return rhs; }
253 if (!match(')')) { return fail("expected ')' after interval"); }
254 return interval(inner.value(), rhs.value(), false, false);
255 }
256 if (!match(')')) { return fail("expected ')'"); }
257 return inner;
258 }
259 if (match('{')) {
260 return parse_set_literal();
261 }
262 if (match('[')) {
263 return parse_interval_literal(true);
264 }
265 if (std::isdigit(static_cast<unsigned char>(peek())) || peek() == '.') {
266 return parse_number();
267 }
268 if (is_ident_start(static_cast<unsigned char>(peek()))) {
269 return parse_identifier_or_call();
270 }
271 return fail("expected expression");
272}

289 auto lower = parse_logical_or();

305ExprResult ExprParser::parse_number() {
306 skip_space();
307 const std::size_t start = pos_;
308 bool saw_digit = false;
309 while (std::isdigit(static_cast<unsigned char>(peek()))) {
310 saw_digit = true;
311 ++pos_;
312 }
313 if (peek() == '.') {
314 ++pos_;
315 while (std::isdigit(static_cast<unsigned char>(peek()))) {
316 saw_digit = true;
317 ++pos_;
318 }
319 }
320 if (!saw_digit) { return fail("invalid number literal"); }
321 if (peek() == 'e' || peek() == 'E') {
322 ++pos_;
323 if (peek() == '+' || peek() == '-') ++pos_;
324 bool saw_exp_digit = false;
325 while (std::isdigit(static_cast<unsigned char>(peek()))) {
326 saw_exp_digit = true;
327 ++pos_;
328 }
329 if (!saw_exp_digit) { return fail("invalid number exponent"); }
330 }
331 try {
332 return rational(Rational(source_.substr(start, pos_ - start)));
333 } catch (const std::bad_alloc&) {
334 return expr_detail::expr_common::expression_failure(CasErrc::ResourceLimit,
335 "number literal allocation failed",
336 kParseOperation);
337 } catch (const std::invalid_argument& error) {
338 return expr_detail::expr_common::expression_failure(
339 CasErrc::ParseError, error.what(), kParseOperation);
340 } catch (const std::out_of_range& error) {
341 return expr_detail::expr_common::expression_failure(
342 CasErrc::ParseError, error.what(), kParseOperation);
343 } catch (const std::exception& error) {
344 return expr_detail::expr_common::expression_failure(
345 CasErrc::InternalInvariant, error.what(), kParseOperation);
346 }

370ExprResult ExprParser::parse_call(const std::string& name) {
371 if (name == "approx") return parse_approx_literal();
372 std::vector<ExprPtr> arguments;
373 bool closed = match(')');
374 if (!closed) {
375 do {
376 auto argument = parse_logical_or();
377 if (!argument) {
378 return argument;
379 }
380 arguments.push_back(argument.value());
381 if (match(')')) {
382 closed = true;
383 break;
384 }
385 } while (match(','));
386 }
387 if (!closed) {
388 return fail("expected ')'");
389 }
390 return apply_function(name, arguments);
391}

393ExprResult ExprParser::parse_identifier_or_call() {
394 auto name = parse_identifier();
395 if (name.empty()) {
396 return fail("expected identifier");
397 }
398 skip_space();
399 if (match('(')) {
400 return parse_call(name);
401 }
402 if (name == "pi" || name == "π") {
403 return pi();
404 }
405 if (name == "e") {

407 }

412 return imaginary_unit();
