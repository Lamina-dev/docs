21 ExprResult parse() {
22 skip_space();
23 if (eof()) {
24 return fail("empty expression");
25 }
26 auto result = parse_logical_or();
27 if (!result) return result;
28 skip_space();
29 if (!eof()) {
30 return fail("unexpected token '" + std::string(1, peek()) + "'");
31 }
32 return result;
33 }

41 char peek() const noexcept {
42 return eof() ? '\0' : source_[pos_];

45 bool match(char c) {
46 skip_space();
47 if (peek() != c) return false;
48 ++pos_;
49 return true;

52 bool match_text(const char* text) {
53 skip_space();
54 const std::size_t start = pos_;
55 for (const char* p = text; *p; ++p) {
56 if (pos_ >= source_.size() || source_[pos_] != *p) {
57 pos_ = start;
58 return false;
59 }
60 ++pos_;
61 }
62 return true;
63 }

65 bool match_keyword(const char* text) {
66 skip_space();
67 const std::size_t start = pos_;
68 if (!match_text(text)) return false;
69 if (is_ident_continue(static_cast<unsigned char>(peek()))) {
70 pos_ = start;
71 return false;
72 }
73 return true;
74 }

76 void skip_space() noexcept {
77 while (!eof() &&
78 std::isspace(static_cast<unsigned char>(source_[pos_]))) {

105 ExprResult parse_logical_or() {
106 auto lhs = parse_logical_and();
107 if (!lhs) return lhs;
108 while (match_keyword("or")) {
109 auto rhs = parse_logical_and();
110 if (!rhs) return rhs;
111 lhs = logical(lhs.value(), rhs.value(), LogicalNode::Op::Or);
112 if (!lhs) return lhs;
113 }
114 return lhs;
115 }

117 ExprResult parse_logical_and() {
118 auto lhs = parse_logical_not();
119 if (!lhs) return lhs;
120 while (match_keyword("and")) {
121 auto rhs = parse_logical_not();
122 if (!rhs) return rhs;
123 lhs = logical(lhs.value(), rhs.value(), LogicalNode::Op::And);
124 if (!lhs) return lhs;
125 }
126 return lhs;
127 }

129 ExprResult parse_logical_not() {
130 if (match_keyword("not") || match('!')) {
131 auto operand = parse_logical_not();
132 if (!operand) return operand;
133 return logical(operand.value(), nullptr, LogicalNode::Op::Not);
134 }
135 return parse_membership();
136 }

138 ExprResult parse_membership() {
139 auto lhs = parse_equality();
140 if (!lhs) return lhs;
141 if (match_keyword("in")) {
142 auto rhs = parse_equality();
143 if (!rhs) return rhs;
144 return membership(lhs.value(), rhs.value(), false);
145 }
146 const auto save = pos_;
147 if (match_keyword("not")) {
148 if (match_keyword("in")) {
149 auto rhs = parse_equality();
150 if (!rhs) return rhs;
151 return membership(lhs.value(), rhs.value(), true);
152 }
153 pos_ = save;
154 }
155 return lhs;
156 }

158 ExprResult parse_equality() {
159 auto lhs = parse_relational();
160 if (!lhs) return lhs;
161 while (true) {
162 RelationOp op;
163 if (match_text("==")) {
164 op = RelationOp::EQ;
165 } else if (match_text("!=")) {
166 op = RelationOp::NEQ;
167 } else {
168 return lhs;
169 }
170 auto rhs = parse_relational();
171 if (!rhs) return rhs;
172 lhs = relational(lhs.value(), rhs.value(), op);
173 if (!lhs) return lhs;
174 }
175 }

177 ExprResult parse_relational() {
178 auto lhs = parse_additive();
179 if (!lhs) return lhs;
180 while (true) {
181 RelationOp op;
182 if (match_text("<=")) {
183 op = RelationOp::LEQ;
184 } else if (match_text(">=")) {
185 op = RelationOp::GEQ;
186 } else if (match('<')) {
187 op = RelationOp::LT;
188 } else if (match('>')) {
189 op = RelationOp::GT;
190 } else {
191 return lhs;
192 }
193 auto rhs = parse_additive();
194 if (!rhs) return rhs;
195 lhs = relational(lhs.value(), rhs.value(), op);
196 if (!lhs) return lhs;
197 }
198 }

200 ExprResult parse_additive() {
201 auto lhs = parse_multiplicative();
202 if (!lhs) return lhs;
203 while (true) {
204 if (match('+')) {
205 auto rhs = parse_multiplicative();
206 if (!rhs) return rhs;
207 lhs = add(lhs.value(), rhs.value());
208 } else if (match('-')) {
209 auto rhs = parse_multiplicative();
210 if (!rhs) return rhs;
211 lhs = sub(lhs.value(), rhs.value());
212 } else {
213 return lhs;
214 }
215 if (!lhs) return lhs;
216 }

219 ExprResult parse_multiplicative() {
220 auto lhs = parse_power();
221 if (!lhs) return lhs;
222 while (true) {
223 if (match_text("**")) {
224 return fail("operator '**' is not supported; use '^'");
225 } else if (match('*')) {
226 auto rhs = parse_power();
227 if (!rhs) return rhs;
228 lhs = mul(lhs.value(), rhs.value());
229 } else if (match('/')) {
230 auto rhs = parse_power();
231 if (!rhs) return rhs;
232 lhs = div(lhs.value(), rhs.value());
233 } else {
234 return lhs;
235 }
236 if (!lhs) return lhs;
237 }

240 ExprResult parse_power() {
241 auto base = parse_unary();
242 if (!base) return base;
243 if (match_text("**")) {
244 return fail("operator '**' is not supported; use '^'");
245 }
246 if (match('^')) {
247 auto exponent = parse_power();
248 if (!exponent) return exponent;
249 return pow(base.value(), exponent.value());
250 }
251 return base;

254 ExprResult parse_unary() {
255 if (match('+')) return parse_unary();
256 if (match('-')) {
257 auto operand = parse_unary();
258 if (!operand) return operand;
259 return neg(operand.value());
260 }
261 return parse_primary();

264 ExprResult parse_primary() {
265 skip_space();
266 if (match('(')) {
267 auto inner = parse_logical_or();
268 if (!inner) return inner;
269 if (match(',')) {
270 auto rhs = parse_logical_or();
271 if (!rhs) return rhs;
272 if (!match(')')) return fail("expected ')' after interval");
273 return interval(inner.value(), rhs.value(), false, false);
274 }
275 if (!match(')')) return fail("expected ')'");
276 return inner;
277 }
278 if (match('{')) {
279 return parse_set_literal();
280 }
281 if (match('[')) {
282 return parse_interval_literal(true);
283 }
284 if (std::isdigit(static_cast<unsigned char>(peek())) || peek() == '.') {
285 return parse_number();
286 }
287 if (is_ident_start(static_cast<unsigned char>(peek()))) {
288 return parse_identifier_or_call();
289 }
290 return fail("expected expression");
291 }

293 ExprResult parse_set_literal() {
294 std::vector<ExprPtr> elements;
295 if (!match('}')) {
296 while (true) {
297 auto element = parse_logical_or();
298 if (!element) return element;
299 elements.push_back(element.value());
300 if (match('}')) break;
301 if (!match(',')) return fail("expected ',' or '}' in set literal");
302 }
303 }
304 return finite_set(elements);
305 }

307 ExprResult parse_interval_literal(bool lower_closed) {
308 auto lower = parse_logical_or();
309 if (!lower) return lower;
310 if (!match(',')) return fail("expected ',' in interval literal");
311 auto upper = parse_logical_or();
312 if (!upper) return upper;
313 bool upper_closed;
314 if (match(']')) {
315 upper_closed = true;
316 } else if (match(')')) {
317 upper_closed = false;
318 } else {
319 return fail("expected ']' or ')' after interval literal");
320 }
321 return interval(lower.value(), upper.value(), lower_closed, upper_closed);
322 }

324 ExprResult parse_number() {
325 skip_space();
326 const std::size_t start = pos_;
327 bool saw_digit = false;
328 while (std::isdigit(static_cast<unsigned char>(peek()))) {
329 saw_digit = true;
330 ++pos_;
331 }
332 if (peek() == '.') {
333 ++pos_;
334 while (std::isdigit(static_cast<unsigned char>(peek()))) {
335 saw_digit = true;
336 ++pos_;
337 }
338 }
339 if (!saw_digit) return fail("invalid number literal");
340 if (peek() == 'e' || peek() == 'E') {
341 ++pos_;
342 if (peek() == '+' || peek() == '-') ++pos_;
343 bool saw_exp_digit = false;
344 while (std::isdigit(static_cast<unsigned char>(peek()))) {
345 saw_exp_digit = true;
346 ++pos_;
347 }
348 if (!saw_exp_digit) return fail("invalid number exponent");
349 }
350 try {
351 return rational(Rational(source_.substr(start, pos_ - start)));
352 } catch (const std::bad_alloc&) {
353 return expression_failure(CasErrc::ResourceLimit,
354 "number literal allocation failed",
355 kParseOperation);
356 } catch (const std::exception& error) {
357 return expression_failure(CasErrc::ParseError, error.what(),
358 kParseOperation);
359 }

362 std::string parse_identifier() {
363 skip_space();
364 const std::size_t start = pos_;
365 if (!is_ident_start(static_cast<unsigned char>(peek()))) return {};
366 ++pos_;
367 while (is_ident_continue(static_cast<unsigned char>(peek()))) ++pos_;
368 return source_.substr(start, pos_ - start);

371 ExprResult parse_identifier_or_call() {
372 auto name = parse_identifier();
373 if (name.empty()) return fail("expected identifier");
374 skip_space();
375 if (match('(')) {
376 std::vector<ExprPtr> arguments;
377 bool closed = match(')');
378 if (!closed) {
379 do {
380 auto argument = parse_logical_or();
381 if (!argument) return argument;
382 arguments.push_back(argument.value());
383 if (match(')')) {
384 closed = true;
385 break;
386 }
387 } while (match(','));
388 }
389 if (!closed) return fail("expected ')'");
390 return apply_function(name, arguments);
391 }
392 if (name == "pi" || name == "π") return pi();
393 if (name == "e") return e();
394 if (name == "phi") return phi();
395 if (is_imaginary_unit_name(name)) return imaginary_unit();
396 return sym(name);
397 }

550 ExprResult apply_function(const std::string& name,
551 const std::vector<ExprPtr>& arguments) {
552 if (name == "sin" && arguments.size() == 1) return sin(arguments[0]);
553 if (name == "cos" && arguments.size() == 1) return cos(arguments[0]);
554 if (name == "tan" && arguments.size() == 1) return tan(arguments[0]);
555 if (name == "asin" && arguments.size() == 1) return asin(arguments[0]);
556 if (name == "acos" && arguments.size() == 1) return acos(arguments[0]);
557 if (name == "atan" && arguments.size() == 1) return atan(arguments[0]);
558 if (name == "sqrt" && arguments.size() == 1) return sqrt(arguments[0]);
559 if (name == "exp" && arguments.size() == 1) return exp(arguments[0]);
560 if ((name == "ln" || name == "log") && arguments.size() == 1) return log(arguments[0]);
561 if (name == "log" && arguments.size() == 2) {
562 try {
563 auto result = SymbolicExpr::log(arguments[0], arguments[1]);
564 return ExprResult::success(std::move(result));
565 } catch (const std::bad_alloc&) {
566 return expression_failure(CasErrc::ResourceLimit,
567 "log expression allocation failed",
568 kParseOperation);
569 } catch (const std::exception& error) {
570 return expression_failure(CasErrc::InvalidArgument, error.what(),
571 kParseOperation);
572 }
573 }
574 if (name == "atan2" && arguments.size() == 2) {
575 try {
576 auto result = SymbolicExpr::atan2(arguments[0], arguments[1]);
577 return ExprResult::success(std::move(result));
578 } catch (const std::bad_alloc&) {
579 return expression_failure(CasErrc::ResourceLimit,
580 "atan2 expression allocation failed",
581 kParseOperation);
582 } catch (const std::exception& error) {
583 return expression_failure(CasErrc::InvalidArgument, error.what(),
584 kParseOperation);
585 }
586 }
587 if (name == "log10" && arguments.size() == 1) return log10(arguments[0]);
588 if (name == "floor" && arguments.size() == 1) return floor(arguments[0]);
589 if (name == "ceil" && arguments.size() == 1) return ceil(arguments[0]);
590 if (name == "round" && arguments.size() == 1) return round(arguments[0]);
591 if (name == "abs" && arguments.size() == 1) return abs(arguments[0]);
592 if (name == "real" && arguments.size() == 1) return real(arguments[0]);
593 if (name == "imag" && arguments.size() == 1) return imag(arguments[0]);
594 if (name == "conj" && arguments.size() == 1) return conj(arguments[0]);
595 if ((name == "Integral" || name == "integral") &&
596 (arguments.size() == 2 || arguments.size() == 4)) {
597 auto variable = std::dynamic_pointer_cast<const VariableNode>(
598 LMCAS::detail::node(arguments[1]));
599 if (!variable) {
600 return fail("Integral variable must be a symbol");
601 }
602 try {
603 auto node = LMCAS::detail::make_node<IntegralNode>(
604 LMCAS::detail::node(arguments[0]), variable->name(),
605 arguments.size() == 4
606 ? LMCAS::detail::node(arguments[2])
607 : nullptr,
608 arguments.size() == 4
609 ? LMCAS::detail::node(arguments[3])
610 : nullptr);
611 return ExprResult::success(
612 LMCAS::detail::make_expression_ptr(std::move(node)));
613 } catch (const std::bad_alloc&) {
614 return expression_failure(
615 CasErrc::ResourceLimit,
616 "Integral expression allocation failed",
617 kParseOperation);
618 } catch (const std::exception& error) {
619 return expression_failure(
620 CasErrc::InvalidArgument, error.what(), kParseOperation);
621 }
622 }
623 if ((name == "limit" || name == "Limit") && arguments.size() == 4) {
624 auto variable = std::dynamic_pointer_cast<const VariableNode>(
625 LMCAS::detail::node(arguments[1]));
626 auto direction = std::dynamic_pointer_cast<const VariableNode>(
627 LMCAS::detail::node(arguments[3]));
628 if (!variable || !direction) {
629 return fail("limit variable and direction must be symbols");
630 }
631 LimitDirection parsed_direction;
632 if (direction->name() == "both") {
633 parsed_direction = LimitDirection::Both;
634 } else if (direction->name() == "below" ||
635 direction->name() == "left" ||
636 direction->name() == "from_below") {
637 parsed_direction = LimitDirection::FromBelow;
638 } else if (direction->name() == "above" ||
639 direction->name() == "right" ||
640 direction->name() == "from_above") {
641 parsed_direction = LimitDirection::FromAbove;
642 } else {
643 return fail("invalid limit direction");
644 }
645 return ExprResult::success(LMCAS::detail::make_expression_ptr(
646 LMCAS::detail::make_node<LimitNode>(
647 LMCAS::detail::node(arguments[0]), variable->name(),
648 LMCAS::detail::node(arguments[2]), parsed_direction)));
649 }
650 if ((name == "rootof" || name == "RootOf") &&
651 arguments.size() == 3) {
652 auto variable = std::dynamic_pointer_cast<const VariableNode>(
653 LMCAS::detail::node(arguments[1]));
654 auto index = std::dynamic_pointer_cast<const NumberNode>(
655 LMCAS::detail::node(arguments[2]));
656 if (!variable || !index ||
657 std::holds_alternative<lmmc_real_t>(index->value())) {
658 return fail("RootOf variable and index are invalid");
659 }
660 BigInt index_value;
661 if (std::holds_alternative<BigInt>(index->value())) {
662 index_value = std::get<BigInt>(index->value());
663 } else {
664 const auto& rational = std::get<Rational>(index->value());
665 if (!rational.is_integer()) {
666 return fail("RootOf index must be an integer");
667 }
668 index_value = rational.to_BigInt();
669 }
670 const auto converted = index_value.try_to_int64();
671 if (!converted || *converted < 0) {
672 return fail("RootOf index is negative or too large");
673 }
674 auto root = make_rootof_checked(
675 arguments[0], variable->name(),
676 static_cast<std::size_t>(*converted));
677 if (!root) return ExprResult::failure(root.error());
678 return ExprResult::success(std::move(root.value()));
679 }
680 if (name == "max" && !arguments.empty()) {
681 return function_node(name, arguments, FunctionNode::FuncType::Max);
682 }
683 if (name == "min" && !arguments.empty()) {
684 return function_node(name, arguments, FunctionNode::FuncType::Min);
685 }
686 if (name == "clamp" && arguments.size() == 3) {
687 return clamp(arguments[0], arguments[1], arguments[2]);
688 }
689 if (name == "sin" || name == "cos" || name == "tan" ||
690 name == "asin" || name == "acos" || name == "atan" ||
691 name == "sqrt" || name == "exp" || name == "ln" ||
692 name == "log" || name == "log10" || name == "floor" ||
693 name == "ceil" || name == "round" || name == "abs" ||
694 name == "real" || name == "imag" || name == "conj" ||
695 name == "atan2" || name == "clamp" ||
696 name == "Integral" || name == "integral" ||
697 name == "Limit" || name == "limit" ||
698 name == "RootOf" || name == "rootof") {
699 return fail("invalid argument count for function '" + name + "'");
700 }
701 try {
702 std::vector<std::shared_ptr<const SymbolicNode>> nodes;
703 nodes.reserve(arguments.size());
