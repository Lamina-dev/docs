17Result<Tribool> parameter_constraint(const ExprPtr& expression, ComputationContext& context) {
18 std::optional<detail::AssumptionFacts> assumptions;
19 if (context.assumptions()) { assumptions.emplace(*context.assumptions()); }
20 const FactsQuery& facts = assumptions ? static_cast<const FactsQuery&>(*assumptions) : detail::no_facts();
21 auto nonzero = detail::query_nonzero_value(detail::node(expression), facts, Domain::Complex, context);
22 if (!nonzero) { return Result<Tribool>::failure(nonzero.error()); }
23 if (nonzero.value() == Tribool::True) { return Tribool::False; }
24 if (nonzero.value() == Tribool::False) { return Tribool::True; }
25 auto residual = check_zero_residual(expression, context);
26 if (!residual) { return Result<Tribool>::failure(residual.error()); }
27 return std::holds_alternative<ProvedZeroResidual>(residual.value()) ?
28 Tribool::True : Tribool::Unknown;
29}

54 auto G_basis = Solver::groebner_basis(input_polys, unknowns);

58 for (const auto& g : G_basis) {
59 auto g_ptr = LMCAS::detail::make_expression_ptr(g);

97 Result<bool> reduce_basis(int position, const ParametricAssignment& partial,
98 ParametricBasis& reduced) const {
99 reduced.reserve(basis_.size());
100 for (const auto& polynomial : basis_) {
101 auto expression = substitute_assignment(polynomial, partial);
102 if (!expression) {
103 return Result<bool>::failure(CasErrc::InternalInvariant,
104 "assignment substitution failed", "solve.parametric");
105 }
106 if (expression->is_zero()) {
107 continue;
108 }
109 if (!depends_on_remaining_unknown(*expression, unknowns_, position)) {
110 auto residual = parameter_constraint(expression, context_);
111 if (!residual) return Result<bool>::failure(residual.error());
112 if (residual.value() == Tribool::False) return false;
113 if (residual.value() != Tribool::True)
114 return Result<bool>::failure(CasErrc::Inconclusive,
115 "remaining parameter constraint is not proved", "solve.parametric");
116 continue;
117 }
118 reduced.push_back(expression);
119 }
120 return true;
121 }

123 Result<std::shared_ptr<SymbolicExpr>> select_target(const ParametricBasis& reduced,
124 int position) const {
125 const auto& variable = unknowns_[position];
126 std::shared_ptr<SymbolicExpr> target;
127 ExprPtr leading;
128 int best_degree = std::numeric_limits<int>::max();
129 bool appears = false;
130 for (const auto& expression : reduced) {
131 if (!contains(*expression, variable)) {
132 continue;
133 }
134 appears = true;
135 if (depends_on_remaining_unknown(*expression, unknowns_, position - 1)) {
136 continue;
137 }
138 auto polynomial = symbolic_to_poly<SymbolicPolyCoeff>(expression, variable);
139 if (!polynomial) {
140 if (polynomial.error().code != CasErrc::UnsupportedExpression) {
141 return Result<std::shared_ptr<SymbolicExpr>>::failure(polynomial.error());
142 }
143 continue;
144 }
145 int degree = polynomial.value().degree();
146 if (degree >= 1 && degree < best_degree) {
147 best_degree = degree;
148 target = expression;
149 leading = polynomial.value().coeffs.back().val;
150 }
151 }
152 if (!target && appears) {
153 return Result<std::shared_ptr<SymbolicExpr>>::failure(CasErrc::Inconclusive,
154 "No convertible polynomial target for remaining variable", "solve.parametric");
155 }
156 if (target) {
157 for (const auto& parameter : parameters_) {
158 if (!contains(*leading, parameter)) continue;
159 auto zero = parameter_constraint(leading, context_);
160 if (!zero) return Result<std::shared_ptr<SymbolicExpr>>::failure(zero.error());
161 if (zero.value() != Tribool::False)
162 return Result<std::shared_ptr<SymbolicExpr>>::failure(CasErrc::Inconclusive,
163 "leading parameter coefficient is not proved nonzero", "solve.parametric");
164 break;
165 }
166 }
167 return target;
168 }

185 ParametricSolutionsResult solve(int position, const ParametricAssignment& partial) const {
186 ParametricBasis reduced;
187 auto consistent = reduce_basis(position, partial, reduced);
188 if (!consistent) return ParametricSolutionsResult::failure(consistent.error());
189 if (!consistent.value()) {
190 return ParametricSolutionList{};
191 }
192 if (position < 0) {
193 return ParametricSolutionList{partial};
194 }
195 auto selected = select_target(reduced, position);
196 if (!selected) return ParametricSolutionsResult::failure(selected.error());
197 if (!selected.value()) {
198 return solve_free_unknown(position, partial);
199 }
200 const auto& variable = unknowns_[position];
201 auto solved = solve_finite_checked(selected.value(), variable, context_);
202 if (!solved) {
203 return ParametricSolutionsResult::failure(solved.error());
204 }
205 auto roots = std::move(solved.value());
206 if (roots.empty()) {
207 return ParametricSolutionList{};
208 }
209 ParametricSolutionList results;
210 for (const auto& root : roots) {
211 auto next = partial;
212 next[variable] = root;
213 auto branch = solve(position - 1, next);
214 if (!branch) {
215 return branch;
216 }
217 auto& values = branch.value();
218 results.insert(results.end(), std::make_move_iterator(values.begin()),
219 std::make_move_iterator(values.end()));
220 }
221 return results;
222 }

227ParametricSolutionsResult ParametricSolver::solve_polynomial_parametric_impl(
228 const std::vector<std::shared_ptr<SymbolicExpr>>& equations,
229 const std::vector<std::string>& unknowns,
230 const std::vector<std::string>& parameters,
231 ComputationContext& context)
232{
233 if (equations.empty() || unknowns.empty()) {
234 return free_unknowns(unknowns);
235 }
236 auto input_polys = parametric_input_polynomials(equations);
237 if (input_polys.empty()) {
238 return free_unknowns(unknowns);
239 }
240 ParametricBasis basis;
241 if (unknowns.size() == 1) {
242 basis.reserve(input_polys.size());
243 for (const auto& polynomial : input_polys)
244 basis.push_back(detail::make_expression_ptr(polynomial));
245 } else {
246 basis = parametric_basis(input_polys, unknowns);
247 }
248 for (const auto& polynomial : basis) {
249 if (!depends_on_remaining_unknown(*polynomial, unknowns, static_cast<int>(unknowns.size()) - 1)) {
250 auto residual = parameter_constraint(polynomial, context);
251 if (!residual) return ParametricSolutionsResult::failure(residual.error());
252 if (residual.value() == Tribool::False)
253 return ParametricSolutionList{};
254 if (residual.value() != Tribool::True)
255 return ParametricSolutionsResult::failure(CasErrc::Inconclusive,
256 "parameter-only constraint is not proved", "solve.parametric");
257 }
258 }
259 ParametricBranchSolver branches(basis, unknowns, parameters, context);
260 ParametricAssignment empty;
261 return branches.solve(static_cast<int>(unknowns.size()) - 1, empty);
262}
