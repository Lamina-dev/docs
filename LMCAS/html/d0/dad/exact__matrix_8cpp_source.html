59Result<void> eliminate_forward(
60 EliminationResult& result, std::size_t coefficient_columns,
61 ComputationContext& context, const std::string& operation) {
62 std::size_t pivot_row = 0;
63 for (std::size_t column = 0;
64 column < coefficient_columns && pivot_row < result.matrix.rows;
65 ++column) {
66 auto pivot = choose_pivot(
67 result.matrix, pivot_row, column, context, operation);
68 if (!pivot) return Result<void>::failure(pivot.error());
69 if (!pivot.value()) continue;
70 if (*pivot.value() != pivot_row) {
71 swap_rows(result.matrix, *pivot.value(), pivot_row);
72 ++result.row_swaps;
73 }
74 auto pivot_value = simplify_expr(result.matrix.at(pivot_row, column));
75 result.pivot_columns.push_back(column);
76 result.pivot_values.push_back(pivot_value);
77 auto reduced = eliminate_below(
78 result.matrix, pivot_row, column, pivot_value, context, operation);
79 if (!reduced) return Result<void>::failure(reduced.error());
80 ++pivot_row;
81 }
82 result.rank = result.pivot_columns.size();
83 return Result<void>::success();
84}

87Result<EliminationResult> eliminate_exact(
88 ExactMatrixData input,
89 std::size_t coefficient_columns,
90 EliminationForm form,
91 ComputationContext& context,
92 const std::string& operation) {
93 auto valid = validate_matrix(input, operation);
94 if (!valid) return Result<EliminationResult>::failure(valid.error());
95 if (coefficient_columns == 0 || coefficient_columns > input.cols) {
96 return Result<EliminationResult>::failure(
97 CasErrc::InvalidArgument,
98 "coefficient column count is outside matrix shape", operation);
99 }
100 try {
101 EliminationResult result;
102 result.matrix = std::move(input);
103 auto forward = eliminate_forward(result, coefficient_columns, context, operation);
104 if (!forward) return Result<EliminationResult>::failure(forward.error());
105
106 if (form == EliminationForm::ReducedRowEchelon) {
107 auto reduced = reduce_above(result, context, operation);
108 if (!reduced) return Result<EliminationResult>::failure(reduced.error());
109 }
110 return Result<EliminationResult>::success(std::move(result));
111 } catch (const std::bad_alloc&) {
112 return Result<EliminationResult>::failure(
113 CasErrc::ResourceLimit,
114 "exact elimination allocation failed", operation);
115 } catch (const std::exception& error) {
116 return Result<EliminationResult>::failure(
117 CasErrc::InternalInvariant, error.what(), operation);
118 }

146Result<ExactMatrixData> inverse_exact(
147 const ExactMatrixData& input,
148 ComputationContext& context,
149 const std::string& operation) {
150 auto valid = validate_matrix(input, operation);
151 if (!valid) return Result<ExactMatrixData>::failure(valid.error());
152 if (input.rows != input.cols) {
153 return Result<ExactMatrixData>::failure(
154 CasErrc::InvalidArgument,
155 "inverse requires a square matrix", operation);
156 }
157 const std::size_t n = input.rows;
158 ExactMatrixData augmented{n, 2 * n, {}};
159 augmented.entries.reserve(n * 2 * n);
160 for (std::size_t row = 0; row < n; ++row) {
161 for (std::size_t column = 0; column < n; ++column) {
162 augmented.entries.push_back(input.at(row, column));
163 }
164 for (std::size_t column = 0; column < n; ++column) {
165 augmented.entries.push_back(
166 SymbolicExpr::number(row == column ? 1 : 0));
167 }
168 }
169 auto eliminated = eliminate_exact(
170 std::move(augmented), n,
171 EliminationForm::ReducedRowEchelon, context, operation);
172 if (!eliminated) return Result<ExactMatrixData>::failure(eliminated.error());
173 if (eliminated.value().rank != n) {
174 return Result<ExactMatrixData>::failure(
175 CasErrc::DomainError,
176 "matrix is singular and has no inverse", operation);
177 }
178 ExactMatrixData inverse{n, n, {}};
179 inverse.entries.reserve(n * n);
180 for (std::size_t row = 0; row < n; ++row) {
181 for (std::size_t column = 0; column < n; ++column) {
182 inverse.entries.push_back(
183 eliminated.value().matrix.at(row, n + column));
184 }
185 }
186 return Result<ExactMatrixData>::success(std::move(inverse));
187}
