15Result<void> validate_matrix(
16 const ExactMatrixData& matrix,
17 const std::string& operation) {
18 if (matrix.rows == 0 || matrix.cols == 0) {
19 return Result<void>::failure(
20 CasErrc::InvalidArgument,
21 "exact matrix dimensions must be non-zero", operation);
22 }
23 if (matrix.rows > std::numeric_limits<std::size_t>::max() / matrix.cols ||
24 matrix.entries.size() != matrix.rows * matrix.cols) {
25 return Result<void>::failure(
26 CasErrc::InvalidArgument,
27 "exact matrix storage does not match its dimensions", operation);
28 }
29 for (const auto& entry : matrix.entries) {
30 if (!entry || !node(entry)) {
31 return Result<void>::failure(
32 CasErrc::InvalidArgument,
33 "exact matrix entries cannot be null", operation);
34 }
35 }
36 return Result<void>::success();
37}

55std::optional<Rational> exact_rational(const ExprPtr& expression) {
56 if (!expression || !node(expression)) return std::nullopt;
57 auto number = std::dynamic_pointer_cast<const NumberNode>(node(expression));
58 if (!number) return std::nullopt;
59 if (std::holds_alternative<BigInt>(number->value())) {
60 return Rational(std::get<BigInt>(number->value()));
61 }
62 if (std::holds_alternative<Rational>(number->value())) {
63 return std::get<Rational>(number->value());
64 }
65 return std::nullopt;
66}

102Result<std::optional<std::size_t>> choose_pivot(
103 const ExactMatrixData& matrix,
104 std::size_t first_row,
105 std::size_t column,
106 ComputationContext& context,
107 const std::string& operation) {
108 bool saw_unknown = false;
109 for (std::size_t row = first_row; row < matrix.rows; ++row) {
110 auto proof = classify_exact_zero(matrix.at(row, column), context, operation);
111 if (!proof) {
112 return Result<std::optional<std::size_t>>::failure(proof.error());
113 }
114 if (proof.value() == ZeroProof::NonZero) {
115 return Result<std::optional<std::size_t>>::success(row);
116 }
117 saw_unknown = saw_unknown || proof.value() == ZeroProof::Unknown;
118 }
119 if (saw_unknown) {
120 return Result<std::optional<std::size_t>>::failure(
121 CasErrc::Inconclusive,
122 "exact pivot is not provably zero or non-zero", operation);
123 }
124 return Result<std::optional<std::size_t>>::success(std::nullopt);
125}

127Result<Rational> rational_determinant_value_impl(
128 std::size_t dimension,
129 std::vector<Rational> values,
130 ComputationContext& context,
131 const std::string& operation) {
132 if (dimension == 0 ||
133 dimension > std::numeric_limits<std::size_t>::max() / dimension ||
134 values.size() != dimension * dimension) {
135 return Result<Rational>::failure(
136 CasErrc::InvalidArgument,
137 "rational determinant storage does not match its dimension",
138 operation);
139 }
140 Rational sign(1);
141 Rational previous(1);
142 if (dimension == 1) return Result<Rational>::success(values[0]);
143 for (std::size_t pivot_column = 0;
144 pivot_column + 1 < dimension; ++pivot_column) {
145 auto budget = context.consume_steps(1, operation);
146 if (!budget) return Result<Rational>::failure(budget.error());
147 std::size_t pivot_row = pivot_column;
148 while (pivot_row < dimension &&
149 values[pivot_row * dimension + pivot_column] == Rational(0)) {
150 ++pivot_row;
151 }
152 if (pivot_row == dimension) {
153 return Result<Rational>::success(Rational(0));
154 }
155 if (pivot_row != pivot_column) {
156 for (std::size_t column = 0; column < dimension; ++column) {
157 std::swap(values[pivot_row * dimension + column],
158 values[pivot_column * dimension + column]);
159 }
160 sign = -sign;
161 }
162 const Rational pivot =
163 values[pivot_column * dimension + pivot_column];
164 for (std::size_t row = pivot_column + 1; row < dimension; ++row) {
165 for (std::size_t column = pivot_column + 1;
166 column < dimension; ++column) {
167 values[row * dimension + column] =
168 (values[row * dimension + column] * pivot -
169 values[row * dimension + pivot_column] *
170 values[pivot_column * dimension + column]) /
171 previous;
172 }
173 values[row * dimension + pivot_column] = Rational(0);
174 }
175 previous = pivot;
176 }
177 return Result<Rational>::success(
178 sign * values[(dimension - 1) * dimension + (dimension - 1)]);
179}

181Result<ExprPtr> rational_determinant(
182 const ExactMatrixData& input,
183 ComputationContext& context,
184 const std::string& operation) {
185 std::vector<Rational> values;
186 values.reserve(input.entries.size());
187 for (const auto& entry : input.entries) {
188 auto value = exact_rational(entry);
189 if (!value) {
190 return Result<ExprPtr>::failure(
191 CasErrc::Inconclusive,
192 "matrix is not entirely rational",
193 operation);
194 }
195 values.push_back(*value);
196 }
197 auto determinant = rational_determinant_value_impl(
198 input.rows, std::move(values), context, operation);
199 if (!determinant) return Result<ExprPtr>::failure(determinant.error());
200 return Result<ExprPtr>::success(
201 SymbolicExpr::number(determinant.value()));
202}

204Result<ExprPtr> determinant_cofactor(
205 const ExactMatrixData& matrix,
206 ComputationContext& context,
207 const std::string& operation) {
208 auto budget = context.consume_steps(1, operation);
209 if (!budget) return Result<ExprPtr>::failure(budget.error());
210 if (matrix.rows == 1) {
211 return Result<ExprPtr>::success(matrix.entries[0]);
212 }
213 auto sum = SymbolicExpr::number(0);
214 for (std::size_t column = 0; column < matrix.cols; ++column) {
215 if (auto coefficient = exact_rational(matrix.at(0, column));
216 coefficient && *coefficient == Rational(0)) {
217 continue;
218 }
219 ExactMatrixData minor{
220 matrix.rows - 1, matrix.cols - 1, {}};
221 minor.entries.reserve(minor.rows * minor.cols);
222 for (std::size_t row = 1; row < matrix.rows; ++row) {
223 for (std::size_t source = 0; source < matrix.cols; ++source) {
224 if (source != column) {
225 minor.entries.push_back(matrix.at(row, source));
226 }
227 }
228 }
229 auto minor_determinant = determinant_cofactor(
230 minor, context, operation);
231 if (!minor_determinant) {
232 return Result<ExprPtr>::failure(minor_determinant.error());
233 }
234 auto term = exact_multiply(
235 matrix.at(0, column), minor_determinant.value());
236 if (column % 2 != 0) term = exact_negate(term);
237 sum = exact_add(sum, term);
238 }
239 return Result<ExprPtr>::success(simplify_expr(sum));
240}

242Result<ExactLinearSolution> rational_linear_solve(
243 const ExactMatrixData& augmented,
244 std::size_t coefficient_columns,
245 ComputationContext& context,
246 const std::string& operation) {
247 auto budget = context.consume_steps(
248 augmented.rows * augmented.cols, operation);
249 if (!budget) {
250 return Result<ExactLinearSolution>::failure(budget.error());
251 }
252 std::vector<Rational> matrix;
253 matrix.reserve(augmented.entries.size());
254 for (const auto& entry : augmented.entries) {
255 auto value = exact_rational(entry);
256 if (!value) {
257 return Result<ExactLinearSolution>::failure(
258 CasErrc::Inconclusive,
259 "rational specialization received a symbolic entry",
260 operation);
261 }
262 matrix.push_back(*value);
263 }
264 const std::size_t columns = augmented.cols;
265 std::vector<std::size_t> pivots;
266 std::size_t pivot_row = 0;
267 for (std::size_t column = 0;
268 column < coefficient_columns && pivot_row < augmented.rows;
269 ++column) {
270 std::size_t selected = pivot_row;
271 while (selected < augmented.rows &&
272 matrix[selected * columns + column] == Rational(0)) {
273 ++selected;
274 }
275 if (selected == augmented.rows) continue;
276 if (selected != pivot_row) {
277 for (std::size_t target = 0; target < columns; ++target) {
278 std::swap(matrix[selected * columns + target],
279 matrix[pivot_row * columns + target]);
280 }
281 }
282 const Rational pivot = matrix[pivot_row * columns + column];
283 for (std::size_t target = 0; target < columns; ++target) {
284 matrix[pivot_row * columns + target] =
285 matrix[pivot_row * columns + target] / pivot;
286 }
287 for (std::size_t row = 0; row < augmented.rows; ++row) {
288 if (row == pivot_row) continue;
289 const Rational factor = matrix[row * columns + column];
290 if (factor == Rational(0)) continue;
291 for (std::size_t target = 0; target < columns; ++target) {
292 matrix[row * columns + target] =
293 matrix[row * columns + target] -
294 factor * matrix[pivot_row * columns + target];
295 }
296 }
297 pivots.push_back(column);
298 ++pivot_row;
299 }
300 for (std::size_t row = 0; row < augmented.rows; ++row) {
301 bool zero = true;
302 for (std::size_t column = 0;
303 column < coefficient_columns; ++column) {
304 zero = zero &&
305 matrix[row * columns + column] == Rational(0);
306 }
307 if (zero &&
308 matrix[row * columns + coefficient_columns] != Rational(0)) {
309 return Result<ExactLinearSolution>::success(
310 InconsistentLinearSolution{});
311 }
312 }
313 std::vector<std::optional<std::size_t>> pivot_rows(
314 coefficient_columns);
315 for (std::size_t row = 0; row < pivots.size(); ++row) {
316 pivot_rows[pivots[row]] = row;
317 }
318 std::vector<std::size_t> free_columns;
319 for (std::size_t column = 0; column < coefficient_columns; ++column) {
320 if (!pivot_rows[column]) free_columns.push_back(column);
321 }
322 std::vector<ExprPtr> particular(
323 coefficient_columns, SymbolicExpr::number(0));
324 for (std::size_t column = 0; column < coefficient_columns; ++column) {
325 if (pivot_rows[column]) {
326 particular[column] = SymbolicExpr::number(
327 matrix[*pivot_rows[column] * columns +
328 coefficient_columns]);
329 }
330 }
331 if (free_columns.empty()) {
332 return Result<ExactLinearSolution>::success(
333 UniqueLinearSolution{std::move(particular)});
334 }
335 std::vector<std::vector<ExprPtr>> basis;
336 for (const auto free_column : free_columns) {
337 std::vector<ExprPtr> vector(
338 coefficient_columns, SymbolicExpr::number(0));
339 vector[free_column] = SymbolicExpr::number(1);
340 for (std::size_t column = 0;
341 column < coefficient_columns; ++column) {
342 if (pivot_rows[column]) {
343 vector[column] = SymbolicExpr::number(
344 -matrix[*pivot_rows[column] * columns + free_column]);
345 }
346 }
347 basis.push_back(std::move(vector));
348 }
349 return Result<ExactLinearSolution>::success(
350 ParametricLinearSolution{
351 std::move(particular), std::move(basis),
352 std::move(free_columns)});
353}

376Result<ZeroProof> classify_exact_zero(
377 const ExprPtr& expression,
378 ComputationContext& context,
379 const std::string& operation) {
380 auto access = context.consume_steps(1, operation);
381 if (!access) return Result<ZeroProof>::failure(access.error());
382 if (!expression || !node(expression)) {
383 return Result<ZeroProof>::failure(
384 CasErrc::InvalidArgument,
385 "zero classification requires an expression", operation);
386 }
387 try {
388 if (contains_approximate_number(expression)) {
389 return Result<ZeroProof>::success(ZeroProof::Unknown);
390 }
391 auto simplified = simplify_expr(expression);
392 if (auto value = exact_rational(simplified)) {
393 return Result<ZeroProof>::success(
394 *value == Rational(0) ? ZeroProof::Zero : ZeroProof::NonZero);
395 }
396 if (auto power =
397 std::dynamic_pointer_cast<const PowerNode>(node(simplified))) {
398 auto base = exact_rational(
399 make_expression_ptr(power->base()));
400 auto exponent = exact_rational(
401 make_expression_ptr(power->exponent()));
402 if (base && exponent) {
403 if (*base > Rational(0)) {
404 return Result<ZeroProof>::success(ZeroProof::NonZero);
405 }
406 if (*base == Rational(0) && *exponent > Rational(0)) {
407 return Result<ZeroProof>::success(ZeroProof::Zero);
408 }
409 }
410 }
411 if (auto function =
412 std::dynamic_pointer_cast<const FunctionNode>(
413 node(simplified));
414 function && function->type() == FunctionNode::FuncType::Sqrt &&
415 function->arguments().size() == 1) {
416 auto argument = exact_rational(
417 make_expression_ptr(function->arguments()[0]));
418 if (argument && *argument >= Rational(0)) {
419 return Result<ZeroProof>::success(
420 *argument == Rational(0)
421 ? ZeroProof::Zero : ZeroProof::NonZero);
422 }
423 }
424 if (auto root =
425 std::dynamic_pointer_cast<const RootOfNode>(
426 node(simplified));
427 root && !root->exact_id().polynomial.coeffs.empty() &&
428 root->exact_id().polynomial.coeffs[0] != Rational(0)) {
429 return Result<ZeroProof>::success(ZeroProof::NonZero);
430 }
431 if (!context.assumptions()) {
432 return Result<ZeroProof>::success(ZeroProof::Unknown);
433 }
434 auto assumption = context.assumptions()->is_nonzero_checked(*simplified);
435 if (!assumption) return Result<ZeroProof>::failure(assumption.error());
436 if (assumption.value() == Tribool::True) {
437 return Result<ZeroProof>::success(ZeroProof::NonZero);
438 }
439 if (assumption.value() == Tribool::False) {
440 return Result<ZeroProof>::success(ZeroProof::Zero);
441 }
442 return Result<ZeroProof>::success(ZeroProof::Unknown);
443 } catch (const std::bad_alloc&) {
444 return Result<ZeroProof>::failure(
445 CasErrc::ResourceLimit,
446 "zero classification allocation failed", operation);
447 } catch (const std::exception& error) {
448 return Result<ZeroProof>::failure(
449 CasErrc::InternalInvariant, error.what(), operation);
450 }
451}

453Result<EliminationResult> eliminate_exact(
454 ExactMatrixData input,
455 std::size_t coefficient_columns,
456 EliminationForm form,
457 ComputationContext& context,
458 const std::string& operation) {
459 auto valid = validate_matrix(input, operation);
460 if (!valid) return Result<EliminationResult>::failure(valid.error());
461 if (coefficient_columns == 0 || coefficient_columns > input.cols) {
462 return Result<EliminationResult>::failure(
463 CasErrc::InvalidArgument,
464 "coefficient column count is outside matrix shape", operation);
465 }
466 try {
467 EliminationResult result;
468 result.matrix = std::move(input);
469 std::size_t pivot_row = 0;
470 for (std::size_t column = 0;
471 column < coefficient_columns && pivot_row < result.matrix.rows;
472 ++column) {
473 auto pivot = choose_pivot(
474 result.matrix, pivot_row, column, context, operation);
475 if (!pivot) return Result<EliminationResult>::failure(pivot.error());
476 if (!pivot.value()) continue;
477 if (*pivot.value() != pivot_row) {
478 swap_rows(result.matrix, *pivot.value(), pivot_row);
479 ++result.row_swaps;
480 }
481 auto pivot_value = simplify_expr(result.matrix.at(pivot_row, column));
482 result.pivot_columns.push_back(column);
483 result.pivot_values.push_back(pivot_value);
484
485 for (std::size_t row = pivot_row + 1; row < result.matrix.rows; ++row) {
486 auto entry_proof = classify_exact_zero(
487 result.matrix.at(row, column), context, operation);
488 if (!entry_proof) {
489 return Result<EliminationResult>::failure(entry_proof.error());
490 }
491 if (entry_proof.value() == ZeroProof::Zero) continue;
492 auto factor = exact_divide(
493 result.matrix.at(row, column), pivot_value);
494 for (std::size_t target = 0; target < result.matrix.cols; ++target) {
495 result.matrix.at(row, target) = exact_subtract(
496 result.matrix.at(row, target),
497 exact_multiply(factor,
498 result.matrix.at(pivot_row, target)));
499 }
500 }
501 ++pivot_row;
502 }
503 result.rank = result.pivot_columns.size();
504
505 if (form == EliminationForm::ReducedRowEchelon) {
506 for (std::size_t pivot_index = result.rank;
507 pivot_index-- > 0;) {
508 const std::size_t row = pivot_index;
509 const std::size_t column = result.pivot_columns[pivot_index];
510 auto pivot_value = simplify_expr(result.matrix.at(row, column));
511 for (std::size_t target = 0; target < result.matrix.cols; ++target) {
512 result.matrix.at(row, target) = exact_divide(
513 result.matrix.at(row, target), pivot_value);
514 }
515 for (std::size_t upper = 0; upper < row; ++upper) {
516 auto proof = classify_exact_zero(
517 result.matrix.at(upper, column), context, operation);
518 if (!proof) {
519 return Result<EliminationResult>::failure(proof.error());
520 }
521 if (proof.value() == ZeroProof::Zero) continue;
522 auto factor = result.matrix.at(upper, column);
523 for (std::size_t target = 0;
524 target < result.matrix.cols; ++target) {
525 result.matrix.at(upper, target) = exact_subtract(
526 result.matrix.at(upper, target),
527 exact_multiply(factor,
528 result.matrix.at(row, target)));
529 }
530 }
531 }
532 }
533 return Result<EliminationResult>::success(std::move(result));
534 } catch (const std::bad_alloc&) {
535 return Result<EliminationResult>::failure(
536 CasErrc::ResourceLimit,
537 "exact elimination allocation failed", operation);
538 } catch (const std::exception& error) {
539 return Result<EliminationResult>::failure(
540 CasErrc::InternalInvariant, error.what(), operation);
541 }
542}

544Result<ExprPtr> determinant_exact(
545 const ExactMatrixData& input,
546 ComputationContext& context,
547 const std::string& operation) {
548 auto valid = validate_matrix(input, operation);
549 if (!valid) return Result<ExprPtr>::failure(valid.error());
550 if (input.rows != input.cols) {
551 return Result<ExprPtr>::failure(
552 CasErrc::InvalidArgument,
553 "determinant requires a square matrix", operation);
554 }
555 for (const auto& entry : input.entries) {
556 if (contains_approximate_number(entry)) {
557 return Result<ExprPtr>::failure(
558 CasErrc::Inconclusive,
559 "exact determinant rejects approximate entries", operation);
560 }
561 }
562 bool all_rational = true;
563 for (const auto& entry : input.entries) {
564 all_rational = all_rational && exact_rational(entry).has_value();
565 }
566 if (all_rational) return rational_determinant(input, context, operation);
567
568 try {
569 ExactMatrixData matrix = input;
570 const std::size_t n = matrix.rows;
571 if (n == 1) return Result<ExprPtr>::success(matrix.entries[0]);
572 auto previous = SymbolicExpr::number(1);
573 std::size_t swaps = 0;
574 for (std::size_t column = 0; column + 1 < n; ++column) {
575 auto pivot = choose_pivot(matrix, column, column, context, operation);
576 if (!pivot) {
577 if (pivot.error().code == CasErrc::Inconclusive) {
578 return determinant_cofactor(input, context, operation);
579 }
580 return Result<ExprPtr>::failure(pivot.error());
581 }
582 if (!pivot.value()) {
583 return Result<ExprPtr>::success(SymbolicExpr::number(0));
584 }
585 if (*pivot.value() != column) {
586 swap_rows(matrix, *pivot.value(), column);
587 ++swaps;
588 }
589 auto pivot_value = matrix.at(column, column);
590 for (std::size_t row = column + 1; row < n; ++row) {
591 for (std::size_t target = column + 1; target < n; ++target) {
592 auto numerator = exact_subtract(
593 exact_multiply(matrix.at(row, target), pivot_value),
594 exact_multiply(matrix.at(row, column),
595 matrix.at(column, target)));
596 matrix.at(row, target) = exact_divide(numerator, previous);
597 }
598 matrix.at(row, column) = SymbolicExpr::number(0);
599 }
600 previous = pivot_value;
601 }
602 auto determinant = matrix.at(n - 1, n - 1);
603 if (swaps % 2 != 0) determinant = exact_negate(determinant);
604 return Result<ExprPtr>::success(simplify_expr(determinant));
605 } catch (const std::bad_alloc&) {
606 return Result<ExprPtr>::failure(
607 CasErrc::ResourceLimit,
608 "exact determinant allocation failed", operation);
609 } catch (const std::exception& error) {
610 return Result<ExprPtr>::failure(
611 CasErrc::InternalInvariant, error.what(), operation);
612 }
613}

640Result<ExactMatrixData> inverse_exact(
641 const ExactMatrixData& input,
642 ComputationContext& context,
643 const std::string& operation) {
644 auto valid = validate_matrix(input, operation);
645 if (!valid) return Result<ExactMatrixData>::failure(valid.error());
646 if (input.rows != input.cols) {
647 return Result<ExactMatrixData>::failure(
648 CasErrc::InvalidArgument,
649 "inverse requires a square matrix", operation);
650 }
651 const std::size_t n = input.rows;
652 ExactMatrixData augmented{n, 2 * n, {}};
653 augmented.entries.reserve(n * 2 * n);
654 for (std::size_t row = 0; row < n; ++row) {
655 for (std::size_t column = 0; column < n; ++column) {
656 augmented.entries.push_back(input.at(row, column));
657 }
658 for (std::size_t column = 0; column < n; ++column) {
659 augmented.entries.push_back(
660 SymbolicExpr::number(row == column ? 1 : 0));
661 }
662 }
663 auto eliminated = eliminate_exact(
664 std::move(augmented), n,
665 EliminationForm::ReducedRowEchelon, context, operation);
666 if (!eliminated) return Result<ExactMatrixData>::failure(eliminated.error());
667 if (eliminated.value().rank != n) {
668 return Result<ExactMatrixData>::failure(
669 CasErrc::DomainError,
670 "matrix is singular and has no inverse", operation);
671 }
672 ExactMatrixData inverse{n, n, {}};
673 inverse.entries.reserve(n * n);
674 for (std::size_t row = 0; row < n; ++row) {
675 for (std::size_t column = 0; column < n; ++column) {
676 inverse.entries.push_back(
677 eliminated.value().matrix.at(row, n + column));
678 }
679 }
680 return Result<ExactMatrixData>::success(std::move(inverse));
681}

683Result<std::vector<Rational>> solve_rational_unique(
684 std::size_t rows,
685 std::size_t coefficient_columns,
686 std::vector<Rational> matrix,
687 ComputationContext& context,
688 const std::string& operation) {
689 const std::size_t columns = coefficient_columns + 1;
690 if (rows == 0 || coefficient_columns == 0 ||
691 matrix.size() != rows * columns) {
692 return Result<std::vector<Rational>>::failure(
693 CasErrc::InvalidArgument,
694 "rational linear-system storage is invalid", operation);
695 }
696 auto budget = context.consume_steps(rows * columns, operation);
697 if (!budget) return Result<std::vector<Rational>>::failure(budget.error());
698 std::vector<std::optional<std::size_t>> pivot_rows(
699 coefficient_columns);
700 std::size_t pivot_row = 0;
701 for (std::size_t column = 0;
702 column < coefficient_columns && pivot_row < rows; ++column) {
703 std::size_t selected = pivot_row;
704 while (selected < rows &&
705 matrix[selected * columns + column] == Rational(0)) {
706 ++selected;
707 }
708 if (selected == rows) continue;
709 if (selected != pivot_row) {
710 for (std::size_t target = 0; target < columns; ++target) {
711 std::swap(matrix[selected * columns + target],
712 matrix[pivot_row * columns + target]);
713 }
714 }
715 const Rational pivot = matrix[pivot_row * columns + column];
716 for (std::size_t target = column; target < columns; ++target) {
717 matrix[pivot_row * columns + target] =
718 matrix[pivot_row * columns + target] / pivot;
719 }
720 for (std::size_t row = 0; row < rows; ++row) {
721 if (row == pivot_row) continue;
722 const Rational factor = matrix[row * columns + column];
723 if (factor == Rational(0)) continue;
724 for (std::size_t target = column; target < columns; ++target) {
725 matrix[row * columns + target] =
726 matrix[row * columns + target] -
727 factor * matrix[pivot_row * columns + target];
728 }
729 }
730 pivot_rows[column] = pivot_row++;
731 }
732 for (std::size_t row = 0; row < rows; ++row) {
733 bool zero = true;
734 for (std::size_t column = 0;
735 column < coefficient_columns; ++column) {
736 zero = zero &&
737 matrix[row * columns + column] == Rational(0);
738 }
739 if (zero && matrix[row * columns + coefficient_columns] != Rational(0)) {
740 return Result<std::vector<Rational>>::failure(
741 CasErrc::DomainError,
742 "rational linear system is inconsistent", operation);
743 }
744 }
745 std::vector<Rational> solution(
746 coefficient_columns, Rational(0));
747 for (std::size_t column = 0; column < coefficient_columns; ++column) {
748 if (!pivot_rows[column]) {
749 return Result<std::vector<Rational>>::failure(
750 CasErrc::Inconclusive,
751 "rational linear system does not have a unique solution",
752 operation);
753 }
754 solution[column] =
755 matrix[*pivot_rows[column] * columns + coefficient_columns];
756 }
757 return Result<std::vector<Rational>>::success(std::move(solution));
758}
759Result<ExactLinearSolution> solve_linear_exact(
760 ExactMatrixData augmented,
761 std::size_t coefficient_columns,
762 ComputationContext& context,
763 const std::string& operation) {
764 auto valid = validate_matrix(augmented, operation);
765
766 if (!valid) return Result<ExactLinearSolution>::failure(valid.error());
767 if (augmented.cols != coefficient_columns + 1) {
768 return Result<ExactLinearSolution>::failure(
769 CasErrc::InvalidArgument,
770 "linear solve requires one augmented right-hand side", operation);
771 }
772 bool all_rational = true;
773 for (const auto& entry : augmented.entries) {
774 all_rational = all_rational && exact_rational(entry).has_value();
775 }
776 if (all_rational) {
777 return rational_linear_solve(
778 augmented, coefficient_columns, context, operation);
779 }
780 auto eliminated = eliminate_exact(
781 std::move(augmented), coefficient_columns,
782 EliminationForm::ReducedRowEchelon, context, operation);
783 if (!eliminated) return Result<ExactLinearSolution>::failure(eliminated.error());
784 auto& rref = eliminated.value().matrix;
785
786 for (std::size_t row = 0; row < rref.rows; ++row) {
787 bool coefficients_zero = true;
788 for (std::size_t column = 0; column < coefficient_columns; ++column) {
789 auto proof = classify_exact_zero(rref.at(row, column), context, operation);
790 if (!proof) return Result<ExactLinearSolution>::failure(proof.error());
791 if (proof.value() == ZeroProof::Unknown) {
792 return Result<ExactLinearSolution>::failure(
793 CasErrc::Inconclusive,
794 "linear-system consistency is not provable", operation);
795 }
796 coefficients_zero = coefficients_zero && proof.value() == ZeroProof::Zero;
797 }
798 if (!coefficients_zero) continue;
799 auto rhs = classify_exact_zero(
800 rref.at(row, coefficient_columns), context, operation);
801 if (!rhs) return Result<ExactLinearSolution>::failure(rhs.error());
802 if (rhs.value() == ZeroProof::Unknown) {
803 return Result<ExactLinearSolution>::failure(
804 CasErrc::Inconclusive,
805 "linear-system right-hand side is not provably zero", operation);
806 }
807 if (rhs.value() == ZeroProof::NonZero) {
808 return Result<ExactLinearSolution>::success(
809 InconsistentLinearSolution{});
810 }
811 }
812
813 std::vector<std::optional<std::size_t>> pivot_row(coefficient_columns);
814 for (std::size_t index = 0;
815 index < eliminated.value().pivot_columns.size(); ++index) {
816 const auto column = eliminated.value().pivot_columns[index];
817 if (column < coefficient_columns) pivot_row[column] = index;
818 }
819 std::vector<std::size_t> free_columns;
820 for (std::size_t column = 0; column < coefficient_columns; ++column) {
821 if (!pivot_row[column]) free_columns.push_back(column);
822 }
823 std::vector<ExprPtr> particular(
824 coefficient_columns, SymbolicExpr::number(0));
825 for (std::size_t column = 0; column < coefficient_columns; ++column) {
826 if (pivot_row[column]) {
827 particular[column] = rref.at(*pivot_row[column], coefficient_columns);
828 }
829 }
830 if (free_columns.empty()) {
831 return Result<ExactLinearSolution>::success(
832 UniqueLinearSolution{std::move(particular)});
833 }
834
835 std::vector<std::vector<ExprPtr>> basis;
836 basis.reserve(free_columns.size());
837 for (const auto free_column : free_columns) {
838 std::vector<ExprPtr> vector(
839 coefficient_columns, SymbolicExpr::number(0));
840 vector[free_column] = SymbolicExpr::number(1);
841 for (std::size_t column = 0; column < coefficient_columns; ++column) {
842 if (pivot_row[column]) {
843 vector[column] = exact_negate(
844 rref.at(*pivot_row[column], free_column));
845 }
846 }
847 basis.push_back(std::move(vector));
848 }
849 return Result<ExactLinearSolution>::success(ParametricLinearSolution{
850 std::move(particular), std::move(basis), std::move(free_columns)});
851}

853Result<std::vector<std::vector<ExprPtr>>> nullspace_exact(
854 ExactMatrixData coefficients,
855 ComputationContext& context,
856 const std::string& operation) {
857 auto valid = validate_matrix(coefficients, operation);
858 if (!valid) {
859 return Result<std::vector<std::vector<ExprPtr>>>::failure(valid.error());
860 }
861 const std::size_t columns = coefficients.cols;
862 auto eliminated = eliminate_exact(
863 std::move(coefficients), columns,
864 EliminationForm::ReducedRowEchelon, context, operation);
865 if (!eliminated) {
866 return Result<std::vector<std::vector<ExprPtr>>>::failure(
867 eliminated.error());
868 }
869 std::set<std::size_t> pivots(
870 eliminated.value().pivot_columns.begin(),
871 eliminated.value().pivot_columns.end());
872 std::vector<std::vector<ExprPtr>> basis;
873 for (std::size_t free_column = 0; free_column < columns; ++free_column) {
874 if (pivots.count(free_column) != 0) continue;
875 std::vector<ExprPtr> vector(columns, SymbolicExpr::number(0));
876 vector[free_column] = SymbolicExpr::number(1);
877 for (std::size_t row = 0;
878 row < eliminated.value().pivot_columns.size(); ++row) {
879 vector[eliminated.value().pivot_columns[row]] = exact_negate(
880 eliminated.value().matrix.at(row, free_column));
881 }
882 basis.push_back(std::move(vector));
883 }
884 return Result<std::vector<std::vector<ExprPtr>>>::success(
885 std::move(basis));
886}
