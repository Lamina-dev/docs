22static bool vector_calculus_evaluate_hessian_at_point(
23 const std::shared_ptr<SymbolicExpr>& H,
24 const std::vector<std::string>&,
25 const std::map<std::string, std::shared_ptr<SymbolicExpr>>& pt,
26 size_t n,
27 std::vector<double>& numeric_H)
28{
29 auto mat_node = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(H));
30 if (!mat_node || mat_node->rows() != n || mat_node->cols() != n) {
31 return false;
32 }
33
34 numeric_H.resize(n * n);
35 for (size_t i = 0; i < n; ++i) {
36 for (size_t j = 0; j < n; ++j) {
37 auto elem_node = mat_node->get(i, j);
38 if (!elem_node) {
39 numeric_H[i * n + j] = 0.0;
40 continue;
41 }
42 auto elem = lamina::detail::make_expression_ptr(elem_node);
43 /// 代入临界点坐标
44 for (const auto& [var_name, val] : pt) {
45 elem = elem->substitute(var_name, val);
46 if (!elem) return false;
47 }
48 elem = elem->simplify();
49 if (!elem) return false;
50
51 if (!vector_calculus_checked_finite_numeric(
52 elem, numeric_H[i * n + j])) {
53 return false;
54 }
55 }
56 }
57 return true;
58}

60/**
61 * @internal
62 * @brief 使用特征值分析对临界点进行分类.
63 *
64 * 对 nxn 实对称矩阵,直接通过数值分析特征值符号判断正定性.
65 * 对于小矩阵(n <= 3),使用解析公式;对于大矩阵,使用特征多项式求解.
66 *
67 * @param[in] numeric_H 数值化的海森矩阵(行优先,nxn)
68 * @param[in] n 矩阵维度
69 * @return 分类字符串: "minimum", "maximum", "saddle", "degenerate"
70 */
71static CriticalPointClassification vector_calculus_classify_critical_point(
72 const std::vector<double>& numeric_H, size_t n)
73{
74 const double tol = 1e-10;
75
76 /// 1x1 情况:直接判断
77 if (n == 1) {
78 double val = numeric_H[0];
79 if (std::abs(val) < tol) return CriticalPointClassification::Degenerate;
80 if (val > 0) return CriticalPointClassification::LocalMinimum;
81 return CriticalPointClassification::LocalMaximum;
82 }
83
84 /// 2x2 情况:使用行列式和迹直接判断
85 if (n == 2) {
86 double a = numeric_H[0], b = numeric_H[1];
87 double c = numeric_H[2], d = numeric_H[3];
88 double det = a * d - b * c;
89 double trace = a + d;
90
91 if (std::abs(det) < tol) return CriticalPointClassification::Degenerate;
92 if (det > 0 && trace > 0) return CriticalPointClassification::LocalMinimum;
93 if (det > 0 && trace < 0) return CriticalPointClassification::LocalMaximum;
94 return CriticalPointClassification::Saddle;
95 }
96
97 /// 一般情况:构建符号矩阵并求特征值
98 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> grid(n,
99 std::vector<std::shared_ptr<SymbolicExpr>>(n));
100 for (size_t i = 0; i < n; ++i) {
101 for (size_t j = 0; j < n; ++j) {
102 /// 使用 Rational 保持 Hessian 系数的精确表示.
103 double val = numeric_H[i * n + j];
104 int int_val = static_cast<int>(std::round(val));
105 if (std::abs(val - int_val) < tol) {
106 grid[i][j] = SymbolicExpr::number(int_val);
107 } else {
108 grid[i][j] = SymbolicExpr::number(val);
109 }
110 }
111 }
112 auto H_mat = SymbolicExpr::matrix(grid);
113
114 auto eigenvalue_result = matrix_eigenvalues_checked(H_mat);
115 if (!eigenvalue_result) {
116 return CriticalPointClassification::Inconclusive;
117 }
118 auto eigenvals = std::move(eigenvalue_result.value());
119
120 bool all_positive = true;
121 bool all_negative = true;
122 bool has_zero = false;
123
124 for (const auto& ev : eigenvals) {
125 if (!ev) {
126 return CriticalPointClassification::Inconclusive;
127 }
128 auto ev_simplified = ev->simplify();
129 double val = 0.0;
130 if (!vector_calculus_checked_finite_numeric(ev_simplified, val)) {
131 return CriticalPointClassification::Inconclusive;
132 }
133
134 if (std::abs(val) < tol) {
135 has_zero = true;
136 all_positive = false;
137 all_negative = false;
138 } else if (val > 0) {
139 all_negative = false;
140 } else {
141 all_positive = false;
142 }
143 }
144
145 if (has_zero) return CriticalPointClassification::Degenerate;
146 if (all_positive) return CriticalPointClassification::LocalMinimum;
147 if (all_negative) return CriticalPointClassification::LocalMaximum;
148 return CriticalPointClassification::Saddle;
149}

151ExtremaResult find_extrema_checked(
152 const std::shared_ptr<SymbolicExpr>& f,
153 const std::vector<std::string>& vars,
154 ComputationContext& context)
155{
156 const std::string operation = "find_extrema";
157 auto valid = vector_calculus_validate_expr_vars(f, vars, context, operation);
158 if (!valid) return ExtremaResult::failure(valid.error());
159 auto distinct = vector_calculus_validate_distinct_vars(vars, vars.size(), context, operation);
160 if (!distinct) return ExtremaResult::failure(distinct.error());
161 auto step = context.consume_steps(vars.size() * vars.size() + vars.size() * 8 + 8,
162 operation);
163 if (!step) return ExtremaResult::failure(step.error());
164
165 try {
166 std::vector<std::shared_ptr<SymbolicExpr>> gradient;
167 gradient.reserve(vars.size());
168 for (const auto& var : vars) {
169 auto partial = vector_calculus_differentiate_strict(f, var, operation);
170 if (!partial) return ExtremaResult::failure(partial.error());
171 gradient.push_back(std::move(lamina::detail::propagate_result(partial)));
172 }
173
174 auto extrema = find_extrema(f, vars);
175 if (extrema.empty()) {
176 return ExtremaResult::failure(
177 CasErrc::Inconclusive,
178 "extrema solver produced no verifiable critical points in the supported domain",
179 operation);
180 }
181
182 for (const auto& cp : extrema) {
183 if (!vector_calculus_point_has_vars(cp.point, vars)) {
184 return ExtremaResult::failure(
185 CasErrc::Inconclusive,
186 "extrema candidate omits one or more variables",
187 operation);
188 }
189 for (const auto& partial : gradient) {
190 if (!vector_calculus_expr_zero_after_substitution(partial, cp.point)) {
191 return ExtremaResult::failure(
192 CasErrc::Inconclusive,
193 "extrema candidate does not verify against the gradient",
194 operation);
195 }
196 }
197 if (cp.classification == CriticalPointClassification::Inconclusive) {
198 return ExtremaResult::failure(
199 CasErrc::Inconclusive,
200 "extrema candidate has no verified classification",
201 operation);
202 }
203 }
204 return ExtremaResult::success(std::move(extrema));
205 } catch (const std::bad_alloc&) {
206 return ExtremaResult::failure(CasErrc::ResourceLimit,
207 "extrema allocation failed",
208 operation);
209 } catch (const std::exception& e) {
210 return ExtremaResult::failure(CasErrc::InternalInvariant,
211 e.what(), operation);
212 }

223std::vector<CriticalPoint> find_extrema(
224 const std::shared_ptr<SymbolicExpr>& f, const std::vector<std::string>& vars)
225{
226 if (!f || vars.empty()) {
227 return {};
228 }
229
230 size_t n = vars.size();
231
232 /// 计算梯度 gradf
233 std::vector<std::shared_ptr<SymbolicExpr>> grad_eqs;
234 grad_eqs.reserve(n);
235 for (const auto& var : vars) {
236 auto partial = f->differentiate(var);
237 if (partial) {
238 partial = partial->simplify();
239 }
240 grad_eqs.push_back(partial);
241 }
242
243 /// 求解 gradf = 0 系统(使用多项式系统求解器)
244 std::vector<std::map<std::string, std::shared_ptr<SymbolicExpr>>> solutions;
245
246 std::vector<SymbolicExpr> poly_eqs;
247 poly_eqs.reserve(n);
248 for (const auto& eq : grad_eqs) {
249 if (eq) poly_eqs.push_back(*eq);
250 }
251 if (poly_eqs.size() == n) {
252 auto checked_solutions =
253 Solver::solve_polynomial_system_checked(poly_eqs, vars);
254 const auto poly_solutions = checked_solutions
255 ? std::move(checked_solutions.value())
256 : std::vector<std::map<std::string, SymbolicExpr>>{};
257 for (const auto& sol : poly_solutions) {
258 std::map<std::string, std::shared_ptr<SymbolicExpr>> pt;
259 for (const auto& [name, val] : sol) {
260 pt[name] = lamina::detail::make_expression_ptr(val);
261 }
262 solutions.push_back(pt);
263 }
264 }
265
266 /// 如果多项式求解器失败,尝试线性求解器
267 if (solutions.empty()) {
268 solutions = SymbolicExpr::solve_system(grad_eqs, vars);
269 }
270
271 if (solutions.empty()) {
272 return {};
273 }
274
275 /// 计算海森矩阵
276 auto H = hessian(f, vars);
277
278 /// 对每个临界点进行分类
279 std::vector<CriticalPoint> result;
280 result.reserve(solutions.size());
281
282 for (const auto& sol : solutions) {
283 CriticalPoint cp;
284 cp.point = sol;
285
286 /// 在临界点处求值海森矩阵
287 std::vector<double> numeric_H;
288 if (vector_calculus_evaluate_hessian_at_point(H, vars, sol, n, numeric_H)) {
289 cp.classification = vector_calculus_classify_critical_point(numeric_H, n);
290 } else {
291 cp.classification = CriticalPointClassification::Inconclusive;
292 }
293
294 result.push_back(std::move(cp));
295 }
296
297 return result;
298}

301static LagrangeResult lagrange_multipliers_strict(
302 const std::shared_ptr<SymbolicExpr>& f,
303 const std::vector<std::shared_ptr<SymbolicExpr>>& constraints,
304 const std::vector<std::string>& vars,
305 const std::string& operation)
306{
307 const size_t n = vars.size();
308 const size_t m = constraints.size();
309
310 VectorField grad_f;
311 grad_f.reserve(n);
312 for (const auto& var : vars) {
313 auto partial = vector_calculus_differentiate_strict(f, var, operation);
314 if (!partial) return LagrangeResult::failure(partial.error());
315 grad_f.push_back(std::move(lamina::detail::propagate_result(partial)));
316 }
317
318 std::vector<VectorField> grad_constraints;
319 grad_constraints.reserve(m);
320 for (const auto& constraint : constraints) {
321 VectorField grad_g;
322 grad_g.reserve(n);
323 for (const auto& var : vars) {
324 auto partial = vector_calculus_differentiate_strict(
325 constraint, var, operation);
326 if (!partial) return LagrangeResult::failure(partial.error());
327 grad_g.push_back(std::move(lamina::detail::propagate_result(partial)));
328 }
329 grad_constraints.push_back(std::move(grad_g));
330 }
331
332 std::vector<std::string> lambda_names;
333 lambda_names.reserve(m);
334 for (size_t k = 0; k < m; ++k) {
335 lambda_names.push_back("lambda_" + std::to_string(k + 1));
336 }
337
338 std::vector<std::shared_ptr<SymbolicExpr>> equations;
339 equations.reserve(n + m);
340 for (size_t i = 0; i < n; ++i) {
341 auto eq = grad_f[i];
342 for (size_t k = 0; k < m; ++k) {
343 auto lambda_var = SymbolicExpr::variable(lambda_names[k]);
344 auto term = SymbolicExpr::multiply(lambda_var, grad_constraints[k][i]);
345 eq = SymbolicExpr::add(
346 eq, SymbolicExpr::multiply(SymbolicExpr::number(-1), term));
347 }
348 auto eq_checked = vector_calculus_simplify_strict(
349 eq, operation, "Lagrange stationarity equation is outside the supported domain");
350 if (!eq_checked) return LagrangeResult::failure(eq_checked.error());
351 equations.push_back(std::move(lamina::detail::propagate_result(eq_checked)));
352 }
353 for (const auto& constraint : constraints) {
354 equations.push_back(constraint);
355 }
356
357 std::vector<std::string> all_vars;
358 all_vars.reserve(n + m);
359 all_vars.insert(all_vars.end(), vars.begin(), vars.end());
360 all_vars.insert(all_vars.end(), lambda_names.begin(), lambda_names.end());
361
362 std::vector<std::map<std::string, std::shared_ptr<SymbolicExpr>>> full_solutions;
363 std::vector<SymbolicExpr> poly_eqs;
364 poly_eqs.reserve(equations.size());
365 for (const auto& eq : equations) {
366 if (!eq || !lamina::detail::node(eq)) {
367 return LagrangeResult::failure(
368 CasErrc::Inconclusive,
369 "Lagrange equation construction failed in the supported domain",
370 operation);
371 }
372 poly_eqs.push_back(*eq);
373 }
374
375 auto checked_solutions =
376 Solver::solve_polynomial_system_checked(poly_eqs, all_vars);
377 const auto poly_solutions = checked_solutions
378 ? std::move(checked_solutions.value())
379 : std::vector<std::map<std::string, SymbolicExpr>>{};
380 for (const auto& sol : poly_solutions) {
381 std::map<std::string, std::shared_ptr<SymbolicExpr>> pt;
382 for (const auto& [name, val] : sol) {
383 pt[name] = lamina::detail::make_expression_ptr(val);
384 }
385 full_solutions.push_back(std::move(pt));
386 }
387
388 if (full_solutions.empty()) {
389 full_solutions = SymbolicExpr::solve_system(equations, all_vars);
390 }
391 if (full_solutions.empty()) {
392 return LagrangeResult::failure(
393 CasErrc::Inconclusive,
394 "Lagrange solver produced no verifiable candidates in the supported domain",
395 operation);
396 }
397
398 std::vector<std::map<std::string, std::shared_ptr<SymbolicExpr>>> verified;
399 std::set<std::string> var_set(vars.begin(), vars.end());
400 for (const auto& full_solution : full_solutions) {
401 if (!vector_calculus_point_has_vars(full_solution, all_vars)) {
402 return LagrangeResult::failure(
403 CasErrc::Inconclusive,
404 "Lagrange candidate omits one or more variables or multipliers",
405 operation);
406 }
407 for (const auto& eq : equations) {
408 if (!vector_calculus_expr_zero_after_substitution(eq, full_solution)) {
409 return LagrangeResult::failure(
410 CasErrc::Inconclusive,
411 "Lagrange candidate does not verify against the full stationarity system",
412 operation);
413 }
414 }
415
416 std::map<std::string, std::shared_ptr<SymbolicExpr>> filtered;
417 for (const auto& [name, val] : full_solution) {
418 if (var_set.count(name)) {
419 filtered[name] = val;
420 }
421 }
422 if (!vector_calculus_point_has_vars(filtered, vars)) {
423 return LagrangeResult::failure(
424 CasErrc::Inconclusive,
425 "Lagrange candidate omits one or more variables",
426 operation);
427 }
428 verified.push_back(std::move(filtered));
429 }
430
431 if (verified.empty()) {
432 return LagrangeResult::failure(
433 CasErrc::Inconclusive,
434 "Lagrange solver produced no verifiable candidates in the supported domain",
435 operation);
436 }
437 return LagrangeResult::success(std::move(verified));
438}

489std::vector<std::map<std::string, std::shared_ptr<SymbolicExpr>>> lagrange_multipliers(
490 const std::shared_ptr<SymbolicExpr>& f,
491 const std::vector<std::shared_ptr<SymbolicExpr>>& constraints,
492 const std::vector<std::string>& vars)
493{
494 if (!f || constraints.empty() || vars.empty()) {
495 return {};
496 }
497
498 size_t n = vars.size();
499 size_t m = constraints.size();
500
501 /// 计算目标函数的梯度
502 VectorField grad_f;
503 grad_f.reserve(n);
504 for (const auto& var : vars) {
505 auto partial = f->differentiate(var);
506 if (partial) partial = partial->simplify();
507 grad_f.push_back(partial);
508 }
509
510 /// 计算每个约束的梯度
511 std::vector<VectorField> grad_constraints;
512 grad_constraints.reserve(m);
513 for (const auto& g : constraints) {
514 VectorField grad_g;
515 grad_g.reserve(n);
516 for (const auto& var : vars) {
517 auto partial = g->differentiate(var);
518 if (partial) partial = partial->simplify();
519 grad_g.push_back(partial);
520 }
521 grad_constraints.push_back(std::move(grad_g));
522 }
523
524 /// 构造乘数变量名
525 std::vector<std::string> lambda_names;
526 lambda_names.reserve(m);
527 for (size_t k = 0; k < m; ++k) {
528 lambda_names.push_back("lambda_" + std::to_string(k + 1));
529 }
530
531 /// 构造方程组: partialf/partialxᵢ - sum lambdaₖ * partialgₖ/partialxᵢ = 0 (对每个 i)
532 /// 加上约束方程: gₖ = 0 (对每个 k)
533 std::vector<std::shared_ptr<SymbolicExpr>> equations;
534 equations.reserve(n + m);
535
536 for (size_t i = 0; i < n; ++i) {
537 /// partialf/partialxᵢ - lambda_1*partialg_1/partialxᵢ - lambda_2*partialg_2/partialxᵢ - ...
538 auto eq = grad_f[i];
539 for (size_t k = 0; k < m; ++k) {
540 auto lambda_var = SymbolicExpr::variable(lambda_names[k]);
541 auto term = SymbolicExpr::multiply(lambda_var, grad_constraints[k][i]);
542 eq = SymbolicExpr::add(eq, SymbolicExpr::multiply(SymbolicExpr::number(-1), term));
543 }
544 if (eq) eq = eq->simplify();
545 equations.push_back(eq);
546 }
547
548 /// 添加约束方程
549 for (const auto& g : constraints) {
550 equations.push_back(g);
551 }
552
553 /// 构造所有未知数列表: 原始变量 + 乘数
554 std::vector<std::string> all_vars;
555 all_vars.reserve(n + m);
556 for (const auto& v : vars) {
557 all_vars.push_back(v);
558 }
559 for (const auto& lam : lambda_names) {
560 all_vars.push_back(lam);
561 }
562
563 /// 使用多项式系统求解器求解增广系统
564 std::vector<std::map<std::string, std::shared_ptr<SymbolicExpr>>> solutions;
565
566 std::vector<SymbolicExpr> poly_eqs;
567 poly_eqs.reserve(equations.size());
568 for (const auto& eq : equations) {
569 if (eq) poly_eqs.push_back(*eq);
570 }
571
572 if (poly_eqs.size() == equations.size()) {
573 auto checked_solutions =
574 Solver::solve_polynomial_system_checked(poly_eqs, all_vars);
575 const auto poly_solutions = checked_solutions
576 ? std::move(checked_solutions.value())
577 : std::vector<std::map<std::string, SymbolicExpr>>{};
578 for (const auto& sol : poly_solutions) {
579 std::map<std::string, std::shared_ptr<SymbolicExpr>> pt;
580 for (const auto& [name, val] : sol) {
581 pt[name] = lamina::detail::make_expression_ptr(val);
582 }
583 solutions.push_back(pt);
584 }
585 }
586
587 /// 如果多项式求解器失败,尝试线性求解器
588 if (solutions.empty()) {
589 solutions = SymbolicExpr::solve_system(equations, all_vars);
590 }
591
592 /// 从解中提取仅原始变量(去除乘数)
593 std::vector<std::map<std::string, std::shared_ptr<SymbolicExpr>>> result;
594 std::set<std::string> var_set(vars.begin(), vars.end());
595
596 for (const auto& sol : solutions) {
597 std::map<std::string, std::shared_ptr<SymbolicExpr>> filtered;
598 for (const auto& [name, val] : sol) {
599 if (var_set.count(name)) {
600 filtered[name] = val;
601 }
602 }
603 if (!filtered.empty()) {
604 result.push_back(std::move(filtered));
605 }
606 }
607
608 return result;
609}
