20static bool vector_calculus_evaluate_hessian_at_point(
21 const std::shared_ptr<SymbolicExpr>& H,
22 const std::vector<std::string>&,
23 const std::map<std::string, std::shared_ptr<SymbolicExpr>>& pt,
24 size_t n,
25 std::vector<double>& numeric_H)
26{
27 auto mat_node = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(H));
28 if (!mat_node || mat_node->rows() != n || mat_node->cols() != n) {
29 return false;
30 }
31
32 numeric_H.resize(n * n);
33 for (size_t i = 0; i < n; ++i) {
34 for (size_t j = 0; j < n; ++j) {
35 auto elem_node = mat_node->get(i, j);
36 if (!elem_node) {
37 numeric_H[i * n + j] = 0.0;
38 continue;
39 }
40 auto elem = lamina::detail::make_expression_ptr(elem_node);
41 /// 代入临界点坐标
42 for (const auto& [var_name, val] : pt) {
43 elem = elem->substitute(var_name, val);
44 if (!elem) return false;
45 }
46 elem = elem->simplify();
47 if (!elem) return false;
48
49 if (!vector_calculus_checked_finite_numeric(
50 elem, numeric_H[i * n + j])) {
51 return false;
52 }
53 }
54 }
55 return true;
56}

58/**
59 * @internal
60 * @brief 使用特征值分析对临界点进行分类。
61 *
62 * 对 n×n 实对称矩阵，直接通过数值分析特征值符号判断正定性。
63 * 对于小矩阵（n ≤ 3），使用解析公式；对于大矩阵，使用特征多项式求解。
64 *
65 * @param[in] numeric_H 数值化的海森矩阵（行优先，n×n）
66 * @param[in] n 矩阵维度
67 * @return 分类字符串: "minimum", "maximum", "saddle", "degenerate"
68 */
69static std::string vector_calculus_classify_critical_point(
70 const std::vector<double>& numeric_H, size_t n)
71{
72 const double tol = 1e-10;
73
74 /// 1×1 情况：直接判断
75 if (n == 1) {
76 double val = numeric_H[0];
77 if (std::abs(val) < tol) return "degenerate";
78 if (val > 0) return "minimum";
79 return "maximum";
80 }
81
82 /// 2×2 情况：使用行列式和迹直接判断
83 if (n == 2) {
84 double a = numeric_H[0], b = numeric_H[1];
85 double c = numeric_H[2], d = numeric_H[3];
86 double det = a * d - b * c;
87 double trace = a + d;
88
89 if (std::abs(det) < tol) return "degenerate";
90 if (det > 0 && trace > 0) return "minimum";
91 if (det > 0 && trace < 0) return "maximum";
92 return "saddle";
93 }
94
95 /// 一般情况：构建符号矩阵并求特征值
96 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> grid(n,
97 std::vector<std::shared_ptr<SymbolicExpr>>(n));
98 for (size_t i = 0; i < n; ++i) {
99 for (size_t j = 0; j < n; ++j) {
100 /// 使用 Rational 避免浮点精度问题
101 double val = numeric_H[i * n + j];
102 int int_val = static_cast<int>(std::round(val));
103 if (std::abs(val - int_val) < tol) {
104 grid[i][j] = SymbolicExpr::number(int_val);
105 } else {
106 grid[i][j] = SymbolicExpr::number(val);
107 }
108 }
109 }
110 auto H_mat = SymbolicExpr::matrix(grid);
111
112 /// 计算特征多项式并求解特征值
113 auto cp = SymbolicExpr::charpoly(H_mat, "lambda");
114 if (!cp) return "degenerate";
115
116 auto eigenvals = SymbolicExpr::solve(cp, "lambda");
117 if (eigenvals.empty()) return "degenerate";
118
119 bool all_positive = true;
120 bool all_negative = true;
121 bool has_zero = false;
122
123 for (const auto& ev : eigenvals) {
124 if (!ev) {
125 return "degenerate";
126 }
127 auto ev_simplified = ev->simplify();
128 double val = 0.0;
129 if (!vector_calculus_checked_finite_numeric(ev_simplified, val)) {
130 return "degenerate";
131 }
132
133 if (std::abs(val) < tol) {
134 has_zero = true;
135 all_positive = false;
136 all_negative = false;
137 } else if (val > 0) {
138 all_negative = false;
139 } else {
140 all_positive = false;
141 }
142 }
143
144 if (has_zero) return "degenerate";
145 if (all_positive) return "minimum";
146 if (all_negative) return "maximum";
147 return "saddle";
148}

150ExtremaResult find_extrema_checked(
151 const std::shared_ptr<SymbolicExpr>& f,
152 const std::vector<std::string>& vars,
153 ComputationContext& context)
154{
155 const std::string operation = "find_extrema";
156 auto valid = vector_calculus_validate_expr_vars(f, vars, context, operation);
157 if (!valid) return ExtremaResult::failure(valid.error());
158 auto distinct = vector_calculus_validate_distinct_vars(vars, vars.size(), context, operation);
159 if (!distinct) return ExtremaResult::failure(distinct.error());
160 auto step = context.consume_steps(vars.size() * vars.size() + vars.size() * 8 + 8,
161 operation);
162 if (!step) return ExtremaResult::failure(step.error());
163
164 try {
165 std::vector<std::shared_ptr<SymbolicExpr>> gradient;
166 gradient.reserve(vars.size());
167 for (const auto& var : vars) {
168 auto partial = vector_calculus_differentiate_strict(f, var, operation);
169 if (!partial) return ExtremaResult::failure(partial.error());
170 gradient.push_back(std::move(partial.value()));
171 }
172
173 auto extrema = find_extrema(f, vars);
174 if (extrema.empty()) {
175 return ExtremaResult::failure(
176 CasErrc::Inconclusive,
177 "extrema solver produced no verifiable critical points in the supported domain",
178 operation);
179 }
180
181 for (const auto& cp : extrema) {
182 if (!vector_calculus_point_has_vars(cp.point, vars)) {
183 return ExtremaResult::failure(
184 CasErrc::Inconclusive,
185 "extrema candidate omits one or more variables",
186 operation);
187 }
188 for (const auto& partial : gradient) {
189 if (!vector_calculus_expr_zero_after_substitution(partial, cp.point)) {
190 return ExtremaResult::failure(
191 CasErrc::Inconclusive,
192 "extrema candidate does not verify against the gradient",
193 operation);
194 }
195 }
196 if (cp.classification.empty()) {
197 return ExtremaResult::failure(
198 CasErrc::Inconclusive,
199 "extrema candidate has no verified classification",
200 operation);
201 }
202 }
203 return ExtremaResult::success(std::move(extrema));
204 } catch (const std::bad_alloc&) {
205 return ExtremaResult::failure(CasErrc::ResourceLimit,
206 "extrema allocation failed",
207 operation);
208 } catch (const std::exception& e) {
209 return ExtremaResult::failure(CasErrc::InternalInvariant,
210 e.what(), operation);
211 }
212}

222std::vector<CriticalPoint> find_extrema(
223 const std::shared_ptr<SymbolicExpr>& f, const std::vector<std::string>& vars)
224{
225 if (!f || vars.empty()) {
226 return {};
227 }
228
229 size_t n = vars.size();
230
231 /// 计算梯度 ∇f
232 std::vector<std::shared_ptr<SymbolicExpr>> grad_eqs;
233 grad_eqs.reserve(n);
234 for (const auto& var : vars) {
235 auto partial = f->differentiate(var);
236 if (partial) {
237 partial = partial->simplify();
238 }
239 grad_eqs.push_back(partial);
240 }
241
242 /// 求解 ∇f = 0 系统（使用多项式系统求解器）
243 std::vector<std::map<std::string, std::shared_ptr<SymbolicExpr>>> solutions;
244
245 std::vector<SymbolicExpr> poly_eqs;
246 poly_eqs.reserve(n);
247 for (const auto& eq : grad_eqs) {
248 if (eq) poly_eqs.push_back(*eq);
249 }
250 if (poly_eqs.size() == n) {
251 auto poly_solutions = Solver::solve_polynomial_system(poly_eqs, vars);
252 for (const auto& sol : poly_solutions) {
253 std::map<std::string, std::shared_ptr<SymbolicExpr>> pt;
254 for (const auto& [name, val] : sol) {
255 pt[name] = lamina::detail::make_expression_ptr(val);
256 }
257 solutions.push_back(pt);
258 }
259 }
260
261 /// 如果多项式求解器失败，尝试线性求解器
262 if (solutions.empty()) {
263 solutions = SymbolicExpr::solve_system(grad_eqs, vars);
264 }
265
266 if (solutions.empty()) {
267 return {};
268 }
269
270 /// 计算海森矩阵
271 auto H = hessian(f, vars);
272 if (!H) return {};
273
274 /// 对每个临界点进行分类
275 std::vector<CriticalPoint> result;
276 result.reserve(solutions.size());
277
278 for (const auto& sol : solutions) {
279 CriticalPoint cp;
280 cp.point = sol;
281
282 /// 在临界点处求值海森矩阵
283 std::vector<double> numeric_H;
284 if (vector_calculus_evaluate_hessian_at_point(H, vars, sol, n, numeric_H)) {
285 cp.classification = vector_calculus_classify_critical_point(numeric_H, n);
286 } else {
287 cp.classification = "degenerate";
288 }
289
290 result.push_back(std::move(cp));
291 }
292
293 return result;
294}

297static LagrangeResult lagrange_multipliers_strict(
298 const std::shared_ptr<SymbolicExpr>& f,
299 const std::vector<std::shared_ptr<SymbolicExpr>>& constraints,
300 const std::vector<std::string>& vars,
301 const std::string& operation)
302{
303 const size_t n = vars.size();
304 const size_t m = constraints.size();
305
306 VectorField grad_f;
307 grad_f.reserve(n);
308 for (const auto& var : vars) {
309 auto partial = vector_calculus_differentiate_strict(f, var, operation);
310 if (!partial) return LagrangeResult::failure(partial.error());
311 grad_f.push_back(std::move(partial.value()));
312 }
313
314 std::vector<VectorField> grad_constraints;
315 grad_constraints.reserve(m);
316 for (const auto& constraint : constraints) {
317 VectorField grad_g;
318 grad_g.reserve(n);
319 for (const auto& var : vars) {
320 auto partial = vector_calculus_differentiate_strict(
321 constraint, var, operation);
322 if (!partial) return LagrangeResult::failure(partial.error());
323 grad_g.push_back(std::move(partial.value()));
324 }
325 grad_constraints.push_back(std::move(grad_g));
326 }
327
328 std::vector<std::string> lambda_names;
329 lambda_names.reserve(m);
330 for (size_t k = 0; k < m; ++k) {
331 lambda_names.push_back("lambda_" + std::to_string(k + 1));
332 }
333
334 std::vector<std::shared_ptr<SymbolicExpr>> equations;
335 equations.reserve(n + m);
336 for (size_t i = 0; i < n; ++i) {
337 auto eq = grad_f[i];
338 for (size_t k = 0; k < m; ++k) {
339 auto lambda_var = SymbolicExpr::variable(lambda_names[k]);
340 auto term = SymbolicExpr::multiply(lambda_var, grad_constraints[k][i]);
341 eq = SymbolicExpr::add(
342 eq, SymbolicExpr::multiply(SymbolicExpr::number(-1), term));
343 }
344 auto eq_checked = vector_calculus_simplify_strict(
345 eq, operation, "Lagrange stationarity equation is outside the supported domain");
346 if (!eq_checked) return LagrangeResult::failure(eq_checked.error());
347 equations.push_back(std::move(eq_checked.value()));
348 }
349 for (const auto& constraint : constraints) {
350 equations.push_back(constraint);
351 }
352
353 std::vector<std::string> all_vars;
354 all_vars.reserve(n + m);
355 all_vars.insert(all_vars.end(), vars.begin(), vars.end());
356 all_vars.insert(all_vars.end(), lambda_names.begin(), lambda_names.end());
357
358 std::vector<std::map<std::string, std::shared_ptr<SymbolicExpr>>> full_solutions;
359 std::vector<SymbolicExpr> poly_eqs;
360 poly_eqs.reserve(equations.size());
361 for (const auto& eq : equations) {
362 if (!eq || !lamina::detail::node(eq)) {
363 return LagrangeResult::failure(
364 CasErrc::Inconclusive,
365 "Lagrange equation construction failed in the supported domain",
366 operation);
367 }
368 poly_eqs.push_back(*eq);
369 }
370
371 auto poly_solutions = Solver::solve_polynomial_system(poly_eqs, all_vars);
372 for (const auto& sol : poly_solutions) {
373 std::map<std::string, std::shared_ptr<SymbolicExpr>> pt;
374 for (const auto& [name, val] : sol) {
375 pt[name] = lamina::detail::make_expression_ptr(val);
376 }
377 full_solutions.push_back(std::move(pt));
378 }
379
380 if (full_solutions.empty()) {
381 full_solutions = SymbolicExpr::solve_system(equations, all_vars);
382 }
383 if (full_solutions.empty()) {
384 return LagrangeResult::failure(
385 CasErrc::Inconclusive,
386 "Lagrange solver produced no verifiable candidates in the supported domain",
387 operation);
388 }
389
390 std::vector<std::map<std::string, std::shared_ptr<SymbolicExpr>>> verified;
391 std::set<std::string> var_set(vars.begin(), vars.end());
392 for (const auto& full_solution : full_solutions) {
393 if (!vector_calculus_point_has_vars(full_solution, all_vars)) {
394 return LagrangeResult::failure(
395 CasErrc::Inconclusive,
396 "Lagrange candidate omits one or more variables or multipliers",
397 operation);
398 }
399 for (const auto& eq : equations) {
400 if (!vector_calculus_expr_zero_after_substitution(eq, full_solution)) {
401 return LagrangeResult::failure(
402 CasErrc::Inconclusive,
403 "Lagrange candidate does not verify against the full stationarity system",
404 operation);
405 }
406 }
407
408 std::map<std::string, std::shared_ptr<SymbolicExpr>> filtered;
409 for (const auto& [name, val] : full_solution) {
410 if (var_set.count(name)) {
411 filtered[name] = val;
412 }
413 }
414 if (!vector_calculus_point_has_vars(filtered, vars)) {
415 return LagrangeResult::failure(
416 CasErrc::Inconclusive,
417 "Lagrange candidate omits one or more variables",
418 operation);
419 }
420 verified.push_back(std::move(filtered));
421 }
422
423 if (verified.empty()) {
424 return LagrangeResult::failure(
425 CasErrc::Inconclusive,
426 "Lagrange solver produced no verifiable candidates in the supported domain",
427 operation);
428 }
429 return LagrangeResult::success(std::move(verified));
430}

481std::vector<std::map<std::string, std::shared_ptr<SymbolicExpr>>> lagrange_multipliers(
482 const std::shared_ptr<SymbolicExpr>& f,
483 const std::vector<std::shared_ptr<SymbolicExpr>>& constraints,
484 const std::vector<std::string>& vars)
485{
486 if (!f || constraints.empty() || vars.empty()) {
487 return {};
488 }
489
490 size_t n = vars.size();
491 size_t m = constraints.size();
492
493 /// 计算目标函数的梯度
494 VectorField grad_f;
495 grad_f.reserve(n);
496 for (const auto& var : vars) {
497 auto partial = f->differentiate(var);
498 if (partial) partial = partial->simplify();
499 grad_f.push_back(partial);
500 }
501
502 /// 计算每个约束的梯度
503 std::vector<VectorField> grad_constraints;
504 grad_constraints.reserve(m);
505 for (const auto& g : constraints) {
506 VectorField grad_g;
507 grad_g.reserve(n);
508 for (const auto& var : vars) {
509 auto partial = g->differentiate(var);
510 if (partial) partial = partial->simplify();
511 grad_g.push_back(partial);
512 }
513 grad_constraints.push_back(std::move(grad_g));
514 }
515
516 /// 构造乘数变量名
517 std::vector<std::string> lambda_names;
518 lambda_names.reserve(m);
519 for (size_t k = 0; k < m; ++k) {
520 lambda_names.push_back("lambda_" + std::to_string(k + 1));
521 }
522
523 /// 构造方程组: ∂f/∂xᵢ - ∑ λₖ · ∂gₖ/∂xᵢ = 0 (对每个 i)
524 /// 加上约束方程: gₖ = 0 (对每个 k)
525 std::vector<std::shared_ptr<SymbolicExpr>> equations;
526 equations.reserve(n + m);
527
528 for (size_t i = 0; i < n; ++i) {
529 /// ∂f/∂xᵢ - λ₁·∂g₁/∂xᵢ - λ₂·∂g₂/∂xᵢ - ...
530 auto eq = grad_f[i];
531 for (size_t k = 0; k < m; ++k) {
532 auto lambda_var = SymbolicExpr::variable(lambda_names[k]);
533 auto term = SymbolicExpr::multiply(lambda_var, grad_constraints[k][i]);
534 eq = SymbolicExpr::add(eq, SymbolicExpr::multiply(SymbolicExpr::number(-1), term));
535 }
536 if (eq) eq = eq->simplify();
537 equations.push_back(eq);
538 }
539
540 /// 添加约束方程
541 for (const auto& g : constraints) {
542 equations.push_back(g);
543 }
544
545 /// 构造所有未知数列表: 原始变量 + 乘数
546 std::vector<std::string> all_vars;
547 all_vars.reserve(n + m);
548 for (const auto& v : vars) {
549 all_vars.push_back(v);
550 }
551 for (const auto& lam : lambda_names) {
552 all_vars.push_back(lam);
553 }
554
555 /// 使用多项式系统求解器求解增广系统
556 std::vector<std::map<std::string, std::shared_ptr<SymbolicExpr>>> solutions;
557
558 std::vector<SymbolicExpr> poly_eqs;
559 poly_eqs.reserve(equations.size());
560 for (const auto& eq : equations) {
561 if (eq) poly_eqs.push_back(*eq);
562 }
563
564 if (poly_eqs.size() == equations.size()) {
565 auto poly_solutions = Solver::solve_polynomial_system(poly_eqs, all_vars);
566 for (const auto& sol : poly_solutions) {
567 std::map<std::string, std::shared_ptr<SymbolicExpr>> pt;
568 for (const auto& [name, val] : sol) {
569 pt[name] = lamina::detail::make_expression_ptr(val);
570 }
571 solutions.push_back(pt);
572 }
573 }
574
575 /// 如果多项式求解器失败，尝试线性求解器
576 if (solutions.empty()) {
577 solutions = SymbolicExpr::solve_system(equations, all_vars);
578 }
579
580 /// 从解中提取仅原始变量（去除乘数）
581 std::vector<std::map<std::string, std::shared_ptr<SymbolicExpr>>> result;
582 std::set<std::string> var_set(vars.begin(), vars.end());
583
584 for (const auto& sol : solutions) {
585 std::map<std::string, std::shared_ptr<SymbolicExpr>> filtered;
586 for (const auto& [name, val] : sol) {
587 if (var_set.count(name)) {
588 filtered[name] = val;
589 }
590 }
591 if (!filtered.empty()) {
592 result.push_back(std::move(filtered));
593 }
594 }
595
596 return result;
597}
