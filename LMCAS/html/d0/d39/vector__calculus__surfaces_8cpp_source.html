82std::shared_ptr<SymbolicExpr> surface_integral_scalar(
83 const std::shared_ptr<SymbolicExpr>& f, const VectorField& parametrization,
84 const std::string& u, const std::string& v,
85 const std::shared_ptr<SymbolicExpr>& u_lower, const std::shared_ptr<SymbolicExpr>& u_upper,
86 const std::shared_ptr<SymbolicExpr>& v_lower, const std::shared_ptr<SymbolicExpr>& v_upper)
87{
88 if (!f) {
89 throw std::invalid_argument("surface_integral_scalar: f must not be null");
90 }
91 if (parametrization.size() != 3) {
92 throw std::invalid_argument(
93 "surface_integral_scalar: parametrization must have 3 components");
94 }
95 if (!u_lower || !u_upper || !v_lower || !v_upper) {
96 throw std::invalid_argument("surface_integral_scalar: bounds must not be null");
97 }
98
99 std::vector<std::string> coord_vars = {"x", "y", "z"};
100
101 auto f_composed = f;
102 for (size_t i = 0; i < 3; ++i) {
103 f_composed = f_composed->substitute(coord_vars[i], parametrization[i]);
104 }
105 f_composed = f_composed->simplify();
106
107 auto cross = vector_calculus_cross_product_partials(parametrization, u, v);
108
109 std::shared_ptr<SymbolicExpr> mag_sq = nullptr;
110 for (size_t i = 0; i < 3; ++i) {
111 auto sq = SymbolicExpr::power(cross[i], SymbolicExpr::number(2));
112 sq = sq->simplify();
113 if (!mag_sq) {
114 mag_sq = sq;
115 } else {
116 mag_sq = SymbolicExpr::add(mag_sq, sq);
117 mag_sq = mag_sq->simplify();
118 }
119 }
120
121 auto magnitude = SymbolicExpr::sqrt(mag_sq);
122 magnitude = magnitude->simplify();
123
124 auto integrand = SymbolicExpr::multiply(f_composed, magnitude);
125 integrand = integrand->simplify();
126
127 Integrator integrator;
128 ComputationContext context;
129 std::vector<IntegrationStep> steps = {
130 {v, v_lower, v_upper},
131 {u, u_lower, u_upper},
132 };
133 auto result = integrate_multiple_checked(*integrand, steps, integrator, context);
134 if (!result) throw std::runtime_error(result.error().message);
135 return LMCAS::detail::make_expression_ptr(result.value());
136}
