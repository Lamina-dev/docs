11VectorField vector_calculus_detail::vector_calculus_cross_product_partials(
12 const VectorField& parametrization,
13 const std::string& u, const std::string& v,
14 detail::RewriteBudget& transform_budget)
15{
16 VectorField r_u;
17 r_u.reserve(3);
18 for (size_t i = 0; i < 3; ++i) {
19 auto deriv = detail::make_expression_ptr(detail::differentiate_node(detail::node(parametrization[i]), u, transform_budget));
20 r_u.push_back(deriv ? detail::make_expression_ptr(detail::normalize_node(detail::node(deriv), transform_budget)) : SymbolicExpr::number(0));
21 }
22
23 VectorField r_v;
24 r_v.reserve(3);
25 for (size_t i = 0; i < 3; ++i) {
26 auto deriv = detail::make_expression_ptr(detail::differentiate_node(detail::node(parametrization[i]), v, transform_budget));
27 r_v.push_back(deriv ? detail::make_expression_ptr(detail::normalize_node(detail::node(deriv), transform_budget)) : SymbolicExpr::number(0));
28 }
29
30 VectorField result;
31 result.reserve(3);
32 for (std::size_t index = 0; index < 3; ++index) {
33 result.push_back(
34 detail::make_expression_ptr(detail::normalize_node(detail::node(vector_calculus_cross_component(r_u, r_v, index)), transform_budget)));
35 }
36 return result;
37}

88std::shared_ptr<SymbolicExpr> surface_integral_scalar(
89 const std::shared_ptr<SymbolicExpr>& f, const VectorField& parametrization,
90 const std::string& u, const std::string& v,
91 const std::shared_ptr<SymbolicExpr>& u_lower, const std::shared_ptr<SymbolicExpr>& u_upper,
92 const std::shared_ptr<SymbolicExpr>& v_lower, const std::shared_ptr<SymbolicExpr>& v_upper)
93{
94 ComputationContext context;
95 detail::RewriteBudget transform_budget(context, (context).limits().max_recursion_depth,
96 (context).limits().max_ast_nodes, "transform");
97 if (!f) {
98 throw std::invalid_argument("surface_integral_scalar: f must not be null");
99 }
100 if (parametrization.size() != 3) {
101 throw std::invalid_argument(
102 "surface_integral_scalar: parametrization must have 3 components");
103 }
104 if (!u_lower || !u_upper || !v_lower || !v_upper) {
105 throw std::invalid_argument("surface_integral_scalar: bounds must not be null");
106 }
107
108 std::vector<std::string> coord_vars = {"x", "y", "z"};
109
110 auto f_composed = f;
111 for (size_t i = 0; i < 3; ++i) {
112 f_composed = detail::make_expression_ptr(detail::substitute_node(detail::node(f_composed), coord_vars[i], detail::node(parametrization[i]), transform_budget));
113 }
114 f_composed = detail::make_expression_ptr(detail::normalize_node(detail::node(f_composed), transform_budget));
115
116 auto cross = vector_calculus_cross_product_partials(parametrization, u, v, transform_budget);
117
118 std::shared_ptr<SymbolicExpr> mag_sq = nullptr;
119 for (size_t i = 0; i < 3; ++i) {
120 auto sq = SymbolicExpr::power(cross[i], SymbolicExpr::number(2));
121 sq = detail::make_expression_ptr(detail::normalize_node(detail::node(sq), transform_budget));
122 if (!mag_sq) {
123 mag_sq = sq;
124 } else {
125 mag_sq = SymbolicExpr::add(mag_sq, sq);
126 mag_sq = detail::make_expression_ptr(detail::normalize_node(detail::node(mag_sq), transform_budget));
127 }
128 }
129
130 auto magnitude = SymbolicExpr::sqrt(mag_sq);
131 magnitude = detail::make_expression_ptr(detail::normalize_node(detail::node(magnitude), transform_budget));
132
133 auto integrand = SymbolicExpr::multiply(f_composed, magnitude);
134 integrand = detail::make_expression_ptr(detail::normalize_node(detail::node(integrand), transform_budget));
135
136 Integrator integrator;
137
138 std::vector<IntegrationStep> steps = {
139 {v, v_lower, v_upper},
140 {u, u_lower, u_upper},
141 };
142 auto result = integrate_multiple_checked(*integrand, steps, integrator, context);
143 if (!result) throw result.error();
144 return LMCAS::detail::make_expression_ptr(result.value());
145}
