19Result<std::shared_ptr<const MatrixNode>> validate_decomposition_matrix(
20 const std::shared_ptr<SymbolicExpr>& A,

41 return Result<std::shared_ptr<const MatrixNode>>::failure(
42 CasErrc::InvalidArgument,
43 "matrix must be square",
44 operation);
45 }
46 return Result<std::shared_ptr<const MatrixNode>>::success(matrix);

75std::optional<Rational> exact_rational_expr(const std::shared_ptr<const SymbolicNode>& node) {
76 if (!node) return std::nullopt;
77 if (auto number = std::dynamic_pointer_cast<const NumberNode>(node)) {
78 if (std::holds_alternative<BigInt>(number->value())) {
79 return Rational(std::get<BigInt>(number->value()));
80 }
81 if (std::holds_alternative<Rational>(number->value())) {
82 return std::get<Rational>(number->value());
83 }
84 return std::nullopt;
85 }
86 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
87 Rational sum(0);
88 for (const auto& operand : add->operands()) {
89 auto value = exact_rational_expr(operand);
90 if (!value) return std::nullopt;
91 sum = sum + *value;
92 }
93 return sum;
94 }
95 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
96 Rational product(1);
97 for (const auto& operand : mul->operands()) {
98 auto value = exact_rational_expr(operand);
99 if (!value) return std::nullopt;
100 product = product * *value;
101 }
102 return product;
103 }
104 if (auto fn = std::dynamic_pointer_cast<const FunctionNode>(node)) {
105 if (fn->type() == FunctionNode::FuncType::Abs && fn->arguments().size() == 1) {
106 auto value = exact_rational_expr(fn->arguments()[0]);
107 if (!value) return std::nullopt;
108 return (*value < Rational(0)) ? (Rational(0) - *value) : *value;
109 }
110 }
111 return std::nullopt;
112}

150Rational exact_det(std::vector<std::vector<Rational>> matrix) {
151 const size_t n = matrix.size();
152 Rational det(1);
153 int sign = 1;
154 for (size_t col = 0; col < n; ++col) {
155 size_t pivot = col;
156 while (pivot < n && matrix[pivot][col] == Rational(0)) {
157 ++pivot;
158 }
159 if (pivot == n) return Rational(0);
160 if (pivot != col) {
161 std::swap(matrix[pivot], matrix[col]);
162 sign = -sign;
163 }
164 const Rational pivot_value = matrix[col][col];
165 det = det * pivot_value;
166 for (size_t row = col + 1; row < n; ++row) {
167 const Rational factor = matrix[row][col] / pivot_value;
168 for (size_t k = col; k < n; ++k) {
169 matrix[row][k] = matrix[row][k] - factor * matrix[col][k];
170 }
171 }
172 }
173 return sign < 0 ? (Rational(0) - det) : det;
174}

176Result<void> prove_exact_spd(
177 const std::shared_ptr<const MatrixNode>& matrix,
178 const std::string& operation) {
179 auto values = exact_rational_matrix(matrix);
180 if (!values) {
181 return Result<void>::failure(
182 CasErrc::Inconclusive,
183 "Cholesky input is outside the exact rational SPD support domain",
184 operation);
185 }
186 const size_t n = values->size();
187 for (size_t row = 0; row < n; ++row) {
188 for (size_t col = row + 1; col < n; ++col) {
189 if ((*values)[row][col] != (*values)[col][row]) {
190 return Result<void>::failure(
191 CasErrc::DomainError,
192 "Cholesky input must be symmetric",
193 operation);
194 }
195 }
196 }
197 for (size_t order = 1; order <= n; ++order) {
198 std::vector<std::vector<Rational>> leading(
199 order, std::vector<Rational>(order, Rational(0)));
200 for (size_t row = 0; row < order; ++row) {
201 for (size_t col = 0; col < order; ++col) {
202 leading[row][col] = (*values)[row][col];
203 }
204 }
205 if (exact_det(std::move(leading)) <= Rational(0)) {
206 return Result<void>::failure(
207 CasErrc::DomainError,
208 "Cholesky input is not positive definite",
209 operation);
210 }
211 }
212 return Result<void>::success();
213}

244Result<void> prove_exact_full_column_rank_qr_support(
245 const std::shared_ptr<const MatrixNode>& matrix,
246 const std::string& operation) {
247 auto values = exact_rational_matrix(matrix);
248 if (!values) {
249 return Result<void>::failure(
250 CasErrc::Inconclusive,
251 "QR input is outside the exact rational full-column-rank support domain",
252 operation);
253 }
254 const size_t rows = values->size();
255 const size_t cols = rows == 0 ? 0 : (*values)[0].size();
256 if (rows < cols) {
257 return Result<void>::failure(
258 CasErrc::Inconclusive,
259 "QR decomposition requires at least as many rows as columns in the supported domain",
260 operation);
261 }
262 std::vector<std::vector<Rational>> gram(
263 cols, std::vector<Rational>(cols, Rational(0)));
264 for (size_t i = 0; i < cols; ++i) {
265 for (size_t j = 0; j < cols; ++j) {
266 Rational sum(0);
267 for (size_t row = 0; row < rows; ++row) {
268 sum = sum + (*values)[row][i] * (*values)[row][j];
269 }
270 gram[i][j] = sum;
271 }
272 }
273 for (size_t order = 1; order <= cols; ++order) {
274 std::vector<std::vector<Rational>> leading(
275 order, std::vector<Rational>(order, Rational(0)));
276 for (size_t row = 0; row < order; ++row) {
277 for (size_t col = 0; col < order; ++col) {
278 leading[row][col] = gram[row][col];
279 }
280 }
281 if (exact_det(std::move(leading)) <= Rational(0)) {
282 return Result<void>::failure(
283 CasErrc::Inconclusive,
284 "QR decomposition requires proven full column rank",
285 operation);
286 }
287 }
288 return Result<void>::success();
289}

300std::optional<std::vector<Rational>> exact_nonnegative_diagonal_entries(
301 const std::shared_ptr<const MatrixNode>& matrix) {
302 auto values = exact_rational_matrix(matrix);
303 if (!values) return std::nullopt;
304 const size_t rows = values->size();
305 const size_t cols = rows == 0 ? 0 : (*values)[0].size();
306 const size_t diag_count = std::min(rows, cols);
307 std::vector<Rational> diagonal(diag_count, Rational(0));
308 for (size_t row = 0; row < rows; ++row) {
309 for (size_t col = 0; col < cols; ++col) {
310 const bool is_diag = row == col && row < diag_count;
311 if (!is_diag && (*values)[row][col] != Rational(0)) {
312 return std::nullopt;
313 }
314 if (is_diag) {
315 if ((*values)[row][col] < Rational(0)) {
316 return std::nullopt;
317 }
318 diagonal[row] = (*values)[row][col];
319 }
320 }
321 }
322 return diagonal;
323}

403bool lu_decomposition(
404 const std::shared_ptr<SymbolicExpr>& A,
405 std::shared_ptr<SymbolicExpr>& L,
406 std::shared_ptr<SymbolicExpr>& U) {
407 auto mat = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(A));
408 if (!mat || mat->rows() != mat->cols()) return false;
409 size_t n = mat->rows();
410
411 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> L_grid(n, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
412 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> U_grid(n, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
413
414 for (size_t i = 0; i < n; i++) {
415 for (size_t k = i; k < n; k++) {
416 auto sum = SymbolicExpr::number(0);
417 for (size_t j = 0; j < i; j++) sum = SymbolicExpr::add(sum, SymbolicExpr::multiply(L_grid[i][j], U_grid[j][k]));
418 auto a_ik = lamina::detail::make_expression_ptr(mat->get(i, k));
419 U_grid[i][k] = SymbolicExpr::add(a_ik, SymbolicExpr::multiply(SymbolicExpr::number(-1), sum));
420 }
421
422 for (size_t k = i; k < n; k++) {
423 if (i == k) {
424 L_grid[i][i] = SymbolicExpr::number(1);
425 } else {
426 auto sum = SymbolicExpr::number(0);
427 for (size_t j = 0; j < i; j++) sum = SymbolicExpr::add(sum, SymbolicExpr::multiply(L_grid[k][j], U_grid[j][i]));
428 auto a_ki = lamina::detail::make_expression_ptr(mat->get(k, i));
429 L_grid[k][i] = SymbolicExpr::divide(SymbolicExpr::add(a_ki, SymbolicExpr::multiply(SymbolicExpr::number(-1), sum)), U_grid[i][i]);
430 }
431 }
432 }
433
434 L = SymbolicExpr::matrix(L_grid);
435 U = SymbolicExpr::matrix(U_grid);
436 return true;
437}

483bool qr_decomposition(
484 const std::shared_ptr<SymbolicExpr>& A,
485 std::shared_ptr<SymbolicExpr>& Q,
486 std::shared_ptr<SymbolicExpr>& R) {
487 auto mat = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(A));
488 if (!mat) return false;
489 size_t m = mat->rows();
490 size_t n = mat->cols();
491
492 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> Q_grid(m, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
493 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> R_grid(n, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
494
495 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> A_cols(n, std::vector<std::shared_ptr<SymbolicExpr>>(m));
496 for (size_t j = 0; j < n; j++) {
497 for (size_t i = 0; i < m; i++) A_cols[j][i] = lamina::detail::make_expression_ptr(mat->get(i, j));
498 }
499
500 for (size_t j = 0; j < n; j++) {
501 std::vector<std::shared_ptr<SymbolicExpr>> u_j = A_cols[j];
502 for (size_t i = 0; i < j; i++) {
503 auto dot = SymbolicExpr::number(0);
504 for (size_t k = 0; k < m; k++) dot = SymbolicExpr::add(dot, SymbolicExpr::multiply(Q_grid[k][i], A_cols[j][k]));
505 R_grid[i][j] = dot;
506
507 for (size_t k = 0; k < m; k++) {
508 u_j[k] = SymbolicExpr::add(u_j[k], SymbolicExpr::multiply(SymbolicExpr::number(-1), SymbolicExpr::multiply(R_grid[i][j], Q_grid[k][i])));
509 }
510 }
511
512 auto norm_sq = SymbolicExpr::number(0);
513 for (size_t k = 0; k < m; k++) norm_sq = SymbolicExpr::add(norm_sq, SymbolicExpr::multiply(u_j[k], u_j[k]));
514 R_grid[j][j] = SymbolicExpr::sqrt(norm_sq);
515
516 for (size_t k = 0; k < m; k++) Q_grid[k][j] = SymbolicExpr::divide(u_j[k], R_grid[j][j]);
517 }
518
519 Q = SymbolicExpr::matrix(Q_grid);
520 R = SymbolicExpr::matrix(R_grid);
521 return true;
522}

591 const std::string operation = "svd_decomposition";
592 auto matrix = validate_decomposition_matrix(A, false, context, operation);
593 if (!matrix) return SVDDecompositionResult::failure(matrix.error());
594 auto budget = context.consume_steps(matrix.value()->rows() * matrix.value()->cols() * 32 + 32,
595 operation);
596 if (!budget) return SVDDecompositionResult::failure(budget.error());
597 auto diagonal = exact_nonnegative_diagonal_entries(matrix.value());
598 if (!diagonal) {
599 return SVDDecompositionResult::failure(
600 CasErrc::Inconclusive,
601 "SVD checked support is currently limited to exact rational nonnegative diagonal matrices",
602 operation);
603 }
604 auto U = identity_matrix_expr(matrix.value()->rows());
605 auto S = rectangular_diagonal_expr(matrix.value()->rows(), matrix.value()->cols(), *diagonal);
606 auto V = identity_matrix_expr(matrix.value()->cols());
607 return SVDDecompositionResult::success(SVDDecomposition{U, S, V});
608}

617static std::vector<std::shared_ptr<SymbolicExpr>> normalize_vec(std::vector<std::shared_ptr<SymbolicExpr>> v) {
618 auto norm_sq = SymbolicExpr::number(0);
619 for (auto& x : v) norm_sq = SymbolicExpr::add(norm_sq, SymbolicExpr::multiply(x, x));
620 auto norm = SymbolicExpr::sqrt(norm_sq);
621 std::vector<std::shared_ptr<SymbolicExpr>> res;
622 for (auto& x : v) res.push_back(SymbolicExpr::divide(x, norm));
623 return res;
624}

626bool svd_decomposition(
627 const std::shared_ptr<SymbolicExpr>& A,
628 std::shared_ptr<SymbolicExpr>& U,
629 std::shared_ptr<SymbolicExpr>& S,
630 std::shared_ptr<SymbolicExpr>& V) {
631
632 auto mat = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(A));
633 if (!mat) return false;
634
635 auto AT = SymbolicExpr::transpose(A);
636 auto ATA = SymbolicExpr::multiply(AT, A);
637
638 auto eigen_V = SymbolicExpr::eigenvectors(ATA);
639 if (eigen_V.empty()) return false; // Fallback fails if matrix > 4x4 or no symbolic roots
640
641 size_t n = mat->cols();
642 size_t m = mat->rows();
643
644 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> V_cols;
645 std::vector<std::shared_ptr<SymbolicExpr>> sigmas;
646
647 for (auto& pair : eigen_V) {
648 auto sigma = SymbolicExpr::sqrt(pair.first);
649 for (auto& vec_mat : pair.second) {
650 auto v_mat_node = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(vec_mat));
651 std::vector<std::shared_ptr<SymbolicExpr>> v_col;
652 for (size_t i = 0; i < n; i++) v_col.push_back(lamina::detail::make_expression_ptr(v_mat_node->get(i, 0)));
653 V_cols.push_back(normalize_vec(v_col));
654 sigmas.push_back(sigma);
655 }
656 }
657
658 if (V_cols.size() != n) return false; // Incomplete basis
659
660 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> U_cols;
661 for (size_t i = 0; i < n; i++) {
662 if (!lamina::detail::node(sigmas[i])->is_zero()) {
663 /// u_i = A v_i / sigma_i
664 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> v_col_mat(n, std::vector<std::shared_ptr<SymbolicExpr>>(1));
665 for (size_t j = 0; j < n; j++) v_col_mat[j][0] = V_cols[i][j];
666 auto Avi = SymbolicExpr::multiply(A, SymbolicExpr::matrix(v_col_mat));
667 auto Avi_node = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(Avi));
668 std::vector<std::shared_ptr<SymbolicExpr>> u_col;
669 for (size_t j = 0; j < m; j++) {
670 u_col.push_back(SymbolicExpr::divide(lamina::detail::make_expression_ptr(Avi_node->get(j, 0)), sigmas[i]));
671 }
672 U_cols.push_back(u_col);
673 }
674 }
675
676 /// Complete U basis with standard basis via Gram-Schmidt if needed (simplified for stub replacement)
677 for (size_t i = 0; i < m && U_cols.size() < m; i++) {
678 std::vector<std::shared_ptr<SymbolicExpr>> ei(m, SymbolicExpr::number(0));
679 ei[i] = SymbolicExpr::number(1);
680 auto ui = ei;
681 for (auto& uj : U_cols) {
682 auto dot = SymbolicExpr::number(0);
683 for (size_t k = 0; k < m; k++) dot = SymbolicExpr::add(dot, SymbolicExpr::multiply(uj[k], ei[k]));
684 for (size_t k = 0; k < m; k++) ui[k] = SymbolicExpr::add(ui[k], SymbolicExpr::multiply(SymbolicExpr::number(-1), SymbolicExpr::multiply(dot, uj[k])));
685 }
686 auto norm_sq = SymbolicExpr::number(0);
687 for (auto& x : ui) norm_sq = SymbolicExpr::add(norm_sq, SymbolicExpr::multiply(x, x));
688 if (!lamina::detail::node(norm_sq)->is_zero()) {
689 U_cols.push_back(normalize_vec(ui));
690 }
691 }
692
693 if (U_cols.size() != m) return false;
694
695 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> U_grid(m, std::vector<std::shared_ptr<SymbolicExpr>>(m));
696 for (size_t i = 0; i < m; i++) {
697 for (size_t j = 0; j < m; j++) U_grid[i][j] = U_cols[j][i];
698 }
699
700 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> V_grid(n, std::vector<std::shared_ptr<SymbolicExpr>>(n));
701 for (size_t i = 0; i < n; i++) {
702 for (size_t j = 0; j < n; j++) V_grid[i][j] = V_cols[j][i];
703 }
704
705 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> S_grid(m, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
706 for (size_t i = 0; i < std::min(m, n); i++) S_grid[i][i] = sigmas[i];
707
708 U = SymbolicExpr::matrix(U_grid);
709 S = SymbolicExpr::matrix(S_grid);
710 V = SymbolicExpr::matrix(V_grid);
711 return true;
712}

714std::shared_ptr<SymbolicExpr> matrix_exp(
715 const std::shared_ptr<SymbolicExpr>& A) {
716 auto mat = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(A));
717 if (!mat || mat->rows() != mat->cols()) return SymbolicExpr::exp(A);
718 size_t n = mat->rows();
719
720 bool is_zero_matrix = true;
721 for (size_t r = 0; r < n && is_zero_matrix; ++r) {
722 for (size_t c = 0; c < n; ++c) {
723 auto entry = lamina::detail::make_expression_ptr(mat->get(r, c))->simplify();
724 if (!lamina::detail::node(entry)->is_zero()) {
725 is_zero_matrix = false;
726 break;
727 }
728 }
729 }
730 if (is_zero_matrix) {
731 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> identity(
732 n, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
733 for (size_t i = 0; i < n; ++i) {
734 identity[i][i] = SymbolicExpr::number(1);
735 }
736 return SymbolicExpr::matrix(identity);
737 }
738
739 auto eigen_V = SymbolicExpr::eigenvectors(A);
740 if (eigen_V.empty()) return SymbolicExpr::exp(A); // Fallback to AST node
741
742 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> P_cols;
743 std::vector<std::shared_ptr<SymbolicExpr>> evals;
744 for (auto& pair : eigen_V) {
745 for (auto& vec : pair.second) {
746 auto v_mat_node = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(vec));
747 std::vector<std::shared_ptr<SymbolicExpr>> v_col;
748 for (size_t i = 0; i < n; i++) v_col.push_back(lamina::detail::make_expression_ptr(v_mat_node->get(i, 0)));
749 P_cols.push_back(v_col);
750 evals.push_back(pair.first);
751 }
752 }
753
754 if (P_cols.size() != n) return SymbolicExpr::exp(A); // Not diagonalizable
755
756 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> P_grid(n, std::vector<std::shared_ptr<SymbolicExpr>>(n));
757 for (size_t i = 0; i < n; i++) {
758 for (size_t j = 0; j < n; j++) P_grid[i][j] = P_cols[j][i];
759 }
760 auto P = SymbolicExpr::matrix(P_grid);
761 auto P_inv = SymbolicExpr::inverse(P);
762 if (!P_inv) return SymbolicExpr::exp(A);
763
764 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> expD_grid(n, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
765 for (size_t i = 0; i < n; i++) expD_grid[i][i] = SymbolicExpr::exp(evals[i]);
766 auto expD = SymbolicExpr::matrix(expD_grid);
767
768 return SymbolicExpr::multiply(P, SymbolicExpr::multiply(expD, P_inv));
769}

782int matrix_rank(const std::shared_ptr<SymbolicExpr>& A) {
783 auto mat = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(A));
784 if (!mat) return -1;
785 size_t m = mat->rows(), n = mat->cols();
786 if (m == 0 || n == 0) return 0;
787
788 /// 构造可变副本（数值化为 double，符号项无法判零时视为非零主元）
789 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> g(m,
790 std::vector<std::shared_ptr<SymbolicExpr>>(n));
791 for (size_t i = 0; i < m; ++i)
792 for (size_t j = 0; j < n; ++j)
793 g[i][j] = lamina::detail::make_expression_ptr(mat->get(i, j))->simplify();
794
795 auto is_zero_expr = [](const std::shared_ptr<SymbolicExpr>& e) {
796 return e && lamina::detail::node(e) && lamina::detail::node(e)->is_zero();
797 };
798
799 int rank = 0;
800 size_t row = 0;
801 for (size_t col = 0; col < n && row < m; ++col) {
802 /// 寻找主元
803 size_t pivot = row;
804 while (pivot < m && is_zero_expr(g[pivot][col])) ++pivot;
805 if (pivot == m) continue;
806 std::swap(g[row], g[pivot]);
807
808 auto pval = g[row][col];
809 /// 消元
810 for (size_t i = 0; i < m; ++i) {
811 if (i == row || is_zero_expr(g[i][col])) continue;
812 auto factor = SymbolicExpr::divide(g[i][col], pval);
813 for (size_t j = col; j < n; ++j) {
814 auto sub = SymbolicExpr::multiply(factor, g[row][j]);
815 g[i][j] = SymbolicExpr::add(g[i][j],
816 SymbolicExpr::multiply(SymbolicExpr::number(-1), sub))->simplify();
817 }
818 }
819 ++row;
820 ++rank;
821 }
822 return rank;
823}

826std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> gram_schmidt(
827 const std::vector<std::vector<std::shared_ptr<SymbolicExpr>>>& vectors,
828 bool normalize) {
829 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> basis;
830 for (const auto& v : vectors) {
831 std::vector<std::shared_ptr<SymbolicExpr>> u = v;
832 for (const auto& b : basis) {
833 /// 投影系数 <v,b>/<b,b>
834 auto vb = SymbolicExpr::number(0);
835 auto bb = SymbolicExpr::number(0);
836 for (size_t k = 0; k < u.size(); ++k) {
837 vb = SymbolicExpr::add(vb, SymbolicExpr::multiply(v[k], b[k]));
838 bb = SymbolicExpr::add(bb, SymbolicExpr::multiply(b[k], b[k]));
839 }
840 if (lamina::detail::node(bb) && lamina::detail::node(bb)->is_zero()) continue;
841 auto coeff = SymbolicExpr::divide(vb, bb);
842 for (size_t k = 0; k < u.size(); ++k) {
843 u[k] = SymbolicExpr::add(u[k],
844 SymbolicExpr::multiply(SymbolicExpr::number(-1),
845 SymbolicExpr::multiply(coeff, b[k])))->simplify();
846 }
847 }
848 /// 检查 u 是否为零向量（线性相关）
849 auto norm_sq = SymbolicExpr::number(0);
850 for (auto& x : u) norm_sq = SymbolicExpr::add(norm_sq, SymbolicExpr::multiply(x, x));
851 norm_sq = norm_sq->simplify();
852 if (lamina::detail::node(norm_sq) && lamina::detail::node(norm_sq)->is_zero()) continue;
853 basis.push_back(u);
854 }
855 if (normalize) {
856 for (auto& u : basis) {
857 auto norm_sq = SymbolicExpr::number(0);
858 for (auto& x : u) norm_sq = SymbolicExpr::add(norm_sq, SymbolicExpr::multiply(x, x));
859 auto norm = SymbolicExpr::sqrt(norm_sq);
860 for (auto& x : u) x = SymbolicExpr::divide(x, norm)->simplify();
861 }
862 }
863 return basis;
864}

868 auto mat = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(A));
869 if (!mat || mat->rows() != mat->cols()) return nullptr;
870 size_t n = mat->rows();
871
872 auto eigen_V = SymbolicExpr::eigenvectors(A);
873 if (eigen_V.empty()) return nullptr;
874
875 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> P_cols;
876 std::vector<std::shared_ptr<SymbolicExpr>> evals;
877 for (auto& pr : eigen_V) {
878 for (auto& vec : pr.second) {
879 auto vnode = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(vec));
880 if (!vnode) return nullptr;
881 std::vector<std::shared_ptr<SymbolicExpr>> col;
882 for (size_t i = 0; i < n; ++i) col.push_back(lamina::detail::make_expression_ptr(vnode->get(i, 0)));
883 P_cols.push_back(col);
884 evals.push_back(pr.first);
885 }
886 }
887 if (P_cols.size() != n) return nullptr;
888
889 /// 检查特征值为正（实数对数存在性）
890 for (auto& ev : evals) {
891 auto se = ev->simplify();
892 if (se->is_number()) {
893 auto d = checked_finite_numeric(se);
894 if (!d || *d <= 0) return nullptr;
895 }
896 }
897
898 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> P_grid(n,
899 std::vector<std::shared_ptr<SymbolicExpr>>(n));
900 for (size_t i = 0; i < n; ++i)
901 for (size_t j = 0; j < n; ++j) P_grid[i][j] = P_cols[j][i];
902 auto P = SymbolicExpr::matrix(P_grid);
903 auto P_inv = SymbolicExpr::inverse(P);
904 if (!P_inv) return nullptr;
905
906 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> logD(n,
907 std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
908 for (size_t i = 0; i < n; ++i) logD[i][i] = SymbolicExpr::ln(evals[i]);
909 auto logDm = SymbolicExpr::matrix(logD);
910
911 return SymbolicExpr::multiply(P, SymbolicExpr::multiply(logDm, P_inv));
912}

936std::shared_ptr<SymbolicExpr> matrix_norm(const std::shared_ptr<SymbolicExpr>& A,
937 const std::string& type) {
938 auto mat = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(A));
939 if (!mat) return nullptr;
940 size_t m = mat->rows(), n = mat->cols();
941
942 auto abs_of = [](const std::shared_ptr<SymbolicExpr>& e) {
943 auto node = lamina::detail::make_node<FunctionNode>(FunctionNode::FuncType::Abs,
944 std::vector<std::shared_ptr<const SymbolicNode>>{lamina::detail::node(e)});
945 return lamina::detail::make_expression_ptr(node);
946 };
947
948 if (type == "frobenius") {
949 auto sum = SymbolicExpr::number(0);
950 for (size_t i = 0; i < m; ++i)
951 for (size_t j = 0; j < n; ++j) {
952 auto e = lamina::detail::make_expression_ptr(mat->get(i, j));
953 sum = SymbolicExpr::add(sum, SymbolicExpr::multiply(e, e));
954 }
955 return SymbolicExpr::sqrt(sum)->simplify();
956 }
957 if (type == "1") {
958 /// 最大列和
959 std::shared_ptr<SymbolicExpr> best = nullptr;
960 std::optional<Rational> best_exact;
961 double best_val = -1;
962 for (size_t j = 0; j < n; ++j) {
963 auto colsum = SymbolicExpr::number(0);
964 std::optional<Rational> exact_sum = Rational(0);
965 for (size_t i = 0; i < m; ++i) {
966 auto entry = lamina::detail::make_expression_ptr(mat->get(i, j));
967 colsum = SymbolicExpr::add(colsum, abs_of(entry));
968 if (exact_sum) {
969 auto exact_entry = exact_rational_expr(entry);
970 if (exact_entry) {
971 *exact_sum = *exact_sum +
972 ((*exact_entry < Rational(0)) ? (Rational(0) - *exact_entry) : *exact_entry);
973 } else {
974 exact_sum.reset();
975 }
976 }
977 }
978 colsum = colsum->simplify();
979 auto exact = exact_sum ? exact_sum : exact_rational_expr(colsum);
980 if (exact && (!best || (best_exact && *exact > *best_exact))) {
981 best = exact_number_expr(*exact);
982 best_exact = *exact;
983 best_val = -1;
984 continue;
985 }
986 if (exact && !best_exact && best) {
987 auto maybe_v = checked_finite_numeric(colsum);
988 if (maybe_v && *maybe_v > best_val) {
989 best = exact_number_expr(*exact);
990 best_val = *maybe_v;
991 }
992 continue;
993 }
994 if (!exact && !best_exact) {
995 auto maybe_v = checked_finite_numeric(colsum);
996 double v = maybe_v ? *maybe_v : -std::numeric_limits<double>::infinity();
997 if (!best || v > best_val) { best = colsum; best_val = v; }
998 }
999 }
1000 return best;
1001 }
1002 if (type == "inf") {
1003 /// 最大行和
1004 std::shared_ptr<SymbolicExpr> best = nullptr;
1005 std::optional<Rational> best_exact;
1006 double best_val = -1;
1007 for (size_t i = 0; i < m; ++i) {
1008 auto rowsum = SymbolicExpr::number(0);
1009 std::optional<Rational> exact_sum = Rational(0);
1010 for (size_t j = 0; j < n; ++j) {
1011 auto entry = lamina::detail::make_expression_ptr(mat->get(i, j));
1012 rowsum = SymbolicExpr::add(rowsum, abs_of(entry));
1013 if (exact_sum) {
1014 auto exact_entry = exact_rational_expr(entry);
1015 if (exact_entry) {
1016 *exact_sum = *exact_sum +
1017 ((*exact_entry < Rational(0)) ? (Rational(0) - *exact_entry) : *exact_entry);
1018 } else {
1019 exact_sum.reset();
1020 }
1021 }
1022 }
1023 rowsum = rowsum->simplify();
1024 auto exact = exact_sum ? exact_sum : exact_rational_expr(rowsum);
1025 if (exact && (!best || (best_exact && *exact > *best_exact))) {
1026 best = exact_number_expr(*exact);
1027 best_exact = *exact;
1028 best_val = -1;
1029 continue;
1030 }
1031 if (exact && !best_exact && best) {
1032 auto maybe_v = checked_finite_numeric(rowsum);
1033 if (maybe_v && *maybe_v > best_val) {
1034 best = exact_number_expr(*exact);
1035 best_val = *maybe_v;
1036 }
1037 continue;
1038 }
1039 if (!exact && !best_exact) {
1040 auto maybe_v = checked_finite_numeric(rowsum);
1041 double v = maybe_v ? *maybe_v : -std::numeric_limits<double>::infinity();
1042 if (!best || v > best_val) { best = rowsum; best_val = v; }
1043 }
1044 }
1045 return best;
1046 }
1047 return nullptr;
1048}

1051std::shared_ptr<SymbolicExpr> quadratic_form_matrix(
1052 const std::shared_ptr<SymbolicExpr>& expr,
1053 const std::vector<std::string>& vars) {
1054 if (!expr) return nullptr;
1055 size_t n = vars.size();
1056 auto e = expr->expand();
1057 if (!e) e = expr;
1058
1059 /// A[i][j] = (1/2) ∂²(expr)/∂xᵢ∂xⱼ （对称矩阵，对角为 ∂²/2 即系数本身的一半*2）
1060 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> grid(n,
1061 std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
1062 for (size_t i = 0; i < n; ++i) {
1063 auto di = e->differentiate(vars[i]);
1064 for (size_t j = 0; j < n; ++j) {
1065 auto dij = di->differentiate(vars[j]);
1066 /// 对二次型，二阶偏导为常数；A_ij = (1/2)·∂²/∂xi∂xj
1067 grid[i][j] = SymbolicExpr::multiply(SymbolicExpr::number(Rational(1, 2)), dij)->simplify();
1068 }
1069 }
1070 return SymbolicExpr::matrix(grid);
1071}

1073std::string classify_quadratic_form(const std::shared_ptr<SymbolicExpr>& A) {
1074 /// 使用特征值列表（来自特征多项式求根），避免脆弱的特征向量求解。
1075 auto evals_expr = SymbolicExpr::eigenvalues(A);
1076 if (!evals_expr) return "unknown";
1077 auto mat_node = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(evals_expr));
1078 if (!mat_node || mat_node->cols() == 0) return "unknown";
1079
1080 std::vector<std::shared_ptr<SymbolicExpr>> evals;
1081 for (size_t i = 0; i < mat_node->cols(); ++i) {
1082 evals.push_back(lamina::detail::make_expression_ptr(mat_node->get(0, i)));
1083 }

1151bool jordan_form(const std::shared_ptr<SymbolicExpr>& A,
1152 std::shared_ptr<SymbolicExpr>& J, std::shared_ptr<SymbolicExpr>& P) {
1153 auto mat = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(A));
1154 if (!mat || mat->rows() != mat->cols()) return false;
1155 size_t n = mat->rows();
1156
1157 bool is_diagonal = true;
1158 for (size_t r = 0; r < n && is_diagonal; ++r) {
1159 for (size_t c = 0; c < n; ++c) {
1160 if (r == c) continue;
1161 auto entry = lamina::detail::make_expression_ptr(mat->get(r, c))->simplify();
1162 if (!lamina::detail::node(entry)->is_zero()) {
1163 is_diagonal = false;
1164 break;
1165 }
1166 }
1167 }
1168 if (is_diagonal) {
1169 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> J_grid(
1170 n, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
1171 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> P_grid(
1172 n, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
1173 for (size_t i = 0; i < n; ++i) {
1174 J_grid[i][i] = lamina::detail::make_expression_ptr(mat->get(i, i))->simplify();
1175 P_grid[i][i] = SymbolicExpr::number(1);
1176 }
1177 J = SymbolicExpr::matrix(J_grid);
1178 P = SymbolicExpr::matrix(P_grid);
1179 return true;
1180 }
1181
1182 auto eigen_V = SymbolicExpr::eigenvectors(A);
1183 if (eigen_V.empty()) return false;
1184
1185 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> P_cols;
1186 std::vector<std::shared_ptr<SymbolicExpr>> diag;
1187 for (auto& pr : eigen_V) {
1188 for (auto& vec : pr.second) {
1189 auto vnode = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(vec));
1190 if (!vnode) return false;
1191 std::vector<std::shared_ptr<SymbolicExpr>> col;
1192 for (size_t i = 0; i < n; ++i) col.push_back(lamina::detail::make_expression_ptr(vnode->get(i, 0)));
1193 P_cols.push_back(col);
1194 diag.push_back(pr.first);
1195 }
1196 }
1197 /// 当特征向量数等于 n 时，矩阵可对角化，Jordan 型即对角阵。
1198 /// 缺陷情形（重根但特征向量不足）当前不支持广义特征向量链，返回 false。
1199 if (P_cols.size() != n) return false;
1200
1201 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> P_grid(n,
1202 std::vector<std::shared_ptr<SymbolicExpr>>(n));
1203 for (size_t i = 0; i < n; ++i)
1204 for (size_t j = 0; j < n; ++j) P_grid[i][j] = P_cols[j][i];
1205 P = SymbolicExpr::matrix(P_grid);
1206
1207 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> J_grid(n,
1208 std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
1209 for (size_t i = 0; i < n; ++i) J_grid[i][i] = diag[i];
1210 J = SymbolicExpr::matrix(J_grid);
1211 return true;
1212}
