22Result<std::shared_ptr<const MatrixNode>> validate_decomposition_matrix(
23 const std::shared_ptr<SymbolicExpr>& A,
24 bool require_square,
25 ComputationContext& context,
26 const std::string& operation)
27{
28 auto step = context.consume_steps(1, operation);
29 if (!step) return Result<std::shared_ptr<const MatrixNode>>::failure(step.error());
30 if (!A || !lamina::detail::node(A)) {
31 return Result<std::shared_ptr<const MatrixNode>>::failure(
32 CasErrc::InvalidArgument,
33 "matrix input cannot be null",
34 operation);
35 }
36 auto matrix = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(A));
37 if (!matrix) {
38 return Result<std::shared_ptr<const MatrixNode>>::failure(
39 CasErrc::InvalidArgument,
40 "input must be a matrix",
41 operation);
42 }
43 if (require_square && matrix->rows() != matrix->cols()) {
44 return Result<std::shared_ptr<const MatrixNode>>::failure(
45 CasErrc::InvalidArgument,
46 "matrix must be square",
47 operation);
48 }
49 return Result<std::shared_ptr<const MatrixNode>>::success(matrix);
50}

72std::optional<Rational> exact_rational_expr(const std::shared_ptr<const SymbolicNode>& node) {
73 if (!node) return std::nullopt;
74 if (auto number = std::dynamic_pointer_cast<const NumberNode>(node)) {
75 if (std::holds_alternative<BigInt>(number->value())) {
76 return Rational(std::get<BigInt>(number->value()));
77 }
78 if (std::holds_alternative<Rational>(number->value())) {
79 return std::get<Rational>(number->value());
80 }
81 return std::nullopt;
82 }
83 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
84 Rational sum(0);
85 for (const auto& operand : add->operands()) {
86 auto value = exact_rational_expr(operand);
87 if (!value) return std::nullopt;
88 sum = sum + *value;
89 }
90 return sum;
91 }
92 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
93 Rational product(1);
94 for (const auto& operand : mul->operands()) {
95 auto value = exact_rational_expr(operand);
96 if (!value) return std::nullopt;
97 product = product * *value;
98 }
99 return product;
100 }
101 if (auto fn = std::dynamic_pointer_cast<const FunctionNode>(node)) {
102 if (fn->type() == FunctionNode::FuncType::Abs && fn->arguments().size() == 1) {
103 auto value = exact_rational_expr(fn->arguments()[0]);
104 if (!value) return std::nullopt;
105 return (*value < Rational(0)) ? (Rational(0) - *value) : *value;
106 }
107 }
108 return std::nullopt;
109}

112 if (!expr || !lamina::detail::node(expr)) return std::nullopt;
113 auto direct = exact_rational_expr(lamina::detail::node(expr));
114 if (direct) return direct;
115 auto simplified = expr->simplify();
116 return simplified ? exact_rational_expr(lamina::detail::node(simplified)) : std::nullopt;

132std::optional<std::vector<std::vector<Rational>>> exact_rational_matrix(
133 const std::shared_ptr<const MatrixNode>& matrix) {
134 if (!matrix) return std::nullopt;
135 std::vector<std::vector<Rational>> values(
136 matrix->rows(), std::vector<Rational>(matrix->cols(), Rational(0)));
137 for (size_t row = 0; row < matrix->rows(); ++row) {
138 for (size_t col = 0; col < matrix->cols(); ++col) {
139 auto value = exact_rational_expr(matrix->get(row, col));
140 if (!value) return std::nullopt;
141 values[row][col] = *value;
142 }
143 }

148Result<void> prove_exact_spd(
149 const std::shared_ptr<const MatrixNode>& matrix,
150 ComputationContext& context,
151 const std::string& operation) {
152 auto values = exact_rational_matrix(matrix);
153 if (!values) {
154 return Result<void>::failure(
155 CasErrc::Inconclusive,
156 "Cholesky input is outside the exact rational SPD support domain",
157 operation);
158 }
159 const size_t n = values->size();
160 for (size_t row = 0; row < n; ++row) {
161 for (size_t col = row + 1; col < n; ++col) {
162 if ((*values)[row][col] != (*values)[col][row]) {
163 return Result<void>::failure(
164 CasErrc::DomainError,
165 "Cholesky input must be symmetric",
166 operation);
167 }
168 }
169 }
170 for (size_t order = 1; order <= n; ++order) {
171 std::vector<Rational> leading;
172 leading.reserve(order * order);
173 for (size_t row = 0; row < order; ++row) {
174 for (size_t col = 0; col < order; ++col) {
175 leading.push_back((*values)[row][col]);
176 }
177 }
178 auto determinant = detail::rational_determinant_exact(
179 order, std::move(leading), context, operation);
180 if (!determinant) return Result<void>::failure(determinant.error());
181 if (determinant.value() <= Rational(0)) {
182 return Result<void>::failure(
183 CasErrc::DomainError,
184 "Cholesky input is not positive definite",
185 operation);
186 }
187 }
188 return Result<void>::success();
189}

192Result<void> prove_exact_full_column_rank_qr_support(
193 const std::shared_ptr<const MatrixNode>& matrix,
194 ComputationContext& context,
195 const std::string& operation) {
196 auto values = exact_rational_matrix(matrix);
197 if (!values) {
198 return Result<void>::failure(
199 CasErrc::Inconclusive,
200 "QR input is outside the exact rational full-column-rank support domain",
201 operation);
202 }
203 const size_t rows = values->size();
204 const size_t cols = rows == 0 ? 0 : (*values)[0].size();
205 if (rows < cols) {
206 return Result<void>::failure(
207 CasErrc::Inconclusive,
208 "QR decomposition requires at least as many rows as columns in the supported domain",
209 operation);
210 }
211 std::vector<std::vector<Rational>> gram(
212 cols, std::vector<Rational>(cols, Rational(0)));
213 for (size_t i = 0; i < cols; ++i) {
214 for (size_t j = 0; j < cols; ++j) {
215 Rational sum(0);
216 for (size_t row = 0; row < rows; ++row) {
217 sum = sum + (*values)[row][i] * (*values)[row][j];
218 }
219 gram[i][j] = sum;
220 }
221 }
222 for (size_t order = 1; order <= cols; ++order) {
223 std::vector<Rational> leading;
224 leading.reserve(order * order);
225 for (size_t row = 0; row < order; ++row) {
226 for (size_t col = 0; col < order; ++col) {
227 leading.push_back(gram[row][col]);
228 }
229 }
230 auto determinant = detail::rational_determinant_exact(
231 order, std::move(leading), context, operation);
232 if (!determinant) return Result<void>::failure(determinant.error());
233 if (determinant.value() <= Rational(0)) {
234 return Result<void>::failure(
235 CasErrc::Inconclusive,
236 "QR decomposition requires proven full column rank",
237 operation);
238 }
239 }
240 return Result<void>::success();
241}

252std::optional<std::vector<Rational>> exact_rectangular_diagonal_entries(
253 const std::shared_ptr<const MatrixNode>& matrix) {
254 auto values = exact_rational_matrix(matrix);
255 if (!values) return std::nullopt;
256 const size_t rows = values->size();
257 const size_t cols = rows == 0 ? 0 : (*values)[0].size();
258 const size_t diag_count = std::min(rows, cols);
259 std::vector<Rational> diagonal(diag_count, Rational(0));
260 for (size_t row = 0; row < rows; ++row) {
261 for (size_t col = 0; col < cols; ++col) {
262 const bool is_diag = row == col && row < diag_count;
263 if (!is_diag && (*values)[row][col] != Rational(0)) {
264 return std::nullopt;
265 }
266 if (is_diag) {
267 diagonal[row] = (*values)[row][col];
268 }
269 }
270 }
271 return diagonal;
272}

274std::optional<std::vector<Rational>> exact_diagonal_entries(
275 const std::shared_ptr<const MatrixNode>& matrix) {
276 auto values = exact_rational_matrix(matrix);
277 if (!values) return std::nullopt;
278 const size_t rows = values->size();
279 const size_t cols = rows == 0 ? 0 : (*values)[0].size();
280 if (rows != cols) return std::nullopt;
281 std::vector<Rational> diagonal(rows, Rational(0));
282 for (size_t row = 0; row < rows; ++row) {
283 for (size_t col = 0; col < cols; ++col) {
284 if (row == col) {
285 diagonal[row] = (*values)[row][col];
286 } else if ((*values)[row][col] != Rational(0)) {
287 return std::nullopt;
288 }
289 }
290 }
291 return diagonal;
292}

320LUDecompositionResult lu_decomposition_checked(
321 const std::shared_ptr<SymbolicExpr>& A,
322 ComputationContext& context)
323{
324 const std::string operation = "lu_decomposition";
325 auto matrix = validate_decomposition_matrix(A, true, context, operation);
326 if (!matrix) return LUDecompositionResult::failure(matrix.error());
327 auto budget = context.consume_steps(matrix.value()->rows() * matrix.value()->cols() * 8 + 8,
328 operation);
329 if (!budget) return LUDecompositionResult::failure(budget.error());
330 auto values = exact_rational_matrix(matrix.value());
331 if (!values) {
332 return LUDecompositionResult::failure(
333 CasErrc::Inconclusive,
334 "exact PLU requires a rational matrix or proved symbolic pivots",
335 operation);
336 }
337 try {
338 const std::size_t n = values->size();
339 auto U_values = *values;
340 std::vector<std::vector<Rational>> L_values(
341 n, std::vector<Rational>(n, Rational(0)));
342 std::vector<std::vector<Rational>> P_values(
343 n, std::vector<Rational>(n, Rational(0)));
344 for (std::size_t i = 0; i < n; ++i) {
345 L_values[i][i] = Rational(1);
346 P_values[i][i] = Rational(1);
347 }
348 for (std::size_t column = 0; column < n; ++column) {
349 std::size_t pivot = column;
350 while (pivot < n && U_values[pivot][column] == Rational(0)) ++pivot;
351 if (pivot == n) {
352 return LUDecompositionResult::failure(
353 CasErrc::DomainError,
354 "PLU decomposition requires a nonsingular matrix",
355 operation);
356 }
357 if (pivot != column) {
358 std::swap(U_values[pivot], U_values[column]);
359 std::swap(P_values[pivot], P_values[column]);
360 for (std::size_t prior = 0; prior < column; ++prior) {
361 std::swap(L_values[pivot][prior], L_values[column][prior]);
362 }
363 }
364 for (std::size_t row = column + 1; row < n; ++row) {
365 const Rational multiplier =
366 U_values[row][column] / U_values[column][column];
367 L_values[row][column] = multiplier;
368 for (std::size_t col = column; col < n; ++col) {
369 U_values[row][col] =
370 U_values[row][col] -
371 multiplier * U_values[column][col];
372 }
373 }
374 }
375 auto to_expression = [&](const std::vector<std::vector<Rational>>& input) {
376 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> grid(
377 n, std::vector<std::shared_ptr<SymbolicExpr>>(n));
378 for (std::size_t row = 0; row < n; ++row) {
379 for (std::size_t col = 0; col < n; ++col) {
380 grid[row][col] = exact_number_expr(input[row][col]);
381 }
382 }
383 return SymbolicExpr::matrix(grid);
384 };
385 return LUDecompositionResult::success(LUDecomposition{
386 to_expression(P_values),
387 to_expression(L_values),
388 to_expression(U_values)});
389 } catch (const std::bad_alloc&) {
390 return LUDecompositionResult::failure(CasErrc::ResourceLimit,
391 "allocation failed while calculating LU decomposition",
392 operation);
393 } catch (const std::exception& ex) {
394 return LUDecompositionResult::failure(CasErrc::InternalInvariant,
395 ex.what(),
396 operation);
397 }
398}

453static bool qr_decomposition_impl(
454 const std::shared_ptr<SymbolicExpr>& A,
455 std::shared_ptr<SymbolicExpr>& Q,
456 std::shared_ptr<SymbolicExpr>& R) {
457 auto mat = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(A));
458 if (!mat) return false;
459 size_t m = mat->rows();
460 size_t n = mat->cols();
461
462 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> Q_grid(m, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
463 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> R_grid(n, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
464
465 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> A_cols(n, std::vector<std::shared_ptr<SymbolicExpr>>(m));
466 for (size_t j = 0; j < n; j++) {
467 for (size_t i = 0; i < m; i++) A_cols[j][i] = lamina::detail::make_expression_ptr(mat->get(i, j));
468 }
469
470 for (size_t j = 0; j < n; j++) {
471 std::vector<std::shared_ptr<SymbolicExpr>> u_j = A_cols[j];
472 for (size_t i = 0; i < j; i++) {
473 auto dot = SymbolicExpr::number(0);
474 for (size_t k = 0; k < m; k++) dot = SymbolicExpr::add(dot, SymbolicExpr::multiply(Q_grid[k][i], A_cols[j][k]));
475 R_grid[i][j] = dot;
476
477 for (size_t k = 0; k < m; k++) {
478 u_j[k] = SymbolicExpr::add(u_j[k], SymbolicExpr::multiply(SymbolicExpr::number(-1), SymbolicExpr::multiply(R_grid[i][j], Q_grid[k][i])));
479 }
480 }
481
482 auto norm_sq = SymbolicExpr::number(0);
483 for (size_t k = 0; k < m; k++) norm_sq = SymbolicExpr::add(norm_sq, SymbolicExpr::multiply(u_j[k], u_j[k]));
484 R_grid[j][j] = SymbolicExpr::sqrt(norm_sq);
485
486 for (size_t k = 0; k < m; k++) Q_grid[k][j] = SymbolicExpr::divide(u_j[k], R_grid[j][j]);
487 }
488
489 Q = SymbolicExpr::matrix(Q_grid);
490 R = SymbolicExpr::matrix(R_grid);
491 return true;
492}

536static bool cholesky_decomposition_impl(
537 const std::shared_ptr<SymbolicExpr>& A,
538 std::shared_ptr<SymbolicExpr>& L) {
539 auto mat = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(A));
540 if (!mat || mat->rows() != mat->cols()) return false;
541 size_t n = mat->rows();
542 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> L_grid(n, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
543 for (size_t i = 0; i < n; i++) {
544 for (size_t j = 0; j <= i; j++) {
545 auto sum = SymbolicExpr::number(0);
546 for (size_t k = 0; k < j; k++) sum = SymbolicExpr::add(sum, SymbolicExpr::multiply(L_grid[i][k], L_grid[j][k]));
547 auto a_ij = lamina::detail::make_expression_ptr(mat->get(i, j));
548 auto diff = SymbolicExpr::add(a_ij, SymbolicExpr::multiply(SymbolicExpr::number(-1), sum));
549 if (i == j) L_grid[i][j] = SymbolicExpr::sqrt(diff);
550 else L_grid[i][j] = SymbolicExpr::divide(diff, L_grid[j][j]);
551 }
552 }
553 L = SymbolicExpr::matrix(L_grid);
554 return true;
555}

557SVDDecompositionResult svd_decomposition_checked(
558 const std::shared_ptr<SymbolicExpr>& A,
559 ComputationContext& context)
560{
561 const std::string operation = "svd_decomposition";
562 auto matrix = validate_decomposition_matrix(A, false, context, operation);
563 if (!matrix) return SVDDecompositionResult::failure(matrix.error());
564 auto budget = context.consume_steps(matrix.value()->rows() * matrix.value()->cols() * 32 + 32,
565 operation);
566 if (!budget) return SVDDecompositionResult::failure(budget.error());
567 auto diagonal = exact_rectangular_diagonal_entries(matrix.value());
568 if (diagonal) {
569 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> signs(
570 matrix.value()->rows(),
571 std::vector<std::shared_ptr<SymbolicExpr>>(
572 matrix.value()->rows(), SymbolicExpr::number(0)));
573 std::vector<Rational> magnitudes = *diagonal;
574 for (std::size_t row = 0; row < matrix.value()->rows(); ++row) {
575 signs[row][row] = SymbolicExpr::number(1);
576 }
577 for (std::size_t index = 0; index < magnitudes.size(); ++index) {
578 if (magnitudes[index] < Rational(0)) {
579 signs[index][index] = SymbolicExpr::number(-1);
580 magnitudes[index] = Rational(0) - magnitudes[index];
581 }
582 }
583 auto U = SymbolicExpr::matrix(signs);
584 auto S = rectangular_diagonal_expr(
585 matrix.value()->rows(), matrix.value()->cols(), magnitudes);
586 auto V = identity_matrix_expr(matrix.value()->cols());
587 return SVDDecompositionResult::success(SVDDecomposition{U, S, V});
588 }
589 return SVDDecompositionResult::failure(
590 CasErrc::Inconclusive,
591 "exact non-diagonal SVD requires a complete proved singular basis",
592 operation);
593}

605 const std::shared_ptr<SymbolicExpr>& A) {
606 auto mat = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(A));
607 if (!mat || mat->rows() != mat->cols()) return SymbolicExpr::exp(A);
608 size_t n = mat->rows();
609
610 bool is_zero_matrix = true;
611 for (size_t r = 0; r < n && is_zero_matrix; ++r) {
612 for (size_t c = 0; c < n; ++c) {
613 auto entry = lamina::detail::make_expression_ptr(mat->get(r, c))->simplify();
614 if (!lamina::detail::node(entry)->is_zero()) {
615 is_zero_matrix = false;
616 break;
617 }
618 }
619 }
620 if (is_zero_matrix) {
621 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> identity(
622 n, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
623 for (size_t i = 0; i < n; ++i) {
624 identity[i][i] = SymbolicExpr::number(1);
625 }
626 return SymbolicExpr::matrix(identity);
627 }
628
629 auto eigenvectors = matrix_eigenvectors_checked(A);
630 auto eigenvalues = matrix_eigenvalues_checked(A);
631 if (!eigenvectors || !eigenvalues) return SymbolicExpr::exp(A);
632 auto P_cols = std::move(eigenvectors.value());
633 auto evals = std::move(eigenvalues.value());
634 if (P_cols.size() != evals.size()) return SymbolicExpr::exp(A);
635
636 if (P_cols.size() != n) return SymbolicExpr::exp(A); // Not diagonalizable
637
638 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> P_grid(n, std::vector<std::shared_ptr<SymbolicExpr>>(n));
639 for (size_t i = 0; i < n; i++) {
640 for (size_t j = 0; j < n; j++) P_grid[i][j] = P_cols[j][i];
641 }
642 auto P = SymbolicExpr::matrix(P_grid);
643 auto inverse_result = matrix_inverse_checked(P);
644 if (!inverse_result) return SymbolicExpr::exp(A);
645 auto P_inv = inverse_result.value();
646
647 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> expD_grid(n, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
648 for (size_t i = 0; i < n; i++) expD_grid[i][i] = SymbolicExpr::exp(evals[i]);
649 auto expD = SymbolicExpr::matrix(expD_grid);
650
651 return SymbolicExpr::multiply(P, SymbolicExpr::multiply(expD, P_inv));
652}

666std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> gram_schmidt(
667 const std::vector<std::vector<std::shared_ptr<SymbolicExpr>>>& vectors,
668 bool normalize) {
669 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> basis;
670 for (const auto& v : vectors) {
671 std::vector<std::shared_ptr<SymbolicExpr>> u = v;
672 for (const auto& b : basis) {
673 /// 投影系数 <v,b>/<b,b>
674 auto vb = SymbolicExpr::number(0);
675 auto bb = SymbolicExpr::number(0);
676 for (size_t k = 0; k < u.size(); ++k) {
677 vb = SymbolicExpr::add(vb, SymbolicExpr::multiply(v[k], b[k]));
678 bb = SymbolicExpr::add(bb, SymbolicExpr::multiply(b[k], b[k]));
679 }
680 if (lamina::detail::node(bb) && lamina::detail::node(bb)->is_zero()) continue;
681 auto coeff = SymbolicExpr::divide(vb, bb);
682 for (size_t k = 0; k < u.size(); ++k) {
683 u[k] = SymbolicExpr::add(u[k],
684 SymbolicExpr::multiply(SymbolicExpr::number(-1),
685 SymbolicExpr::multiply(coeff, b[k])))->simplify();
686 }
687 }
688 /// 检查 u 是否为零向量(线性相关)
689 auto norm_sq = SymbolicExpr::number(0);
690 for (auto& x : u) norm_sq = SymbolicExpr::add(norm_sq, SymbolicExpr::multiply(x, x));
691 norm_sq = norm_sq->simplify();
692 if (lamina::detail::node(norm_sq) && lamina::detail::node(norm_sq)->is_zero()) continue;
693 basis.push_back(u);
694 }
695 if (normalize) {
696 for (auto& u : basis) {
697 auto norm_sq = SymbolicExpr::number(0);
698 for (auto& x : u) norm_sq = SymbolicExpr::add(norm_sq, SymbolicExpr::multiply(x, x));
699 auto norm = SymbolicExpr::sqrt(norm_sq);
700 for (auto& x : u) x = SymbolicExpr::divide(x, norm)->simplify();
701 }
702 }
703 return basis;
704}

708 auto mat = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(A));

768std::shared_ptr<SymbolicExpr> matrix_norm(const std::shared_ptr<SymbolicExpr>& A,
769 const std::string& type) {
770 auto mat = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(A));
771 if (!mat) return nullptr;
772 size_t m = mat->rows(), n = mat->cols();
773
774 auto abs_of = [](const std::shared_ptr<SymbolicExpr>& e) {
775 auto node = lamina::detail::make_node<FunctionNode>(FunctionNode::FuncType::Abs,
776 std::vector<std::shared_ptr<const SymbolicNode>>{lamina::detail::node(e)});
777 return lamina::detail::make_expression_ptr(node);
778 };
779
780 if (type == "frobenius") {
781 auto sum = SymbolicExpr::number(0);
782 for (size_t i = 0; i < m; ++i)
783 for (size_t j = 0; j < n; ++j) {
784 auto e = lamina::detail::make_expression_ptr(mat->get(i, j));
785 sum = SymbolicExpr::add(sum, SymbolicExpr::multiply(e, e));
786 }
787 return SymbolicExpr::sqrt(sum)->simplify();
788 }
789 if (type == "1") {
790 /// 最大列和
791 std::shared_ptr<SymbolicExpr> best = nullptr;
792 std::optional<Rational> best_exact;
793 double best_val = -1;
794 for (size_t j = 0; j < n; ++j) {
795 auto colsum = SymbolicExpr::number(0);
796 std::optional<Rational> exact_sum = Rational(0);
797 for (size_t i = 0; i < m; ++i) {
798 auto entry = lamina::detail::make_expression_ptr(mat->get(i, j));
799 colsum = SymbolicExpr::add(colsum, abs_of(entry));
800 if (exact_sum) {
801 auto exact_entry = exact_rational_expr(entry);
802 if (exact_entry) {
803 *exact_sum = *exact_sum +
804 ((*exact_entry < Rational(0)) ? (Rational(0) - *exact_entry) : *exact_entry);
805 } else {
806 exact_sum.reset();
807 }
808 }
809 }
810 colsum = colsum->simplify();
811 auto exact = exact_sum ? exact_sum : exact_rational_expr(colsum);
812 if (exact && (!best || (best_exact && *exact > *best_exact))) {
813 best = exact_number_expr(*exact);
814 best_exact = *exact;
815 best_val = -1;
816 continue;
817 }
818 if (exact && !best_exact && best) {
819 auto maybe_v = checked_finite_numeric(colsum);
820 if (maybe_v && *maybe_v > best_val) {
821 best = exact_number_expr(*exact);
822 best_val = *maybe_v;
823 }
824 continue;
825 }
826 if (!exact && !best_exact) {
827 auto maybe_v = checked_finite_numeric(colsum);
828 double v = maybe_v ? *maybe_v : -std::numeric_limits<double>::infinity();
829 if (!best || v > best_val) { best = colsum; best_val = v; }
830 }
831 }
832 return best;
833 }
834 if (type == "inf") {
835 /// 最大行和
836 std::shared_ptr<SymbolicExpr> best = nullptr;
837 std::optional<Rational> best_exact;
838 double best_val = -1;
839 for (size_t i = 0; i < m; ++i) {
840 auto rowsum = SymbolicExpr::number(0);
841 std::optional<Rational> exact_sum = Rational(0);
842 for (size_t j = 0; j < n; ++j) {
843 auto entry = lamina::detail::make_expression_ptr(mat->get(i, j));
844 rowsum = SymbolicExpr::add(rowsum, abs_of(entry));
845 if (exact_sum) {
846 auto exact_entry = exact_rational_expr(entry);
847 if (exact_entry) {
848 *exact_sum = *exact_sum +
849 ((*exact_entry < Rational(0)) ? (Rational(0) - *exact_entry) : *exact_entry);
850 } else {
851 exact_sum.reset();
852 }
853 }
854 }
855 rowsum = rowsum->simplify();
856 auto exact = exact_sum ? exact_sum : exact_rational_expr(rowsum);
857 if (exact && (!best || (best_exact && *exact > *best_exact))) {
858 best = exact_number_expr(*exact);
859 best_exact = *exact;
860 best_val = -1;
861 continue;
862 }
863 if (exact && !best_exact && best) {
864 auto maybe_v = checked_finite_numeric(rowsum);
865 if (maybe_v && *maybe_v > best_val) {
866 best = exact_number_expr(*exact);
867 best_val = *maybe_v;
868 }
869 continue;
870 }
871 if (!exact && !best_exact) {
872 auto maybe_v = checked_finite_numeric(rowsum);
873 double v = maybe_v ? *maybe_v : -std::numeric_limits<double>::infinity();
874 if (!best || v > best_val) { best = rowsum; best_val = v; }
875 }
876 }
877 return best;
878 }
879 return nullptr;
880}

883std::shared_ptr<SymbolicExpr> quadratic_form_matrix(
884 const std::shared_ptr<SymbolicExpr>& expr,
885 const std::vector<std::string>& vars) {
886 if (!expr) return nullptr;
887 size_t n = vars.size();
888 auto e = expr->expand();
889 if (!e) e = expr;
890
891 /// A[i][j] = (1/2) partial^2(expr)/partialxᵢpartialxⱼ (对称矩阵,对角为 partial^2/2 即系数本身的一半*2)
892 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> grid(n,
893 std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
894 for (size_t i = 0; i < n; ++i) {
895 auto di = e->differentiate(vars[i]);
896 for (size_t j = 0; j < n; ++j) {
897 auto dij = di->differentiate(vars[j]);
898 /// 对二次型,二阶偏导为常数;A_ij = (1/2)*partial^2/partialxipartialxj
899 grid[i][j] = SymbolicExpr::multiply(SymbolicExpr::number(Rational(1, 2)), dij)->simplify();
900 }
901 }
902 return SymbolicExpr::matrix(grid);
903}

905std::string classify_quadratic_form(const std::shared_ptr<SymbolicExpr>& A) {
906 /// 使用特征多项式的根列表分类二次型,直接依据特征值符号.
907 auto eigenvalues = matrix_eigenvalues_checked(A);
908 if (!eigenvalues) return "unknown";
909 auto evals = std::move(eigenvalues.value());
910 if (evals.empty()) return "unknown";
911
912 bool all_pos = true, all_neg = true, any_zero = false;
913 bool has_pos = false, has_neg = false;
914 const double eps = 1e-9;
915 for (auto& ev : evals) {
916 auto se = ev->simplify();
917 auto maybe_d = checked_finite_numeric(se);
918 if (!maybe_d) return "unknown";
919 double d = *maybe_d;
920 if (d > eps) { all_neg = false; has_pos = true; }
921 else if (d < -eps) { all_pos = false; has_neg = true; }
922 else { any_zero = true; all_pos = false; all_neg = false; }
923 }
924 if (all_pos) return "positive_definite";
925 if (all_neg) return "negative_definite";
926 if (has_pos && has_neg) return "indefinite";
927 if (has_pos && any_zero) return "positive_semidefinite";
928 if (has_neg && any_zero) return "negative_semidefinite";
929 return "indefinite";
930}

933JordanDecompositionResult jordan_form_checked(
934 const std::shared_ptr<SymbolicExpr>& A,
935 ComputationContext& context)
936{
937 const std::string operation = "jordan_form";
938 auto matrix = validate_decomposition_matrix(A, true, context, operation);
939 if (!matrix) return JordanDecompositionResult::failure(matrix.error());
940 auto budget = context.consume_steps(matrix.value()->rows() * matrix.value()->cols() * 48 + 32,
941 operation);
942 if (!budget) return JordanDecompositionResult::failure(budget.error());
943 try {
944 if (auto diagonal = exact_diagonal_entries(matrix.value())) {
945 auto J = rectangular_diagonal_expr(
946 matrix.value()->rows(), matrix.value()->cols(), *diagonal);
947 auto P = identity_matrix_expr(matrix.value()->rows());
948 return JordanDecompositionResult::success(
949 JordanDecomposition{J, P});
950 }
951 auto exact = exact_rational_matrix(matrix.value());
952 if (!exact) {
953 return JordanDecompositionResult::failure(
954 CasErrc::Inconclusive,
955 "exact Jordan form requires rational entries or proved symbolic chains",
956 operation);
957 }
958 if (exact->size() == 2 && (*exact)[0].size() == 2 &&
959 (*exact)[0][0] == (*exact)[1][1] &&
960 (*exact)[1][0] == Rational(0) &&
961 (*exact)[0][1] != Rational(0)) {
962 return JordanDecompositionResult::success(JordanDecomposition{
963 A, identity_matrix_expr(2)});
964 }
965 std::shared_ptr<SymbolicExpr> J;
966 std::shared_ptr<SymbolicExpr> P;
967 if (!jordan_form_impl(A, J, P)) {
968 return JordanDecompositionResult::failure(
969 CasErrc::Inconclusive,
970 "exact rational Jordan chains could not be constructed",
971 operation);
972 }
973 return JordanDecompositionResult::success(
974 JordanDecomposition{std::move(J), std::move(P)});
975 } catch (const std::bad_alloc&) {
976 return JordanDecompositionResult::failure(
977 CasErrc::ResourceLimit,
978 "allocation failed while calculating Jordan form",
979 operation);
980 } catch (const std::exception& ex) {
981 return JordanDecompositionResult::failure(CasErrc::InternalInvariant,
982 ex.what(),
983 operation);
984 }
985}

994static bool jordan_form_impl(const std::shared_ptr<SymbolicExpr>& A,
995 std::shared_ptr<SymbolicExpr>& J, std::shared_ptr<SymbolicExpr>& P) {
996 auto mat = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(A));
997 if (!mat || mat->rows() != mat->cols()) return false;
998 size_t n = mat->rows();
999
1000 bool is_diagonal = true;
1001 for (size_t r = 0; r < n && is_diagonal; ++r) {
1002 for (size_t c = 0; c < n; ++c) {
1003 if (r == c) continue;
1004 auto entry = lamina::detail::make_expression_ptr(mat->get(r, c))->simplify();
1005 if (!lamina::detail::node(entry)->is_zero()) {
1006 is_diagonal = false;
1007 break;
1008 }
1009 }
1010 }
1011 if (is_diagonal) {
1012 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> J_grid(
1013 n, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
1014 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> P_grid(
1015 n, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
1016 for (size_t i = 0; i < n; ++i) {
1017 J_grid[i][i] = lamina::detail::make_expression_ptr(mat->get(i, i))->simplify();
1018 P_grid[i][i] = SymbolicExpr::number(1);
1019 }
1020 J = SymbolicExpr::matrix(J_grid);
1021 P = SymbolicExpr::matrix(P_grid);
1022 return true;
1023 }
1024
1025 auto eigenvectors = matrix_eigenvectors_checked(A);
1026 auto eigenvalues = matrix_eigenvalues_checked(A);
1027 if (!eigenvectors || !eigenvalues) return false;
1028 auto P_cols = std::move(eigenvectors.value());
1029 auto diag = std::move(eigenvalues.value());
1030 if (P_cols.size() != diag.size()) return false;
1031 /// 特征向量数等于 n 时矩阵可对角化,Jordan 型即对角阵;
1032 /// 向量数小于 n 时返回 false,表示广义特征向量链位于当前支持域之外.
1033 if (P_cols.size() != n) return false;
1034
1035 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> P_grid(n,
1036 std::vector<std::shared_ptr<SymbolicExpr>>(n));
1037 for (size_t i = 0; i < n; ++i)
1038 for (size_t j = 0; j < n; ++j) P_grid[i][j] = P_cols[j][i];
1039 P = SymbolicExpr::matrix(P_grid);
1040
1041 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> J_grid(n,
1042 std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
1043 for (size_t i = 0; i < n; ++i) J_grid[i][i] = diag[i];
1044 J = SymbolicExpr::matrix(J_grid);
1045 return true;
1046}
