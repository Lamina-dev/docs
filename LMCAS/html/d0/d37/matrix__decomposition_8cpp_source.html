20NormSum matrix_axis_sum(const MatrixNode& matrix, std::size_t axis, bool columns) {
21 auto sum = SymbolicExpr::number(0);
22 std::optional<Rational> exact_sum = Rational(0);
23 const auto count = columns ? matrix.rows() : matrix.cols();
24 for (std::size_t index = 0; index < count; ++index) {
25 auto entry = detail::make_expression_ptr(columns
26 ? matrix.get(index, axis) : matrix.get(axis, index));
27 sum = SymbolicExpr::add(sum, abs_expression(entry));
28 if (exact_sum) {
29 auto exact_entry = exact_rational_expr(entry);
30 if (exact_entry) {
31 *exact_sum = *exact_sum + ((*exact_entry < Rational(0))
32 ? (Rational(0) - *exact_entry) : *exact_entry);
33 } else {
34 exact_sum.reset();
35 }
36 }
37 }
38 sum = sum->simplify();
39 return {sum, exact_sum ? exact_sum : exact_rational_expr(sum)};
40}

43 std::optional<Rational> maximum_exact;
44 std::vector<std::shared_ptr<const SymbolicNode>> candidates;
45 const auto count = columns ? matrix.cols() : matrix.rows();

47 auto candidate = matrix_axis_sum(matrix, axis, columns);
48 if (candidate.exact) {
49 if (!maximum_exact || *candidate.exact > *maximum_exact)
50 maximum_exact = *candidate.exact;

52 candidates.push_back(detail::node(candidate.expression));
