109 Poly reduce(Poly p, const std::vector<Poly>& G) {
110 Poly r(p.num_vars);
111
112 while (!p.is_zero()) {
113 bool reduced = false;
114 Monomial p_lm = p.lead_monomial();
115
116 for (const auto& g : G) {
117 if (divides_mono(g.lead_monomial(), p_lm)) {
118
119 Term factor;
120 factor.coeff = p.lead_coeff() / g.lead_coeff();
121 factor.mono = div_mono(p_lm, g.lead_monomial());
122
123 Poly term_g = mul_poly_term(g, factor);
124 p = sub_poly(p, term_g);
125 reduced = true;
126 break;
127 }
128 }
129
130 if (!reduced) {
131 Term lt = p.lead_term();
132 r.add_term(lt.mono, lt.coeff);
133
134 p.terms.erase(p.terms.begin());
135 }
136 }
137 return r;
