21void PrintVisitor::visit(const AddNode& node) {
22 if (node.operands().empty()) {
23 buffer << "0";
24 return;
25 }
26 for (size_t i = 0; i < node.operands().size(); ++i) {
27 if (i > 0) {
28 std::ostringstream sub;
29 PrintVisitor sub_v;
30 node.operands()[i]->accept(sub_v);
31 std::string s = sub_v.get_result();
32 if (!s.empty() && s[0] == '-') {
33 buffer << " - " << s.substr(1);
34 } else {
35 buffer << " + " << s;
36 }
37 } else {
38 node.operands()[i]->accept(*this);
39 }
40 }

43void PrintVisitor::visit(const MultiplyNode& node) {
44 if (node.operands().empty()) {
45 buffer << "1";
46 return;
47 }
48 for (size_t i = 0; i < node.operands().size(); ++i) {
49 if (i > 0) buffer << "*";
50
51 bool needs_parens = std::dynamic_pointer_cast<const AddNode>(node.operands()[i]) != nullptr ||
52 std::dynamic_pointer_cast<const PowerNode>(node.operands()[i]) != nullptr;
53
54 if (!needs_parens) {
55 if (auto num = std::dynamic_pointer_cast<const NumberNode>(node.operands()[i])) {
56 if (std::holds_alternative<Rational>(num->value())) {
57 needs_parens = true;
58 }
59 }
60 }
61
62 if (needs_parens) buffer << "(";
63 node.operands()[i]->accept(*this);
64 if (needs_parens) buffer << ")";
65 }

68void PrintVisitor::visit(const PowerNode& node) {
69 bool base_parens = dynamic_cast<const AddNode*>(node.base().get()) ||
70 dynamic_cast<const MultiplyNode*>(node.base().get()) ||
71 dynamic_cast<const PowerNode*>(node.base().get());
72 if (const auto* number = dynamic_cast<const NumberNode*>(node.base().get())) {
73 const auto& value = number->value();
74 base_parens = std::holds_alternative<Rational>(value) ||
75 (std::holds_alternative<BigInt>(value) &&
76 std::get<BigInt>(value).IsNegative()) ||
77 (std::holds_alternative<double>(value) &&
78 std::signbit(std::get<double>(value)));
79 }
80 if (base_parens) buffer << "(";
81 node.base()->accept(*this);
82 if (base_parens) buffer << ")";
83 buffer << "^";
84 bool exp_parens = dynamic_cast<const AddNode*>(node.exponent().get()) ||
85 dynamic_cast<const MultiplyNode*>(node.exponent().get()) ||
86 dynamic_cast<const PowerNode*>(node.exponent().get());
87 if (const auto* number = dynamic_cast<const NumberNode*>(node.exponent().get())) {
88 const auto* rational = std::get_if<Rational>(&number->value());
89 exp_parens = rational && !rational->is_integer();
90 }
91 if (exp_parens) buffer << "(";
92 node.exponent()->accept(*this);
93 if (exp_parens) buffer << ")";
94}
