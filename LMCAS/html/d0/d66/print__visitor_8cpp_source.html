10PrintVisitor::Precedence PrintVisitor::precedence(const SymbolicNode& node) {
11 if (dynamic_cast<const AddNode*>(&node)) { return Precedence::Add; }
12 if (dynamic_cast<const MultiplyNode*>(&node)) { return Precedence::Multiply; }
13 if (dynamic_cast<const PowerNode*>(&node)) { return Precedence::Power; }
14 if (dynamic_cast<const MembershipNode*>(&node)) { return Precedence::Membership; }
15 if (const auto* relation = dynamic_cast<const RelationalNode*>(&node)) {
16 return relation->op() == RelationOp::EQ || relation->op() == RelationOp::NEQ
17 ? Precedence::Equality : Precedence::Relation;
18 }
19 if (const auto* number = dynamic_cast<const NumberNode*>(&node)) {
20 if (const auto* rational = std::get_if<Rational>(&number->value())) {
21 if (!rational->is_integer()) { return Precedence::Multiply; }
22 if (*rational < Rational(0)) { return Precedence::Unary; }
23 }
24 if (const auto* integer = std::get_if<BigInt>(&number->value())) {
25 if (integer->is_negative()) { return Precedence::Unary; }
26 }
27 }
28 return Precedence::Atom;
29}

31void PrintVisitor::print_operand(const SymbolicNode& node, Precedence parent,
32 bool parenthesize_equal) {
33 const auto child = precedence(node);
34 const bool parentheses = child < parent ||
35 (parenthesize_equal && child == parent) ||
36 (parent == Precedence::Power && parenthesize_equal && child == Precedence::Unary);
37 if (parentheses) { buffer << "("; }
38 node.accept(*this);
39 if (parentheses) { buffer << ")"; }
40}

42void PrintVisitor::visit(const NumberNode& node) {
43 if (std::holds_alternative<BigInt>(node.value())) {
44 buffer << std::get<BigInt>(node.value()).to_string();
45 } else if (std::holds_alternative<Rational>(node.value())) {
46 buffer << std::get<Rational>(node.value()).to_string();
47 } else {
48 const double value = std::get<double>(node.value());
49 if (!std::isfinite(value)) {
50 buffer << value;
51 return;
52 }
53 char token[64];
54 const auto result = std::to_chars(token, token + sizeof(token), value,
55 std::chars_format::general, std::numeric_limits<double>::max_digits10);
56 if (result.ec != std::errc{}) {
57 throw std::logic_error("finite double formatting failed");
58 }
59 buffer << "approx(";
60 if (value == 0 && std::signbit(value)) buffer << "-0";
61 else buffer.write(token, result.ptr - token);
62 buffer << ")";
63 }
64}
