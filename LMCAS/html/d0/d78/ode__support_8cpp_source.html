20 * @internal
21 * @brief 尝试将表达式求值为 double。
22 * @return 若表达式为纯数值返回其值，否则返回 NaN。
23 */
24double try_eval_double(const std::shared_ptr<SymbolicExpr>& expr) {
25 if (!expr || !LMCAS::detail::node(expr)) {
26 return std::numeric_limits<double>::quiet_NaN();
27 }
28 if (expr->is_number()) {
29 auto node = std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(expr));
30 if (!node) {
31 return std::numeric_limits<double>::quiet_NaN();
32 }
33 if (std::holds_alternative<BigInt>(node->value())) {
34 return std::get<BigInt>(node->value()).to_double();
35 }
36 if (std::holds_alternative<Rational>(node->value())) {
37 return std::get<Rational>(node->value()).to_double();
38 }
39 return static_cast<double>(std::get<lmmc_real_t>(node->value()));
40 }
41 return std::numeric_limits<double>::quiet_NaN();
42}

145Result<void> validate_numeric_ode_coefficients(
146 const std::vector<double>& coeffs,
147 std::size_t min_size,
148 std::size_t max_size,
149 const std::string& operation)
150{
151 if (coeffs.size() < min_size || coeffs.size() > max_size) {
152 return Result<void>::failure(CasErrc::InvalidArgument,
153 "ODE coefficient list has unsupported size",
154 operation);
155 }
156 if (!std::isfinite(coeffs.front()) || coeffs.front() == 0.0) {
157 return Result<void>::failure(CasErrc::InvalidArgument,
158 "ODE leading coefficient must be finite and nonzero",
159 operation);
160 }
161 for (double coeff : coeffs) {
162 if (!std::isfinite(coeff)) {
163 return Result<void>::failure(CasErrc::InvalidArgument,
164 "ODE coefficients must be finite",
165 operation);
166 }
167 }
168 return Result<void>::success();
169}
