5std::optional<std::shared_ptr<const SymbolicNode>> LimitVisitor::exact_ratio_at_point(
6 const std::shared_ptr<const SymbolicNode>& num,
7 const std::shared_ptr<const SymbolicNode>& den) {
8 auto nl = exact_leading_at_point(num);
9 auto dl = exact_leading_at_point(den);
10 if (nl && dl && dl->first != Rational(0)) {
11 if (nl->first == Rational(0) || nl->second > dl->second)
12 { return detail::make_node<NumberNode>(BigInt(0)); }
13 Rational coefficient = nl->first / dl->first;
14 if (nl->second == dl->second) { return detail::make_node<NumberNode>(coefficient); }
15 if (direction.empty() && (dl->second - nl->second) % 2) { return std::shared_ptr<const SymbolicNode>{}; }
16 int sign = coefficient > Rational(0) ? 1 : -1;
17 if (direction == "-" && (dl->second - nl->second) % 2) { sign = -sign; }
18 return signed_infinity(sign);
19 }
20 return std::nullopt;
21}
