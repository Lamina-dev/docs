53ExprPtr canonicalize_power(const PowerNode& power,
54 const detail::SymbolicNodePtr& original,
55 detail::RewriteBudget& budget) {
56 auto exponent = exact_small_integer_node(power.exponent(), 0, 16);
57 if (exponent) {
58 auto base = canonicalize_complex_product(
59 *detail::make_expression_ptr(power.base()), budget);
60 if (std::dynamic_pointer_cast<const ComplexNode>(detail::node(base))) {
61 budget.require_nodes(1);

81 void accumulate(const ExprPtr& factor, detail::RewriteBudget& budget) {
82 ExprPtr factor_real;
83 ExprPtr factor_imag;
84 if (auto complex = std::dynamic_pointer_cast<const ComplexNode>(
85 detail::node(factor))) {
86 saw_complex = true;
87 factor_real = detail::make_expression_ptr(complex->real());
88 factor_imag = detail::make_expression_ptr(complex->imag());
89 } else {
90 factor_real = factor;
91 budget.require_nodes(1);
92 factor_imag = SymbolicExpr::number(0);
93 }
94 const auto ac_size = normalization_check_arithmetic<MultiplyNode>(
95 &budget, 0, detail::node(real), detail::node(factor_real));
96 auto ac = SymbolicExpr::multiply(real, factor_real);
97 const auto bd_size = normalization_check_arithmetic<MultiplyNode>(
98 &budget, 0, detail::node(imag), detail::node(factor_imag));
99 auto bd = SymbolicExpr::multiply(imag, factor_imag);
100 const auto ad_size = normalization_check_arithmetic<MultiplyNode>(
101 &budget, 0, detail::node(real), detail::node(factor_imag));
102 auto ad = SymbolicExpr::multiply(real, factor_imag);
103 const auto bc_size = normalization_check_arithmetic<MultiplyNode>(
104 &budget, 0, detail::node(imag), detail::node(factor_real));
105 auto bc = SymbolicExpr::multiply(imag, factor_real);
106 const auto negative_size = budget.append_size(1, bd_size);
107 budget.append_size(budget.append_size(1, ac_size), negative_size);
108 auto next_real = normalize_equivalence(SymbolicExpr::add(
109 ac, SymbolicExpr::multiply(SymbolicExpr::number(-1), bd)), budget);
110 budget.append_size(budget.append_size(1, ad_size), bc_size);
111 auto next_imag = normalize_equivalence(SymbolicExpr::add(ad, bc), budget);
112 real = std::move(next_real);
113 imag = std::move(next_imag);
114 }

117ExprPtr canonicalize_product(const MultiplyNode& multiply,
118 const detail::SymbolicNodePtr& original,
119 detail::RewriteBudget& budget) {
120 std::vector<ExprPtr> factors;
121 bool changed = false;
122 bool saw_complex = false;
123 factors.reserve(multiply.operands().size());
124 for (const auto& operand : multiply.operands()) {
125 auto canonical = canonicalize_complex_product(
126 *detail::make_expression_ptr(operand), budget);
127 changed = changed || detail::node(canonical) != operand;
128 saw_complex = saw_complex ||
129 static_cast<bool>(std::dynamic_pointer_cast<const ComplexNode>(
130 detail::node(canonical)));
131 factors.push_back(std::move(canonical));
132 }
133 if (!changed && !saw_complex) {
134 budget.measure(original);
135 return detail::make_expression_ptr(original);
136 }
137 if (!saw_complex) {
138 std::vector<detail::SymbolicNodePtr> operands;
139 std::size_t size = 0;
140 operands.reserve(factors.size());
141 for (const auto& factor : factors) {
142 size = budget.append_size(size, budget.measure(detail::node(factor)));
143 operands.push_back(detail::node(factor));
144 }
145 return normalize_equivalence(detail::make_expression_ptr(
146 SymbolicFactory::create_multiply(std::move(operands), &budget)), budget);
147 }
148 ComplexProduct product;
149 for (const auto& factor : factors) {
150 product.accumulate(factor, budget);
151 }
152 return rewritten_complex(product.real, product.imag, original, budget);
153}

157ExprPtr canonicalize_complex_product(const SymbolicExpr& expression,
158 detail::RewriteBudget& budget) {
159 detail::RewriteScope scope(&budget);
160 const auto& node = detail::node(expression);
161 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
162 return canonicalize_sum(*add, node, budget);
163 }
164 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
165 return canonicalize_power(*power, node, budget);
166 }
167 if (auto complex = std::dynamic_pointer_cast<const ComplexNode>(node)) {
168 auto real = canonicalize_complex_product(
169 *detail::make_expression_ptr(complex->real()), budget);
170 auto imag = canonicalize_complex_product(
171 *detail::make_expression_ptr(complex->imag()), budget);
172 return rewritten_complex(real, imag, node, budget);
173 }
174 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
175 return canonicalize_product(*multiply, node, budget);
176 }
177 budget.measure(node);
178 return detail::make_expression_ptr(node);
179}
