22 std::shared_ptr<const SymbolicNode> try_match(std::shared_ptr<const SymbolicNode> node) {
23 auto current_expr = LMCAS::detail::expression_from_node(node);
24 const auto& rules = engine.get_rules();
25 const AssumptionContext* ctx = context.assumptions().get();
26 for (const auto& rule : rules) {
27 MatchMap bindings;
28
29 if (Matcher::match(rule.pattern, current_expr, rule.wildcards, bindings)) {
30
31 if (rule.condition && !rule.condition(bindings)) {
32 continue;
33 }
34 if (rule.assumption_condition &&
35 (!ctx || !rule.assumption_condition(bindings, ctx))) {
36 continue;
37 }
38
39 SymbolicExpr new_expr = Matcher::replace(rule.replacement, bindings, true);
40
41 changed = true;
42 return LMCAS::detail::node(new_expr);
43 }
44 }
45 return node;
46 }

388Result<SymbolicExpr> RewriteEngine::apply_step_checked(
389 const SymbolicExpr& expr,
390 ComputationContext& context) const {
391 auto budget = context.consume_steps(1, "rewrite.apply_step");
392 if (!budget) return Result<SymbolicExpr>::failure(budget.error());
393 if (!LMCAS::detail::node(expr)) {
394 return Result<SymbolicExpr>::failure(
395 CasErrc::InvalidArgument, "rewrite expression must not be null",
396 "rewrite.apply_step");
397 }
398 RewriteVisitor v(*this, context);
399 LMCAS::detail::node(expr)->accept(v);
400 return Result<SymbolicExpr>::success(
401 LMCAS::detail::expression_from_node(v.get_result()));
402}

404Result<SymbolicExpr> RewriteEngine::apply_checked(
405 const SymbolicExpr& expr,
406 ComputationContext& context,
407 int max_iterations) const {
408 if (max_iterations < 0) {
409 return Result<SymbolicExpr>::failure(
410 CasErrc::InvalidArgument, "rewrite iteration count must be non-negative",
411 "rewrite.apply");
412 }
413 SymbolicExpr current = expr;
414 for (int i = 0; i < max_iterations; ++i) {
415 auto next_result = apply_step_checked(current, context);
416 if (!next_result) return next_result;
417 SymbolicExpr next = std::move(next_result.value());
418 if (LMCAS::detail::node(next)->equals(*LMCAS::detail::node(current))) {
419 return Result<SymbolicExpr>::success(std::move(current));
420 }
421 current = next;
422 }
423 return Result<SymbolicExpr>::success(std::move(current));
424}
425
426Result<SymbolicExpr> RewriteEngine::apply_step(const SymbolicExpr& expr) const {
427 ComputationContext context;
428 return apply_step_checked(expr, context);
429}
