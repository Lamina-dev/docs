72/**
73 * @brief 二因子二次 Hensel 提升的状态结构.
74 *
75 * 存储一对因子 g, h 及其 Bezout 系数 s, t,满足:
76 * - f == g * h (mod modulus)
77 * - s * g + t * h == 1 (mod modulus)
78 *
79 * 系数按升幂存储:coeffs[i] 对应 x^i 的系数.
80 */
81struct HenselLiftPair {
82 std::vector<BigInt> g; /**< 第一个因子的系数。 */
83 std::vector<BigInt> h; /**< 第二个因子的系数。 */
84 std::vector<BigInt> s; /**< g 的 Bezout 系数。 */
85 std::vector<BigInt> t; /**< h 的 Bezout 系数。 */
86 BigInt modulus; /**< 当前模数。 */
87};

89/**
90 * @brief 执行一步二次 Hensel 提升:mod m -> mod m^2.
91 *
92 * 给定 f == g*h (mod m) 且 s*g + t*h == 1 (mod m),
93 * 计算 g', h', s', t' 使得 f == g'*h' (mod m^2) 且 s'*g' + t'*h' == 1 (mod m^2).
94 *
95 * @param[in] f 原始多项式系数向量(升幂排列)
96 * @param[in] current 当前提升状态(g, h, s, t, modulus=m)
97 * @return 提升后的状态(g', h', s', t', modulus=m^2)
98 *
99 * @pre f == g*h (mod m)
100 * @pre s*g + t*h == 1 (mod m)
101 * @pre deg(s) < deg(h), deg(t) < deg(g)
102 *
103 * @see Zassenhaus, H. "On Hensel factorization, I."
104 * Journal of Number Theory, 1(3), 1969.
105 */
106LMCAS_API HenselLiftPair hl_two_factor_lift(
107 const std::vector<BigInt>& f,
108 const HenselLiftPair& current);
