5static bool validate_integration_steps(const std::vector<IntegrationStep>& steps) {
6 if (steps.empty()) return false;
7
8 for (size_t i = 0; i < steps.size(); ++i) {
9 const auto& s = steps[i];
10 if (s.variable.empty()) return false;
11
12 // Bounds must be either both null (indefinite) or both set (definite).
13 const bool has_lower = static_cast<bool>(s.lower);
14 const bool has_upper = static_cast<bool>(s.upper);
15 if (has_lower != has_upper) return false;
16
17 // Reject duplicate variables.
18 for (size_t j = 0; j < i; ++j) {
19 if (steps[j].variable == s.variable) return false;
20 }
21 }
22 return true;
23}
