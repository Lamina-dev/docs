77 void FGLMPoly::normalize() {
78
79 std::vector<std::pair<Monomial, Rational>> cleaned;
80 for (auto& [m, c] : terms) {
81 if (c.is_zero()) continue;
82 if (!cleaned.empty() && cleaned.back().first == m) {
83 cleaned.back().second = cleaned.back().second + c;
84 if (cleaned.back().second.is_zero()) {
85 cleaned.pop_back();
86 }
87 } else {
88 cleaned.emplace_back(m, c);
89 }
90 }

94 FGLMPoly FGLMPoly::reduce(const std::vector<FGLMPoly>& basis,
95 const MonomialOrder& order) const {
96 FGLMPoly remainder(num_vars);
97 FGLMPoly polynomial = *this;
98 polynomial.sort_terms(order);
99 polynomial.normalize();
100
101 while (!polynomial.is_zero()) {
102 const Monomial previous_leading = polynomial.lead_monomial();
103 if (!reduce_leading_term(
104 polynomial, basis, previous_leading, order)) {
105 remainder.add_term(
106 previous_leading, polynomial.lead_coeff());
107 polynomial.terms.erase(polynomial.terms.begin());
108 polynomial.normalize();
109 }
110 if (!polynomial.is_zero() &&
111 !order(previous_leading, polynomial.lead_monomial())) {
112 throw std::runtime_error(
113 "FGLM polynomial reduction did not decrease the leading monomial");
114 }
115 }
116 remainder.sort_terms(order);
117 remainder.normalize();
118 return remainder;
119 }
