13Result<detail::SymbolicNodePtr> select_continuity_branch(
14 const PiecewiseNode& piecewise, const FactsQuery& facts,
15 const AssumptionContext& assumptions, ComputationContext& context) {
16 for (const auto& branch : piecewise.branches()) {
17 auto access = context.consume_steps(1, "continuity_at");
18 if (!access) { return Result<detail::SymbolicNodePtr>::failure(access.error()); }
19 auto defined = detail::query_definedness(branch.condition, facts, Domain::Real, context);
20 if (!defined) { return Result<detail::SymbolicNodePtr>::failure(defined.error()); }
21 const auto truth = defined.value() == Tribool::True
22 ? assumptions.evaluate_condition(detail::expression_from_node(branch.condition))
23 : Tribool::Unknown;
24 if (truth == Tribool::Unknown) {
25 return Result<detail::SymbolicNodePtr>::failure(CasErrc::Inconclusive,
26 "point-value branch is undecided", "continuity_at");
27 }
28 if (truth == Tribool::True) { return branch.expression; }
29 }
30 return piecewise.default_expr();
31}

33Result<ExprPtr> continuity_point_value(
34 ExprPtr value, const FactsQuery& facts, ComputationContext& context) {
35 const AssumptionContext empty_assumptions;

46ContinuityResult classify_continuity_point_value(
47 const ExprPtr& limit, const ExprPtr& value,
48 const FactsQuery& facts, ComputationContext& context) {
49 auto defined = detail::query_definedness(detail::node(value), facts, Domain::Real, context);
50 if (!defined) { return ContinuityResult::failure(defined.error()); }
51 if (defined.value() == Tribool::False) {
52 return ContinuityResult::success(ContinuityType::Removable);
53 }
54 if (defined.value() != Tribool::True) {
55 return ContinuityResult::failure(CasErrc::Inconclusive,
56 "point-value definedness is unknown", "continuity_at");
57 }
58 auto same = detail::compare_pointwise_values(limit, value, context, Domain::Real);
59 if (!same) { return ContinuityResult::failure(same.error()); }
60 if (same.value() == Tribool::True) {
61 return ContinuityResult::success(ContinuityType::Continuous);
62 }
63 if (same.value() == Tribool::False) {
64 return ContinuityResult::success(ContinuityType::Removable);
65 }
66 return ContinuityResult::failure(CasErrc::Inconclusive,
67 "point value could not be compared with the limit", "continuity_at");
68}

70ContinuityResult continuity_with_limit(
71 const ExprPtr& f, const std::string& var, const ExprPtr& point,
72 const ExprPtr& limit, ComputationContext& context) {
73 auto value = f->substitute(var, point);
74 if (!value) {
75 return ContinuityResult::failure(CasErrc::InternalInvariant,
76 "substitution produced no value", "continuity_at");
77 }
78 std::optional<detail::AssumptionFacts> assumed;
79 if (context.assumptions()) { assumed.emplace(*context.assumptions()); }
80 const auto& facts = assumed ? static_cast<const FactsQuery&>(*assumed) : detail::no_facts();
81 auto selected = continuity_point_value(value, facts, context);
82 if (!selected) { return ContinuityResult::failure(selected.error()); }
83 if (!selected.value()) { return ContinuityResult::success(ContinuityType::Removable); }
84 return classify_continuity_point_value(limit, selected.value(), facts, context);
85}

87ContinuityResult continuity_from_limits(
88 const ExprPtr& f, const std::string& var, const ExprPtr& point,
89 const LimitOutcome& left, const LimitOutcome& right, ComputationContext& context) {
90 const auto* left_finite = std::get_if<FiniteLimit>(&left);
91 const auto* right_finite = std::get_if<FiniteLimit>(&right);
92 if (!left_finite || !right_finite) {
93 return ContinuityResult::success(ContinuityType::Essential);
94 }
95 auto sides = detail::compare_pointwise_values(
96 left_finite->value, right_finite->value, context, Domain::Real);
97 if (!sides) { return ContinuityResult::failure(sides.error()); }
98 if (sides.value() == Tribool::False) {
99 return ContinuityResult::success(ContinuityType::Jump);
100 }
101 if (sides.value() != Tribool::True) {
102 return ContinuityResult::failure(CasErrc::Inconclusive,
103 "one-sided values could not be compared", "continuity_at");
104 }
105 return continuity_with_limit(f, var, point, left_finite->value, context);
106}

111 const std::shared_ptr<SymbolicExpr>& point, ComputationContext& context) {
112 const std::string operation = "continuity_at";

117 }
118 auto input = calculus_utils_validate_expr(f, var, context, operation);
119 if (!input) { return ContinuityResult::failure(input.error()); }

121 if (!left) { return ContinuityResult::failure(left.error()); }
122 auto right = limit_checked(f, var, point, LimitDirection::FromAbove, context);
123 if (!right) { return ContinuityResult::failure(right.error()); }
124 return continuity_from_limits(f, var, point, left.value().value, right.value().value, context);
125 } catch (const std::bad_alloc&) {
126 return ContinuityResult::failure(CasErrc::ResourceLimit, "continuity analysis allocation failed", operation);
127 } catch (const std::exception& ex) {
128 return ContinuityResult::failure(CasErrc::InternalInvariant, ex.what(), operation);
129 }
