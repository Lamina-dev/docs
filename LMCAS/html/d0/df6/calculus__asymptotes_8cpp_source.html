12static std::shared_ptr<const SymbolicNode> denominator_factor(
13 const std::shared_ptr<const SymbolicNode>& node)
14{
15 auto power = std::dynamic_pointer_cast<const PowerNode>(node);
16 if (!power) {
17 return nullptr;
18 }
19 auto number = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
20 if (!number) {
21 return nullptr;
22 }
23 double exponent = 0;
24 if (std::holds_alternative<lmmc_real_t>(number->value())) {
25 exponent = std::get<lmmc_real_t>(number->value());
26 } else if (std::holds_alternative<BigInt>(number->value())) {
27 exponent = std::get<BigInt>(number->value()).to_double();
28 } else if (std::holds_alternative<Rational>(number->value())) {
29 exponent = std::get<Rational>(number->value()).to_double();
30 }
31 if (!(exponent < 0)) {
32 return nullptr;
33 }
34 if (exponent == -1.0) {
35 return power->base();
36 }
37 auto positive = LMCAS::detail::make_node<NumberNode>(BigInt(static_cast<int>(-exponent)));
38 return LMCAS::detail::make_node<PowerNode>(power->base(), positive);
39}

41static std::shared_ptr<SymbolicExpr> calculus_utils_extract_denominator(
42 const std::shared_ptr<SymbolicExpr>& expr)
43{
44 if (!expr || !LMCAS::detail::node(expr)) {
45 return nullptr;
46 }
47 auto factor = denominator_factor(LMCAS::detail::node(expr));
48 if (factor) {
49 return LMCAS::detail::make_expression_ptr(factor);
50 }
51 auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(LMCAS::detail::node(expr));
52 if (!multiply) {
53 return nullptr;
54 }
55 std::vector<std::shared_ptr<const SymbolicNode>> factors;
56 for (const auto& operand : multiply->operands()) {
57 auto denominator = denominator_factor(operand);
58 if (denominator) factors.push_back(std::move(denominator));
59 }
60 if (factors.empty()) {
61 return nullptr;
62 }
63 if (factors.size() == 1) {
64 return LMCAS::detail::make_expression_ptr(factors[0]);
65 }
66 return LMCAS::detail::make_expression_ptr(
67 LMCAS::detail::make_node<MultiplyNode>(factors));
68}

82static bool calculus_utils_is_minus_one(const std::shared_ptr<const NumberNode>& number)
83{
84 if (!number) {
85 return false;
86 }
87 if (std::holds_alternative<BigInt>(number->value())) {
88 return std::get<BigInt>(number->value()) == BigInt(-1);
89 }
90 if (std::holds_alternative<Rational>(number->value())) {
91 return std::get<Rational>(number->value()) == Rational(-1);
92 }
93 return std::get<lmmc_real_t>(number->value()) == -1.0;

96static bool calculus_utils_split_rational(
97 const std::shared_ptr<SymbolicExpr>& expr,
98 std::shared_ptr<SymbolicExpr>& numerator,
99 std::shared_ptr<SymbolicExpr>& denominator)
100{
101 if (!expr || !LMCAS::detail::node(expr)) {
102 return false;
103 }
104
105 std::vector<std::shared_ptr<const SymbolicNode>> num_factors;
106 std::vector<std::shared_ptr<const SymbolicNode>> den_factors;
107
108 auto collect_factor = [&](const std::shared_ptr<const SymbolicNode>& factor) {
109 if (auto power = std::dynamic_pointer_cast<const PowerNode>(factor)) {
110 auto exponent = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
111 if (calculus_utils_is_minus_one(exponent)) {
112 den_factors.push_back(power->base());
113 return;
114 }
115 }
116 num_factors.push_back(factor);
117 };
118
119 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(LMCAS::detail::node(expr))) {
120 for (const auto& factor : multiply->operands()) collect_factor(factor);
121 } else {
122 collect_factor(LMCAS::detail::node(expr));
123 }
124
125 if (den_factors.empty()) {
126 return false;
127 }
128 numerator = calculus_utils_product(num_factors);
129 denominator = calculus_utils_product(den_factors);
130 return numerator && denominator;
131}

133static bool calculus_utils_oblique_from_rational(
134 const std::shared_ptr<SymbolicExpr>& f,
135 const std::string& var,
136 std::shared_ptr<SymbolicExpr>& slope,
137 std::shared_ptr<SymbolicExpr>& intercept)
138{
139 std::shared_ptr<SymbolicExpr> numerator;
140 std::shared_ptr<SymbolicExpr> denominator;
141 if (!calculus_utils_split_rational(f, numerator, denominator)) {
142 return false;
143 }
144
145 try {
146 auto num_poly = symbolic_to_poly<Rational>(numerator->expand()->simplify(), var);
147 auto den_poly = symbolic_to_poly<Rational>(denominator->expand()->simplify(), var);
148 if (!num_poly || !den_poly || den_poly.value().is_zero()) {
149 return false;
150 }
151 auto [quotient, remainder] = num_poly.value().div_mod(den_poly.value());
152 (void)remainder;
153 if (quotient.degree() != 1 || quotient.coeffs.size() < 2) {
154 return false;
155 }
156 intercept = SymbolicExpr::number(quotient.coeffs[0])->simplify();
157 slope = SymbolicExpr::number(quotient.coeffs[1])->simplify();
158 return !slope->is_zero();
159 } catch (const std::invalid_argument&) {
160 return false;
161 } catch (const std::out_of_range&) {
162 return false;
163 }

165static SymbolicExprVectorResult asymptote_candidates(
166 const std::shared_ptr<SymbolicExpr>& denominator,
167 const std::string& var, ComputationContext& context,
168 const std::string& operation)
169{
170 auto solved = solve_equation(denominator, var, context, SolveOptions{});
171 if (!solved) {
172 return SymbolicExprVectorResult::failure(solved.error());
173 }
174 const auto& zero_set = solved.value();
175 const auto* finite = std::get_if<FiniteSolutions>(&zero_set);
176 if (!std::holds_alternative<EmptySolutions>(zero_set) && !finite) {
177 return SymbolicExprVectorResult::failure(
178 CasErrc::Inconclusive,
179 "asymptote denominator zeros are outside the finite exact support domain",
180 operation);
181 }
182 std::vector<std::shared_ptr<SymbolicExpr>> zeros;
183 if (!finite) {
184 return SymbolicExprVectorResult::success(std::move(zeros));
185 }
186 zeros.reserve(finite->values.size());
187 for (const auto& solution : finite->values) {
188 if (!solution.value || !LMCAS::detail::node(solution.value)) {
189 return SymbolicExprVectorResult::failure(
190 CasErrc::InternalInvariant,
191 "checked solver returned a null vertical-asymptote candidate", operation);
192 }
193 auto candidate = rootof_simplify(solution.value);
194 if (!candidate || !LMCAS::detail::node(candidate)) {
195 return SymbolicExprVectorResult::failure(
196 CasErrc::InternalInvariant,
197 "checked solver returned a malformed vertical-asymptote candidate", operation);
198 }
199 if (LMCAS::detail::contains_node_type<RootOfNode>(LMCAS::detail::node(candidate))) {
200 return SymbolicExprVectorResult::failure(
201 CasErrc::Inconclusive,
202 "vertical-asymptote candidate could not be reduced to an exact explicit point",
203 operation);
204 }
205 zeros.push_back(candidate);
206 }
207 return SymbolicExprVectorResult::success(std::move(zeros));
208}

210static Result<bool> is_vertical_asymptote(
211 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
212 const std::shared_ptr<SymbolicExpr>& point, ComputationContext& context)
213{
214 auto right = limit_checked(f, var, point, LimitDirection::FromAbove, context);
215 if (!right && right.error().code != CasErrc::Inconclusive) {
216 return Result<bool>::failure(right.error());
217 }
218 auto left = limit_checked(f, var, point, LimitDirection::FromBelow, context);
219 if (!left && left.error().code != CasErrc::Inconclusive) {
220 return Result<bool>::failure(left.error());
221 }
222 auto infinite = [](const LimitResult& value) {
223 return value && (std::holds_alternative<PositiveInfinityLimit>(value.value().value) ||
224 std::holds_alternative<NegativeInfinityLimit>(value.value().value));
225 };
226 if (infinite(right) || infinite(left)) { return true; }
227 if (!right) { return Result<bool>::failure(right.error()); }
228 if (!left) { return Result<bool>::failure(left.error()); }
229 return false;
230}

232static Result<void> collect_vertical_asymptotes(
233 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
234 ComputationContext& context, AsymptoteResult& result,
235 const std::string& operation)
236{
237 auto denominator = calculus_utils_extract_denominator(f);
238 if (!denominator) {
239 return Result<void>::success();
240 }
241 auto zeros = asymptote_candidates(denominator, var, context, operation);
242 if (!zeros) {
243 return Result<void>::failure(zeros.error());
244 }
245 for (const auto& point : zeros.value()) {
246 if (!point || calculus_utils_is_infinity(point)) {
247 continue;
248 }
249 auto vertical = is_vertical_asymptote(f, var, point, context);
250 if (!vertical) {
251 return Result<void>::failure(vertical.error());
252 }
253 if (vertical.value()) result.vertical.push_back(point);
254 }
255 return Result<void>::success();
256}

271static Result<void> collect_oblique_asymptote(
272 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
273 const std::shared_ptr<SymbolicExpr>& x_expr,
274 const std::shared_ptr<SymbolicExpr>& infinity, ComputationContext& context,
275 AsymptoteResult& result)
276{
277 auto quotient = SymbolicExpr::multiply(f,
278 SymbolicExpr::power(x_expr, SymbolicExpr::number(-1)));
279 auto slope_result = limit_expression_checked(
280 quotient, var, infinity, LimitDirection::Both, context);
281 if (!slope_result) {
282 return Result<void>::failure(slope_result.error());
283 }
284 auto slope = std::move(slope_result.value());
285 if (slope) slope = slope->simplify();
286 if (!slope || calculus_utils_is_infinity(slope) || slope->is_zero()) {
287 return Result<void>::success();
288 }
289 auto linear = SymbolicExpr::multiply(slope, x_expr);
290 auto remainder = SymbolicExpr::add(f,
291 SymbolicExpr::multiply(SymbolicExpr::number(-1), linear));
292 auto intercept_result = limit_expression_checked(
293 remainder, var, infinity, LimitDirection::Both, context);
294 if (!intercept_result) {
295 return Result<void>::failure(intercept_result.error());
296 }
297 auto intercept = std::move(intercept_result.value());
298 if (intercept) intercept = intercept->simplify();
299 if (intercept && !calculus_utils_is_infinity(intercept)) {
300 if (!has_oblique_asymptote(result, slope, intercept)) {
301 result.oblique.emplace_back(slope, intercept);
302 }
303 }
304 return Result<void>::success();
305}

315static Result<void> collect_infinite_asymptotes(
316 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
317 const std::shared_ptr<SymbolicExpr>& pos_inf,
318 const std::shared_ptr<SymbolicExpr>& neg_inf,
319 ComputationContext& context, AsymptoteResult& result)
320{
321 auto positive = limit_expression_checked(f, var, pos_inf, LimitDirection::Both, context);
322 if (!positive) {
323 return Result<void>::failure(positive.error());
324 }
325 auto negative = limit_expression_checked(f, var, neg_inf, LimitDirection::Both, context);
326 if (!negative) {
327 return Result<void>::failure(negative.error());
328 }
329 auto lim_pos = std::move(positive.value());
330 auto lim_neg = std::move(negative.value());
331 const bool horizontal_pos = simplify_finite_limit(lim_pos);
332 const bool horizontal_neg = simplify_finite_limit(lim_neg);
333 if (horizontal_pos) result.horizontal.push_back(lim_pos);
334 if (horizontal_neg) {
335 if (!horizontal_pos || !calculus_utils_expr_equal(lim_pos, lim_neg)) {
336 result.horizontal.push_back(lim_neg);
337 }
338 }
339 auto x_expr = SymbolicExpr::variable(var);
340 if (!horizontal_pos && !horizontal_neg) {
341 std::shared_ptr<SymbolicExpr> slope;
342 std::shared_ptr<SymbolicExpr> intercept;
343 if (calculus_utils_oblique_from_rational(f, var, slope, intercept)) {
344 result.oblique.emplace_back(slope, intercept);
345 return Result<void>::success();
346 }
347 }
348 if (!horizontal_pos) {
349 auto status = collect_oblique_asymptote(f, var, x_expr, pos_inf, context, result);
350 if (!status) {
351 return status;
352 }
353 }
354 if (!horizontal_neg) {
355 return collect_oblique_asymptote(f, var, x_expr, neg_inf, context, result);
356 }
357 return Result<void>::success();
358}

363{

388 }
