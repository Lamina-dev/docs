59bool is_explicit_approx_real(const std::shared_ptr<SymbolicExpr>& expr, double& out) {
60 if (!expr || !lamina::detail::node(expr)) return false;
61 auto number = std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(expr));
62 if (!number || !std::holds_alternative<lmmc_real_t>(number->value())) return false;
63 out = static_cast<double>(std::get<lmmc_real_t>(number->value()));
64 return std::isfinite(out);
65}

248ComplexRootsResult solve_complex_nth_root_checked(std::shared_ptr<SymbolicExpr> c, int n,
249 ComputationContext& context) {
250 auto vc = validate_expr(c, "c", context, kComplexNthRootOperation);
251 if (!vc) return ComplexRootsResult::failure(vc.error());
252 if (n <= 0) {
253 return ComplexRootsResult::failure(
254 CasErrc::InvalidArgument, "root degree must be positive", kComplexNthRootOperation);
255 }
256 auto budget = context.consume_steps(static_cast<std::size_t>(n) * 8 + 8,
257 kComplexNthRootOperation);
258 if (!budget) return ComplexRootsResult::failure(budget.error());
259
260 double val = 0.0;
261 if (!is_explicit_approx_real(c, val)) {
262 return ComplexRootsResult::failure(
263 CasErrc::Inconclusive,
264 "checked complex nth roots currently require an explicit approximate real input",
265 kComplexNthRootOperation);
266 }
267
268 return checked_construct<std::vector<ComplexSymbolic>>([&]() {
269 double r = std::abs(val);
270 double theta = std::arg(std::complex<double>(val, 0.0));
271 std::vector<ComplexSymbolic> roots;
272 roots.reserve(static_cast<std::size_t>(n));
273 for (int k = 0; k < n; ++k) {
274 double root_r = std::pow(r, 1.0 / n);
275 double root_theta = (theta + 2 * LMMC_CONST_PI * k) / n;
276 roots.push_back(lamina::detail::propagate_result(complex_exp_form_checked(
277 SymbolicExpr::number(root_r),
278 SymbolicExpr::number(root_theta), context)));
279 }
280 return roots;
281 }, kComplexNthRootOperation);
282}

288ComplexRootsResult solve_complex_quadratic_checked(
289 std::shared_ptr<SymbolicExpr> a, std::shared_ptr<SymbolicExpr> b,
290 std::shared_ptr<SymbolicExpr> c, ComputationContext& context) {
291 auto va = validate_expr(a, "a", context, kComplexQuadraticOperation);
292 if (!va) return ComplexRootsResult::failure(va.error());
293 auto vb = validate_expr(b, "b", context, kComplexQuadraticOperation);
294 if (!vb) return ComplexRootsResult::failure(vb.error());
295 auto vc = validate_expr(c, "c", context, kComplexQuadraticOperation);
296 if (!vc) return ComplexRootsResult::failure(vc.error());
297 return checked_construct<std::vector<ComplexSymbolic>>([&]() {
298 auto four_ac = SymbolicExpr::multiply(
299 SymbolicExpr::number(4), SymbolicExpr::multiply(a, c));
300 auto delta = SymbolicExpr::add(
301 SymbolicExpr::power(b, SymbolicExpr::number(2)),
302 SymbolicExpr::multiply(SymbolicExpr::number(-1), four_ac));
303 auto sqrt_delta = SymbolicExpr::sqrt(delta);
304 auto denominator = SymbolicExpr::multiply(SymbolicExpr::number(2), a);
305 auto real_denominator =
306 make_complex(denominator, SymbolicExpr::number(0));
307 auto first_numerator = make_complex(
308 SymbolicExpr::add(
309 SymbolicExpr::multiply(SymbolicExpr::number(-1), b),
310 sqrt_delta),
311 SymbolicExpr::number(0));
312 auto second_numerator = make_complex(
313 SymbolicExpr::add(
314 SymbolicExpr::multiply(SymbolicExpr::number(-1), b),
315 SymbolicExpr::multiply(SymbolicExpr::number(-1), sqrt_delta)),
316 SymbolicExpr::number(0));
317 auto first = detail::propagate_result(
318 complex_div_checked(first_numerator, real_denominator, context));
319 auto second = detail::propagate_result(
320 complex_div_checked(second_numerator, real_denominator, context));
321 return std::vector<ComplexSymbolic>{std::move(first), std::move(second)};
322 }, kComplexQuadraticOperation);
323}
