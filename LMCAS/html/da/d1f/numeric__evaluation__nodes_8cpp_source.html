8Result<ApproxReal> NumericEvaluator::number(const NumberNode& node) {
9 double value = 0.0;
10 try {
11 if (std::holds_alternative<lmmc_real_t>(node.value())) {
12 value = static_cast<double>(std::get<lmmc_real_t>(node.value()));
13 } else if (std::holds_alternative<BigInt>(node.value())) {
14 value = std::get<BigInt>(node.value()).to_double();
15 } else {
16 value = std::get<Rational>(node.value()).to_double();
17 }
18 } catch (const std::exception& error) {
19 return numeric_failure(CasErrc::NumericFailure, error.what());
20 }
21 if (!std::isfinite(value)) {
22 return numeric_failure(CasErrc::NumericFailure,
23 "exact number cannot be represented as a finite double");
24 }
25 if (std::holds_alternative<lmmc_real_t>(node.value())) {
26 return Result<ApproxReal>::success(ApproxReal{value, 0.0, NumericStatus::Finite});
27 }
28 return numeric_approximation(value);
29}

53Result<ApproxReal> NumericEvaluator::sum(const AddNode& node) {
54 double value = 0.0;
55 bool infinite = false;
56 for (const auto& operand : node.operands()) {
57 auto evaluated = evaluate(operand);
58 if (!evaluated) return evaluated;
59 value += evaluated.value().value;
60 infinite = infinite || !evaluated.value().is_finite();
61 if (std::isinf(value) && !infinite) {
62 return numeric_failure(CasErrc::NumericFailure, "numeric evaluation produced a nonfinite result");
63 }
64 }
65 if (std::isinf(value) && infinite) {
66 return Result<ApproxReal>::success(ApproxReal{
67 value, 0.0, value > 0 ? NumericStatus::PositiveInfinity : NumericStatus::NegativeInfinity});
68 }
69 return numeric_approximation(value);
70}

102Result<ApproxReal> NumericEvaluator::extremum(const FunctionNode& node) {
103 if (node.arguments().empty()) {
104 return numeric_failure(CasErrc::InvalidArgument, "min/max requires an argument");
105 }
106 auto first = evaluate(node.arguments().front());
107 if (!first) return first;
108 double value = first.value().value;
109 for (std::size_t i = 1; i < node.arguments().size(); ++i) {
110 auto next = evaluate(node.arguments()[i]);
111 if (!next) return next;
112 value = node.type() == FunctionNode::FuncType::Max
113 ? std::max(value, next.value().value)
114 : std::min(value, next.value().value);
115 }
116 return numeric_approximation(value);
117}
