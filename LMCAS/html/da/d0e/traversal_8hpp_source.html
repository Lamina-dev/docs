12/** Scope information for an AST node that introduces one lexical binder. */
13struct BinderView {
14 std::string_view bound_name;
15 SymbolicNodePtr scoped_body;
16 std::vector<SymbolicNodePtr> outside_scope;
17};
18
19inline std::optional<BinderView> binder_view(const SymbolicNode& node) {
20 if (const auto* sum = dynamic_cast<const SummationNode*>(&node)) {
21 return BinderView{sum->index_var(), sum->body(),
22 {sum->lower_bound(), sum->upper_bound()}};
23 }
24 if (const auto* product = dynamic_cast<const ProductNode*>(&node)) {
25 return BinderView{product->index_var(), product->body(),
26 {product->lower_bound(), product->upper_bound()}};
27 }
28 if (const auto* integral = dynamic_cast<const IntegralNode*>(&node)) {
29 std::vector<SymbolicNodePtr> bounds;
30 if (integral->lower()) bounds.push_back(integral->lower());
31 if (integral->upper()) bounds.push_back(integral->upper());
32 return BinderView{integral->variable(), integral->body(), std::move(bounds)};
33 }
34 if (const auto* transform = dynamic_cast<const TransformNode*>(&node)) {
35 return BinderView{transform->source_var(), transform->body(),
36 {transform->target()}};
37 }
38 if (const auto* quantifier = dynamic_cast<const QuantifierNode*>(&node)) {
39 return BinderView{quantifier->bound_var(), quantifier->predicate(),
40 {quantifier->domain()}};
41 }
42 if (const auto* set_builder = dynamic_cast<const SetBuilderNode*>(&node)) {
43 return BinderView{set_builder->element_var(), set_builder->predicate(),
44 {set_builder->domain()}};
45 }
46 if (const auto* limit = dynamic_cast<const LimitNode*>(&node)) {
47 return BinderView{limit->variable(), limit->body(), {limit->point()}};
48 }
49 return std::nullopt;
50}

52/** Exhaustive depth-first traversal with correct coverage for every AST child. */
53class RecursiveSymbolicVisitor : public SymbolicVisitor {
54protected:
55 void visit_child(const SymbolicNodePtr& child) {
56 if (child) child->accept(*this);
57 }
58
59 template <class Range>
60 void visit_children(const Range& children) {
61 for (const auto& child : children) visit_child(child);
62 }
63
64public:
65 void visit(const NumberNode&) override {}
66 void visit(const VariableNode&) override {}
67 void visit(const AddNode& node) override { visit_children(node.operands()); }
68 void visit(const MultiplyNode& node) override { visit_children(node.operands()); }
69 void visit(const PowerNode& node) override {
70 visit_child(node.base());
71 visit_child(node.exponent());
72 }
73 void visit(const FunctionNode& node) override { visit_children(node.arguments()); }
74 void visit(const UninterpretedFunctionNode& node) override {
75 visit_children(node.arguments());
76 }
77 void visit(const MatrixNode& node) override {
78 if (const auto* dense = std::get_if<MatrixNode::DenseStorage>(&node.storage())) {
79 visit_children(*dense);
80 return;
81 }
82 for (const auto& [index, value] :
83 std::get<MatrixNode::SparseStorage>(node.storage())) {
84 (void)index;
85 visit_child(value);
86 }
87 }
88 void visit(const RelationalNode& node) override {
89 visit_child(node.left());
90 visit_child(node.right());
91 }
92 void visit(const LogicalNode& node) override {
93 visit_child(node.left());
94 visit_child(node.right());
95 }
96 void visit(const PiecewiseNode& node) override {
97 for (const auto& branch : node.branches()) {
98 visit_child(branch.expression);
99 visit_child(branch.condition);
100 }
101 visit_child(node.default_expr());
102 }
103 void visit(const SummationNode& node) override {
104 visit_child(node.lower_bound());
105 visit_child(node.upper_bound());
106 visit_child(node.body());
107 }
108 void visit(const ProductNode& node) override {
109 visit_child(node.lower_bound());
110 visit_child(node.upper_bound());
111 visit_child(node.body());
112 }
113 void visit(const TransformNode& node) override {
114 visit_child(node.target());
115 visit_child(node.body());
116 }
117 void visit(const QuantifierNode& node) override {
118 visit_child(node.domain());
119 visit_child(node.predicate());
120 }
121 void visit(const SetBuilderNode& node) override {
122 visit_child(node.domain());
123 visit_child(node.predicate());
124 }
125 void visit(const FiniteSetNode& node) override { visit_children(node.elements()); }
126 void visit(const IntervalNode& node) override {
127 visit_child(node.lower());
128 visit_child(node.upper());
129 }
130 void visit(const MembershipNode& node) override {
131 visit_child(node.element());
132 visit_child(node.set());
133 }
134 void visit(const QuantityNode& node) override { visit_child(node.value()); }
135 void visit(const ComplexNode& node) override {
136 visit_child(node.real());
137 visit_child(node.imag());
138 }
139 void visit(const IntegralNode& node) override {
140 visit_child(node.lower());

169/** Immutable, path-copying AST rewrite. Unchanged subtrees retain pointer identity. */
170class SymbolicRewriter : public SymbolicVisitor {
171public:
172 SymbolicNodePtr rewrite(const SymbolicNodePtr& node) {
173 if (!node) return nullptr;
174 const auto previous_current = current_;
175 const auto previous_result = result_;
176 current_ = node;
177 result_.reset();
178 node->accept(*this);
179 auto rewritten = result_;
180 current_ = previous_current;
181 result_ = previous_result;
182 if (!rewritten) throw std::logic_error("SymbolicRewriter visit did not set a result");
183 return rewritten;
184 }
185
186protected:
187 SymbolicNodePtr current() const noexcept { return current_; }
188 void set_result(SymbolicNodePtr result) { result_ = std::move(result); }
189
190 SymbolicNodePtr rewrite_child(const SymbolicNodePtr& child, bool& changed) {
191 auto rewritten = rewrite(child);
192 changed = changed || rewritten != child;
193 return rewritten;
194 }
195
196 template <class Range>
197 std::vector<SymbolicNodePtr> rewrite_children(const Range& children, bool& changed) {
198 std::vector<SymbolicNodePtr> rewritten;
199 rewritten.reserve(children.size());
200 for (const auto& child : children) {
201 rewritten.push_back(rewrite_child(child, changed));
202 }
203 return rewritten;
204 }
205
206public:
207 void visit(const NumberNode&) override { set_result(current()); }
208 void visit(const VariableNode&) override { set_result(current()); }
209 void visit(const AddNode& node) override {
210 bool changed = false;
211 auto operands = rewrite_children(node.operands(), changed);
212 set_result(changed ? SymbolicFactory::create_add(std::move(operands)) : current());
213 }
214 void visit(const MultiplyNode& node) override {
215 bool changed = false;
216 auto operands = rewrite_children(node.operands(), changed);
217 set_result(changed ? SymbolicFactory::create_multiply(std::move(operands)) : current());
218 }
219 void visit(const PowerNode& node) override {
220 bool changed = false;
221 auto base = rewrite_child(node.base(), changed);
222 auto exponent = rewrite_child(node.exponent(), changed);
223 set_result(changed ? SymbolicFactory::create_power(base, exponent) : current());
224 }
225 void visit(const FunctionNode& node) override {
226 bool changed = false;
227 auto arguments = rewrite_children(node.arguments(), changed);
228 set_result(changed ? make_node<FunctionNode>(node.type(), std::move(arguments)) : current());
229 }
230 void visit(const UninterpretedFunctionNode& node) override {
231 bool changed = false;
232 auto arguments = rewrite_children(node.arguments(), changed);
233 set_result(changed ? make_node<UninterpretedFunctionNode>(node.name(), std::move(arguments))
234 : current());
235 }
236 void visit(const MatrixNode& node) override {
237 bool changed = false;
238 if (const auto* dense = std::get_if<MatrixNode::DenseStorage>(&node.storage())) {
239 auto entries = rewrite_children(*dense, changed);
240 set_result(changed ? make_node<MatrixNode>(node.rows(), node.cols(), std::move(entries))
241 : current());
242 return;
243 }
244 auto entries = std::get<MatrixNode::SparseStorage>(node.storage());
245 for (auto& [index, value] : entries) {
246 (void)index;
247 value = rewrite_child(value, changed);
248 }
249 set_result(changed ? make_node<MatrixNode>(node.rows(), node.cols(), std::move(entries))
250 : current());
251 }
252 void visit(const RelationalNode& node) override {
253 bool changed = false;
254 auto left = rewrite_child(node.left(), changed);
255 auto right = rewrite_child(node.right(), changed);
256 set_result(changed ? make_node<RelationalNode>(left, right, node.op()) : current());
257 }
258 void visit(const LogicalNode& node) override {
259 bool changed = false;
260 auto left = rewrite_child(node.left(), changed);
261 auto right = node.right() ? rewrite_child(node.right(), changed) : nullptr;
262 set_result(changed ? make_node<LogicalNode>(left, right, node.op()) : current());
263 }
264 void visit(const PiecewiseNode& node) override {
265 bool changed = false;
266 std::vector<PiecewiseNode::Branch> branches;
267 branches.reserve(node.branches().size());
268 for (const auto& branch : node.branches()) {
269 branches.push_back({rewrite_child(branch.expression, changed),
270 rewrite_child(branch.condition, changed)});
271 }
272 auto fallback = node.default_expr()
273 ? rewrite_child(node.default_expr(), changed) : nullptr;
274 set_result(changed ? make_node<PiecewiseNode>(std::move(branches), fallback) : current());
275 }
276 void visit(const SummationNode& node) override {
277 bool changed = false;
278 auto body = rewrite_child(node.body(), changed);
279 auto lower = rewrite_child(node.lower_bound(), changed);
280 auto upper = rewrite_child(node.upper_bound(), changed);
281 set_result(changed ? make_node<SummationNode>(body, node.index_var(), lower, upper)
282 : current());
283 }
284 void visit(const ProductNode& node) override {
285 bool changed = false;
286 auto body = rewrite_child(node.body(), changed);
287 auto lower = rewrite_child(node.lower_bound(), changed);
288 auto upper = rewrite_child(node.upper_bound(), changed);
289 set_result(changed ? make_node<ProductNode>(body, node.index_var(), lower, upper)
290 : current());
291 }
292 void visit(const TransformNode& node) override {
293 bool changed = false;
294 auto body = rewrite_child(node.body(), changed);
295 auto target = rewrite_child(node.target(), changed);
296 set_result(changed ? make_node<TransformNode>(node.transform_type(), body,
297 node.source_var(), target)
298 : current());
299 }
300 void visit(const QuantifierNode& node) override {
301 bool changed = false;
302 auto domain = rewrite_child(node.domain(), changed);
303 auto predicate = rewrite_child(node.predicate(), changed);
304 set_result(changed ? make_node<QuantifierNode>(node.quantifier_type(), node.bound_var(),
305 domain, predicate)
306 : current());
307 }
308 void visit(const SetBuilderNode& node) override {
309 bool changed = false;
310 auto domain = rewrite_child(node.domain(), changed);
311 auto predicate = rewrite_child(node.predicate(), changed);
312 set_result(changed ? make_node<SetBuilderNode>(node.element_var(), domain, predicate)
313 : current());
314 }
315 void visit(const FiniteSetNode& node) override {
316 bool changed = false;
317 auto elements = rewrite_children(node.elements(), changed);
318 set_result(changed ? make_node<FiniteSetNode>(std::move(elements)) : current());
319 }
320 void visit(const IntervalNode& node) override {
321 bool changed = false;
322 auto lower = rewrite_child(node.lower(), changed);
323 auto upper = rewrite_child(node.upper(), changed);
324 set_result(changed ? make_node<IntervalNode>(lower, upper,
325 node.lower_closed(), node.upper_closed())
326 : current());
327 }
328 void visit(const MembershipNode& node) override {
329 bool changed = false;
330 auto element = rewrite_child(node.element(), changed);
331 auto set = rewrite_child(node.set(), changed);
332 set_result(changed ? make_node<MembershipNode>(element, set) : current());
333 }
334 void visit(const QuantityNode& node) override {
335 bool changed = false;
336 auto value = rewrite_child(node.value(), changed);
337 set_result(changed ? make_node<QuantityNode>(value, node.dimension(),
338 node.scale_to_base(), node.display_unit())
339 : current());
340 }
341 void visit(const ComplexNode& node) override {
342 bool changed = false;
343 auto real = rewrite_child(node.real(), changed);
344 auto imag = rewrite_child(node.imag(), changed);
345 set_result(changed ? SymbolicFactory::create_complex(real, imag) : current());
346 }
347 void visit(const IntegralNode& node) override {
348 bool changed = false;
349 auto body = rewrite_child(node.body(), changed);
350 auto lower = node.lower() ? rewrite_child(node.lower(), changed) : nullptr;
351 auto upper = node.upper() ? rewrite_child(node.upper(), changed) : nullptr;
352 set_result(changed ? make_node<IntegralNode>(body, node.variable(), lower, upper)
353 : current());
354 }
355 void visit(const LimitNode& node) override {
356 bool changed = false;
357 auto body = rewrite_child(node.body(), changed);
358 auto point = rewrite_child(node.point(), changed);
359 set_result(changed ? make_node<LimitNode>(
360 body, node.variable(), point, node.direction())
361 : current());
362 }
363 void visit(const RootOfNode&) override { set_result(current()); }
364
365private:
366 SymbolicNodePtr current_;
367 SymbolicNodePtr result_;
368};
