14static bool te_is_power_of_var(const std::shared_ptr<SymbolicExpr>& e, const std::string& v, BigInt& n) {
15 if (!e || !LMCAS::detail::node(e)) {
16 return false;
17 }
18 auto vn = std::dynamic_pointer_cast<const VariableNode>(LMCAS::detail::node(e));
19 if (vn && !vn->is_constant() && vn->name() == v) { n = BigInt(1); return true; }
20 auto pw = std::dynamic_pointer_cast<const PowerNode>(LMCAS::detail::node(e));
21 if (!pw) {
22 return false;
23 }
24 auto bv = std::dynamic_pointer_cast<const VariableNode>(pw->base());
25 if (!bv || bv->is_constant() || bv->name() != v) {
26 return false;
27 }
28 return try_get_integer_value(std::dynamic_pointer_cast<const NumberNode>(pw->exponent()), n) &&
29 !n.is_negative();

48static bool te_extract_exp(const std::shared_ptr<SymbolicExpr>& e, const std::string& t,
49 std::shared_ptr<SymbolicExpr>& a_out, std::shared_ptr<SymbolicExpr>& rem_out) {
50 if (!e || !LMCAS::detail::node(e)) {
51 return false;
52 }
53 auto fn = std::dynamic_pointer_cast<const FunctionNode>(LMCAS::detail::node(e));
54 if (fn && fn->type() == FunctionNode::FuncType::Exp && !fn->arguments().empty()) {
55 a_out = te_linear_coefficient(fn->arguments()[0], t);
56 if (a_out) {
57 rem_out = detail::make_expression_ptr(SymbolicFactory::create_number(BigInt(1)));
58 return true;
59 }
60 }
61 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(LMCAS::detail::node(e));
62 if (mul) {
63 for (size_t i = 0; i < mul->operands().size(); ++i) {
64 auto fac = LMCAS::detail::make_expression_ptr(mul->operands()[i]);
65 std::shared_ptr<SymbolicExpr> at, rt;
66 if (te_extract_exp(fac, t, at, rt)) {
67 std::vector<std::shared_ptr<const SymbolicNode>> rest = mul->operands();
68 rest.erase(rest.begin() + static_cast<std::ptrdiff_t>(i));
69 rem_out = detail::make_expression_ptr(
70 SymbolicFactory::create_multiply(std::move(rest)));
71 a_out = at; return true;
72 }
73 }
74 }
75 return false;
76}

78static std::shared_ptr<SymbolicExpr> te_laplace_lookup(
79 const std::shared_ptr<SymbolicExpr>& f, const std::string& t,
80 const std::string& s, ComputationContext& context) {
81 if (!f || !LMCAS::detail::node(f)) return nullptr;

83 if (!te_depends_on(f, t)) return SymbolicExpr::divide(f, sv);
84 BigInt n;
85 if (te_is_power_of_var(f, t, n)) {
86 return SymbolicExpr::divide(SymbolicExpr::number(te_factorial(n, context)),
87 SymbolicExpr::power(sv, SymbolicExpr::number(n + BigInt(1))));
88 }
89 bool is_sin = false; std::shared_ptr<SymbolicExpr> freq;
90 if (te_is_trig(f, t, is_sin, freq)) {
91 auto den = SymbolicExpr::add(SymbolicExpr::power(sv, SymbolicExpr::number(2)), SymbolicExpr::power(freq, SymbolicExpr::number(2)));
92 return is_sin ? SymbolicExpr::divide(freq, den) : SymbolicExpr::divide(sv, den);
93 }
94 bool is_sinh = false; std::shared_ptr<SymbolicExpr> hf;
95 if (te_is_hyp(f, t, is_sinh, hf)) {
96 auto den = SymbolicExpr::add(SymbolicExpr::power(sv, SymbolicExpr::number(2)), SymbolicExpr::multiply(SymbolicExpr::number(-1), SymbolicExpr::power(hf, SymbolicExpr::number(2))));
97 return is_sinh ? SymbolicExpr::divide(hf, den) : SymbolicExpr::divide(sv, den);
98 }
99 std::shared_ptr<SymbolicExpr> a, rem;
100 if (te_extract_exp(f, t, a, rem) && !te_depends_on(rem, t))
101 return SymbolicExpr::divide(rem, SymbolicExpr::add(sv, SymbolicExpr::multiply(SymbolicExpr::number(-1), a)));
102 return nullptr;
103}

152std::shared_ptr<SymbolicExpr> transform_detail::laplace_transform_core(
153 const std::shared_ptr<SymbolicExpr>& f, const std::string& t,
154 const std::string& s, ComputationContext& context) {
155 auto step = context.consume_steps(1, "laplace_transform.recursive");
156 if (!step) {
157 return nullptr;
158 }
159 if (!f || !LMCAS::detail::node(f)) {
160 return nullptr;
161 }
162 if (te_is_zero(f)) {
163 return SymbolicExpr::number(0);
164 }
165 if (!te_depends_on(f, t)) {
166 return SymbolicExpr::divide(f, SymbolicExpr::variable(s));
167 }
168 auto add = std::dynamic_pointer_cast<const AddNode>(LMCAS::detail::node(f));
169 if (add) {
170 return te_laplace_sum(f, *add, t, s, context);
171 }
172 auto [coeff, body] = te_split_coeff(f, t);
173 if (!coeff->is_one()) {
174 auto lt_body = laplace_transform_core(body, t, s, context);
175 if (lt_body) {
176 return SymbolicExpr::multiply(coeff, lt_body);
177 }
178 }
179 std::shared_ptr<SymbolicExpr> a_shift, remainder;
180 if (te_extract_exp(f, t, a_shift, remainder)) {
181 if (te_depends_on(remainder, t)) {
182 auto lt_rem = laplace_transform_core(remainder, t, s, context);
183 if (lt_rem) {
184 return lt_rem->substitute(s, SymbolicExpr::add(SymbolicExpr::variable(s), SymbolicExpr::multiply(SymbolicExpr::number(-1), a_shift)));
185 }
186 } else {
187 return SymbolicExpr::divide(remainder, SymbolicExpr::add(SymbolicExpr::variable(s), SymbolicExpr::multiply(SymbolicExpr::number(-1), a_shift)));
188 }
189 }
190 auto lookup = te_laplace_lookup(f, t, s, context);
191 if (lookup) {
192 return lookup;
193 }
194 return te_unevaluated_laplace(f, t, s);
195}
