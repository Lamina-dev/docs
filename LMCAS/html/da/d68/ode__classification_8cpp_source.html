20/**
21 * @internal
22 * @brief 尝试将表达式求值为 double。
23 * @return 若表达式为纯数值返回其值，否则返回 NaN。
24 */
25double try_eval_double(const std::shared_ptr<SymbolicExpr>& expr) {
26 if (!expr || !lamina::detail::node(expr)) return std::numeric_limits<double>::quiet_NaN();
27 if (expr->is_number()) {
28 auto node = std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(expr));
29 if (!node) return std::numeric_limits<double>::quiet_NaN();
30 if (std::holds_alternative<BigInt>(node->value()))
31 return std::get<BigInt>(node->value()).to_double();
32 if (std::holds_alternative<Rational>(node->value()))
33 return std::get<Rational>(node->value()).to_double();
34 return static_cast<double>(std::get<lmmc_real_t>(node->value()));
35 }
36 return std::numeric_limits<double>::quiet_NaN();
37}

230bool is_separable(
231 const std::shared_ptr<SymbolicExpr>& rhs,
232 const std::string& x,
233 const std::string& y)
234{
235 if (!rhs || !lamina::detail::node(rhs)) return true;
236
237 bool has_x = expression_depends_on_variable(lamina::detail::node(rhs), x);
238 bool has_y = expression_depends_on_variable(lamina::detail::node(rhs), y);
239
240 /// 若只依赖一个变量或都不依赖，则可分离
241 if (!has_x || !has_y) return true;
242
243 /// 检查乘法形式 f(x)*g(y)
244 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(rhs));
245 if (mul) {
246 /// 将因子分为仅含 x 的和仅含 y 的
247 bool all_separable = true;
248 for (const auto& factor : mul->operands()) {
249 bool fx = expression_depends_on_variable(factor, x);
250 bool fy = expression_depends_on_variable(factor, y);
251 if (fx && fy) {
252 all_separable = false;
253 break;
254 }
255 }
256 if (all_separable) return true;
257 }
258
259 /// 检查除法形式 f(x)/g(y) 或 g(y)/f(x)
260 auto pow = std::dynamic_pointer_cast<const PowerNode>(lamina::detail::node(rhs));
261 if (pow) {
262 auto exp_node = std::dynamic_pointer_cast<const NumberNode>(pow->exponent());
263 if (exp_node) {
264 double exp_val = try_eval_double(lamina::detail::make_expression_ptr(pow->exponent()));
265 int eq;
266 lmmc_double_nearly_equal_tol(exp_val, -1.0, 1e-12, 1e-12, &eq);
267 if (eq) {
268 /// rhs = base^(-1) = 1/base; 若 base 仅含一个变量则可分离
269 bool base_x = expression_depends_on_variable(pow->base(), x);
270 bool base_y = expression_depends_on_variable(pow->base(), y);
271 if ((base_x && !base_y) || (!base_x && base_y)) return true;
272 }
273 }
274 }
275
276 return false;
277}

280bool is_linear_first_order(
281 const std::shared_ptr<SymbolicExpr>& rhs,
282 const std::string&,
283 const std::string& y,
284 std::shared_ptr<SymbolicExpr>& P,
285 std::shared_ptr<SymbolicExpr>& Q)
286{
287 if (!rhs || !lamina::detail::node(rhs)) {
288 P = SymbolicExpr::number(0);
289 Q = SymbolicExpr::number(0);
290 return true;
291 }
292
293 /// 方程形式: y' = rhs(x, y)
294 /// 线性形式: y' + P(x)*y = Q(x) → y' = Q(x) - P(x)*y → rhs = Q(x) - P(x)*y
295 /// 即 rhs 关于 y 是线性的: rhs = A(x) + B(x)*y，其中 Q = A, P = -B
296
297 /// 若 rhs 不依赖 y，则 P=0, Q=rhs
298 if (!expression_depends_on_variable(lamina::detail::node(rhs), y)) {
299 P = SymbolicExpr::number(0);
300 Q = rhs;
301 return true;
302 }
303
304 /// 对 rhs 关于 y 求导，若结果不依赖 y，则 rhs 关于 y 是线性的
305 auto drhs_dy = rhs->differentiate(y);
306 if (!drhs_dy || expression_depends_on_variable(lamina::detail::node(drhs_dy), y)) {
307 return false;
308 }
309
310 /// rhs = A(x) + B(x)*y，其中 B(x) = ∂rhs/∂y
311 /// 计算 A(x) = rhs|_{y=0}
312 auto A = rhs->substitute(y, SymbolicExpr::number(0));
313 if (!A) return false;
314
315 /// 验证 A 不依赖 y
316 if (expression_depends_on_variable(lamina::detail::node(A), y)) return false;
317
318 /// B(x) = drhs_dy（已验证不依赖 y）
319 /// 线性形式: y' = A(x) + B(x)*y → y' - B(x)*y = A(x) → y' + (-B(x))*y = A(x)
320 /// 所以 P = -B(x), Q = A(x)
321 P = SymbolicExpr::multiply(SymbolicExpr::number(-1), drhs_dy)->simplify();
322 Q = A->simplify();
323 return true;
324}

327bool is_homogeneous_ode(
328 const std::shared_ptr<SymbolicExpr>& rhs,
329 const std::string& x,
330 const std::string& y)
331{
332 if (!rhs || !lamina::detail::node(rhs)) return false;
333
334 /// 齐次方程: f(tx, ty) = f(x, y) 对所有 t 成立
335 /// 用 t=2 进行数值测试：f(2x, 2y) 应等于 f(x, y)
336 auto t_val = SymbolicExpr::number(2);
337 auto tx = SymbolicExpr::multiply(t_val, SymbolicExpr::variable(x));
338 auto ty = SymbolicExpr::multiply(t_val, SymbolicExpr::variable(y));
339
340 auto f_scaled = rhs->substitute(x, tx);
341 f_scaled = f_scaled->substitute(y, ty);
342 f_scaled = f_scaled->simplify();
343
344 /// 计算 f_scaled - rhs，若化简为零则齐次
345 auto diff = SymbolicExpr::add(f_scaled,
346 SymbolicExpr::multiply(SymbolicExpr::number(-1), rhs));
347 diff = diff->simplify();
348
349 if (diff->is_zero()) return true;
350
351 /// 数值验证：在具体点 (x=1, y=1) 和 (x=2, y=3) 测试
352 auto test_at = [&](double xv, double yv) -> bool {
353 auto f_orig = rhs->substitute(x, SymbolicExpr::number(xv));
354 f_orig = f_orig->substitute(y, SymbolicExpr::number(yv));
355 f_orig = f_orig->simplify();
356
357 double t_test = 2.0;
358 auto f_sc = rhs->substitute(x, SymbolicExpr::number(t_test * xv));
359 f_sc = f_sc->substitute(y, SymbolicExpr::number(t_test * yv));
360 f_sc = f_sc->simplify();
361
362 double v_orig = try_eval_double(f_orig);
363 double v_sc = try_eval_double(f_sc);
364
365 if (std::isnan(v_orig) || std::isnan(v_sc)) return false;
366 if (std::abs(v_orig) < 1e-15 && std::abs(v_sc) < 1e-15) return true;
367 if (std::abs(v_orig) < 1e-15) return false;
368
369 int eq;
370 lmmc_double_nearly_equal_tol(v_orig, v_sc, 1e-9, 1e-9, &eq);
371 return eq != 0;
372 };
373
374 /// 在多个点测试
375 if (test_at(1.0, 1.0) && test_at(2.0, 3.0) && test_at(0.5, 1.5)) {
376 return true;
377 }
378
379 return false;
380}

383bool is_bernoulli_ode(
384 const std::shared_ptr<SymbolicExpr>& rhs,
385 const std::string&,
386 const std::string& y,
387 std::shared_ptr<SymbolicExpr>& P,
388 std::shared_ptr<SymbolicExpr>& Q,
389 int& n)
390{
391 if (!rhs || !lamina::detail::node(rhs)) return false;
392 if (!expression_depends_on_variable(lamina::detail::node(rhs), y)) return false;
393
394 /// Bernoulli: y' + P(x)*y = Q(x)*y^n → y' = -P(x)*y + Q(x)*y^n
395 /// 即 rhs = -P(x)*y + Q(x)*y^n = y*(-P(x) + Q(x)*y^{n-1})
396
397 /// 尝试 rhs / y 并检查结果是否为 A(x) + B(x)*y^m 形式
398 /// 其中 m = n-1, P = -A, Q = B, n = m+1
399
400 /// 首先检查 rhs 是否含 y 的幂次
401 /// 对 rhs 关于 y 求两次导，检查是否为 y 的幂函数
402 auto d1 = rhs->differentiate(y);
403 if (!d1) return false;
404 auto d2 = d1->differentiate(y);
405 if (!d2) return false;
406
407 /// 若 d2 不依赖 y，则 rhs 关于 y 最多是二次的
408 /// 对于 Bernoulli，我们需要 rhs = A(x)*y + B(x)*y^n
409 /// 尝试特定的 n 值: 2, 3, -1
410 for (int test_n : {2, 3, -1, 4}) {
411 /// rhs 应为 -P(x)*y + Q(x)*y^n
412 /// 令 rhs / y = -P(x) + Q(x)*y^{n-1}
413 /// 若 n=2: rhs/y = -P(x) + Q(x)*y → 关于 y 线性
414 /// 若 n=3: rhs/y = -P(x) + Q(x)*y^2 → 关于 y 二次
415
416 /// 计算 rhs 在 y=1 和 y=2 处的值来推断结构
417 auto rhs_at_y1 = rhs->substitute(y, SymbolicExpr::number(1))->simplify();
418 auto rhs_at_y0 = rhs->substitute(y, SymbolicExpr::number(0))->simplify();
419
420 /// 若 rhs(x, 0) = 0，则 rhs 含 y 因子
421 if (!rhs_at_y0->is_zero()) continue;
422
423 /// rhs = y * h(x, y)，计算 h = rhs / y
424 /// h(x, y) = -P(x) + Q(x)*y^{n-1}
425 /// h(x, 0) = -P(x)
426 /// h(x, 1) = -P(x) + Q(x)
427
428 /// 用 y 除 rhs：对 rhs 做 substitute 检查
429 /// 实际上，若 rhs(x,0)=0，则 rhs 含 y 因子
430 /// 计算 ∂rhs/∂y|_{y=0} = h(x, 0) = -P(x)
431 auto h_at_0 = d1->substitute(y, SymbolicExpr::number(0))->simplify();
432 if (expression_depends_on_variable(lamina::detail::node(h_at_0), y)) continue;
433
434 /// 对于 Bernoulli n=test_n:
435 /// rhs = -P*y + Q*y^n
436 /// d(rhs)/dy = -P + n*Q*y^{n-1}
437 /// d(rhs)/dy|_{y=0} = -P (对 n>=2)
438 /// d²(rhs)/dy²|_{y=0} = n*(n-1)*Q*y^{n-2}|_{y=0}
439 /// 对 n=2: = 2*Q
440 /// 对 n=3: = 0 (需要 y=0 时 y^1 = 0)
441
442 if (test_n == 2) {
443 /// d²rhs/dy² = 2*Q(x) (常数关于 y)
444 auto d2_simplified = d2->simplify();
445 if (expression_depends_on_variable(lamina::detail::node(d2_simplified), y)) continue;
446
447 /// 验证三阶导为零
448 auto d3 = d2->differentiate(y)->simplify();
449 if (!d3->is_zero()) continue;
450
451 /// P = -(d1|_{y=0}), Q = d2/2
452 P = SymbolicExpr::multiply(SymbolicExpr::number(-1), h_at_0)->simplify();
453 Q = SymbolicExpr::divide(d2_simplified, SymbolicExpr::number(2))->simplify();
454
455 /// 验证 Q 不依赖 y
456 if (expression_depends_on_variable(lamina::detail::node(Q), y)) continue;
457
458 n = 2;
459 return true;
460 }
461
462 if (test_n == 3) {
463 /// rhs = -P*y + Q*y^3
464 /// d1 = -P + 3*Q*y^2
465 /// d2 = 6*Q*y
466 /// d3 = 6*Q
467 auto d2_at_0 = d2->substitute(y, SymbolicExpr::number(0))->simplify();
468 if (!d2_at_0->is_zero()) continue;
469
470 auto d3 = d2->differentiate(y)->simplify();
471 if (expression_depends_on_variable(lamina::detail::node(d3), y)) continue;
472
473 auto d4 = d3->differentiate(y)->simplify();
474 if (!d4->is_zero()) continue;
475
476 P = SymbolicExpr::multiply(SymbolicExpr::number(-1), h_at_0)->simplify();
477 Q = SymbolicExpr::divide(d3, SymbolicExpr::number(6))->simplify();
478
479 if (expression_depends_on_variable(lamina::detail::node(Q), y)) continue;
480
481 n = 3;
482 return true;
483 }
484
485 /// 对于 n=-1 和 n=4，使用数值验证
486 /// rhs = -P*y + Q*y^n
487 /// 在 y=1: rhs(x,1) = -P + Q
488 /// 在 y=2: rhs(x,2) = -2P + Q*2^n
489 /// 在 y=3: rhs(x,3) = -3P + Q*3^n
490 /// 从两个方程解出 P 和 Q，用第三个验证
491 }
492
493 return false;
494}

497bool is_exact_ode(
498 const std::shared_ptr<SymbolicExpr>& M,
499 const std::shared_ptr<SymbolicExpr>& N,
500 const std::string& x,
501 const std::string& y)
502{
503 if (!M || !N) return false;
504
505 /// 恰当条件: ∂M/∂y = ∂N/∂x
506 auto dM_dy = M->differentiate(y);
507 auto dN_dx = N->differentiate(x);
508
509 if (!dM_dy || !dN_dx) return false;
510
511 dM_dy = dM_dy->simplify();
512 dN_dx = dN_dx->simplify();
513 if (dM_dy->compare(dN_dx) == 0) return true;
514
515 /// 计算差值并化简
516 auto diff = SymbolicExpr::add(dM_dy,
517 SymbolicExpr::multiply(SymbolicExpr::number(-1), dN_dx));
518 diff = diff->simplify();
519
520 if (diff->is_zero()) return true;
521
522 /// 数值验证：在几个点检查
523 auto eval_at = [&](double xv, double yv) -> bool {
524 auto d = diff->substitute(x, SymbolicExpr::number(xv));
525 d = d->substitute(y, SymbolicExpr::number(yv));
526 d = d->simplify();
527 double val = try_eval_double(d);
528 if (std::isnan(val)) return false;
529 int eq;
530 lmmc_double_nearly_equal_tol(val, 0.0, 1e-9, 1e-9, &eq);
531 return eq != 0;
532 };
533
534 if (eval_at(1.0, 1.0) && eval_at(2.0, 3.0) && eval_at(0.5, -1.0)) {
535 return true;
536 }
537
538 return false;
539}

554bool is_euler_equation(
555 const std::vector<std::shared_ptr<SymbolicExpr>>& coeffs,
556 const std::string& x,
557 std::vector<double>& euler_consts)
558{
559 /// Euler 方程: 第 k 阶导数的系数为 a_k * x^k
560 /// coeffs[0] 对应最高阶 (阶数 n)，coeffs[i] 对应阶数 n-i
561 int n = static_cast<int>(coeffs.size()) - 1;
562 euler_consts.clear();
563 euler_consts.resize(coeffs.size(), 0.0);
564
565 for (int i = 0; i <= n; ++i) {
566 int order = n - i; // 该系数对应的导数阶数
567
568 if (!coeffs[i] || coeffs[i]->is_zero()) {
569 euler_consts[i] = 0.0;
570 continue;
571 }
572
573 if (order == 0) {
574 /// 零阶项系数应为常数
575 if (expression_depends_on_variable(lamina::detail::node(coeffs[i]), x)) return false;
576 double val = try_eval_double(coeffs[i]);
577 if (std::isnan(val)) return false;
578 euler_consts[i] = val;
579 continue;
580 }
581
582 /// 第 order 阶导数的系数应为 a_k * x^order
583 /// 除以 x^order 后应为常数
584 auto x_power = SymbolicExpr::power(
585 SymbolicExpr::variable(x),
586 SymbolicExpr::number(order));
587 auto ratio = SymbolicExpr::divide(coeffs[i], x_power)->simplify();
588
589 if (expression_depends_on_variable(lamina::detail::node(ratio), x)) {
590 /// 数值验证：在 x=1 和 x=2 处检查比值是否相同
591 auto at_1 = ratio->substitute(x, SymbolicExpr::number(1.0))->simplify();
592 auto at_2 = ratio->substitute(x, SymbolicExpr::number(2.0))->simplify();
593 double v1 = try_eval_double(at_1);
594 double v2 = try_eval_double(at_2);
595 if (std::isnan(v1) || std::isnan(v2)) return false;
596 int eq;
597 lmmc_double_nearly_equal_tol(v1, v2, 1e-9, 1e-9, &eq);
598 if (!eq) return false;
599 euler_consts[i] = v1;
600 } else {
601 double val = try_eval_double(ratio);
602 if (std::isnan(val)) return false;
603 euler_consts[i] = val;
604 }
605 }
606
607 return true;
608}

611ODEClassification classify_first_order_ode(
612 const std::shared_ptr<SymbolicExpr>& rhs,
613 const std::string& x,
614 const std::string& y)
615{
616 ODEClassification result;
617 result.order = 1;
618
619 if (!rhs || !lamina::detail::node(rhs)) {
620 result.type = ODEType::Separable;
621 return result;
622 }
623
624 /// 1. 检测可分离变量
625 if (is_separable(rhs, x, y)) {
626 result.type = ODEType::Separable;
627 return result;
628 }
629
630 /// 2. 检测一阶线性
631 std::shared_ptr<SymbolicExpr> P, Q;
632 if (is_linear_first_order(rhs, x, y, P, Q)) {
633 result.type = ODEType::Linear1;
634 result.P_coeff = P;
635 result.Q_coeff = Q;
636 return result;
637 }
638
639 /// 3. 检测齐次方程
640 if (is_homogeneous_ode(rhs, x, y)) {
641 result.type = ODEType::Homogeneous;
642 return result;
643 }
644
645 /// 4. 检测 Bernoulli
646 std::shared_ptr<SymbolicExpr> bP, bQ;
647 int bn;
648 if (is_bernoulli_ode(rhs, x, y, bP, bQ, bn)) {
649 result.type = ODEType::Bernoulli;
650 result.bernoulli_P = bP;
651 result.bernoulli_Q = bQ;
652 result.bernoulli_n = bn;
653 return result;
654 }
655
656 /// 5. 检测恰当方程
657 /// 将 y' = rhs 改写为 M + N*y' = 0 形式: -rhs + y' = 0
658 /// 即 M = -rhs, N = 1（标准形式 M*dx + N*dy = 0 → M + N*dy/dx = 0）
659 /// 更一般地: 若 rhs = f(x,y)，则 -f(x,y)*dx + dy = 0
660 /// M = -f(x,y), N = 1
661 auto neg_rhs = SymbolicExpr::multiply(SymbolicExpr::number(-1), rhs)->simplify();
662 auto one = SymbolicExpr::number(1);
663 if (is_exact_ode(neg_rhs, one, x, y)) {
664 result.type = ODEType::Exact;
665 result.exact_M = neg_rhs;
666 result.exact_N = one;
667 return result;
668 }
669
670 result.type = ODEType::Unknown;
671 return result;
672}

674
675ODEClassification classify_higher_order_ode(
676 const std::vector<std::shared_ptr<SymbolicExpr>>& coeffs,
677 const std::shared_ptr<SymbolicExpr>& forcing,
678 const std::string& x,
679 const std::string&)
680{
681 ODEClassification result;
682 result.order = static_cast<int>(coeffs.size()) - 1;
683
684 if (coeffs.empty()) {
685 result.type = ODEType::Unknown;
686 return result;
687 }
688
689 /// 1. 检测常系数
690 if (is_constant_coefficient(coeffs, x)) {
691 if (result.order == 2) {
692 result.type = ODEType::Linear2_ConstCoeff;
693 } else {
694 result.type = ODEType::HigherOrder_ConstCoeff;
695 }
696
697 /// 提取数值系数
698 result.const_coeffs.clear();
699 for (const auto& c : coeffs) {
700 double val = try_eval_double(c);
701 result.const_coeffs.push_back(std::isnan(val) ? 0.0 : val);
702 }
703 result.forcing_func = forcing;
704 return result;
705 }
706
707 /// 2. 检测 Euler 方程
708 std::vector<double> euler_consts;
709 if (is_euler_equation(coeffs, x, euler_consts)) {
710 result.type = ODEType::Euler;
711 result.euler_coeffs = euler_consts;
712 result.euler_forcing = forcing;
713 return result;
714 }
715
716 result.type = ODEType::Unknown;
717 return result;
718}
