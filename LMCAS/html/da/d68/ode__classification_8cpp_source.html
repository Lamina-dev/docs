26/**
27 * @internal
28 * @brief 尝试将表达式求值为 double。
29 * @return 若表达式为纯数值返回其值，否则返回 NaN。
30 */
31double try_eval_double(const std::shared_ptr<SymbolicExpr>& expr) {
32 if (!expr || !LMCAS::detail::node(expr)) return std::numeric_limits<double>::quiet_NaN();
33 if (expr->is_number()) {
34 auto node = std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(expr));
35 if (!node) return std::numeric_limits<double>::quiet_NaN();
36 if (std::holds_alternative<BigInt>(node->value()))
37 return std::get<BigInt>(node->value()).to_double();
38 if (std::holds_alternative<Rational>(node->value()))
39 return std::get<Rational>(node->value()).to_double();
40 return static_cast<double>(std::get<lmmc_real_t>(node->value()));
41 }
42 return std::numeric_limits<double>::quiet_NaN();
43}

243bool is_separable(
244 const std::shared_ptr<SymbolicExpr>& rhs,
245 const std::string& x,
246 const std::string& y)
247{
248 if (!rhs || !LMCAS::detail::node(rhs)) return false;
249 if (!valid_classifier_variables(x, y)) return false;
250
251 bool has_x = expression_depends_on_variable(LMCAS::detail::node(rhs), x);
252 bool has_y = expression_depends_on_variable(LMCAS::detail::node(rhs), y);
253
254 /// 若只依赖一个变量或都不依赖，则可分离
255 if (!has_x || !has_y) return true;
256
257 /// 检查乘法形式 f(x)*g(y)
258 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(LMCAS::detail::node(rhs));
259 if (mul) {
260 /// 将因子分为仅含 x 的和仅含 y 的
261 bool all_separable = true;
262 for (const auto& factor : mul->operands()) {
263 bool fx = expression_depends_on_variable(factor, x);
264 bool fy = expression_depends_on_variable(factor, y);
265 if (fx && fy) {
266 all_separable = false;
267 break;
268 }
269 }
270 if (all_separable) return true;
271 }
272
273 /// 检查除法形式 f(x)/g(y) 或 g(y)/f(x)
274 auto pow = std::dynamic_pointer_cast<const PowerNode>(LMCAS::detail::node(rhs));
275 if (pow) {
276 auto exp_node = std::dynamic_pointer_cast<const NumberNode>(pow->exponent());
277 if (exp_node) {
278 double exp_val = try_eval_double(LMCAS::detail::make_expression_ptr(pow->exponent()));
279 int eq;
280 lmmc_double_nearly_equal_tol(exp_val, -1.0, 1e-12, 1e-12, &eq);
281 if (eq) {
282 /// rhs = base^(-1) = 1/base; 若 base 仅含一个变量则可分离
283 bool base_x = expression_depends_on_variable(pow->base(), x);
284 bool base_y = expression_depends_on_variable(pow->base(), y);
285 if ((base_x && !base_y) || (!base_x && base_y)) return true;
286 }
287 }
288 }
289
290 return false;
291}

294bool is_linear_first_order(
295 const std::shared_ptr<SymbolicExpr>& rhs,
296 const std::string& x,
297 const std::string& y,
298 std::shared_ptr<SymbolicExpr>& P,
299 std::shared_ptr<SymbolicExpr>& Q)
300{
301 P.reset();
302 Q.reset();
303 if (!rhs || !LMCAS::detail::node(rhs) ||
304 !valid_classifier_variables(x, y)) {
305 return false;
306 }
307
308 /// 方程形式: y' = rhs(x, y)
309 /// 线性形式: y' + P(x)*y = Q(x) → y' = Q(x) - P(x)*y → rhs = Q(x) - P(x)*y
310 /// 即 rhs 关于 y 是线性的: rhs = A(x) + B(x)*y，其中 Q = A, P = -B
311
312 /// 若 rhs 不依赖 y，则 P=0, Q=rhs
313 if (!expression_depends_on_variable(LMCAS::detail::node(rhs), y)) {
314 P = SymbolicExpr::number(0);
315 Q = rhs;
316 return true;
317 }
318
319 /// 对 rhs 关于 y 求导，若结果不依赖 y，则 rhs 关于 y 是线性的
320 auto drhs_dy = rhs->differentiate(y);
321 if (!drhs_dy || expression_depends_on_variable(LMCAS::detail::node(drhs_dy), y)) {
322 return false;
323 }
324
325 /// rhs = A(x) + B(x)*y，其中 B(x) = ∂rhs/∂y
326 /// 计算 A(x) = rhs|_{y=0}
327 auto A = rhs->substitute(y, SymbolicExpr::number(0));
328 if (!A) return false;
329
330 /// 验证 A 不依赖 y
331 if (expression_depends_on_variable(LMCAS::detail::node(A), y)) return false;
332
333 /// B(x) = drhs_dy（已验证不依赖 y）
334 /// 线性形式: y' = A(x) + B(x)*y → y' - B(x)*y = A(x) → y' + (-B(x))*y = A(x)
335 /// 所以 P = -B(x), Q = A(x)
336 P = SymbolicExpr::multiply(SymbolicExpr::number(-1), drhs_dy)->simplify();
337 Q = A->simplify();
338 return true;
339}

342bool is_homogeneous_ode(
343 const std::shared_ptr<SymbolicExpr>& rhs,
344 const std::string& x,
345 const std::string& y)
346{
347 if (!rhs || !LMCAS::detail::node(rhs)) return false;
348 if (!valid_classifier_variables(x, y)) return false;
349
350 /// 齐次方程: f(tx, ty) = f(x, y) 对所有 t 成立
351 /// 用 t=2 进行数值测试：f(2x, 2y) 应等于 f(x, y)
352 auto t_val = SymbolicExpr::number(2);
353 auto tx = SymbolicExpr::multiply(t_val, SymbolicExpr::variable(x));
354 auto ty = SymbolicExpr::multiply(t_val, SymbolicExpr::variable(y));
355
356 auto f_scaled = rhs->substitute(x, tx);
357 f_scaled = f_scaled->substitute(y, ty);
358 f_scaled = f_scaled->simplify();
359
360 /// 计算 f_scaled - rhs，若化简为零则齐次
361 auto diff = SymbolicExpr::add(f_scaled,
362 SymbolicExpr::multiply(SymbolicExpr::number(-1), rhs));
363 diff = diff->simplify();
364
365 if (diff->is_zero()) return true;
366
367 ComputationContext context;
368 auto verified = check_zero_residual(diff, context);
369 return verified &&
370 std::holds_alternative<ProvedZeroResidual>(verified.value());
371}

374static std::optional<int> bernoulli_integer_exponent(const NumberNode& number)
375{
376 std::optional<std::int64_t> value;
377 if (std::holds_alternative<BigInt>(number.value())) {
378 value = std::get<BigInt>(number.value()).try_to_int64();
379 } else if (std::holds_alternative<Rational>(number.value())) {
380 const auto& rational = std::get<Rational>(number.value());
381 if (!rational.is_integer()) return std::nullopt;
382 value = rational.to_BigInt().try_to_int64();
383 } else {
384 const lmmc_real_t approximate =
385 std::get<lmmc_real_t>(number.value());
386 if (!std::isfinite(approximate) ||
387 approximate != std::floor(approximate) ||
388 approximate < std::numeric_limits<int>::min() ||
389 approximate > std::numeric_limits<int>::max()) {
390 return std::nullopt;
391 }
392 return static_cast<int>(approximate);
393 }
394 if (!value || *value < std::numeric_limits<int>::min() ||
395 *value > std::numeric_limits<int>::max()) {
396 return std::nullopt;
397 }
398 return static_cast<int>(*value);
399}

401static bool extract_bernoulli_monomial(
402 const std::shared_ptr<const SymbolicNode>& term,
403 const std::string& y,
404 int& exponent,
405 std::shared_ptr<SymbolicExpr>& coefficient)
406{
407 exponent = 0;
408 coefficient = SymbolicExpr::number(1);
409 std::vector<std::shared_ptr<const SymbolicNode>> factors;
410 if (auto product = std::dynamic_pointer_cast<const MultiplyNode>(term)) {
411 factors = product->operands();
412 } else {
413 factors.push_back(term);
414 }
415
416 for (const auto& factor : factors) {
417 if (auto variable =
418 std::dynamic_pointer_cast<const VariableNode>(factor);
419 variable && variable->name() == y) {
420 if (exponent == std::numeric_limits<int>::max()) return false;
421 ++exponent;
422 continue;
423 }
424
425 if (auto power = std::dynamic_pointer_cast<const PowerNode>(factor)) {
426 auto base =
427 std::dynamic_pointer_cast<const VariableNode>(power->base());
428 auto power_value =
429 std::dynamic_pointer_cast<const NumberNode>(power->exponent());
430 if (base && base->name() == y && power_value) {
431 auto value = bernoulli_integer_exponent(*power_value);
432 if (!value ||
433 (*value > 0 &&
434 exponent > std::numeric_limits<int>::max() - *value) ||
435 (*value < 0 &&
436 exponent < std::numeric_limits<int>::min() - *value)) {
437 return false;
438 }
439 exponent += *value;
440 continue;
441 }
442 }
443
444 if (expression_depends_on_variable(factor, y)) return false;
445 coefficient = SymbolicExpr::multiply(
446 coefficient, LMCAS::detail::make_expression_ptr(factor))->simplify();
447 }
448 return true;
449}

451bool is_bernoulli_ode(
452 const std::shared_ptr<SymbolicExpr>& rhs,
453 const std::string& x,
454 const std::string& y,
455 std::shared_ptr<SymbolicExpr>& P,
456 std::shared_ptr<SymbolicExpr>& Q,
457 int& n)
458{
459 P.reset();
460 Q.reset();
461 n = 0;
462 if (!rhs || !LMCAS::detail::node(rhs) ||
463 !valid_classifier_variables(x, y)) return false;
464 if (!expression_depends_on_variable(LMCAS::detail::node(rhs), y)) {
465 return false;
466 }
467
468 std::vector<std::shared_ptr<const SymbolicNode>> terms;
469 if (auto sum =
470 std::dynamic_pointer_cast<const AddNode>(LMCAS::detail::node(rhs))) {
471 terms = sum->operands();
472 } else {
473 terms.push_back(LMCAS::detail::node(rhs));
474 }
475
476 std::map<int, std::shared_ptr<SymbolicExpr>> coefficients;
477 for (const auto& term : terms) {
478 int exponent = 0;
479 std::shared_ptr<SymbolicExpr> coefficient;
480 if (!extract_bernoulli_monomial(
481 term, y, exponent, coefficient)) {
482 return false;
483 }
484 auto& combined = coefficients[exponent];
485 combined = combined
486 ? SymbolicExpr::add(combined, coefficient)->simplify()
487 : coefficient->simplify();
488 }
489
490 auto linear_coefficient = SymbolicExpr::number(0);
491 std::shared_ptr<SymbolicExpr> nonlinear_coefficient;
492 int nonlinear_exponent = 0;
493 for (auto& [exponent, coefficient] : coefficients) {
494 coefficient = coefficient->simplify();
495 if (coefficient->is_zero()) continue;
496 if (exponent == 1) {
497 linear_coefficient = coefficient;
498 continue;
499 }
500 if (exponent == 0 || nonlinear_coefficient) return false;
501 nonlinear_exponent = exponent;
502 nonlinear_coefficient = coefficient;
503 }
504
505 if (!nonlinear_coefficient ||
506 nonlinear_exponent == 0 || nonlinear_exponent == 1) {
507 return false;
508 }
509
510 P = SymbolicExpr::multiply(
511 SymbolicExpr::number(-1), linear_coefficient)->simplify();
512 Q = nonlinear_coefficient->simplify();
513 n = nonlinear_exponent;
514 return true;
515}

518bool is_exact_ode(
519 const std::shared_ptr<SymbolicExpr>& M,
520 const std::shared_ptr<SymbolicExpr>& N,
521 const std::string& x,
522 const std::string& y)
523{
524 if (!M || !LMCAS::detail::node(M) ||
525 !N || !LMCAS::detail::node(N) ||
526 !valid_classifier_variables(x, y)) return false;
527
528 /// 恰当条件: ∂M/∂y = ∂N/∂x
529 auto dM_dy = M->differentiate(y);
530 auto dN_dx = N->differentiate(x);
531
532 if (!dM_dy || !dN_dx) return false;
533
534 dM_dy = dM_dy->simplify();
535 dN_dx = dN_dx->simplify();
536 if (dM_dy->compare(dN_dx) == 0) return true;
537
538 /// 计算差值并化简
539 auto diff = SymbolicExpr::add(dM_dy,
540 SymbolicExpr::multiply(SymbolicExpr::number(-1), dN_dx));
541 diff = diff->simplify();
542
543 if (diff->is_zero()) return true;
544
545 ComputationContext context;
546 auto verified = check_zero_residual(diff, context);
547 return verified &&
548 std::holds_alternative<ProvedZeroResidual>(verified.value());
549
550}

566bool is_euler_equation(
567 const std::vector<std::shared_ptr<SymbolicExpr>>& coeffs,
568 const std::string& x,
569 std::vector<double>& euler_consts)
570{
571 euler_consts.clear();
572 if (coeffs.empty() || x.empty()) return false;
573 if (!coeffs.front() || !LMCAS::detail::node(coeffs.front()) ||
574 coeffs.front()->is_zero()) {
575 return false;
576 }
577
578 const int highest_order = static_cast<int>(coeffs.size()) - 1;
579 std::vector<double> extracted(coeffs.size(), 0.0);
580 for (int index = 0; index <= highest_order; ++index) {
581 const int derivative_order = highest_order - index;
582 if (!coeffs[index] || !LMCAS::detail::node(coeffs[index])) {
583 return false;
584 }
585 if (coeffs[index]->is_zero()) continue;
586
587 std::shared_ptr<SymbolicExpr> ratio;
588 if (derivative_order == 0) {
589 ratio = coeffs[index]->simplify();
590 } else {
591 auto x_power = SymbolicExpr::power(
592 SymbolicExpr::variable(x),
593 SymbolicExpr::number(derivative_order));
594 ratio = SymbolicExpr::divide(
595 coeffs[index], x_power)->simplify();
596 }
597 if (!ratio || !LMCAS::detail::node(ratio)) return false;
598
599 if (expression_depends_on_variable(
600 LMCAS::detail::node(ratio), x)) {
601 return false;
602 }
603
604 const double value = try_eval_double(ratio);
605 if (!std::isfinite(value)) return false;
606 extracted[index] = value;
607 }
608
609 euler_consts = std::move(extracted);
610 return true;
611}

614ODEClassification classify_first_order_ode(
615 const std::shared_ptr<SymbolicExpr>& rhs,
616 const std::string& x,
617 const std::string& y)
618{
619 ODEClassification result;
620 result.order = 1;
621
622 if (!rhs || !LMCAS::detail::node(rhs)) {
623 result.type = ODEType::Unknown;
624 return result;
625 }
626
627 /// 1. 检测可分离变量
628 if (is_separable(rhs, x, y)) {
629 result.type = ODEType::Separable;
630 return result;
631 }
632
633 /// 2. 检测一阶线性
634 std::shared_ptr<SymbolicExpr> P, Q;
635 if (is_linear_first_order(rhs, x, y, P, Q)) {
636 result.type = ODEType::Linear1;
637 result.P_coeff = P;
638 result.Q_coeff = Q;
639 return result;
640 }
641
642 /// 3. 检测齐次方程
643 if (is_homogeneous_ode(rhs, x, y)) {
644 result.type = ODEType::Homogeneous;
645 return result;
646 }
647
648 /// 4. 检测 Bernoulli
649 std::shared_ptr<SymbolicExpr> bP, bQ;
650 int bn;
651 if (is_bernoulli_ode(rhs, x, y, bP, bQ, bn)) {
652 result.type = ODEType::Bernoulli;
653 result.bernoulli_P = bP;
654 result.bernoulli_Q = bQ;
655 result.bernoulli_n = bn;
656 return result;
657 }
658
659 /// 5. 检测恰当方程
660 /// 将 y' = rhs 改写为 M + N*y' = 0 形式: -rhs + y' = 0
661 /// 即 M = -rhs, N = 1（标准形式 M*dx + N*dy = 0 → M + N*dy/dx = 0）
662 /// 更一般地: 若 rhs = f(x,y)，则 -f(x,y)*dx + dy = 0
663 /// M = -f(x,y), N = 1
664 auto neg_rhs = SymbolicExpr::multiply(SymbolicExpr::number(-1), rhs)->simplify();
665 auto one = SymbolicExpr::number(1);
666 if (is_exact_ode(neg_rhs, one, x, y)) {
667 result.type = ODEType::Exact;
668 result.exact_M = neg_rhs;
669 result.exact_N = one;
670 return result;
671 }
672
673 result.type = ODEType::Unknown;
674 return result;
675}

678ODEClassification classify_higher_order_ode(
679 const std::vector<std::shared_ptr<SymbolicExpr>>& coeffs,
680 const std::shared_ptr<SymbolicExpr>& forcing,
681 const std::string& x,
682 const std::string&)
683{
684 ODEClassification result;
685 result.order = static_cast<int>(coeffs.size()) - 1;
686
687 if (coeffs.empty()) {
688 result.type = ODEType::Unknown;
689 return result;
690 }
691
692 /// 1. 检测当前数值分类结果能够表示的常系数
693 if (is_constant_coefficient(coeffs, x)) {
694 std::vector<double> numeric_coeffs;
695 numeric_coeffs.reserve(coeffs.size());
696 for (const auto& c : coeffs) {
697 double val = try_eval_double(c);
698 if (!std::isfinite(val)) {
699 result.type = ODEType::Unknown;
700 return result;
701 }
702 numeric_coeffs.push_back(val);
703 }
704
705 result.type = result.order == 2
706 ? ODEType::Linear2_ConstCoeff
707 : ODEType::HigherOrder_ConstCoeff;
708 result.const_coeffs = std::move(numeric_coeffs);
709 result.forcing_func = forcing;
710 return result;
711 }
712
713 /// 2. 检测 Euler 方程
714 std::vector<double> euler_consts;
715 if (is_euler_equation(coeffs, x, euler_consts)) {
716 result.type = ODEType::Euler;
717 result.euler_coeffs = euler_consts;
718 result.euler_forcing = forcing;
719 return result;
720 }
721
722 result.type = ODEType::Unknown;
723 return result;
724}
