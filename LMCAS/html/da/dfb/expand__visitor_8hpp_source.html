28 void visit(const AddNode& node) override {
29 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
30 for (auto& op : node.operands()) {
31 op->accept(*this);
32 new_ops.push_back(result);
33 }
34 result = LMCAS::detail::make_node<AddNode>(new_ops);
35 }
36
37 void visit(const MultiplyNode& node) override {
38 std::vector<std::shared_ptr<const SymbolicNode>> expanded_ops;
39 for (auto& op : node.operands()) {
40 op->accept(*this);
41 expanded_ops.push_back(result);
42 }
43
44 if (expanded_ops.empty()) {
45 result = LMCAS::detail::make_node<NumberNode>(BigInt(1));
46 return;
47 }
48
49 NormalizationVisitor norm;
50 std::shared_ptr<const SymbolicNode> current = expanded_ops[0];
51 for (size_t i = 1; i < expanded_ops.size(); ++i) {
52 current = norm.expand_product(current, expanded_ops[i]);
53 }
54 result = current;
55 }
56
57 void visit(const PowerNode& node) override {
58 node.base()->accept(*this);
59 auto expanded_base = result;
60 node.exponent()->accept(*this);
61 auto expanded_exp = result;
62
63 if (auto exp_num = std::dynamic_pointer_cast<const NumberNode>(expanded_exp)) {
64 long long e_val = -1;
65 if (std::holds_alternative<BigInt>(exp_num->value())) e_val = std::get<BigInt>(exp_num->value()).to_int();
66 else if (std::holds_alternative<lmmc_real_t>(exp_num->value())) e_val = (long long)std::get<lmmc_real_t>(exp_num->value());
67 else if (std::holds_alternative<Rational>(exp_num->value())) e_val = (long long)std::get<Rational>(exp_num->value()).to_double();
68
69 if (e_val == 0) {
70 result = LMCAS::detail::make_node<NumberNode>(BigInt(1));
71 return;
72 }
73 if (e_val > 0 && e_val < 20) {
74 NormalizationVisitor norm;
75 std::shared_ptr<const SymbolicNode> current = expanded_base;
76 for (int i = 1; i < (int)e_val; ++i) {
77 current = norm.expand_product(current, expanded_base);
78 }
79 result = current;
80 return;
81 }
82 // Exponent too large for expansion (>= 20) or negative — keep as PowerNode
83 }
84 result = LMCAS::detail::make_node<PowerNode>(expanded_base, expanded_exp);
85 }

87 void visit(const FunctionNode& node) override {
88 std::vector<std::shared_ptr<const SymbolicNode>> new_args;
89 for (auto& arg : node.arguments()) {
90 arg->accept(*this);
91 new_args.push_back(result);
92 }
93 result = LMCAS::detail::make_node<FunctionNode>(node.type(), new_args);
94 }
95 void visit(const UninterpretedFunctionNode& node) override {

99 argument->accept(*this);
100 arguments.push_back(result);
101 }
102 result = LMCAS::detail::make_node<UninterpretedFunctionNode>(
103 node.name(), std::move(arguments));
104 }

106 void visit(const MatrixNode& node) override {
107 result = node.clone();
108 }
109 void visit(const IntegralNode& node) override {
110 node.body()->accept(*this);
111 auto body = result;
112 std::shared_ptr<const SymbolicNode> lower;
113 std::shared_ptr<const SymbolicNode> upper;
114 if (node.lower()) {
115 node.lower()->accept(*this);
116 lower = result;
117 node.upper()->accept(*this);
118 upper = result;
119 }
120 result = LMCAS::detail::make_node<IntegralNode>(
121 body, node.variable(), lower, upper);
122 }

124 void visit(const RelationalNode& node) override {
125 node.left()->accept(*this);
126 auto head = result;
127 node.right()->accept(*this);
128 auto tail = result;
129 result = LMCAS::detail::make_node<RelationalNode>(head, tail, node.op());
130 }
131
132 void visit(const LogicalNode& node) override {
133 node.left()->accept(*this);
134 auto head = result;
135 std::shared_ptr<const SymbolicNode> tail = nullptr;
136 if (node.right()) {
137 node.right()->accept(*this);
138 tail = result;
139 }
140 result = LMCAS::detail::make_node<LogicalNode>(head, tail, node.op());
141 }

143 void visit(const PiecewiseNode& node) override {
144 std::vector<PiecewiseNode::Branch> branches;
145 branches.reserve(node.branches().size());
146 for (const auto& branch : node.branches()) {
147 branch.expression->accept(*this);
148 auto expression = result;
149 branch.condition->accept(*this);
150 auto condition = result;
151 branches.push_back({expression, condition});
152 }
153 std::shared_ptr<const SymbolicNode> default_expr = nullptr;
154 if (node.default_expr()) {
155 node.default_expr()->accept(*this);
156 default_expr = result;
157 }
158 result = LMCAS::detail::make_node<PiecewiseNode>(std::move(branches), default_expr);
159 }

161 void visit(const SummationNode& node) override {
162 node.body()->accept(*this);
163 auto body = result;
164 node.lower_bound()->accept(*this);
165 auto lower = result;
166 node.upper_bound()->accept(*this);
167 auto upper = result;
168 result = LMCAS::detail::make_node<SummationNode>(body, node.index_var(), lower, upper);
169 }

179 }

182 node.body()->accept(*this);

206 void visit(const FiniteSetNode& node) override {
207 std::vector<std::shared_ptr<const SymbolicNode>> elements;
208 elements.reserve(node.elements().size());
209 for (const auto& element : node.elements()) {
210 element->accept(*this);
211 elements.push_back(result);
212 }
213 result = LMCAS::detail::make_node<FiniteSetNode>(std::move(elements));
214 }
215
216 void visit(const IntervalNode& node) override {
217 node.lower()->accept(*this);
218 auto lower = result;
219 node.upper()->accept(*this);
220 result = LMCAS::detail::make_node<IntervalNode>(
221 lower, result, node.lower_closed(), node.upper_closed());
222 }
223
224 void visit(const MembershipNode& node) override {
225 node.element()->accept(*this);
226 auto element = result;
227 node.set()->accept(*this);
228 result = LMCAS::detail::make_node<MembershipNode>(element, result);
229 }
230
231 void visit(const QuantityNode& node) override {
232 node.value()->accept(*this);
233 result = LMCAS::detail::make_node<QuantityNode>(
234 result, node.dimension(), node.scale_to_base(), node.display_unit());
235 }
236
237 void visit(const ComplexNode& node) override {
238 node.real()->accept(*this);
239 auto expanded_r = result;
240 node.imag()->accept(*this);
241 auto expanded_i = result;
242 result = SymbolicFactory::create_complex(expanded_r, expanded_i);
243 }
244 void visit(const LimitNode& node) override {
245 node.body()->accept(*this);
246 auto body = result;
247 node.point()->accept(*this);
248 result = LMCAS::detail::make_node<LimitNode>(
249 body, node.variable(), result, node.direction());
250 }
