71 void visit(const MatrixNode& node) override {
72 std::size_t size = 1;
73 if (rewrite_budget()) {
74 if (const auto* dense = std::get_if<MatrixNode::DenseStorage>(&node.storage())) {
75 check_child_count(dense->size());
76 for (const auto& element : *dense) {
77 append_child(size, element);
78 }
79 } else {
80 const auto& sparse = std::get<MatrixNode::SparseStorage>(node.storage());
81 check_child_count(sparse.size());
82 for (const auto& entry : sparse) {
83 append_child(size, entry.second);
84 }
85 }
86 }
87 set_result(node.clone(), size);

119 void visit(const PiecewiseNode& node) override {
120 std::vector<PiecewiseNode::Branch> branches;
121 if (rewrite_budget()) {
122 auto minimum = rewrite_budget()->append_size(1, node.branches().size());
123 rewrite_budget()->append_size(minimum, node.branches().size());
124 }
125 branches.reserve(node.branches().size());
126 std::size_t size = 1;
127 for (const auto& branch : node.branches()) {
128 auto expression = expand_child(branch.expression, size);
129 auto condition = expand_child(branch.condition, size);
130 branches.push_back({expression, condition});
131 }
132 std::shared_ptr<const SymbolicNode> default_expr = nullptr;
133 if (node.default_expr()) {
134 default_expr = expand_child(node.default_expr(), size);
135 }
136 set_result(LMCAS::detail::make_node<PiecewiseNode>(
137 std::move(branches), default_expr), size);
138 }

253 std::vector<std::shared_ptr<const SymbolicNode>> expand_children(
254 const std::vector<std::shared_ptr<const SymbolicNode>>& children, std::size_t& size) {
255 check_child_count(children.size(), size);
256 std::vector<std::shared_ptr<const SymbolicNode>> expanded;
257 expanded.reserve(children.size());
258 for (const auto& child : children) {
259 auto expanded_child = expand_child(child, size);
260 expanded.push_back(std::move(expanded_child));
261 }
262 return expanded;
263 }
