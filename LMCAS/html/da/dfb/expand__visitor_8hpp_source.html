26 void visit(const AddNode& node) override {
27 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
28 for (auto& op : node.operands()) {
29 op->accept(*this);
30 new_ops.push_back(result);
31 }

35 void visit(const MultiplyNode& node) override {
36 std::vector<std::shared_ptr<const SymbolicNode>> expanded_ops;
37 for (auto& op : node.operands()) {
38 op->accept(*this);
39 expanded_ops.push_back(result);
40 }
41
42 if (expanded_ops.empty()) {
43 result = lamina::detail::make_node<NumberNode>(BigInt(1));
44 return;
45 }
46
47 NormalizationVisitor norm;
48 std::shared_ptr<const SymbolicNode> current = expanded_ops[0];
49 for (size_t i = 1; i < expanded_ops.size(); ++i) {
50 current = norm.expand_product(current, expanded_ops[i]);
51 }
52 result = current;
53 }

55 void visit(const PowerNode& node) override {
56 node.base()->accept(*this);
57 auto expanded_base = result;
58 node.exponent()->accept(*this);
59 auto expanded_exp = result;
60
61 if (auto exp_num = std::dynamic_pointer_cast<const NumberNode>(expanded_exp)) {
62 long long e_val = -1;
63 if (std::holds_alternative<BigInt>(exp_num->value())) e_val = std::get<BigInt>(exp_num->value()).to_int();
64 else if (std::holds_alternative<lmmc_real_t>(exp_num->value())) e_val = (long long)std::get<lmmc_real_t>(exp_num->value());
65 else if (std::holds_alternative<Rational>(exp_num->value())) e_val = (long long)std::get<Rational>(exp_num->value()).to_double();
66
67 if (e_val == 0) {
68 result = lamina::detail::make_node<NumberNode>(BigInt(1));
69 return;
70 }
71 if (e_val > 0 && e_val < 20) {
72 NormalizationVisitor norm;
73 std::shared_ptr<const SymbolicNode> current = expanded_base;
74 for (int i = 1; i < (int)e_val; ++i) {
75 current = norm.expand_product(current, expanded_base);
76 }
77 result = current;
78 return;
79 }
80 // Exponent too large for expansion (>= 20) or negative — keep as PowerNode
81 }
82 result = lamina::detail::make_node<PowerNode>(expanded_base, expanded_exp);
83 }

141 void visit(const PiecewiseNode& node) override {
142 std::vector<PiecewiseNode::Branch> branches;
143 branches.reserve(node.branches().size());
144 for (const auto& branch : node.branches()) {
145 branch.expression->accept(*this);
146 auto expression = result;
147 branch.condition->accept(*this);
148 auto condition = result;
149 branches.push_back({expression, condition});
150 }
151 std::shared_ptr<const SymbolicNode> default_expr = nullptr;
152 if (node.default_expr()) {
153 node.default_expr()->accept(*this);
154 default_expr = result;
155 }
156 result = lamina::detail::make_node<PiecewiseNode>(std::move(branches), default_expr);
157 }
