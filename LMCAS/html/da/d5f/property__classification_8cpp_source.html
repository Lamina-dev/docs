12bool definiteness_conflicts(Definiteness existing, Definiteness incoming) {
13 using D = Definiteness;
14 struct Conflict {
15 D existing;
16 std::initializer_list<D> incoming;
17 };
18 const Conflict conflicts[] = {
19 {D::PositiveDefinite, {D::NegativeDefinite, D::NegativeSemiDefinite, D::Indefinite}},
20 {D::NegativeDefinite, {D::PositiveDefinite, D::PositiveSemiDefinite, D::Indefinite}},
21 {D::PositiveSemiDefinite, {D::NegativeDefinite, D::Indefinite}},
22 {D::NegativeSemiDefinite, {D::PositiveDefinite, D::Indefinite}},
23 {D::Indefinite, {D::PositiveDefinite, D::NegativeDefinite,
24 D::PositiveSemiDefinite, D::NegativeSemiDefinite}}
25 };
26 for (const auto& conflict : conflicts) {
27 if (existing != conflict.existing) {
28 continue;
29 }
30 return std::find(conflict.incoming.begin(), conflict.incoming.end(), incoming)
31 != conflict.incoming.end();
32 }
33 return false;
34}

82void PropertyStore::declare_finiteness_unchecked(const std::string& symbol, Finiteness f) {
83 auto& props = properties_[symbol];

85 if (props.finiteness == f) {
86 return;
87 }

127void PropertyStore::declare_definiteness_unchecked(const std::string& symbol, Definiteness d) {
128 auto& props = properties_[symbol];
129
130 if (props.definiteness == d) {
131 return;
132 }
133 if (d == Definiteness::Unknown) {
134 props.definiteness = d;
135 return;
136 }
137
138 if (props.definiteness != Definiteness::Unknown) {
139 if (definiteness_conflicts(props.definiteness, d)) {
140 throw std::invalid_argument(
141 "Contradiction for symbol '" + symbol +
142 "': definiteness conflict between existing and new declaration");
143 }
144 if (props.definiteness == Definiteness::PositiveSemiDefinite &&
145 d == Definiteness::PositiveDefinite) {
146 props.definiteness = d;
147 return;
148 }
149 if (props.definiteness == Definiteness::NegativeSemiDefinite &&
150 d == Definiteness::NegativeDefinite) {
151 props.definiteness = d;
152 return;
153 }
154 if (props.definiteness == Definiteness::PositiveDefinite &&
155 d == Definiteness::PositiveSemiDefinite) {
156 return; /** @brief 保留更强的正定声明。 */
157 }
158 if (props.definiteness == Definiteness::NegativeDefinite &&
159 d == Definiteness::NegativeSemiDefinite) {
160 return; /** @brief 保留更强的负定声明。 */
161 }
162 }
163
164 props.definiteness = d;
165}

202PropertyStoreResult PropertyStore::declare_periodic_checked(
203 const std::string& symbol, const std::string& variable,
204 const SymbolicExpr& period) {
205 if (symbol.empty() || variable.empty() || !detail::node(period)) {
206 return PropertyStoreResult::failure(CasErrc::InvalidArgument,
207 "symbol, independent variable, and period must be nonempty", "declare_periodic");
208 }
209 ComputationContext context;
210 auto defined = detail::query_definedness(
211 detail::node(period), detail::no_facts(), Domain::Real, context);
212 if (!defined) return PropertyStoreResult::failure(defined.error());
213 auto positive = detail::query_positive_value(
214 detail::node(period), detail::no_facts(), context);
215 if (!positive) return PropertyStoreResult::failure(positive.error());
216 if (defined.value() == Tribool::False || positive.value() == Tribool::False) {
217 return PropertyStoreResult::failure(CasErrc::InvalidArgument,
218 "period must be a positive defined real value", "declare_periodic");
219 }
220 return checked_property_update(*this, symbol, "declare_periodic",
221 [&](PropertyStore& candidate) {
222 candidate.declare_periodic_unchecked(symbol, variable, period);
223 });
224}
