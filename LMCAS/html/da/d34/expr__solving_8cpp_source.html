33static ExprSetResult project_finite_expr_set(const ExprPtr& original,
34 const std::string& variable,
35 ComputationContext& context,
36 const SolveOptions& options) {
37 auto solved = solve_set(original, variable, context, options);
38 if (!solved) {
39 return ExprSetResult::failure(solved.error());
40 }
41
42 const auto& solution_set = solved.value();
43 if (std::holds_alternative<EmptySolutions>(solution_set)) {
44 return expr_set({});
45 }
46 const auto* finite = std::get_if<FiniteSolutions>(&solution_set);
47 if (!finite) {
48 return expr_set_failure(
49 CasErrc::Inconclusive,
50 "solution set is not a finite enumerable set<Expr>",
51 kSolveExprSetOperation);
52 }
53
54 std::vector<ExprPtr> elements;
55 elements.reserve(finite->values.size());
56 for (const auto& solution : finite->values) {
57 if (!solution.conditions.empty()) {
58 return expr_set_failure(CasErrc::Inconclusive,
59 "conditional finite solutions cannot be lowered to set<Expr>",
60 kSolveExprSetOperation);
61 }
62 elements.push_back(solution.value);
63 }
64 return expr_set(std::move(elements));
65}
