50ExpressionResult matrix_reflection_checked(double angle, int dim,
51 ComputationContext& context) {
52 const std::string operation = "matrix_reflection";
53 auto step = context.consume_steps(1, operation);
54 if (!step) return ExpressionResult::failure(step.error());
55 if (dim == 2) {
56 if (!std::isfinite(angle)) {
57 return ExpressionResult::failure(CasErrc::InvalidArgument,
58 "reflection angle must be finite", operation);
59 }
60 const double cosine = std::cos(angle);
61 const double sine = std::sin(angle);
62 try {
63 auto c = SymbolicExpr::number(cosine * cosine - sine * sine);
64 auto s = SymbolicExpr::number(2.0 * sine * cosine);
65 return require_matrix_result(
66 SymbolicExpr::matrix({{c, s}, {s, SymbolicExpr::multiply(SymbolicExpr::number(-1), c)}}),
67 operation);
68 } catch (const CasError& error) {
69 return ExpressionResult::failure(error);
70 } catch (const std::bad_alloc&) {
71 return ExpressionResult::failure(CasErrc::ResourceLimit,
72 "allocation failed while constructing reflection matrix", operation);
73 } catch (const std::exception& error) {
74 return ExpressionResult::failure(CasErrc::InternalInvariant,
75 error.what(), operation);
76 }
77 }
78 return unsupported_matrix_dimension(operation);
79}
