41static std::optional<double> finite_numeric_value(
42 const std::shared_ptr<SymbolicExpr>& expr) {
43 auto evaluated = detail::try_finite_numeric(expr);
44 if (!evaluated) return std::nullopt;
45 auto simplified = expr->simplify();
46 if (simplified && lamina::detail::node(simplified)) {
47 if (auto num = std::dynamic_pointer_cast<const NumberNode>(
48 lamina::detail::node(simplified))) {
49 if (std::holds_alternative<Rational>(num->value())) {
50 const auto& value = std::get<Rational>(num->value());
51 if (!value.get_numerator().is_zero() && *evaluated == 0.0) {
52 return std::nullopt;

104

202 SymbolicExpr::number(real_part),

213 double cos_arg = -q_val / (2.0 * r_val);
214
215 if (cos_arg > 1.0) cos_arg = 1.0;
216 if (cos_arg < -1.0) cos_arg = -1.0;
217 double theta_val = std::acos(cos_arg);
218 double two_cbrt_r = 2.0 * std::cbrt(r_val);
219
220 for (int k = 0; k < 3; ++k) {
221 double angle = (theta_val + 2.0 * k * LMMC_CONST_PI) / 3.0;
222 double t_k_val = two_cbrt_r * std::cos(angle);
223 double x_k_val = t_k_val - shift_val;
224 roots.push_back(SymbolicExpr::number(x_k_val));
225 }
226 }
227 } else {

229 auto q_half = SymbolicExpr::divide(q, num(2));
230 auto p_third = SymbolicExpr::divide(p, num(3));
231 auto D_expr = SymbolicExpr::add(
232 SymbolicExpr::power(q_half, num(2)),
233 SymbolicExpr::power(p_third, num(3)))->simplify();
234
235 auto neg_q_half = negate(q_half)->simplify();
236 auto sqrt_D = SymbolicExpr::sqrt(D_expr);
237
238 auto u = cbrt_expr(SymbolicExpr::add(neg_q_half, sqrt_D))->simplify();
239 auto v = cbrt_expr(sub(neg_q_half, sqrt_D))->simplify();
240
241 auto t1 = SymbolicExpr::add(u, v);
242 auto x1 = sub(t1, shift)->simplify();
243
244 auto neg_half_sum = SymbolicExpr::divide(negate(SymbolicExpr::add(u, v)), num(2));
245 auto diff_uv = sub(u, v);
246 auto sqrt3_half = SymbolicExpr::divide(SymbolicExpr::sqrt(num(3)), num(2));
247 auto imag_coeff = SymbolicExpr::multiply(sqrt3_half, diff_uv);
248
249 auto i_unit = SymbolicExpr::sqrt(num(-1));
250 auto x2 = sub(SymbolicExpr::add(neg_half_sum, SymbolicExpr::multiply(i_unit, imag_coeff)), shift)->simplify();
251 auto x3 = sub(sub(neg_half_sum, SymbolicExpr::multiply(i_unit, imag_coeff)), shift)->simplify();
252
253 roots = { x1, x2, x3 };
254 }
255
256 return roots;

588

591 results.push_back(sub(pos_y, shift)->simplify());
592 results.push_back(sub(neg_y, shift)->simplify());

597 auto p2 = SymbolicExpr::power(p, num(2));

599 auto eight_p = SymbolicExpr::multiply(num(8), p);
600 auto two_p2_minus_8r = sub(SymbolicExpr::multiply(num(2), p2), SymbolicExpr::multiply(num(8), r))->simplify();
601 auto neg_q2 = negate(q2)->simplify();

607 m = cubic_roots[0];
608 }

610 if (!m) return {};

612 auto two_m = SymbolicExpr::multiply(num(2), m);
613 auto s = SymbolicExpr::sqrt(two_m)->simplify();
614

616
617 auto m_plus_p_half = SymbolicExpr::add(m, p_half)->simplify();
618

620

622

624 auto quad2_c = SymbolicExpr::add(m_plus_p_half, q_over_2s)->simplify();

626 auto y_roots1 = solve_quadratic_internal(num(1), s, quad1_c);
627 auto y_roots2 = solve_quadratic_internal(num(1), neg_s, quad2_c);
628
629 std::vector<std::shared_ptr<SymbolicExpr>> results;
630 for (const auto& y : y_roots1) {
631 results.push_back(sub(y, shift)->simplify());
632 }
633 for (const auto& y : y_roots2) {
634 results.push_back(sub(y, shift)->simplify());
635 }

640static std::vector<BigInt> positive_divisors(const BigInt& n) {
641 std::vector<BigInt> divs;
642 if (n.is_zero()) {
643
644 return divs;
645 }
646 BigInt abs_n = n.Abs();
647
648 BigInt i(1);
649 BigInt limit(1000);
650
651 while (i * i <= abs_n && i <= limit) {
652 BigInt rem = abs_n % i;
653 if (rem.is_zero()) {
654 divs.push_back(i);
655 BigInt counterpart = abs_n / i;
656 if (counterpart != i) {
657 divs.push_back(counterpart);
658 }
659 }
660 i = i + BigInt(1);
661 }
662
663 std::sort(divs.begin(), divs.end());
664 return divs;
665}

667std::vector<std::pair<Polynomial<Rational>, int>> square_free_factorization(
668 const Polynomial<Rational>& poly) {
669
670 std::vector<std::pair<Polynomial<Rational>, int>> factors;
671
672 if (poly.is_zero() || poly.degree() <= 0) {
673 return factors;
674 }
675
676 Polynomial<Rational> f = poly.make_monic();
677
678 if (f.degree() == 1) {
679 factors.push_back({f, 1});
680 return factors;
681 }
682
683 Polynomial<Rational> f_prime = f.differentiate();
684 Polynomial<Rational> g = Polynomial<Rational>::gcd(f, f_prime);
685
686 if (g.degree() <= 0) {
687 factors.push_back({f, 1});
688 return factors;
689 }
690
691 auto [w, w_rem] = f.div_mod(g);
692
693 int multiplicity = 1;
694
695 while (w.degree() > 0) {
696
697 Polynomial<Rational> y = Polynomial<Rational>::gcd(g, w);
698
699 auto [z, z_rem] = w.div_mod(y);
700
701 if (z.degree() > 0) {
702
703 factors.push_back({z.make_monic(), multiplicity});
704 }
705
706 auto [g_next, g_rem] = g.div_mod(y);
707 g = g_next;
708 w = y;
709 multiplicity++;
710 }
711
712 if (g.degree() > 0) {
713 factors.push_back({g.make_monic(), multiplicity});
714 }
715
716 return factors;
717}

719static std::vector<std::shared_ptr<SymbolicExpr>> solve_closed_form_poly(
720 const Polynomial<SymbolicPolyCoeff>& poly,
721 const std::string& var)
722{
723 int deg = poly.degree();
724 if (deg <= 0) return {};
725
726 auto get_coeff = [&](int d) -> std::shared_ptr<SymbolicExpr> {
727 if (d < 0 || d > deg) return SymbolicExpr::number(0);
728 return poly.coeffs[d].val ? poly.coeffs[d].val : SymbolicExpr::number(0);
729 };
730
731 if (deg == 1) {
732 auto a = get_coeff(1);
733 auto b = get_coeff(0);
734 auto neg_b = SymbolicExpr::multiply(b, SymbolicExpr::number(-1));
735 return { SymbolicExpr::divide(neg_b, a)->simplify() };
736 } else if (deg == 2) {
737 return solve_quadratic_internal(get_coeff(2), get_coeff(1), get_coeff(0));
738 } else if (deg == 3) {
739 return solve_cubic(get_coeff(3), get_coeff(2), get_coeff(1), get_coeff(0), var);
740 } else if (deg == 4) {
741 return solve_quartic(get_coeff(4), get_coeff(3), get_coeff(2), get_coeff(1), get_coeff(0), var);
742 }
743
744 return {};
745}

748 const Polynomial<SymbolicPolyCoeff>& sym_poly,
749 Polynomial<Rational>& out_poly)
750{
751 out_poly = Polynomial<Rational>(sym_poly.variable_name);
752 out_poly.coeffs.resize(sym_poly.coeffs.size(), Rational(0));
753
754 for (size_t i = 0; i < sym_poly.coeffs.size(); ++i) {
755 auto coeff_expr = sym_poly.coeffs[i].val;
756 if (!coeff_expr) {
757 out_poly.coeffs[i] = Rational(0);
758 continue;
759 }
760
761 auto simplified = coeff_expr->simplify();
762
763 if (lamina::detail::node(simplified)) {
764 struct VarCheck : public lamina::detail::RecursiveSymbolicVisitor {
765 bool found = false;
766 void visit(const NumberNode&) override {}
767 void visit(const VariableNode&) override { found = true; }
768 void visit(const AddNode& n) override { for (auto& op : n.operands()) { if (found) return; op->accept(*this); } }
769 void visit(const MultiplyNode& n) override { for (auto& op : n.operands()) { if (found) return; op->accept(*this); } }
770 void visit(const PowerNode& n) override { n.base()->accept(*this); if (!found) n.exponent()->accept(*this); }
771 void visit(const FunctionNode& n) override { for (auto& arg : n.arguments()) { if (found) return; arg->accept(*this); } }
772 void visit(const MatrixNode&) override {}
773 void visit(const RelationalNode& n) override { n.left()->accept(*this); if (!found) n.right()->accept(*this); }
774 void visit(const LogicalNode& n) override { n.left()->accept(*this); if (!found && n.right()) n.right()->accept(*this); }
775 void visit(const PiecewiseNode& n) override {
776 for (const auto& branch : n.branches()) {
777 if (found) return;
778 branch.expression->accept(*this);
779 if (!found) branch.condition->accept(*this);
780 }
781 if (!found && n.default_expr()) n.default_expr()->accept(*this);
782 }
783 void visit(const SummationNode& n) override { n.body()->accept(*this); if (!found) n.lower_bound()->accept(*this); if (!found) n.upper_bound()->accept(*this); }
784 void visit(const ProductNode& n) override { n.body()->accept(*this); if (!found) n.lower_bound()->accept(*this); if (!found) n.upper_bound()->accept(*this); }
785 void visit(const TransformNode& n) override { n.body()->accept(*this); }
786 void visit(const QuantifierNode& n) override { n.domain()->accept(*this); if (!found) n.predicate()->accept(*this); }
787 void visit(const SetBuilderNode& n) override { n.domain()->accept(*this); if (!found) n.predicate()->accept(*this); }
788 void visit(const ComplexNode& n) override { n.real()->accept(*this); if (!found) n.imag()->accept(*this); }
789 void visit(const FiniteSetNode& n) override { for (const auto& e : n.elements()) { if (found) return; e->accept(*this); } }
790 void visit(const IntervalNode& n) override { n.lower()->accept(*this); if (!found) n.upper()->accept(*this); }
791 void visit(const MembershipNode& n) override { n.element()->accept(*this); if (!found) n.set()->accept(*this); }
792 void visit(const QuantityNode& n) override { n.value()->accept(*this); }
793 } checker;
794 lamina::detail::node(simplified)->accept(checker);
795 if (checker.found) return false;
796 }
797
798 if (auto num_node = std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(simplified))) {
799 if (std::holds_alternative<Rational>(num_node->value())) {
800 out_poly.coeffs[i] = std::get<Rational>(num_node->value());
801 } else if (std::holds_alternative<BigInt>(num_node->value())) {
802 out_poly.coeffs[i] = Rational(std::get<BigInt>(num_node->value()));
803 } else {
804 out_poly.coeffs[i] = Rational::from_double(std::get<double>(num_node->value()));
805 }
806 } else if (simplified->is_zero()) {
807 out_poly.coeffs[i] = Rational(0);
808 } else if (simplified->is_one()) {
809 out_poly.coeffs[i] = Rational(1);
810 } else {
811 ComputationContext context;
812 auto evaluated = evaluate_numeric(*simplified, NumericBindings{}, context);
813 if (!evaluated || !evaluated.value().is_finite() ||
814 !std::isfinite(evaluated.value().value)) {
815 return false;
816 }
817 out_poly.coeffs[i] = Rational::from_double(evaluated.value().value);
818 }
819 }
820
821 out_poly.trim();
822 return true;
823}

836std::vector<std::shared_ptr<SymbolicExpr>> solve_by_factoring(
837 const Polynomial<SymbolicPolyCoeff>& poly,
838 const std::string& var) {
839
840 if (poly.is_zero() || poly.degree() < 1) return {};
841
842 if (poly.degree() <= 4) {
843 return solve_closed_form_poly(poly, var);
844 }
845
846 std::vector<std::shared_ptr<SymbolicExpr>> results;
847
848 Polynomial<Rational> rat_poly;
849 if (!convert_to_rational_poly(poly, rat_poly)) {
850
851 return make_rootof_solutions(poly, var);
852 }
853
854 auto rational_roots = find_rational_roots(rat_poly);
855
856 for (const auto& r : rational_roots) {
857 results.push_back(SymbolicExpr::number(r));
858 }
859
860 Polynomial<Rational> quotient = rat_poly;
861 for (const auto& r : rational_roots) {
862 Polynomial<Rational> linear_factor({-r, Rational(1)}, var);
863 auto [q, rem] = quotient.div_mod(linear_factor);
864 quotient = q;
865 }
866
867 if (quotient.degree() <= 0) {
868 return results;
869 }
870
871 if (quotient.degree() <= 4) {
872 auto sym_quotient = rational_to_symbolic_poly(quotient);
873 auto factor_roots = solve_closed_form_poly(sym_quotient, var);
874 results.insert(results.end(), factor_roots.begin(), factor_roots.end());
875 return results;
876 }
877
878 auto sqfree_factors = square_free_factorization(quotient);
879
880 for (const auto& [factor, multiplicity] : sqfree_factors) {
881 if (factor.degree() <= 0) continue;
882
883 std::vector<std::shared_ptr<SymbolicExpr>> factor_solutions;
884
885 if (factor.degree() <= 4) {
886
887 auto sym_factor = rational_to_symbolic_poly(factor);
888 factor_solutions = solve_closed_form_poly(sym_factor, var);
889 } else {
890
891 auto sym_factor = rational_to_symbolic_poly(factor);
892 factor_solutions = make_rootof_solutions(sym_factor, var);
893 }
894
895 for (int m = 0; m < multiplicity; ++m) {
896 results.insert(results.end(), factor_solutions.begin(), factor_solutions.end());
897 }
898 }
899
900 return results;
901}

903std::vector<Rational> find_rational_roots(const Polynomial<Rational>& poly) {
904 std::vector<Rational> roots;
905
906 if (poly.is_zero() || poly.degree() < 1) {
907 return roots;
908 }
909
910 Polynomial<Rational> current = poly;
911
912 while (current.degree() >= 1 && current.coeffs[0] == Rational(0)) {
913 roots.push_back(Rational(0));
914
915 std::vector<Rational> new_coeffs(current.coeffs.begin() + 1, current.coeffs.end());
916 current = Polynomial<Rational>(new_coeffs, current.variable_name);
917 }
918
919 while (current.degree() >= 1) {
920
921 Rational a0 = current.coeffs[0];
922 Rational an = current.lead_coeff();
923
924 if (a0 == Rational(0)) {
925
926 roots.push_back(Rational(0));
927 std::vector<Rational> new_coeffs(current.coeffs.begin() + 1, current.coeffs.end());
928 current = Polynomial<Rational>(new_coeffs, current.variable_name);
929 continue;
930 }
931
932 BigInt lcm_den(1);
933 for (const auto& c : current.coeffs) {
934 BigInt d = c.get_denominator();
935 lcm_den = BigInt::lcm(lcm_den, d);
936 }
937
938 BigInt int_a0 = (a0 * Rational(lcm_den)).get_numerator();
939 BigInt int_an = (an * Rational(lcm_den)).get_numerator();
940
941 std::vector<BigInt> p_divs = positive_divisors(int_a0);
942 std::vector<BigInt> q_divs = positive_divisors(int_an);
943
944 if (p_divs.empty() || q_divs.empty()) {
945 break;
946 }
947
948 bool found_root = false;
949 Rational found_r;
950
951 for (const auto& p : p_divs) {
952 for (const auto& q : q_divs) {
953
954 Rational candidate_pos(p, q);
955 Rational candidate_neg(-p, q);
956
957 if (current.eval(candidate_pos) == Rational(0)) {
958 found_root = true;
959 found_r = candidate_pos;
960 break;
961 }
962 if (current.eval(candidate_neg) == Rational(0)) {
963 found_root = true;
964 found_r = candidate_neg;
965 break;
966 }
967 }
968 if (found_root) break;
969 }
970
971 if (!found_root) {
972 break;
973 }
974
975 Polynomial<Rational> linear_factor({-found_r, Rational(1)}, current.variable_name);
976
977 while (current.degree() >= 1) {
978 auto [quotient, remainder] = current.div_mod(linear_factor);
979
980 if (!remainder.is_zero()) {
981 break;
982 }
983 roots.push_back(found_r);
984 current = quotient;
985
986 if (current.degree() < 1 || current.eval(found_r) != Rational(0)) {
987 break;
988 }
989 }
990 }
991
992 return roots;
993}
