12static std::vector<std::shared_ptr<SymbolicExpr>> solve_closed_form_poly(
13 const Polynomial<SymbolicPolyCoeff>& poly,
14 const std::string& var)
15{
16 int deg = poly.degree();
17 if (deg <= 0) return {};
18
19 auto get_coeff = [&](int d) -> std::shared_ptr<SymbolicExpr> {
20 if (d < 0 || d > deg) return SymbolicExpr::number(0);
21 return poly.coeffs[d].val ? poly.coeffs[d].val : SymbolicExpr::number(0);
22 };
23
24 if (deg == 1) {
25 auto a = get_coeff(1);
26 auto b = get_coeff(0);
27 auto neg_b = SymbolicExpr::multiply(b, SymbolicExpr::number(-1));
28 return { SymbolicExpr::divide(neg_b, a)->simplify() };
29 } else if (deg == 2) {
30 return solve_quadratic_internal(get_coeff(2), get_coeff(1), get_coeff(0));
31 } else if (deg == 3) {
32 return solve_cubic(get_coeff(3), get_coeff(2), get_coeff(1), get_coeff(0), var);
33 } else if (deg == 4) {
34 return solve_quartic(get_coeff(4), get_coeff(3), get_coeff(2), get_coeff(1), get_coeff(0), var);
35 }
36
37 return {};
38}

51static void append_factor_solutions(const Polynomial<Rational>& factor, int multiplicity,

62std::vector<std::shared_ptr<SymbolicExpr>> solve_by_factoring(
63 const Polynomial<SymbolicPolyCoeff>& poly,
64 const std::string& var) {
65
66 if (poly.is_zero() || poly.degree() < 1) { return {}; }
67
68 if (poly.degree() <= 2) {
69 return solve_closed_form_poly(poly, var);
70 }
71
72 std::vector<std::shared_ptr<SymbolicExpr>> results;
73
74 Polynomial<Rational> rat_poly;
75 if (!convert_to_rational_poly(poly, rat_poly)) {
76 if (poly.degree() <= 4) {
77 return solve_closed_form_poly(poly, var);
78 }
79 return make_rootof_solutions(poly, var);
80 }
81
82 // Keep rational boundaries exact before cubic/quartic floating formulas.
83 auto rational_roots = bounded_rational_root_search(rat_poly) ?
84 find_rational_roots(rat_poly) : std::vector<Rational>{};
85
86 for (const auto& r : rational_roots) {
87 results.push_back(SymbolicExpr::number(r));
88 }
89
90 Polynomial<Rational> quotient = rat_poly;
91 for (const auto& r : rational_roots) {
92 Polynomial<Rational> linear_factor({-r, Rational(1)}, var);
93 auto [q, rem] = quotient.div_mod(linear_factor);
94 if (!rem.is_zero()) { throw std::logic_error("solve_by_factoring: inexact root division"); }
95 quotient = q;
96 }
97
98 if (quotient.degree() <= 0) {
99 return results;
100 }
101
102 if (quotient.degree() <= 4) {
103 auto sym_quotient = rational_to_symbolic_poly(quotient);
104 auto factor_roots = solve_closed_form_poly(sym_quotient, var);
105 results.insert(results.end(), factor_roots.begin(), factor_roots.end());
106 return results;
107 }
108
109 auto sqfree_factors = square_free_factorization(quotient);
110
111 for (const auto& [factor, multiplicity] : sqfree_factors) {
112 append_factor_solutions(factor, multiplicity, var, results);
113 }
114
115 return results;
116}
