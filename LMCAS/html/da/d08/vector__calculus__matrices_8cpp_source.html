110std::shared_ptr<SymbolicExpr> hessian(
111 const std::shared_ptr<SymbolicExpr>& f, const std::vector<std::string>& vars)
112{
113 size_t n = vars.size();
114
115 std::vector<std::shared_ptr<SymbolicExpr>> first_partials;
116 first_partials.reserve(n);
117 for (size_t i = 0; i < n; ++i) {
118 first_partials.push_back(f->differentiate(vars[i]));
119 }
120
121 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> grid(n,
122 std::vector<std::shared_ptr<SymbolicExpr>>(n));
123
124 for (size_t i = 0; i < n; ++i) {
125 for (size_t j = i; j < n; ++j) {
126 auto second_partial = first_partials[i]->differentiate(vars[j]);
127 second_partial = second_partial->simplify();
128 grid[i][j] = second_partial;
129 if (i != j) {
130 grid[j][i] = second_partial;
131 }
132 }
133 }
134
135 return SymbolicExpr::matrix(grid);
136}
