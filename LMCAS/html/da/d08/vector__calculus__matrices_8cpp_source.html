57 std::size_t nodes = 1;
58 transform_budget.require_nodes(nodes);
59
60 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> grid;

63 for (size_t i = 0; i < m; ++i) {

119std::shared_ptr<SymbolicExpr> vector_calculus_detail::hessian_impl(
120 const std::shared_ptr<SymbolicExpr>& f, const std::vector<std::string>& vars,
121 detail::RewriteBudget& transform_budget)
122{
123 size_t n = vars.size();
124 std::size_t nodes = 1;
125 transform_budget.require_nodes(nodes);
126
127 std::vector<std::shared_ptr<SymbolicExpr>> first_partials;
128 first_partials.reserve(n);
129 for (size_t i = 0; i < n; ++i) {
130 first_partials.push_back(detail::make_expression_ptr(detail::differentiate_node(detail::node(f), vars[i], transform_budget)));
131 }
132
133 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> grid(n,
134 std::vector<std::shared_ptr<SymbolicExpr>>(n));
135
136 for (size_t i = 0; i < n; ++i) {
137 for (size_t j = i; j < n; ++j) {
138 auto second_partial = detail::make_expression_ptr(detail::differentiate_node(detail::node(first_partials[i]), vars[j], transform_budget));
139 second_partial = detail::make_expression_ptr(detail::normalize_node(detail::node(second_partial), transform_budget));
140 const auto partial_nodes = transform_budget.measure(detail::node(second_partial));
141 nodes = transform_budget.append_size(nodes, partial_nodes);
142 if (i != j) nodes = transform_budget.append_size(nodes, partial_nodes);
143 grid[i][j] = second_partial;
144 if (i != j) {
145 grid[j][i] = second_partial;
146 }
147 }
148 }
149
150 return SymbolicExpr::matrix(grid);
151}
