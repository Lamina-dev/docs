20/**
21 * Per-operation limits for APIs that accept a ComputationContext.
22 *
23 * Context-free BigInt and symbolic constructors are intentionally unbounded;
24 * use checked, context-aware entry points when processing untrusted input.

26struct ResourceLimits {

44class DiagnosticEngine {
45public:
46 using Consumer = std::function<void(const Diagnostic&)>;
47
48 explicit DiagnosticEngine(std::size_t limit, Consumer consumer = {})
49 : limit_(limit), consumer_(std::move(consumer)) {}
50
51 Result<void> emit(Diagnostic diagnostic) {
52 if (diagnostics_.size() >= limit_) {
53 return Result<void>::failure(CasErrc::ResourceLimit,
54 "diagnostic budget exhausted",
55 diagnostic.operation);
56 }
57 bool inserted = false;
58 try {
59 diagnostics_.push_back(std::move(diagnostic));
60 inserted = true;
61 if (consumer_) consumer_(diagnostics_.back());
62 } catch (const std::bad_alloc&) {
63 if (inserted) diagnostics_.pop_back();
64 return Result<void>::failure(CasErrc::ResourceLimit,
65 "diagnostic allocation failed");
66 } catch (...) {
67 if (inserted) diagnostics_.pop_back();
68 return Result<void>::failure(CasErrc::InternalInvariant,
69 "diagnostic consumer failed");
70 }
71 return Result<void>::success();
72 }
73
74 void set_consumer(Consumer consumer) { consumer_ = std::move(consumer); }
75
76 const std::vector<Diagnostic>& diagnostics() const noexcept {
77 return diagnostics_;
78 }
79
80private:
81 std::size_t limit_;
82 Consumer consumer_;
83 std::vector<Diagnostic> diagnostics_;
84};

112 Result<void> consume_steps(std::size_t amount = 1,
113 const std::string& operation = {}) {
114 auto access = check_access(operation);
115 if (!access) return access;
116 if (steps_used_ > limits_.max_steps ||

118 return Result<void>::failure(CasErrc::ResourceLimit,

131 }
132 ++recursion_depth_;
133 return Result<void>::success();
