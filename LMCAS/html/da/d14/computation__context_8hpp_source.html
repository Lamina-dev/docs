38class DiagnosticEngine {
39public:
40 using Consumer = std::function<void(const Diagnostic&)>;
41
42 explicit DiagnosticEngine(std::size_t limit, Consumer consumer = {})
43 : limit_(limit), consumer_(std::move(consumer)) {}
44
45 Result<void> emit(Diagnostic diagnostic) {
46 if (diagnostics_.size() >= limit_) {
47 return Result<void>::failure(CasErrc::ResourceLimit,
48 "diagnostic budget exhausted",
49 diagnostic.operation);
50 }
51 bool inserted = false;
52 try {
53 diagnostics_.push_back(std::move(diagnostic));
54 inserted = true;
55 if (consumer_) consumer_(diagnostics_.back());
56 } catch (const std::bad_alloc&) {
57 if (inserted) diagnostics_.pop_back();
58 return Result<void>::failure(CasErrc::ResourceLimit,
59 "diagnostic allocation failed");
60 } catch (...) {
61 if (inserted) diagnostics_.pop_back();
62 return Result<void>::failure(CasErrc::InternalInvariant,
63 "diagnostic consumer failed");
64 }
65 return Result<void>::success();
66 }
67
68 void set_consumer(Consumer consumer) { consumer_ = std::move(consumer); }
69
70 const std::vector<Diagnostic>& diagnostics() const noexcept {
71 return diagnostics_;
72 }
73
74private:
75 std::size_t limit_;
76 Consumer consumer_;
77 std::vector<Diagnostic> diagnostics_;
78};
