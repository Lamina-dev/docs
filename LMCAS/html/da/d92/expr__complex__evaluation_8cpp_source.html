27ApproxReal approx_part(double value) {
28 ApproxReal part;
29 part.value = value;
30 part.absolute_error = std::numeric_limits<double>::epsilon() *
31 std::max(1.0, std::abs(value)) * 4.0;

63Result<ApproxComplex> add_complex(const ApproxComplex& lhs,
64 const ApproxComplex& rhs) {
65 auto result = checked_complex(lhs.real.value + rhs.real.value,
66 lhs.imag.value + rhs.imag.value,
67 kEvalComplexOperation);
68 if (!result) return result;
69 result.value().real.absolute_error +=
70 lhs.real.absolute_error + rhs.real.absolute_error;
71 result.value().imag.absolute_error +=
72 lhs.imag.absolute_error + rhs.imag.absolute_error;
73 return result;
74}

76Result<ApproxComplex> multiply_complex(const ApproxComplex& lhs,
77 const ApproxComplex& rhs) {
78 const double a = lhs.real.value;
79 const double b = lhs.imag.value;
80 const double c = rhs.real.value;
81 const double d = rhs.imag.value;
82 const lmmc_complex_t left{a, b};
83 const lmmc_complex_t right{c, d};
84 lmmc_complex_t product;
85 if (lmmc_complex_mul(&left, &right, &product) != LMMC_STATUS_OK) {
86 return complex_failure(CasErrc::NumericFailure,
87 "complex multiplication requires finite components and result",
88 kEvalComplexOperation);
89 }
90 auto result = checked_complex(product.real, product.imag, kEvalComplexOperation);
91 if (!result) return result;
92 result.value().real.absolute_error +=
93 std::abs(c) * lhs.real.absolute_error +
94 std::abs(a) * rhs.real.absolute_error +
95 std::abs(d) * lhs.imag.absolute_error +
96 std::abs(b) * rhs.imag.absolute_error;
97 result.value().imag.absolute_error +=
98 std::abs(d) * lhs.real.absolute_error +
99 std::abs(a) * rhs.imag.absolute_error +
100 std::abs(c) * lhs.imag.absolute_error +
101 std::abs(b) * rhs.real.absolute_error;
102 return result;
103}

105Result<ApproxComplex> divide_complex(const ApproxComplex& lhs,
106 const ApproxComplex& rhs) {
107 const double a = lhs.real.value;
108 const double b = lhs.imag.value;
109 const double c = rhs.real.value;
110 const double d = rhs.imag.value;
111 if (c == 0.0 && d == 0.0) {
112 return complex_failure(CasErrc::DomainError,
113 "complex division by zero",
114 kEvalComplexOperation);
115 }
116 const lmmc_complex_t left{a, b};
117 const lmmc_complex_t right{c, d};
118 lmmc_complex_t quotient;
119 if (lmmc_complex_div(&left, &right, &quotient) != LMMC_STATUS_OK) {
120 return complex_failure(CasErrc::NumericFailure,
121 "complex division requires finite components and result",
122 kEvalComplexOperation);
123 }
124 return checked_complex(quotient.real, quotient.imag, kEvalComplexOperation);
125}

131Result<ApproxComplex> evaluate_complex_node(
132 const std::shared_ptr<const SymbolicNode>& node,
133 const NumericBindings& bindings,
134 ComputationContext& context) {
135 auto entered = context.enter_recursion(kEvalComplexOperation);
136 if (!entered) return Result<ApproxComplex>::failure(entered.error());
137 struct RecursionExit {
138 ComputationContext& context;
139 ~RecursionExit() { context.leave_recursion(); }
140 } recursion_exit{context};
141 if (!node) {
142 return complex_failure(CasErrc::InvalidArgument,
143 "expression contains a null node",
144 kEvalComplexOperation);
145 }
146
147 if (auto complex_node = std::dynamic_pointer_cast<const ComplexNode>(node)) {
148 auto real = evaluate_numeric(
149 *LMCAS::detail::make_expression_ptr(complex_node->real()),
150 bindings, context);
151 if (!real) return eval_complex_failure(real.error());
152 auto imag = evaluate_numeric(
153 *LMCAS::detail::make_expression_ptr(complex_node->imag()),
154 bindings, context);
155 if (!imag) return eval_complex_failure(imag.error());
156 if (!real.value().is_finite() || !imag.value().is_finite()) {
157 return complex_failure(CasErrc::NumericFailure,
158 "complex components must be finite",
159 kEvalComplexOperation);
160 }
161 return Result<ApproxComplex>::success(
162 ApproxComplex{real.value(), imag.value()});
163 }
164
165 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(node)) {
166 if (is_imaginary_unit_name(variable->name())) {
167 return Result<ApproxComplex>::success(approx_complex(0.0, 1.0));
168 }
169 }
170
171 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
172 auto sum = Result<ApproxComplex>::success(approx_complex(0.0, 0.0));
173 for (const auto& operand : add->operands()) {
174 auto term = evaluate_complex_node(operand, bindings, context);
175 if (!term) return term;
176 sum = add_complex(sum.value(), term.value());
177 if (!sum) return sum;
178 }
179 return sum;
180 }
181
182 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
183 auto product = Result<ApproxComplex>::success(approx_complex(1.0, 0.0));
184 for (const auto& operand : multiply->operands()) {
185 auto factor = evaluate_complex_node(operand, bindings, context);
186 if (!factor) return factor;
187 product = multiply_complex(product.value(), factor.value());
188 if (!product) return product;
189 }
190 return product;
191 }
192
193 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
194 auto base = evaluate_complex_node(power->base(), bindings, context);
195 if (!base) return base;
196 auto exponent_expression = simplify(
197 LMCAS::detail::make_expression_ptr(power->exponent()), context);
198 if (!exponent_expression) {
199 return eval_complex_failure(exponent_expression.error());
200 }
201 /**
202 * Exact rational exponents retain their integer-domain classification
203 * through normalization and conversion to the numeric power kernel.
204 * @see David Goldberg, "What Every Computer Scientist Should Know About
205 * Floating-Point Arithmetic" (1991), Floating-point Formats.
206 * https://docs.oracle.com/cd/E19957-01/806-3568/ncg_goldberg.html
207 */
208 if (auto number = std::dynamic_pointer_cast<const NumberNode>(
209 LMCAS::detail::node(exponent_expression.value()))) {
210 if (const auto* rational = std::get_if<Rational>(&number->value());
211 rational && rational->get_denominator() != BigInt(1)) {
212 return complex_failure(
213 CasErrc::UnsupportedExpression,
214 "complex evaluation requires an integer exponent",
215 kEvalComplexOperation);
216 }
217 }
218 auto exponent = evaluate_numeric(
219 *exponent_expression.value(), bindings, context);
220 if (!exponent) return eval_complex_failure(exponent.error());
221 if (!exponent.value().is_finite() ||
222 !std::isfinite(exponent.value().value)) {
223 return complex_failure(CasErrc::NumericFailure,
224 "complex power exponent must be finite",
225 kEvalComplexOperation);
226 }
227 const double exponent_value = exponent.value().value;
228 if (!is_integer_double(exponent_value) ||
229 std::abs(exponent_value) > 64.0) {
230 return complex_failure(CasErrc::UnsupportedExpression,
231 "complex evaluation only supports integer powers with |n| <= 64",
232 kEvalComplexOperation);
233 }
234 const int exponent_int = static_cast<int>(exponent_value);
235 unsigned exponent_magnitude =
236 static_cast<unsigned>(std::abs(exponent_int));
237 auto factor = base;
238 if (exponent_int < 0) {
239 factor = divide_complex(approx_complex(1.0, 0.0), base.value());
240 if (!factor) return factor;
241 }
242 auto result =
243 Result<ApproxComplex>::success(approx_complex(1.0, 0.0));
244 while (exponent_magnitude != 0U) {
245 if ((exponent_magnitude & 1U) != 0U) {
246 result = multiply_complex(result.value(), factor.value());
247 if (!result) return result;
248 }
249 exponent_magnitude >>= 1U;
250 if (exponent_magnitude != 0U) {
251 factor = multiply_complex(factor.value(), factor.value());
252 if (!factor) return factor;
253 }
254 }
255 return result;
256 }
257
258 return real_to_complex(
259 evaluate_numeric(*LMCAS::detail::make_expression_ptr(node),
260 bindings, context));
261}

286Result<ApproxComplex> eval_complex(const SymbolicExpr& expression,
287 const NumericBindings& bindings,
288 ComputationContext& context) {
289 try {
290 if (!LMCAS::detail::node(expression)) {
291 return complex_failure(CasErrc::InvalidArgument,
292 "cannot evaluate an empty expression as complex",
293 kEvalComplexOperation);
294 }
295 return evaluate_complex_node(LMCAS::detail::node(expression),
296 bindings, context);
297 } catch (const std::bad_alloc&) {
298 return complex_failure(CasErrc::ResourceLimit,
299 "complex evaluation allocation failed",
300 kEvalComplexOperation);
301 } catch (const std::exception& error) {
302 return complex_failure(CasErrc::InternalInvariant, error.what(),
303 kEvalComplexOperation);
304 }
305}
