29 void visit(const NumberNode& n) override {
30 const auto& value = n.value();
31 if (const auto* integer = std::get_if<BigInt>(&value)) {
32 put("integer");
33 put_integer(*integer);
34 } else if (const auto* fraction = std::get_if<Rational>(&value)) {
35 put("rational");
36 put_rational(*fraction);
37 } else {
38 put("binary64");
39 static_assert(sizeof(lmmc_real_t) == sizeof(std::uint64_t), "binary64 required");
40 std::uint64_t bits;
41 auto number = std::get<lmmc_real_t>(value);
42 std::memcpy(&bits, &number, sizeof(bits));
43 char hex[16];
44 constexpr char digits[] = "0123456789abcdef";
45 for (int i = 15; i >= 0; --i) {
46 hex[i] = digits[bits & 15];
47 bits >>= 4;
48 }
49 put(std::string_view(hex, sizeof(hex)));
50 }
51 }

145 bool needs_scale(const QuantityNode& n, const NodePtr& base) {
146 if (n.scale_to_base() != Rational(1)) return true;
147 if (dynamic_cast<const MultiplyNode*>(base.get())) return true;
148 const auto* number = dynamic_cast<const NumberNode*>(base.get());
149 return number && std::holds_alternative<Rational>(number->value());
150 }
151 bool collect_exact_factors(const std::vector<NodePtr>& factors,
152 Rational& coefficient, std::vector<NodePtr>& remaining) {
153 bool exact_factor = false;
154 for (const auto& factor : factors) {
155 const auto* number = dynamic_cast<const NumberNode*>(factor.get());
156 if (!number || std::holds_alternative<lmmc_real_t>(number->value())) {
157 remaining.push_back(factor);
158 continue;
159 }
160 const Rational value = std::holds_alternative<BigInt>(number->value())
161 ? Rational(std::get<BigInt>(number->value()))
162 : std::get<Rational>(number->value());
163 check(context_.require_integer_bits(value.get_numerator().bit_length(), kEncode));
164 check(context_.require_integer_bits(value.get_denominator().bit_length(), kEncode));
165 coefficient = coefficient * value;
166 check(context_.require_integer_bits(coefficient.get_numerator().bit_length(), kEncode));
167 check(context_.require_integer_bits(coefficient.get_denominator().bit_length(), kEncode));
168 exact_factor = true;
169 }
170 return exact_factor;
171 }
172 NodePtr base_value(const QuantityNode& n) {
173 NodePtr base = n.value();
174 if (!needs_scale(n, base)) return base;
175 Rational coefficient = n.scale_to_base();
176 check(context_.require_integer_bits(coefficient.get_numerator().bit_length(), kEncode));
177 check(context_.require_integer_bits(coefficient.get_denominator().bit_length(), kEncode));
178 std::vector<NodePtr> factors;
179 if (const auto* product = dynamic_cast<const MultiplyNode*>(base.get()))
180 factors.assign(product->operands().begin(), product->operands().end());
181 else
182 factors.push_back(base);
183 std::vector<NodePtr> remaining;
184 remaining.reserve(factors.size() + 1);
185 const bool exact_factor = collect_exact_factors(factors, coefficient, remaining);
186 return scaled_value(base, coefficient, remaining, exact_factor);
187 }
188 NodePtr scaled_value(const NodePtr& base, const Rational& coefficient,
189 std::vector<NodePtr>& remaining, bool exact_factor) {
190 if (exact_factor || coefficient != Rational(1)) {
191 if (coefficient != Rational(1) || remaining.empty()) {
192 remaining.push_back(coefficient.is_integer()
193 ? SymbolicFactory::create_number(coefficient.to_bigint())
194 : SymbolicFactory::create_number(coefficient));
195 }
196 return remaining.size() == 1 ? remaining.front()
197 : SymbolicFactory::create_multiply(std::move(remaining));
198 }
199 return base;
200 }
201 void put(std::string_view field) {
202 check(context_.consume_steps(1, kEncode));
203 char digits[std::numeric_limits<std::size_t>::digits10 + 2];
204 auto end = std::to_chars(digits, digits + sizeof(digits), field.size()).ptr;
205 const auto width = static_cast<std::size_t>(end - digits);
206 const auto extra = width + 2;
207 if (field.size() > context_.limits().max_input_bytes ||
208 output_.size() > context_.limits().max_input_bytes - field.size() ||
209 extra > context_.limits().max_input_bytes - field.size() - output_.size()) {
210 throw CasError{CasErrc::ResourceLimit, "serialized expression byte limit exceeded", kEncode};
211 }
212 output_.append(digits, width);
213 output_ += ':';
214 output_.append(field.data(), field.size());
215 output_ += ',';
216 }
217 void put_size(std::size_t value) {
218 char digits[std::numeric_limits<std::size_t>::digits10 + 2];
219 auto end = std::to_chars(digits, digits + sizeof(digits), value).ptr;
220 put(std::string_view(digits, static_cast<std::size_t>(end - digits)));
221 }
222 void put_integer(const BigInt& value) {
223 check(context_.require_integer_bits(value.bit_length(), kEncode));
224 put(value.to_string());
225 }
226 void put_rational(const Rational& value) {
227 put_integer(value.get_numerator());
228 put_integer(value.get_denominator());
229 }
230 template <typename Children>
231 void emit_list(const Children& children) {
232 put_size(children.size());
233 for (const auto& child : children) emit(child);
234 }
235 void emit(const NodePtr& node) {
236 Recursion depth(context_, kEncode);
237 if (!node) throw CasError{CasErrc::InvalidArgument, "null expression node", kEncode};
238 try {
239 node->accept(*this);
240 } catch (const std::runtime_error& error) {
241 if (std::string_view(error.what()) == "AST traversal depth limit exceeded")
242 throw CasError{CasErrc::ResourceLimit, error.what(), kEncode};
243 throw;
244 }
245 }

248 std::string output_;

254LMCAS_API Result<std::string> serialize_expr(const ExprPtr& expression,
255 ComputationContext& context) {
256 auto step = context.consume_steps(1, kEncode);
257 if (!step) return Result<std::string>::failure(step.error());
258 if (!expression || !detail::node(expression))
259 return Result<std::string>::failure(CasErrc::InvalidArgument,
260 "expression cannot be null", kEncode);
261 try {
262 return Writer(context).finish(detail::node(expression));
263 } catch (const CasError& error) {
264 return Result<std::string>::failure(error);
265 } catch (const std::bad_alloc&) {
266 return Result<std::string>::failure(CasErrc::ResourceLimit,
267 "expression serialization allocation failed", kEncode);
268 } catch (const std::exception& error) {
269 return Result<std::string>::failure(CasErrc::UnsupportedExpression,
270 error.what(), kEncode);
271 }
272}
