14static std::vector<std::shared_ptr<SymbolicExpr>> tf_restore_factors(
15 const std::vector<Polynomial<Rational>>& true_factors,
16 const TransSubstitutionResult& sub_result,
17 const std::shared_ptr<SymbolicExpr>& pyth_simplified) {
18 /// --- Phase 6: 逆换元 ---
19 std::vector<std::shared_ptr<SymbolicExpr>> symbolic_factors;
20 symbolic_factors.reserve(true_factors.size());
21
22 for (const auto& poly_factor : true_factors) {
23 /// 将多项式因子转回符号表达式
24 auto sym_factor = poly_to_symbolic(poly_factor);
25 if (!sym_factor) continue;
26
27 /// 逆换元:将不定元替换回原始超越表达式
28 auto back_sub = tf_back_substitute(sym_factor, sub_result.mappings);
29 if (back_sub) {
30 symbolic_factors.push_back(back_sub);
31 }
32 }
33
34 /// 若逆换元后因子数 <= 1,分解无效
35 if (symbolic_factors.size() <= 1) {
36 return {pyth_simplified};
37 }
38
39 /// --- Phase 6.2: 化简与常数提取 ---
40 auto final_factors = tf_simplify_factors(symbolic_factors);
41
42 /// 若化简后仅剩一个非常数因子,分解无效
43 size_t non_const_count = 0;
44 for (const auto& f : final_factors) {
45 if (f && !f->is_number()) non_const_count++;
46 }
47 if (non_const_count <= 1 && final_factors.size() <= 1) {
48 return {pyth_simplified};
49 }
50
51 return final_factors;
52}

54static Result<std::vector<std::shared_ptr<SymbolicExpr>>> tf_factor_substitution(
55 const TransSubstitutionResult& sub_result, const std::string& var,
56 const std::shared_ptr<SymbolicExpr>& pyth_simplified,
57 ComputationContext& context) {
58 using FactorsResult = Result<std::vector<std::shared_ptr<SymbolicExpr>>>;
59 std::vector<std::string> indeterminates;
60 indeterminates.reserve(sub_result.mappings.size());
61 for (const auto& m : sub_result.mappings) {
62 indeterminates.push_back(m.indeterminate);
63 }
64 auto built = tf_build_polynomial(
65 sub_result.poly_expr, indeterminates, var);
66 if (!built) { return FactorsResult::failure(built.error()); }
67 const auto& poly_result = built.value();
68 if (!poly_result.success) {
69 return FactorsResult::success({pyth_simplified});
70 }
71 if (poly_result.poly.degree() <= 1) {
72 return FactorsResult::success({pyth_simplified});
73 }
74 auto factored =
75 factor_univariate_bridge_checked(poly_result.poly, context);
76 if (!factored) {
77 return FactorsResult::failure(factored.error());
78 }
79 const auto& factorization = factored.value();
80 if (factorization.completeness != Completeness::Complete ||
81 factorization.value.size() <= 1) {
82 return FactorsResult::success({pyth_simplified});
83 }
84 return tf_restore_factors(
85 factorization.value, sub_result, pyth_simplified);
86}

108 if (!exp_factors.empty()) {

111

115

117 if (sub_result.mappings.empty()) {

119 }
120 if (tf_is_linear_irreducible(sub_result, var)) {
121 return FactorsResult::success({pyth_simplified});
122 }

126}
