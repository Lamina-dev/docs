24bool square_root_is_imaginary(const FunctionNode& function) {
25 if (function.type() != FunctionNode::FuncType::Sqrt || function.arguments().size() != 1) {
26 return false;
27 }
28 auto number = std::dynamic_pointer_cast<const NumberNode>(function.arguments()[0]);
29 if (!number) {
30 return false;
31 }
32 const auto& value = number->value();
33 if (const auto* integer = std::get_if<BigInt>(&value)) {
34 return *integer == BigInt(-1) || integer->is_negative();
35 }
36 if (const auto* rational = std::get_if<Rational>(&value)) {
37 return *rational == Rational(-1) || *rational < Rational(0);
38 }
39 if (const auto* real = std::get_if<lmmc_real_t>(&value)) {
40 int equal;
41 lmmc_double_nearly_equal(*real, -1.0, &equal);
42 return equal || (std::isfinite(*real) && *real < 0.0);
43 }
44 return false;
45}

47bool power_is_imaginary(const PowerNode& power) {
48 auto base = std::dynamic_pointer_cast<const NumberNode>(power.base());
49 auto exponent = std::dynamic_pointer_cast<const NumberNode>(power.exponent());
50 if (!base || !exponent) {
51 return false;
52 }
53 double base_value = 0.0;
54 bool base_numeric = solver_number_value(*base, base_value);
55 double exponent_value = 0.0;
56 bool exponent_numeric = solver_number_value(*exponent, exponent_value);
57 if (!base_numeric || !exponent_numeric || base_value >= 0.0) {
58 return false;
59 }
60 return base_value < 0.0 && std::abs(exponent_value - std::round(exponent_value)) > 1e-12;
61}
