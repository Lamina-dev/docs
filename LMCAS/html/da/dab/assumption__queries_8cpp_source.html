104AssumptionTriboolResult AssumptionContext::is_continuous_checked(
105 const std::string& symbol,
106 const Interval& interval) const {
107 constexpr const char* operation = "is_continuous";
108 if (symbol.empty()) {
109 return AssumptionTriboolResult::failure(
110 CasErrc::InvalidArgument, "symbol name must not be empty", operation);
111 }
112 for (auto it = scope_stack_.rbegin(); it != scope_stack_.rend(); ++it) {
113 auto result = it->properties.is_continuous_checked(symbol, interval);
114 if (!result) {
115 return AssumptionTriboolResult::failure(result.error());
116 }
117 if (result.value()) {
118 return AssumptionTriboolResult::success(Tribool::True);
119 }
120 }
121 return AssumptionTriboolResult::success(Tribool::Unknown);
122}
