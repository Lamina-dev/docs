136InferenceTriboolResult InferenceEngine::infer_power_property_checked(
137 const PowerNode& node, Sign target, ComputationContext& context) const {
138 try {
139 auto defined = Impl::real_definedness(*this, node, context);
140 if (!defined || defined.value() != Tribool::True) return defined;
141 auto base_expr = LMCAS::detail::expression_from_node(node.base());
142 auto exp_expr = LMCAS::detail::expression_from_node(node.exponent());
143 auto exp_num = std::dynamic_pointer_cast<const NumberNode>(node.exponent());
144
145 auto zero = impl_->zero_power_sign(*this, base_expr, exp_num.get(), target, context);
146 if (!zero) {
147 return zero;
148 }
149 if (zero.value() != Tribool::Unknown) {
150 return zero;
151 }
152
153 auto positive = impl_->positive_power_sign(*this, base_expr, exp_expr, target, context);
154 if (!positive) {
155 return positive;
156 }
157 if (positive.value() != Tribool::Unknown) {
158 return positive;
159 }
160
161 if (exp_num) {
162 return impl_->integer_power_sign(*this, base_expr, *exp_num, target, context);
163 }
164 return InferenceTriboolResult::success(Tribool::Unknown);
165 } catch (const CasError& error) {
166 return InferenceTriboolResult::failure(error);
167 } catch (const std::bad_alloc&) {
168 return InferenceTriboolResult::failure(
169 CasErrc::ResourceLimit,
170 "power sign inference allocation failed",
171 "infer_power_property_checked");
172 } catch (const std::exception& ex) {
173 return InferenceTriboolResult::failure(
174 CasErrc::InternalInvariant,
175 ex.what(),
176 "infer_power_property_checked");
177 }
