14void BigInt::realloc_to(mp_size_t new_alloc) {
15 if (new_alloc <= capacity_) return;
16 const std::size_t requested = static_cast<std::size_t>(new_alloc);
17 if (requested > std::numeric_limits<std::size_t>::max() - 3) {
18 throw std::length_error("BigInt allocation size overflow");
19 }
20 const std::size_t rounded = (requested + 3) & ~std::size_t(3);
21 if (rounded > std::numeric_limits<std::size_t>::max() / sizeof(mp_limb_t) ||
22 rounded > static_cast<std::size_t>(std::numeric_limits<mp_size_t>::max())) {
23 throw std::length_error("BigInt allocation byte size overflow");
24 }

31 }
32 if (data_) ::operator delete[](data_);

34 capacity_ = static_cast<mp_size_t>(rounded);

109void BigInt::import_unsigned(std::uint64_t magnitude) {
110 zero();
111 if (magnitude == 0) {
112 return;
113 }
114 constexpr auto limb_bits = std::numeric_limits<mp_limb_t>::digits;
115 constexpr auto value_bits = std::numeric_limits<std::uint64_t>::digits;
116 realloc_to((value_bits + limb_bits - 1) / limb_bits);
117 do {
118 data_[size_++] = static_cast<mp_limb_t>(magnitude);
119 if constexpr (limb_bits < value_bits) {
120 magnitude >>= limb_bits % value_bits;
121 } else {
122 magnitude = 0;
123 }
124 } while (magnitude != 0);
125 sign_ = POSITIVE;
126}

144BigInt::BigInt(const std::string& str) {
145 if (str.empty()) {
146 throw std::invalid_argument("BigInt: decimal string must contain digits");
147 }
148 size_t start = 0;
149 int sign = POSITIVE;
150 if (str[0] == '-') {
151 sign = NEGATIVE;
152 start = 1;
153 } else if (str[0] == '+') {
154 start = 1;
155 }
156
157 if (start == str.length()) {
158 throw std::invalid_argument("BigInt: sign must be followed by digits");
159 }
160
161 size_t len = str.length() - start;
162
163 std::vector<mp_byte_t> digit_buf(len);
164 for (size_t i = 0; i < len; ++i) {
165 char c = str[start + i];
166 if (c < '0' || c > '9') {
167 throw std::invalid_argument(
168 "BigInt: invalid character '" + std::string(1, c) +
169 "' at position " + std::to_string(start + i) +
170 " in input \"" + str + "\"");
171 }
172 uint8_t d = static_cast<uint8_t>(c - '0');
173
174 digit_buf[len - 1 - i] = d;
175 }
176
177 mp_size_t needed = len / 19 + 2;
178 realloc_to(needed);
179 LMCAS::detail::ensure_lmmc_lifecycle();
180
181 size_ = lmmp_from_str_(data_, digit_buf.data(), len, 10);
182
183 if (size_ == 0) {
184 zero();
185 } else {
186 sign_ = sign;
187 }
188 normalize();
189 }
