99void serialize_symbol_properties(std::ostream& out, const PropertyStore& props,
100 const std::string& sym) {
101 Domain dom = props.get_domain(sym);
102 if (dom != Domain::Complex) {
103 out << "DOMAIN " << sym << " " << property_detail::domain_str(dom) << "\n";
104 }
105 auto signs = props.get_signs(sym);
106 for (Sign s : signs) {
107 out << "SIGN " << sym << " " << sign_to_string(s) << "\n";
108 }
109 Parity par = props.get_parity(sym);
110 if (par != Parity::Unknown) {
111 out << "PARITY " << sym << " " << parity_to_string(par) << "\n";
112 }
113 Boundedness bnd = props.get_boundedness(sym);
114 if (bnd != Boundedness::Unknown) {
115 out << "BOUNDED " << sym << " " << boundedness_to_string(bnd);
116 if (bnd == Boundedness::Bounded) {
117 const auto bounds = props.get_bounds(sym);
118 if (bounds) {
119 out << " " << interval_to_string(*bounds);
120 }
121 }
122 out << "\n";
123 }
124 if (props.is_transcendental(sym)) {
125 out << "TRANSCENDENTAL " << sym << "\n";
126 }
127 Finiteness fin = props.get_finiteness(sym);
128 if (fin != Finiteness::Unknown) {
129 out << "FINITENESS " << sym << " " << finiteness_to_string(fin) << "\n";
130 }
131 Definiteness def = props.get_definiteness(sym);
132 if (def != Definiteness::Unknown) {
133 out << "DEFINITENESS " << sym << " " << definiteness_to_string(def) << "\n";
134 }
135 for (const auto& [variable, period] : props.get_period_decls(sym)) {
136 out << "PERIODIC " << sym << " " << variable << " "
137 << period.to_string() << "\n";
138 }
139}

141void serialize_symbol_intervals(std::ostream& out, const PropertyStore& props,
142 const std::string& sym) {

153

155 for (const auto& decl : mono_decls) {
156 out << "MONOTONICITY " << sym << " " << decl.variable << " "
157 << interval_to_string(decl.interval) << " "
158 << monotonicity_to_string(decl.type) << "\n";
159 }

164std::string AssumptionContext::serialize() const {
165 std::ostringstream out;
166
167 for (int scope_idx = 0; scope_idx < static_cast<int>(scope_stack_.size()); ++scope_idx) {
168 out << "SCOPE " << scope_idx << "\n";
169 const auto& scope = scope_stack_[scope_idx];
170 const auto& props = scope.properties;
171 auto symbols = props.get_all_symbols();
172 for (const auto& sym : symbols) {
173 serialize_symbol_properties(out, props, sym);
174 }
175 for (const auto& sym : symbols) {
176 serialize_symbol_intervals(out, props, sym);
177 }
178 const auto& relations = scope.relations.get_relations();
179 for (const auto i : scope.relations.declaration_order_) {
180 const auto& rel = relations[i];
181 std::string lhs_str = rel.lhs.to_string();
182 std::string rhs_str = rel.rhs.to_string();
183 out << "RELATION " << lhs_str << " " << relop_to_string(rel.op)
184 << " " << rhs_str << "\n";
185 }
186 for (const auto& cond : scope.conditionals) {
187 std::string cond_str = cond.condition.to_string();
188 std::string concl_str = cond.conclusion.to_string();
189 out << "CONDITIONAL (" << cond_str << ") => (" << concl_str << ")\n";
190 }
191 }
192
193 out << "END\n";
194 return out.str();
195}
