57/** @brief 符号多项式算法使用的表达式系数。 */
58struct SymbolicPolyCoeff {
59 std::shared_ptr<SymbolicExpr> val;
60
61 SymbolicPolyCoeff() : val(SymbolicExpr::number(0)) {}
62 explicit SymbolicPolyCoeff(int value) : val(SymbolicExpr::number(value)) {}
63 SymbolicPolyCoeff(std::shared_ptr<SymbolicExpr> value) : val(std::move(value)) {}
64
65 bool operator==(const SymbolicPolyCoeff& other) const {
66 if (!val || !other.val) return false;
67 if (val == other.val) return true;
68 auto difference = SymbolicExpr::add(
69 val,
70 SymbolicExpr::multiply(other.val, SymbolicExpr::number(-1)));
71 return difference->simplify()->is_zero();
72 }
73
74 bool operator!=(const SymbolicPolyCoeff& other) const {
75 return !(*this == other);
76 }
77
78 SymbolicPolyCoeff operator+(const SymbolicPolyCoeff& other) const {
79 return SymbolicPolyCoeff(SymbolicExpr::add(val, other.val));
80 }
81
82 SymbolicPolyCoeff operator-(const SymbolicPolyCoeff& other) const {
83 return SymbolicPolyCoeff(SymbolicExpr::add(
84 val,
85 SymbolicExpr::multiply(other.val, SymbolicExpr::number(-1))));
86 }
87
88 SymbolicPolyCoeff operator*(const SymbolicPolyCoeff& other) const {
89 return SymbolicPolyCoeff(SymbolicExpr::multiply(val, other.val));
90 }
91
92 SymbolicPolyCoeff operator/(const SymbolicPolyCoeff& other) const {
93 return SymbolicPolyCoeff(SymbolicExpr::divide(val, other.val));
94 }
95
96 SymbolicPolyCoeff operator-() const {
97 return SymbolicPolyCoeff(
98 SymbolicExpr::multiply(val, SymbolicExpr::number(-1)));
99 }
100
101 std::string to_string() const {
102 return val ? val->to_string() : "0";
103 }
104
105 friend SymbolicPolyCoeff abs(const SymbolicPolyCoeff& value) {
106 return value;
107 }
108
109 bool operator<(const SymbolicPolyCoeff&) const {
110 return false;
111 }
112};
