43Result<void> verify_quadratic_boundary(
44 const Polynomial<Rational>& polynomial,
45 const std::shared_ptr<SymbolicExpr>& boundary,
46 const Rational& rational_part,
47 const Rational& radical_coefficient,
48 const Rational& radicand,
49 ComputationContext& context) {
50 auto steps = context.consume_steps(2, kCheckedInequalityOperation);
51 if (!steps) return steps;
52
53 if (!boundary || !lamina::detail::node(boundary) || polynomial.coeffs.size() < 3) {
54 return Result<void>::failure(
55 CasErrc::InternalInvariant,
56 "quadratic boundary verification could not construct an expression",
57 kCheckedInequalityOperation);
58 }
59
60 const Rational& constant = polynomial.coeffs[0];
61 const Rational& linear = polynomial.coeffs[1];
62 const Rational& quadratic = polynomial.coeffs[2];
63 const Rational residual_rational =
64 quadratic * (rational_part * rational_part +
65 radical_coefficient * radical_coefficient * radicand) +
66 linear * rational_part + constant;
67 const Rational residual_radical =
68 Rational(2) * quadratic * rational_part * radical_coefficient +
69 linear * radical_coefficient;
70 if (residual_rational != Rational(0) ||
71 residual_radical != Rational(0)) {
72 return Result<void>::failure(
73 CasErrc::Inconclusive,
74 "quadratic boundary could not be verified by exact substitution",
75 kCheckedInequalityOperation);
76 }
77 return Result<void>::success();
78}

82Result<IntervalUnion> InequalitySolver::solve_exact_quadratic_inequality(
83 const Polynomial<Rational>& polynomial,
84 InequalityType type,
85 ComputationContext& context) {
86 const Rational& constant = polynomial.coeffs[0];
87 const Rational& linear = polynomial.coeffs[1];
88 const Rational& quadratic = polynomial.coeffs[2];
89 if (quadratic == Rational(0)) {
90 return Result<IntervalUnion>::failure(
91 CasErrc::InternalInvariant,
92 "quadratic polynomial has a zero leading coefficient",
93 kCheckedInequalityOperation);
94 }
95
96 auto steps = context.consume_steps(3, kCheckedInequalityOperation);
97 if (!steps) return Result<IntervalUnion>::failure(steps.error());
98 const Rational discriminant = linear * linear - Rational(4) * quadratic * constant;
99 const bool leading_positive = !rational_is_negative(quadratic);
100 const bool wants_positive = type == InequalityType::GreaterThan ||
101 type == InequalityType::GreaterEqual;
102 const bool strict = type == InequalityType::GreaterThan ||
103 type == InequalityType::LessThan;
104
105 if (rational_is_negative(discriminant)) {
106 const bool satisfied = wants_positive == leading_positive;
107 return Result<IntervalUnion>::success(
108 satisfied ? IntervalUnion::entire_line() : IntervalUnion::empty());
109 }
110
111 const Rational denominator = Rational(2) * quadratic;
112 const Rational neg_linear = Rational(0) - linear;
113 const Rational rational_part = neg_linear / denominator;
114 if (discriminant == Rational(0)) {
115 auto root = SymbolicExpr::number(rational_part);
116 auto verified = verify_quadratic_boundary(
117 polynomial, root, rational_part, Rational(0), Rational(0), context);
118 if (!verified) return Result<IntervalUnion>::failure(verified.error());
119
120 if (wants_positive == leading_positive) {
121 if (!strict) return Result<IntervalUnion>::success(IntervalUnion::entire_line());
122 std::vector<Interval> rays{
123 Interval{Endpoint::neg_inf(), Endpoint::open(root)},
124 Interval{Endpoint::open(root), Endpoint::pos_inf()}
125 };
126 return Result<IntervalUnion>::success(
127 IntervalUnion::from_checked_normalized(std::move(rays)));
128 }
129 if (strict) return Result<IntervalUnion>::success(IntervalUnion::empty());
130 return Result<IntervalUnion>::success(
131 IntervalUnion::from_checked_normalized({Interval::point(root)}));
132 }
133
134 auto nodes = context.reserve_nodes(20, kCheckedInequalityOperation);
135 if (!nodes) return Result<IntervalUnion>::failure(nodes.error());
136 auto sqrt_discriminant = SymbolicExpr::sqrt(
137 SymbolicExpr::number(discriminant))->simplify();
138 const Rational first_radical_coefficient = Rational(-1) / denominator;
139 const Rational second_radical_coefficient = Rational(1) / denominator;
140 auto first_formula = SymbolicExpr::add(
141 SymbolicExpr::number(rational_part),
142 SymbolicExpr::multiply(SymbolicExpr::number(first_radical_coefficient),
143 sqrt_discriminant))->simplify();
144 auto second_formula = SymbolicExpr::add(
145 SymbolicExpr::number(rational_part),
146 SymbolicExpr::multiply(SymbolicExpr::number(second_radical_coefficient),
147 sqrt_discriminant))->simplify();
148 auto lower = leading_positive ? first_formula : second_formula;
149 auto upper = leading_positive ? second_formula : first_formula;
150
151 auto first_verified = verify_quadratic_boundary(
152 polynomial, first_formula, rational_part,
153 first_radical_coefficient, discriminant, context);
154 if (!first_verified) return Result<IntervalUnion>::failure(first_verified.error());
155 auto second_verified = verify_quadratic_boundary(
156 polynomial, second_formula, rational_part,
157 second_radical_coefficient, discriminant, context);
158 if (!second_verified) return Result<IntervalUnion>::failure(second_verified.error());
159
160 const bool outside = wants_positive == leading_positive;
161 if (outside) {
162 std::vector<Interval> rays{
163 Interval{Endpoint::neg_inf(), finite_endpoint(lower, strict)},
164 Interval{finite_endpoint(upper, strict), Endpoint::pos_inf()}
165 };
166 return Result<IntervalUnion>::success(
167 IntervalUnion::from_checked_normalized(std::move(rays)));
168 }
169 return Result<IntervalUnion>::success(IntervalUnion::from_checked_normalized({
170 Interval{finite_endpoint(lower, strict), finite_endpoint(upper, strict)}
171 }));

175Result<IntervalUnion> solve_exact_affine_inequality_impl(
176 const Polynomial<Rational>& polynomial,
177 InequalityType type,
178 ComputationContext& context) {
179 auto step = context.consume_steps(1, kCheckedInequalityOperation);
180 if (!step) return Result<IntervalUnion>::failure(step.error());
181
182 if (polynomial.degree() <= 0) {
183 const Rational value = polynomial.is_zero()
184 ? Rational(0)
185 : polynomial.coeffs[0];
186 return Result<IntervalUnion>::success(
187 constant_satisfies(value, type)
188 ? IntervalUnion::entire_line()
189 : IntervalUnion::empty());
190 }
191
192 if (polynomial.degree() != 1 || polynomial.coeffs.size() < 2) {
193 return Result<IntervalUnion>::failure(
194 CasErrc::Inconclusive,
195 "checked inequality solving currently supports exact polynomials through degree two",
196 kCheckedInequalityOperation);
197 }
198
199 const Rational& intercept = polynomial.coeffs[0];
200 const Rational& slope = polynomial.coeffs[1];
201 if (slope == Rational(0)) {
202 return Result<IntervalUnion>::failure(
203 CasErrc::InternalInvariant,
204 "affine polynomial has a zero leading coefficient",
205 kCheckedInequalityOperation);
206 }
207
208 const Rational boundary = (Rational(0) - intercept) / slope;
209 auto boundary_expr = SymbolicExpr::number(boundary);
210 auto nodes = context.reserve_nodes(1, kCheckedInequalityOperation);
211 if (!nodes) return Result<IntervalUnion>::failure(nodes.error());
212
213 const bool wants_greater = type == InequalityType::GreaterThan ||
214 type == InequalityType::GreaterEqual;
215 const bool strict = type == InequalityType::GreaterThan ||
216 type == InequalityType::LessThan;
217 const bool solution_above = wants_greater != rational_is_negative(slope);
218 const Endpoint finite = strict
219 ? Endpoint::open(boundary_expr)
220 : Endpoint::closed(boundary_expr);
221 const Interval interval = solution_above
222 ? Interval{finite, Endpoint::pos_inf()}
223 : Interval{Endpoint::neg_inf(), finite};
224 return Result<IntervalUnion>::success(IntervalUnion::from_single(interval));
225}

234std::shared_ptr<SymbolicExpr> rational_polynomial_expression(
235 const Polynomial<Rational>& polynomial) {
236 auto variable = SymbolicExpr::variable(polynomial.variable_name);
237 std::shared_ptr<SymbolicExpr> sum = SymbolicExpr::number(0);
238 for (int degree = 0; degree <= polynomial.degree(); ++degree) {
239 if (polynomial.coeffs[degree] == Rational(0)) continue;
240 auto term = SymbolicExpr::number(polynomial.coeffs[degree]);
241 if (degree > 0) {
242 auto power = degree == 1
243 ? variable
244 : SymbolicExpr::power(variable, SymbolicExpr::number(degree));
245 term = SymbolicExpr::multiply(term, power);
246 }
247 sum = SymbolicExpr::add(sum, term);
248 }
249 return sum->simplify();
250}

252Result<IntervalUnion> solve_exact_polynomial_inequality_impl(
253 const Polynomial<Rational>& polynomial,
254 InequalityType type,
255 ComputationContext& context) {
256 auto factors = square_free_factorization(polynomial);
257 std::vector<ExactInequalityRoot> roots;
258 try {
259 for (const auto& [factor, multiplicity] : factors) {
260 if (factor.degree() < 1) continue;
261 auto isolated = isolate_real_roots_checked(factor, context);
262 if (!isolated) {
263 return Result<IntervalUnion>::failure(isolated.error());
264 }
265 const auto& intervals = isolated.value();
266 auto factor_expression = rational_polynomial_expression(factor);
267 for (std::size_t index = 0; index < intervals.size(); ++index) {
268 auto algebraic = lamina::detail::make_exact_real_algebraic(
269 factor, index, static_cast<std::size_t>(multiplicity), context);
270 if (!algebraic) {
271 return Result<IntervalUnion>::failure(algebraic.error());
272 }
273 roots.push_back(ExactInequalityRoot{
274 SymbolicExpr::root_of(
275 factor_expression, polynomial.variable_name,
276 static_cast<int>(index)),
277 std::move(algebraic.value()),
278 multiplicity});
279 }
280 }
281 std::sort(roots.begin(), roots.end(),
282 [](const ExactInequalityRoot& lhs, const ExactInequalityRoot& rhs) {
283 return lhs.algebraic.lower < rhs.algebraic.lower;
284 });
285 } catch (const std::bad_alloc&) {
286 return Result<IntervalUnion>::failure(
287 CasErrc::ResourceLimit,
288 "polynomial inequality root allocation failed",
289 kCheckedInequalityOperation);
290 } catch (const std::exception& error) {
291 return Result<IntervalUnion>::failure(
292 CasErrc::InternalInvariant, error.what(),
293 kCheckedInequalityOperation);
294 }
295
296 const bool wants_positive =
297 type == InequalityType::GreaterThan ||
298 type == InequalityType::GreaterEqual;
299 const bool strict =
300 type == InequalityType::GreaterThan ||
301 type == InequalityType::LessThan;
302 auto sign_satisfies = [&](int sign_value) {
303 return wants_positive ? sign_value > 0 : sign_value < 0;
304 };
305
306 int sign_value = rational_is_negative(polynomial.lead_coeff()) ? -1 : 1;
307 if ((polynomial.degree() & 1) != 0) sign_value = -sign_value;
308
309 std::vector<Interval> intervals;
310 Endpoint lower = Endpoint::neg_inf();
311 for (const auto& root : roots) {
312 if (sign_satisfies(sign_value)) {
313 intervals.push_back(Interval{
314 lower, Endpoint::open(root.expression)});
315 }
316 if (!strict) {
317 intervals.push_back(Interval::point(root.expression));
318 }
319 lower = Endpoint::open(root.expression);
320 if ((root.multiplicity & 1) != 0) sign_value = -sign_value;
321 }
322 if (sign_satisfies(sign_value)) {
323 intervals.push_back(Interval{lower, Endpoint::pos_inf()});
324 }
325 return IntervalUnion::from_intervals_checked(
326 std::move(intervals), context);
327}

401 iv.upper = Endpoint::pos_inf();

410 InequalityType type,

412 const std::vector<int>&) {

415

417 bool is_strict = (type == InequalityType::GreaterThan || type == InequalityType::LessThan);
418 int target_sign = want_positive ? 1 : -1;
419
420 for (const auto& entry : chart) {
421 if (entry.sign == target_sign) {
422 result_intervals.push_back(entry.interval);
423 }
424 }
425
426 if (!is_strict) {
427
428 for (size_t i = 0; i < roots.size(); ++i) {
429

431 for (auto& iv : result_intervals) {
432

436 merged = true;

441 if (roots_equal(iv.lower.value, roots[i])) {
442 iv.lower.is_open = false;
443 merged = true;
444 }
445 }
446 }
447
448 if (!merged) {
449 result_intervals.push_back(Interval::point(roots[i]));
450 }
451 }
452 }
