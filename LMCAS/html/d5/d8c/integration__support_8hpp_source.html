74inline std::optional<int> integration_bounded_exponent(
75 const NumberNode& number, int minimum_exponent, int maximum_exponent) {
76 const auto narrow = [minimum_exponent, maximum_exponent](const BigInt& exponent)
77 -> std::optional<int> {
78 if (exponent < BigInt(minimum_exponent) || exponent > BigInt(maximum_exponent)) {
79 return std::nullopt;
80 }
81 const auto bounded_exponent = exponent.try_to_int64();
82 if (!bounded_exponent) { return std::nullopt; }
83 return static_cast<int>(*bounded_exponent);
84 };
85 const auto& value = number.value();
86 if (const auto* exponent = std::get_if<BigInt>(&value)) { return narrow(*exponent); }
87 if (const auto* exponent = std::get_if<Rational>(&value)) {
88 if (!exponent->is_integer()) { return std::nullopt; }
89 return narrow(exponent->to_bigint());
90 }
91 return std::nullopt;
92}

115template <typename Transform>
116std::shared_ptr<SymbolicExpr> lookup_integral(
117 const std::vector<detail::IntegrationEntry>& table, const SymbolicExpr& matched,
118 const std::string& matched_variable, detail::RewriteBudget& budget, Transform&& transform) {
119 for (const auto& rule : table) {
120 const auto* entry = &rule;
121 budget.consume();
122 MatchMap bindings;
123 if (!Matcher::match(entry->pattern, matched, entry->wildcards, bindings)) { continue; }
124 if (bindings.find("__Add_REST__") != bindings.end() ||
125 bindings.find("__Mul_REST__") != bindings.end()) {
126 continue;
127 }
128 const bool complete = std::all_of(entry->wildcards.begin(), entry->wildcards.end(),
129 [&](const std::string& name) { return bindings.find(name) != bindings.end(); });
130 if (!complete || (entry->condition && !entry->condition(bindings, matched_variable, budget.context()))) {
131 continue;
132 }
133 auto instantiated = detail::make_expression_ptr(
134 Matcher::replace(entry->result, bindings, false));
135 budget.measure(detail::node(instantiated));
136 if (contains_unevaluated_integral(detail::node(instantiated))) {
137 continue;
138 }
139 auto candidate = transform(std::move(instantiated));
140 if (!candidate) { continue; }
141 return detail::make_expression_ptr(detail::normalize_node(detail::node(candidate), budget));
142 }
143 return nullptr;
144}
