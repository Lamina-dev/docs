63inline bool contains_unevaluated_integral(
64 const std::shared_ptr<const SymbolicNode>& node) {
65 if (!node) {
66 return false;
67 }
68 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(node)) {
69 if (function->type() == FunctionNode::FuncType::Calculus_Integral) {
70 return true;
71 }
72 for (const auto& argument : function->arguments()) {
73 if (contains_unevaluated_integral(argument)) {
74 return true;
75 }
76 }
77 } else if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
78 for (const auto& operand : add->operands()) {
79 if (contains_unevaluated_integral(operand)) {
80 return true;
81 }
82 }
83 } else if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
84 for (const auto& operand : multiply->operands()) {
85 if (contains_unevaluated_integral(operand)) {
86 return true;
87 }
88 }
89 } else if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
90 return contains_unevaluated_integral(power->base()) ||
91 contains_unevaluated_integral(power->exponent());
92 }
93 return false;
94}
