46inline std::optional<int> integration_bounded_exponent(
47 const NumberNode& number, int minimum_exponent, int maximum_exponent) {
48 const auto narrow = [minimum_exponent, maximum_exponent](const BigInt& exponent)
49 -> std::optional<int> {
50 if (exponent < BigInt(minimum_exponent) || exponent > BigInt(maximum_exponent)) {
51 return std::nullopt;
52 }
53 const auto bounded_exponent = exponent.try_to_int64();
54 if (!bounded_exponent) { return std::nullopt; }
55 return static_cast<int>(*bounded_exponent);
56 };
57 const auto& value = number.value();
58 if (const auto* exponent = std::get_if<BigInt>(&value)) { return narrow(*exponent); }
59 if (const auto* exponent = std::get_if<Rational>(&value)) {
60 if (!exponent->is_integer()) { return std::nullopt; }
61 return narrow(exponent->to_bigint());
62 }
63 if (const auto* exponent = std::get_if<lmmc_real_t>(&value)) {
64 if (!std::isfinite(*exponent)) { return std::nullopt; }
65 const Rational exact_exponent = Rational::from_double(*exponent);
66 if (!exact_exponent.is_integer()) { return std::nullopt; }
67 return narrow(exact_exponent.to_bigint());
68 }
69 return std::nullopt;
70}
