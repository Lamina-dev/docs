26 * The imaginary unit in LMCAS is represented as sqrt(-1), i.e., a FunctionNode
27 * with type Sqrt and a single argument that is a NumberNode with value -1.

50 void visit(const PowerNode& node) override {
51 if (found) return;
52
53 // Check if this is a negative base raised to a fractional exponent
54 // (e.g., (-4)^0.5 which produces an imaginary result)
55 auto base_num = std::dynamic_pointer_cast<const NumberNode>(node.base());
56 auto exp_num = std::dynamic_pointer_cast<const NumberNode>(node.exponent());
57 if (base_num && exp_num) {
58 double base_val = 0.0;
59 bool base_is_numeric = false;
60 if (std::holds_alternative<BigInt>(base_num->value())) {
61 base_val = std::get<BigInt>(base_num->value()).to_double();
62 base_is_numeric = true;
63 } else if (std::holds_alternative<Rational>(base_num->value())) {
64 base_val = std::get<Rational>(base_num->value()).to_double();
65 base_is_numeric = true;
66 } else if (std::holds_alternative<lmmc_real_t>(base_num->value())) {
67 base_val = std::get<lmmc_real_t>(base_num->value());
68 base_is_numeric = true;
69 }
70
71 double exp_val = 0.0;
72 bool exp_is_numeric = false;
73 if (std::holds_alternative<BigInt>(exp_num->value())) {
74 exp_val = std::get<BigInt>(exp_num->value()).to_double();
75 exp_is_numeric = true;
76 } else if (std::holds_alternative<Rational>(exp_num->value())) {
77 exp_val = std::get<Rational>(exp_num->value()).to_double();
78 exp_is_numeric = true;
79 } else if (std::holds_alternative<lmmc_real_t>(exp_num->value())) {
80 exp_val = std::get<lmmc_real_t>(exp_num->value());
81 exp_is_numeric = true;
82 }
83
84 if (base_is_numeric && exp_is_numeric) {
85 // Negative base with non-integer exponent → imaginary
86 if (base_val < 0.0) {
87 double rounded_exp = std::round(exp_val);
88 if (std::abs(exp_val - rounded_exp) > 1e-12) {
89 found = true;
90 return;
91 }
92 }
93 }
94 }
95
96 node.base()->accept(*this);
97 if (found) return;
98 node.exponent()->accept(*this);
99 }

101 void visit(const FunctionNode& node) override {
102 // Check if this is sqrt(-1) — the imaginary unit
103 if (node.type() == FunctionNode::FuncType::Sqrt && node.arguments().size() == 1) {
104 auto num = std::dynamic_pointer_cast<const NumberNode>(node.arguments()[0]);
105 if (num) {
106 if (std::holds_alternative<BigInt>(num->value())) {
107 if (std::get<BigInt>(num->value()) == BigInt(-1)) {
108 found = true;
109 return;
110 }
111 } else if (std::holds_alternative<Rational>(num->value())) {
112 if (std::get<Rational>(num->value()) == Rational(-1)) {
113 found = true;
114 return;
115 }
116 } else if (std::holds_alternative<lmmc_real_t>(num->value())) {
117 lmmc_real_t v = std::get<lmmc_real_t>(num->value());
118 int eq;
119 lmmc_double_nearly_equal(v, -1.0, &eq);
120 if (eq) {
121 found = true;
122 return;
123 }
124 }
125 }
126 }
127 // Also check for sqrt of any negative number (produces imaginary result)
128 if (node.type() == FunctionNode::FuncType::Sqrt && node.arguments().size() == 1) {
129 auto num = std::dynamic_pointer_cast<const NumberNode>(node.arguments()[0]);
130 if (num) {
131 if (std::holds_alternative<BigInt>(num->value())) {
132 if (std::get<BigInt>(num->value()).IsNegative()) {
133 found = true;
134 return;
135 }
136 } else if (std::holds_alternative<Rational>(num->value())) {
137 if (std::get<Rational>(num->value()) < Rational(0)) {
138 found = true;
139 return;
140 }
141 } else if (std::holds_alternative<lmmc_real_t>(num->value())) {
142 lmmc_real_t v = std::get<lmmc_real_t>(num->value());
143 if (std::isfinite(v) && v < 0.0) {
144 found = true;
145 return;
146 }
147 }
148 }
149 }
150 // Recurse into arguments
151 for (auto& arg : node.arguments()) {
152 if (found) return;
153 arg->accept(*this);
154 }

237/// Check if a solution expression contains imaginary components (sqrt of negative).
238static bool contains_imaginary(const std::shared_ptr<SymbolicExpr>& expr) {
239 if (!expr || !LMCAS::detail::node(expr)) return false;
240 ContainsImaginaryVisitor visitor;
241 LMCAS::detail::node(expr)->accept(visitor);
242 return visitor.found;
243}

245/// Try to extract a numeric double value from a solution expression.
246/// Returns true if the expression is a pure numeric value, and sets out_value.
247static bool try_get_numeric_value(const std::shared_ptr<SymbolicExpr>& expr, double& out_value) {
248 if (!expr || !LMCAS::detail::node(expr)) return false;
249 auto num = std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(expr));
250 if (!num) return false;
251
252 if (std::holds_alternative<BigInt>(num->value())) {
253 out_value = std::get<BigInt>(num->value()).to_double();
254 return true;
255 }
256 if (std::holds_alternative<Rational>(num->value())) {
257 out_value = std::get<Rational>(num->value()).to_double();
258 return true;
259 }
260 if (std::holds_alternative<lmmc_real_t>(num->value())) {
261 out_value = std::get<lmmc_real_t>(num->value());
262 return true;
263 }
264 return false;
265}

267/// Check if a numeric value is an integer (within tolerance).
268static bool is_integer_value(double v) {
269 if (!std::isfinite(v)) return false;
270 int eq;
271 lmmc_double_nearly_equal_tol(v, std::round(v), 1e-12, 1e-12, &eq);
272 return eq != 0;
273}

277static AssumptionSolveResult solve_with_assumptions_impl(
278 const std::shared_ptr<SymbolicExpr>& equation,
279 const std::string& variable,
280 const AssumptionContext* ctx,
281 ComputationContext& context)
282{
283 // Step 1: Compute all solutions using the existing solver
284 auto solved = solve_finite_checked(
285 std::const_pointer_cast<SymbolicExpr>(equation), variable, context);
286 if (!solved) return AssumptionSolveResult::failure(solved.error());
287 auto all_solutions = std::move(solved.value());
288
289 // Step 2: If no context provided, return all solutions unfiltered
290 if (!ctx) {
291 return all_solutions;
292 }
293
294 // Step 3: Get the domain for the variable from the context
295 Domain domain = ctx->get_domain(variable);
296
297 // Step 4: Apply domain filtering as post-processing
298 std::vector<std::shared_ptr<SymbolicExpr>> filtered;
299 filtered.reserve(all_solutions.size());
300
301 // Check if we have any sign constraints to apply
302 bool has_sign_constraint = ctx->has_sign(variable, Sign::NonNegative) ||
303 ctx->has_sign(variable, Sign::Positive) ||
304 ctx->has_sign(variable, Sign::Negative) ||
305 ctx->has_sign(variable, Sign::NonPositive);
306
307 // If domain is Complex (default/no restriction) and no sign constraints, return all
308 if (domain == Domain::Complex && !has_sign_constraint) {
309 return all_solutions;
310 }
311
312 for (const auto& sol : all_solutions) {
313 if (!sol) continue;
314
315 bool exclude = false;
316
317 // --- Real domain filter ---
318 // Exclude solutions containing imaginary components
319 if (domain == Domain::Real || domain == Domain::Algebraic ||
320 domain == Domain::Rational ||
321 domain == Domain::Integer || domain == Domain::Natural ||
322 domain == Domain::PositiveInt) {
323 if (contains_imaginary(sol)) {
324 exclude = true;
325 }
326 }
327
328 if (!exclude) {
329 double numeric_val = 0.0;
330 bool is_numeric = try_get_numeric_value(sol, numeric_val);
331
332 // --- Integer domain filter ---
333 // Exclude non-integer numeric solutions
334 if (domain == Domain::Integer || domain == Domain::Natural ||
335 domain == Domain::PositiveInt) {
336 if (is_numeric && !is_integer_value(numeric_val)) {
337 exclude = true;
338 }
339 }
340
341 // --- Natural domain filter (non-negative integers) ---
342 // Exclude negative solutions
343 if (!exclude && (domain == Domain::Natural || domain == Domain::PositiveInt)) {
344 if (is_numeric && numeric_val < 0.0) {
345 exclude = true;
346 }
347 }
348
349 // --- PositiveInt domain filter ---
350 // Exclude zero and negative solutions
351 if (!exclude && domain == Domain::PositiveInt) {
352 if (is_numeric) {
353 int eq;
354 lmmc_double_nearly_equal(numeric_val, 0.0, &eq);
355 if (eq || numeric_val <= 0.0) {
356 exclude = true;
357 }
358 }
359 }
360
361 // --- NonNegative sign check (for Real domain with NonNegative sign) ---
362 // Check if the variable has NonNegative sign assumption
363 if (!exclude && is_numeric) {
364 if (ctx->has_sign(variable, Sign::NonNegative) && numeric_val < 0.0) {
365 exclude = true;
366 }
367 if (ctx->has_sign(variable, Sign::Positive)) {
368 int eq;
369 lmmc_double_nearly_equal(numeric_val, 0.0, &eq);
370 if (eq || numeric_val <= 0.0) {
371 exclude = true;
372 }
373 }
374 if (ctx->has_sign(variable, Sign::Negative) && numeric_val >= 0.0) {
375 exclude = true;
376 }
377 if (ctx->has_sign(variable, Sign::NonPositive) && numeric_val > 0.0) {
378 exclude = true;
379 }
380 }
381 }
382
383 if (!exclude) {
384 filtered.push_back(sol);
385 }
386 }
387
388 return filtered;
389}

391AssumptionSolveResult solve_with_assumptions_checked(
392 const std::shared_ptr<SymbolicExpr>& equation,
393 const std::string& variable,
394 const AssumptionContext* assumptions,
395 ComputationContext& context)
396{
397 constexpr const char* operation = "solve_with_assumptions";
398 if (!equation || variable.empty()) {
399 return AssumptionSolveResult::failure(
400 CasErrc::InvalidArgument,
401 "assumption-aware solve requires an equation and variable",
402 operation);
403 }
404 auto budget = context.consume_steps(1, operation);
405 if (!budget) return AssumptionSolveResult::failure(budget.error());
406 try {
407 return solve_with_assumptions_impl(
408 equation, variable, assumptions, context);
409 } catch (const std::bad_alloc&) {
410 return AssumptionSolveResult::failure(
411 CasErrc::ResourceLimit,
412 "allocation failed while solving with assumptions",
413 operation);
414 } catch (const std::exception& ex) {
415 return AssumptionSolveResult::failure(
416 CasErrc::InternalInvariant, ex.what(), operation);
417 }
418}

421 const std::shared_ptr<SymbolicExpr>& equation,
422 const std::string& variable,

425 ComputationContext context;
