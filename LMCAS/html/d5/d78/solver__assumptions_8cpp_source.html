59static bool numeric_domain_excludes(double value, Domain domain) {
60 if (domain == Domain::Integer || domain == Domain::Natural || domain == Domain::PositiveInt) {
61 if (!is_integer_value(value)) {
62 return true;
63 }
64 }
65 if (domain == Domain::Natural || domain == Domain::PositiveInt) {
66 if (value < 0.0) {
67 return true;
68 }
69 }
70 if (domain == Domain::PositiveInt) {
71 int equal;
72 lmmc_double_nearly_equal(value, 0.0, &equal);
73 if (equal || value <= 0.0) {

76 }

80static bool numeric_sign_excludes(double value, const AssumptionContext& ctx,
81 const std::string& variable) {
82 bool exclude = false;
83 if (ctx.has_sign(variable, Sign::NonNegative) && value < 0.0) {
84 exclude = true;
85 }
86 if (ctx.has_sign(variable, Sign::Positive)) {
87 int equal;
88 lmmc_double_nearly_equal(value, 0.0, &equal);
89 if (equal || value <= 0.0) {
90 exclude = true;
91 }
92 }
93 if (ctx.has_sign(variable, Sign::Negative) && value >= 0.0) {
94 exclude = true;
95 }
96 if (ctx.has_sign(variable, Sign::NonPositive) && value > 0.0) {
97 exclude = true;
98 }
99 return exclude;
100}

111static const NumberNode* exact_scalar_radicand(const SymbolicNode* node) {

127static bool collect_scalar_radical(const SymbolicNode* node, Rational& coefficient,
128 const NumberNode*& radicand) {
129 const auto* product = dynamic_cast<const MultiplyNode*>(node);
130 if (!product) {
131 radicand = exact_scalar_radicand(node);
132 return radicand != nullptr;
133 }
134 for (const auto& factor : product->operands()) {
135 if (const auto* number = exact_scalar_number(factor.get())) {
136 coefficient = coefficient * exact_number_as_rational(*number);
137 } else {
138 if (radicand) {
139 return false;
140 }
141 radicand = exact_scalar_radicand(factor.get());
142 if (!radicand) {
143 return false;
144 }
145 }
146 }
147 return radicand != nullptr;
148}

166static bool exact_radical_domain_excludes(const SymbolicNode* node, Domain domain) {
167 if (domain != Domain::Rational && domain != Domain::Integer &&
168 domain != Domain::Natural && domain != Domain::PositiveInt) {
169 return false;
170 }
171 Rational coefficient(1);
172 const NumberNode* radicand = nullptr;
173 if (!collect_scalar_radical(node, coefficient, radicand)) {
174 return false;
175 }
176 if (coefficient == Rational(0)) {
177 return exact_rational_domain_excludes(coefficient, domain);
178 }
179 if (exact_number_as_rational(*radicand) < Rational(0)) {
180 return false;
181 }
182 const auto root = normalization_exact_square_root(*radicand);
183 if (!root) {
184 return true;
185 }
186 return exact_rational_domain_excludes(
187 coefficient * exact_number_as_rational(*root), domain);

190static Result<bool> solution_is_excluded(const std::shared_ptr<SymbolicExpr>& solution,
191 Domain domain, const AssumptionContext& ctx,
192 const std::string& variable,
193 ComputationContext& context) {
194 if (domain_requires_real(domain)) {
195 if (const auto* root = dynamic_cast<const RootOfNode*>(detail::node(solution).get())) {
196 auto isolated = detail::isolate_exact_root(
197 root->exact_id(), context, "solve_with_assumptions");
198 if (!isolated) return Result<bool>::failure(isolated.error());
199 if (std::holds_alternative<detail::ComplexIsolation>(isolated.value())) {
200 return true;
201 }
202 } else if (solver_detail::contains_imaginary(solution)) {
203 return true;
204 }
205 }
206 if (exact_radical_domain_excludes(detail::node(solution).get(), domain)) {
207 return true;
208 }
209 double value = 0.0;
210 if (!try_get_numeric_value(solution, value)) {
211 return false;
212 }
213 if (numeric_domain_excludes(value, domain)) {
214 return true;
215 }
216 return numeric_sign_excludes(value, ctx, variable);
217}

219static AssumptionSolveResult solve_with_assumptions_impl(
220 const std::shared_ptr<SymbolicExpr>& equation,
221 const std::string& variable,
222 const AssumptionContext* ctx,
223 ComputationContext& context)
224{
225 auto solved = solve_finite_checked(
226 std::const_pointer_cast<SymbolicExpr>(equation), variable, context);
227 if (!solved) {
228 return AssumptionSolveResult::failure(solved.error());
229 }
230 auto all_solutions = std::move(solved.value());
231
232 if (!ctx) {
233 return all_solutions;
234 }
235
236 Domain domain = ctx->get_domain(variable);
237
238 std::vector<std::shared_ptr<SymbolicExpr>> filtered;
239 filtered.reserve(all_solutions.size());
240
241 bool has_sign_constraint = ctx->has_sign(variable, Sign::NonNegative) ||
242 ctx->has_sign(variable, Sign::Positive) ||
243 ctx->has_sign(variable, Sign::Negative) ||
244 ctx->has_sign(variable, Sign::NonPositive);
245
246 if (domain == Domain::Complex && !has_sign_constraint) {
247 return all_solutions;
248 }
249
250 for (const auto& sol : all_solutions) {
251 if (!sol) {
252 continue;
253 }
254
255 auto excluded = solution_is_excluded(sol, domain, *ctx, variable, context);
256 if (!excluded) return AssumptionSolveResult::failure(excluded.error());
257 if (!excluded.value()) {
258 filtered.push_back(sol);
259 }
260 }
261
262 return filtered;
263}

265AssumptionSolveResult solve_with_assumptions_checked(
266 const std::shared_ptr<SymbolicExpr>& equation,
267 const std::string& variable,
268 const AssumptionContext* assumptions,
269 ComputationContext& context)
270{
271 constexpr const char* operation = "solve_with_assumptions";
272 if (!equation || variable.empty()) {
273 return AssumptionSolveResult::failure(
274 CasErrc::InvalidArgument,
275 "assumption-aware solve requires an equation and variable",
276 operation);
277 }
278 auto budget = context.consume_steps(1, operation);
279 if (!budget) {
280 return AssumptionSolveResult::failure(budget.error());
281 }
282 try {
283 return solve_with_assumptions_impl(
284 equation, variable, assumptions, context);
285 } catch (const std::bad_alloc&) {
286 return AssumptionSolveResult::failure(
287 CasErrc::ResourceLimit,
288 "allocation failed while solving with assumptions",
289 operation);
290 } catch (const std::exception& ex) {
291 return AssumptionSolveResult::failure(
292 CasErrc::InternalInvariant, ex.what(), operation);
293 }
294}
