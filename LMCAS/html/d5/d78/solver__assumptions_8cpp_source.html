23 * @brief Visitor that checks if an expression tree contains an imaginary unit.
24 *
25 * The imaginary unit in LMCAS is represented as sqrt(-1), i.e., a FunctionNode
26 * with type Sqrt and a single argument that is a NumberNode with value -1.
27 */

49 void visit(const PowerNode& node) override {
50 if (found) return;
51
52 // Check if this is a negative base raised to a fractional exponent
53 // (e.g., (-4)^0.5 which produces an imaginary result)
54 auto base_num = std::dynamic_pointer_cast<const NumberNode>(node.base());
55 auto exp_num = std::dynamic_pointer_cast<const NumberNode>(node.exponent());
56 if (base_num && exp_num) {
57 double base_val = 0.0;
58 bool base_is_numeric = false;
59 if (std::holds_alternative<BigInt>(base_num->value())) {
60 base_val = std::get<BigInt>(base_num->value()).to_double();
61 base_is_numeric = true;
62 } else if (std::holds_alternative<Rational>(base_num->value())) {
63 base_val = std::get<Rational>(base_num->value()).to_double();
64 base_is_numeric = true;
65 } else if (std::holds_alternative<lmmc_real_t>(base_num->value())) {
66 base_val = std::get<lmmc_real_t>(base_num->value());
67 base_is_numeric = true;
68 }
69
70 double exp_val = 0.0;
71 bool exp_is_numeric = false;
72 if (std::holds_alternative<BigInt>(exp_num->value())) {
73 exp_val = std::get<BigInt>(exp_num->value()).to_double();
74 exp_is_numeric = true;
75 } else if (std::holds_alternative<Rational>(exp_num->value())) {
76 exp_val = std::get<Rational>(exp_num->value()).to_double();
77 exp_is_numeric = true;
78 } else if (std::holds_alternative<lmmc_real_t>(exp_num->value())) {
79 exp_val = std::get<lmmc_real_t>(exp_num->value());
80 exp_is_numeric = true;
81 }
82
83 if (base_is_numeric && exp_is_numeric) {
84 // Negative base with non-integer exponent → imaginary
85 if (base_val < 0.0) {
86 double rounded_exp = std::round(exp_val);
87 if (std::abs(exp_val - rounded_exp) > 1e-12) {
88 found = true;
89 return;
90 }
91 }
92 }
93 }
94
95 node.base()->accept(*this);
96 if (found) return;
97 node.exponent()->accept(*this);
98 }

100 void visit(const FunctionNode& node) override {
101 // Check if this is sqrt(-1) — the imaginary unit
102 if (node.type() == FunctionNode::FuncType::Sqrt && node.arguments().size() == 1) {
103 auto num = std::dynamic_pointer_cast<const NumberNode>(node.arguments()[0]);
104 if (num) {
105 if (std::holds_alternative<BigInt>(num->value())) {
106 if (std::get<BigInt>(num->value()) == BigInt(-1)) {
107 found = true;
108 return;
109 }
110 } else if (std::holds_alternative<Rational>(num->value())) {
111 if (std::get<Rational>(num->value()) == Rational(-1)) {
112 found = true;
113 return;
114 }
115 } else if (std::holds_alternative<lmmc_real_t>(num->value())) {
116 lmmc_real_t v = std::get<lmmc_real_t>(num->value());
117 int eq;
118 lmmc_double_nearly_equal(v, -1.0, &eq);
119 if (eq) {
120 found = true;
121 return;
122 }
123 }
124 }
125 }
126 // Also check for sqrt of any negative number (produces imaginary result)
127 if (node.type() == FunctionNode::FuncType::Sqrt && node.arguments().size() == 1) {
128 auto num = std::dynamic_pointer_cast<const NumberNode>(node.arguments()[0]);
129 if (num) {
130 if (std::holds_alternative<BigInt>(num->value())) {
131 if (std::get<BigInt>(num->value()).IsNegative()) {
132 found = true;
133 return;
134 }
135 } else if (std::holds_alternative<Rational>(num->value())) {
136 if (std::get<Rational>(num->value()) < Rational(0)) {
137 found = true;
138 return;
139 }
140 } else if (std::holds_alternative<lmmc_real_t>(num->value())) {
141 lmmc_real_t v = std::get<lmmc_real_t>(num->value());
142 if (std::isfinite(v) && v < 0.0) {
143 found = true;
144 return;
145 }
146 }
147 }
148 }
149 // Recurse into arguments
150 for (auto& arg : node.arguments()) {
151 if (found) return;
152 arg->accept(*this);
153 }
154 }

236/// Check if a solution expression contains imaginary components (sqrt of negative).
237static bool contains_imaginary(const std::shared_ptr<SymbolicExpr>& expr) {
238 if (!expr || !lamina::detail::node(expr)) return false;
239 ContainsImaginaryVisitor visitor;
240 lamina::detail::node(expr)->accept(visitor);
241 return visitor.found;
242}

244/// Try to extract a numeric double value from a solution expression.
245/// Returns true if the expression is a pure numeric value, and sets out_value.
246static bool try_get_numeric_value(const std::shared_ptr<SymbolicExpr>& expr, double& out_value) {
247 if (!expr || !lamina::detail::node(expr)) return false;
248 auto num = std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(expr));
249 if (!num) return false;
250
251 if (std::holds_alternative<BigInt>(num->value())) {
252 out_value = std::get<BigInt>(num->value()).to_double();
253 return true;
254 }
255 if (std::holds_alternative<Rational>(num->value())) {
256 out_value = std::get<Rational>(num->value()).to_double();
257 return true;
258 }
259 if (std::holds_alternative<lmmc_real_t>(num->value())) {
260 out_value = std::get<lmmc_real_t>(num->value());
261 return true;
262 }
263 return false;
264}

266/// Check if a numeric value is an integer (within tolerance).
267static bool is_integer_value(double v) {
268 if (!std::isfinite(v)) return false;
269 int eq;
270 lmmc_double_nearly_equal_tol(v, std::round(v), 1e-12, 1e-12, &eq);
271 return eq != 0;
272}

276std::vector<std::shared_ptr<SymbolicExpr>> solve_with_assumptions(
277 const std::shared_ptr<SymbolicExpr>& equation,
278 const std::string& variable,
279 const AssumptionContext* ctx)
280{
281 // Step 1: Compute all solutions using the existing solver
282 auto all_solutions = SymbolicExpr::solve(
283 std::const_pointer_cast<SymbolicExpr>(equation), variable);
284
285 // Step 2: If no context provided, return all solutions unfiltered
286 if (!ctx) {
287 return all_solutions;
288 }
289
290 // Step 3: Get the domain for the variable from the context
291 Domain domain = ctx->get_domain(variable);
292
293 // Step 4: Apply domain filtering as post-processing
294 std::vector<std::shared_ptr<SymbolicExpr>> filtered;
295 filtered.reserve(all_solutions.size());
296
297 // Check if we have any sign constraints to apply
298 bool has_sign_constraint = ctx->has_sign(variable, Sign::NonNegative) ||
299 ctx->has_sign(variable, Sign::Positive) ||
300 ctx->has_sign(variable, Sign::Negative) ||
301 ctx->has_sign(variable, Sign::NonPositive);
302
303 // If domain is Complex (default/no restriction) and no sign constraints, return all
304 if (domain == Domain::Complex && !has_sign_constraint) {
305 return all_solutions;
306 }
307
308 for (const auto& sol : all_solutions) {
309 if (!sol) continue;
310
311 bool exclude = false;
312
313 // --- Real domain filter ---
314 // Exclude solutions containing imaginary components
315 if (domain == Domain::Real || domain == Domain::Algebraic ||
316 domain == Domain::Rational ||
317 domain == Domain::Integer || domain == Domain::Natural ||
318 domain == Domain::PositiveInt) {
319 if (contains_imaginary(sol)) {
320 exclude = true;
321 }
322 }
323
324 if (!exclude) {
325 double numeric_val = 0.0;
326 bool is_numeric = try_get_numeric_value(sol, numeric_val);
327
328 // --- Integer domain filter ---
329 // Exclude non-integer numeric solutions
330 if (domain == Domain::Integer || domain == Domain::Natural ||
331 domain == Domain::PositiveInt) {
332 if (is_numeric && !is_integer_value(numeric_val)) {
333 exclude = true;
334 }
335 }
336
337 // --- Natural domain filter (non-negative integers) ---
338 // Exclude negative solutions
339 if (!exclude && (domain == Domain::Natural || domain == Domain::PositiveInt)) {
340 if (is_numeric && numeric_val < 0.0) {
341 exclude = true;
342 }
343 }
344
345 // --- PositiveInt domain filter ---
346 // Exclude zero and negative solutions
347 if (!exclude && domain == Domain::PositiveInt) {
348 if (is_numeric) {
349 int eq;
350 lmmc_double_nearly_equal(numeric_val, 0.0, &eq);
351 if (eq || numeric_val <= 0.0) {
352 exclude = true;
353 }
354 }
355 }
356
357 // --- NonNegative sign check (for Real domain with NonNegative sign) ---
358 // Check if the variable has NonNegative sign assumption
359 if (!exclude && is_numeric) {
360 if (ctx->has_sign(variable, Sign::NonNegative) && numeric_val < 0.0) {
361 exclude = true;
362 }
363 if (ctx->has_sign(variable, Sign::Positive)) {
364 int eq;
365 lmmc_double_nearly_equal(numeric_val, 0.0, &eq);
366 if (eq || numeric_val <= 0.0) {
367 exclude = true;
368 }
369 }
370 if (ctx->has_sign(variable, Sign::Negative) && numeric_val >= 0.0) {
371 exclude = true;
372 }
373 if (ctx->has_sign(variable, Sign::NonPositive) && numeric_val > 0.0) {
374 exclude = true;
375 }
376 }
377 }
378
379 if (!exclude) {
380 filtered.push_back(sol);
381 }
382 }
383
384 return filtered;
385}
