131 static void implications(ValueFacts& result) {
132 if (result.nonnegative == T::True && result.real != T::False) {
133 result.real = T::True;
134 if (result.nonzero == T::True) result.positive = T::True;
135 }
136 if (result.real == T::False) {
137 result.positive = result.nonnegative = T::False;
138 } else if (result.positive == T::True) {
139 result.real = result.nonzero = result.nonnegative = T::True;
140 } else if (result.nonzero == T::False) {
141 result.real = result.nonnegative = T::True;
142 result.positive = T::False;
143 } else if (result.real == T::True && result.nonnegative == T::False) {
144 result.nonzero = T::True;
145 result.positive = T::False;
146 }
147 }

149 void infer(const Node& node, ValueFacts& result) {
150 implications(result);
151 if (result.positive == T::Unknown) { result.positive = checked(facts_.is_positive(node, context_)); }
152 if (error) { return; }
153 implications(result);
154 if (result.real == T::Unknown) { result.real = checked(facts_.is_real(node, context_)); }
155 if (error) { return; }
156 if (result.nonzero == T::Unknown) { result.nonzero = checked(facts_.is_nonzero(node, context_)); }
157 if (error) { return; }
158 if (result.nonnegative == T::Unknown) { result.nonnegative = checked(facts_.is_nonnegative(node, context_)); }
159 if (error) { return; }
160 if (result.positive == T::Unknown || result.nonnegative == T::Unknown) {
161 auto negative = checked(facts_.is_negative(node, context_));
162 if (error) { return; }
163 if (negative == T::True) {
164 result.real = result.nonzero = T::True;
165 result.positive = result.nonnegative = T::False;
166 }
167 }
168 implications(result);

171 struct ArithmeticReduction {
172 T all_real = T::True, all_nonnegative = T::True, all_nonzero = T::True;
173 bool any_positive = false, any_zero = false, all_zero = true, signs_known = true;
174 unsigned negative_parity = 0;
175 std::size_t nonreal_count = 0;
176 bool classified_reality = true, other_factors_nonzero = true;
177
178 ValueFacts finish(ValueFacts result, bool product) const {
179 if (all_real == T::True) { result.real = T::True; }
180 if (nonreal_count == 1 && classified_reality && (!product || other_factors_nonzero))
181 { result.real = T::False; }
182 if (product) {
183 if (any_zero) { result.nonzero = T::False; }
184 else if (all_nonzero == T::True) { result.nonzero = T::True; }
185 if (signs_known) {
186 result.positive = result.nonnegative = negative_parity ? T::False : T::True;
187 }
188 } else if (all_nonnegative == T::True) {
189 result.nonnegative = T::True;
190 if (any_positive) { result.positive = T::True; }
191 if (all_zero) { result.nonzero = T::False; }
192 }
193 return result;
194 }

197 ValueFacts arithmetic(const std::vector<Node>& operands, bool product, bool values) {
198 ValueFacts result;
199 result.defined = T::True;
200 ArithmeticReduction reduction;
201 for (const auto& operand : operands) {
202 auto child = walk(operand, values);
203 if (error) { return {}; }
204 result.defined = conjunction(result.defined, child.defined);
205 if (!values) { continue; }
206 reduction.all_real = conjunction(reduction.all_real, child.real);
207 reduction.all_nonnegative = conjunction(reduction.all_nonnegative, child.nonnegative);
208 reduction.all_nonzero = conjunction(reduction.all_nonzero, child.nonzero);
209 reduction.any_positive |= child.positive == T::True;
210 reduction.any_zero |= child.nonzero == T::False;
211 reduction.all_zero &= child.nonzero == T::False;
212 if (child.real == T::False) { ++reduction.nonreal_count; }
213 else if (child.real == T::Unknown) { reduction.classified_reality = false; }
214 else if (child.nonzero != T::True) { reduction.other_factors_nonzero = false; }
215 if (child.positive == T::True) {
216 continue;
217 } else if (child.real == T::True && child.nonnegative == T::False) {
218 reduction.negative_parity ^= 1;
219 } else {
220 reduction.signs_known = false;
221 }
222 }
223 if (!values) { return result; }
224 return reduction.finish(result, product);
225 }

227 ValueFacts power(const PowerNode& node, bool values) {
228 const auto exponent_number = std::dynamic_pointer_cast<const NumberNode>(node.exponent());
229 BigInt integer;
230 const bool integral = exponent_number && try_get_integer_value(exponent_number, integer);
231 bool needs_base_value = values || !integral;
232 if (!needs_base_value) {
233 needs_base_value = integer.is_negative() ||
234 (integer.is_zero() && domain_ == Domain::Real);
235 }
236 auto base = walk(node.base(), needs_base_value);
237 if (error) { return {}; }
238 auto exponent = walk(node.exponent(), !integral);
239 if (error) { return {}; }
240 ValueFacts result;
241 result.defined = conjunction(base.defined, exponent.defined);
242 if (integral) {
243 return integral_power(node, base, integer, result);
244 }
245 if (exponent_number) {
246 return nonintegral_power(node, base, exponent, result, true);
247 }
248 if (exponent.nonzero == T::False) {
249 return integral_power(node, base, BigInt(0), result);
250 }
251 const auto exact_exponent = closed_rational_value(node.exponent(), domain_, context_);
252 if (exact_exponent && exact_exponent->is_integer()) {
253 return integral_power(node, base, exact_exponent->to_bigint(), result);
254 }
255 return nonintegral_power(node, base, exponent, result, exact_exponent.has_value());

258 ValueFacts integral_power(const PowerNode& node, const ValueFacts& base, const BigInt& integer, ValueFacts result) {
259 const bool zero = integer == BigInt(0);
260 if (integer.is_negative() || (zero && domain_ == Domain::Real)) {
261 require(result, base.nonzero, [&] {
262 return relation(node.base(), number(0), RelationOp::NEQ);
263 });
264 }
265 if (zero) {
266 result.real = result.nonzero = result.positive = result.nonnegative = T::True;
267 } else {
268 if (base.real == T::True) { result.real = T::True; }
269 const T nonzero = project_ && integer.is_negative() ? T::True : base.nonzero;
270 result.nonzero = nonzero;
271 if (base.real == T::True) {
272 if (integer % BigInt(2) == BigInt(0)) {
273 result.nonnegative = T::True;
274 result.positive = nonzero;
275 } else {
276 result.positive = base.positive;
277 result.nonnegative = base.nonnegative;
278 }
279 }
280 }
281 return result;

284 T nonintegral_domain(const ValueFacts& base, const ValueFacts& exponent, bool known_noninteger, bool& unresolved_integer_domain) {
285 T allowed;
286 unresolved_integer_domain = false;
287 if (domain_ == Domain::Real) {
288 allowed = disjunction(conjunction(base.positive, exponent.real),
289 conjunction(base.nonnegative, exponent.positive));
290 unresolved_integer_domain = !known_noninteger && base.nonnegative != T::True;
291 if (unresolved_integer_domain && allowed != T::True) { allowed = T::Unknown; }
292 if (known_noninteger && base.real == T::True && base.nonnegative == T::False)
293 { allowed = T::False; }
294 if (base.nonzero == T::False && exponent.positive == T::False &&
295 exponent.real == T::True) { allowed = T::False; }
296 } else {
297 const T zero_nonnegative = conjunction(
298 base.nonzero == T::False ? T::True :
299 base.nonzero == T::True ? T::False : T::Unknown,
300 exponent.nonnegative);
301 allowed = disjunction(base.nonzero, zero_nonnegative);
302 }
303 return allowed;
304 }

357 ValueFacts elementary(const FunctionNode& node, bool values) {
358 const auto& arguments = node.arguments();
359 if (node.type() == FT::Infinity) { return {T::False}; }
360 const bool logarithm_base = node.type() == FT::Log && arguments.size() == 2;
361 if (arguments.size() != 1 && !logarithm_base) {
362 expressible = false;
363 return {};
364 }
365 const auto& argument = arguments[0];
366 const bool restricted = restricted_argument(node.type());
367 auto child = walk(argument, values || restricted);
368 if (error) { return {}; }
369 ValueFacts result;
370 result.defined = child.defined;
371 return elementary_value(node, child, result);

388 ValueFacts elementary_value(const FunctionNode& node, const ValueFacts& child, ValueFacts result) {
389 switch (node.type()) {
390 case FT::Exp: {
391 result.nonzero = T::True;
392 if (child.real == T::True)
393 { result.real = result.positive = result.nonnegative = T::True; }
394 return result;
395 }
396 case FT::Sin: { break; }
397 case FT::Cos: { break; }
398 case FT::Sinh: { break; }
399 case FT::Cosh: { break; }
400 case FT::Erf: { break; }
401 case FT::Si: { break; }
402 default: { return restricted_function(node, child, result); }
403 }
404 if (child.real == T::True) { result.real = T::True; }
405 if (child.nonzero == T::False) {
406 result.nonzero = node.type() == FT::Cos || node.type() == FT::Cosh
407 ? T::True : T::False;
408 if (result.nonzero == T::True) { result.positive = T::True; }
409 }
410 return result;

413 ValueFacts restricted_function(const FunctionNode& node, const ValueFacts& child, ValueFacts result) {
414 switch (node.type()) {
415 case FT::Ln: { return logarithm(node.arguments(), child, result); }
416 case FT::Log: { return logarithm(node.arguments(), child, result); }
417 case FT::Sqrt: { return real_valued_function(node, child, result); }
418 case FT::Abs: { return real_valued_function(node, child, result); }
419 case FT::ComplexAbs: { return real_valued_function(node, child, result); }
420 case FT::Tan: { return trigonometric_domain(node, child, result); }
421 case FT::Sec: { return trigonometric_domain(node, child, result); }
422 case FT::Cot: { return trigonometric_domain(node, child, result); }
423 case FT::Csc: { return trigonometric_domain(node, child, result); }
424 case FT::ArcSin: { return trigonometric_domain(node, child, result); }
425 case FT::ArcCos: { return trigonometric_domain(node, child, result); }
426 case FT::Sgn: { return real_valued_function(node, child, result); }
427 case FT::ArcTan: { return trigonometric_domain(node, child, result); }
428 case FT::LambertW: { return lambert_w_domain(node, child, result); }
429 default: {
430 // Unsupported singularities and conditional branches remain undecidable.
431 result.defined = conjunction(result.defined, T::Unknown);
432 expressible = false;
433 break;
434 }
435 }
436 return result;

581 ValueFacts structural(const Node& node, bool values) {
582 if (const auto* number = dynamic_cast<const NumberNode*>(node.get())) {
583 const T nz = number->is_zero() ? T::False : T::True;
584 const T positive = number->is_positive() ? T::True : T::False;
585 return {T::True, T::True, nz, positive,
586 nz == T::False || positive == T::True ? T::True : T::False};
587 }
588 if (const auto* variable = dynamic_cast<const VariableNode*>(node.get())) {
589 return variable_facts(*variable);
590 }
591 if (const auto* add = dynamic_cast<const AddNode*>(node.get()))
592 { return arithmetic(add->operands(), false, values); }
593 if (const auto* multiply = dynamic_cast<const MultiplyNode*>(node.get()))
594 { return arithmetic(multiply->operands(), true, values); }
595 if (const auto* pow = dynamic_cast<const PowerNode*>(node.get())) { return power(*pow, values); }
596 if (const auto* fn = dynamic_cast<const FunctionNode*>(node.get())) { return elementary(*fn, values); }
597 if (const auto* complex = dynamic_cast<const ComplexNode*>(node.get())) {
598 return complex_facts(*complex);
599 }
600 if (const auto* root = dynamic_cast<const RootOfNode*>(node.get())) {
601 return root_facts(*root, values);
602 }
603 if (const auto* relation = dynamic_cast<const RelationalNode*>(node.get())) {

621 ValueFacts complex_facts(const ComplexNode& complex) {
622 auto real = walk(complex.real(), true);
623 if (error) { return {}; }
624 auto imag = walk(complex.imag(), true);
625 if (error) { return {}; }
626 ValueFacts result;
627 result.defined = conjunction(conjunction(real.defined, imag.defined),
628 conjunction(real.real, imag.real));
629 if (real.real == T::Unknown || imag.real == T::Unknown) { expressible = false; }
630 if (domain_ == Domain::Real) {
631 const T zero = imag.nonzero == T::False ? T::True :
632 imag.nonzero == T::True ? T::False : T::Unknown;
633 require(result, zero, [&] {
634 return relation(complex.imag(), number(0), RelationOp::EQ);
635 }, false);
636 }
637 result.nonzero = disjunction(real.nonzero, imag.nonzero);
638 result.real = imag.nonzero == T::False ? T::True :
639 imag.nonzero == T::True ? T::False : T::Unknown;
640 if (result.real == T::True) {
641 result.positive = real.positive;
642 result.nonnegative = real.nonnegative;
643 }
644 return result;
645 }

647 ValueFacts root_facts(const RootOfNode& root, bool values) {
648 ValueFacts result;
649 result.defined = T::True; /**< ExactRootId 已在构造时校验。 */
650 if (values || domain_ == Domain::Real) {
651 auto isolation = isolate_exact_root(root.exact_id(), context_, "facts.rootof");
652 if (!isolation) {
653 error = isolation.error();
654 return {};
655 }
656 if (const auto* real = std::get_if<RealIsolation>(&isolation.value())) {
657 result.real = T::True;
658 if (real->value.lower > Rational(0)) { result.positive = T::True; }
659 else if (real->value.upper < Rational(0)) { result.nonnegative = T::False; }
660 else if (real->value.lower == Rational(0) && real->value.upper == Rational(0))
661 { result.nonzero = T::False; }
662 } else {
663 result.real = T::False;
664 result.nonzero = T::True;
665 if (domain_ == Domain::Real) { result.defined = T::False; }
666 }
667 }
668 return result;
669 }

714Result<std::optional<std::vector<std::shared_ptr<SymbolicExpr>>>> domain_constraints(
715 const Node& node, const FactsQuery& facts, Domain domain, ComputationContext& context) {
716 using Output = std::optional<std::vector<std::shared_ptr<SymbolicExpr>>>;
717 if (domain != Domain::Real && domain != Domain::Complex) {
718 return Result<Output>::failure(CasErrc::InvalidArgument,
719 "value domain must be Real or Complex", "facts.domain");
720 }
721 try {
722 DomainTraversal traversal(facts, domain, context, true);
723 auto result = traversal.walk(node);
724 if (traversal.error) return Result<Output>::failure(*traversal.error);
725 if (result.defined == T::False) {
726 if (domain == Domain::Real && !traversal.violated_real_restriction) {
727 auto complex_defined = query_definedness(node, facts, Domain::Complex, context);
728 if (!complex_defined) return Result<Output>::failure(complex_defined.error());
729 if (complex_defined.value() != T::False) {
730 auto real_value = query_real_value(node, facts, context);
731 if (!real_value) return Result<Output>::failure(real_value.error());
732 if (real_value.value() == T::Unknown) return Output{};
733 }
734 }
735 auto impossible = traversal.false_condition();
736 if (traversal.error) return Result<Output>::failure(*traversal.error);
737 traversal.conditions.clear();
738 traversal.conditions.push_back(make_expression_ptr(std::move(impossible)));
739 } else if (!traversal.expressible) {
740 return Output{};
741 }
742 return Output{std::move(traversal.conditions)};
743 } catch (const CasError& error) {
744 return Result<Output>::failure(error);
745 } catch (const std::bad_alloc&) {
746 return Result<Output>::failure(CasErrc::ResourceLimit,
747 "domain projection allocation failed", "facts.domain");
748 }
