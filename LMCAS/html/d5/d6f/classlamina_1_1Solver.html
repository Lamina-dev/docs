912 for (int i = 0; i <= var_pos && i < (int)variables.size(); ++i) {
913 if (contains(*r, variables[i])) {

929 }

933 std::shared_ptr<SymbolicExpr> target = nullptr;
934 int best_deg = std::numeric_limits<int>::max();
935
936 for (const auto& r : reduced) {
937 if (!contains(*r, curr_var)) continue;
938 curr_var_appears = true;
939
940 bool has_other = false;
941 for (int i = 0; i < var_pos; ++i) {
942 if (contains(*r, variables[i])) {
943 has_other = true;
944 break;
945 }
946 }

950 int deg = poly.degree();

955 }

957 if (!curr_var_appears) {
958 auto next_partial = partial;
959 next_partial.insert_or_assign(curr_var, *SymbolicExpr::variable(curr_var));

961 }

966 if (roots.empty()) return {};

969 for (const auto& r : roots) {
970 auto next_partial = partial;
971 next_partial.insert_or_assign(curr_var, *r);
972 auto sub_res = self(self, var_pos - 1, next_partial);
973 results.insert(results.end(), sub_res.begin(), sub_res.end());
974 }
975 return results;
976 };
977
978 std::map<std::string, SymbolicExpr> empty;
979 auto candidates = solve_rec(solve_rec, static_cast<int>(variables.size()) - 1, empty);
980
981 if (denom_constraints.empty()) return candidates;

984 filtered.reserve(candidates.size());

988 for (const auto& den : denom_constraints) {
989 auto sub = den;
990 for (const auto& [name, val] : sol) {
991 sub = sub->substitute(name, lamina::detail::make_expression_ptr(val));
992 if (!sub) break;
993 }
994 if (!sub) continue;
995 sub = sub->simplify();
996 if (sub && sub->is_zero()) {
997 ok = false;
998 break;
999 }
1000 }
1001 if (ok) filtered.push_back(sol);
1002 }
1003
1004 return filtered;
