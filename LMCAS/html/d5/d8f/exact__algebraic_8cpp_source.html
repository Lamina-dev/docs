17Polynomial<Rational> monic(Polynomial<Rational> polynomial) {
18 polynomial.trim();
19 if (polynomial.is_zero()) return polynomial;
20 const Rational leading = polynomial.lead_coeff();
21 for (auto& coefficient : polynomial.coeffs) coefficient = coefficient / leading;
22 polynomial.trim();
23 return polynomial;
24}

44ExactRealAlgebraicResult make_exact_real_algebraic(
45 Polynomial<Rational> polynomial,
46 std::size_t root_index,
47 std::size_t multiplicity,
48 ComputationContext& context) {
49 if (polynomial.degree() < 1) {
50 return ExactRealAlgebraicResult::failure(
51 CasErrc::InvalidArgument,
52 "an algebraic value requires a non-constant polynomial",
53 kOperation);
54 }
55 if (multiplicity == 0) {
56 return ExactRealAlgebraicResult::failure(
57 CasErrc::InvalidArgument,
58 "algebraic root multiplicity must be positive",
59 kOperation);
60 }
61 auto step = context.consume_steps(1, kOperation);
62 if (!step) return ExactRealAlgebraicResult::failure(step.error());
63
64 try {
65 polynomial = monic(std::move(polynomial));
66 auto rational_roots = find_rational_roots(polynomial);
67 std::sort(rational_roots.begin(), rational_roots.end());
68 rational_roots.erase(
69 std::unique(rational_roots.begin(), rational_roots.end()),
70 rational_roots.end());
71 if (rational_roots.size() ==
72 static_cast<std::size_t>(polynomial.degree())) {
73 if (root_index >= rational_roots.size()) {
74 return ExactRealAlgebraicResult::failure(
75 CasErrc::InvalidArgument,
76 "real algebraic root index is out of range",
77 kOperation);
78 }
79 return ExactRealAlgebraicResult::success(ExactRealAlgebraic{
80 std::move(polynomial), rational_roots[root_index],
81 rational_roots[root_index], root_index, multiplicity});
82 }
83 auto isolated = isolate_real_roots_exact(
84 polynomial, context, kOperation);
85 if (!isolated) {
86 return ExactRealAlgebraicResult::failure(isolated.error());
87 }
88 const auto& intervals = isolated.value();
89 if (root_index >= intervals.size()) {
90 return ExactRealAlgebraicResult::failure(
91 CasErrc::InvalidArgument,
92 "real algebraic root index is out of range",
93 kOperation);
94 }
95 return ExactRealAlgebraicResult::success(ExactRealAlgebraic{
96 std::move(polynomial), intervals[root_index].first,
97 intervals[root_index].second, root_index, multiplicity});
98 } catch (const std::bad_alloc&) {
99 return ExactRealAlgebraicResult::failure(
100 CasErrc::ResourceLimit,
101 "algebraic root isolation allocation failed",
102 kOperation);
103 } catch (const std::exception& error) {
104 return ExactRealAlgebraicResult::failure(
105 CasErrc::InternalInvariant, error.what(), kOperation);
106 }

109Result<void> refine_exact_real_algebraic(
110 ExactRealAlgebraic& value,
111 ComputationContext& context,
112 const std::string& operation) {
113 auto step = context.consume_steps(1, operation);
114 if (!step) return step;
115 if (value.lower == value.upper) return Result<void>::success();
116
117 const Rational lower_value = value.polynomial.eval(value.lower);
118 if (lower_value == Rational(0)) {
119 value.upper = value.lower;
120 return Result<void>::success();
121 }
122 const Rational upper_value = value.polynomial.eval(value.upper);
123 if (upper_value == Rational(0)) {
124 value.lower = value.upper;
125 return Result<void>::success();
126 }
127
128 const Rational midpoint = (value.lower + value.upper) / Rational(2);
129 const Rational midpoint_value = value.polynomial.eval(midpoint);
130 if (midpoint_value == Rational(0)) {
131 value.lower = midpoint;
132 value.upper = midpoint;
133 return Result<void>::success();
134 }
135
136 if (sign(lower_value) != sign(midpoint_value)) {
137 value.upper = midpoint;
138 } else {
139 value.lower = midpoint;
140 }
141 return Result<void>::success();
142}

144Result<void> refine_exact_real_algebraic_to_tolerance(
145 ExactRealAlgebraic& value,
146 double absolute_tolerance,
147 double relative_tolerance,
148 ComputationContext& context,
149 const std::string& operation) {
150 if (!(absolute_tolerance > 0.0) ||
151 !(relative_tolerance > 0.0) ||
152 !std::isfinite(absolute_tolerance) ||
153 !std::isfinite(relative_tolerance)) {
154 return Result<void>::failure(
155 CasErrc::InvalidArgument,
156 "algebraic refinement tolerances must be finite and positive",
157 operation);
158 }
159 if (value.lower == value.upper) return Result<void>::success();
160
161 try {
162
163 const Rational base = value.lower;
164 const Rational span = value.upper - value.lower;
165 const double base_double = base.to_double();
166 const double upper_double = value.upper.to_double();
167 auto evaluate_double = [&](double point) {
168 double result = 0.0;
169 for (std::size_t position = value.polynomial.coeffs.size();
170 position-- > 0;) {
171 result = result * point +
172 value.polynomial.coeffs[position].to_double();
173 }
174 return result;
175 };
176 double approximate_lower = base_double;
177 double approximate_upper = upper_double;
178 double lower_value = evaluate_double(approximate_lower);
179 for (int iteration = 0; iteration < 128; ++iteration) {
180 const double midpoint =
181 (approximate_lower + approximate_upper) * 0.5;
182 const double midpoint_value = evaluate_double(midpoint);
183 if (!std::isfinite(midpoint_value) || midpoint_value == 0.0) {
184 approximate_lower = midpoint;
185 approximate_upper = midpoint;
186 break;
187 }
188 if ((lower_value < 0.0) != (midpoint_value < 0.0)) {
189 approximate_upper = midpoint;
190 } else {
191 approximate_lower = midpoint;
192 lower_value = midpoint_value;
193 }
194 }
195 const double approximation =
196 (approximate_lower + approximate_upper) * 0.5;
197 const double target_tolerance = absolute_tolerance +
198 relative_tolerance * std::abs(approximation);
199 const double span_double = span.to_double();
200 int initial_depth = 1;
201 if (span_double > 2.0 * target_tolerance) {
202 initial_depth = static_cast<int>(std::ceil(std::log2(
203 span_double / (2.0 * target_tolerance)))) + 2;
204 }
205 if (initial_depth > 60) {
206 return Result<void>::failure(
207 CasErrc::NumericFailure,
208 "requested algebraic tolerance is finer than the double evaluation boundary",
209 operation);
210 }
211
212 const double fraction = std::clamp(
213 (approximation - base_double) / span_double, 0.0, 1.0);
214 for (int depth = std::max(1, initial_depth); depth <= 60; ++depth) {
215 auto step = context.consume_steps(1, operation);
216 if (!step) return step;
217 const unsigned long long scale_value = 1ULL << depth;
218 unsigned long long center = static_cast<unsigned long long>(
219 std::floor(fraction * static_cast<double>(scale_value)));
220 if (center >= scale_value) center = scale_value - 1;
221 for (int radius = 0; radius <= 16; ++radius) {
222 for (int direction : {radius == 0 ? 0 : -1,
223 radius == 0 ? 0 : 1}) {
224 if (radius == 0 && direction != 0) continue;
225 const long long candidate_signed =
226 static_cast<long long>(center) +
227 static_cast<long long>(direction * radius);
228 if (candidate_signed < 0 ||
229 static_cast<unsigned long long>(candidate_signed) >=
230 scale_value) {
231 continue;
232 }
233 const auto candidate =
234 static_cast<unsigned long long>(candidate_signed);
235 const BigInt scale_integer(scale_value);
236 const Rational lower = base + span * Rational(
237 BigInt(candidate), scale_integer);
238 const Rational upper = base + span * Rational(
239 BigInt(candidate + 1), scale_integer);
240 auto exact_count = count_real_roots_exact(
241 value.polynomial, lower, upper,
242 context, operation);
243 if (!exact_count) {
244 return Result<void>::failure(exact_count.error());
245 }
246 const bool contains_root = exact_count.value() == 1;
247 if (contains_root) {
248 value.lower = lower;
249 value.upper = upper;
250 return Result<void>::success();
251 }
252 if (radius == 0) break;
253 }
254 }
255 }
256 return Result<void>::failure(
257 CasErrc::InternalInvariant,
258 "exact projection refinement could not certify the suggested bracket",
259 operation);
260 } catch (const std::bad_alloc&) {
261 return Result<void>::failure(
262 CasErrc::ResourceLimit,
263 "algebraic refinement allocation failed",
264 operation);
265 } catch (const std::exception& error) {
266 return Result<void>::failure(
267 CasErrc::InternalInvariant, error.what(), operation);
268 }

271Result<bool> equal_exact_real_algebraic(
272 ExactRealAlgebraic lhs,
273 ExactRealAlgebraic rhs,
274 ComputationContext& context) {
275 auto step = context.consume_steps(1, "exact_algebraic.equal");
276 if (!step) return Result<bool>::failure(step.error());
277 if (!overlaps(lhs.lower, lhs.upper, rhs.lower, rhs.upper)) {
278 return Result<bool>::success(false);
279 }
280 if (same_polynomial(lhs.polynomial, rhs.polynomial)) {
281 return Result<bool>::success(lhs.root_index == rhs.root_index);
282 }
283
284 const Rational overlap_lower =
285 lhs.lower < rhs.lower ? rhs.lower : lhs.lower;
286 const Rational overlap_upper =
287 lhs.upper < rhs.upper ? lhs.upper : rhs.upper;
288 if (overlap_lower == overlap_upper &&
289 (lhs.polynomial.eval(overlap_lower) != Rational(0) ||
290 rhs.polynomial.eval(overlap_lower) != Rational(0))) {
291 return Result<bool>::success(false);
292 }
293
294 try {
295 auto common = monic(Polynomial<Rational>::gcd(
296 lhs.polynomial, rhs.polynomial));
297 if (common.degree() < 1) return Result<bool>::success(false);
298 auto isolated = isolate_real_roots_exact(
299 common, context, "exact_algebraic.equal");
300 if (!isolated) return Result<bool>::failure(isolated.error());
301 for (const auto& interval : isolated.value()) {
302 if (overlaps(lhs.lower, lhs.upper, interval.first, interval.second) &&
303 overlaps(rhs.lower, rhs.upper, interval.first, interval.second)) {
304 return Result<bool>::success(true);
305 }
306 }
307 return Result<bool>::success(false);
308 } catch (const std::bad_alloc&) {
309 return Result<bool>::failure(
310 CasErrc::ResourceLimit,
311 "algebraic equality allocation failed",
312 "exact_algebraic.equal");
313 } catch (const std::exception& error) {
314 return Result<bool>::failure(
315 CasErrc::InternalInvariant,
316 error.what(),
317 "exact_algebraic.equal");
318 }

321Result<int> compare_exact_real_algebraic(
322 ExactRealAlgebraic lhs,
323 ExactRealAlgebraic rhs,
324 ComputationContext& context) {
325 auto equal = equal_exact_real_algebraic(lhs, rhs, context);
326 if (!equal) return Result<int>::failure(equal.error());
327 if (equal.value()) return Result<int>::success(0);
328
329 while (true) {
330 if (lhs.upper < rhs.lower) return Result<int>::success(-1);
331 if (rhs.upper < lhs.lower) return Result<int>::success(1);
332 const double lhs_width = (lhs.upper - lhs.lower).to_double();
333 const double rhs_width = (rhs.upper - rhs.lower).to_double();
334 const double width = std::max(lhs_width, rhs_width);
335 if (!(width > 0.0) || !std::isfinite(width)) {
336 return Result<int>::failure(
337 CasErrc::InternalInvariant,
338 "distinct algebraic values retained an inseparable enclosure",
339 "exact_algebraic.compare");
340 }
341 if (lhs_width > 0.0) {
342 auto refined = refine_exact_real_algebraic_to_tolerance(
343 lhs, width / 16.0, 1e-15, context,
344 "exact_algebraic.compare");
345 if (!refined) return Result<int>::failure(refined.error());
346 }
347 if (rhs_width > 0.0) {
348 auto refined = refine_exact_real_algebraic_to_tolerance(
349 rhs, width / 16.0, 1e-15, context,
350 "exact_algebraic.compare");
351 if (!refined) return Result<int>::failure(refined.error());
352 }
353 }
