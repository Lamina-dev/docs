253void DifferentiationVisitor::differentiate_atan2(const FunctionNode& node) {
254 const auto& y = node.arguments()[0];
255 const auto& x = node.arguments()[1];
256 y->accept(*this);
257 auto dy = result;
258 x->accept(*this);
259 auto dx = result;
260 auto numerator = SymbolicFactory::create_add({
261 SymbolicFactory::create_multiply({x, dy}),
262 SymbolicFactory::create_multiply({
263 SymbolicFactory::create_number(BigInt(-1)), y, dx})
264 });
265 auto two = SymbolicFactory::create_number(BigInt(2));
266 auto denominator = SymbolicFactory::create_add({
267 LMCAS::detail::make_node<PowerNode>(x, two),
268 LMCAS::detail::make_node<PowerNode>(y, two)
269 });
270 auto inverse = LMCAS::detail::make_node<PowerNode>(
271 denominator, SymbolicFactory::create_number(BigInt(-1)));
272 result = SymbolicFactory::create_multiply({numerator, inverse});

275void DifferentiationVisitor::visit(const FunctionNode& node) {
276 if (node.type() == FunctionNode::FuncType::Atan2 && node.arguments().size() == 2) {
277 differentiate_atan2(node);
278 return;
279 }
280 if (node.type() == FunctionNode::FuncType::Log && node.arguments().size() == 2) {
281 const auto& arg = node.arguments()[0];
282 if (!valid_log_base(node.arguments()[1])) {
283 unsupported("FunctionNode");
284 }
285 arg->accept(*this);
286 auto d_arg = result;
287 if (d_arg->is_zero()) return;
288 auto inverse = logarithmic_outer(node);
289 result = SymbolicFactory::create_multiply({std::move(d_arg), inverse});
290 return;
291 }
292 if (node.arguments().size() != 1) unsupported("FunctionNode");
293 node.arguments()[0]->accept(*this);
294 auto d_arg = result;
295 if (d_arg->is_zero()) {
296 return;
297 }
298
299 auto d_outer = trigonometric_outer(node);
300 if (!d_outer) d_outer = inverse_trigonometric_outer(node);
301 if (!d_outer) d_outer = hyperbolic_outer(node);
302 if (!d_outer) d_outer = elementary_outer(node);
303 if (!d_outer) d_outer = special_outer(node);
304 if (!d_outer) unsupported("FunctionNode");
305 result = SymbolicFactory::create_multiply({std::move(d_outer), std::move(d_arg)});
306}
