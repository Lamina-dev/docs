9static std::optional<NumericCubicDepression> depress_numeric_cubic(
10 double a, double b, double c, double d) {
11 double p_val = 0.0;
12 double q_val = 0.0;
13 double shift_val = 0.0;
14 bool numeric_depression = false;
15 const double max_coefficient = std::max({
16 std::abs(a), std::abs(b), std::abs(c), std::abs(d)});
17 /**
18 * @brief 统一按二进制幂缩放，将最大系数绝对值置于 [1,2)。
19 * 缩放后处于正规范围的数值保留二进制有效位。
20 * @see David Goldberg, "What Every Computer Scientist Should Know About
21 * Floating-Point Arithmetic" (1991)，定理 7 的证明。
22 * https://docs.oracle.com/cd/E19957-01/806-3568/ncg_goldberg.html
23 * @see S. Ghaderpanah and S. Klasa, "Polynomial Scaling" (1990),
24 * SIAM Journal on Numerical Analysis 27(1), 117-135（缩放背景）。
25 * https://doi.org/10.1137/0727007
26 */
27 const int coefficient_exponent = std::ilogb(max_coefficient);
28 const double an = std::scalbn(a, -coefficient_exponent);
29 const double bn = std::scalbn(b, -coefficient_exponent);
30 const double cn = std::scalbn(c, -coefficient_exponent);
31 const double dn = std::scalbn(d, -coefficient_exponent);
32 if (an == 0.0) return std::nullopt;
33 const double an_squared = an * an;
34 const double an_cubed = an_squared * an;
35 const double bn_squared = bn * bn;
36 const double p_left = 3.0 * an * cn;
37 double p_numerator = std::fma(-bn, bn, p_left);
38 const double p_roundoff =
39 16.0 * std::numeric_limits<double>::epsilon() *
40 (std::abs(p_left) + std::abs(bn_squared));
41 if (std::abs(p_numerator) <= p_roundoff) {
42 p_numerator = 0.0;
43 }
44
45 const double q_first = 2.0 * bn_squared * bn;
46 const double q_second = -9.0 * an * bn * cn;
47 const double q_third = 27.0 * an_squared * dn;
48 double q_numerator = (q_first + q_second) + q_third;
49 const double q_roundoff =
50 32.0 * std::numeric_limits<double>::epsilon() *
51 (std::abs(q_first) + std::abs(q_second) +
52 std::abs(q_third));
53 if (std::abs(q_numerator) <= q_roundoff) {
54 q_numerator = 0.0;
55 }
56
57 const double p_denominator = 3.0 * an_squared;
58 const double q_denominator = 27.0 * an_cubed;
59 if (p_denominator == 0.0 || q_denominator == 0.0) return std::nullopt;
60 p_val = p_numerator / p_denominator;
61 q_val = q_numerator / q_denominator;
62 shift_val = bn / (3.0 * an);
63 numeric_depression =
64 std::isfinite(p_val) && std::isfinite(q_val) &&
65 std::isfinite(shift_val);
66 if (!numeric_depression) return std::nullopt;
67 return NumericCubicDepression{p_val, q_val, shift_val};
68}

83static std::vector<std::shared_ptr<SymbolicExpr>> cubic_one_real_root(
84 double discriminant, double q_half, double root_scale, double shift_val) {
85 const double sqrt_discriminant = std::sqrt(discriminant);
86 const double u = std::cbrt(-q_half + sqrt_discriminant);
87 const double v = std::cbrt(-q_half - sqrt_discriminant);
88 const double real_sum = (u + v) * root_scale;
89 const double x1_value = real_sum - shift_val;
90 const double real_part = -real_sum / 2.0 - shift_val;
91 const double imaginary_part =
92 std::sqrt(3.0) * (u - v) * root_scale / 2.0;
93
94 auto x1 = SymbolicExpr::number(x1_value);
95 auto i_unit = SymbolicExpr::sqrt(num(-1));
96 auto x2 = SymbolicExpr::add(
97 SymbolicExpr::number(real_part),
98 SymbolicExpr::multiply(
99 SymbolicExpr::number(imaginary_part), i_unit))->simplify();
100 auto x3 = sub(
101 SymbolicExpr::number(real_part),
102 SymbolicExpr::multiply(
103 SymbolicExpr::number(imaginary_part), i_unit))->simplify();
104 return {x1, x2, x3};
105}

107static std::vector<std::shared_ptr<SymbolicExpr>> cubic_three_real_roots(
108 double p_normalized, double q_normalized, double root_scale, double shift_val) {
109 std::vector<std::shared_ptr<SymbolicExpr>> roots;
110 const double radius =
111 std::sqrt(-(p_normalized * p_normalized * p_normalized) / 27.0);
112 double cosine_argument = -q_normalized / (2.0 * radius);
113 cosine_argument = std::clamp(cosine_argument, -1.0, 1.0);
114 const double theta = std::acos(cosine_argument);
115 const double amplitude = 2.0 * std::cbrt(radius) * root_scale;
116
117 for (int k = 0; k < 3; ++k) {
118 const double angle =
119 (theta + 2.0 * k * LMMC_CONST_PI) / 3.0;
120 roots.push_back(SymbolicExpr::number(
121 amplitude * std::cos(angle) - shift_val));
122 }
123 return roots;
124}

126std::vector<std::shared_ptr<SymbolicExpr>> numeric_cubic_roots(
127 const NumericCubicDepression& depression) {
128 const auto& [p_val, q_val, shift_val] = depression;
129 const double root_scale = std::max(
130 std::sqrt(std::abs(p_val)),
131 std::cbrt(std::abs(q_val)));
132 if (root_scale == 0.0) {
133 auto root = SymbolicExpr::number(-shift_val);
134 return {root, root, root};
135 }
136 const double p_normalized = (p_val / root_scale) / root_scale;
137 const double q_normalized =
138 ((q_val / root_scale) / root_scale) / root_scale;
139 const double q_half = q_normalized / 2.0;
140 const double p_third = p_normalized / 3.0;
141 const double discriminant_q = q_half * q_half;
142 const double discriminant_p = p_third * p_third * p_third;
143 const double discriminant = discriminant_q + discriminant_p;
144 const double discriminant_tolerance =
145 32.0 * std::numeric_limits<double>::epsilon() *
146 (std::abs(discriminant_q) + std::abs(discriminant_p));
147 if (std::abs(discriminant) <= discriminant_tolerance) {
148 const double cbrt_q_half = std::cbrt(q_half);
149 const double t1 = -2.0 * cbrt_q_half * root_scale;
150 const double t2 = cbrt_q_half * root_scale;
151
152 auto x1 = SymbolicExpr::number(t1 - shift_val);
153 auto x2 = SymbolicExpr::number(t2 - shift_val);
154 return {x1, x2, x2};
155 }
156 if (discriminant > 0.0) {
157 return cubic_one_real_root(discriminant, q_half, root_scale, shift_val);
158 }
159 return cubic_three_real_roots(p_normalized, q_normalized, root_scale, shift_val);
160}
