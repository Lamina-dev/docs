92static Result<Tribool> logical_predicate_truth(const LogicalNode& logical,
93 const SegmentFacts& facts, ComputationContext& context) {
94 auto left = predicate_truth(logical.left(), facts, context);
95 if (!left) { return left; }
96 const auto negate = [](Tribool truth) {
97 return truth == Tribool::True ? Tribool::False :
98 truth == Tribool::False ? Tribool::True : Tribool::Unknown;
99 };
100 if (logical.op() == LogicalNode::Op::Not) { return negate(left.value()); }
101 auto right = predicate_truth(logical.right(), facts, context);
102 if (!right) { return right; }
103 auto a = left.value(), b = right.value();
104 if (logical.op() == LogicalNode::Op::Implies) { a = negate(a); }
105 return binary_logical_truth(logical.op(), a, b);
106}

108static Result<Tribool> relational_predicate_truth(const RelationalNode& relation,
109 const SegmentFacts& facts, ComputationContext& context) {
110 auto difference = sym_sub(detail::expression_from_node(relation.left()),
111 detail::expression_from_node(relation.right()));
112 auto value = facts.value_sign(detail::node(difference), context);
113 if (!value) { return Result<Tribool>::failure(value.error()); }
114 if (!value.value()) { return Tribool::Unknown; }
115 const int s = *value.value();
116 bool truth;
117 switch (relation.op()) {
118 case RelationOp::EQ: { truth = s == 0; break; }
119 case RelationOp::NEQ: { truth = s != 0; break; }
120 case RelationOp::LT: { truth = s < 0; break; }
121 case RelationOp::LEQ: { truth = s <= 0; break; }
122 case RelationOp::GT: { truth = s > 0; break; }
123 case RelationOp::GEQ: { truth = s >= 0; break; }
124 default: { return Tribool::Unknown; }
125 }
126 return truth ? Tribool::True : Tribool::False;
127}

144public:
145 SegmentExpression(const SegmentFacts& facts, ComputationContext& context) : facts_(facts), context_(context) {}
146 std::optional<CasError> error;
147 void visit(const PiecewiseNode& piecewise) override {
148 for (const auto& branch : piecewise.branches()) {
149 auto truth = predicate_truth(branch.condition, facts_, context_);
150 if (!truth) { error = truth.error(); set_result(current()); return; }
151 if (truth.value() == Tribool::Unknown) {
152 error = CasError{CasErrc::Inconclusive, "piecewise branch is undecided", operation};
153 set_result(current());
154 return;
155 }
156 if (truth.value() == Tribool::True) {
157 set_result(rewrite(branch.expression));
158 return;
159 }
160 }
161 if (piecewise.default_expr()) set_result(rewrite(piecewise.default_expr()));
162 else {
163 error = CasError{CasErrc::DomainError, "piecewise function has no value on the segment",
164 "integrate.definite.convergence"};
165 set_result(current());
166 }
167 }
168 void visit(const PowerNode& node) override {
169 bool changed = false;
170 auto base = rewrite_child(node.base(), changed);
171 auto exponent = rewrite_child(node.exponent(), changed);
172 auto rewritten = changed ? detail::make_node<PowerNode>(base, exponent) : current();
173 if (!specialize_absolute(absolute_argument(*rewritten))) set_result(rewritten);
174 }
175 void visit(const FunctionNode& node) override {
176 bool changed = false;
177 auto arguments = rewrite_children(node.arguments(), changed);
178 auto rewritten = changed ? detail::make_node<FunctionNode>(node.type(), std::move(arguments)) : current();
179 if (!specialize_absolute(absolute_argument(*rewritten))) set_result(rewritten);
180 }
181private:
182 bool specialize_absolute(const Node& argument) {
183 if (!argument) return false;
184 auto real = detail::query_real_value(argument, facts_, context_);
185 if (!real) { error = real.error(); set_result(current()); return true; }
186 auto value = facts_.value_sign(argument, context_);
187 if (!value) { error = value.error(); set_result(current()); return true; }
188 if (!value.value()) {
189 auto nonnegative = detail::query_nonnegative_value(argument, facts_, context_);
190 if (!nonnegative) { error = nonnegative.error(); set_result(current()); return true; }
191 if (nonnegative.value() == Tribool::True) value.value() = 1;
192 else if (nonnegative.value() == Tribool::False) value.value() = -1;

195 set_result(*value.value() < 0

197 detail::make_expression_ptr(argument))) : argument);

224static Result<void> collect_predicate_roots(const Node& condition, const std::string& var,
225 std::vector<Algebraic>& roots, ComputationContext& context) {
226 auto step = context.consume_steps(1, operation);
227 if (!step) { return step; }
228 if (const auto* logical = dynamic_cast<const LogicalNode*>(condition.get())) {
229 auto left = collect_predicate_roots(logical->left(), var, roots, context);
230 if (!left) { return left; }
231 if (logical->right()) { return collect_predicate_roots(logical->right(), var, roots, context); }
232 return Result<void>::success();
233 }
234 const auto* relation = dynamic_cast<const RelationalNode*>(condition.get());
235 if (!relation) { return undecided<void>(); }
236 auto difference = sym_sub(detail::expression_from_node(relation->left()),
237 detail::expression_from_node(relation->right()));
238 auto rational = rational_checked(detail::node(difference), var, context);
239 if (!rational) { return Result<void>::failure(rational.error()); }
240 if (!rational.value()) { return undecided<void>(); }
241 for (const auto* polynomial : {&rational.value()->numerator, &rational.value()->denominator}) {
242 if (polynomial->degree() <= 0) { continue; }
243 auto isolated = roots_checked(*polynomial, context);
244 if (!isolated) { return Result<void>::failure(isolated.error()); }
245 roots.insert(roots.end(), isolated.value().begin(), isolated.value().end());
246 }
247 return Result<void>::success();

328static Result<std::vector<Algebraic>> ordered_partition_roots(
329 std::vector<Algebraic>& roots, const Boundary& a, const Boundary& b,
330 ComputationContext& context) {
331 std::vector<Algebraic> ordered;
332 for (auto& root : roots) {
333 auto left = compare_boundaries(a, Boundary{0, root}, context);
334 auto right = compare_boundaries(Boundary{0, root}, b, context);
335 if (!left) { return Result<std::vector<Algebraic>>::failure(left.error()); }
336 if (!right) { return Result<std::vector<Algebraic>>::failure(right.error()); }
337 if (left.value() >= 0 || right.value() >= 0) { continue; }
338 auto position = ordered.begin();
339 bool duplicate = false;
340 for (; position != ordered.end(); ++position) {
341 auto comparison = compare_boundaries(Boundary{0, root}, Boundary{0, *position}, context);
342 if (!comparison) { return Result<std::vector<Algebraic>>::failure(comparison.error()); }
343 if (comparison.value() == 0) { duplicate = true; break; }
344 if (comparison.value() < 0) { break; }
345 }
346 if (!duplicate) { ordered.insert(position, std::move(root)); }
347 }
348 return ordered;
349}

351Result<std::vector<SymbolicExpr>> definite_partition_checked(const SymbolicExpr& original,
352 const std::string& var, const SymbolicExpr& lower, const SymbolicExpr& upper,
353 ComputationContext& context) {
354 auto assumptions = real_assumptions(var, context);
355 if (!assumptions) { return Result<std::vector<SymbolicExpr>>::failure(assumptions.error()); }
356 detail::AssumptionFacts facts(assumptions.value());
357 std::vector<std::shared_ptr<SymbolicExpr>> conditions;
358 auto captured = original_conditions_checked(detail::node(original), facts, conditions, context);
359 if (!captured) { return Result<std::vector<SymbolicExpr>>::failure(captured.error()); }
360 if (conditions.empty()) { return std::vector<SymbolicExpr>{lower, upper}; }
361 auto a = boundary_checked(lower, context);
362 auto b = boundary_checked(upper, context);
363 if (!a) { return Result<std::vector<SymbolicExpr>>::failure(a.error()); }
364 if (!b) { return Result<std::vector<SymbolicExpr>>::failure(b.error()); }
365 std::vector<Algebraic> roots;
366 for (const auto& condition : conditions) {
367 auto collected = collect_predicate_roots(detail::node(condition), var, roots, context);
368 if (!collected) { return Result<std::vector<SymbolicExpr>>::failure(collected.error()); }
369 }
370 auto ordered = ordered_partition_roots(roots, a.value(), b.value(), context);
371 if (!ordered) { return Result<std::vector<SymbolicExpr>>::failure(ordered.error()); }
372 std::vector<SymbolicExpr> result{lower};
373 for (const auto& root : ordered.value()) {
374 if (root.is_rational()) { result.push_back(*SymbolicExpr::number(root.lower)); }
375 else {
376 auto expression = make_rootof_checked(poly_to_symbolic(root.polynomial), var, root.root_index, context);
377 if (!expression) { return Result<std::vector<SymbolicExpr>>::failure(expression.error()); }
378 result.push_back(*expression.value());
379 }
380 }
381 result.push_back(upper);
382 return result;
383}

385static Result<Tribool> defined_on_interval(const Node& node, const FactsQuery& facts,
386 ComputationContext& context) {
387 auto defined = detail::query_definedness(node, facts, Domain::Real, context);
388 if (!defined || defined.value() != Tribool::Unknown) { return defined; }
389 const auto* interval = dynamic_cast<const SegmentFacts*>(&facts);
390 if (!interval) { return defined; }
391 auto conditions = detail::domain_constraints(node, facts, Domain::Real, context);
392 if (!conditions) { return Result<Tribool>::failure(conditions.error()); }
393 if (!conditions.value()) { return Tribool::Unknown; }
394 bool unknown = false;
395 for (const auto& condition : *conditions.value()) {
396 auto truth = predicate_truth(detail::node(condition), *interval, context);
397 if (!truth) { return truth; }
398 if (truth.value() == Tribool::False) { return Tribool::False; }
399 unknown |= truth.value() == Tribool::Unknown;
400 }
401 return unknown ? Tribool::Unknown : Tribool::True;
402}

436static Result<Tribool> structurally_continuous(const Node& node, const FactsQuery& facts,
437 ComputationContext& context) {
438 auto step = context.consume_steps(1, operation);
439 if (!step) { return Result<Tribool>::failure(step.error()); }
440 auto defined = defined_on_interval(node, facts, context);
441 if (!defined || defined.value() != Tribool::True) { return defined; }
442 if (dynamic_cast<const NumberNode*>(node.get()) || dynamic_cast<const VariableNode*>(node.get()) ||
443 dynamic_cast<const RootOfNode*>(node.get())) { return Tribool::True; }
444 const auto* sum = dynamic_cast<const AddNode*>(node.get());
445 const auto* product = dynamic_cast<const MultiplyNode*>(node.get());
446 const auto* function = dynamic_cast<const FunctionNode*>(node.get());
447 const auto* power = dynamic_cast<const PowerNode*>(node.get());
448 if (power) {
449 auto base = structurally_continuous(power->base(), facts, context);
450 if (!base || base.value() != Tribool::True) { return base; }
451 return structurally_continuous(power->exponent(), facts, context);
452 }
453 if (sum) { return continuous_children(sum->operands(), facts, context); }
454 if (product) { return continuous_children(product->operands(), facts, context); }
455 if (function) {
456 if (!continuous_function_family(function->type())) { return Tribool::Unknown; }
457 return continuous_children(function->arguments(), facts, context);
458 }
459 return Tribool::Unknown;
460}

525 auto result = limit_checked(detail::make_expression_ptr(primitive), var, detail::make_expression_ptr(point), direction, context);
526 if (!result) return Result<SymbolicExpr>::failure(result.error());
527 const auto& outcome = result.value().value;
528 if (const auto* finite = std::get_if<FiniteLimit>(&outcome)) return *finite->value;
529 if (std::holds_alternative<PositiveInfinityLimit>(outcome)) return *SymbolicExpr::infinity(1);
530 if (std::holds_alternative<NegativeInfinityLimit>(outcome)) return *SymbolicExpr::infinity(-1);
531 return divergent<SymbolicExpr>();
532}

534static ExpressionResult certify_primitive_derivative(SymbolicExpr& candidate,
535 const std::string& var, const SymbolicExpr& a, const SymbolicExpr& b,
536 const Result<Boundary>& lower, const Result<Boundary>& upper,
537 ComputationContext& context) {
538 if (!lower && !is_undecided(lower.error())) {
539 return ExpressionResult::failure(lower.error());
540 }
541 if (!upper && !is_undecided(upper.error())) {
542 return ExpressionResult::failure(upper.error());
543 }
544 if (lower && upper) {
545 auto assumptions = real_assumptions(var, context);
546 if (!assumptions) { return ExpressionResult::failure(assumptions.error()); }
547 SegmentFacts facts(assumptions.value(), var, lower.value(), upper.value());
548 RealPrimitiveCandidate rewrite(facts, context);
549 candidate = detail::expression_from_node(rewrite.rewrite(detail::node(candidate)));
550 if (rewrite.error) { return ExpressionResult::failure(*rewrite.error); }
551 }
552 auto primitive_domain = certify_continuity_checked(candidate, var, a, b, context);
553 if (!primitive_domain) { return ExpressionResult::failure(primitive_domain.error()); }
554 if (primitive_domain.value() != Tribool::True) { return undecided<std::shared_ptr<SymbolicExpr>>(); }
555 auto derivative = candidate.differentiate(var);
556 if (!derivative) { return ExpressionResult::failure(CasErrc::InternalInvariant,
557 "primitive differentiation returned no expression", operation); }
558 auto derivative_domain = certify_continuity_checked(*derivative, var, a, b, context);
559 if (!derivative_domain) { return ExpressionResult::failure(derivative_domain.error()); }
560 if (derivative_domain.value() != Tribool::True) { return undecided<std::shared_ptr<SymbolicExpr>>(); }
561 return derivative;
562}

564static Result<void> prove_rational_derivative(const SymbolicExpr& effective,
565 const std::shared_ptr<SymbolicExpr>& derivative, const std::string& var,
566 ComputationContext& context) {
567 auto f = rational_checked(detail::node(effective), var, context);
568 auto d = rational_checked(detail::node(derivative), var, context);
569 if (!f) { return Result<void>::failure(f.error()); }
570 if (!d) { return Result<void>::failure(d.error()); }
571 if (!f.value() || !d.value()) { return undecided<void>(); }
572 auto left = product_checked(f.value()->numerator, d.value()->denominator, context);
573 auto right = product_checked(d.value()->numerator, f.value()->denominator, context);
574 if (!left) { return Result<void>::failure(left.error()); }
575 if (!right) { return Result<void>::failure(right.error()); }
576 auto proof = check_zero_residual(poly_to_symbolic(left.value() - right.value()), context);
577 if (!proof) { return Result<void>::failure(proof.error()); }
578 if (!std::holds_alternative<ProvedZeroResidual>(proof.value())) { return undecided<void>(); }
579 return Result<void>::success();
580}

582static Result<void> prove_segment_derivative(SymbolicExpr& effective,
583 std::shared_ptr<SymbolicExpr>& derivative, const std::string& var,
584 const Result<Boundary>& lower, const Result<Boundary>& upper,
585 ComputationContext& context) {
586 if (lower && upper) {
587 auto assumptions = real_assumptions(var, context);
588 if (!assumptions) { return Result<void>::failure(assumptions.error()); }
589 SegmentFacts facts(assumptions.value(), var, lower.value(), upper.value());
590 NormalizationVisitor normalize_derivative(context, facts);
591 detail::node(derivative)->accept(normalize_derivative);
592 if (facts.error) { return Result<void>::failure(*facts.error); }
593 derivative = detail::make_expression_ptr(normalize_derivative.get_result());
594 NormalizationVisitor normalize_original(context, facts);
595 detail::node(effective)->accept(normalize_original);
596 if (facts.error) { return Result<void>::failure(*facts.error); }
597 effective = detail::expression_from_node(normalize_original.get_result());
598 }
599 auto residual = sym_sub(*derivative, effective);
600 if (lower && upper) {
601 auto assumptions = real_assumptions(var, context);
602 if (!assumptions) { return Result<void>::failure(assumptions.error()); }
603 SegmentFacts facts(assumptions.value(), var, lower.value(), upper.value());
604 NormalizationVisitor normalize_residual(context, facts);
605 detail::node(residual)->accept(normalize_residual);
606 if (facts.error) { return Result<void>::failure(*facts.error); }
607 residual = detail::make_expression_ptr(normalize_residual.get_result());
608 }
609 auto proof = check_zero_residual(residual, context);
610 if (!proof) { return Result<void>::failure(proof.error()); }
611 if (!std::holds_alternative<ProvedZeroResidual>(proof.value())) {
612 return prove_rational_derivative(effective, derivative, var, context);
613 }
614 return Result<void>::success();
615}

617static Result<SymbolicExpr> evaluate_segment_endpoints(const SymbolicExpr& candidate,
618 const std::string& var, const SymbolicExpr& a, const SymbolicExpr& b,
619 bool concrete_endpoints, ComputationContext& context) {
620 if (!concrete_endpoints) {
621 auto high = candidate.substitute(var, detail::make_expression_ptr(b));
622 auto low = candidate.substitute(var, detail::make_expression_ptr(a));
623 if (!high || !low) { return Result<SymbolicExpr>::failure(CasErrc::InternalInvariant,
624 "primitive substitution returned no expression", operation); }
625 return *sym_sub(*high, *low)->simplify();
626 }
627 auto high = boundary_value_checked(candidate, var, b, LimitDirection::FromBelow, context);
628 auto low = boundary_value_checked(candidate, var, a, LimitDirection::FromAbove, context);
629 if (!high) { return high; }
630 if (!low) { return low; }
631 const int hs = infinity_sign(detail::node(high.value()));
632 const int ls = -infinity_sign(detail::node(low.value()));
633 if (hs && ls && hs != ls) { return divergent<SymbolicExpr>(); }
634 if (hs || ls) { return *SymbolicExpr::infinity(hs ? hs : ls); }
635 return *sym_sub(high.value(), low.value())->simplify();
636}

638Result<SymbolicExpr> definite_segment_checked(const SymbolicExpr& original,
639 const SymbolicExpr& primitive, const std::string& var,
640 const SymbolicExpr& a, const SymbolicExpr& b, ComputationContext& context) {
641 auto continuous = certify_continuity_checked(original, var, a, b, context);
642 if (!continuous) { return Result<SymbolicExpr>::failure(continuous.error()); }
643 if (continuous.value() == Tribool::False) { return divergent<SymbolicExpr>(); }
644 if (continuous.value() != Tribool::True) { return undecided<SymbolicExpr>(); }
645 auto effective = segment_expression_checked(original, var, a, b, context);
646 if (!effective) { return effective; }
647 SymbolicExpr candidate = primitive;
648 auto lower = boundary_checked(a, context);
649 auto upper = boundary_checked(b, context);
650 auto derivative = certify_primitive_derivative(candidate, var, a, b, lower, upper, context);
651 if (!derivative) { return Result<SymbolicExpr>::failure(derivative.error()); }
652 auto proof = prove_segment_derivative(effective.value(), derivative.value(), var, lower, upper, context);
653 if (!proof) { return Result<SymbolicExpr>::failure(proof.error()); }
654 return evaluate_segment_endpoints(candidate, var, a, b, lower && upper, context);
655}
