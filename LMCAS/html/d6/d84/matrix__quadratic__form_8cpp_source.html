13bool valid_variables(const std::vector<std::string>& variables) {
14 if (variables.empty()) return false;
15 for (std::size_t index = 0; index < variables.size(); ++index) {
16 if (variables[index].empty()) return false;
17 for (std::size_t previous = 0; previous < index; ++previous) {
18 if (variables[index] == variables[previous]) return false;
19 }
20 }
21 return true;
22}

37std::shared_ptr<SymbolicExpr> reconstruct_quadratic_form(
38 const std::vector<std::vector<std::shared_ptr<SymbolicExpr>>>& matrix,
39 const std::vector<std::shared_ptr<SymbolicExpr>>& variables) {
40 auto reconstructed = SymbolicExpr::number(0);
41 for (std::size_t row = 0; row < matrix.size(); ++row) {
42 for (std::size_t column = 0; column < matrix.size(); ++column) {
43 if (detail::node(matrix[row][column])->is_zero()) continue;
44 auto term = SymbolicExpr::multiply(
45 matrix[row][column],
46 SymbolicExpr::multiply(variables[row], variables[column]));
47 reconstructed = SymbolicExpr::add(reconstructed, term);
48 }
49 }
50 return reconstructed;
51}

66std::optional<SymbolicMatrix> quadratic_coefficients(
67 const std::shared_ptr<SymbolicExpr>& expanded,
68 const std::vector<std::string>& variables, ComputationContext& context) {
69
70 detail::RewriteBudget transform_budget(context, (context).limits().max_recursion_depth,
71 (context).limits().max_ast_nodes, "transform");
72 const auto half = SymbolicExpr::number(Rational(1, 2));
73 SymbolicMatrix matrix(
74 variables.size(),
75 std::vector<std::shared_ptr<SymbolicExpr>>(variables.size()));
76 for (std::size_t row = 0; row < variables.size(); ++row) {
77 auto first_derivative = detail::make_expression_ptr(detail::differentiate_node(detail::node(expanded), variables[row], transform_budget));
78 if (!first_derivative) {
79 return std::nullopt;
80 }
81 for (std::size_t column = 0; column < variables.size(); ++column) {
82 auto second_derivative =
83 detail::make_expression_ptr(detail::differentiate_node(detail::node(first_derivative), variables[column], transform_budget));
84 if (!second_derivative) {
85 return std::nullopt;
86 }
87 auto coefficient =
88 detail::make_expression_ptr(detail::normalize_node(detail::node(SymbolicExpr::multiply(half, second_derivative)), transform_budget));
89 if (!independent_of_variables(coefficient, variables)) {
90 return std::nullopt;
91 }
92 matrix[row][column] = std::move(coefficient);
93 }
94 }
95 return matrix;

99 const std::shared_ptr<SymbolicExpr>& expanded,
100 const SymbolicMatrix& matrix,
101 const std::vector<std::shared_ptr<SymbolicExpr>>& variables, ComputationContext& context) {
102
103 detail::RewriteBudget transform_budget(context, (context).limits().max_recursion_depth,
104 (context).limits().max_ast_nodes, "transform");
105 auto reconstructed = reconstruct_quadratic_form(matrix, variables);
106 auto residual = SymbolicExpr::add(
107 expanded,
108 SymbolicExpr::multiply(

110 residual = detail::make_expression_ptr(detail::expand_node(detail::node(residual), transform_budget));
111 if (residual) {

113 }
114 return residual && detail::node(residual) &&
