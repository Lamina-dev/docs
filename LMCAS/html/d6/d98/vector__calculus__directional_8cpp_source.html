10static std::shared_ptr<SymbolicExpr> vector_calculus_magnitude_squared(
11 const VectorField& v,
12 detail::RewriteBudget& transform_budget)
13{
14 std::shared_ptr<SymbolicExpr> sum = nullptr;
15 for (const auto& comp : v) {
16 if (!comp) {
17 continue;
18 }
19 auto sq = SymbolicExpr::power(comp, SymbolicExpr::number(2));
20 sq = detail::make_expression_ptr(detail::normalize_node(detail::node(sq), transform_budget));
21 if (!sum) {
22 sum = sq;
23 } else {
24 sum = SymbolicExpr::add(sum, sq);
25 sum = detail::make_expression_ptr(detail::normalize_node(detail::node(sum), transform_budget));
26 }
27 }
28 if (!sum) {
29 return SymbolicExpr::number(0);
30 }
31 return detail::make_expression_ptr(detail::normalize_node(detail::node(sum), transform_budget));
32}

34static std::shared_ptr<SymbolicExpr> vector_calculus_single_dir_deriv(
35 const std::shared_ptr<SymbolicExpr>& f,
36 const std::vector<std::string>& vars,
37 const VectorField& unit_dir,
38 detail::RewriteBudget& transform_budget)
39{
40 std::shared_ptr<SymbolicExpr> sum = nullptr;
41 for (size_t i = 0; i < vars.size(); ++i) {
42 auto partial = detail::make_expression_ptr(detail::differentiate_node(detail::node(f), vars[i], transform_budget));
43 if (!partial) {
44 continue;
45 }
46 partial = detail::make_expression_ptr(detail::normalize_node(detail::node(partial), transform_budget));
47 auto term = SymbolicExpr::multiply(partial, unit_dir[i]);
48 term = detail::make_expression_ptr(detail::normalize_node(detail::node(term), transform_budget));
49 if (!sum) {
50 sum = term;
51 } else {
52 sum = SymbolicExpr::add(sum, term);
53 sum = detail::make_expression_ptr(detail::normalize_node(detail::node(sum), transform_budget));
54 }
55 }
56 if (!sum) {
57 return SymbolicExpr::number(0);
58 }
59 return detail::make_expression_ptr(detail::normalize_node(detail::node(sum), transform_budget));
60}

63 const std::shared_ptr<SymbolicExpr>& f,
64 const std::vector<std::string>& vars,
65 const VectorField& direction,
66 int order,
67 ComputationContext& context)
68{
69 detail::RewriteBudget transform_budget(context, context.limits().max_recursion_depth,
70 context.limits().max_ast_nodes, "directional_derivative");
71 const std::string operation = "directional_derivative";
72 auto valid = vector_calculus_validate_expr_vars(f, vars, context, operation);
73 if (!valid) {

75 }
76 auto dir_valid = vector_calculus_validate_field_vars(direction, vars, context, operation);
77 if (!dir_valid) {

82 "directional derivative order must be positive",

91 operation);

95 "direction vector cannot be zero",
96 operation);
97 }

99 operation);
100 if (!step) {
101 return VectorCalculusExprResult::failure(step.error());
102 }
103 return vector_calculus_wrap_expr(
104 directional_derivative_impl(f, vars, direction, order, transform_budget), operation);
105 } catch (const CasError& error) {
106 return VectorCalculusExprResult::failure(error);
107 } catch (const std::bad_alloc&) {
108 return VectorCalculusExprResult::failure(CasErrc::ResourceLimit,
109 "vector calculus allocation failed",
110 operation);
111 } catch (const std::length_error& error) {
112 return VectorCalculusExprResult::failure(CasErrc::ResourceLimit, error.what(), operation);
113 } catch (const std::exception& e) {

128std::shared_ptr<SymbolicExpr> vector_calculus_detail::directional_derivative_impl(
129 const std::shared_ptr<SymbolicExpr>& f, const std::vector<std::string>& vars,
130 const VectorField& direction, int order,
131 detail::RewriteBudget& transform_budget)
132{
133 if (!f) {
134 throw std::invalid_argument("directional_derivative: f must not be null");
135 }
136 if (direction.size() != vars.size()) {
137 throw std::invalid_argument(
138 "directional_derivative: direction and vars must have the same dimension");
139 }
140 if (order < 1) {
141 throw std::invalid_argument(
142 "directional_derivative: order must be >= 1");
143 }
144
145 auto mag_sq = vector_calculus_magnitude_squared(direction, transform_budget);
146 if (!mag_sq || mag_sq->is_zero()) {
147 return nullptr;
148 }
149
150 auto magnitude = SymbolicExpr::sqrt(mag_sq);
151 magnitude = detail::make_expression_ptr(detail::normalize_node(detail::node(magnitude), transform_budget));
152 VectorField unit_dir;
153 unit_dir.reserve(direction.size());
154 for (const auto& comp : direction) {
155 if (!comp || comp->is_zero()) {
156 unit_dir.push_back(SymbolicExpr::number(0));
157 } else {
158 auto u_comp = SymbolicExpr::multiply(
159 comp,
160 SymbolicExpr::power(magnitude, SymbolicExpr::number(-1)));
161 u_comp = detail::make_expression_ptr(detail::normalize_node(detail::node(u_comp), transform_budget));
162 unit_dir.push_back(u_comp);
163 }
164 }
165
166 auto result = f;
167 for (int k = 0; k < order; ++k) {
168 result = vector_calculus_single_dir_deriv(result, vars, unit_dir, transform_budget);
169 if (!result) {
170 return SymbolicExpr::number(0);
171 }
172 }
173
174 return result;
175}
