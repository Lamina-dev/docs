11 const VectorField& v)
12{

14 for (const auto& comp : v) {
15 if (!comp) {
16 continue;
17 }
18 auto sq = SymbolicExpr::power(comp, SymbolicExpr::number(2));

124std::shared_ptr<SymbolicExpr> directional_derivative(
125 const std::shared_ptr<SymbolicExpr>& f, const std::vector<std::string>& vars,
126 const VectorField& direction, int order)
127{
128 if (!f) {
129 throw std::invalid_argument("directional_derivative: f must not be null");
130 }
131 if (direction.size() != vars.size()) {
132 throw std::invalid_argument(
133 "directional_derivative: direction and vars must have the same dimension");
134 }
135 if (order < 1) {
136 throw std::invalid_argument(
137 "directional_derivative: order must be >= 1");
138 }
139
140 auto mag_sq = vector_calculus_magnitude_squared(direction);
141 if (!mag_sq || mag_sq->is_zero()) {
142 return nullptr;
143 }
144
145 auto magnitude = SymbolicExpr::sqrt(mag_sq);
146 magnitude = magnitude->simplify();
147 VectorField unit_dir;
148 unit_dir.reserve(direction.size());
149 for (const auto& comp : direction) {
150 if (!comp || comp->is_zero()) {
151 unit_dir.push_back(SymbolicExpr::number(0));
152 } else {
153 auto u_comp = SymbolicExpr::multiply(
154 comp,
155 SymbolicExpr::power(magnitude, SymbolicExpr::number(-1)));
156 u_comp = u_comp->simplify();
157 unit_dir.push_back(u_comp);
158 }
159 }
160
161 auto result = f;
162 for (int k = 0; k < order; ++k) {
163 result = vector_calculus_single_dir_deriv(result, vars, unit_dir);
164 if (!result) {
165 return SymbolicExpr::number(0);
166 }
167 }
168
169 return result;
170}
