7/**
8 * @brief Recursively replace |var| with var in an AST when the variable is known Positive.
9 *
10 * Traverses the expression tree and replaces any FunctionNode(Abs, [arg]) where
11 * arg is exactly the integration variable with just the variable itself.
12 * This is valid when the AssumptionContext confirms the variable is Positive.
13 */
14static std::shared_ptr<const SymbolicNode> simplify_abs_positive(
15 const std::shared_ptr<const SymbolicNode>& node, const std::string& var) {
16 if (!node) return node;
17
18 if (auto fn = std::dynamic_pointer_cast<const FunctionNode>(node)) {
19 if (fn->type() == FunctionNode::FuncType::Abs && fn->arguments().size() == 1) {
20 // Check if the argument is exactly the integration variable
21 if (auto vn = std::dynamic_pointer_cast<const VariableNode>(fn->arguments()[0])) {
22 if (vn->name() == var) {
23 // |x| → x when x is Positive
24 return fn->arguments()[0];
25 }
26 }
27 }
28 // Recurse into function arguments
29 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
30 bool changed = false;
31 for (auto& op : fn->arguments()) {
32 auto new_op = simplify_abs_positive(op, var);
33 if (new_op != op) changed = true;
34 new_ops.push_back(new_op);
35 }
36 if (changed) {
37 return lamina::detail::make_node<FunctionNode>(fn->type(), new_ops);
38 }
39 return node;
40 }
41
42 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
43 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
44 bool changed = false;
45 for (auto& op : add->operands()) {
46 auto new_op = simplify_abs_positive(op, var);
47 if (new_op != op) changed = true;
48 new_ops.push_back(new_op);
49 }
50 if (changed) return lamina::detail::make_node<AddNode>(new_ops);
51 return node;
52 }
53
54 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
55 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
56 bool changed = false;
57 for (auto& op : mul->operands()) {
58 auto new_op = simplify_abs_positive(op, var);
59 if (new_op != op) changed = true;
60 new_ops.push_back(new_op);
61 }
62 if (changed) return lamina::detail::make_node<MultiplyNode>(new_ops);
63 return node;
64 }
65
66 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
67 auto new_base = simplify_abs_positive(pow->base(), var);
68 auto new_exp = simplify_abs_positive(pow->exponent(), var);
69 if (new_base != pow->base() || new_exp != pow->exponent()) {
70 return lamina::detail::make_node<PowerNode>(new_base, new_exp);
71 }
72 return node;
73 }
74
75 // Leaf nodes (NumberNode, VariableNode, etc.) — no change
76 return node;
77}

79/**
80 * @brief Apply assumption-aware simplifications to an integrand.
81 *
82 * When the AssumptionContext indicates the integration variable is Positive,
83 * replaces |var| with var throughout the integrand. Returns the original
84 * expression unchanged if no simplifications apply.
85 */
86static SymbolicExpr apply_assumption_simplifications(
87 const SymbolicExpr& expr, const std::string& var,
88 const AssumptionContext* ctx) {
89 if (!ctx) return expr;
90
91 // Check if the integration variable is known Positive
92 SymbolicExpr var_expr = *SymbolicExpr::variable(var);
93 Tribool var_positive = detail::propagate_result(ctx->is_positive(var_expr));
94
95 if (var_positive == Tribool::True) {
96 auto new_root = simplify_abs_positive(lamina::detail::node(expr), var);
97 if (new_root != lamina::detail::node(expr)) {
98 return lamina::detail::expression_from_node(new_root);
99 }
100 }
101
102 return expr;
103}

105IntegrationStrategyResult IntegrationStrategy::try_integrate(
106 const SymbolicExpr& expr,
107 const std::string& var,
108 Integrator& integrator,
109 ComputationContext& computation,
110 int depth) {
111 const std::string operation = "integrate.strategy." + name();
112 auto access = computation.consume_steps(0, operation);
113 if (!access) return IntegrationStrategyResult::failure(access.error());
114 try {
115 auto expression = try_integrate_raw(
116 expr, var, integrator, computation, depth);
117 auto final_access = computation.consume_steps(0, operation);
118 if (!final_access) {
119 return IntegrationStrategyResult::failure(final_access.error());
120 }
121 if (!expression) {
122 return IntegrationStrategyResult::success(
123 IntegrationNotApplicable{});
124 }
125 return IntegrationStrategyResult::success(IntegrationCandidate{
126 std::move(expression), name()});
127 } catch (const detail::ResultPropagation& propagation) {
128 return IntegrationStrategyResult::failure(propagation.error());
129 } catch (const std::bad_alloc&) {
130 return IntegrationStrategyResult::failure(
131 CasErrc::ResourceLimit,
132 "integration strategy allocation failed", operation);
133 } catch (const std::exception& error) {
134 return IntegrationStrategyResult::failure(
135 CasErrc::InternalInvariant, error.what(), operation);

196void Integrator::resolve_cycle(std::shared_ptr<SymbolicExpr>& result, size_t cycle_idx) {
197 std::string cycle_var = "INT_CYCLE_" + std::to_string(cycle_idx);
198 if (depends_on_integration_variable(*result, cycle_var)) {
199 auto B = result->differentiate(cycle_var)->simplify();
200 auto A = result->substitute(cycle_var, SymbolicExpr::number(0))->simplify();
201 auto one_minus_B = sym_sub(*SymbolicExpr::number(1), *B)->simplify();
202 if (!one_minus_B->is_zero()) {
203 result = SymbolicExpr::divide(A, one_minus_B);
204 }

208Result<std::shared_ptr<SymbolicExpr>> Integrator::apply_linearity(
209 const SymbolicExpr& expr, const std::string& var,
210 ComputationContext& context, int depth) {
211
212 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(expr))) {
213 std::vector<std::shared_ptr<const SymbolicNode>> constants;
214 std::vector<std::shared_ptr<const SymbolicNode>> dependents;
215
216 for (auto& op : mul->operands()) {
217 auto term = lamina::detail::expression_from_node(op);
218 if (!depends_on_integration_variable(term, var)) {
219 constants.push_back(op);
220 } else {
221 dependents.push_back(op);
222 }
223 }
224
225 if (!constants.empty() && dependents.size() < mul->operands().size()) {
226 auto const_part = (constants.size() == 1)
227 ? lamina::detail::expression_from_node(constants[0])
228 : lamina::detail::expression_from_node(
229 lamina::detail::make_node<MultiplyNode>(constants));
230 SymbolicExpr dep_part = (dependents.empty()) ?
231 *SymbolicExpr::number(1) :
232 ((dependents.size() == 1)
233 ? lamina::detail::expression_from_node(dependents[0])
234 : lamina::detail::expression_from_node(
235 lamina::detail::make_node<MultiplyNode>(dependents)));
236 if (dependents.size() > 1) {
237 std::vector<std::shared_ptr<const SymbolicNode>> exponents;
238 exponents.reserve(dependents.size());
239 bool all_exponential = true;
240 for (const auto& dependent : dependents) {
241 auto function =
242 std::dynamic_pointer_cast<const FunctionNode>(dependent);
243 if (!function ||
244 function->type() != FunctionNode::FuncType::Exp ||
245 function->arguments().size() != 1) {
246 all_exponential = false;
247 break;
248 }
249 exponents.push_back(function->arguments()[0]);
250 }
251 if (all_exponential) {
252 dep_part = lamina::detail::expression_from_node(
253 lamina::detail::make_node<FunctionNode>(
254 FunctionNode::FuncType::Exp,
255 std::vector<std::shared_ptr<const SymbolicNode>>{
256 SymbolicFactory::create_add(
257 std::move(exponents))}));
258 }
259 }
260
261 auto int_part = integrate_recursive(dep_part, var, context, depth + 1);
262 if (!int_part) {
263 return Result<std::shared_ptr<SymbolicExpr>>::failure(
264 int_part.error());
265 }
266 return Result<std::shared_ptr<SymbolicExpr>>::success(
267 SymbolicExpr::multiply(
268 lamina::detail::make_expression_ptr(const_part),
269 int_part.value()));
270 }
271 }
272
273 if (auto add = std::dynamic_pointer_cast<const AddNode>(
274 lamina::detail::node(expr))) {
275 std::vector<std::shared_ptr<const SymbolicNode>> results;
276 for (auto& op : add->operands()) {
277 auto term = lamina::detail::expression_from_node(op);
278 auto int_term = integrate_recursive(
279 term, var, context, depth + 1);
280 if (!int_term) {
281 return Result<std::shared_ptr<SymbolicExpr>>::failure(
282 int_term.error());
283 }
284 results.push_back(lamina::detail::node(int_term.value()));
285 }
286 return Result<std::shared_ptr<SymbolicExpr>>::success(
287 lamina::detail::make_expression_ptr(
288 lamina::detail::make_node<AddNode>(results)));
289 }
290 return Result<std::shared_ptr<SymbolicExpr>>::success(nullptr);
291}

293Result<std::shared_ptr<SymbolicExpr>> Integrator::dispatch_strategies(
294 const SymbolicExpr& expr, const std::string& var,
295 ComputationContext& context, int depth) {
296 for (auto& strategy : strategies_) {
297 auto budget = context.consume_steps(1, "integrate.strategy");
298 if (!budget) {
299 return Result<std::shared_ptr<SymbolicExpr>>::failure(
300 budget.error());
301 }
302 auto attempt = strategy->try_integrate(
303 expr, var, *this, context, depth);
304 if (!attempt) {
305 return Result<std::shared_ptr<SymbolicExpr>>::failure(
306 attempt.error());
307 }
308 if (auto* candidate =
309 std::get_if<IntegrationCandidate>(&attempt.value())) {
310 if (!strategy->requires_residual_verification()) {
311 return Result<std::shared_ptr<SymbolicExpr>>::success(
312 candidate->expression);
313 }
314 auto derivative = candidate->expression
315 ? candidate->expression->differentiate(var) : nullptr;
316 if (!derivative) {
317 if (depth > 0) {
318 return Result<std::shared_ptr<SymbolicExpr>>::success(
319 nullptr);
320 }
321 continue;
322 }
323 auto normalized_derivative = derivative->simplify();
324 auto normalized_integrand = make_expr_ptr(expr)->simplify();
325 if (normalized_derivative && normalized_integrand &&
326 lamina::detail::node(normalized_derivative)->equals(
327 *lamina::detail::node(normalized_integrand))) {
328 return Result<std::shared_ptr<SymbolicExpr>>::success(
329 candidate->expression);
330 }
331 auto delta = SymbolicExpr::add(
332 derivative,
333 SymbolicExpr::multiply(
334 SymbolicExpr::number(-1), make_expr_ptr(expr)));
335 delta = delta->cancel()->simplify();
336 if (delta && delta->is_zero()) {
337 return Result<std::shared_ptr<SymbolicExpr>>::success(
338 candidate->expression);
339 }
340 if (depth > 0) {
341 return Result<std::shared_ptr<SymbolicExpr>>::success(nullptr);
342 }
343 }
344 }
345 return Result<std::shared_ptr<SymbolicExpr>>::success(nullptr);
346}

348Result<std::shared_ptr<SymbolicExpr>> Integrator::integrate_recursive(
349 const SymbolicExpr& expr, const std::string& var,
350 ComputationContext& context, int depth) {
351 auto entered = context.enter_recursion("integrate.recursive");
352 if (!entered) {
353 return Result<std::shared_ptr<SymbolicExpr>>::failure(
354 entered.error());
355 }
356 struct RecursionExit {
357 ComputationContext& context;
358 ~RecursionExit() { context.leave_recursion(); }
359 } recursion_exit{context};
360
361 auto linear_result = apply_linearity(expr, var, context, depth);
362 if (!linear_result) {
363 return Result<std::shared_ptr<SymbolicExpr>>::failure(
364 linear_result.error());
365 }
366 if (linear_result.value()) return linear_result;
367
368 auto cycle_result = check_cycle(expr, var);
369 if (cycle_result) {
370 return Result<std::shared_ptr<SymbolicExpr>>::success(
371 std::move(cycle_result));
372 }
373
374 cycle_state_.history.push_back(expr);
375 const size_t my_idx = cycle_state_.history.size() - 1;
376
377 if (expr.is_number() || !depends_on_integration_variable(expr, var)) {
378 cycle_state_.history.pop_back();
379 return Result<std::shared_ptr<SymbolicExpr>>::success(
380 SymbolicExpr::multiply(
381 make_expr_ptr(expr), SymbolicExpr::variable(var)));
382 }
383
384 auto result = dispatch_strategies(expr, var, context, depth);
385 cycle_state_.history.pop_back();
386 if (!result) {
387 return Result<std::shared_ptr<SymbolicExpr>>::failure(result.error());
388 }
389 if (!result.value()) {
390 return Result<std::shared_ptr<SymbolicExpr>>::success(
391 make_integral_node(expr, var));
392 }
393
394 resolve_cycle(result.value(), my_idx);
395 return result;
396}

398Result<SymbolicExpr> Integrator::integrate_checked(
399 const SymbolicExpr& expr,
400 const std::string& var_name,
401 ComputationContext& context) {
402
403 if (var_name.empty()) {
404 return Result<SymbolicExpr>::failure(
405 CasErrc::InvalidArgument, "integration variable must not be empty",
406 "integrate");
407 }
408 auto access = context.consume_steps(1, "integrate");
409 if (!access) return Result<SymbolicExpr>::failure(access.error());
410
411 const bool top_level_query = query_depth_ == 0;
412 if (top_level_query) cycle_state_.history.clear();
413 ++query_depth_;
414 struct QueryExit {
415 Integrator& integrator;
416 ~QueryExit() { --integrator.query_depth_; }
417 } query_exit{*this};
418
419 if (auto pw = std::dynamic_pointer_cast<const PiecewiseNode>(lamina::detail::node(expr))) {
420 std::vector<PiecewiseNode::Branch> new_brs;
421 for (const auto& br : pw->branches()) {
422 PiecewiseNode::Branch nb;
423 auto branch = integrate_checked(
424 lamina::detail::expression_from_node(br.expression), var_name, context);
425 if (!branch) return branch;
426 nb.expression = lamina::detail::node(branch.value());
427 nb.condition = br.condition;
428 new_brs.push_back(nb);
429 }
430 std::shared_ptr<const SymbolicNode> new_def = nullptr;
431 if (pw->default_expr()) {
432 auto default_result = integrate_checked(
433 lamina::detail::expression_from_node(pw->default_expr()), var_name, context);
434 if (!default_result) return default_result;
435 new_def = lamina::detail::node(default_result.value());
436 }
437 return Result<SymbolicExpr>::success(lamina::detail::expression_from_node(
438 lamina::detail::make_node<PiecewiseNode>(std::move(new_brs), new_def)));
439 }
440
441 SymbolicExpr working_expr = expr;
442 try {
443 working_expr = apply_assumption_simplifications(
444 expr, var_name, context.assumptions().get());
445 auto normalized = working_expr.simplify();
446 if (!normalized) {
447 return Result<SymbolicExpr>::failure(
448 CasErrc::InternalInvariant,
449 "integrand normalization produced no expression",
450 "integrate.preprocess");
451 }
452 working_expr = *normalized;
453 } catch (const std::exception& error) {
454 return Result<SymbolicExpr>::failure(
455 CasErrc::InternalInvariant, error.what(), "integrate.preprocess");
456 }
457
458 auto recursive = integrate_recursive(
459 working_expr, var_name, context, 0);
460 if (!recursive) {
461 return Result<SymbolicExpr>::failure(recursive.error());
462 }
463 auto result = std::move(recursive.value());
464 auto final_access = context.consume_steps(0, "integrate");
465 if (!final_access) return Result<SymbolicExpr>::failure(final_access.error());
466 if (!result) {
467 return Result<SymbolicExpr>::failure(
468 CasErrc::InternalInvariant, "integration produced no result", "integrate");
469 }
470 return Result<SymbolicExpr>::success(*result);
471}

480Result<SymbolicExpr> Integrator::integrate_def_checked(
481 const SymbolicExpr& expr,
482 const std::string& var_name,
483 const SymbolicExpr& lower,
484 const SymbolicExpr& upper,
485 ComputationContext& context) {
486
487 auto access = context.consume_steps(1, "integrate.definite");
488 if (!access) return Result<SymbolicExpr>::failure(access.error());
489 SymbolicExpr simp_expr_val = *apply_assumption_simplifications(
490 expr, var_name, context.assumptions().get()).simplify();
491 bool is_inv_x = false;
492
493 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(lamina::detail::node(simp_expr_val))) {
494 if (auto v = std::dynamic_pointer_cast<const VariableNode>(pow->base())) {
495 if (v->name() == var_name) {
496 if (auto en = std::dynamic_pointer_cast<const NumberNode>(pow->exponent())) {
497 int eq_minus_one = 0;
498 if (std::holds_alternative<lmmc_real_t>(en->value())) {
499 lmmc_double_nearly_equal_tol(std::get<lmmc_real_t>(en->value()), -1.0, 1e-9, 1e-9, &eq_minus_one);
500 }
501 if ((std::holds_alternative<lmmc_real_t>(en->value()) && eq_minus_one != 0) ||
502 (std::holds_alternative<BigInt>(en->value()) && std::get<BigInt>(en->value()).to_int() == -1) ||
503 (std::holds_alternative<Rational>(en->value()) && std::get<Rational>(en->value()).to_double() == -1.0)) {
504 is_inv_x = true;
505 }
506 }
507 }
508 }
509 }
510
511 bool numeric_bounds = (lamina::detail::node(lower) && std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(lower))) &&
512 (lamina::detail::node(upper) && std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(upper)));
513
514 if (is_inv_x && numeric_bounds) {
515 auto l_checked = try_checked_numeric_constant(lower, context);
516 auto u_checked = try_checked_numeric_constant(upper, context);
517 if (l_checked && u_checked) {
518 double l_val = *l_checked;
519 double u_val = *u_checked;
520 int eq_l, eq_u;
521 lmmc_double_nearly_equal_tol(l_val, 0.0, 1e-9, 1e-9, &eq_l);
522 lmmc_double_nearly_equal_tol(u_val, 0.0, 1e-9, 1e-9, &eq_u);
523 if (!eq_l && l_val < 0 && !eq_u && u_val > 0) {
524 auto t = lamina::detail::make_expression_ptr(*SymbolicExpr::variable("t"));
525 auto zero = lamina::detail::make_expression_ptr(*SymbolicExpr::number(0));
526 auto int_left_result = integrate_def_checked(
527 expr, var_name, lower, *t, context);
528 if (!int_left_result) return int_left_result;
529 auto int_left = int_left_result.value();
530 auto lim_left_result = limit_expression_checked(
531 std::make_shared<SymbolicExpr>(int_left), "t", zero,
532 LimitDirection::FromBelow, context);
533 if (!lim_left_result) {
534 return Result<SymbolicExpr>::failure(
535 lim_left_result.error());
536 }
537 auto lim_left = lim_left_result.value();
538 auto int_right_result = integrate_def_checked(
539 expr, var_name, *t, upper, context);
540 if (!int_right_result) return int_right_result;
541 auto int_right = int_right_result.value();
542 auto lim_right_result = limit_expression_checked(
543 std::make_shared<SymbolicExpr>(int_right), "t", zero,
544 LimitDirection::FromAbove, context);
545 if (!lim_right_result) {
546 return Result<SymbolicExpr>::failure(
547 lim_right_result.error());
548 }
549 auto lim_right = lim_right_result.value();
550 if (lim_left && lim_right) {
551 return Result<SymbolicExpr>::success(
552 *SymbolicExpr::add(lim_left, lim_right));
553 }
554 }
555 }
556 }
557
558 auto indefinite_result = integrate_checked(expr, var_name, context);
559 if (!indefinite_result) return indefinite_result;
560 SymbolicExpr indefinite = indefinite_result.value();
561
562 if (std::dynamic_pointer_cast<const IntegralNode>(
563 lamina::detail::node(indefinite))) {
564 return Result<SymbolicExpr>::success(
565 lamina::detail::expression_from_node(
566 lamina::detail::make_node<IntegralNode>(
567 lamina::detail::node(expr), var_name,
568 lamina::detail::node(lower),
569 lamina::detail::node(upper))));
570 }
571
572 auto F_b = indefinite.substitute(var_name, make_expr_ptr(upper));
573 auto F_a = indefinite.substitute(var_name, make_expr_ptr(lower));
574 auto result = sym_sub(*F_b, *F_a);
575 return Result<SymbolicExpr>::success(*result->simplify());
576}
