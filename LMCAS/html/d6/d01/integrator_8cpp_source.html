7/**
8 * @brief Recursively replace |var| with var in an AST when the variable is known Positive.
9 *
10 * Traverses the expression tree and replaces any FunctionNode(Abs, [arg]) where
11 * arg is exactly the integration variable with just the variable itself.
12 * This is valid when the AssumptionContext confirms the variable is Positive.
13 */
14static std::shared_ptr<const SymbolicNode> simplify_abs_positive(
15 const std::shared_ptr<const SymbolicNode>& node, const std::string& var) {
16 if (!node) return node;
17
18 if (auto fn = std::dynamic_pointer_cast<const FunctionNode>(node)) {
19 if (fn->type() == FunctionNode::FuncType::Abs && fn->arguments().size() == 1) {
20 // Check if the argument is exactly the integration variable
21 if (auto vn = std::dynamic_pointer_cast<const VariableNode>(fn->arguments()[0])) {
22 if (vn->name() == var) {
23 // |x| → x when x is Positive
24 return fn->arguments()[0];
25 }
26 }
27 }
28 // Recurse into function arguments
29 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
30 bool changed = false;
31 for (auto& op : fn->arguments()) {
32 auto new_op = simplify_abs_positive(op, var);
33 if (new_op != op) changed = true;
34 new_ops.push_back(new_op);
35 }
36 if (changed) {
37 return lamina::detail::make_node<FunctionNode>(fn->type(), new_ops);
38 }
39 return node;
40 }
41
42 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
43 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
44 bool changed = false;
45 for (auto& op : add->operands()) {
46 auto new_op = simplify_abs_positive(op, var);
47 if (new_op != op) changed = true;
48 new_ops.push_back(new_op);
49 }
50 if (changed) return lamina::detail::make_node<AddNode>(new_ops);
51 return node;
52 }
53
54 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
55 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
56 bool changed = false;
57 for (auto& op : mul->operands()) {
58 auto new_op = simplify_abs_positive(op, var);
59 if (new_op != op) changed = true;
60 new_ops.push_back(new_op);
61 }
62 if (changed) return lamina::detail::make_node<MultiplyNode>(new_ops);
63 return node;
64 }
65
66 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
67 auto new_base = simplify_abs_positive(pow->base(), var);
68 auto new_exp = simplify_abs_positive(pow->exponent(), var);
69 if (new_base != pow->base() || new_exp != pow->exponent()) {
70 return lamina::detail::make_node<PowerNode>(new_base, new_exp);
71 }
72 return node;
73 }
74
75 // Leaf nodes (NumberNode, VariableNode, etc.) — no change
76 return node;
77}

79/**
80 * @brief Apply assumption-aware simplifications to an integrand.
81 *
82 * When the AssumptionContext indicates the integration variable is Positive,
83 * replaces |var| with var throughout the integrand. Returns the original
84 * expression unchanged if no simplifications apply.
85 */
86static SymbolicExpr apply_assumption_simplifications(
87 const SymbolicExpr& expr, const std::string& var,
88 const AssumptionContext* ctx) {
89 if (!ctx) return expr;
90
91 // Check if the integration variable is known Positive
92 SymbolicExpr var_expr = *SymbolicExpr::variable(var);
93 Tribool var_positive = ctx->is_positive(var_expr);
94
95 if (var_positive == Tribool::True) {
96 auto new_root = simplify_abs_positive(lamina::detail::node(expr), var);
97 if (new_root != lamina::detail::node(expr)) {
98 return lamina::detail::expression_from_node(new_root);
99 }
100 }
101
102 return expr;
103}

155void Integrator::resolve_cycle(std::shared_ptr<SymbolicExpr>& result, size_t cycle_idx) {
156 std::string cycle_var = "INT_CYCLE_" + std::to_string(cycle_idx);
157 if (depends_on_integration_variable(*result, cycle_var)) {
158 auto B = result->differentiate(cycle_var)->simplify();

160 auto one_minus_B = sym_sub(*SymbolicExpr::number(1), *B)->simplify();
161 if (!one_minus_B->is_zero()) {
162 result = SymbolicExpr::divide(A, one_minus_B);

167std::shared_ptr<SymbolicExpr> Integrator::apply_linearity(
168 const SymbolicExpr& expr, const std::string& var,
169 ComputationContext& context, int depth) {
170
171 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(expr))) {
172 std::vector<std::shared_ptr<const SymbolicNode>> constants;
173 std::vector<std::shared_ptr<const SymbolicNode>> dependents;
174
175 for (auto& op : mul->operands()) {
176 auto term = lamina::detail::expression_from_node(op);
177 if (!depends_on_integration_variable(term, var)) {
178 constants.push_back(op);
179 } else {
180 dependents.push_back(op);
181 }
182 }
183
184 if (!constants.empty() && dependents.size() < mul->operands().size()) {
185 auto const_part = (constants.size() == 1)
186 ? lamina::detail::expression_from_node(constants[0])
187 : lamina::detail::expression_from_node(
188 lamina::detail::make_node<MultiplyNode>(constants));
189 SymbolicExpr dep_part = (dependents.empty()) ?
190 *SymbolicExpr::number(1) :
191 ((dependents.size() == 1)
192 ? lamina::detail::expression_from_node(dependents[0])
193 : lamina::detail::expression_from_node(
194 lamina::detail::make_node<MultiplyNode>(dependents)));
195
196 auto int_part = integrate_recursive(dep_part, var, context, depth + 1);
197 if (!int_part) return nullptr;
198 return SymbolicExpr::multiply(
199 lamina::detail::make_expression_ptr(const_part),
200 int_part);
201 }
202 }
203
204 if (auto add = std::dynamic_pointer_cast<const AddNode>(lamina::detail::node(expr))) {
205 std::vector<std::shared_ptr<const SymbolicNode>> results;
206 for (auto& op : add->operands()) {
207 auto term = lamina::detail::expression_from_node(op);
208 auto int_term = integrate_recursive(term, var, context, depth + 1);
209 if (!int_term) return nullptr;
210 results.push_back(lamina::detail::node(int_term));
211 }
212 return lamina::detail::make_expression_ptr(lamina::detail::make_node<AddNode>(results));
213 }
214
215 return nullptr;
216}

232 error.what(),

237 return result;
238 }

243std::shared_ptr<SymbolicExpr> Integrator::integrate_recursive(
244 const SymbolicExpr& expr, const std::string& var,
245 ComputationContext& context, int depth) {
246
247 auto entered = context.enter_recursion("integrate.recursive");
248 if (!entered) return nullptr;
249 struct RecursionExit {
250 ComputationContext& context;
251 ~RecursionExit() { context.leave_recursion(); }
252 } recursion_exit{context};
253
254 auto linear_result = apply_linearity(expr, var, context, depth);
255 if (linear_result) return linear_result;
256
257 auto cycle_result = check_cycle(expr, var);
258 if (cycle_result) return cycle_result;
259
260 cycle_state_.history.push_back(expr);
261 size_t my_idx = cycle_state_.history.size() - 1;
262
263 if (expr.is_number() || !depends_on_integration_variable(expr, var)) {
264 cycle_state_.history.pop_back();
265 return SymbolicExpr::multiply(make_expr_ptr(expr), SymbolicExpr::variable(var));
266 }
267
268 auto result = dispatch_strategies(expr, var, context, depth);
269
270 cycle_state_.history.pop_back();
271
272 if (!result) {
273 return make_integral_node(expr, var);
274 }
275
276 resolve_cycle(result, my_idx);
277
278 return result;
279}

281Result<SymbolicExpr> Integrator::integrate_checked(
282 const SymbolicExpr& expr,
283 const std::string& var_name,
284 ComputationContext& context) {
285
286 if (var_name.empty()) {
287 return Result<SymbolicExpr>::failure(
288 CasErrc::InvalidArgument, "integration variable must not be empty",
289 "integrate");
290 }
291 auto access = context.consume_steps(1, "integrate");
292 if (!access) return Result<SymbolicExpr>::failure(access.error());
293
294 const bool top_level_query = query_depth_ == 0;
295 if (top_level_query) cycle_state_.history.clear();
296 ++query_depth_;
297 struct QueryExit {
298 Integrator& integrator;
299 ~QueryExit() { --integrator.query_depth_; }
300 } query_exit{*this};
301
302 if (auto pw = std::dynamic_pointer_cast<const PiecewiseNode>(lamina::detail::node(expr))) {
303 std::vector<PiecewiseNode::Branch> new_brs;
304 for (const auto& br : pw->branches()) {
305 PiecewiseNode::Branch nb;
306 auto branch = integrate_checked(
307 lamina::detail::expression_from_node(br.expression), var_name, context);
308 if (!branch) return branch;
309 nb.expression = lamina::detail::node(branch.value());
310 nb.condition = br.condition;
311 new_brs.push_back(nb);
312 }
313 std::shared_ptr<const SymbolicNode> new_def = nullptr;
314 if (pw->default_expr()) {
315 auto default_result = integrate_checked(
316 lamina::detail::expression_from_node(pw->default_expr()), var_name, context);
317 if (!default_result) return default_result;
318 new_def = lamina::detail::node(default_result.value());
319 }
320 return Result<SymbolicExpr>::success(lamina::detail::expression_from_node(
321 lamina::detail::make_node<PiecewiseNode>(std::move(new_brs), new_def)));
322 }
323
324 SymbolicExpr working_expr = expr;
325 try {
326 working_expr = apply_assumption_simplifications(
327 expr, var_name, context.assumptions().get());
328 auto normalized = working_expr.simplify();
329 if (!normalized) {
330 return Result<SymbolicExpr>::failure(
331 CasErrc::InternalInvariant,
332 "integrand normalization produced no expression",
333 "integrate.preprocess");
334 }
335 working_expr = *normalized;
336 } catch (const std::exception& error) {
337 return Result<SymbolicExpr>::failure(
338 CasErrc::InternalInvariant, error.what(), "integrate.preprocess");
339 }
340
341 std::shared_ptr<SymbolicExpr> result;
342 try {
343 result = integrate_recursive(working_expr, var_name, context, 0);
344 } catch (const std::exception& error) {
345 return Result<SymbolicExpr>::failure(
346 CasErrc::InternalInvariant, error.what(), "integrate.recursive");
347 }
348 if (!context.diagnostics().empty()) {
349 const auto& diagnostic = context.diagnostics().back();
350 if (diagnostic.severity == DiagnosticSeverity::Error &&
351 diagnostic.operation.find("integrate.strategy.") == 0) {
352 return Result<SymbolicExpr>::failure(
353 CasErrc::InternalInvariant,
354 diagnostic.message,
355 diagnostic.operation);
356 }
357 }
358 auto final_access = context.consume_steps(0, "integrate");
359 if (!final_access) return Result<SymbolicExpr>::failure(final_access.error());
360 if (!result) {
361 return Result<SymbolicExpr>::failure(
362 CasErrc::InternalInvariant, "integration produced no result", "integrate");
363 }
364 return Result<SymbolicExpr>::success(*result);
365}

379Result<SymbolicExpr> Integrator::integrate_def_checked(
380 const SymbolicExpr& expr,
381 const std::string& var_name,
382 const SymbolicExpr& lower,
383 const SymbolicExpr& upper,
384 ComputationContext& context) {
385
386 auto access = context.consume_steps(1, "integrate.definite");
387 if (!access) return Result<SymbolicExpr>::failure(access.error());
388 SymbolicExpr simp_expr_val = *apply_assumption_simplifications(
389 expr, var_name, context.assumptions().get()).simplify();
390 bool is_inv_x = false;
391
392 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(lamina::detail::node(simp_expr_val))) {
393 if (auto v = std::dynamic_pointer_cast<const VariableNode>(pow->base())) {
394 if (v->name() == var_name) {
395 if (auto en = std::dynamic_pointer_cast<const NumberNode>(pow->exponent())) {
396 int eq_minus_one = 0;
397 if (std::holds_alternative<lmmc_real_t>(en->value())) {
398 lmmc_double_nearly_equal_tol(std::get<lmmc_real_t>(en->value()), -1.0, 1e-9, 1e-9, &eq_minus_one);
399 }
400 if ((std::holds_alternative<lmmc_real_t>(en->value()) && eq_minus_one != 0) ||
401 (std::holds_alternative<BigInt>(en->value()) && std::get<BigInt>(en->value()).to_int() == -1) ||
402 (std::holds_alternative<Rational>(en->value()) && std::get<Rational>(en->value()).to_double() == -1.0)) {
403 is_inv_x = true;
404 }
405 }
406 }
407 }
408 }
409
410 bool numeric_bounds = (lamina::detail::node(lower) && std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(lower))) &&
411 (lamina::detail::node(upper) && std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(upper)));
412
413 if (is_inv_x && numeric_bounds) {
414 auto l_checked = try_checked_numeric_constant(lower);
415 auto u_checked = try_checked_numeric_constant(upper);
416 if (l_checked && u_checked) {
417 double l_val = *l_checked;
418 double u_val = *u_checked;
419 int eq_l, eq_u;
420 lmmc_double_nearly_equal_tol(l_val, 0.0, 1e-9, 1e-9, &eq_l);
421 lmmc_double_nearly_equal_tol(u_val, 0.0, 1e-9, 1e-9, &eq_u);
422 if (!eq_l && l_val < 0 && !eq_u && u_val > 0) {
423 auto t = lamina::detail::make_expression_ptr(*SymbolicExpr::variable("t"));
424 auto zero = lamina::detail::make_expression_ptr(*SymbolicExpr::number(0));
425 auto int_left_result = integrate_def_checked(
426 expr, var_name, lower, *t, context);
427 if (!int_left_result) return int_left_result;
428 auto int_left = int_left_result.value();
429 auto lim_left = int_left.limit("t", zero, "-");
430 auto int_right_result = integrate_def_checked(
431 expr, var_name, *t, upper, context);
432 if (!int_right_result) return int_right_result;
433 auto int_right = int_right_result.value();
434 auto lim_right = int_right.limit("t", zero, "+");
435 if (lim_left && lim_right) {
436 return Result<SymbolicExpr>::success(
437 *SymbolicExpr::add(lim_left, lim_right));
438 }
439 }
440 }
441 }
442
443 auto indefinite_result = integrate_checked(expr, var_name, context);
444 if (!indefinite_result) return indefinite_result;
445 SymbolicExpr indefinite = indefinite_result.value();
446
447 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(indefinite))) {
448 if (func->type() == FunctionNode::FuncType::Calculus_Integral) {
449 std::vector<std::shared_ptr<const SymbolicNode>> args;
450 args.push_back(lamina::detail::node(expr));
451 args.push_back(lamina::detail::node(SymbolicExpr::variable(var_name)));
452 args.push_back(lamina::detail::node(lower));
453 args.push_back(lamina::detail::node(upper));
454 return Result<SymbolicExpr>::success(
455 lamina::detail::expression_from_node(lamina::detail::make_node<FunctionNode>(
456 FunctionNode::FuncType::Calculus_Integral, args)));
457 }
458 }
459
460 auto F_b = indefinite.substitute(var_name, make_expr_ptr(upper));
461 auto F_a = indefinite.substitute(var_name, make_expr_ptr(lower));
462 auto result = sym_sub(*F_b, *F_a);
463 return Result<SymbolicExpr>::success(*result->simplify());
464}
