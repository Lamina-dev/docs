6IntegrationStrategyResult IntegrationStrategy::try_integrate(
7 const SymbolicExpr& expr,
8 const std::string& var,
9 Integrator& integrator,
10 ComputationContext& computation,
11 int depth) {
12 const std::string operation = "integrate.strategy." + name();
13 auto access = computation.consume_steps(0, operation);
14 if (!access) { return IntegrationStrategyResult::failure(access.error()); }
15 try {
16 auto raw_result = try_integrate_raw(
17 expr, var, integrator, computation, depth);
18 if (!raw_result) {
19 return IntegrationStrategyResult::failure(raw_result.error());
20 }
21 auto expression = std::move(raw_result.value());
22 auto final_access = computation.consume_steps(0, operation);
23 if (!final_access) {
24 return IntegrationStrategyResult::failure(final_access.error());
25 }
26 if (!expression) {

28 IntegrationNotApplicable{});

31 std::move(expression), name()});

33 return IntegrationStrategyResult::failure(
34 CasErrc::ResourceLimit,
35 "integration strategy allocation failed", operation);
36 } catch (const std::exception& error) {
37 return IntegrationStrategyResult::failure(
38 CasErrc::InternalInvariant, error.what(), operation);

99void Integrator::resolve_cycle(std::shared_ptr<SymbolicExpr>& result, size_t cycle_idx) {
100 std::string cycle_var = "INT_CYCLE_" + std::to_string(cycle_idx);
101 if (depends_on_integration_variable(*result, cycle_var)) {
102 auto B = result->differentiate(cycle_var)->simplify();

111Result<std::shared_ptr<SymbolicExpr>> Integrator::dispatch_strategies(
112 const SymbolicExpr& expr, const std::string& var,
113 ComputationContext& context, int depth) {
114 for (auto& strategy : strategies_) {
115 auto budget = context.consume_steps(1, "integrate.strategy");
116 if (!budget) {
117 return Result<std::shared_ptr<SymbolicExpr>>::failure(
118 budget.error());
119 }
120 auto attempt = strategy->try_integrate(
121 expr, var, *this, context, depth);
122 if (!attempt) {
123 return Result<std::shared_ptr<SymbolicExpr>>::failure(
124 attempt.error());
125 }
126 if (auto* candidate =
127 std::get_if<IntegrationCandidate>(&attempt.value())) {
128 if (!strategy->requires_residual_verification()) {
129 return Result<std::shared_ptr<SymbolicExpr>>::success(
130 candidate->expression);
131 }
132 auto derivative = candidate->expression
133 ? candidate->expression->differentiate(var) : nullptr;
134 if (!derivative) {
135 if (depth > 0) {
136 return Result<std::shared_ptr<SymbolicExpr>>::success(
137 nullptr);
138 }
139 continue;
140 }
141 auto normalized_derivative = derivative->simplify();
142 auto normalized_integrand = detail::make_expression_ptr(expr)->simplify();
143 if (normalized_derivative && normalized_integrand &&
144 LMCAS::detail::node(normalized_derivative)->equals(
145 *LMCAS::detail::node(normalized_integrand))) {
146 return Result<std::shared_ptr<SymbolicExpr>>::success(
147 candidate->expression);
148 }
149 auto delta = SymbolicExpr::add(
150 derivative,
151 SymbolicExpr::multiply(
152 SymbolicExpr::number(-1), detail::make_expression_ptr(expr)));
153 auto proof = check_zero_residual(delta, context);
154 if (!proof) return Result<std::shared_ptr<SymbolicExpr>>::failure(proof.error());
155 if (std::holds_alternative<ProvedZeroResidual>(proof.value())) {
156 return Result<std::shared_ptr<SymbolicExpr>>::success(
157 candidate->expression);
158 }
159 if (depth > 0) {
160 return Result<std::shared_ptr<SymbolicExpr>>::success(nullptr);
161 }
162 }
163 }
164 return Result<std::shared_ptr<SymbolicExpr>>::success(nullptr);
165}

167Result<std::shared_ptr<SymbolicExpr>> Integrator::integrate_recursive(
168 const SymbolicExpr& expr, const std::string& var,
169 ComputationContext& context, int depth) {
170 auto entered = context.enter_recursion("integrate.recursive");
171 if (!entered) {
172 return Result<std::shared_ptr<SymbolicExpr>>::failure(
173 entered.error());
174 }
175 struct RecursionExit {
176 ComputationContext& context;
177 ~RecursionExit() { context.leave_recursion(); }
178 } recursion_exit{context};
179
180 auto linear_result = apply_linearity(expr, var, context, depth);
181 if (!linear_result) {
182 return Result<std::shared_ptr<SymbolicExpr>>::failure(
183 linear_result.error());
184 }
185 if (linear_result.value()) { return linear_result; }
186
187 auto cycle_result = check_cycle(expr, var);
188 if (cycle_result) {
189 return Result<std::shared_ptr<SymbolicExpr>>::success(
190 std::move(cycle_result));
191 }
192
193 cycle_state_.history.push_back(expr);
194 const size_t my_idx = cycle_state_.history.size() - 1;
195
196 if (expr.is_number() || !depends_on_integration_variable(expr, var)) {
197 cycle_state_.history.pop_back();
198 return Result<std::shared_ptr<SymbolicExpr>>::success(
199 SymbolicExpr::multiply(
200 detail::make_expression_ptr(expr), SymbolicExpr::variable(var)));
201 }
202
203 auto result = dispatch_strategies(expr, var, context, depth);
204 cycle_state_.history.pop_back();
205 if (!result) {
206 return Result<std::shared_ptr<SymbolicExpr>>::failure(result.error());
207 }
208 if (!result.value()) {
209 return Result<std::shared_ptr<SymbolicExpr>>::success(
210 make_integral_node(expr, var));
211 }
212
213 resolve_cycle(result.value(), my_idx);
214 return result;
215}

218 const SymbolicExpr& expr,
219 const std::string& var_name,
220 ComputationContext& context) {
221
222 if (var_name.empty()) {
223 return Result<SymbolicExpr>::failure(
224 CasErrc::InvalidArgument, "integration variable must not be empty",
225 "integrate");

234 Integrator& integrator;

241 PiecewiseNode::Branch nb;

248 }

253 if (!default_result) { return default_result; }

255 }

257 LMCAS::detail::make_node<PiecewiseNode>(std::move(new_brs), new_def)));
258 }

261 try {
262 auto assumed = apply_assumption_simplifications(
263 expr, var_name, context.assumptions().get());
264 if (!assumed) { return assumed; }
265 working_expr = std::move(assumed.value());
266 auto normalized = working_expr.simplify();
267 if (!normalized) {
268 return Result<SymbolicExpr>::failure(
269 CasErrc::InternalInvariant,
270 "integrand normalization produced no expression",
271 "integrate.preprocess");
272 }
273 working_expr = *normalized;
274 } catch (const std::exception& error) {
275 return Result<SymbolicExpr>::failure(
276 CasErrc::InternalInvariant, error.what(), "integrate.preprocess");
277 }
278
279 auto recursive = integrate_recursive(
280 working_expr, var_name, context, 0);
281 if (!recursive) {
282 return Result<SymbolicExpr>::failure(recursive.error());
283 }
284 auto result = std::move(recursive.value());
285 auto final_access = context.consume_steps(0, "integrate");
286 if (!final_access) { return Result<SymbolicExpr>::failure(final_access.error()); }
287 if (!result) {
288 return Result<SymbolicExpr>::failure(
289 CasErrc::InternalInvariant, "integration produced no result", "integrate");
290 }
291 return Result<SymbolicExpr>::success(*result);
292}
