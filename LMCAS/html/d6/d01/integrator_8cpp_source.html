7/**
8 * @brief Recursively replace |var| with var in an AST when the variable is known Positive.
9 *
10 * Traverses the expression tree and replaces any FunctionNode(Abs, [arg]) where
11 * arg is exactly the integration variable with just the variable itself.
12 * This is valid when the AssumptionContext confirms the variable is Positive.
13 */
14static std::shared_ptr<const SymbolicNode> simplify_abs_positive(
15 const std::shared_ptr<const SymbolicNode>& node, const std::string& var) {
16 if (!node) return node;
17
18 if (auto fn = std::dynamic_pointer_cast<const FunctionNode>(node)) {
19 if (fn->type() == FunctionNode::FuncType::Abs && fn->arguments().size() == 1) {
20 // Check if the argument is exactly the integration variable
21 if (auto vn = std::dynamic_pointer_cast<const VariableNode>(fn->arguments()[0])) {
22 if (vn->name() == var) {
23 // |x| → x when x is Positive
24 return fn->arguments()[0];
25 }
26 }
27 }
28 // Recurse into function arguments
29 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
30 bool changed = false;
31 for (auto& op : fn->arguments()) {
32 auto new_op = simplify_abs_positive(op, var);
33 if (new_op != op) changed = true;
34 new_ops.push_back(new_op);
35 }
36 if (changed) {
37 return LMCAS::detail::make_node<FunctionNode>(fn->type(), new_ops);
38 }
39 return node;
40 }
41
42 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
43 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
44 bool changed = false;
45 for (auto& op : add->operands()) {
46 auto new_op = simplify_abs_positive(op, var);
47 if (new_op != op) changed = true;
48 new_ops.push_back(new_op);
49 }
50 if (changed) return LMCAS::detail::make_node<AddNode>(new_ops);
51 return node;
52 }
53
54 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
55 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
56 bool changed = false;
57 for (auto& op : mul->operands()) {
58 auto new_op = simplify_abs_positive(op, var);
59 if (new_op != op) changed = true;
60 new_ops.push_back(new_op);
61 }
62 if (changed) return LMCAS::detail::make_node<MultiplyNode>(new_ops);
63 return node;
64 }
65
66 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
67 auto new_base = simplify_abs_positive(pow->base(), var);
68 auto new_exp = simplify_abs_positive(pow->exponent(), var);
69 if (new_base != pow->base() || new_exp != pow->exponent()) {
70 return LMCAS::detail::make_node<PowerNode>(new_base, new_exp);
71 }
72 return node;
73 }
74
75 // Leaf nodes (NumberNode, VariableNode, etc.) — no change
76 return node;
77}

79/**
80 * @brief Apply assumption-aware simplifications to an integrand.
81 *
82 * When the AssumptionContext indicates the integration variable is Positive,
83 * replaces |var| with var throughout the integrand. Returns the original
84 * expression unchanged if no simplifications apply.
85 */
86static Result<SymbolicExpr> apply_assumption_simplifications(
87 const SymbolicExpr& expr, const std::string& var,
88 const AssumptionContext* ctx) {
89 if (!ctx) return expr;
90
91 SymbolicExpr var_expr = *SymbolicExpr::variable(var);
92 auto positive = ctx->is_positive(var_expr);
93 if (!positive) {
94 return Result<SymbolicExpr>::failure(positive.error());
95 }
96
97 if (positive.value() == Tribool::True) {
98 auto new_root = simplify_abs_positive(
99 LMCAS::detail::node(expr), var);
100 if (new_root != LMCAS::detail::node(expr)) {
101 return LMCAS::detail::expression_from_node(new_root);
102 }
103 }
104
105 return expr;
106}

108IntegrationStrategyResult IntegrationStrategy::try_integrate(
109 const SymbolicExpr& expr,
110 const std::string& var,
111 Integrator& integrator,
112 ComputationContext& computation,
113 int depth) {
114 const std::string operation = "integrate.strategy." + name();
115 auto access = computation.consume_steps(0, operation);
116 if (!access) return IntegrationStrategyResult::failure(access.error());
117 try {
118 auto raw_result = try_integrate_raw(
119 expr, var, integrator, computation, depth);
120 if (!raw_result) {
121 return IntegrationStrategyResult::failure(raw_result.error());
122 }
123 auto expression = std::move(raw_result.value());
124 auto final_access = computation.consume_steps(0, operation);
125 if (!final_access) {
126 return IntegrationStrategyResult::failure(final_access.error());
127 }
128 if (!expression) {
129 return IntegrationStrategyResult::success(
130 IntegrationNotApplicable{});
131 }
132 return IntegrationStrategyResult::success(IntegrationCandidate{
133 std::move(expression), name()});
134 } catch (const std::bad_alloc&) {
135 return IntegrationStrategyResult::failure(
136 CasErrc::ResourceLimit,
137 "integration strategy allocation failed", operation);
138 } catch (const std::exception& error) {
139 return IntegrationStrategyResult::failure(
140 CasErrc::InternalInvariant, error.what(), operation);
141 }

192 for (size_t i = 0; i < cycle_state_.history.size(); ++i) {
193 auto ratio = SymbolicExpr::divide(make_expr_ptr(expr), make_expr_ptr(cycle_state_.history[i]))->simplify();
194 if (!depends_on_integration_variable(*ratio, var)) {

201void Integrator::resolve_cycle(std::shared_ptr<SymbolicExpr>& result, size_t cycle_idx) {
202 std::string cycle_var = "INT_CYCLE_" + std::to_string(cycle_idx);
203 if (depends_on_integration_variable(*result, cycle_var)) {
204 auto B = result->differentiate(cycle_var)->simplify();
205 auto A = result->substitute(cycle_var, SymbolicExpr::number(0))->simplify();
206 auto one_minus_B = sym_sub(*SymbolicExpr::number(1), *B)->simplify();
207 if (!one_minus_B->is_zero()) {
208 result = SymbolicExpr::divide(A, one_minus_B);
209 }
210 }

213Result<std::shared_ptr<SymbolicExpr>> Integrator::apply_linearity(
214 const SymbolicExpr& expr, const std::string& var,
215 ComputationContext& context, int depth) {
216
217 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(LMCAS::detail::node(expr))) {
218 std::vector<std::shared_ptr<const SymbolicNode>> constants;
219 std::vector<std::shared_ptr<const SymbolicNode>> dependents;
220
221 for (auto& op : mul->operands()) {
222 auto term = LMCAS::detail::expression_from_node(op);
223 if (!depends_on_integration_variable(term, var)) {
224 constants.push_back(op);
225 } else {
226 dependents.push_back(op);
227 }
228 }
229
230 if (!constants.empty() && dependents.size() < mul->operands().size()) {
231 auto const_part = (constants.size() == 1)
232 ? LMCAS::detail::expression_from_node(constants[0])
233 : LMCAS::detail::expression_from_node(
234 LMCAS::detail::make_node<MultiplyNode>(constants));
235 SymbolicExpr dep_part = (dependents.empty()) ?
236 *SymbolicExpr::number(1) :
237 ((dependents.size() == 1)
238 ? LMCAS::detail::expression_from_node(dependents[0])
239 : LMCAS::detail::expression_from_node(
240 LMCAS::detail::make_node<MultiplyNode>(dependents)));
241 if (dependents.size() > 1) {
242 std::vector<std::shared_ptr<const SymbolicNode>> exponents;
243 exponents.reserve(dependents.size());
244 bool all_exponential = true;
245 for (const auto& dependent : dependents) {
246 auto function =
247 std::dynamic_pointer_cast<const FunctionNode>(dependent);
248 if (!function ||
249 function->type() != FunctionNode::FuncType::Exp ||
250 function->arguments().size() != 1) {
251 all_exponential = false;
252 break;
253 }
254 exponents.push_back(function->arguments()[0]);
255 }
256 if (all_exponential) {
257 dep_part = LMCAS::detail::expression_from_node(
258 LMCAS::detail::make_node<FunctionNode>(
259 FunctionNode::FuncType::Exp,
260 std::vector<std::shared_ptr<const SymbolicNode>>{
261 SymbolicFactory::create_add(
262 std::move(exponents))}));
263 }
264 }
265
266 auto int_part = integrate_recursive(dep_part, var, context, depth + 1);
267 if (!int_part) {
268 return Result<std::shared_ptr<SymbolicExpr>>::failure(
269 int_part.error());
270 }
271 return Result<std::shared_ptr<SymbolicExpr>>::success(
272 SymbolicExpr::multiply(
273 LMCAS::detail::make_expression_ptr(const_part),
274 int_part.value()));
275 }
276 }
277
278 if (auto add = std::dynamic_pointer_cast<const AddNode>(
279 LMCAS::detail::node(expr))) {
280 std::vector<std::shared_ptr<const SymbolicNode>> results;
281 for (auto& op : add->operands()) {
282 auto term = LMCAS::detail::expression_from_node(op);
283 auto int_term = integrate_recursive(
284 term, var, context, depth + 1);
285 if (!int_term) {
286 return Result<std::shared_ptr<SymbolicExpr>>::failure(
287 int_term.error());
288 }
289 results.push_back(LMCAS::detail::node(int_term.value()));
290 }
291 return Result<std::shared_ptr<SymbolicExpr>>::success(
292 LMCAS::detail::make_expression_ptr(
293 LMCAS::detail::make_node<AddNode>(results)));
294 }
295 return Result<std::shared_ptr<SymbolicExpr>>::success(nullptr);
296}

298Result<std::shared_ptr<SymbolicExpr>> Integrator::dispatch_strategies(
299 const SymbolicExpr& expr, const std::string& var,
300 ComputationContext& context, int depth) {
301 for (auto& strategy : strategies_) {
302 auto budget = context.consume_steps(1, "integrate.strategy");
303 if (!budget) {
304 return Result<std::shared_ptr<SymbolicExpr>>::failure(
305 budget.error());
306 }
307 auto attempt = strategy->try_integrate(
308 expr, var, *this, context, depth);
309 if (!attempt) {
310 return Result<std::shared_ptr<SymbolicExpr>>::failure(
311 attempt.error());
312 }
313 if (auto* candidate =
314 std::get_if<IntegrationCandidate>(&attempt.value())) {
315 if (!strategy->requires_residual_verification()) {
316 return Result<std::shared_ptr<SymbolicExpr>>::success(
317 candidate->expression);
318 }
319 auto derivative = candidate->expression
320 ? candidate->expression->differentiate(var) : nullptr;
321 if (!derivative) {
322 if (depth > 0) {
323 return Result<std::shared_ptr<SymbolicExpr>>::success(
324 nullptr);
325 }
326 continue;
327 }
328 auto normalized_derivative = derivative->simplify();
329 auto normalized_integrand = make_expr_ptr(expr)->simplify();
330 if (normalized_derivative && normalized_integrand &&
331 LMCAS::detail::node(normalized_derivative)->equals(
332 *LMCAS::detail::node(normalized_integrand))) {
333 return Result<std::shared_ptr<SymbolicExpr>>::success(
334 candidate->expression);
335 }
336 auto delta = SymbolicExpr::add(
337 derivative,
338 SymbolicExpr::multiply(
339 SymbolicExpr::number(-1), make_expr_ptr(expr)));
340 delta = delta->cancel()->simplify();
341 if (delta && delta->is_zero()) {
342 return Result<std::shared_ptr<SymbolicExpr>>::success(
343 candidate->expression);
344 }
345 if (depth > 0) {
346 return Result<std::shared_ptr<SymbolicExpr>>::success(nullptr);
347 }
348 }
349 }
350 return Result<std::shared_ptr<SymbolicExpr>>::success(nullptr);
351}

353Result<std::shared_ptr<SymbolicExpr>> Integrator::integrate_recursive(
354 const SymbolicExpr& expr, const std::string& var,
355 ComputationContext& context, int depth) {
356 auto entered = context.enter_recursion("integrate.recursive");
357 if (!entered) {
358 return Result<std::shared_ptr<SymbolicExpr>>::failure(
359 entered.error());
360 }
361 struct RecursionExit {
362 ComputationContext& context;
363 ~RecursionExit() { context.leave_recursion(); }
364 } recursion_exit{context};
365
366 auto linear_result = apply_linearity(expr, var, context, depth);
367 if (!linear_result) {
368 return Result<std::shared_ptr<SymbolicExpr>>::failure(
369 linear_result.error());
370 }
371 if (linear_result.value()) return linear_result;
372
373 auto cycle_result = check_cycle(expr, var);
374 if (cycle_result) {
375 return Result<std::shared_ptr<SymbolicExpr>>::success(
376 std::move(cycle_result));
377 }
378
379 cycle_state_.history.push_back(expr);
380 const size_t my_idx = cycle_state_.history.size() - 1;
381
382 if (expr.is_number() || !depends_on_integration_variable(expr, var)) {
383 cycle_state_.history.pop_back();
384 return Result<std::shared_ptr<SymbolicExpr>>::success(
385 SymbolicExpr::multiply(
386 make_expr_ptr(expr), SymbolicExpr::variable(var)));
387 }
388
389 auto result = dispatch_strategies(expr, var, context, depth);
390 cycle_state_.history.pop_back();
391 if (!result) {
392 return Result<std::shared_ptr<SymbolicExpr>>::failure(result.error());
393 }
394 if (!result.value()) {
395 return Result<std::shared_ptr<SymbolicExpr>>::success(
396 make_integral_node(expr, var));
397 }
398
399 resolve_cycle(result.value(), my_idx);
400 return result;
401}

403Result<SymbolicExpr> Integrator::integrate_checked(
404 const SymbolicExpr& expr,
405 const std::string& var_name,
406 ComputationContext& context) {
407
408 if (var_name.empty()) {
409 return Result<SymbolicExpr>::failure(
410 CasErrc::InvalidArgument, "integration variable must not be empty",
411 "integrate");
412 }
413 auto access = context.consume_steps(1, "integrate");
414 if (!access) return Result<SymbolicExpr>::failure(access.error());
415
416 const bool top_level_query = query_depth_ == 0;
417 if (top_level_query) cycle_state_.history.clear();
418 ++query_depth_;
419 struct QueryExit {
420 Integrator& integrator;
421 ~QueryExit() { --integrator.query_depth_; }
422 } query_exit{*this};
423
424 if (auto pw = std::dynamic_pointer_cast<const PiecewiseNode>(LMCAS::detail::node(expr))) {
425 std::vector<PiecewiseNode::Branch> new_brs;
426 for (const auto& br : pw->branches()) {
427 PiecewiseNode::Branch nb;
428 auto branch = integrate_checked(
429 LMCAS::detail::expression_from_node(br.expression), var_name, context);
430 if (!branch) return branch;
431 nb.expression = LMCAS::detail::node(branch.value());
432 nb.condition = br.condition;
433 new_brs.push_back(nb);
434 }
435 std::shared_ptr<const SymbolicNode> new_def = nullptr;
436 if (pw->default_expr()) {
437 auto default_result = integrate_checked(
438 LMCAS::detail::expression_from_node(pw->default_expr()), var_name, context);
439 if (!default_result) return default_result;
440 new_def = LMCAS::detail::node(default_result.value());
441 }
442 return Result<SymbolicExpr>::success(LMCAS::detail::expression_from_node(
443 LMCAS::detail::make_node<PiecewiseNode>(std::move(new_brs), new_def)));
444 }
445
446 SymbolicExpr working_expr = expr;
447 try {
448 auto assumed = apply_assumption_simplifications(
449 expr, var_name, context.assumptions().get());
450 if (!assumed) return assumed;
451 working_expr = std::move(assumed.value());
452 auto normalized = working_expr.simplify();
453 if (!normalized) {
454 return Result<SymbolicExpr>::failure(
455 CasErrc::InternalInvariant,
456 "integrand normalization produced no expression",
457 "integrate.preprocess");
458 }
459 working_expr = *normalized;
460 } catch (const std::exception& error) {
461 return Result<SymbolicExpr>::failure(
462 CasErrc::InternalInvariant, error.what(), "integrate.preprocess");
463 }
464
465 auto recursive = integrate_recursive(
466 working_expr, var_name, context, 0);
467 if (!recursive) {
468 return Result<SymbolicExpr>::failure(recursive.error());
469 }
470 auto result = std::move(recursive.value());
471 auto final_access = context.consume_steps(0, "integrate");
472 if (!final_access) return Result<SymbolicExpr>::failure(final_access.error());
473 if (!result) {
474 return Result<SymbolicExpr>::failure(
475 CasErrc::InternalInvariant, "integration produced no result", "integrate");
476 }
477 return Result<SymbolicExpr>::success(*result);
478}

487Result<SymbolicExpr> Integrator::integrate_def_checked(
488 const SymbolicExpr& expr,
489 const std::string& var_name,
490 const SymbolicExpr& lower,
491 const SymbolicExpr& upper,
492 ComputationContext& context) {
493
494 auto access = context.consume_steps(1, "integrate.definite");
495 if (!access) return Result<SymbolicExpr>::failure(access.error());
496 SymbolicExpr simp_expr_val = expr;
497 try {
498 auto assumed = apply_assumption_simplifications(
499 expr, var_name, context.assumptions().get());
500 if (!assumed) return assumed;
501 auto simplified = assumed.value().simplify();
502 if (!simplified) {
503 return Result<SymbolicExpr>::failure(
504 CasErrc::InternalInvariant,
505 "definite integrand normalization produced no expression",
506 "integrate.definite.preprocess");
507 }
508 simp_expr_val = *simplified;
509 } catch (const std::exception& error) {
510 return Result<SymbolicExpr>::failure(
511 CasErrc::InternalInvariant, error.what(),
512 "integrate.definite.preprocess");
513 }
514 bool is_inv_x = false;
515
516 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(LMCAS::detail::node(simp_expr_val))) {
517 if (auto v = std::dynamic_pointer_cast<const VariableNode>(pow->base())) {
518 if (v->name() == var_name) {
519 if (auto en = std::dynamic_pointer_cast<const NumberNode>(pow->exponent())) {
520 int eq_minus_one = 0;
521 if (std::holds_alternative<lmmc_real_t>(en->value())) {
522 lmmc_double_nearly_equal_tol(std::get<lmmc_real_t>(en->value()), -1.0, 1e-9, 1e-9, &eq_minus_one);
523 }
524 if ((std::holds_alternative<lmmc_real_t>(en->value()) && eq_minus_one != 0) ||
525 (std::holds_alternative<BigInt>(en->value()) && std::get<BigInt>(en->value()).to_int() == -1) ||
526 (std::holds_alternative<Rational>(en->value()) && std::get<Rational>(en->value()).to_double() == -1.0)) {
527 is_inv_x = true;
528 }
529 }
530 }
531 }
532 }
533
534 bool numeric_bounds = (LMCAS::detail::node(lower) && std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(lower))) &&
535 (LMCAS::detail::node(upper) && std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(upper)));
536
537 if (is_inv_x && numeric_bounds) {
538 auto l_checked = try_checked_numeric_constant(lower, context);
539 auto u_checked = try_checked_numeric_constant(upper, context);
540 if (l_checked && u_checked) {
541 double l_val = *l_checked;
542 double u_val = *u_checked;
543 int eq_l, eq_u;
544 lmmc_double_nearly_equal_tol(l_val, 0.0, 1e-9, 1e-9, &eq_l);
545 lmmc_double_nearly_equal_tol(u_val, 0.0, 1e-9, 1e-9, &eq_u);
546 if (!eq_l && l_val < 0 && !eq_u && u_val > 0) {
547 auto t = LMCAS::detail::make_expression_ptr(*SymbolicExpr::variable("t"));
548 auto zero = LMCAS::detail::make_expression_ptr(*SymbolicExpr::number(0));
549 auto int_left_result = integrate_def_checked(
550 expr, var_name, lower, *t, context);
551 if (!int_left_result) return int_left_result;
552 auto int_left = int_left_result.value();
553 auto lim_left_result = limit_expression_checked(
554 std::make_shared<SymbolicExpr>(int_left), "t", zero,
555 LimitDirection::FromBelow, context);
556 if (!lim_left_result) {
557 return Result<SymbolicExpr>::failure(
558 lim_left_result.error());
559 }
560 auto lim_left = lim_left_result.value();
561 auto int_right_result = integrate_def_checked(
562 expr, var_name, *t, upper, context);
563 if (!int_right_result) return int_right_result;
564 auto int_right = int_right_result.value();
565 auto lim_right_result = limit_expression_checked(
566 std::make_shared<SymbolicExpr>(int_right), "t", zero,
567 LimitDirection::FromAbove, context);
568 if (!lim_right_result) {
569 return Result<SymbolicExpr>::failure(
570 lim_right_result.error());
571 }
572 auto lim_right = lim_right_result.value();
573 if (lim_left && lim_right) {
574 return Result<SymbolicExpr>::success(
575 *SymbolicExpr::add(lim_left, lim_right));
576 }
577 }
578 }
579 }
580
581 auto indefinite_result = integrate_checked(expr, var_name, context);
582 if (!indefinite_result) return indefinite_result;
583 SymbolicExpr indefinite = indefinite_result.value();
584
585 if (std::dynamic_pointer_cast<const IntegralNode>(
586 LMCAS::detail::node(indefinite))) {
587 return Result<SymbolicExpr>::success(
588 LMCAS::detail::expression_from_node(
589 LMCAS::detail::make_node<IntegralNode>(
590 LMCAS::detail::node(expr), var_name,
591 LMCAS::detail::node(lower),
592 LMCAS::detail::node(upper))));
593 }
594
595 try {
596 auto F_b = indefinite.substitute(var_name, make_expr_ptr(upper));
597 auto F_a = indefinite.substitute(var_name, make_expr_ptr(lower));
598 if (!F_b || !F_a) {
599 return Result<SymbolicExpr>::failure(
600 CasErrc::InternalInvariant,
601 "definite integration substitution produced no expression",
602 "integrate.definite.evaluate");
603 }
604 auto result = sym_sub(*F_b, *F_a);
605 auto simplified = result ? result->simplify() : nullptr;
606 if (!simplified) {
607 return Result<SymbolicExpr>::failure(
608 CasErrc::InternalInvariant,
609 "definite integration simplification produced no expression",
610 "integrate.definite.evaluate");
611 }
612 return Result<SymbolicExpr>::success(*simplified);
613 } catch (const std::exception& error) {
614 return Result<SymbolicExpr>::failure(
615 CasErrc::InternalInvariant, error.what(),
616 "integrate.definite.evaluate");
617 }
618}
