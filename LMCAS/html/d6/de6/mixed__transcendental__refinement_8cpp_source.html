29 std::optional<NumericRoot> bisect(
30 ComputationContext* context, std::optional<CasError>* failure,
31 bool* complete) {
32 if (std::isnan(f_lo_) || std::isnan(f_hi_) || f_lo_ * f_hi_ > 0.0) {
33 return root_at(x_, evaluate(x_), 1);
34 }
35 for (int i = 1; i <= opts_.max_newton_iterations; ++i) {
36 if (!consume_mixed_step(context, failure, complete)) {
37 return std::nullopt;
38 }
39 const auto mid = midpoint();
40 const auto f_mid = evaluate(mid);
41 if (std::isnan(f_mid)) {
42 hi_ = mid;
43 continue;
44 }
45 remember(mid, f_mid, i);
46 auto root = root_at(mid, f_mid, i);
47 if (root) {
48 return root;
49 }
50 if (f_lo_ * f_mid < 0.0) {
51 hi_ = mid;
52 } else {
53 lo_ = mid;
54 f_lo_ = f_mid;
55 }
56 if (hi_ - lo_ < opts_.tolerance) {
57 const auto final_x = midpoint();
58 remember(final_x, evaluate(final_x), i);
59 break;
60 }
61 }
62 return best_root();
63 }

65 std::optional<NumericRoot> newton(
66 ComputationContext* context, std::optional<CasError>* failure,
67 bool* complete) {
68 if (!resample_endpoints()) {
69 return std::nullopt;
70 }
71 for (int i = 1; i <= opts_.max_newton_iterations; ++i) {
72 if (!consume_mixed_step(context, failure, complete)) {
73 return std::nullopt;
74 }
75 const auto fx = evaluate(x_);
76 if (std::isnan(fx)) {
77 x_ = midpoint();
78 continue;
79 }
80 remember(x_, fx, i);
81 auto root = root_at(x_, fx, i);
82 if (root) {
83 return root;
84 }
85 const auto dfx = mixed_evaluate_at(derivative_, var_, x_);
86 if (std::isnan(dfx) || std::fabs(dfx) < 1e-15) {
87 bisection_step();
88 continue;
89 }
90 if (accept_newton_step(fx, dfx)) {
91 continue;
92 }
93 bisection_step();
94 if (hi_ - lo_ < opts_.tolerance) {
95 const auto final_x = midpoint();
96 root = root_at(final_x, evaluate(final_x), i);
97 if (root) {
98 return root;
99 }
100 break;
101 }
102 }
103 return best_root();
104 }

137 bool resample_endpoints() {
138 if (std::isnan(f_lo_)) {
139 f_lo_ = evaluate(lo_ + (hi_ - lo_) * 0.01);
140 lo_ = lo_ + (hi_ - lo_) * 0.01;
141 }
142 if (std::isnan(f_hi_)) {
143 f_hi_ = evaluate(hi_ - (hi_ - lo_) * 0.01);
144 hi_ = hi_ - (hi_ - lo_) * 0.01;
145 }
146 return !std::isnan(f_lo_) && !std::isnan(f_hi_);
147 }

149 void bisection_step() {
150 const auto mid = midpoint();
151 const auto f_mid = evaluate(mid);
152 if (!std::isnan(f_mid)) {
153 if (!std::isnan(f_lo_) && f_lo_ * f_mid < 0.0) {
154 hi_ = mid;
155 f_hi_ = f_mid;
156 } else {
157 lo_ = mid;
158 f_lo_ = f_mid;
159 }
160 }
161 x_ = midpoint();
162 }

164 bool accept_newton_step(lmmc_real_t fx, lmmc_real_t dfx) {
165 const auto next = x_ - fx / dfx;
166 if (next < lo_ || next > hi_) {
167 return false;
168 }
169 const auto f_next = evaluate(next);
170 if (std::isnan(f_next) || std::fabs(f_next) > std::fabs(fx)) {
171 return false;
172 }
173 if (!std::isnan(f_lo_) && fx * f_lo_ < 0.0) {
174 hi_ = x_;
175 f_hi_ = fx;
176 } else if (!std::isnan(f_hi_) && fx * f_hi_ < 0.0) {
177 lo_ = x_;
178 f_lo_ = fx;
179 }
180 x_ = next;
181 return true;
182 }
