47 * @brief Check if an operator participates in transitive closure.
48 * Only GT and GEQ form transitive chains.

50bool is_transitive_op(RelationOp op) {
51 return op == RelationOp::GT || op == RelationOp::GEQ;

54/**
55 * @brief Combine two transitive operators according to the combination rules:
56 * GT+GT→GT, GT+GEQ→GT, GEQ+GT→GT, GEQ+GEQ→GEQ.
57 */
58RelationOp combine_ops(RelationOp op1, RelationOp op2) {
59 if (op1 == RelationOp::GEQ && op2 == RelationOp::GEQ) {
60 return RelationOp::GEQ;
61 }
62 return RelationOp::GT;

74RelationStoreResult declare_comparison_sign(const std::string& symbol,
75 RelationOp op, PropertyStore& store) {
76 struct SignRule {
77 RelationOp op;
78 Sign sign;
79 };
80 static constexpr SignRule rules[] = {
81 {RelationOp::GT, Sign::Positive},
82 {RelationOp::GEQ, Sign::NonNegative},
83 {RelationOp::LT, Sign::Negative},
84 {RelationOp::LEQ, Sign::NonPositive},
85 {RelationOp::NEQ, Sign::NonZero}
86 };
87 for (const auto& rule : rules) {
88 if (rule.op == op) return store.declare_sign(symbol, rule.sign);
89 }
90 return RelationStoreResult::success();
91}

94RelationStoreResult derive_comparison_sign(const SymbolicExpr& lhs,
95 const SymbolicExpr& rhs,
96 RelationOp op, PropertyStore& store) {
97 const auto variable = std::dynamic_pointer_cast<const VariableNode>(LMCAS::detail::node(lhs));
98 const auto number = std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(rhs));
99 if (variable && !variable->is_constant() && number && number->is_zero()) {
100 return declare_comparison_sign(variable->name(), op, store);
101 }
102 const auto zero = std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(lhs));
103 const auto reversed_variable = std::dynamic_pointer_cast<const VariableNode>(LMCAS::detail::node(rhs));
104 if (zero && zero->is_zero() && reversed_variable &&
105 !reversed_variable->is_constant()) {
106 return declare_comparison_sign(
107 reversed_variable->name(), detail::reversed_relation(op), store);
108 }
109 return RelationStoreResult::success();
110}

114RelationStoreResult RelationStore::add_relation_unchecked(
115 const SymbolicExpr& lhs,
116 const SymbolicExpr& rhs,
117 RelationOp op,
118 PropertyStore& prop_store) {
119 auto index = find_relation(lhs, rhs, op);
120 if (!index) {
121 index = relations_.size();
122 relations_.push_back(Relation{lhs, rhs, op});
123 proofs_.push_back(RelationProofs{});
124 }
125 const bool already_declared = proofs_[*index].declared;
126 if (!already_declared) {
127 proofs_[*index].declared = true;
128 declaration_order_.push_back(*index);
129 }
130
131 if (!LMCAS::detail::node(lhs) || !LMCAS::detail::node(rhs)) {
132 return RelationStoreResult::success();
133 }
134
135 auto derived = derive_comparison_sign(lhs, rhs, op, prop_store);
136 if (!derived) return derived;
137
138 // Compute transitive closure for GT/GEQ relations
139 if (is_transitive_op(op)) {
140 auto closure = compute_transitive_closure(*index, prop_store);
141 if (!closure) return RelationStoreResult::failure(closure.error());
142 if (already_declared && closure.value()) {
143 // 重复声明可以推进受限闭包，序列化需重放这次操作。
144 declaration_order_.push_back(*index);
145 }
146 }
147 return RelationStoreResult::success();
148}

158RelationStoreResult RelationStore::add_relation_checked(
159 const SymbolicExpr& lhs,
160 const SymbolicExpr& rhs,
161 RelationOp op,
162 PropertyStore& prop_store) {
163 if (!LMCAS::detail::node(lhs)) {
164 return RelationStoreResult::failure(
165 CasErrc::InvalidArgument, "relation lhs must not be null", "add_relation");
166 }
167 if (!LMCAS::detail::node(rhs)) {
168 return RelationStoreResult::failure(
169 CasErrc::InvalidArgument, "relation rhs must not be null", "add_relation");
170 }
171
172 try {
173 RelationStore relation_candidate = *this;
174 PropertyStore property_candidate = prop_store;
175 auto inserted = relation_candidate.add_relation_unchecked(
176 lhs, rhs, op, property_candidate);
177 if (!inserted) return inserted;
178 *this = std::move(relation_candidate);
179 prop_store = std::move(property_candidate);
180 } catch (const std::bad_alloc&) {
181 return RelationStoreResult::failure(
182 CasErrc::ResourceLimit, "relation-store allocation failed", "add_relation");
183 } catch (const std::invalid_argument& ex) {
184 return RelationStoreResult::failure(CasErrc::InvalidArgument, ex.what(), "add_relation");
185 } catch (const std::exception& ex) {
186 return RelationStoreResult::failure(
187 CasErrc::InternalInvariant, ex.what(), "add_relation");
188 }
189
190 return RelationStoreResult::success();

198RelationStoreResult RelationStore::enqueue_transitive_deduction(
199 const Relation& deduced, Premises premises,
200 PropertyStore& prop_store, TransitiveWork& work) {
201 if (premises[1] < premises[0]) std::swap(premises[0], premises[1]);
202 if (const auto index = find_relation(deduced.lhs, deduced.rhs, deduced.op)) {
203 if (premises[0] == *index || premises[1] == *index) {
204 return RelationStoreResult::success();
205 }
206 auto& alternatives = proofs_[*index].alternatives;
207 if (std::find(alternatives.begin(), alternatives.end(), premises) ==
208 alternatives.end()) {
209 alternatives.push_back(premises);
210 work.changed = true;
211 }
212 return RelationStoreResult::success();
213 }
214 if (work.deductions == MAX_TRANSITIVE_DEDUCTIONS) {
215 return RelationStoreResult::success();
216 }
217 relations_.push_back(deduced);
218 proofs_.push_back(RelationProofs{false, {premises}});
219 ++work.deductions;
220 work.changed = true;
221 auto derived = derive_comparison_sign(
222 deduced.lhs, deduced.rhs, deduced.op, prop_store);
223 if (!derived) return derived;
224 work.queue.push(relations_.size() - 1);
225 return RelationStoreResult::success();
226}

228Result<bool> RelationStore::compute_transitive_closure(
229 std::size_t new_relation, PropertyStore& prop_store) {
230 TransitiveWork work;
231 work.queue.push(new_relation);
232 while (!work.queue.empty()) {
233 const auto current_index = work.queue.front();
234 const auto current = relations_[current_index];
235 work.queue.pop();
236 /**
237 * @brief 双向推导共用同一快照，先正向后反向。
238 * @note 插入可能使向量引用失效，跨插入仅保留索引或副本。
239 */
240 const std::size_t relation_count = relations_.size();
241 for (std::size_t i = 0; i < relation_count; ++i) {
242 if (!is_transitive_op(relations_[i].op)) continue;
243 if (!expr_equals(current.rhs, relations_[i].lhs)) continue;
244 auto result = enqueue_transitive_deduction(
245 {current.lhs, relations_[i].rhs, combine_ops(current.op, relations_[i].op)},
246 {current_index, i}, prop_store, work);
247 if (!result) return Result<bool>::failure(result.error());
248 }
249 for (std::size_t i = 0; i < relation_count; ++i) {
250 if (!is_transitive_op(relations_[i].op)) continue;
251 if (!expr_equals(relations_[i].rhs, current.lhs)) continue;
252 auto result = enqueue_transitive_deduction(
253 {relations_[i].lhs, current.rhs, combine_ops(relations_[i].op, current.op)},
254 {i, current_index}, prop_store, work);
255 if (!result) return Result<bool>::failure(result.error());
256 }
257 }
258 return Result<bool>::success(work.changed);
259}

265std::optional<std::size_t> RelationStore::find_relation(

267 for (std::size_t i = 0; i < relations_.size(); ++i) {
268 const auto& relation = relations_[i];

270 expr_equals(relation.rhs, rhs)) {

272 }
273 }
