67std::optional<int> integer_value_from_node(const std::shared_ptr<const SymbolicNode>& node)
68{
69 BigInt integer;
70 if (!try_get_integer_value(std::dynamic_pointer_cast<const NumberNode>(node), integer)) {
71 return std::nullopt;
72 }
73 auto value = integer.try_to_int64();
74 if (!value || *value < std::numeric_limits<int>::min() ||
75 *value > std::numeric_limits<int>::max()) {
76 return std::nullopt;
77 }
78 return static_cast<int>(*value);

81static std::optional<int> laurent_power_of_power(
82 const PowerNode& power, const std::string& var) {
83 auto exponent = integer_value_from_node(power.exponent());
84 if (!exponent) {
85 return std::nullopt;
86 }
87 auto base = std::dynamic_pointer_cast<const VariableNode>(power.base());
88 if (base && !base->is_constant() && base->name() == var) {
89 return *exponent;
90 }
91 auto nested = supported_laurent_integer_power(power.base(), var);
92 if (!nested) {
93 return std::nullopt;
94 }
95 const BigInt product = BigInt(*nested) * BigInt(*exponent);
96 if (product < BigInt(std::numeric_limits<int>::min()) ||
97 product > BigInt(std::numeric_limits<int>::max())) {
98 return std::nullopt;
99 }
100 return static_cast<int>(*product.try_to_int64());
101}

103std::optional<int> supported_laurent_integer_power(
104 const std::shared_ptr<const SymbolicNode>& node,
105 const std::string& var)
106{
107 if (!node) {
108 return std::nullopt;
109 }
110 if (std::dynamic_pointer_cast<const NumberNode>(node)) {
111 return 0;
112 }
113 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(node)) {
114 return !variable->is_constant() && variable->name() == var
115 ? std::optional<int>(1) : std::nullopt;
116 }
117 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
118 return laurent_power_of_power(*power, var);
119 }
120 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
121 BigInt total_power(0);
122 for (const auto& operand : multiply->operands()) {
123 if (std::dynamic_pointer_cast<const NumberNode>(operand)) {
124 continue;
125 }
126 auto power = supported_laurent_integer_power(operand, var);
127 if (!power) {
128 return std::nullopt;
129 }
130 total_power += BigInt(*power);
131 }
132 if (total_power < BigInt(std::numeric_limits<int>::min()) ||
133 total_power > BigInt(std::numeric_limits<int>::max())) {
134 return std::nullopt;
135 }
136 return static_cast<int>(*total_power.try_to_int64());
137 }
138 return std::nullopt;

162std::vector<std::shared_ptr<SymbolicExpr>> power_series_add(
163 const std::vector<std::shared_ptr<SymbolicExpr>>& a,
164 const std::vector<std::shared_ptr<SymbolicExpr>>& b) {
165 size_t len = std::max(a.size(), b.size());
166 std::vector<std::shared_ptr<SymbolicExpr>> result(len);
167 for (size_t i = 0; i < len; ++i) {
168 auto ai = (i < a.size() && a[i]) ? a[i] : SymbolicExpr::number(0);
169 auto bi = (i < b.size() && b[i]) ? b[i] : SymbolicExpr::number(0);
170 result[i] = SymbolicExpr::add(ai, bi)->simplify();
171 }
172 return result;
173}

175PowerSeriesResult power_series_multiply_checked(
176 const std::vector<std::shared_ptr<SymbolicExpr>>& a,
177 const std::vector<std::shared_ptr<SymbolicExpr>>& b, int order,
178 ComputationContext& context) {
179 const std::string operation = "power_series_multiply";
180 auto order_check = validate_power_series_order(order, context, operation);
181 if (!order_check) {
182 return PowerSeriesResult::failure(order_check.error());
183 }
184 auto a_check = validate_power_series_coefficients(a, operation, "left series");
185 if (!a_check) {
186 return PowerSeriesResult::failure(a_check.error());
187 }
188 auto b_check = validate_power_series_coefficients(b, operation, "right series");
189 if (!b_check) {
190 return PowerSeriesResult::failure(b_check.error());
191 }
192
193 size_t n = static_cast<size_t>(order);
194 std::vector<std::shared_ptr<SymbolicExpr>> result(n, SymbolicExpr::number(0));
195 for (size_t k = 0; k < n; ++k) {
196 auto step = context.consume_steps(1, operation);
197 if (!step) {
198 return PowerSeriesResult::failure(step.error());
199 }
200 std::vector<std::shared_ptr<const SymbolicNode>> terms;
201 for (size_t j = 0; j <= k; ++j) {
202 if (j >= a.size() || (k-j) >= b.size()) {
203 continue;
204 }
205 auto aj = a[j] ? a[j] : SymbolicExpr::number(0);
206 auto bkj = b[k-j] ? b[k-j] : SymbolicExpr::number(0);
207 terms.push_back(LMCAS::detail::node(SymbolicExpr::multiply(aj, bkj)));
208 }
209 if (terms.empty()) {
210 result[k] = SymbolicExpr::number(0);
211 } else if (terms.size() == 1) {
212 result[k] = LMCAS::detail::make_expression_ptr(terms[0])->simplify();
213 } else {
214 result[k] = LMCAS::detail::make_expression_ptr(
215 LMCAS::detail::make_node<AddNode>(terms))->simplify();
216 }
217 }
218 return PowerSeriesResult::success(std::move(result));
219}

229static Result<void> composition_inner_powers(
230 const std::vector<std::shared_ptr<SymbolicExpr>>& g,
231 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>>& gp,
232 int order, ComputationContext& context, const std::string& operation) {
233 const size_t n = gp.size();
234 gp[0].resize(n, SymbolicExpr::number(0));
235 gp[0][0] = SymbolicExpr::number(1);
236 if (n > 1) {
237 gp[1].resize(n, SymbolicExpr::number(0));
238 for (size_t i = 0; i < std::min(g.size(), n); ++i) {
239 gp[1][i] = g[i] ? g[i] : SymbolicExpr::number(0);
240 }
241 }
242 for (size_t k = 2; k < n; ++k) {
243 auto step = context.consume_steps(1, operation);
244 if (!step) {
245 return step;
246 }
247 auto multiplied = power_series_multiply_checked(gp[k-1], gp[1], order, context);
248 if (!multiplied) {
249 return Result<void>::failure(multiplied.error());
250 }
251 gp[k] = std::move(multiplied.value());
252 }
253 return Result<void>::success();
254}

256PowerSeriesResult power_series_compose_checked(
257 const std::vector<std::shared_ptr<SymbolicExpr>>& f,
258 const std::vector<std::shared_ptr<SymbolicExpr>>& g, int order,
259 ComputationContext& context) {
260 const std::string operation = "power_series_compose";
261 auto order_check = validate_power_series_order(order, context, operation);
262 if (!order_check) {
263 return PowerSeriesResult::failure(order_check.error());
264 }
265 auto f_check = validate_power_series_coefficients(f, operation, "outer series");
266 if (!f_check) {
267 return PowerSeriesResult::failure(f_check.error());
268 }
269 auto g_check = validate_power_series_coefficients(g, operation, "inner series");
270 if (!g_check) {
271 return PowerSeriesResult::failure(g_check.error());
272 }
273
274 size_t n = static_cast<size_t>(order);
275 if (!g.empty() && !g[0]->is_zero()) {
276 return PowerSeriesResult::failure(
277 CasErrc::DomainError,
278 "power series composition around zero requires g(0) = 0",
279 operation);
280 }
281 std::vector<std::shared_ptr<SymbolicExpr>> result(n, SymbolicExpr::number(0));
282 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> gp(n);
283 auto powers = composition_inner_powers(g, gp, order, context, operation);
284 if (!powers) {
285 return PowerSeriesResult::failure(powers.error());
286 }
287 for (size_t k = 0; k < std::min(f.size(), n); ++k) {
288 auto step = context.consume_steps(1, operation);
289 if (!step) {
290 return PowerSeriesResult::failure(step.error());
291 }
292 auto fk = f[k] ? f[k] : SymbolicExpr::number(0);
293 if (fk->is_zero()) {
294 continue;
295 }
296 for (size_t i = 0; i < n && i < gp[k].size(); ++i)
297 result[i] = SymbolicExpr::add(result[i], SymbolicExpr::multiply(fk, gp[k][i]))->simplify();
298 }
299 return PowerSeriesResult::success(std::move(result));
300}

311std::shared_ptr<SymbolicExpr> symbolic_sum(
312 const std::shared_ptr<SymbolicExpr>& body, const std::string& index,
313 const std::shared_ptr<SymbolicExpr>& lower, const std::shared_ptr<SymbolicExpr>& upper) {
314 if (!body || !lower || !upper) {
315 return nullptr;
316 }
317 BigInt lv, uv;
318 if (try_get_integer_value(
319 std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(lower)), lv) &&
320 try_get_integer_value(
321 std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(upper)), uv)) {
322 if (uv < lv) {
323 return SymbolicExpr::number(0);
324 }
325 const BigInt span = uv - lv;
326 if (span < BigInt(100)) {
327 const auto iterations = *span.try_to_uint64() + 1;
328 std::vector<std::shared_ptr<const SymbolicNode>> terms;
329 BigInt k = lv;
330 const BigInt increment(1);
331 for (std::uint64_t iteration = 0; iteration < iterations; ++iteration, k += increment) {
332 auto val = SymbolicExpr::number(k);
333 auto term = body->substitute(index, val);
334 if (term) { term = term->simplify(); terms.push_back(LMCAS::detail::node(term)); }
335 }
336 if (terms.empty()) {
337 return SymbolicExpr::number(0);
338 }
339 if (terms.size() == 1) {
340 return LMCAS::detail::make_expression_ptr(terms[0])->simplify();
341 }
342 return LMCAS::detail::make_expression_ptr(LMCAS::detail::make_node<AddNode>(terms))->simplify();
343 }
344 }
345 auto k_var = SymbolicExpr::variable(index);
346 if (body->to_string() == k_var->to_string()) {
347 auto u = upper; auto l = lower;
348 auto u1 = SymbolicExpr::add(u, SymbolicExpr::number(1));
349 auto su = SymbolicExpr::divide(SymbolicExpr::multiply(u, u1), SymbolicExpr::number(2));
350 auto lm1 = SymbolicExpr::add(l, SymbolicExpr::number(-1));
351 auto sl = SymbolicExpr::divide(SymbolicExpr::multiply(lm1, l), SymbolicExpr::number(2));
352 return SymbolicExpr::add(su, SymbolicExpr::multiply(SymbolicExpr::number(-1), sl))->simplify();
353 }
354 return LMCAS::detail::make_expression_ptr(LMCAS::detail::make_node<SummationNode>(LMCAS::detail::node(body), index, LMCAS::detail::node(lower), LMCAS::detail::node(upper)));
355}

357std::shared_ptr<SymbolicExpr> symbolic_product(
358 const std::shared_ptr<SymbolicExpr>& body, const std::string& index,
359 const std::shared_ptr<SymbolicExpr>& lower, const std::shared_ptr<SymbolicExpr>& upper) {
360 if (!body || !lower || !upper) {
361 return nullptr;
362 }
363 BigInt lv, uv;
364 if (try_get_integer_value(
365 std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(lower)), lv) &&
366 try_get_integer_value(
367 std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(upper)), uv)) {
368 if (uv < lv) {
369 return SymbolicExpr::number(1);
370 }
371 const BigInt span = uv - lv;
372 if (span < BigInt(50)) {
373 const auto iterations = *span.try_to_uint64() + 1;
374 auto result = SymbolicExpr::number(1);
375 BigInt k = lv;
376 const BigInt increment(1);
377 for (std::uint64_t iteration = 0; iteration < iterations; ++iteration, k += increment) {
378 auto val = SymbolicExpr::number(k);
379 auto term = body->substitute(index, val);
380 if (term) { term = term->simplify(); result = SymbolicExpr::multiply(result, term); }
381 }
382 return result->simplify();
383 }
384 }
385 return LMCAS::detail::make_expression_ptr(LMCAS::detail::make_node<ProductNode>(LMCAS::detail::node(body), index, LMCAS::detail::node(lower), LMCAS::detail::node(upper)));
386}
