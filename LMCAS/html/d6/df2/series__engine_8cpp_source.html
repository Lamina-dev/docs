48static bool series_extract_alternating(const std::shared_ptr<const SymbolicNode>& node,
49 const std::string& n,
50 std::shared_ptr<SymbolicExpr>& remainder) {
51 auto pow = std::dynamic_pointer_cast<const PowerNode>(node);
52 if (pow) {
53 auto base_num = std::dynamic_pointer_cast<const NumberNode>(pow->base());
54 auto exp_var = std::dynamic_pointer_cast<const VariableNode>(pow->exponent());
55 if (base_num && exp_var && exp_var->name() == n) {
56 double base_val = 0.0;
57 if (std::holds_alternative<BigInt>(base_num->value()))
58 base_val = std::get<BigInt>(base_num->value()).to_double();
59 else if (std::holds_alternative<Rational>(base_num->value()))
60 base_val = std::get<Rational>(base_num->value()).to_double();
61 else
62 base_val = static_cast<double>(std::get<lmmc_real_t>(base_num->value()));
63 if (std::abs(base_val + 1.0) < 1e-12) {
64 remainder = SymbolicExpr::number(1);
65 return true;
66 }
67 }
68 }
69 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node);
70 if (mul) {
71 for (size_t i = 0; i < mul->operands().size(); ++i) {
72 auto pw = std::dynamic_pointer_cast<const PowerNode>(mul->operands()[i]);
73 if (!pw) continue;
74 auto base_num = std::dynamic_pointer_cast<const NumberNode>(pw->base());
75 auto exp_var = std::dynamic_pointer_cast<const VariableNode>(pw->exponent());
76 if (!base_num || !exp_var || exp_var->name() != n) continue;
77 double base_val = 0.0;
78 if (std::holds_alternative<BigInt>(base_num->value()))
79 base_val = std::get<BigInt>(base_num->value()).to_double();
80 else if (std::holds_alternative<Rational>(base_num->value()))
81 base_val = std::get<Rational>(base_num->value()).to_double();
82 else
83 base_val = static_cast<double>(std::get<lmmc_real_t>(base_num->value()));
84 if (std::abs(base_val + 1.0) < 1e-12) {
85 std::vector<std::shared_ptr<const SymbolicNode>> rest;
86 for (size_t j = 0; j < mul->operands().size(); ++j)
87 if (j != i) rest.push_back(mul->operands()[j]);
88 if (rest.empty()) remainder = SymbolicExpr::number(1);
89 else if (rest.size() == 1) remainder = lamina::detail::make_expression_ptr(rest[0]);
90 else remainder = lamina::detail::make_expression_ptr(lamina::detail::make_node<MultiplyNode>(rest));
91 return true;
92 }
93 }
94 }
95 return false;
96}

98static bool series_detect_trig_oscillation(const std::shared_ptr<const SymbolicNode>& node,
99 const std::string& n,
100 std::shared_ptr<SymbolicExpr>& amplitude) {
101 auto func = std::dynamic_pointer_cast<const FunctionNode>(node);
102 if (func && (func->type() == FunctionNode::FuncType::Sin || func->type() == FunctionNode::FuncType::Cos)) {
103 if (!func->arguments().empty()) {
104 auto arg_expr = lamina::detail::make_expression_ptr(func->arguments()[0]);
105 auto arg_limit = arg_expr->limit(n, SymbolicExpr::infinity());
106 if (series_is_infinity(arg_limit)) { amplitude = SymbolicExpr::number(1); return true; }
107 }
108 }
109 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node);
110 if (!mul) return false;
111 for (size_t i = 0; i < mul->operands().size(); ++i) {
112 auto f = std::dynamic_pointer_cast<const FunctionNode>(mul->operands()[i]);
113 if (!f || (f->type() != FunctionNode::FuncType::Sin && f->type() != FunctionNode::FuncType::Cos)) continue;
114 if (f->arguments().empty()) continue;
115 auto arg_expr = lamina::detail::make_expression_ptr(f->arguments()[0]);
116 auto arg_limit = arg_expr->limit(n, SymbolicExpr::infinity());
117 if (!series_is_infinity(arg_limit)) continue;
118 std::vector<std::shared_ptr<const SymbolicNode>> rest;
119 for (size_t j = 0; j < mul->operands().size(); ++j)
120 if (j != i) rest.push_back(mul->operands()[j]);
121 if (rest.empty()) amplitude = SymbolicExpr::number(1);
122 else if (rest.size() == 1) amplitude = lamina::detail::make_expression_ptr(rest[0]);
123 else amplitude = lamina::detail::make_expression_ptr(lamina::detail::make_node<MultiplyNode>(rest));
124 return true;
125 }
126 return false;
127}

141static bool try_get_int(const std::shared_ptr<SymbolicExpr>& expr, long long& out) {
142 if (!expr || !lamina::detail::node(expr)) return false;
143 auto num = std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(expr));
144 if (!num) return false;
145 if (std::holds_alternative<BigInt>(num->value())) {
146 out = static_cast<long long>(std::get<BigInt>(num->value()).to_int());
147 return true;
148 }
149 if (std::holds_alternative<Rational>(num->value())) {
150 auto& r = std::get<Rational>(num->value());
151 if (r.get_denominator() == BigInt(1)) {
152 out = static_cast<long long>(r.get_numerator().to_int());
153 return true;
154 }
155 return false;
156 }
157 double d = static_cast<double>(std::get<lmmc_real_t>(num->value()));
158 if (d == std::floor(d) && std::abs(d) < 1e15) { out = static_cast<long long>(d); return true; }
159 return false;
160}

164static int detect_parity(const std::shared_ptr<SymbolicExpr>& f, const std::string& var) {
165 if (!f) return 0;
166 auto neg_x = SymbolicExpr::multiply(SymbolicExpr::number(-1), SymbolicExpr::variable(var));
167 auto f_neg = f->substitute(var, neg_x);
168 if (!f_neg) return 0;
169 f_neg = f_neg->simplify();
170 auto f_s = f->simplify();
171 auto diff_even = SymbolicExpr::add(f_neg, SymbolicExpr::multiply(SymbolicExpr::number(-1), f_s));
172 if (diff_even) { diff_even = diff_even->simplify(); if (diff_even->is_zero()) return 1; }
173 auto diff_odd = SymbolicExpr::add(f_neg, f_s);
174 if (diff_odd) { diff_odd = diff_odd->simplify(); if (diff_odd->is_zero()) return -1; }
175 return 0;
176}

267std::optional<int> integer_value_from_node(const std::shared_ptr<const SymbolicNode>& node)
268{
269 auto number = std::dynamic_pointer_cast<const NumberNode>(node);
270 if (!number) return std::nullopt;
271 if (std::holds_alternative<BigInt>(number->value())) {
272 return std::get<BigInt>(number->value()).to_int();
273 }
274 if (std::holds_alternative<Rational>(number->value())) {
275 const auto& rational = std::get<Rational>(number->value());
276 if (rational.get_denominator() != BigInt(1)) return std::nullopt;
277 return rational.get_numerator().to_int();
278 }
279 double value = static_cast<double>(std::get<lmmc_real_t>(number->value()));
280 if (!std::isfinite(value) || value != std::floor(value) ||
281 std::abs(value) > static_cast<double>(std::numeric_limits<int>::max())) {
282 return std::nullopt;
283 }
284 return static_cast<int>(value);

287std::optional<int> supported_laurent_integer_power(
288 const std::shared_ptr<const SymbolicNode>& node,
289 const std::string& var)
290{
291 if (!node) return std::nullopt;
292 if (std::dynamic_pointer_cast<const NumberNode>(node)) return 0;
293 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(node)) {
294 return variable->name() == var ? std::optional<int>(1) : std::nullopt;
295 }
296 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
297 auto base = std::dynamic_pointer_cast<const VariableNode>(power->base());
298 if (!base || base->name() != var) return std::nullopt;
299 return integer_value_from_node(power->exponent());
300 }
301 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
302 int total_power = 0;
303 for (const auto& operand : multiply->operands()) {
304 if (std::dynamic_pointer_cast<const NumberNode>(operand)) continue;
305 auto power = supported_laurent_integer_power(operand, var);
306 if (!power) return std::nullopt;
307 total_power += *power;
308 }
309 return total_power;
310 }
311 return std::nullopt;
312}

368std::shared_ptr<SymbolicExpr> convergence_radius(
369 const std::vector<std::shared_ptr<SymbolicExpr>>& coefficients, const std::string& var) {
370 (void)var;
371 if (coefficients.size() < 2) return SymbolicExpr::infinity();
372 std::vector<size_t> nz;
373 for (size_t i = 0; i < coefficients.size(); ++i)
374 if (coefficients[i] && !coefficients[i]->is_zero()) nz.push_back(i);
375 if (nz.size() < 2) return SymbolicExpr::infinity();
376 std::vector<double> ratios;
377 for (size_t i = 0; i + 1 < nz.size(); ++i) {
378 auto a = coefficients[nz[i]]; auto b = coefficients[nz[i+1]];
379 if (!a || !b || b->is_zero()) continue;
380 if (series_is_number(a) && series_is_number(b)) {
381 double va = std::abs(series_get_double(a)), vb = std::abs(series_get_double(b));
382 if (vb > 1e-300) ratios.push_back(va / vb);
383 }
384 }
385 if (!ratios.empty()) {
386 double r = ratios.back();
387 if (std::isinf(r) || r > 1e15) return SymbolicExpr::infinity();
388 return SymbolicExpr::number(r);
389 }
390 return SymbolicExpr::infinity();
391}

436ConvergenceInfo convergence_test(
437 const std::shared_ptr<SymbolicExpr>& general_term, const std::string& index_var) {
438 if (!general_term || !lamina::detail::node(general_term)) return {ConvergenceResult::Inconclusive, ""};
439 if (auto power = std::dynamic_pointer_cast<const PowerNode>(lamina::detail::node(general_term))) {
440 auto base_var = std::dynamic_pointer_cast<const VariableNode>(power->base());
441 auto exponent = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
442 if (base_var && base_var->name() == index_var && exponent) {
443 double p = 0.0;
444 if (std::holds_alternative<BigInt>(exponent->value())) {
445 p = std::get<BigInt>(exponent->value()).to_double();
446 } else if (std::holds_alternative<Rational>(exponent->value())) {
447 p = std::get<Rational>(exponent->value()).to_double();
448 } else {
449 p = static_cast<double>(std::get<lmmc_real_t>(exponent->value()));
450 }
451 if (p < -1.0) return {ConvergenceResult::Convergent, "p-series"};
452 if (p >= -1.0 && p < 0.0) return {ConvergenceResult::Divergent, "p-series"};
453 }
454 }
455 auto n = SymbolicExpr::variable(index_var);
456 auto n1 = SymbolicExpr::add(n, SymbolicExpr::number(1));
457 auto inf = SymbolicExpr::infinity();
458 auto a_n1 = general_term->substitute(index_var, n1);
459 if (a_n1) {
460 a_n1 = a_n1->simplify();
461 auto ratio = SymbolicExpr::divide(a_n1, general_term);
462 if (ratio) {
463 ratio = ratio->simplify();
464 auto abs_r = lamina::detail::make_expression_ptr(lamina::detail::make_node<FunctionNode>(
465 FunctionNode::FuncType::Abs, std::vector<std::shared_ptr<const SymbolicNode>>{lamina::detail::node(ratio)}));
466 abs_r = abs_r->simplify();
467 auto lim = abs_r->limit(index_var, inf);
468 if (lim) {
469 auto ls = lim->simplify();
470 if (ls && series_is_number(ls)) {
471 double v = series_get_double(ls);
472 if (v < 1.0 - 1e-12) return {ConvergenceResult::Convergent, "ratio"};
473 if (v > 1.0 + 1e-12) return {ConvergenceResult::Divergent, "ratio"};
474 }
475 if (ls && ls->is_zero()) return {ConvergenceResult::Convergent, "ratio"};
476 if (series_is_infinity(ls)) return {ConvergenceResult::Divergent, "ratio"};
477 }
478 }
479 }
480 return {ConvergenceResult::Inconclusive, ""};
481}

483std::vector<std::shared_ptr<SymbolicExpr>> power_series_add(
484 const std::vector<std::shared_ptr<SymbolicExpr>>& a,
485 const std::vector<std::shared_ptr<SymbolicExpr>>& b) {
486 size_t len = std::max(a.size(), b.size());
487 std::vector<std::shared_ptr<SymbolicExpr>> result(len);
488 for (size_t i = 0; i < len; ++i) {
489 auto ai = (i < a.size() && a[i]) ? a[i] : SymbolicExpr::number(0);
490 auto bi = (i < b.size() && b[i]) ? b[i] : SymbolicExpr::number(0);
491 result[i] = SymbolicExpr::add(ai, bi)->simplify();
492 }
493 return result;
494}

496PowerSeriesResult power_series_multiply_checked(
497 const std::vector<std::shared_ptr<SymbolicExpr>>& a,
498 const std::vector<std::shared_ptr<SymbolicExpr>>& b, int order,
499 ComputationContext& context) {
500 const std::string operation = "power_series_multiply";
501 auto order_check = validate_power_series_order(order, context, operation);
502 if (!order_check) return PowerSeriesResult::failure(order_check.error());
503 auto a_check = validate_power_series_coefficients(a, operation, "left series");
504 if (!a_check) return PowerSeriesResult::failure(a_check.error());
505 auto b_check = validate_power_series_coefficients(b, operation, "right series");
506 if (!b_check) return PowerSeriesResult::failure(b_check.error());
507
508 size_t n = static_cast<size_t>(order);
509 std::vector<std::shared_ptr<SymbolicExpr>> result(n, SymbolicExpr::number(0));
510 for (size_t k = 0; k < n; ++k) {
511 auto step = context.consume_steps(1, operation);
512 if (!step) return PowerSeriesResult::failure(step.error());
513 std::vector<std::shared_ptr<const SymbolicNode>> terms;
514 for (size_t j = 0; j <= k; ++j) {
515 if (j >= a.size() || (k-j) >= b.size()) continue;
516 auto aj = a[j] ? a[j] : SymbolicExpr::number(0);
517 auto bkj = b[k-j] ? b[k-j] : SymbolicExpr::number(0);
518 terms.push_back(lamina::detail::node(SymbolicExpr::multiply(aj, bkj)));
519 }
520 if (terms.empty()) result[k] = SymbolicExpr::number(0);
521 else if (terms.size() == 1) result[k] = lamina::detail::make_expression_ptr(terms[0])->simplify();
522 else result[k] = lamina::detail::make_expression_ptr(lamina::detail::make_node<AddNode>(terms))->simplify();
523 }
524 return PowerSeriesResult::success(std::move(result));
525}

534std::vector<std::shared_ptr<SymbolicExpr>> power_series_multiply(
535 const std::vector<std::shared_ptr<SymbolicExpr>>& a,
536 const std::vector<std::shared_ptr<SymbolicExpr>>& b, int order) {
537 if (order <= 0) return {};
538 size_t n = static_cast<size_t>(order);
539 std::vector<std::shared_ptr<SymbolicExpr>> result(n, SymbolicExpr::number(0));
540 for (size_t k = 0; k < n; ++k) {
541 std::vector<std::shared_ptr<const SymbolicNode>> terms;
542 for (size_t j = 0; j <= k; ++j) {
543 if (j >= a.size() || (k-j) >= b.size()) continue;
544 auto aj = a[j] ? a[j] : SymbolicExpr::number(0);
545 auto bkj = b[k-j] ? b[k-j] : SymbolicExpr::number(0);

623std::shared_ptr<SymbolicExpr> fourier_series(
624 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
625 const std::shared_ptr<SymbolicExpr>& period, int n_terms) {
626 if (!f || !period || n_terms < 0) return nullptr;
627
628 /// 周期 T，半周期 L = T/2，基频 w = 2π/T
629 auto T = period;
630 auto pi = lamina::detail::make_expression_ptr(lamina::detail::make_node<VariableNode>("pi"));
631 auto two = SymbolicExpr::number(2);
632 auto L = SymbolicExpr::divide(T, two); // 半周期
633 auto half_lo = SymbolicExpr::multiply(SymbolicExpr::number(-1), L);
634 auto half_hi = L;
635 auto w = SymbolicExpr::divide(SymbolicExpr::multiply(two, pi), T); // 2π/T
636
637 int parity = detect_parity(f, var); // 1=even, -1=odd, 0=unknown
638
639 Integrator integrator;
640 auto x = SymbolicExpr::variable(var);
641
642 /// a0 = (1/L) ∫_{-L}^{L} f dx ; 常数项 a0/2
643 std::shared_ptr<SymbolicExpr> a0 = SymbolicExpr::number(0);
644 {
645 auto integ = integrator.integrate_def(*f, var, *half_lo, *half_hi);
646 a0 = SymbolicExpr::divide(lamina::detail::make_expression_ptr(lamina::detail::node(integ)), L)->simplify();
647 }
648 auto result = SymbolicExpr::divide(a0, two); // a0/2
649
650 for (int k = 1; k <= n_terms; ++k) {
651 auto kw = SymbolicExpr::multiply(SymbolicExpr::number(k), w); // k·2π/T
652 auto arg = SymbolicExpr::multiply(kw, x);
653
654 /// a_k：奇函数时为 0
655 if (parity != -1) {
656 auto integrand = SymbolicExpr::multiply(f, SymbolicExpr::cos(arg));
657 auto integ = integrator.integrate_def(*integrand, var, *half_lo, *half_hi);
658 auto ak = SymbolicExpr::divide(lamina::detail::make_expression_ptr(lamina::detail::node(integ)), L)->simplify();
659 if (!(lamina::detail::node(ak) && lamina::detail::node(ak)->is_zero())) {
660 result = SymbolicExpr::add(result, SymbolicExpr::multiply(ak, SymbolicExpr::cos(arg)));
661 }
662 }
663 /// b_k：偶函数时为 0
664 if (parity != 1) {
665 auto integrand = SymbolicExpr::multiply(f, SymbolicExpr::sin(arg));
666 auto integ = integrator.integrate_def(*integrand, var, *half_lo, *half_hi);
667 auto bk = SymbolicExpr::divide(lamina::detail::make_expression_ptr(lamina::detail::node(integ)), L)->simplify();
668 if (!(lamina::detail::node(bk) && lamina::detail::node(bk)->is_zero())) {
669 result = SymbolicExpr::add(result, SymbolicExpr::multiply(bk, SymbolicExpr::sin(arg)));
670 }
671 }
672 }
673 return result->simplify();
674}

806LaurentResult laurent_series_full(
807 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
808 const std::shared_ptr<SymbolicExpr>& center, int order_neg, int order_pos) {
809 LaurentResult res{nullptr, SingularityType::Removable, 0, SymbolicExpr::number(0)};
810 if (!f || !center) return res;
811
812 auto x = SymbolicExpr::variable(var);
813 /// (x - center)
814 auto shift = SymbolicExpr::add(x, SymbolicExpr::multiply(SymbolicExpr::number(-1), center));
815
816 /// 寻找极点阶数 m：使 (x-c)^m · f 在 c 处解析（有限）。
817 int pole_order = 0;
818 std::shared_ptr<SymbolicExpr> regular = f;
819 {
820 auto test = f;
821 const int MAX_M = (order_neg > 0 ? order_neg : 5);
822 /// 先判断 f 在 center 是否已解析
823 auto f_at = f->substitute(var, center);
824 if (f_at) f_at = f_at->simplify();
825 bool finite = f_at && f_at->is_number();
826 if (!finite) {
827 for (int m = 1; m <= MAX_M; ++m) {
828 auto powm = SymbolicExpr::power(shift, SymbolicExpr::number(m));
829 auto g = SymbolicExpr::multiply(powm, f)->simplify();
830 auto g_at = g->substitute(var, center);
831 if (g_at) g_at = g_at->simplify();
832 if (g_at && g_at->is_number()) {
833 pole_order = m;
834 regular = g; // (x-c)^m f(x)，在 c 处解析
835 break;
836 }
837 }
838 }
839 }
840
841 /// 对 regular（解析部分）做 Taylor 展开
842 int taylor_order = order_pos + pole_order + 1;
843 if (taylor_order < 1) taylor_order = 1;
844 auto taylor = regular->series(var, center, taylor_order);
845 if (!taylor) return res;
846
847 /// Laurent = taylor / (x-c)^pole_order
848 std::shared_ptr<SymbolicExpr> laurent = taylor;
849 if (pole_order > 0) {
850 auto powm = SymbolicExpr::power(shift, SymbolicExpr::number(pole_order));
851 laurent = SymbolicExpr::divide(taylor, powm)->simplify();
852 }
853
854 res.series = laurent->simplify();
855 res.pole_order = pole_order;
856 if (pole_order == 0) res.singularity = SingularityType::Removable;
857 else res.singularity = SingularityType::Pole;
858
859 /// 留数 = Taylor 展开中 (x-c)^(pole_order-1) 的系数
860 if (pole_order >= 1) {
861 /// a_{m-1} = (1/(m-1)!) lim_{x->c} d^{m-1}/dx^{m-1} [regular]
862 auto deriv = regular;
863 for (int i = 0; i < pole_order - 1; ++i) deriv = deriv->differentiate(var);
864 /// 用极限求值，稳健处理 0/0 形式（如 z/(z(z+1)) 在 z=0）。
865 auto val = deriv->limit(var, center);
866 if (!val) val = deriv->substitute(var, center);
867 long long fact = 1;
868 for (int i = 2; i <= pole_order - 1; ++i) fact *= i;
869 res.residue = SymbolicExpr::divide(val, SymbolicExpr::number((int)fact))->simplify();
870 }
871
872 return res;
873}

876 const std::shared_ptr<SymbolicExpr>& f, const std::string& var, int order) {
877 if (!f || order < 0) return nullptr;
878 /// 通过 x = 1/t 替换，在 t=0 处做 Taylor 展开，再回代 t = 1/x，
879 /// 得到按 x 递减幂次的渐近展开。
880 auto x = SymbolicExpr::variable(var);

882 auto t = SymbolicExpr::variable(tname);

884 auto g = f->substitute(var, inv_t);
885 if (!g) return nullptr;
886 g = g->simplify();
887 auto taylor_t = g->series(tname, SymbolicExpr::number(0), order + 1);
888 if (!taylor_t) return nullptr;

890 auto inv_x = SymbolicExpr::divide(SymbolicExpr::number(1), x);
891 auto back = taylor_t->substitute(tname, inv_x);
892 if (!back) return nullptr;
893 return back->simplify();
