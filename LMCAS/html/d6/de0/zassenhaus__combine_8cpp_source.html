24/**
25 * @brief 将 BigInt 系数归约到对称表示 [-m/2, m/2).
26 *
27 * @param[in] c 待归约的系数
28 * @param[in] m 模数(正整数)
29 * @return 对称表示下的归约值
30 * @internal
31 */
32static BigInt zc_symmetric_mod(const BigInt& c, const BigInt& m) {
33 if (m.is_zero()) return c;
34
35 BigInt r = c % m;
36 if (r.IsNegative()) {
37 r = r + m;
38 }
39
40 BigInt half_m = m / BigInt(2);
41 if (r > half_m) {
42 r = r - m;
43 }
44 return r;
45}

47/**
48 * 在模 m 下乘以按字典序枚举的提升因子子集.
49 */
50static Result<std::vector<BigInt>> zc_subset_product(
51 const std::vector<Polynomial<BigInt>>& lifted_factors,
52 const std::vector<size_t>& indices,
53 const BigInt& mod,
54 ComputationContext& context) {
55 std::vector<BigInt> product = {BigInt(1)};
56
57 for (size_t index : indices) {
58 auto step = context.consume_steps(1, "zassenhaus_combine");
59 if (!step) {
60 return Result<std::vector<BigInt>>::failure(step.error());
61 }
62 if (index >= lifted_factors.size() ||
63 lifted_factors[index].coeffs.empty()) {
64 return Result<std::vector<BigInt>>::failure(
65 CasErrc::InvalidArgument, "invalid lifted-factor index or zero factor",
66 "zassenhaus_combine");
67 }
68 const auto& factor_coeffs = lifted_factors[index].coeffs;
69 std::vector<BigInt> next(product.size() + factor_coeffs.size() - 1,
70 BigInt(0));
71 for (size_t i = 0; i < product.size(); ++i) {
72 if (product[i].is_zero()) continue;
73 for (size_t j = 0; j < factor_coeffs.size(); ++j) {
74 if (factor_coeffs[j].is_zero()) continue;
75 next[i + j] = next[i + j] + product[i] * factor_coeffs[j];
76 }
77 }
78 for (auto& coefficient : next) {
79 coefficient = zc_symmetric_mod(coefficient, mod);
80 }
81 while (!next.empty() && next.back().is_zero()) next.pop_back();
82 product = std::move(next);
83 }
84 return Result<std::vector<BigInt>>::success(std::move(product));
85}

99 const std::string& var) {
100
101 std::vector<Rational> rat_coeffs;

109/**
110 * @brief 对 BigInt 系数执行有理数重构.
111 *
112 * 给定整数 a 和模数 m,使用扩展欧几里得算法寻找有理数 p/q 满足:
113 * - a == p/q (mod m)
114 * - |p| <= floor(sqrt(m/2)), |q| <= floor(sqrt(m/2))
115 *
116 * 当模数超出 int64_t 范围时使用 BigInt 算术;否则委托给
117 * modular_arithmetic.hpp 中的 int64_t 版本以获得更好性能.
118 *
119 * @param[in] a 待重构的整数系数(已归约到对称表示)
120 * @param[in] m 模数(正整数,通常为 p^k)
121 * @param[out] num 输出分子
122 * @param[out] den 输出分母
123 * @return 满足界约束且 gcd=1 时返回 true;false 表示重构未决
124 * @internal
125 */
126static bool zc_rational_reconstruction(
127 const BigInt& a,
128 const BigInt& m,
129 BigInt& num,
130 BigInt& den) {
131
132 if (m.is_zero() || m.IsNegative()) return false;
133
134 /// 对于可精确表示的小模数,委托给 int64_t 版本
135 const auto a_small = a.try_to_int64();
136 const auto m_small = m.try_to_int64();
137 if (a_small && m_small) {
138 const int64_t a_val = *a_small;
139 const int64_t m_val = *m_small;
140
141 auto reconstructed =
142 lamina::rational_reconstruction_checked(a_val, m_val);
143 if (!reconstructed) return false;
144 const auto [p, q] = reconstructed.value();
145 num = BigInt(static_cast<long long>(p));
146 den = BigInt(static_cast<long long>(q));
147 return true;
148 }
149
150 /// BigInt 版本的有理重构
151 /// 将 a 归约到 [0, m)
152 BigInt a_mod = a % m;
153 if (a_mod.IsNegative()) {
154 a_mod = a_mod + m;
155 }
156
157 /// 计算界 bound = floor(sqrt(m/2))
158 BigInt half_m = m / BigInt(2);
159 BigInt bound = half_m.sqrt();
160 if (bound.is_zero()) bound = BigInt(1);
161
162 /// 扩展欧几里得算法
163 BigInt r0 = m, r1 = a_mod;
164 BigInt s0 = BigInt(0), s1 = BigInt(1);
165
166 while (r1 > bound) {
167 BigInt q = r0 / r1;
168 BigInt r_new = r0 - q * r1;
169 BigInt s_new = s0 - q * s1;
170
171 r0 = r1;
172 r1 = r_new;
173 s0 = s1;
174 s1 = s_new;
175 }
176
177 BigInt p = r1;
178 BigInt q_val = s1;
179
180 /// 确保分母为正
181 if (q_val.IsNegative()) {
182 p = -p;
183 q_val = -q_val;
184 }
185
186 /// 验证界约束
187 if (p.Abs() > bound || q_val.is_zero() || q_val > bound) {
188 return false;
189 }
190
191 /// 验证 gcd(|p|, q) == 1
192 BigInt g = BigInt::gcd(p.Abs(), q_val);
193 if (g != BigInt(1)) {
194 return false;
195 }
196
197 num = p;
198 den = q_val;
199 return true;
200}

202/**
203 * @brief 对子集乘积的所有系数执行有理重构,构造有理系数多项式.
204 *
205 * 对 product_coeffs 中的每个 BigInt 系数调用 zc_rational_reconstruction,
206 * 若所有系数均成功重构,则返回对应的有理系数多项式.
207 * 若任一系数重构失败,则回退到整数系数直接转换.
208 *
209 * @param[in] product_coeffs 子集乘积的 BigInt 系数向量(对称表示)
210 * @param[in] mod 模数 p^k
211 * @param[in] var 变量名
212 * @return 有理系数多项式(通过有理重构或整数直接转换)
213 * @internal
214 */
215static Polynomial<Rational> zc_reconstruct_candidate(
216 const std::vector<BigInt>& product_coeffs,
217 const BigInt& mod,
218 const std::string& var) {
219
220 std::vector<Rational> rat_coeffs;
221 rat_coeffs.reserve(product_coeffs.size());
222
223 /// 尝试对每个系数执行有理重构
224 for (const auto& c : product_coeffs) {
225 BigInt num, den;
226 if (zc_rational_reconstruction(c, mod, num, den)) {
227 rat_coeffs.emplace_back(num, den);
228 } else {
229 /// 任一系数重构失败,回退到整数系数
230 return zc_bigint_to_rational_poly(product_coeffs, var);
231 }
232 }
233
234 return Polynomial<Rational>(rat_coeffs, var);
235}

237/**
238 * @brief 检验候选因子是否整除原多项式(精确有理除法).
239 *
240 * 执行多项式带余除法 f / candidate,若余数为零则整除.
241 *
242 * @param[in] f 原多项式
243 * @param[in] candidate 候选因子
244 * @return 整除返回 true
245 * @internal
246 */
247static bool zc_divides_exactly(
248 const Polynomial<Rational>& f,
249 const Polynomial<Rational>& candidate) {
250
251 if (candidate.is_zero()) return false;
252 if (candidate.degree() > f.degree()) return false;
253 if (candidate.degree() == 0) return true;
254
255 auto [quotient, remainder] = f.div_mod(candidate);
256 return remainder.is_zero();
257}

271static bool zc_next_combination(std::vector<size_t>& positions, size_t count) {
272 const size_t width = positions.size();
273 for (size_t cursor = width; cursor > 0; --cursor) {
274 const size_t i = cursor - 1;
275 if (positions[i] < count - width + i) {
276 ++positions[i];
277 for (size_t j = i + 1; j < width; ++j) {
278 positions[j] = positions[j - 1] + 1;
279 }
280 return true;
281 }
282 }
283 return false;
284}

295static Result<void> zc_validate_lifted_product(
296 const Polynomial<Rational>& poly,
297 const std::vector<Polynomial<BigInt>>& lifted_factors,
298 const BigInt& modulus,
299 ComputationContext& context) {
300 if (lifted_factors.empty()) return Result<void>::success();
301
302 std::vector<size_t> all_indices(lifted_factors.size());
303 for (size_t i = 0; i < all_indices.size(); ++i) all_indices[i] = i;
304 auto product_result =
305 zc_subset_product(lifted_factors, all_indices, modulus, context);
306 if (!product_result) return Result<void>::failure(product_result.error());
307 auto product = std::move(product_result.value());
308
309 BigInt denominator_lcm(1);
310 for (const Rational& coefficient : poly.coeffs) {
311 denominator_lcm =
312 BigInt::lcm(denominator_lcm, coefficient.get_denominator());
313 }
314 std::vector<BigInt> expected;
315 expected.reserve(poly.coeffs.size());
316 for (const Rational& coefficient : poly.coeffs) {
317 BigInt integer_coefficient =
318 coefficient.get_numerator() *
319 (denominator_lcm / coefficient.get_denominator());
320 expected.push_back(zc_symmetric_mod(integer_coefficient, modulus));
321 }
322 while (!expected.empty() && expected.back().is_zero()) expected.pop_back();
323 while (!product.empty() && product.back().is_zero()) product.pop_back();
324 if (!(product == expected)) {
325 return Result<void>::failure(
326 CasErrc::InvalidArgument,
327 "lifted factors are not congruent to the normalized polynomial",
328 "zassenhaus_combine");
329 }
330 return Result<void>::success();
331}

333static ZassenhausResult zc_finalize(
334 const Polynomial<Rational>& original,
335 std::vector<Polynomial<Rational>> factors,
336 Polynomial<Rational> remaining,
337 Completeness completeness,
338 std::string reason) {
339 if (!remaining.is_zero() && remaining.degree() > 0) {
340 factors.push_back(remaining.make_monic());
341 }
342
343 if (!original.is_zero()) {
344 Polynomial<Rational> product({Rational(1)}, original.variable_name);
345 for (const auto& factor : factors) product = product * factor;
346 if (!(product == original.make_monic())) {
347 return ZassenhausResult::failure(
348 CasErrc::InternalInvariant,
349 "accepted Zassenhaus factors do not reconstruct the polynomial",
350 "zassenhaus_combine");
351 }
352 } else if (!factors.empty()) {
353 return ZassenhausResult::failure(
354 CasErrc::InternalInvariant,
355 "zero polynomial cannot have a nonempty monic factorization",
356 "zassenhaus_combine");
357 }
358
359 return ZassenhausResult::success(
360 MathResult<std::vector<Polynomial<Rational>>>{
361 std::move(factors), completeness, std::move(reason)});
362}

364static ZassenhausResult zassenhaus_combine_impl(
365 const Polynomial<Rational>& poly,
366 const std::vector<Polynomial<BigInt>>& lifted_factors,
367 const BigInt& reconstruction_modulus,
368 ComputationContext& context) {
369 if (reconstruction_modulus.is_zero() ||
370 reconstruction_modulus.IsNegative()) {
371 return ZassenhausResult::failure(
372 CasErrc::InvalidArgument,
373 "reconstruction modulus must be positive",
374 "zassenhaus_combine");
375 }
376 if (poly.is_zero()) {
377 if (!lifted_factors.empty()) {
378 return ZassenhausResult::failure(
379 CasErrc::InvalidArgument,
380 "zero polynomial requires an empty lifted-factor list",
381 "zassenhaus_combine");
382 }
383 return ZassenhausResult::success(
384 MathResult<std::vector<Polynomial<Rational>>>{{},
385 Completeness::Complete, {}});
386 }
387 if (lifted_factors.empty()) {
388 return zc_finalize(poly, {}, poly.make_monic(),
389 Completeness::Inconclusive,
390 "no lifted factors were provided");
391 }
392
393 auto validation = zc_validate_lifted_product(
394 poly, lifted_factors, reconstruction_modulus, context);
395 if (!validation) {
396 if (validation.error().code == CasErrc::ResourceLimit) {
397 return zc_finalize(poly, {}, poly.make_monic(),
398 Completeness::Inconclusive,
399 "budget exhausted while validating lifted factors");
400 }
401 return ZassenhausResult::failure(validation.error());
402 }
403
404 std::vector<Polynomial<Rational>> true_factors;
405 std::vector<size_t> active(lifted_factors.size());
406 for (size_t i = 0; i < active.size(); ++i) active[i] = i;
407 Polynomial<Rational> remaining = poly.make_monic();
408
409 for (;;) {
410 bool found = false;
411 if (active.size() <= 1 || remaining.degree() <= 1) break;
412
413 for (size_t subset_size = 1;
414 subset_size <= active.size() / 2 && !found;
415 ++subset_size) {
416 std::vector<size_t> positions(subset_size);
417 for (size_t i = 0; i < subset_size; ++i) positions[i] = i;
418 do {
419 auto candidate_step =
420 context.consume_steps(1, "zassenhaus_combine");
421 if (!candidate_step) {
422 if (candidate_step.error().code == CasErrc::ResourceLimit) {
423 return zc_finalize(
424 poly, std::move(true_factors), std::move(remaining),
425 Completeness::Inconclusive,
426 "budget exhausted during bounded combination enumeration");
427 }
428 return ZassenhausResult::failure(candidate_step.error());
429 }
430
431 std::vector<size_t> indices =
432 zc_select_indices(active, positions);
433 auto product_result = zc_subset_product(
434 lifted_factors, indices, reconstruction_modulus, context);
435 if (!product_result) {
436 if (product_result.error().code == CasErrc::ResourceLimit) {
437 return zc_finalize(
438 poly, std::move(true_factors), std::move(remaining),
439 Completeness::Inconclusive,
440 "budget exhausted during candidate multiplication");
441 }
442 return ZassenhausResult::failure(product_result.error());
443 }
444 if (product_result.value().empty()) continue;
445
446 Polynomial<Rational> candidate = zc_make_primitive(
447 zc_reconstruct_candidate(product_result.value(),
448 reconstruction_modulus,
449 poly.variable_name));
450 if (candidate.is_zero() || candidate.degree() <= 0 ||
451 candidate.degree() >= remaining.degree() ||
452 !zc_divides_exactly(remaining, candidate)) {
453 continue;
454 }
455
456 auto [quotient, remainder] = remaining.div_mod(candidate);
457 if (!remainder.is_zero()) {
458 return ZassenhausResult::failure(
459 CasErrc::InternalInvariant,
460 "exact divisor produced a nonzero remainder",
461 "zassenhaus_combine");
462 }
463 true_factors.push_back(candidate);
464 remaining = std::move(quotient);
465 for (size_t cursor = positions.size(); cursor > 0; --cursor) {
466 active.erase(active.begin() +
467 static_cast<std::ptrdiff_t>(positions[cursor - 1]));
468 }
469 found = true;
470 break;
471 } while (zc_next_combination(positions, active.size()));
472 }
473 if (!found) break;
474 }
475
476 return zc_finalize(poly, std::move(true_factors), std::move(remaining),
477 Completeness::Complete, {});
478}
