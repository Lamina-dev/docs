44/**
45 * @brief 扩展欧几里得算法，求 gcd(a, b) 及 Bezout 系数
46 * @param a 第一个整数
47 * @param b 第二个整数
48 * @param s 输出 Bezout 系数 s，满足 a*s + b*t = gcd(a,b)
49 * @param t 输出 Bezout 系数 t
50 * @return gcd(a, b)
51 */
52inline int64_t extended_gcd(int64_t a, int64_t b, int64_t& s, int64_t& t) {
53 int64_t old_r = a, r = b;
54 int64_t old_s = 1, ss = 0;
55 int64_t old_t = 0, tt = 1;
56
57 while (r != 0) {
58 int64_t q = old_r / r;
59 int64_t tmp = r;
60 r = old_r - q * r;
61 old_r = tmp;
62
63 tmp = ss;
64 ss = old_s - q * ss;
65 old_s = tmp;
66
67 tmp = tt;
68 tt = old_t - q * tt;
69 old_t = tmp;
70 }
71
72 s = old_s;
73 t = old_t;
74 return old_r;
75}

107/** @brief 模整数类，封装模 p 下的算术运算 */
108class ModInt {
109 int64_t val_;
110 int64_t mod_;
111
112public:
113 /** @brief 默认构造，值为 0，模为 2 */
114 ModInt() : val_(0), mod_(2) {}
115
116 /**
117 * @brief 构造模整数
118 * @param v 整数值（自动取模归约到 [0, p)）
119 * @param p 模数
120 */
121 ModInt(int64_t v, int64_t p) : mod_(p) {
122 val_ = v % p;
123 if (val_ < 0) val_ += p;
124 }
125
126 /** @brief 获取当前值 */
127 int64_t value() const { return val_; }
128
129 /** @brief 获取模数 */
130 int64_t modulus() const { return mod_; }
131
132 /** @brief 模加法 */
133 ModInt operator+(const ModInt& other) const {
134 return ModInt(val_ + other.val_, mod_);
135 }
136
137 /** @brief 模减法 */
138 ModInt operator-(const ModInt& other) const {
139 return ModInt(val_ - other.val_ + mod_, mod_);
140 }
141
142 /** @brief 模乘法 */
143 ModInt operator*(const ModInt& other) const {
144
145 uint64_t q;
146 uint64_t result = lmmp_mulmod_ulong_(
147 static_cast<uint64_t>(val_),
148 static_cast<uint64_t>(other.val_),
149 static_cast<uint64_t>(mod_), &q);
150 return ModInt(static_cast<int64_t>(result), mod_);
151 }
152
153 /** @brief 模除法（乘以逆元） */
154 ModInt operator/(const ModInt& other) const {
155 return *this * other.inverse();
156 }
157
158 /** @brief 取负 */
159 ModInt operator-() const {
160 return ModInt(val_ == 0 ? 0 : mod_ - val_, mod_);
161 }
162
163 /** @brief 判等 */
164 bool operator==(const ModInt& other) const {
165 return val_ == other.val_ && mod_ == other.mod_;
166 }
167
168 /** @brief 判不等 */
169 bool operator!=(const ModInt& other) const {
170 return !(*this == other);
171 }
172
173 /** @brief 小于比较（先比模数，再比值） */
174 bool operator<(const ModInt& other) const {
175 if (mod_ != other.mod_) return mod_ < other.mod_;
176 return val_ < other.val_;
177 }
178
179 /** @brief 大于比较 */
180 bool operator>(const ModInt& other) const {
181 return other < *this;
182 }
183
184 /** @brief 判断是否为零 */
185 bool is_zero() const { return val_ == 0; }
186
187 /**
188 * @brief 求模逆元
189 * @return 当前值在模 mod_ 下的乘法逆元
190 * @throw std::domain_error 若值为 0 或不可逆
191 */
192 ModInt inverse() const {
193 if (val_ == 0) {
194 throw std::domain_error("ModInt::inverse(): zero is not invertible");
195 }
196 int64_t s, t;
197 int64_t g = extended_gcd(val_, mod_, s, t);
198 if (g != 1) {
199 throw std::domain_error(
200 "ModInt::inverse(): element is not invertible (gcd != 1)");
201 }
202 return ModInt(s, mod_);
203 }
204
205 /**
206 * @brief 模幂运算
207 * @param base 底数
208 * @param exp 指数（可为负，负指数先求逆）
209 * @return base^exp mod p
210 */
211 static ModInt pow(ModInt base, int64_t exp) {
212 uint64_t power = 0;
213 if (exp < 0) {
214 base = base.inverse();
215 power = static_cast<uint64_t>(-(exp + 1)) + 1;
216 } else {
217 power = static_cast<uint64_t>(exp);
218 }
219 ModInt result(1, base.mod_);
220 while (power != 0) {
221 if ((power & 1U) != 0) result = result * base;
222 power >>= 1U;
223 if (power != 0) base = base * base;
224 }
225 return result;
226 }

229/**
230 * @brief 中国剩余定理（两模数）
231 * @param r1 第一个余数
232 * @param m1 第一个模数
233 * @param r2 第二个余数
234 * @param m2 第二个模数
235 * @return pair(合并余数, 合并模数 m1*m2)
236 * @throw std::domain_error 若模数不互素
237 */
238inline std::pair<int64_t, int64_t> crt(int64_t r1, int64_t m1,
239 int64_t r2, int64_t m2) {
240 int64_t s, t;
241 int64_t g = extended_gcd(m1, m2, s, t);
242 if (g != 1) {
243 throw std::domain_error("crt(): moduli must be coprime");
244 }
245
246 int64_t diff = ((r2 - r1) % m2 + m2) % m2;
247
248 int64_t s_mod = ((s % m2) + m2) % m2;
249
250 uint64_t q_unused;
251 uint64_t factor = lmmp_mulmod_ulong_(
252 static_cast<uint64_t>(s_mod),
253 static_cast<uint64_t>(diff),
254 static_cast<uint64_t>(m2), &q_unused);
255
256#if defined(__GNUC__) || defined(__clang__)
257 #pragma GCC diagnostic push
258 #pragma GCC diagnostic ignored "-Wpedantic"
259 __int128 x = static_cast<__int128>(r1) + static_cast<__int128>(m1) * factor;
260 __int128 bigM = static_cast<__int128>(m1) * m2;
261 x = ((x % bigM) + bigM) % bigM;
262 #pragma GCC diagnostic pop
263 return {static_cast<int64_t>(x), static_cast<int64_t>(bigM)};
264#else
265 int64_t M = m1 * m2;
266 int64_t x = r1 + m1 * static_cast<int64_t>(factor);
267 x = ((x % M) + M) % M;
268 return {x, M};
269#endif
270}

272/**
273 * @brief Checked Chinese remainder theorem for two positive coprime moduli.
274 */
275inline CrtResult crt_checked(int64_t r1, int64_t m1,
276 int64_t r2, int64_t m2,
277 ComputationContext& context) {
278 constexpr const char* operation = "crt";
279 auto step = context.consume_steps(1, operation);
280 if (!step) return CrtResult::failure(step.error());
281 if (m1 <= 0 || m2 <= 0) {
282 return CrtResult::failure(CasErrc::InvalidArgument,
283 "CRT moduli must be positive", operation);
284 }
285
286 int64_t product = 0;
287 if (!modular_detail::checked_positive_product(m1, m2, product)) {
288 return CrtResult::failure(CasErrc::ResourceLimit,
289 "CRT modulus product exceeds int64 range", operation);
290 }
291
292 int64_t s = 0;
293 int64_t t = 0;
294 int64_t g = extended_gcd(m1, m2, s, t);
295 if (g < 0) g = -g;
296 if (g != 1) {
297 return CrtResult::failure(CasErrc::InvalidArgument,
298 "CRT moduli must be coprime", operation);
299 }
300
301#if defined(__GNUC__) || defined(__clang__)
302 #pragma GCC diagnostic push
303 #pragma GCC diagnostic ignored "-Wpedantic"
304 const __int128 big_m2 = static_cast<__int128>(m2);
305 const __int128 diff = ((static_cast<__int128>(r2) - r1) % big_m2 + big_m2) % big_m2;
306 const __int128 s_mod = ((static_cast<__int128>(s) % big_m2) + big_m2) % big_m2;
307 __int128 factor = (s_mod * diff) % big_m2;
308 __int128 x = static_cast<__int128>(r1) + static_cast<__int128>(m1) * factor;
309 const __int128 big_product = static_cast<__int128>(product);
310 x = ((x % big_product) + big_product) % big_product;
311 #pragma GCC diagnostic pop
312 return CrtResult::success({static_cast<int64_t>(x), product});
313#else
314 try {
315 return CrtResult::success(crt(r1, m1, r2, m2));
316 } catch (const std::bad_alloc&) {
317 return CrtResult::failure(CasErrc::ResourceLimit,
318 "CRT allocation failed", operation);
319 } catch (const std::exception& ex) {
320 return CrtResult::failure(CasErrc::InternalInvariant, ex.what(), operation);
321 }
322#endif
323}

325inline CrtResult crt_checked(int64_t r1, int64_t m1, int64_t r2, int64_t m2) {

330/**
331 * @brief 多模数中国剩余定理
332 * @param residues 余数列表
333 * @param primes 对应的模数列表（两两互素）
334 * @return pair(合并余数, 合并模数)
335 * @throw std::invalid_argument 若输入为空或大小不匹配
336 */
337inline std::pair<int64_t, int64_t> multi_crt(const std::vector<int64_t>& residues,
338 const std::vector<int64_t>& primes) {
339 if (residues.empty() || residues.size() != primes.size()) {
340 throw std::invalid_argument("multi_crt(): residues and primes must be non-empty and same size");
341 }
342
343 int64_t combined = residues[0];
344 int64_t modulus = primes[0];
345
346 for (size_t i = 1; i < residues.size(); ++i) {
347 auto [x, m] = crt(combined, modulus, residues[i], primes[i]);
348 combined = x;
349 modulus = m;
350 }
351
352 return {combined, modulus};
353}

355/**
356 * @brief Checked CRT for a non-empty list of pairwise coprime positive moduli.
357 */
358inline CrtResult multi_crt_checked(const std::vector<int64_t>& residues,
359 const std::vector<int64_t>& primes,
360 ComputationContext& context) {
361 constexpr const char* operation = "multi_crt";
362 auto step = context.consume_steps(residues.size() + 1, operation);
363 if (!step) return CrtResult::failure(step.error());
364 if (residues.empty() || residues.size() != primes.size()) {
365 return CrtResult::failure(
366 CasErrc::InvalidArgument,
367 "multi_crt residues and moduli must be non-empty and the same size",
368 operation);
369 }
370 for (int64_t modulus : primes) {
371 if (modulus <= 0) {
372 return CrtResult::failure(CasErrc::InvalidArgument,
373 "multi_crt moduli must be positive", operation);
374 }
375 }
376
377 int64_t combined = residues[0];
378 int64_t modulus = primes[0];
379 for (size_t i = 1; i < residues.size(); ++i) {
380 auto merged = crt_checked(combined, modulus, residues[i], primes[i], context);
381 if (!merged) return merged;
382 combined = merged.value().first;
383 modulus = merged.value().second;
384 }
385
386 return CrtResult::success({combined, modulus});
387}

396/**
397 * @brief Checked rational reconstruction. Returns Inconclusive when no
398 * supported coprime numerator/denominator can be reconstructed.
399 */
400inline RationalReconstructionResult rational_reconstruction_checked(
401 int64_t x,
402 int64_t m,
403 ComputationContext& context) {
404 constexpr const char* operation = "rational_reconstruction";
405 auto step = context.consume_steps(1, operation);
406 if (!step) return RationalReconstructionResult::failure(step.error());
407 if (m <= 0) {
408 return RationalReconstructionResult::failure(
409 CasErrc::InvalidArgument,
410 "rational reconstruction modulus must be positive",
411 operation);
412 }
413
414 x = ((x % m) + m) % m;
415 int64_t bound = static_cast<int64_t>(
416 std::floor(std::sqrt(static_cast<double>(m) / 2.0)));
417 if (bound == 0) bound = 1;
418 int64_t r0 = m;
419 int64_t r1 = x;
420 int64_t s0 = 0;
421 int64_t s1 = 1;
422 while (r1 > bound) {
423 step = context.consume_steps(1, operation);
424 if (!step) return RationalReconstructionResult::failure(step.error());
425 const int64_t quotient = r0 / r1;
426 const int64_t next_r = r0 - quotient * r1;
427 const int64_t next_s = s0 - quotient * s1;
428 r0 = r1;
429 r1 = next_r;
430 s0 = s1;
431 s1 = next_s;
432 }
433 int64_t numerator = r1;
434 int64_t denominator = s1;
435 if (denominator < 0) {
436 numerator = -numerator;
437 denominator = -denominator;
438 }
439 int64_t unused_s = 0;
440 int64_t unused_t = 0;
441 const bool outside_bound =
442 std::abs(numerator) >= bound || denominator == 0 ||
443 denominator >= bound;
444 const bool not_coprime = !outside_bound &&
445 extended_gcd(std::abs(numerator), denominator,
446 unused_s, unused_t) != 1;
447 if (outside_bound || not_coprime) {
448 return RationalReconstructionResult::failure(
449 CasErrc::Inconclusive,
450 "no rational reconstruction exists in the supported bound",
451 operation);
452 }
453 return RationalReconstructionResult::success(
454 {numerator, denominator});
455}

472/**
473 * @brief 判断素数 p 是否为"好素数"（不整除任何首项系数）
474 * @param p 待检测素数
475 * @param leading_coeffs 首项系数列表
476 * @return 若 p 不整除任何系数则返回 true
477 */
478inline bool is_good_prime(int64_t p, const std::vector<int64_t>& leading_coeffs) {
479 return std::none_of(leading_coeffs.begin(), leading_coeffs.end(),
480 [p](int64_t c) { return (c % p) == 0; });
481}

483/**
484 * @brief 生成指定数量的好素数
485 * @param count 需要的素数个数
486 * @param leading_coeffs 首项系数列表（用于过滤）
487 * @param start_from 搜索起始值
488 * @return 好素数列表
489 */
490inline std::vector<int64_t> generate_good_primes(int count,
491 const std::vector<int64_t>& leading_coeffs,
492 uint64_t start_from = 1000000000ULL) {
493 std::vector<int64_t> result;
494 result.reserve(count);
495 uint64_t candidate = start_from;
496 while ((int)result.size() < count) {
497 candidate = lmmp_next_prime_ulong_(candidate);
498 if (is_good_prime(static_cast<int64_t>(candidate), leading_coeffs)) {
499 result.push_back(static_cast<int64_t>(candidate));
500 }
501 }
502 return result;
503}
