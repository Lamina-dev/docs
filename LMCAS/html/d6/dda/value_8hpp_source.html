145 /** @brief Checked conversion to a floating-point value. */
146 LMCAS::Result<lmmc_real_t> as_number_checked() const {
147 if (type == Type::Infinity) {
148 lmmc_real_t inf;
149 lmmc_inf(&inf);
150 const auto sign = std::get_if<int>(&data);
151 return LMCAS::Result<lmmc_real_t>::success(
152 (sign && *sign > 0) ? inf : -inf);
153 }
154 if (type == Type::Int) {
155 return LMCAS::Result<lmmc_real_t>::success(
156 static_cast<lmmc_real_t>(std::get<int>(data)));
157 }
158 if (type == Type::Float) {
159 return LMCAS::Result<lmmc_real_t>::success(
160 std::get<lmmc_real_t>(data));
161 }
162 if (type == Type::BigInt) {
163 return LMCAS::Result<lmmc_real_t>::success(
164 std::get<::LMCAS::BigInt>(data).to_double());
165 }
166 if (type == Type::Rational) {
167 return LMCAS::Result<lmmc_real_t>::success(
168 std::get<::LMCAS::Rational>(data).to_double());
169 }
170 if (type == Type::Irrational) {
171 return LMCAS::Result<lmmc_real_t>::success(
172 std::get<::LMCAS::Irrational>(data).to_double());
173 }
174 if (type == Type::Symbolic) {
175 const auto& expression =
176 std::get<std::shared_ptr<SymbolicExpr>>(data);
177 auto evaluated = LMCAS::evaluate_numeric(*expression);
178 if (!evaluated) {
179 return LMCAS::Result<lmmc_real_t>::failure(
180 evaluated.error());
181 }
182 if (!evaluated.value().is_finite() ||
183 !std::isfinite(evaluated.value().value)) {
184 return LMCAS::Result<lmmc_real_t>::failure(
185 LMCAS::CasErrc::NumericFailure,
186 "symbolic value did not evaluate to a finite number",
187 "Value::as_number_checked");
188 }
189 return LMCAS::Result<lmmc_real_t>::success(
190 static_cast<lmmc_real_t>(evaluated.value().value));
191 }
192 return LMCAS::Result<lmmc_real_t>::failure(
193 LMCAS::CasErrc::InvalidArgument,
194 "value is not numeric", "Value::as_number_checked");
195 }

282 /** @brief Checked conversion to a symbolic expression. */
283 LMCAS::Result<std::shared_ptr<SymbolicExpr>> as_symbolic_checked() const {
284 if (type == Type::Infinity) {
285 const auto sign = std::get_if<int>(&data);
286 return LMCAS::Result<std::shared_ptr<SymbolicExpr>>::success(
287 SymbolicExpr::infinity(sign ? *sign : 1));
288 }
289 if (type == Type::Symbolic) {
290 return LMCAS::Result<std::shared_ptr<SymbolicExpr>>::success(
291 std::get<std::shared_ptr<SymbolicExpr>>(data));
292 }
293 if (type == Type::Float) {
294 return LMCAS::Result<std::shared_ptr<SymbolicExpr>>::success(
295 SymbolicExpr::number(std::get<lmmc_real_t>(data)));
296 }
297 if (type == Type::Int || type == Type::Rational ||
298 type == Type::BigInt) {
299 auto rational = as_rational_checked();
300 if (!rational) {
301 return LMCAS::Result<std::shared_ptr<SymbolicExpr>>::failure(
302 rational.error());
303 }
304 return LMCAS::Result<std::shared_ptr<SymbolicExpr>>::success(
305 SymbolicExpr::number(rational.value()));
306 }
307 if (type == Type::Irrational) {
308 return LMCAS::Result<std::shared_ptr<SymbolicExpr>>::success(
309 std::get<::LMCAS::Irrational>(data).to_symbolic());
310 }
311 if (type == Type::Matrix) {
312 const auto& matrix =
313 std::get<std::vector<std::vector<Value>>>(data);
314 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>>
315 symbolic_matrix;
316 symbolic_matrix.reserve(matrix.size());
317 for (const auto& row : matrix) {
318 std::vector<std::shared_ptr<SymbolicExpr>> symbolic_row;
319 symbolic_row.reserve(row.size());
320 for (const auto& value : row) {
321 auto symbolic = value.as_symbolic_checked();
322 if (!symbolic) {
323 return LMCAS::Result<
324 std::shared_ptr<SymbolicExpr>>::failure(
325 symbolic.error());
326 }
327 symbolic_row.push_back(std::move(symbolic.value()));
328 }
329 symbolic_matrix.push_back(std::move(symbolic_row));
330 }
331 return LMCAS::Result<std::shared_ptr<SymbolicExpr>>::success(
332 SymbolicExpr::matrix(symbolic_matrix));
333 }
334 return LMCAS::Result<std::shared_ptr<SymbolicExpr>>::failure(
335 LMCAS::CasErrc::InvalidArgument,
336 "value cannot be converted to a symbolic expression",
337 "Value::as_symbolic_checked");
338 }

359 /**
360 * @brief 将值转换为可读字符串
361 * @return 格式化的字符串表示
362 */
363 std::string to_string() const {
364 if (type == Type::String) return _str_cache;
365 if (!_str_cache.empty()) return _str_cache;
366 switch (type) {
367 case Type::Infinity: {
368 auto sign = std::get_if<int>(&data);
369 return (sign && *sign > 0) ? "inf" : "-inf";
370 }
371 case Type::Null:
372 return "null";
373 case Type::Int:
374 return std::to_string(std::get<int>(data));
375 case Type::Float: {
376 lmmc_real_t val = std::get<lmmc_real_t>(data);
377 std::string str = std::to_string(val);
378 str.erase(str.find_last_not_of('0') + 1, std::string::npos);
379 str.erase(str.find_last_not_of('.') + 1, std::string::npos);
380 return str;
381 }
382 case Type::Array: {
383 std::string res = "[";
384 const auto& arr = std::get<std::vector<Value>>(data);
385 for (size_t i = 0; i < arr.size(); ++i) {
386 if (i) res += ", ";
387 res += arr[i].to_string();
388 }
389 res += "]";
390 return res;
391 }
392 case Type::Matrix: {
393 std::string res = "[";
394 const auto& mat = std::get<std::vector<std::vector<Value>>>(data);
395 for (size_t i = 0; i < mat.size(); ++i) {
396 if (i) res += ", ";
397 res += "[";
398 for (size_t j = 0; j < mat[i].size(); ++j) {
399 if (j) res += ", ";
400 res += mat[i][j].to_string();
401 }
402 res += "]";
403 }
404 res += "]";
405 return res;
406 }
407 case Type::BigInt:
408 return std::get<::LMCAS::BigInt>(data).to_string();
409 case Type::Rational:
410 return std::get<::LMCAS::Rational>(data).to_string();
411 case Type::Irrational:
412 return std::get<::LMCAS::Irrational>(data).to_string();
413 case Type::Symbolic: {
414 const auto& expression =
415 std::get<std::shared_ptr<SymbolicExpr>>(data);
416 if (!expression) return "<invalid-symbolic>";
417 return expression->to_string();
418 }
419 default:
420 return "<unknown>";
421 }

424 bool operator==(const Value& other) const {
425 if (type != other.type) return false;
426 if (type == Type::String) {
427 return _str_cache == other._str_cache;
428 }
429 if (data.index() != other.data.index()) return false;
430 return std::visit([&other](const auto& val1) -> bool {
431 using T = std::decay_t<decltype(val1)>;
432 if constexpr (std::is_same_v<T, std::nullptr_t>) {
433 return true;
434 } else if constexpr (std::is_same_v<T, std::shared_ptr<SymbolicExpr>>) {
435 if (!val1 && !std::get<T>(other.data)) return true;
436 if (!val1 || !std::get<T>(other.data)) return false;
437 return val1->compare(std::get<T>(other.data)) == 0;
438 } else {
439 return val1 == std::get<T>(other.data);
440 }
441 }, data);
442 }
443
444 bool operator<(const Value& other) const {
445 if (type != other.type) return static_cast<int>(type) < static_cast<int>(other.type);
446 if (type == Type::String) {
447 return _str_cache < other._str_cache;
448 }
449 if (data.index() != other.data.index()) return data.index() < other.data.index();
450 return std::visit([&other](const auto& val1) -> bool {
451 using T = std::decay_t<decltype(val1)>;
452 if constexpr (std::is_same_v<T, std::nullptr_t>) {
453 return false;
454 } else if constexpr (std::is_same_v<T, std::shared_ptr<SymbolicExpr>>) {
455 if (!val1 && !std::get<T>(other.data)) return false;
456 if (!val1) return true;
457 if (!std::get<T>(other.data)) return false;
458 return val1->compare(std::get<T>(other.data)) < 0;
459 } else {
460 return val1 < std::get<T>(other.data);
461 }
462 }, data);
463 }

465 /** @brief Checked element-wise vector addition. */
466 LMCAS::Result<Value> vector_add_checked(const Value& other) const {
467 if (!is_array() || !other.is_array()) {
468 return LMCAS::Result<Value>::failure(
469 LMCAS::CasErrc::InvalidArgument,
470 "vector addition requires two arrays",
471 "Value::vector_add_checked");
472 }
473 const auto& lhs = std::get<std::vector<Value>>(data);
474 const auto& rhs = std::get<std::vector<Value>>(other.data);
475 if (lhs.size() != rhs.size()) {
476 return LMCAS::Result<Value>::failure(
477 LMCAS::CasErrc::DimensionMismatch,
478 "vector lengths do not match",
479 "Value::vector_add_checked");
480 }
481 std::vector<Value> values;
482 values.reserve(lhs.size());
483 for (size_t index = 0; index < lhs.size(); ++index) {
484 auto left = lhs[index].as_number_checked();
485 if (!left) return LMCAS::Result<Value>::failure(left.error());
486 auto right = rhs[index].as_number_checked();
487 if (!right) return LMCAS::Result<Value>::failure(right.error());
488 values.emplace_back(left.value() + right.value());
489 }
490 return LMCAS::Result<Value>::success(Value(values));
491 }

532 /** @brief Checked matrix multiplication. */
533 LMCAS::Result<Value> matrix_multiply_checked(
534 const Value& other) const {
535 if (!is_matrix() || !other.is_matrix()) {
536 return LMCAS::Result<Value>::failure(
537 LMCAS::CasErrc::InvalidArgument,
538 "matrix multiplication requires two matrices",
539 "Value::matrix_multiply_checked");
540 }
541 const auto& lhs =
542 std::get<std::vector<std::vector<Value>>>(data);
543 const auto& rhs =
544 std::get<std::vector<std::vector<Value>>>(other.data);
545 if (lhs.empty() || rhs.empty() ||
546 lhs.front().size() != rhs.size()) {
547 return LMCAS::Result<Value>::failure(
548 LMCAS::CasErrc::DimensionMismatch,
549 "matrix dimensions are not multiplicatively compatible",
550 "Value::matrix_multiply_checked");
551 }
552 const size_t rows = lhs.size();
553 const size_t columns = rhs.front().size();
554 const size_t inner = lhs.front().size();
555 std::vector<std::vector<Value>> values(
556 rows, std::vector<Value>(columns, Value(0.0)));
557 for (size_t row = 0; row < rows; ++row) {
558 for (size_t column = 0; column < columns; ++column) {
559 lmmc_real_t sum = 0.0;
560 for (size_t index = 0; index < inner; ++index) {
561 auto left = lhs[row][index].as_number_checked();
562 if (!left) {
563 return LMCAS::Result<Value>::failure(left.error());
564 }
565 auto right = rhs[index][column].as_number_checked();
566 if (!right) {
567 return LMCAS::Result<Value>::failure(right.error());
568 }
569 sum += left.value() * right.value();
570 }
571 values[row][column] = Value(sum);
572 }
573 }
574 return LMCAS::Result<Value>::success(Value(values));
575 }
