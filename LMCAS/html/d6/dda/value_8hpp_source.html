101 bool is_numeric() const {
102 if (type == Type::Int || type == Type::Float ||
103 type == Type::BigInt || type == Type::Rational ||
104 type == Type::Irrational) {
105 return true;
106 }
107 // Symbolic 表达式只要能化简为数值节点，就视为可走 numeric 路径；否则不视为
108 // numeric，避免上层算子把无法求值的符号悄悄塞进数值路径。
109 if (type == Type::Symbolic) {
110 const auto& sp = std::get<std::shared_ptr<SymbolicExpr>>(data);
111 if (!sp) return false;
112 try {
113 auto simp = sp->simplify();
114 return simp && simp->is_number();
115 } catch (...) {
116 return false;
117 }
118 }
119 return false;
120 }

122 /**
123 * @brief 将值转换为浮点数
124 * @return 浮点数近似值，非数值类型返回 0.0
125 */
126 lmmc_real_t as_number() const {
127 if (type == Type::Infinity) {
128 lmmc_real_t inf;
129 lmmc_inf(&inf);
130 auto sign = std::get_if<int>(&data);
131 return (sign && *sign > 0) ? inf : -inf;
132 }
133 if (type == Type::Int) return static_cast<lmmc_real_t>(std::get<int>(data));
134 if (type == Type::Float) return std::get<lmmc_real_t>(data);
135 if (type == Type::BigInt) {
136 // 直接用 BigInt::to_double() 即可，避免先 to_int() 后再转 double 在大数上失真。
137 return std::get<::BigInt>(data).to_double();
138 }
139 if (type == Type::Rational) {
140 return std::get<::Rational>(data).to_double();
141 }
142 if (type == Type::Irrational) {
143 return std::get<::Irrational>(data).to_double();
144 }
145 if (type == Type::Symbolic) {
146 const auto& expr = std::get<std::shared_ptr<SymbolicExpr>>(data);
147 if (!expr) {
148 throw std::runtime_error(
149 "numeric evaluation failed: symbolic value is null");
150 }
151 auto evaluated = lamina::evaluate_numeric(*expr);
152 if (!evaluated || !evaluated.value().is_finite() ||
153 !std::isfinite(evaluated.value().value)) {
154 throw std::runtime_error(
155 "numeric evaluation failed: " +
156 (evaluated ? std::string("non-finite result")
157 : evaluated.error().message));
158 }
159 return static_cast<lmmc_real_t>(evaluated.value().value);
160 }
161 return 0.0;
162 }

198 /**
199 * @brief 将值转换为符号表达式
200 * @return 符号表达式智能指针，不可转换时返回数值 0
201 */
202 std::shared_ptr<SymbolicExpr> as_symbolic() const {
203 if (type == Type::Infinity) {
204 auto sign = std::get_if<int>(&data);
205 return SymbolicExpr::infinity(sign ? *sign : 1);
206 }
207 if (type == Type::Symbolic) return std::get<std::shared_ptr<SymbolicExpr>>(data);
208 if (type == Type::Int || type == Type::Float || type == Type::Rational || type == Type::BigInt) {
209 return SymbolicExpr::number(as_rational());
210 }
211 if (type == Type::Irrational) {
212 return as_irrational().to_symbolic();
213 }
214 if (type == Type::Matrix) {
215 const auto& mat = std::get<std::vector<std::vector<Value>>>(data);
216 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> sym_mat;
217 for (const auto& row : mat) {
218 std::vector<std::shared_ptr<SymbolicExpr>> sym_row;
219 for (const auto& val : row) {
220 sym_row.push_back(val.as_symbolic());
221 }
222 sym_mat.push_back(sym_row);
223 }
224 return SymbolicExpr::matrix(sym_mat);
225 }
226 return SymbolicExpr::number(0);
227 }

241 /**
242 * @brief 将值转换为可读字符串
243 * @return 格式化的字符串表示
244 */
245 std::string to_string() const {
246 if (type == Type::String) return _str_cache;
247 if (!_str_cache.empty()) return _str_cache;
248 switch (type) {
249 case Type::Infinity: {
250 auto sign = std::get_if<int>(&data);
251 return (sign && *sign > 0) ? "inf" : "-inf";
252 }
253 case Type::Null:
254 return "null";
255 case Type::Int:
256 return std::to_string(std::get<int>(data));
257 case Type::Float: {
258 lmmc_real_t val = std::get<lmmc_real_t>(data);
259 std::string str = std::to_string(val);
260 str.erase(str.find_last_not_of('0') + 1, std::string::npos);
261 str.erase(str.find_last_not_of('.') + 1, std::string::npos);
262 return str;
263 }
264 case Type::Array: {
265 std::string res = "[";
266 const auto& arr = std::get<std::vector<Value>>(data);
267 for (size_t i = 0; i < arr.size(); ++i) {
268 if (i) res += ", ";
269 res += arr[i].to_string();
270 }
271 res += "]";
272 return res;
273 }
274 case Type::Matrix: {
275 std::string res = "[";
276 const auto& mat = std::get<std::vector<std::vector<Value>>>(data);
277 for (size_t i = 0; i < mat.size(); ++i) {
278 if (i) res += ", ";
279 res += "[";
280 for (size_t j = 0; j < mat[i].size(); ++j) {
281 if (j) res += ", ";
282 res += mat[i][j].to_string();
283 }
284 res += "]";
285 }
286 res += "]";
287 return res;
288 }
289 case Type::BigInt:
290 return std::get<::BigInt>(data).to_string();
291 case Type::Rational:
292 return std::get<::Rational>(data).to_string();
293 case Type::Irrational:
294 return std::get<::Irrational>(data).to_string();
295 case Type::Symbolic:
296 return std::get<std::shared_ptr<SymbolicExpr>>(data)->to_string();
297 default:
298 return "<unknown>";
299 }

302 bool operator==(const Value& other) const {
303 if (type != other.type) return false;
304 if (type == Type::String) {
305 return _str_cache == other._str_cache;
306 }
307 if (data.index() != other.data.index()) return false;
308 return std::visit([&other](const auto& val1) -> bool {
309 using T = std::decay_t<decltype(val1)>;
310 if constexpr (std::is_same_v<T, std::nullptr_t>) {
311 return true;
312 } else if constexpr (std::is_same_v<T, std::shared_ptr<SymbolicExpr>>) {
313 if (!val1 && !std::get<T>(other.data)) return true;
314 if (!val1 || !std::get<T>(other.data)) return false;
315 return val1->compare(std::get<T>(other.data)) == 0;
316 } else {
317 return val1 == std::get<T>(other.data);
318 }
319 }, data);
320 }
321
322 bool operator<(const Value& other) const {
323 if (type != other.type) return static_cast<int>(type) < static_cast<int>(other.type);
324 if (type == Type::String) {
325 return _str_cache < other._str_cache;
326 }
327 if (data.index() != other.data.index()) return data.index() < other.data.index();
328 return std::visit([&other](const auto& val1) -> bool {
329 using T = std::decay_t<decltype(val1)>;
330 if constexpr (std::is_same_v<T, std::nullptr_t>) {
331 return false;
332 } else if constexpr (std::is_same_v<T, std::shared_ptr<SymbolicExpr>>) {
333 if (!val1 && !std::get<T>(other.data)) return false;
334 if (!val1) return true;
335 if (!std::get<T>(other.data)) return false;
336 return val1->compare(std::get<T>(other.data)) < 0;
337 } else {
338 return val1 < std::get<T>(other.data);
339 }
340 }, data);
341 }

385 /**
386 * @brief 矩阵乘法
387 * @param other 右矩阵
388 * @return 乘积矩阵
389 */
390 Value matrix_multiply(const Value& other) const {
391 if (!is_matrix() || !other.is_matrix()) return Value();
392 const auto& a = std::get<std::vector<std::vector<Value>>>(data);
393 const auto& b = std::get<std::vector<std::vector<Value>>>(other.data);
394 if (a.empty() || b.empty() || a[0].size() != b.size()) return Value();
395 size_t rows = a.size();
396 size_t cols = b[0].size();
397 size_t inner = a[0].size();
398 std::vector<std::vector<Value>> result(rows, std::vector<Value>(cols, Value(0.0)));
399 for (size_t i = 0; i < rows; ++i) {
400 for (size_t j = 0; j < cols; ++j) {
401 lmmc_real_t sum = 0.0;
402 for (size_t k = 0; k < inner; ++k) {
403 if (!a[i][k].is_numeric() || !b[k][j].is_numeric()) return Value();
404 sum += a[i][k].as_number() * b[k][j].as_number();
405 }
406 result[i][j] = Value(sum);
407 }
408 }
409 return Value(result);
410 }
