50Result<bool> domain_contains_node(
51 NumberDomain domain,
52 const std::shared_ptr<const SymbolicNode>& node) {
53 if (!node) {
54 return bool_failure(CasErrc::InvalidArgument,
55 "domain membership element must not be null",
56 kNumberDomainOperation);
57 }
58 if (domain == NumberDomain::Expressions) {
59 return Result<bool>::success(true);
60 }
61
62 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(node)) {
63 if (is_imaginary_unit_name(variable->name())) {
64 return Result<bool>::success(domain == NumberDomain::Complexes);
65 }
66 }
67
68 if (auto number = std::dynamic_pointer_cast<const NumberNode>(node)) {
69 const auto& value = number->value();
70 if (std::holds_alternative<BigInt>(value)) {
71 return Result<bool>::success(true);
72 }
73 if (std::holds_alternative<Rational>(value)) {
74 const auto& rational = std::get<Rational>(value);
75 switch (domain) {
76 case NumberDomain::Integers:
77 return Result<bool>::success(rational.is_integer());
78 case NumberDomain::Rationals:
79 case NumberDomain::Reals:
80 case NumberDomain::Complexes:
81 case NumberDomain::Expressions:
82 return Result<bool>::success(true);
83 }
84 }
85 const lmmc_real_t real = std::get<lmmc_real_t>(value);
86 if (!std::isfinite(static_cast<double>(real))) {
87 return bool_failure(CasErrc::NumericFailure,
88 "domain membership requires finite numeric literals",
89 kNumberDomainOperation);
90 }
91 switch (domain) {
92 case NumberDomain::Integers:
93 case NumberDomain::Rationals:
94 return Result<bool>::success(false);
95 case NumberDomain::Reals:
96 case NumberDomain::Complexes:
97 case NumberDomain::Expressions:
98 return Result<bool>::success(true);
99 }
100 }
101
102 if (auto complex_node = std::dynamic_pointer_cast<const ComplexNode>(node)) {
103 if (domain == NumberDomain::Complexes) {
104 auto real_part =
105 domain_contains_node(NumberDomain::Reals, complex_node->real());
106 if (!real_part) return real_part;
107 auto imag_part =
108 domain_contains_node(NumberDomain::Reals, complex_node->imag());
109 if (!imag_part) return imag_part;
110 return Result<bool>::success(real_part.value() && imag_part.value());
111 }
112
113 if (!complex_node->imag()->is_zero()) {
114 return Result<bool>::success(false);
115 }
116 return domain_contains_node(domain, complex_node->real());
117 }
118
119 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
120 for (const auto& operand : add->operands()) {
121 auto member = domain_contains_node(domain, operand);
122 if (!member || !member.value()) return member;
123 }
124 return Result<bool>::success(true);
125 }
126
127 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
128 for (const auto& operand : multiply->operands()) {
129 auto member = domain_contains_node(domain, operand);
130 if (!member || !member.value()) return member;
131 }
132 return Result<bool>::success(true);
133 }
134
135 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
136 auto exponent = std::dynamic_pointer_cast<const NumberNode>(
137 power->exponent());
138 if (!exponent || std::holds_alternative<lmmc_real_t>(exponent->value())) {
139 return bool_failure(
140 CasErrc::Inconclusive,
141 "domain membership for powers requires an exact exponent",
142 kNumberDomainOperation);
143 }
144 Rational exponent_value = std::holds_alternative<BigInt>(exponent->value())
145 ? Rational(std::get<BigInt>(exponent->value()))
146 : std::get<Rational>(exponent->value());
147 if (!exponent_value.is_integer()) {
148 auto exact_base = exact_number_value(power->base());
149 if (exponent_value == Rational(1, 2) && exact_base) {
150 if (*exact_base < Rational(0)) {
151 return Result<bool>::success(
152 domain == NumberDomain::Complexes);
153 }
154 if (domain == NumberDomain::Reals ||
155 domain == NumberDomain::Complexes) {
156 return Result<bool>::success(true);
157 }
158 const bool rational_root =
159 exact_base->get_numerator().is_perfect_square() &&
160 exact_base->get_denominator().is_perfect_square();
161 return Result<bool>::success(
162 domain == NumberDomain::Rationals
163 ? rational_root
164 : rational_root &&
165 exact_base->get_denominator() == BigInt(1));
166 }
167 return bool_failure(
168 CasErrc::Inconclusive,
169 "non-integer power membership requires a branch/domain proof",
170 kNumberDomainOperation);
171 }
172 auto base_member = domain_contains_node(domain, power->base());
173 if (!base_member || !base_member.value()) return base_member;
174 if (domain == NumberDomain::Integers &&
175 exponent_value < Rational(0)) {
176 return Result<bool>::success(false);
177 }
178 return Result<bool>::success(true);
179 }
180
181 if (std::dynamic_pointer_cast<const RootOfNode>(node)) {
182 auto expression = lamina::detail::make_expression_ptr(node);
183 if (domain == NumberDomain::Complexes) {
184 auto value = rootof_evaluate_complex_checked(expression);
185 return value ? Result<bool>::success(true)
186 : Result<bool>::failure(value.error());
187 }
188 ComputationContext root_context;
189 auto value = rootof_evaluate_checked(expression, root_context);
190 if (value) return Result<bool>::success(domain == NumberDomain::Reals);
191 if (value.error().code == CasErrc::DomainError) {
192 return Result<bool>::success(false);
193 }
194 return Result<bool>::failure(value.error());
195 }
196
197 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(node)) {
198 if (function->type() == FunctionNode::FuncType::Sqrt &&
199 function->arguments().size() == 1) {
200 auto exact = exact_number_value(function->arguments()[0]);
201 if (exact) {
202 const bool negative = *exact < Rational(0);
203 if (domain == NumberDomain::Complexes) {
204 return Result<bool>::success(true);
205 }
206 if (negative) return Result<bool>::success(false);
207 if (domain == NumberDomain::Reals) {
208 return Result<bool>::success(true);
209 }
210 const bool rational_root =
211 exact->get_numerator().is_perfect_square() &&
212 exact->get_denominator().is_perfect_square();
213 if (domain == NumberDomain::Rationals) {
214 return Result<bool>::success(rational_root);
215 }
216 if (domain == NumberDomain::Integers) {
217 return Result<bool>::success(
218 rational_root && exact->get_denominator() == BigInt(1));
219 }
220 }
221 }
222 if (function->arguments().size() == 1 &&
223 (function->type() == FunctionNode::FuncType::Sin ||
224 function->type() == FunctionNode::FuncType::Cos ||
225 function->type() == FunctionNode::FuncType::Exp ||
226 function->type() == FunctionNode::FuncType::Sinh ||
227 function->type() == FunctionNode::FuncType::Cosh ||
228 function->type() == FunctionNode::FuncType::Tanh ||
229 function->type() == FunctionNode::FuncType::Abs)) {
230 if (domain == NumberDomain::Integers ||
231 domain == NumberDomain::Rationals) {
232 return Result<bool>::success(false);
233 }
234 return domain_contains_node(
235 domain == NumberDomain::Complexes
236 ? NumberDomain::Complexes
237 : NumberDomain::Reals,
238 function->arguments()[0]);
239 }
240 }
241
242 if (domain == NumberDomain::Complexes) {
243 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
244 for (const auto& operand : add->operands()) {
245 auto member = domain_contains_node(domain, operand);
246 if (!member || !member.value()) return member;
247 }
248 return Result<bool>::success(true);
249 }
250
251 if (auto multiply =
252 std::dynamic_pointer_cast<const MultiplyNode>(node)) {
253 for (const auto& operand : multiply->operands()) {
254 auto member = domain_contains_node(domain, operand);
255 if (!member || !member.value()) return member;
256 }
257 return Result<bool>::success(true);
258 }
259
260 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
261 if (!exact_small_integer_node(power->exponent(), 1, 64)) {
262 return bool_failure(
263 CasErrc::Inconclusive,
264 "complex domain membership for powers requires a positive exact integer exponent",
265 kNumberDomainOperation);
266 }
267 auto base_member = domain_contains_node(domain, power->base());
268 if (!base_member) return base_member;
269 return Result<bool>::success(base_member.value());
270 }
271 }
272
273 return bool_failure(CasErrc::Inconclusive,
274 "domain membership is only decidable for numeric and explicit complex expressions",
275 kNumberDomainOperation);
276}
