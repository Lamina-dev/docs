14// Test whether `node` is a single-argument FunctionNode of the given type
15// whose argument is exactly the integration variable.
16bool sf_is_fn_of_var(const std::shared_ptr<const SymbolicNode>& node,
17 FunctionNode::FuncType t,
18 const std::string& var) {
19 auto fn = std::dynamic_pointer_cast<const FunctionNode>(node);
20 if (!fn || fn->type() != t) { return false; }
21 if (fn->arguments().size() != 1) { return false; }
22 auto v = std::dynamic_pointer_cast<const VariableNode>(fn->arguments()[0]);
23 return v && !v->is_constant() && v->name() == var;
24}

26// Detect a 1/x factor: a PowerNode whose base is the integration variable and
27// exponent is the integer -1.
28bool sf_is_inv_var(const std::shared_ptr<const SymbolicNode>& node,
29 const std::string& var) {
30 auto pw = std::dynamic_pointer_cast<const PowerNode>(node);
31 if (!pw) { return false; }
32 auto b = std::dynamic_pointer_cast<const VariableNode>(pw->base());
33 if (!b || b->is_constant() || b->name() != var) { return false; }
34 auto en = std::dynamic_pointer_cast<const NumberNode>(pw->exponent());
35 if (!en) { return false; }
36 if (std::holds_alternative<BigInt>(en->value())) {
37 return std::get<BigInt>(en->value()) == BigInt(-1);
38 }
39 if (std::holds_alternative<Rational>(en->value())) {
40 return std::get<Rational>(en->value()) == Rational(-1);
41 }
42 if (std::holds_alternative<lmmc_real_t>(en->value())) {
43 lmmc_real_t d = std::get<lmmc_real_t>(en->value());
44 int eq = 0;
45 lmmc_double_nearly_equal_tol(d, -1.0, 1e-12, 1e-12, &eq);
46 return eq != 0;
47 }
48 return false;
49}

51// Detect a 1/ln(var) factor, i.e. PowerNode(ln(var), -1).
52bool sf_is_inv_ln_var(const std::shared_ptr<const SymbolicNode>& node,
53 const std::string& var) {
54 auto pw = std::dynamic_pointer_cast<const PowerNode>(node);
55 if (!pw) { return false; }
56 if (!sf_is_fn_of_var(pw->base(), FunctionNode::FuncType::Ln, var)) { return false; }
57 auto en = std::dynamic_pointer_cast<const NumberNode>(pw->exponent());
58 if (!en) { return false; }
59 if (std::holds_alternative<BigInt>(en->value())) {
60 return std::get<BigInt>(en->value()) == BigInt(-1);
61 }
62 if (std::holds_alternative<Rational>(en->value())) {
63 return std::get<Rational>(en->value()) == Rational(-1);
64 }
65 if (std::holds_alternative<lmmc_real_t>(en->value())) {
66 lmmc_real_t d = std::get<lmmc_real_t>(en->value());
67 int eq = 0;
68 lmmc_double_nearly_equal_tol(d, -1.0, 1e-12, 1e-12, &eq);
69 return eq != 0;
70 }
71 return false;

74// Split a node into (factors, has_inv_var) where the inv-var factor is removed
75// from `factors` if present. Returns false if there is more than one inv-var
76// factor (which would be 1/x^2 and is not the form we handle here).
77bool sf_split_inv_var(const std::shared_ptr<const SymbolicNode>& node,
78 const std::string& var,
79 std::vector<std::shared_ptr<const SymbolicNode>>& other_factors,
80 bool& has_inv_var) {
81 other_factors.clear();
82 has_inv_var = false;
83 std::vector<std::shared_ptr<const SymbolicNode>> factors;
84 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
85 factors = mul->operands();
86 } else {
87 factors.push_back(node);
88 }
89 for (const auto& f : factors) {
90 if (sf_is_inv_var(f, var)) {
91 if (has_inv_var) { return false; } /** @brief 多个 1/x 因子形成 1/x^2 类幂次，超出此路径的处理范围。 */
92 has_inv_var = true;
93 } else {
94 other_factors.push_back(f);
95 }
96 }
97 return true;
98}

100// Detect exp(-c*var^2) where c is a constant w.r.t. var that is rational and
101// strictly positive. On success, sets `c_out` to that rational coefficient.
102//
103// We accept exp(arg) where the argument is a polynomial in var of degree
104// exactly 2, no constant or linear term, and the leading rational coefficient
105// is strictly negative. (We then return its absolute value as c.)
106Result<bool> sf_match_exp_neg_quad(const std::shared_ptr<const SymbolicNode>& node,
107 const std::string& var,
108 Rational& c_out) {
109 auto fn = std::dynamic_pointer_cast<const FunctionNode>(node);
110 if (!fn || fn->type() != FunctionNode::FuncType::Exp) { return false; }
111 if (fn->arguments().size() != 1) { return false; }
112 auto arg = LMCAS::detail::expression_from_node(fn->arguments()[0]);
113 auto converted = symbolic_to_poly<Rational>(LMCAS::detail::make_expression_ptr(arg), var);
114 if (!converted) {
115 if (converted.error().code == CasErrc::UnsupportedExpression) { return false; }
116 return Result<bool>::failure(converted.error());
117 }
118 auto poly = std::move(converted.value());
119 if (poly.degree() != 2) { return false; }
120 if (poly.coeffs.size() < 3) { return false; }
121 // Constant and linear terms must be zero.
122 if (!(poly.coeffs[0] == Rational(0))) { return false; }
123 if (!(poly.coeffs[1] == Rational(0))) { return false; }
124 Rational a2 = poly.coeffs[2];
125 // Need a strictly negative coefficient: exp(-c*x^2) with c > 0.
126 if (!(a2 < Rational(0))) { return false; }
127 c_out = Rational(0) - a2; // c = -a2 > 0
128 return true;
129}

133Result<std::shared_ptr<SymbolicExpr>> SpecialFunctionStrategy::try_integrate_raw(
134 const SymbolicExpr& expr, const std::string& var, Integrator& ctx,
135 ComputationContext&, int depth) {
136 (void)ctx;
137 (void)depth;
138
139 using FT = FunctionNode::FuncType;
140
141 auto v = SymbolicExpr::variable(var);
142
143 if (sf_is_inv_ln_var(LMCAS::detail::node(expr), var)) {
144 auto li = make_unary_function(FT::Li, v);
145 auto simp = li->simplify();
146 return simp ? simp : li;
147 }
148
149 {
150 Rational c_rat;
151 auto matched = sf_match_exp_neg_quad(LMCAS::detail::node(expr), var, c_rat);
152 if (!matched) {
153 return Result<std::shared_ptr<SymbolicExpr>>::failure(matched.error());
154 }
155 if (matched.value()) {
156 // Result: (sqrt(pi) / (2 * sqrt(c))) * erf(sqrt(c) * x)
157 auto sqrt_pi = sf_sqrt_pi();
158 auto sqrt_c = SymbolicExpr::sqrt(SymbolicExpr::number(c_rat));
159 auto scaled_x = SymbolicExpr::multiply(sqrt_c, v);
160 auto erf_node = make_unary_function(FT::Erf, scaled_x);
161
162 auto two = SymbolicExpr::number(2);
163 auto two_sqrt_c = SymbolicExpr::multiply(two, sqrt_c);
164 auto coeff = SymbolicExpr::multiply(
165 sqrt_pi,
166 SymbolicExpr::power(two_sqrt_c, SymbolicExpr::number(-1)));
167 auto result = SymbolicExpr::multiply(coeff, erf_node);
168 auto simp = result->simplify();
169 return simp ? simp : result;
170 }
171 }

173 // These all have shape (something)*1/x, where the "something" is a
174 // FunctionNode of var. Other 1/x patterns (e.g. 1/x alone, x*1/x) are
175 // not our concern.
176 {
177 std::vector<std::shared_ptr<const SymbolicNode>> others;
178 bool has_inv = false;
179 if (sf_split_inv_var(LMCAS::detail::node(expr), var, others, has_inv) && has_inv && others.size() == 1) {
180 const auto& other = others[0];
181 if (sf_is_fn_of_var(other, FT::Exp, var)) {
182 auto ei = make_unary_function(FT::Ei, v);
183 auto simp = ei->simplify();
184 return simp ? simp : ei;
185 }
186 if (sf_is_fn_of_var(other, FT::Sin, var)) {
187 auto si = make_unary_function(FT::Si, v);
188 auto simp = si->simplify();
189 return simp ? simp : si;
190 }
191 if (sf_is_fn_of_var(other, FT::Cos, var)) {
192 auto ci = make_unary_function(FT::Ci, v);
193 auto simp = ci->simplify();
194 return simp ? simp : ci;
195 }
196 }
197 }
