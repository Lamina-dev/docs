21// Test whether `node` is a single-argument FunctionNode of the given type
22// whose argument is exactly the integration variable.
23bool sf_is_fn_of_var(const std::shared_ptr<const SymbolicNode>& node,
24 FunctionNode::FuncType t,
25 const std::string& var) {
26 auto fn = std::dynamic_pointer_cast<const FunctionNode>(node);
27 if (!fn || fn->type() != t) return false;
28 if (fn->arguments().size() != 1) return false;
29 auto v = std::dynamic_pointer_cast<const VariableNode>(fn->arguments()[0]);
30 return v && v->name() == var;
31}

33// Detect a 1/x factor: a PowerNode whose base is the integration variable and
34// exponent is the integer -1.
35bool sf_is_inv_var(const std::shared_ptr<const SymbolicNode>& node,
36 const std::string& var) {
37 auto pw = std::dynamic_pointer_cast<const PowerNode>(node);
38 if (!pw) return false;
39 auto b = std::dynamic_pointer_cast<const VariableNode>(pw->base());
40 if (!b || b->name() != var) return false;
41 auto en = std::dynamic_pointer_cast<const NumberNode>(pw->exponent());
42 if (!en) return false;
43 if (std::holds_alternative<BigInt>(en->value())) {
44 return std::get<BigInt>(en->value()) == BigInt(-1);
45 }
46 if (std::holds_alternative<Rational>(en->value())) {
47 return std::get<Rational>(en->value()) == Rational(-1);
48 }
49 if (std::holds_alternative<lmmc_real_t>(en->value())) {
50 lmmc_real_t d = std::get<lmmc_real_t>(en->value());
51 int eq = 0;
52 lmmc_double_nearly_equal_tol(d, -1.0, 1e-12, 1e-12, &eq);
53 return eq != 0;
54 }
55 return false;
56}

58// Detect a 1/ln(var) factor, i.e. PowerNode(ln(var), -1).
59bool sf_is_inv_ln_var(const std::shared_ptr<const SymbolicNode>& node,
60 const std::string& var) {
61 auto pw = std::dynamic_pointer_cast<const PowerNode>(node);
62 if (!pw) return false;
63 if (!sf_is_fn_of_var(pw->base(), FunctionNode::FuncType::Ln, var)) return false;
64 auto en = std::dynamic_pointer_cast<const NumberNode>(pw->exponent());
65 if (!en) return false;
66 if (std::holds_alternative<BigInt>(en->value())) {
67 return std::get<BigInt>(en->value()) == BigInt(-1);
68 }
69 if (std::holds_alternative<Rational>(en->value())) {
70 return std::get<Rational>(en->value()) == Rational(-1);
71 }
72 if (std::holds_alternative<lmmc_real_t>(en->value())) {
73 lmmc_real_t d = std::get<lmmc_real_t>(en->value());
74 int eq = 0;
75 lmmc_double_nearly_equal_tol(d, -1.0, 1e-12, 1e-12, &eq);
76 return eq != 0;
77 }
78 return false;

81// Split a node into (factors, has_inv_var) where the inv-var factor is removed
82// from `factors` if present. Returns false if there is more than one inv-var
83// factor (which would be 1/x^2 and is not the form we handle here).
84bool sf_split_inv_var(const std::shared_ptr<const SymbolicNode>& node,
85 const std::string& var,
86 std::vector<std::shared_ptr<const SymbolicNode>>& other_factors,
87 bool& has_inv_var) {
88 other_factors.clear();
89 has_inv_var = false;
90 std::vector<std::shared_ptr<const SymbolicNode>> factors;
91 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
92 factors = mul->operands();
93 } else {
94 factors.push_back(node);
95 }
96 for (const auto& f : factors) {
97 if (sf_is_inv_var(f, var)) {
98 if (has_inv_var) return false; // multiple 1/x factors (1/x^2 form)
99 has_inv_var = true;
100 } else {
101 other_factors.push_back(f);
102 }
103 }
104 return true;
105}

107// Detect exp(-c*var^2) where c is a constant w.r.t. var that is rational and
108// strictly positive. On success, sets `c_out` to that rational coefficient.
109//
110// We accept exp(arg) where the argument is a polynomial in var of degree
111// exactly 2, no constant or linear term, and the leading rational coefficient
112// is strictly negative. (We then return its absolute value as c.)
113bool sf_match_exp_neg_quad(const std::shared_ptr<const SymbolicNode>& node,
114 const std::string& var,
115 Rational& c_out) {
116 auto fn = std::dynamic_pointer_cast<const FunctionNode>(node);
117 if (!fn || fn->type() != FunctionNode::FuncType::Exp) return false;
118 if (fn->arguments().size() != 1) return false;
119 auto arg = lamina::detail::expression_from_node(fn->arguments()[0]);
120 Polynomial<Rational> poly;
121 try {
122 poly = symbolic_to_poly<Rational>(lamina::detail::make_expression_ptr(arg), var);
123 } catch (const std::invalid_argument&) {
124 return false;
125 } catch (const std::out_of_range&) {
126 return false;
127 } catch (const std::runtime_error&) {
128 return false;
129 }
130 if (poly.degree() != 2) return false;
131 if (poly.coeffs.size() < 3) return false;
132 // Constant and linear terms must be zero.
133 if (!(poly.coeffs[0] == Rational(0))) return false;
134 if (!(poly.coeffs[1] == Rational(0))) return false;
135 Rational a2 = poly.coeffs[2];
136 // Need a strictly negative coefficient: exp(-c*x^2) with c > 0.
137 if (!(a2 < Rational(0))) return false;
138 c_out = Rational(0) - a2; // c = -a2 > 0
139 return true;
140}

144std::shared_ptr<SymbolicExpr> SpecialFunctionStrategy::try_integrate_raw(
145 const SymbolicExpr& expr, const std::string& var, Integrator& ctx,
146 ComputationContext&, int depth) {
147 (void)ctx;
148 (void)depth;
149
150 using FT = FunctionNode::FuncType;
151
152 auto v = SymbolicExpr::variable(var);
153
154 if (sf_is_inv_ln_var(lamina::detail::node(expr), var)) {
155 auto li = sf_make_fn(FT::Li, v);
156 auto simp = li->simplify();
157 return simp ? simp : li;
158 }
159
160 {
161 Rational c_rat;
162 if (sf_match_exp_neg_quad(lamina::detail::node(expr), var, c_rat)) {
163 // Result: (sqrt(pi) / (2 * sqrt(c))) * erf(sqrt(c) * x)
164 auto sqrt_pi = sf_sqrt_pi();
165 auto sqrt_c = SymbolicExpr::sqrt(SymbolicExpr::number(c_rat));
166 auto scaled_x = SymbolicExpr::multiply(sqrt_c, v);
167 auto erf_node = sf_make_fn(FT::Erf, scaled_x);
168
169 auto two = SymbolicExpr::number(2);
170 auto two_sqrt_c = SymbolicExpr::multiply(two, sqrt_c);
171 auto coeff = SymbolicExpr::multiply(
172 sqrt_pi,
173 SymbolicExpr::power(two_sqrt_c, SymbolicExpr::number(-1)));
174 auto result = SymbolicExpr::multiply(coeff, erf_node);
175 auto simp = result->simplify();
176 return simp ? simp : result;
177 }
178 }
179
180 // These all have shape (something)*1/x, where the "something" is a
181 // FunctionNode of var. Other 1/x patterns (e.g. 1/x alone, x*1/x) are
182 // not our concern.
183 {
184 std::vector<std::shared_ptr<const SymbolicNode>> others;
185 bool has_inv = false;
186 if (sf_split_inv_var(lamina::detail::node(expr), var, others, has_inv) && has_inv && others.size() == 1) {
187 const auto& other = others[0];
188 if (sf_is_fn_of_var(other, FT::Exp, var)) {
189 auto ei = sf_make_fn(FT::Ei, v);
190 auto simp = ei->simplify();
191 return simp ? simp : ei;
192 }
193 if (sf_is_fn_of_var(other, FT::Sin, var)) {
194 auto si = sf_make_fn(FT::Si, v);
195 auto simp = si->simplify();
196 return simp ? simp : si;
197 }
198 if (sf_is_fn_of_var(other, FT::Cos, var)) {
199 auto ci = sf_make_fn(FT::Ci, v);
200 auto simp = ci->simplify();
201 return simp ? simp : ci;
202 }
203 }
204 }
205
206 return nullptr;
207}
