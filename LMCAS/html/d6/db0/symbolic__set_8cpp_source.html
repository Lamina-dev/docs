26bool exact_value(const std::shared_ptr<const SymbolicNode>& node, Rational& value) {
27 auto number = std::dynamic_pointer_cast<const NumberNode>(node);
28 if (!number) return false;
29 if (std::holds_alternative<BigInt>(number->value())) {

53SymbolicSetResult make_finite_set(
54 std::vector<std::shared_ptr<SymbolicExpr>> elements,
55 ComputationContext& context) {
56 auto step = consume(context);
57 if (!step) return SymbolicSetResult::failure(step.error());
58 std::vector<std::shared_ptr<const SymbolicNode>> nodes;
59 nodes.reserve(elements.size());
60 for (const auto& element : elements) {
61 if (!element || !detail::node(element)) {
62 return failure(CasErrc::InvalidArgument,
63 "finite set elements cannot be null");
64 }
65 nodes.push_back(detail::node(element));
66 }
67 return from_nodes(std::move(nodes));

70SymbolicSetResult make_interval(
71 const std::shared_ptr<SymbolicExpr>& lower,
72 const std::shared_ptr<SymbolicExpr>& upper,
73 bool lower_closed,
74 bool upper_closed,
75 ComputationContext& context) {
76 auto step = consume(context);
77 if (!step) return SymbolicSetResult::failure(step.error());
78 if (!lower || !upper || !detail::node(lower) || !detail::node(upper)) {
79 return failure(CasErrc::InvalidArgument, "interval endpoints cannot be null");
80 }
81 Rational lower_value;
82 Rational upper_value;
83 if (exact_value(detail::node(lower), lower_value) &&
84 exact_value(detail::node(upper), upper_value) &&
85 upper_value < lower_value) {
86 return failure(CasErrc::DomainError,
87 "interval lower endpoint exceeds upper endpoint");
88 }
89 try {
90 return SymbolicSetResult::success(detail::make_expression_ptr(

98SymbolicSetResult make_membership(
99 const std::shared_ptr<SymbolicExpr>& element,
100 const std::shared_ptr<SymbolicExpr>& set,
101 ComputationContext& context) {
102 auto step = consume(context);
103 if (!step) return SymbolicSetResult::failure(step.error());
104 if (!element || !set || !detail::node(element) || !detail::node(set)) {
105 return failure(CasErrc::InvalidArgument, "membership operands cannot be null");
106 }
107 if (auto finite = finite_set_node(*set)) {
108 return SymbolicSetResult::success(SymbolicExpr::number(
109 finite->contains(*detail::node(element)) ? 1 : 0));
110 }
111 if (auto interval = std::dynamic_pointer_cast<const IntervalNode>(detail::node(set))) {
112 Rational value;
113 Rational lower;
114 Rational upper;
115 if (exact_value(detail::node(element), value) &&
116 exact_value(interval->lower(), lower) && exact_value(interval->upper(), upper)) {
117 const bool above = lower < value || (interval->lower_closed() && lower == value);
118 const bool below = value < upper || (interval->upper_closed() && value == upper);
119 return SymbolicSetResult::success(SymbolicExpr::number(above && below ? 1 : 0));
120 }
121 }
122 return SymbolicSetResult::success(detail::make_expression_ptr(
123 detail::make_node<MembershipNode>(detail::node(element), detail::node(set))));
124}

126SymbolicSetResult finite_set_union(
127 const SymbolicExpr& lhs, const SymbolicExpr& rhs,
128 ComputationContext& context) {
129 auto step = consume(context);
130 if (!step) return SymbolicSetResult::failure(step.error());
131 auto left = finite_set_node(lhs);

134 "set union requires two finite sets");

137 return from_nodes(std::move(elements));
