27bool exact_value(const std::shared_ptr<const SymbolicNode>& node, Rational& value) {
28 auto number = std::dynamic_pointer_cast<const NumberNode>(node);
29 if (!number) return false;
30 if (std::holds_alternative<BigInt>(number->value())) {
31 value = Rational(std::get<BigInt>(number->value()));
32 return true;
33 }
34 if (std::holds_alternative<Rational>(number->value())) {
35 value = std::get<Rational>(number->value());
36 return true;
37 }
38 return false;

52std::optional<bool> exact_interval_membership(
53 const std::shared_ptr<const SymbolicNode>& element, const IntervalNode& interval) {
54 Rational value;
55 Rational lower;
56 Rational upper;
57 if (!exact_value(element, value)) {
58 return std::nullopt;
59 }
60 if (!exact_value(interval.lower(), lower) || !exact_value(interval.upper(), upper)) {
61 return std::nullopt;
62 }
63 if (value < lower || upper < value) {
64 return false;
65 }
66 if (value == lower && !interval.lower_closed()) {
67 return false;
68 }
69 return value != upper || interval.upper_closed();
70}

74SymbolicSetResult make_finite_set(
75 std::vector<std::shared_ptr<SymbolicExpr>> elements,
76 ComputationContext& context) {
77 auto step = consume(context);
78 if (!step) return SymbolicSetResult::failure(step.error());
79 std::vector<std::shared_ptr<const SymbolicNode>> nodes;
80 nodes.reserve(elements.size());
81 for (const auto& element : elements) {
82 if (!element || !detail::node(element)) {

86 nodes.push_back(detail::node(element));
87 }
88 return from_nodes(std::move(nodes));

91SymbolicSetResult make_interval(
92 const std::shared_ptr<SymbolicExpr>& lower,
93 const std::shared_ptr<SymbolicExpr>& upper,
94 bool lower_closed,
95 bool upper_closed,
96 ComputationContext& context) {
97 auto step = consume(context);
98 if (!step) {
99 return SymbolicSetResult::failure(step.error());
100 }
101 if (!lower || !upper || !detail::node(lower) || !detail::node(upper)) {
102 return failure(CasErrc::InvalidArgument, "interval endpoints cannot be null");
103 }
104 Rational lower_value;
105 Rational upper_value;
106 if (exact_value(detail::node(lower), lower_value) &&
107 exact_value(detail::node(upper), upper_value) &&
108 upper_value < lower_value) {
109 return failure(CasErrc::InvalidArgument,
110 "interval lower endpoint exceeds upper endpoint");
111 }
112 try {
113 return SymbolicSetResult::success(detail::make_expression_ptr(
114 detail::make_node<IntervalNode>(detail::node(lower), detail::node(upper),
115 lower_closed, upper_closed)));
116 } catch (const std::bad_alloc&) {
117 return failure(CasErrc::ResourceLimit, "interval allocation failed");
118 }

121SymbolicSetResult make_membership(
122 const std::shared_ptr<SymbolicExpr>& element,
123 const std::shared_ptr<SymbolicExpr>& set,
124 ComputationContext& context) {
125 auto step = consume(context);
126 if (!step) {
127 return SymbolicSetResult::failure(step.error());
128 }
129 if (!element || !set || !detail::node(element) || !detail::node(set)) {
130 return failure(CasErrc::InvalidArgument, "membership operands cannot be null");
131 }
132 if (auto finite = finite_set_node(*set)) {
133 return SymbolicSetResult::success(SymbolicExpr::number(
134 finite->contains(*detail::node(element)) ? 1 : 0));
135 }
136 if (auto interval = std::dynamic_pointer_cast<const IntervalNode>(detail::node(set))) {
137 if (auto contained = exact_interval_membership(detail::node(element), *interval)) {
138 return SymbolicSetResult::success(SymbolicExpr::number(*contained ? 1 : 0));
139 }
140 }
141 return SymbolicSetResult::success(detail::make_expression_ptr(
142 detail::make_node<MembershipNode>(detail::node(element), detail::node(set))));
143}

146 const SymbolicExpr& lhs, const SymbolicExpr& rhs,
147 ComputationContext& context) {
