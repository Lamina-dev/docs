15 const std::shared_ptr<SymbolicExpr>& lower,
16 const std::shared_ptr<SymbolicExpr>& upper,
17 ComputationContext& context,
18 const std::string& operation)
19{
20 auto step = context.consume_steps(1, operation);

36ExpressionResult simplify_geometry_checked(std::shared_ptr<SymbolicExpr> expr,
37 const std::string& operation,
38 const std::string& message, ComputationContext& context)
39try {
40
41 detail::RewriteBudget transform_budget(context, (context).limits().max_recursion_depth,
42 (context).limits().max_ast_nodes, "transform");
43 if (!expr || !LMCAS::detail::node(expr)) {
44 return ExpressionResult::failure(CasErrc::Inconclusive,
45 message, operation);
46 }
47 auto simplified = detail::make_expression_ptr(detail::normalize_node(detail::node(expr), transform_budget));
48 if (!simplified || !LMCAS::detail::node(simplified)) {
49 return ExpressionResult::failure(CasErrc::Inconclusive,
50 message, operation);
51 }

133std::shared_ptr<SymbolicExpr> squared_profile(const std::shared_ptr<SymbolicExpr>& f, ComputationContext& context) {
134
135 detail::RewriteBudget transform_budget(context, (context).limits().max_recursion_depth,
136 (context).limits().max_ast_nodes, "transform");
137 if (f) {
138 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(LMCAS::detail::node(f))) {
139 if (func->type() == FunctionNode::FuncType::Sqrt && func->arguments().size() == 1) {
140 return detail::make_expression_ptr(detail::normalize_node(func->arguments()[0], transform_budget));
141 }
142 }
143 if (auto power = std::dynamic_pointer_cast<const PowerNode>(LMCAS::detail::node(f))) {
144 auto exponent = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
145 if (exponent && std::holds_alternative<Rational>(exponent->value())) {
146 const auto& rational = std::get<Rational>(exponent->value());
147 if (rational.get_numerator() == BigInt(1) &&
148 rational.get_denominator() == BigInt(2)) {
149 return detail::make_expression_ptr(detail::normalize_node(power->base(), transform_budget));
150 }
151 }
152 }
153 }
154 return detail::make_expression_ptr(detail::normalize_node(detail::node(SymbolicExpr::power(f, SymbolicExpr::number(2))), transform_budget));
155}
