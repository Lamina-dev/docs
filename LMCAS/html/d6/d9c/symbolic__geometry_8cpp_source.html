38ExpressionResult simplify_geometry_checked(std::shared_ptr<SymbolicExpr> expr,
39 const std::string& operation,
40 const std::string& message)
41{
42 if (!expr || !lamina::detail::node(expr)) {
43 return ExpressionResult::failure(CasErrc::Inconclusive,
44 message, operation);

46 auto simplified = expr->simplify();

77ExpressionResult definite_integral_geometry_checked(
78 const std::shared_ptr<SymbolicExpr>& integrand,
79 const std::string& variable,
80 const std::shared_ptr<SymbolicExpr>& lower_bound,
81 const std::shared_ptr<SymbolicExpr>& upper_bound,
82 const std::string& operation)
83{
84 if (!integrand || !lamina::detail::node(integrand)) {
85 return ExpressionResult::failure(
86 CasErrc::Inconclusive,
87 "geometry integrand is outside the supported domain",
88 operation);
89 }
90 auto integral = integrand->integrate(variable);
91 if (!integral || !lamina::detail::node(integral) ||
92 contains_unevaluated_integral(lamina::detail::node(integral))) {
93 return ExpressionResult::failure(
94 CasErrc::Inconclusive,
95 "geometry integral is outside the supported domain",
96 operation);
97 }
98 auto upper = integral->substitute(variable, upper_bound);
99 auto lower = integral->substitute(variable, lower_bound);
100 auto result = SymbolicExpr::add(
101 upper,
102 SymbolicExpr::multiply(SymbolicExpr::number(-1), lower));
103 return simplify_geometry_checked(
104 result, operation,
105 "geometry definite integral result is outside the supported domain");
106}

108std::shared_ptr<SymbolicExpr> squared_profile(const std::shared_ptr<SymbolicExpr>& f) {
109 if (f) {
110 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(f))) {
111 if (func->type() == FunctionNode::FuncType::Sqrt && func->arguments().size() == 1) {
112 return lamina::detail::make_expression_ptr(func->arguments()[0])->simplify();
113 }
114 }
115 if (auto power = std::dynamic_pointer_cast<const PowerNode>(lamina::detail::node(f))) {
116 auto exponent = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
117 if (exponent && std::holds_alternative<Rational>(exponent->value())) {
118 const auto& rational = std::get<Rational>(exponent->value());
119 if (rational.get_numerator() == BigInt(1) &&
120 rational.get_denominator() == BigInt(2)) {
121 return lamina::detail::make_expression_ptr(power->base())->simplify();
122 }
123 }
124 }
125 }
126 return SymbolicExpr::power(f, SymbolicExpr::number(2))->simplify();
127}
