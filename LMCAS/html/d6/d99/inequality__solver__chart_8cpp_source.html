65static std::vector<Interval> select_open_intervals(
66 const std::vector<std::shared_ptr<SymbolicExpr>>& roots,
67 const std::vector<int>& signs, int target_sign) {
68 std::vector<Interval> intervals;
69 for (size_t index = 0; index < signs.size(); ++index) {
70 if (signs[index] != target_sign) {
71 continue;
72 }
73 Interval interval;
74 interval.lower = index == 0 ? Endpoint::neg_inf() : Endpoint::open(roots[index - 1]);
75 interval.upper = index == roots.size() ? Endpoint::pos_inf() : Endpoint::open(roots[index]);
76 intervals.push_back(interval);
77 }
