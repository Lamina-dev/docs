19std::shared_ptr<SymbolicExpr> rd_integrate_polynomial(
20 const Polynomial<Rational>& quot, const std::string& var) {
21 auto result = SymbolicExpr::number(0);
22 if (!quot.is_zero()) {
23 for (size_t k = 0; k < quot.coeffs.size(); ++k) {
24 if (quot.coeffs[k] == Rational(0)) { continue; }
25 const BigInt integrated_power = BigInt(k) + BigInt(1);
26 Rational coeff = quot.coeffs[k] / Rational(integrated_power);
27 std::shared_ptr<SymbolicExpr> term;
28 auto v = SymbolicExpr::variable(var);
29 auto pw = SymbolicExpr::power(v, SymbolicExpr::number(integrated_power));
30 if (coeff == Rational(1)) {
31 term = pw;
32 } else {
33 term = SymbolicExpr::multiply(SymbolicExpr::number(coeff), pw);
34 }
35 result = SymbolicExpr::add(result, term);
36 }
37 }
38 return result;
39}

67Result<detail::IntegralParts> rd_integrate_proper_fraction(
68 const std::string& var,
69 const Polynomial<Rational>& P, const Polynomial<Rational>& Q,
70 const SymbolicExpr& zero, ComputationContext& computation) {
71 Polynomial<Rational> quot(Q.variable_name), rem = P;
72 if (P.degree() >= Q.degree()) {
73 auto divided = P.div_mod(Q);
74 quot = std::move(divided.first);
75 rem = std::move(divided.second);
76 }
77
78 detail::IntegralParts result{*rd_integrate_polynomial(quot, var), zero};
79 if (rem.is_zero()) { return result; }
80
81 std::vector<std::pair<Polynomial<Rational>, int>> factors;
82 if (!detail::factor_denominator(Q, factors)) {
83 result.remainder = *SymbolicExpr::divide(poly_to_symbolic(rem), poly_to_symbolic(Q));
84 return result;
85 }
86
87 std::vector<Polynomial<Rational>> numerators;
88 auto coefficients = detail::solve_coefficients(rem, Q, factors, numerators, computation);
89 if (!coefficients) {
90 return Result<detail::IntegralParts>::failure(coefficients.error());
91 }
92 if (!coefficients.value()) {
93 result.remainder = *SymbolicExpr::divide(poly_to_symbolic(rem), poly_to_symbolic(Q));
94 return result;
95 }
96 return rd_add_fraction_primitives(var, factors, numerators, std::move(result), computation);
97}
