16// Build (var - r) as a SymbolicExpr.
17inline std::shared_ptr<SymbolicExpr> rd_var_minus(const std::string& var, const Rational& r) {
18 auto v = SymbolicExpr::variable(var);
19 if (r == Rational(0)) return v;
20 auto neg_r = SymbolicExpr::number(Rational(0) - r);
21 return SymbolicExpr::add(v, neg_r);

24// Test whether a rational-coefficient poly is the zero polynomial.
25inline bool rd_is_zero_poly(const Polynomial<Rational>& p) {
26 if (p.coeffs.empty()) return true;
27 for (const auto& c : p.coeffs) if (!(c == Rational(0))) return false;
28 return true;

31// Convert a Polynomial<Rational> into a SymbolicExpr in the given variable.
32inline std::shared_ptr<SymbolicExpr> rd_poly_to_sym(
33 const Polynomial<Rational>& p, const std::string& var) {
34 if (rd_is_zero_poly(p)) return SymbolicExpr::number(0);
35 std::vector<std::shared_ptr<SymbolicExpr>> terms;
36 auto v = SymbolicExpr::variable(var);
37 for (size_t i = 0; i < p.coeffs.size(); ++i) {
38 if (p.coeffs[i] == Rational(0)) continue;
39 std::shared_ptr<SymbolicExpr> t;
40 if (i == 0) {
41 t = rd_num_rat(p.coeffs[i]);
42 } else if (i == 1) {
43 if (p.coeffs[i] == Rational(1)) {
44 t = v;
45 } else {
46 t = SymbolicExpr::multiply(rd_num_rat(p.coeffs[i]), v);
47 }
48 } else {
49 auto pw = SymbolicExpr::power(v, rd_num_int(static_cast<long long>(i)));
50 if (p.coeffs[i] == Rational(1)) {
51 t = pw;
52 } else {
53 t = SymbolicExpr::multiply(rd_num_rat(p.coeffs[i]), pw);
54 }
55 }
56 terms.push_back(t);
57 }
58 if (terms.empty()) return SymbolicExpr::number(0);
59 if (terms.size() == 1) return terms[0];
60 auto res = terms[0];
61 for (size_t i = 1; i < terms.size(); ++i) res = SymbolicExpr::add(res, terms[i]);
62 return res;
63}

65// Recursively decompose a SymbolicNode into (numerator-polys, denominator-polys)
66// over the rationals. Each polynomial is assumed to be a polynomial in `var`
67// with rational coefficients. Returns false on any non-rational sub-expression
68// (functions, irrational/symbolic exponents, variables other than the integration
69// variable). Multiplication aggregates by appending; division comes from
70// PowerNode with negative integer exponent. Numbers and the integration
71// variable are converted directly via symbolic_to_poly.
72bool rd_collect_rational(const std::shared_ptr<const SymbolicNode>& node,
73 const std::string& var,
74 std::vector<Polynomial<Rational>>& num,
75 std::vector<Polynomial<Rational>>& den) {
76 if (!node) return false;
77
78 // A NumberNode is a constant; convert directly.
79 if (auto n = std::dynamic_pointer_cast<const NumberNode>(node)) {
80 Polynomial<Rational> p =
81 symbolic_to_poly<Rational>(lamina::detail::make_expression_ptr(n), var);
82 num.push_back(p);
83 return true;
84 }
85
86 // The integration variable, or any other variable that's actually a constant.
87 if (auto v = std::dynamic_pointer_cast<const VariableNode>(node)) {
88 if (v->name() == var) {
89 num.push_back(Polynomial<Rational>({Rational(0), Rational(1)}, var));
90 return true;
91 }
92 // Unknown symbolic constant: cannot lift safely into rational poly.
93 return false;
94 }
95
96 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
97 // Addition: combine all summands over a common denominator.
98 // We collect each operand as P_i/Q_i and assemble
99 // sum P_i * (prod_{j != i} Q_j) / prod_j Q_j
100 std::vector<std::pair<std::vector<Polynomial<Rational>>,
101 std::vector<Polynomial<Rational>>>> parts;
102 parts.reserve(add->operands().size());
103 for (const auto& op : add->operands()) {
104 std::vector<Polynomial<Rational>> sub_num, sub_den;
105 if (!rd_collect_rational(op, var, sub_num, sub_den)) return false;
106 parts.push_back({std::move(sub_num), std::move(sub_den)});
107 }
108
109 // Combine: numerator = sum_i (prod_k num_k_i) * prod_{j != i} (prod_k den_k_j)
110 // denominator = prod_i (prod_k den_k_i)
111 Polynomial<Rational> total_num(var); // start as 0
112 Polynomial<Rational> total_den({Rational(1)}, var);
113
114 // Pre-compute Q_i (denominator product per term) and total denominator.
115 std::vector<Polynomial<Rational>> Q_per_term;
116 Q_per_term.reserve(parts.size());
117 for (const auto& [pn, pd] : parts) {
118 Polynomial<Rational> q({Rational(1)}, var);
119 for (const auto& d : pd) q = q * d;
120 Q_per_term.push_back(q);
121 total_den = total_den * q;
122 }
123
124 for (size_t i = 0; i < parts.size(); ++i) {
125 Polynomial<Rational> p({Rational(1)}, var);
126 for (const auto& nn : parts[i].first) p = p * nn;
127 // multiply by prod_{j != i} Q_j
128 for (size_t j = 0; j < parts.size(); ++j) {
129 if (j == i) continue;
130 p = p * Q_per_term[j];
131 }
132 total_num = total_num + p;
133 }
134
135 num.push_back(total_num);
136 den.push_back(total_den);
137 return true;
138 }
139
140 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
141 for (const auto& op : mul->operands()) {
142 if (!rd_collect_rational(op, var, num, den)) return false;
143 }
144 return true;
145 }
146
147 if (auto pw = std::dynamic_pointer_cast<const PowerNode>(node)) {
148 // Only integer exponents (negative or non-negative) are allowed.
149 auto en = std::dynamic_pointer_cast<const NumberNode>(pw->exponent());
150 if (!en) return false;
151
152 long long exp_v = 0;
153 bool ok = false;
154 if (std::holds_alternative<BigInt>(en->value())) {
155 const auto& bi = std::get<BigInt>(en->value());
156 // Bound to keep matrix sizes sane.
157 int v = bi.to_int();
158 if (v >= -64 && v <= 64) { ok = true; exp_v = v; }
159 } else if (std::holds_alternative<Rational>(en->value())) {
160 const auto& r = std::get<Rational>(en->value());
161 if (r.is_integer()) {
162 int v = r.to_BigInt().to_int();
163 if (v >= -64 && v <= 64) { ok = true; exp_v = v; }
164 }
165 } else if (std::holds_alternative<lmmc_real_t>(en->value())) {
166 lmmc_real_t d = std::get<lmmc_real_t>(en->value());
167 if (std::isfinite(d) && d == std::floor(d) && d >= -64.0 && d <= 64.0) {
168 ok = true; exp_v = static_cast<long long>(d);
169 }
170 }
171 if (!ok) return false;
172
173 // Build the base polynomial.
174 std::vector<Polynomial<Rational>> bn, bd;
175 if (!rd_collect_rational(pw->base(), var, bn, bd)) return false;
176
177 // base_num = product of bn ; base_den = product of bd
178 Polynomial<Rational> base_num({Rational(1)}, var);
179 for (const auto& p : bn) base_num = base_num * p;
180 Polynomial<Rational> base_den({Rational(1)}, var);
181 for (const auto& p : bd) base_den = base_den * p;
182
183 if (exp_v == 0) {
184 num.push_back(Polynomial<Rational>({Rational(1)}, var));
185 return true;
186 }
187 long long e_abs = std::llabs(exp_v);
188 Polynomial<Rational> n_pow({Rational(1)}, var);
189 Polynomial<Rational> d_pow({Rational(1)}, var);
190 for (long long i = 0; i < e_abs; ++i) {
191 n_pow = n_pow * base_num;
192 d_pow = d_pow * base_den;
193 }
194 if (exp_v > 0) {
195 num.push_back(n_pow);
196 den.push_back(d_pow);
197 } else {
198 // (P/Q)^(-k) = (Q/P)^k
199 num.push_back(d_pow);
200 den.push_back(n_pow);
201 }
202 return true;
203 }
204
205 // FunctionNode etc. -> not a rational function in `var`.
206 return false;
207}

211bool RationalDecompositionStrategy::extract_rational(
212 const SymbolicExpr& expr, const std::string& var,
213 Polynomial<Rational>& P_out, Polynomial<Rational>& Q_out) {
214
215 std::vector<Polynomial<Rational>> nums, dens;
216 if (!rd_collect_rational(lamina::detail::node(expr), var, nums, dens)) return false;
217
218 Polynomial<Rational> P({Rational(1)}, var);
219 for (const auto& p : nums) P = P * p;
220 Polynomial<Rational> Q({Rational(1)}, var);
221 for (const auto& p : dens) Q = Q * p;
222
223 if (rd_is_zero_poly(Q)) return false; // 0 in denominator, refuse
224
225 // Reduce by GCD to keep things small (and keep variable name consistent).
226 if (!rd_is_zero_poly(P)) {
227 Polynomial<Rational> g = Polynomial<Rational>::gcd(P, Q);
228 if (!rd_is_zero_poly(g) && g.degree() >= 1) {
229 auto pq = P.div_mod(g);
230 auto qq = Q.div_mod(g);
231 if (rd_is_zero_poly(pq.second) && rd_is_zero_poly(qq.second)) {
232 P = pq.first;
233 Q = qq.first;
234 }
235 }
236 }
237
238 // Make Q monic (rescale numerator accordingly).
239 if (Q.degree() >= 0) {
240 Rational lc = Q.lead_coeff();
241 if (!(lc == Rational(1)) && !(lc == Rational(0))) {
242 Polynomial<Rational> P_scaled(var);
243 P_scaled.coeffs.reserve(P.coeffs.size());
244 for (const auto& c : P.coeffs) P_scaled.coeffs.push_back(c / lc);
245 Polynomial<Rational> Q_scaled(var);
246 Q_scaled.coeffs.reserve(Q.coeffs.size());
247 for (const auto& c : Q.coeffs) Q_scaled.coeffs.push_back(c / lc);
248 P_scaled.trim();
249 Q_scaled.trim();
250 P = P_scaled;
251 Q = Q_scaled;
252 }
253 }
254
255 P_out = P;
256 Q_out = Q;
257 return true;
258}

265 quotient_out = qr.first;

269bool RationalDecompositionStrategy::factor_denominator(
270 const Polynomial<Rational>& Q,
271 std::vector<std::pair<Polynomial<Rational>, int>>& factors_out) {
272 factors_out.clear();
273 if (Q.degree() <= 0) return false;
274
275 // Step 1: square-free factorization to peel off multiplicities.
276 auto sqfree = square_free_factorization(Q);
277 if (sqfree.empty()) return false;
278
279 // Step 2: for each square-free factor, peel rational roots, leaving an
280 // irreducible-quadratic cofactor at most. Anything that cannot be
281 // reduced to factors of degree <= 2 forces this step to fail.
282 for (auto& [piece, mult] : sqfree) {
283 Polynomial<Rational> current = piece.make_monic();
284
285 if (current.degree() == 0) {
286 // pure constant factor; nothing to integrate, ignore
287 continue;
288 }
289 if (current.degree() == 1 || current.degree() == 2) {
290 // Irreducible quadratics can stay as-is; quadratic discriminant
291 // determines splitting later in solve_coefficients/integrate_term.
292 // For deg 2 we still try rational roots to split into two linears
293 // if possible.
294 if (current.degree() == 2) {
295 auto roots = find_rational_roots(current);
296 if (!roots.empty()) {
297 for (const auto& r : roots) {
298 Polynomial<Rational> linear({Rational(0) - r, Rational(1)}, current.variable_name);
299 auto qr = current.div_mod(linear);
300 if (!rd_is_zero_poly(qr.second)) {
301 return false;
302 }
303 factors_out.push_back({linear, mult});
304 current = qr.first;
305 if (current.degree() == 0) break;
306 }
307 }
308 if (current.degree() == 2) {
309 // Confirmed irreducible quadratic over Q.
310 factors_out.push_back({current.make_monic(), mult});
311 current = Polynomial<Rational>({Rational(1)}, current.variable_name);
312 } else if (current.degree() == 1) {
313 factors_out.push_back({current.make_monic(), mult});
314 current = Polynomial<Rational>({Rational(1)}, current.variable_name);
315 }
316 continue;
317 }
318 // degree 1
319 factors_out.push_back({current, mult});
320 continue;
321 }
322
323 // degree >= 3: try to split off rational linear factors.
324 auto roots = find_rational_roots(current);
325 for (const auto& r : roots) {
326 Polynomial<Rational> linear({Rational(0) - r, Rational(1)}, current.variable_name);
327 auto qr = current.div_mod(linear);
328 if (rd_is_zero_poly(qr.second)) {
329 factors_out.push_back({linear, mult});
330 current = qr.first;
331 if (current.degree() <= 0) break;
332 }
333 }
334
335 if (current.degree() <= 0) continue;
336 if (current.degree() == 1) {
337 factors_out.push_back({current.make_monic(), mult});
338 continue;
339 }
340 if (current.degree() == 2) {
341 factors_out.push_back({current.make_monic(), mult});
342 continue;
343 }
344 // Higher-degree irreducible factor over Q: cannot handle in this
345 // strategy. Caller will return an unevaluated integral.
346 return false;
347 }
348 return true;
349}

351bool RationalDecompositionStrategy::solve_coefficients(
352 const Polynomial<Rational>& P,
353 const Polynomial<Rational>& Q,
354 const std::vector<std::pair<Polynomial<Rational>, int>>& factors,
355 std::vector<Polynomial<Rational>>& numerators_out) {
356
357 numerators_out.clear();
358
359 // Map each (factor index, power) to a list of unknowns.
360 // Linear factor (x - r)^k contributes k unknowns A_1..A_k (constant numerators).
361 // Quadratic factor q(x)^k contributes 2k unknowns: (B_l, C_l) per power l.
362 //
363 // We also remember, for each (factor index, power l), the partial-fraction
364 // term's *full* numerator polynomial (constant or linear) as a function of
365 // the unknowns. Then we multiply each term by the missing portion of Q to
366 // get a polynomial whose coefficients (in x) are linear combinations of the
367 // unknowns; equating to P gives the linear system.
368
369 // Total number of unknowns.
370 size_t num_unknowns = 0;
371 std::vector<std::pair<int, int>> ifac_ipow; // (factor_index, power index l in [1..mult])
372 std::vector<int> kind_per_unknown_block; // 1 -> linear, 2 -> quadratic
373 std::vector<size_t> unknown_offset_per_term;
374 unknown_offset_per_term.reserve(factors.size());
375
376 for (size_t i = 0; i < factors.size(); ++i) {
377 unknown_offset_per_term.push_back(num_unknowns);
378 const auto& [fpoly, mult] = factors[i];
379 if (fpoly.degree() == 1) {
380 num_unknowns += static_cast<size_t>(mult);
381 } else if (fpoly.degree() == 2) {
382 num_unknowns += static_cast<size_t>(2 * mult);
383 } else {
384 return false;
385 }
386 }
387 if (num_unknowns == 0) return false;
388
389 const std::string& var = Q.variable_name;
390 int N = Q.degree();
391 if (N < 0) return false;
392
393 // Build the matrix: each column is the contribution (coefficient vector
394 // of x^0..x^{N-1}) of one unknown to the LHS polynomial; each row is one
395 // equation indexed by the power of x.
396 // Number of equations = N (size of polynomial of degree < N for the RHS P
397 // when properly partial-fractioned: deg(P) < N).
398 size_t rows = static_cast<size_t>(N);
399 size_t cols = num_unknowns + 1; // augmented column for P
400
401 // Initialize augmented matrix as Rational.
402 std::vector<std::vector<Rational>> M(rows, std::vector<Rational>(cols, Rational(0)));
403
404 // RHS column: coefficients of P (degree < N expected; if deg(P) >= N
405 // then we have a problem — caller must run poly_divide first).
406 if (P.degree() >= N) return false;
407 for (size_t k = 0; k < rows; ++k) {
408 Rational c = (k < P.coeffs.size()) ? P.coeffs[k] : Rational(0);
409 M[k][num_unknowns] = c;
410 }
411
412 // For each factor, for each multiplicity power l, build the contribution
413 // poly = numerator_term * (Q / fpoly^l). Each unknown in this term
414 // contributes one column.
415 for (size_t i = 0; i < factors.size(); ++i) {
416 const auto& [fpoly, mult] = factors[i];
417 // Q / fpoly^l for l = 1..mult; build incrementally: start with Q/fpoly^mult
418 // by dividing Q by fpoly mult times.
419 Polynomial<Rational> remaining = Q;
420 for (int t = 0; t < mult; ++t) {
421 auto qr = remaining.div_mod(fpoly);
422 if (!rd_is_zero_poly(qr.second)) {
423 // Q does not divide cleanly by this power of factor -> bug.
424 return false;
425 }
426 remaining = qr.first;
427 }
428 // remaining now equals Q / fpoly^mult.
429
430 // Now build for each power l = 1..mult:
431 // term coefficient pattern is multiplied by Q / fpoly^l
432 // = remaining * fpoly^(mult - l)
433 Polynomial<Rational> fp_pow({Rational(1)}, var);
434 // fp_pow starts as fpoly^0 = 1; we will iterate l from mult down to 1
435 // so we increment fp_pow by * fpoly each time after using it.
436 // Pre-build the array of coefficient polynomials for each l in 1..mult.
437 std::vector<Polynomial<Rational>> q_over_fpow_l(mult + 1, Polynomial<Rational>(var));
438 // index by l from 1..mult: q_over_fpow_l[l] = remaining * fpoly^(mult - l)
439 Polynomial<Rational> cur({Rational(1)}, var);
440 for (int l = mult; l >= 1; --l) {
441 // when l = mult: cur = fpoly^0 = 1, factor = remaining * 1 = remaining
442 // when decreasing l: factor = remaining * fpoly^(mult - l)
443 Polynomial<Rational> factor_poly = remaining * cur;
444 q_over_fpow_l[l] = factor_poly;
445 cur = cur * fpoly;
446 }
447
448 size_t col_off = unknown_offset_per_term[i];
449
450 if (fpoly.degree() == 1) {
451 // Linear factor. Each l contributes one unknown A_l (constant
452 // numerator). Coefficient column = q_over_fpow_l[l].
453 for (int l = 1; l <= mult; ++l) {
454 const auto& qpl = q_over_fpow_l[l];
455 size_t this_col = col_off + static_cast<size_t>(l - 1);
456 for (size_t k = 0; k < rows; ++k) {
457 Rational c = (k < qpl.coeffs.size()) ? qpl.coeffs[k] : Rational(0);
458 M[k][this_col] = c;
459 }
460 }
461 } else if (fpoly.degree() == 2) {
462 // Quadratic factor. Each l contributes two unknowns (B_l, C_l).
463 // Numerator at power l = B_l * x + C_l ; the coefficient column for
464 // B_l is q_over_fpow_l[l] shifted by 1 (multiplied by x), for C_l
465 // is q_over_fpow_l[l] (multiplied by 1).
466 for (int l = 1; l <= mult; ++l) {
467 const auto& qpl = q_over_fpow_l[l];
468 size_t base_col = col_off + static_cast<size_t>(2 * (l - 1));
469 size_t b_col = base_col; // for B_l
470 size_t c_col = base_col + 1; // for C_l
471 // C_l contributes qpl directly.
472 for (size_t k = 0; k < rows; ++k) {
473 Rational c = (k < qpl.coeffs.size()) ? qpl.coeffs[k] : Rational(0);
474 M[k][c_col] = c;
475 }
476 // B_l contributes qpl * x = qpl with index shifted by 1.
477 for (size_t k = 0; k < rows; ++k) {
478 if (k == 0) {
479 M[k][b_col] = Rational(0);
480 } else {
481 size_t src = k - 1;
482 Rational c = (src < qpl.coeffs.size()) ? qpl.coeffs[src] : Rational(0);
483 M[k][b_col] = c;
484 }
485 }
486 }
487 }
488 }
489
490 // Solve via Gaussian elimination on Rational matrix.
491 size_t m = rows;
492 size_t n = cols;
493 int sign = 1;
494 (void)sign;
495 size_t pivot_row = 0;
496 std::vector<size_t> pivot_col_for_row(m, static_cast<size_t>(-1));
497 for (size_t col = 0; col + 1 < n && pivot_row < m; ++col) {
498 size_t max_row = pivot_row;
499 bool found = false;
500 while (max_row < m) {
501 if (!(M[max_row][col] == Rational(0))) { found = true; break; }
502 ++max_row;
503 }
504 if (!found) continue;
505 if (max_row != pivot_row) std::swap(M[pivot_row], M[max_row]);
506 pivot_col_for_row[pivot_row] = col;
507 Rational pivot = M[pivot_row][col];
508 // Normalize pivot row.
509 for (size_t k = col; k < n; ++k) {
510 M[pivot_row][k] = M[pivot_row][k] / pivot;
511 }
512 // Eliminate in other rows.
513 for (size_t i = 0; i < m; ++i) {
514 if (i == pivot_row) continue;
515 Rational f = M[i][col];
516 if (f == Rational(0)) continue;
517 for (size_t k = col; k < n; ++k) {
518 M[i][k] = M[i][k] - f * M[pivot_row][k];
519 }
520 }
521 ++pivot_row;
522 }
523
524 // Check for inconsistency: any row with all-zero LHS but nonzero RHS.
525 for (size_t r = 0; r < m; ++r) {
526 bool all_zero_lhs = true;
527 for (size_t c = 0; c + 1 < n; ++c) {
528 if (!(M[r][c] == Rational(0))) { all_zero_lhs = false; break; }
529 }
530 if (all_zero_lhs && !(M[r][n - 1] == Rational(0))) return false;
531 }
532
533 // Extract solution: requires that each unknown has a pivot column.
534 std::vector<Rational> sol(num_unknowns, Rational(0));
535 std::vector<bool> pivoted(num_unknowns, false);
536 for (size_t r = 0; r < m; ++r) {
537 size_t pc = pivot_col_for_row[r];
538 if (pc == static_cast<size_t>(-1)) continue;
539 if (pc >= num_unknowns) continue;
540 sol[pc] = M[r][n - 1];
541 pivoted[pc] = true;
542 }
543 for (size_t i = 0; i < num_unknowns; ++i) {
544 if (!pivoted[i]) return false; // underdetermined -> can't form unique partial fractions
545 }
546
547 // Re-package solution into per-(factor, power) numerator polynomials.
548 numerators_out.reserve(num_unknowns);
549 for (size_t i = 0; i < factors.size(); ++i) {
550 const auto& [fpoly, mult] = factors[i];
551 size_t col_off = unknown_offset_per_term[i];
552 if (fpoly.degree() == 1) {
553 for (int l = 1; l <= mult; ++l) {
554 Rational A = sol[col_off + static_cast<size_t>(l - 1)];
555 numerators_out.push_back(Polynomial<Rational>({A}, Q.variable_name));
556 }
557 } else { // degree 2
558 for (int l = 1; l <= mult; ++l) {
559 size_t base_col = col_off + static_cast<size_t>(2 * (l - 1));
560 Rational B = sol[base_col];
561 Rational C = sol[base_col + 1];
562 numerators_out.push_back(
563 Polynomial<Rational>({C, B}, Q.variable_name));
564 }
565 }
566 }
567 return true;
568}

570std::shared_ptr<SymbolicExpr> RationalDecompositionStrategy::integrate_term(
571 const Polynomial<Rational>& numerator,
572 const Polynomial<Rational>& factor,
573 int power, const std::string& var) {
574
575 if (rd_is_zero_poly(numerator)) return SymbolicExpr::number(0);
576
577 if (factor.degree() == 1) {
578 // factor = x - r (monic linear). r = -coeffs[0].
579 Rational r = Rational(0) - factor.coeffs[0];
580 Rational A = (numerator.coeffs.size() > 0) ? numerator.coeffs[0] : Rational(0);
581 if (A == Rational(0)) return SymbolicExpr::number(0);
582
583 if (power == 1) {
584 // A * ln|x - r|. We use ln(x - r) here (matching existing strategies
585 // which omit the absolute value).
586 auto inner = rd_var_minus(var, r);
587 return SymbolicExpr::multiply(rd_num_rat(A), SymbolicExpr::ln(inner));
588 }
589 // power >= 2: -A / ((power-1) * (x - r)^(power-1))
590 Rational coeff = Rational(0) - A / Rational(BigInt(power - 1));
591 auto base = rd_var_minus(var, r);
592 auto exp = rd_num_int(power - 1);
593 auto pw = SymbolicExpr::power(base, exp);
594 auto inv_pw = SymbolicExpr::power(pw, rd_num_int(-1));
595 return SymbolicExpr::multiply(rd_num_rat(coeff), inv_pw);
596 }
597
598 if (factor.degree() == 2) {
599 // factor = x^2 + p*x + q (monic quadratic). discriminant = p^2 - 4q.
600 Rational q = (factor.coeffs.size() > 0) ? factor.coeffs[0] : Rational(0);
601 Rational p = (factor.coeffs.size() > 1) ? factor.coeffs[1] : Rational(0);
602
603 // Verify it's irreducible: discriminant must be < 0. (If it's >= 0
604 // we should have factored it; but be defensive.)
605 Rational disc = p * p - Rational(4) * q;
606
607 // numerator = B*x + C
608 Rational B = (numerator.coeffs.size() > 1) ? numerator.coeffs[1] : Rational(0);
609 Rational C = (numerator.coeffs.size() > 0) ? numerator.coeffs[0] : Rational(0);
610
611 if (power >= 2) {
612 // Higher powers of irreducible quadratics: leave as unevaluated
613 // integral; this triggers the safe fallback in try_integrate.
614 // Build the symbolic integrand for the unevaluated node.
615 auto num_sym = rd_poly_to_sym(numerator, var);
616 auto den_sym_base = rd_poly_to_sym(factor, var);
617 auto den_pw = SymbolicExpr::power(den_sym_base, rd_num_int(power));
618 auto inv_den = SymbolicExpr::power(den_pw, rd_num_int(-1));
619 auto integrand = SymbolicExpr::multiply(num_sym, inv_den);
620 std::vector<std::shared_ptr<const SymbolicNode>> args;
621 args.push_back(lamina::detail::node(integrand));
622 args.push_back(lamina::detail::node(SymbolicExpr::variable(var)));
623 return lamina::detail::make_expression_ptr(
624 lamina::detail::make_node<FunctionNode>(FunctionNode::FuncType::Calculus_Integral, args));
625 }
626
627 // power == 1: ∫ (B x + C) / (x^2 + p x + q) dx
628 // Split numerator: (B x + C) = (B/2) * (2x + p) + (C - B p / 2)
629 // ∫ (B/2)(2x+p)/(x^2+px+q) dx = (B/2) * ln(x^2+px+q)
630 // ∫ (C - B p / 2) / (x^2 + p x + q) dx
631 // = (C - B p / 2) * (2 / sqrt(4q - p^2)) * arctan( (2x + p) / sqrt(4q - p^2) )
632 std::shared_ptr<SymbolicExpr> result = SymbolicExpr::number(0);
633
634 Rational B_half = B / Rational(2);
635 if (!(B_half == Rational(0))) {
636 auto den_sym = rd_poly_to_sym(factor, var);
637 auto ln_part = SymbolicExpr::ln(den_sym);
638 auto term_log = SymbolicExpr::multiply(rd_num_rat(B_half), ln_part);
639 result = SymbolicExpr::add(result, term_log);
640 }
641
642 Rational atan_coeff = C - (B * p) / Rational(2);
643 if (!(atan_coeff == Rational(0))) {
644 // s2 = -disc = 4q - p^2 (positive for irreducible factor).
645 Rational s2 = Rational(0) - disc;
646 // Build sqrt(s2) symbolically (works even when s2 isn't a perfect square).
647 auto s2_sym = rd_num_rat(s2);
648 auto sqrt_s2 = SymbolicExpr::sqrt(s2_sym);
649 // 2x + p
650 auto two_x = SymbolicExpr::multiply(rd_num_int(2), SymbolicExpr::variable(var));
651 std::shared_ptr<SymbolicExpr> two_x_plus_p;
652 if (p == Rational(0)) {
653 two_x_plus_p = two_x;
654 } else {
655 two_x_plus_p = SymbolicExpr::add(two_x, rd_num_rat(p));
656 }
657 auto inv_sqrt = SymbolicExpr::power(sqrt_s2, rd_num_int(-1));
658 auto arctan_arg = SymbolicExpr::multiply(two_x_plus_p, inv_sqrt);
659 auto atan_part = make_arctan(arctan_arg);
660 auto two = rd_num_int(2);
661 auto coeff_part = SymbolicExpr::multiply(
662 SymbolicExpr::multiply(rd_num_rat(atan_coeff), two), inv_sqrt);
663 auto term_atan = SymbolicExpr::multiply(coeff_part, atan_part);
664 result = SymbolicExpr::add(result, term_atan);
665 }
666
667 return result;
668 }
669
670 // Should not reach here (factor degree > 2 is rejected by factor_denominator).
671 return SymbolicExpr::number(0);
672}

674std::shared_ptr<SymbolicExpr> RationalDecompositionStrategy::try_integrate(
675 const SymbolicExpr& expr, const std::string& var, Integrator& ctx,
676 ComputationContext&, int depth) {
677 (void)ctx;
678 (void)depth;
679
680 Polynomial<Rational> P, Q;
681 try {
682 if (!extract_rational(expr, var, P, Q)) {
683 return nullptr; // not rational -> let next strategy try
684 }
685 } catch (...) {
686 return nullptr;
687 }
688
689 // Defensive: if Q is zero or constant, this is not the right strategy.
690 if (rd_is_zero_poly(Q) || Q.degree() < 1) return nullptr;
691
692 /// PartialFractionStrategy 处理 deg<=1 的简单情形；degree>=2 交给本策略，
693 /// 因为它的 integrate_term 对不可约二次因子（→ arctan + ln）是完整的，
694 /// 而 PartialFraction 对不可约二次式会失败（留下未求值积分）。
695 if (Q.degree() < 2) return nullptr;
696 try {
697 // Long division if needed.
698 Polynomial<Rational> quot, rem;
699 if (P.degree() >= Q.degree()) {
700 poly_divide(P, Q, quot, rem);
701 } else {
702 quot = Polynomial<Rational>(Q.variable_name);
703 rem = P;
704 }
705
706 // Factor denominator.
707 std::vector<std::pair<Polynomial<Rational>, int>> factors;
708 if (!factor_denominator(Q, factors)) {
709 // Cannot factor over Q -> return unevaluated integral node.
710 return Integrator::depends_on(expr, var)
711 ? lamina::detail::make_expression_ptr(
712 lamina::detail::make_node<FunctionNode>(
713 FunctionNode::FuncType::Calculus_Integral,
714 std::vector<std::shared_ptr<const SymbolicNode>>{
715 lamina::detail::node(expr),
716 lamina::detail::node(SymbolicExpr::variable(var))}))
717 : nullptr;
718 }
719 if (factors.empty()) {
720 // Means Q is constant after factoring, so really there's nothing left;
721 // integrate quot only.
722 if (rd_is_zero_poly(quot)) return SymbolicExpr::number(0);
723 // ∫ quot(x) dx = poly_integral(quot)
724 // fall through; handled below
725 }
726
727 // Solve coefficients on the proper part rem / Q.
728 std::vector<Polynomial<Rational>> numerators;
729 if (!rd_is_zero_poly(rem)) {
730 if (!solve_coefficients(rem, Q, factors, numerators)) {
731 return lamina::detail::make_expression_ptr(
732 lamina::detail::make_node<FunctionNode>(
733 FunctionNode::FuncType::Calculus_Integral,
734 std::vector<std::shared_ptr<const SymbolicNode>>{
735 lamina::detail::node(expr),
736 lamina::detail::node(SymbolicExpr::variable(var))}));
737 }
738 }
739
740 // Build the result piece by piece.
741 std::shared_ptr<SymbolicExpr> result = SymbolicExpr::number(0);
742
743 // Polynomial part from long division.
744 if (!rd_is_zero_poly(quot)) {
745 // Antiderivative of x^k is x^(k+1) / (k+1).
746 for (size_t k = 0; k < quot.coeffs.size(); ++k) {
747 if (quot.coeffs[k] == Rational(0)) continue;
748 Rational coeff = quot.coeffs[k] / Rational(BigInt(static_cast<long long>(k + 1)));
749 std::shared_ptr<SymbolicExpr> term;
750 auto v = SymbolicExpr::variable(var);
751 auto pw = SymbolicExpr::power(v, rd_num_int(static_cast<long long>(k + 1)));
752 if (coeff == Rational(1)) {
753 term = pw;
754 } else {
755 term = SymbolicExpr::multiply(rd_num_rat(coeff), pw);
756 }
757 result = SymbolicExpr::add(result, term);
758 }
759 }
760
761 // Partial-fraction terms.
762 if (!numerators.empty()) {
763 size_t idx = 0;
764 for (size_t i = 0; i < factors.size(); ++i) {
765 const auto& [fpoly, mult] = factors[i];
766 for (int l = 1; l <= mult; ++l, ++idx) {
767 if (idx >= numerators.size()) break;
768 auto term = integrate_term(numerators[idx], fpoly, l, var);
769 if (!term) {
770 // Fallback: unevaluated integral over the original.
771 return lamina::detail::make_expression_ptr(
772 lamina::detail::make_node<FunctionNode>(
773 FunctionNode::FuncType::Calculus_Integral,
774 std::vector<std::shared_ptr<const SymbolicNode>>{
775 lamina::detail::node(expr),
776 lamina::detail::node(SymbolicExpr::variable(var))}));
777 }
778 result = SymbolicExpr::add(result, term);
779 }
780 }
781 }
782
783 auto simplified = result->simplify();
784 if (simplified) {
785 // If the simplified form contains an unevaluated integral marker
786 // (e.g. high-power irreducible quadratic), prefer keeping the
787 // un-simplified result so downstream consumers can still read the
788 // partial-fraction form.
789 return simplified;
790 }
791 return result;
792 } catch (...) {
793 // Any unexpected runtime failure -> return unevaluated integral.
794 return lamina::detail::make_expression_ptr(
795 lamina::detail::make_node<FunctionNode>(
796 FunctionNode::FuncType::Calculus_Integral,
797 std::vector<std::shared_ptr<const SymbolicNode>>{
798 lamina::detail::node(expr),
799 lamina::detail::node(SymbolicExpr::variable(var))}));
800 }
801}
