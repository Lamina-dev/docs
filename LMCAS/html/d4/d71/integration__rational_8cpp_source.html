6void rd_reduce_reciprocal(RationalDecompositionStrategy& strategy,
7 Polynomial<Rational>& P, Polynomial<Rational>& Q) {
8 if (!P.is_zero() && P.degree() > 0 && P.degree() < Q.degree()) {
9 Polynomial<Rational> reduced_denominator;
10 Polynomial<Rational> remainder;
11 strategy.poly_divide(Q, P, reduced_denominator, remainder);
12 if (remainder.is_zero()) {
13 P = Polynomial<Rational>(
14 std::vector<Rational>{Rational(1)}, P.variable_name);
15 Q = std::move(reduced_denominator);
16 }

20std::shared_ptr<SymbolicExpr> rd_integrate_polynomial(
21 const Polynomial<Rational>& quot, const std::string& var) {
22 auto result = SymbolicExpr::number(0);
23 if (!quot.is_zero()) {
24 for (size_t k = 0; k < quot.coeffs.size(); ++k) {
25 if (quot.coeffs[k] == Rational(0)) { continue; }
26 const BigInt integrated_power = BigInt(k) + BigInt(1);
27 Rational coeff = quot.coeffs[k] / Rational(integrated_power);
28 std::shared_ptr<SymbolicExpr> term;
29 auto v = SymbolicExpr::variable(var);
30 auto pw = SymbolicExpr::power(v, SymbolicExpr::number(integrated_power));
31 if (coeff == Rational(1)) {
32 term = pw;
33 } else {
34 term = SymbolicExpr::multiply(SymbolicExpr::number(coeff), pw);
35 }
36 result = SymbolicExpr::add(result, term);
37 }
38 }
39 return result;
40}

42std::shared_ptr<SymbolicExpr> rd_add_fraction_primitives(
43 RationalDecompositionStrategy& strategy, const SymbolicExpr& expr, const std::string& var,
44 const std::vector<std::pair<Polynomial<Rational>, int>>& factors,
45 const std::vector<Polynomial<Rational>>& numerators,
46 std::shared_ptr<SymbolicExpr> result) {
47 if (!numerators.empty()) {
48 size_t idx = 0;
49 for (size_t i = 0; i < factors.size(); ++i) {
50 const auto& [fpoly, mult] = factors[i];
51 for (int l = 1; l <= mult; ++l, ++idx) {
52 if (idx >= numerators.size()) { break; }
53 auto term = strategy.integrate_term(numerators[idx], fpoly, l, var);
54 if (!term) {
55 return LMCAS::detail::make_expression_ptr(
56 LMCAS::detail::make_node<IntegralNode>(
57 LMCAS::detail::node(expr), var));
58 }
59 result = SymbolicExpr::add(result, term);
60 }
61 }
62 }
63 auto simplified = result->simplify();
64 if (simplified) {
65 return simplified;
66 }
67 return result;

70Result<std::shared_ptr<SymbolicExpr>> rd_integrate_proper_fraction(
71 RationalDecompositionStrategy& strategy, const SymbolicExpr& expr, const std::string& var,
72 const Polynomial<Rational>& P, const Polynomial<Rational>& Q,
73 ComputationContext& computation) {
74 Polynomial<Rational> quot, rem;
75 if (P.degree() >= Q.degree()) {
76 strategy.poly_divide(P, Q, quot, rem);
77 } else {
78 quot = Polynomial<Rational>(Q.variable_name);
79 rem = P;
80 }
81
82 std::vector<std::pair<Polynomial<Rational>, int>> factors;
83 if (!strategy.factor_denominator(Q, factors)) {
84 /// Q 上因式分解未决时保留未求值积分节点.
85 return Integrator::depends_on(expr, var)
86 ? LMCAS::detail::make_expression_ptr(
87 LMCAS::detail::make_node<IntegralNode>(
88 LMCAS::detail::node(expr), var))
89 : nullptr;
90 }
91 if (factors.empty() && quot.is_zero()) {
92 return SymbolicExpr::number(0);
93 }
94
95 std::vector<Polynomial<Rational>> numerators;
96 if (!rem.is_zero()) {
97 auto coefficients = strategy.solve_coefficients(
98 rem, Q, factors, numerators, computation);
99 if (!coefficients) {
100 return Result<std::shared_ptr<SymbolicExpr>>::failure(
101 coefficients.error());
102 }
103 if (!coefficients.value()) {
104 return LMCAS::detail::make_expression_ptr(
105 LMCAS::detail::make_node<IntegralNode>(
106 LMCAS::detail::node(expr), var));
107 }
108 }
109 auto result = rd_integrate_polynomial(quot, var);
110 return rd_add_fraction_primitives(strategy, expr, var, factors, numerators, result);
111}

114Result<std::shared_ptr<SymbolicExpr>> RationalDecompositionStrategy::try_integrate_raw(
115 const SymbolicExpr& expr, const std::string& var, Integrator& ctx,
116 ComputationContext& computation, int depth) {
117 (void)ctx;
118 (void)depth;
119
120 Polynomial<Rational> P, Q;
121 auto extracted = extract_rational(expr, var, P, Q);
122 if (!extracted) {
123 return Result<std::shared_ptr<SymbolicExpr>>::failure(extracted.error());
124 }
125 if (!extracted.value()) { return nullptr; }
126 rd_reduce_reciprocal(*this, P, Q);
127 if (Q.is_zero() || Q.degree() < 1) { return nullptr; }
128
129 /**
130 * @brief 常数分子与一次分母直接积分，其余情形分解后逐项积分。
131 * @note 不可约二次因子的原函数可含 arctan 与 ln。
132 */
133 if (Q.degree() == 1 && P.degree() <= 0 &&
134 !P.coeffs.empty() && Q.coeffs.size() > 1 &&
135 Q.coeffs[1] != Rational(0)) {
136 auto coefficient = SymbolicExpr::number(
137 P.coeffs[0] / Q.coeffs[1]);
138 auto logarithm = SymbolicExpr::ln(poly_to_symbolic(Q));
139 return SymbolicExpr::multiply(
140 coefficient, logarithm)->simplify();
141 }
142 if (Q.degree() < 2) { return nullptr; }
143 return rd_integrate_proper_fraction(*this, expr, var, P, Q, computation);
144}
