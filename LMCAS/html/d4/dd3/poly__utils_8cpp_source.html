27template <>
28Result<BigInt> extract_coeff_value<BigInt>(
29 const std::shared_ptr<SymbolicExpr>& coefficient) {
30 if (!coefficient || !detail::node(coefficient)) {
31 return Result<BigInt>::failure(CasErrc::InvalidArgument,
32 "Coefficient must have an expression", "polynomial.coefficient");
33 }
34 auto simplified = coefficient->simplify();
35 if (auto number = std::dynamic_pointer_cast<const NumberNode>(
36 LMCAS::detail::node(simplified))) {
37 if (std::holds_alternative<BigInt>(number->value())) {
38 return std::get<BigInt>(number->value());
39 }
40 if (std::holds_alternative<Rational>(number->value())) {
41 const Rational& value = std::get<Rational>(number->value());
42 if (value.is_integer()) return value.to_bigint();
43 }
44 if (std::holds_alternative<lmmc_real_t>(number->value())) {
45 const lmmc_real_t value = std::get<lmmc_real_t>(number->value());
46 if (std::isfinite(value) && value == std::floor(value)) {
47 return Rational::from_double(value).to_bigint();
48 }
49 }
50 }
51 return Result<BigInt>::failure(CasErrc::UnsupportedExpression,
52 "Coefficient is not an integer", "polynomial.coefficient");
53}

86template <typename T>
87bool polynomial_product_fits(const Polynomial<T>& left, const Polynomial<T>& right) {
88 if (left.is_zero() || right.is_zero()) return true;
89 const auto max_terms = static_cast<std::size_t>(std::numeric_limits<int>::max());
90 return left.coeffs.size() <= max_terms &&
91 right.coeffs.size() - 1 <= max_terms - left.coeffs.size();
92}

96void checked_coefficient_count(std::size_t count, detail::RewriteBudget& budget) {
97 if (count > static_cast<std::size_t>(std::numeric_limits<int>::max())) {
98 throw CasError{CasErrc::ResourceLimit,
99 "Polynomial coefficient storage limit exceeded", "polynomial.convert"};
100 }
101 auto terms = budget.context().require_expansion_terms(count, "polynomial.convert");
102 if (!terms) { throw terms.error(); }
103 budget.require_nodes(count);
104}

106std::size_t checked_product_count(std::size_t left, std::size_t right,
107 detail::RewriteBudget& budget) {
108 if (left == 0 || right == 0) return 0;
109 const auto maximum = static_cast<std::size_t>(std::numeric_limits<int>::max());
110 if (left > maximum || right - 1 > maximum - left) {
111 throw CasError{CasErrc::ResourceLimit,
112 "Polynomial coefficient storage limit exceeded", "polynomial.convert"};
113 }
114 const auto count = left + right - 1;
115 checked_coefficient_count(count, budget);
116 return count;
117}

147Polynomial<SymbolicPolyCoeff> add_checked_coefficients(
148 const Polynomial<SymbolicPolyCoeff>& left, const Polynomial<SymbolicPolyCoeff>& right,
149 detail::RewriteBudget& budget) {
150 const auto count = std::max(left.coeffs.size(), right.coeffs.size());
151 checked_coefficient_count(count, budget);
152 Polynomial<SymbolicPolyCoeff> result(left.variable_name);
153 result.coeffs.reserve(count);
154 std::size_t nodes = 0;
155 for (std::size_t index = 0; index < count; ++index) {
156 detail::SymbolicNodePtr coefficient;
157 if (index < left.coeffs.size() && index < right.coeffs.size()) {
158 coefficient = coefficient_arithmetic<AddNode>(
159 detail::node(left.coeffs[index].val), detail::node(right.coeffs[index].val), budget);
160 } else {
161 const auto& source = index < left.coeffs.size() ? left : right;
162 coefficient = detail::node(source.coeffs[index].val);
163 }
164 nodes = budget.append_size(nodes, budget.measure(coefficient));
165 result.coeffs.emplace_back(detail::make_expression_ptr(std::move(coefficient)));
166 }
167 trim_checked_coefficients(result, budget);
168 return result;
169}

171Polynomial<SymbolicPolyCoeff> multiply_checked_coefficients(
172 const Polynomial<SymbolicPolyCoeff>& left, const Polynomial<SymbolicPolyCoeff>& right,
173 detail::RewriteBudget& budget) {
174 const auto count = checked_product_count(left.coeffs.size(), right.coeffs.size(), budget);
175 Polynomial<SymbolicPolyCoeff> result(left.variable_name);
176 result.coeffs.reserve(count);
177 std::size_t nodes = 0;
178 for (std::size_t degree = 0; degree < count; ++degree) {
179 detail::SymbolicNodePtr coefficient;
180 std::size_t coefficient_nodes = 0;
181 const auto first = degree < right.coeffs.size() ? 0 : degree - right.coeffs.size() + 1;
182 const auto last = std::min(degree, left.coeffs.size() - 1);
183 for (std::size_t index = first; index <= last; ++index) {
184 auto product = coefficient_arithmetic<MultiplyNode>(
185 detail::node(left.coeffs[index].val),
186 detail::node(right.coeffs[degree - index].val), budget);
187 if (coefficient) {
188 coefficient = coefficient_arithmetic<AddNode>(coefficient, product, budget);
189 } else {
190 coefficient.swap(product);
191 }
192 coefficient_nodes = budget.measure(coefficient);
193 budget.append_size(nodes, coefficient_nodes);
194 }
195 nodes = budget.append_size(nodes, coefficient_nodes);
196 result.coeffs.emplace_back(detail::make_expression_ptr(std::move(coefficient)));
197 }
198 trim_checked_coefficients(result, budget);
199 return result;
200}

202Result<void> check_zero_power_base(
203 const detail::SymbolicNodePtr& base, ComputationContext& context) {
204 const auto& facts = detail::no_facts();
205 auto real_defined = detail::query_definedness(base, facts, Domain::Real, context);
206 if (!real_defined) { return Result<void>::failure(real_defined.error()); }
207 auto complex_defined = detail::query_definedness(base, facts, Domain::Complex, context);
208 if (!complex_defined) { return Result<void>::failure(complex_defined.error()); }
209 auto nonzero = detail::query_nonzero_value(base, facts, Domain::Real, context);
210 if (!nonzero) { return Result<void>::failure(nonzero.error()); }
211 if (real_defined.value() != Tribool::True ||
212 complex_defined.value() != Tribool::True || nonzero.value() != Tribool::True) {
213 return Result<void>::failure(CasErrc::UnsupportedExpression,
214 "Zero power requires a defined nonzero base", "polynomial.convert");
215 }
216 return Result<void>::success();
217}

219Result<void> check_power_coefficient_count(
220 std::size_t base_count, int exponent, detail::RewriteBudget* budget) {
221 if (!budget || base_count == 0) { return Result<void>::success(); }
222 const auto degree = base_count - 1;
223 const auto maximum = static_cast<std::size_t>(std::numeric_limits<int>::max());
224 const auto count = static_cast<std::size_t>(exponent);
225 if (degree > (maximum - 1) / count) {
226 return Result<void>::failure(CasErrc::ResourceLimit,
227 "Polynomial coefficient storage limit exceeded", "polynomial.convert");
228 }
229 checked_coefficient_count(degree * count + 1, *budget);
230 return Result<void>::success();
231}

260template <typename T>
261Result<Polynomial<T>> symbolic_power_to_poly(
262 const PowerNode& power, const std::string& variable,
263 ComputationContext* context, detail::RewriteBudget* budget) {
264 detail::RewriteScope scope(budget);
265 BigInt exponent;
266 if (!try_get_integer_value(
267 std::dynamic_pointer_cast<const NumberNode>(power.exponent()), exponent) ||
268 exponent.is_negative()) {
269 return Result<Polynomial<T>>::failure(CasErrc::UnsupportedExpression,
270 "Polynomial powers require nonnegative integer exponents", "polynomial.convert");
271 }
272 const auto bounded_exponent = exponent.try_to_int64();
273 if (!bounded_exponent || *bounded_exponent >= 1000) {
274 return Result<Polynomial<T>>::failure(CasErrc::ResourceLimit,
275 "Polynomial power expansion limit exceeded", "polynomial.convert");
276 }
277 const int exponent_value = static_cast<int>(*bounded_exponent);
278
279 Polynomial<T> result(variable);
280 if (budget) {
281 checked_coefficient_count(1, *budget);
282 normalization_check_children(budget, 1);
283 }
284 result.coeffs.emplace_back(1);
285 if (exponent_value == 0) {
286 auto defined = check_zero_power_base(power.base(), *context);
287 if (!defined) { return Result<Polynomial<T>>::failure(defined.error()); }
288 return result;
289 }
290 const auto base = symbolic_to_poly_recursive<T>(power.base(), variable, context, budget);
291 if (!base) { return Result<Polynomial<T>>::failure(base.error()); }
292 auto count = check_power_coefficient_count(base.value().coeffs.size(), exponent_value, budget);
293 if (!count) { return Result<Polynomial<T>>::failure(count.error()); }
294 for (int index = 0; index < exponent_value; ++index) {
295 if constexpr (std::is_same_v<T, SymbolicPolyCoeff>) {
296 if (budget) {
297 result = multiply_checked_coefficients(result, base.value(), *budget);
298 continue;
299 }
300 }
301 if (!polynomial_product_fits(result, base.value())) {
302 return Result<Polynomial<T>>::failure(CasErrc::ResourceLimit,
303 "Polynomial coefficient storage limit exceeded", "polynomial.convert");
304 }
305 result = result * base.value();
306 }
307 return result;
308}

359template <typename T>
360Result<Polynomial<T>> symbolic_to_poly_recursive(
361 const std::shared_ptr<const SymbolicNode>& node,
362 const std::string& variable, ComputationContext* context, detail::RewriteBudget* budget) {
363 detail::RewriteScope scope(budget);
364 if (!node) {
365 return Result<Polynomial<T>>::failure(CasErrc::InvalidArgument,
366 "Expression node must not be null", "polynomial.convert");
367 }
368 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
369 BigInt exponent;
370 if (try_get_integer_value(
371 std::dynamic_pointer_cast<const NumberNode>(power->exponent()), exponent) &&
372 exponent.is_zero()) {
373 return symbolic_power_to_poly<T>(*power, variable, context, budget);
374 }
375 }
376 if (!expression_depends_on_variable(node, variable, budget)) {
377 return constant_to_poly<T>(node, variable, budget);
378 }
379
380 if (auto symbol = std::dynamic_pointer_cast<const VariableNode>(node)) {
381 if (!symbol->is_constant() && symbol->name() == variable) {
382 if constexpr (std::is_same_v<T, SymbolicPolyCoeff>) {
383 if (budget) {
384 checked_coefficient_count(2, *budget);
385 normalization_check_children(budget, 2);
386 Polynomial<T> result(variable);
387 result.coeffs.reserve(2);
388 result.coeffs.emplace_back(0);
389 result.coeffs.emplace_back(1);
390 return result;
391 }
392 }
393 return Polynomial<T>({T(0), T(1)}, variable);
394 }
395 }
396
397 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
398 return symbolic_sum_to_poly<T>(*add, variable, context, budget);
399 }
400
401 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
402 return symbolic_product_to_poly<T>(*multiply, variable, context, budget);
403 }
404
405 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
406 return symbolic_power_to_poly<T>(*power, variable, context, budget);
407 }
408
409 return Result<Polynomial<T>>::failure(CasErrc::UnsupportedExpression,
410 "Expression is not a polynomial in the requested variable", "polynomial.convert");
411}

461template <typename T>
462LMCAS_API std::shared_ptr<SymbolicExpr> poly_to_symbolic(
463 const Polynomial<T>& polynomial) {
464 if (polynomial.is_zero()) return SymbolicExpr::number(0);
465
466 std::vector<std::shared_ptr<SymbolicExpr>> terms;
467 for (std::size_t degree = 0; degree < polynomial.coeffs.size(); ++degree) {
468 if (polynomial.coeffs[degree] == T(0)) continue;
469
470 std::shared_ptr<SymbolicExpr> coefficient;
471 if constexpr (std::is_same_v<T, SymbolicPolyCoeff>) {
472 coefficient = polynomial.coeffs[degree].val
473 ? polynomial.coeffs[degree].val
474 : SymbolicExpr::number(0);
475 } else {
476 coefficient = SymbolicExpr::number(polynomial.coeffs[degree]);
477 }
478
479 if (degree == 0) {
480 terms.push_back(coefficient);
481 continue;
482 }
483
484 auto variable = SymbolicExpr::variable(polynomial.variable_name);
485 auto variable_part = degree == 1
486 ? variable
487 : SymbolicExpr::power(
488 variable, SymbolicExpr::number(BigInt(static_cast<std::uint64_t>(degree))));
489 if (polynomial.coeffs[degree] == T(1)) {
490 terms.push_back(variable_part);
491 } else if (polynomial.coeffs[degree] == T(-1)) {
492 terms.push_back(SymbolicExpr::multiply(
493 SymbolicExpr::number(-1), variable_part));
494 } else {
495 terms.push_back(SymbolicExpr::multiply(coefficient, variable_part));
496 }
497 }
498
499 std::reverse(terms.begin(), terms.end());
500 if (terms.empty()) return SymbolicExpr::number(0);
501 auto result = terms.front();
502 for (std::size_t index = 1; index < terms.size(); ++index) {
503 result = SymbolicExpr::add(result, terms[index]);
504 }
505 return result;
506}
