123std::optional<Result<ApproxReal>> special(FunctionNode::FuncType type, double x) {
124 switch (type) {
125 case FunctionNode::FuncType::Erf: {

128 case FunctionNode::FuncType::LambertW: {
129 lmmc_real_t value = 0.0;
130 const auto status = lmmc_lambertw(static_cast<lmmc_real_t>(x), &value);
131 if (status == LMMC_STATUS_INVALID_ARGUMENT) {
132 return numeric_failure(CasErrc::DomainError, "LambertW argument is outside the real branch");
133 }
134 if (status == LMMC_STATUS_CONVERGENCE_FAILED) {
135 return numeric_failure(CasErrc::NumericFailure, "LambertW failed to certify convergence");
136 }
137 if (status == LMMC_STATUS_NUMERICAL_FAILURE) {
138 return numeric_failure(CasErrc::NumericFailure, "LambertW encountered a nonfinite numerical value");
139 }
140 if (status != LMMC_STATUS_OK) {
141 return numeric_failure(CasErrc::InternalInvariant, "Unexpected LambertW status");
142 }
143 return numeric_approximation(static_cast<double>(value));
144 }
145 case FunctionNode::FuncType::RealPart:
146 case FunctionNode::FuncType::Conjugate: {
147 return numeric_approximation(x);

149 case FunctionNode::FuncType::ComplexArg: {

153 return numeric_approximation(0.0);

166Result<ApproxReal> evaluate_unary_numeric(FunctionNode::FuncType type, double x) {
167 auto evaluated = trigonometric(type, x);
168 if (!evaluated) {
169 evaluated = inverse_trigonometric(type, x);
170 }
171 if (!evaluated) {
172 evaluated = hyperbolic(type, x);
173 }
174 if (!evaluated) {
175 evaluated = elementary(type, x);
176 }
177 if (!evaluated) {
178 evaluated = special(type, x);
179 }
180 if (!evaluated) {
181 return numeric_failure(CasErrc::UnsupportedExpression,
182 "function is not supported by real numeric evaluation");
183 }
184 return std::move(*evaluated);
185}
