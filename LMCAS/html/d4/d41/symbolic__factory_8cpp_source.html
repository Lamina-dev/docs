13 std::vector<std::shared_ptr<const SymbolicNode>> ops,
14 detail::RewriteBudget* budget) {
15 if (ops.empty()) {
16 normalization_check_children(budget, 1);
17 return create_number(::LMCAS::BigInt(0));
18 }
19 std::size_t nodes = 0;
20 for (const auto& op : ops) {
21 if (!op) {
22 throw std::invalid_argument("create_add operand cannot be null");
23 }
24 if (budget) {
25 nodes = budget->append_size(nodes, budget->measure(op));
26 }
27 }
28 if (ops.size() == 1) {
29 return ops[0];
30 }
31
32 std::vector<std::shared_ptr<const SymbolicNode>> flat_ops;
33 nodes = 0;
34 flat_ops.reserve(ops.size());
35 for (const auto& op : ops) {
36 if (op->is_zero()) {
37 continue;
38 }
39 if (auto add = std::dynamic_pointer_cast<const AddNode>(op)) {
40 if (budget) {
41 for (const auto& child : add->operands()) {
42 nodes = budget->append_size(nodes, budget->measure(child));
43 }
44 }
45 flat_ops.insert(flat_ops.end(), add->operands().begin(), add->operands().end());
46 } else {
47 normalization_append(budget, nodes, flat_ops, op);
48 }
49 }
50
51 if (flat_ops.empty()) {
52 normalization_check_children(budget, 1);
53 return create_number(::LMCAS::BigInt(0));
54 }
55 if (flat_ops.size() == 1) {
56 return flat_ops[0];
57 }
58 normalization_check_arithmetic<AddNode>(budget, flat_ops, nodes);

64detail::SymbolicNodePtr discardable_zero_factor(
65 const std::vector<detail::SymbolicNodePtr>& operands, detail::RewriteBudget* budget) {
66 const auto zero = std::find_if(operands.begin(), operands.end(), [](const auto& operand) {
67 return operand->is_zero();
68 });
69 if (zero == operands.end()) { return nullptr; }
70 std::optional<ComputationContext> local_context;
71 if (!budget) { local_context.emplace(); }
72 auto& context = budget ? budget->context() : *local_context;
73 if (std::all_of(operands.begin(), operands.end(), [&](const auto& operand) {
74 return normalization_can_discard(operand, detail::no_facts(), Domain::Real, context);
75 })) {
76 return *zero;
77 }
78 return nullptr;

139std::shared_ptr<const SymbolicNode> SymbolicFactory::create_power(std::shared_ptr<const SymbolicNode> base, std::shared_ptr<const SymbolicNode> exponent) {
140 if (!base || !exponent) {
141 throw std::invalid_argument("create_power operands cannot be null");
142 }
143 ComputationContext context;
144 if (exponent->is_zero()) {
145 if (normalization_can_discard(base, detail::no_facts(), Domain::Real, context) &&
146 normalization_proved(detail::query_nonzero_value(base, detail::no_facts(), Domain::Complex, context))) {
147 return create_number(::LMCAS::BigInt(1));
148 }
149 return LMCAS::detail::make_node<PowerNode>(std::move(base), std::move(exponent));
150 }
151 if (exponent->is_one()) {
152 return base;
153 }
154 if (base->is_zero()) {
155 if (normalization_can_discard(exponent, detail::no_facts(), Domain::Real, context) &&
156 normalization_proved(detail::query_positive_value(exponent, detail::no_facts(), context))) {
157 return create_number(::LMCAS::BigInt(0));
158 }
159 return LMCAS::detail::make_node<PowerNode>(std::move(base), std::move(exponent));
160 }
161 if (base->is_one() && normalization_can_discard(exponent, detail::no_facts(), Domain::Real, context)) {
162 return create_number(::LMCAS::BigInt(1));
163 }
164 return LMCAS::detail::make_node<PowerNode>(std::move(base), std::move(exponent));
165}
