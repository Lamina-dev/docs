34static TransformEngineResult te_convolve_sum(
35 const AddNode& addition, const std::shared_ptr<SymbolicExpr>& other,
36 bool sum_is_left, const std::string& variable, ComputationContext& context) {
37 std::shared_ptr<SymbolicExpr> accumulated;
38 for (const auto& operand : addition.operands()) {
39 auto term = detail::make_expression_ptr(operand);
40 auto transformed = sum_is_left
41 ? te_convolve_core(term, other, variable, context)
42 : te_convolve_core(other, term, variable, context);
43 if (!transformed) return transformed;
44 auto expression = transformed.value().value.expression;
45 accumulated = accumulated ? SymbolicExpr::add(accumulated, expression) : expression;
46 }
47 return te_wrap_transform_result(
48 accumulated ? accumulated->simplify() : SymbolicExpr::number(0), "convolve");
49}

51static TransformEngineResult te_convolve_core(
52 const std::shared_ptr<SymbolicExpr>& f,
53 const std::shared_ptr<SymbolicExpr>& g,
54 const std::string& variable,
55 ComputationContext& context)
56{
57 constexpr const char* operation = "convolve";
58 auto step = context.consume_steps(1, operation);
59 if (!step) return TransformEngineResult::failure(step.error());
60
61 if (te_is_zero(f) || te_is_zero(g)) {
62 return te_wrap_transform_result(SymbolicExpr::number(0), operation);
63 }
64
65 auto left_sum = std::dynamic_pointer_cast<const AddNode>(detail::node(f));
66 if (left_sum) {
67 return te_convolve_sum(*left_sum, g, true, variable, context);
68 }
69 auto right_sum = std::dynamic_pointer_cast<const AddNode>(detail::node(g));
70 if (right_sum) {
71 return te_convolve_sum(*right_sum, f, false, variable, context);
72 }
73
74 auto left_parts = te_split_coeff(f, variable);
75 if (te_depends_on(left_parts.second, variable) &&
76 left_parts.second->to_string() != f->to_string()) {
77 auto inner =
78 te_convolve_core(left_parts.second, g, variable, context);
79 if (!inner) return inner;
80 auto scaled = SymbolicExpr::multiply(
81 left_parts.first, inner.value().value.expression)->simplify();
82 return te_wrap_transform_result(scaled, operation);
83 }
84 auto right_parts = te_split_coeff(g, variable);
85 if (te_depends_on(right_parts.second, variable) &&
86 right_parts.second->to_string() != g->to_string()) {
87 auto inner =
88 te_convolve_core(f, right_parts.second, variable, context);
89 if (!inner) return inner;
90 auto scaled = SymbolicExpr::multiply(
91 right_parts.first, inner.value().value.expression)->simplify();
92 return te_wrap_transform_result(scaled, operation);
93 }
94
95 const auto left = te_match_scaled_gaussian(f, variable);
96 const auto right = te_match_scaled_gaussian(g, variable);
97 if (!left || !right) {
98 return TransformEngineResult::failure(
99 CasErrc::Inconclusive,
100 "当前双边卷积规则不支持该表达式对", operation);
101 }
102
103 const double rate_sum = left->rate + right->rate;
104 const double output_rate = left->rate * right->rate / rate_sum;
105 auto scale = SymbolicExpr::multiply(
106 SymbolicExpr::multiply(left->coefficient, right->coefficient),
107 SymbolicExpr::sqrt(SymbolicExpr::divide(
108 LMCAS::detail::make_expression_ptr(
109 LMCAS::detail::make_node<VariableNode>("pi", true)),
110 SymbolicExpr::number(rate_sum))));
111 auto square = SymbolicExpr::power(
112 SymbolicExpr::variable(variable), SymbolicExpr::number(2));
113 auto exponent = SymbolicExpr::multiply(
114 SymbolicExpr::number(-output_rate), square);
115 auto gaussian = SymbolicExpr::exp(exponent);
116 return te_wrap_transform_result(
117 SymbolicExpr::multiply(scale, gaussian)->simplify(), operation);
118}
