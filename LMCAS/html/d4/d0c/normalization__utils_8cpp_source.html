61BigInt get_node_degree_helper(const std::shared_ptr<const SymbolicNode>& node) {
62 if (!node) {
63 return BigInt(0);
64 }
65 if (std::dynamic_pointer_cast<const VariableNode>(node)) {
66 return BigInt(1);
67 }
68 if (auto p = std::dynamic_pointer_cast<const PowerNode>(node)) {
69 const auto e = std::dynamic_pointer_cast<const NumberNode>(p->exponent());
70 BigInt degree;
71 if (try_get_integer_value(e, degree)) {
72 return degree;
73 }
74 return BigInt(1);
75 }
76 if (auto m = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
77 BigInt d(0);
78 for (auto& op : m->operands()) d += get_node_degree_helper(op);
79 return d;
80 }
81 return BigInt(0);
82}

182bool get_pi_coeff(const std::shared_ptr<const SymbolicNode>& node, Rational& k) {
183 const auto variable = std::dynamic_pointer_cast<const VariableNode>(node);
184 if (variable && is_pi_variable(*variable)) {
185 k = Rational(1);
186 return true;
187 }
188 const auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node);
189 if (!multiply) {
190 return false;
191 }
192 bool has_pi = false;
193 k = Rational(1);
194 for (const auto& operand : multiply->operands()) {
195 if (const auto variable = std::dynamic_pointer_cast<const VariableNode>(operand)) {
196 if (has_pi || !is_pi_variable(*variable)) return false;
197 has_pi = true;
198 continue;
199 }
200 const auto number = std::dynamic_pointer_cast<const NumberNode>(operand);
201 if (!number) {
202 return false;
203 }
204 if (const auto* rational = std::get_if<Rational>(&number->value())) {
205 k = k * *rational;
206 continue;
207 }
208 const auto* integer = std::get_if<BigInt>(&number->value());
209 if (!integer) {
210 return false;
211 }
212 k = k * Rational(*integer);
213 }
214 return has_pi;
215}
