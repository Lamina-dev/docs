48bool try_get_integer_value(
49 const std::shared_ptr<const NumberNode>& node, BigInt& value) {
50 if (!node) {
51 return false;
52 }
53 if (const auto* integer = std::get_if<BigInt>(&node->value())) {
54 value = *integer;
55 return true;
56 }
57 if (const auto* rational = std::get_if<Rational>(&node->value())) {
58 if (rational->get_denominator() != BigInt(1)) {
59 return false;
60 }
61 value = rational->get_numerator();
62 return true;
63 }
64 const auto real = std::get<lmmc_real_t>(node->value());
65 if (!std::isfinite(real) || std::trunc(real) != real) {
66 return false;
67 }
68 value = Rational::from_double(real).get_numerator();
69 return true;
70}

73 * @brief 计算 AST 节点的多项式次数
74 * @param node 输入节点
75 * @return 节点对应的多项式次数

77BigInt get_node_degree_helper(const std::shared_ptr<const SymbolicNode>& node) {
78 if (!node) {
79 return BigInt(0);
80 }
81 if (std::dynamic_pointer_cast<const VariableNode>(node)) {
82 return BigInt(1);
83 }
84 if (auto p = std::dynamic_pointer_cast<const PowerNode>(node)) {
85 const auto e = std::dynamic_pointer_cast<const NumberNode>(p->exponent());
86 BigInt degree;
87 if (try_get_integer_value(e, degree)) {
88 return degree;
89 }
90 return BigInt(1);
91 }
92 if (auto m = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
93 BigInt d(0);
94 for (auto& op : m->operands()) d += get_node_degree_helper(op);
95 return d;
96 }
97 return BigInt(0);
98}

124bool NodeCompare::operator()(const std::shared_ptr<const SymbolicNode>& lhs, const std::shared_ptr<const SymbolicNode>& rhs) const {

148 return lhs->type_priority() < rhs->type_priority();

203 const Rational r1 = exact_number_as_rational(*a);
204 const Rational r2 = exact_number_as_rational(*b);
205 return LMCAS::detail::make_node<NumberNode>(r1 + r2);

209 BigInt i2;
210 try_get_integer_value(a, i1);
211 try_get_integer_value(b, i2);
212 return LMCAS::detail::make_node<NumberNode>(i1 + i2);
213}

246bool get_pi_coeff(const std::shared_ptr<const SymbolicNode>& node, Rational& k) {
247 const auto variable = std::dynamic_pointer_cast<const VariableNode>(node);
248 if (variable && is_pi_variable(*variable)) {
249 k = Rational(1);
250 return true;
251 }
252 const auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node);
253 if (!multiply) {
254 return false;
255 }
256 bool has_pi = false;
257 k = Rational(1);
258 for (const auto& operand : multiply->operands()) {
259 if (const auto variable = std::dynamic_pointer_cast<const VariableNode>(operand)) {
260 if (has_pi || !is_pi_variable(*variable)) return false;
261 has_pi = true;
262 continue;
263 }
264 const auto number = std::dynamic_pointer_cast<const NumberNode>(operand);
265 if (!number) {
266 return false;
267 }
268 if (const auto* rational = std::get_if<Rational>(&number->value())) {
269 k = k * *rational;
270 continue;
271 }
272 const auto* integer = std::get_if<BigInt>(&number->value());
273 if (!integer) {
274 return false;
275 }
276 k = k * Rational(*integer);
277 }
278 return has_pi;
279}
