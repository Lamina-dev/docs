12std::shared_ptr<const NumberNode> collectable_power_exponent(
13 const std::shared_ptr<const SymbolicNode>& factor,
14 std::shared_ptr<const SymbolicNode>& base, const FactsQuery& facts, Domain domain,
15 ComputationContext& context) {
16 auto exponent = detail::make_node<NumberNode>(BigInt(1));
17 const auto power = std::dynamic_pointer_cast<const PowerNode>(factor);
18 if (!power) { return exponent; }
19 const auto numeric = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
20 BigInt integer;
21 if (!try_get_integer_value(numeric, integer)) { return exponent; }
22 if (integer <= BigInt(0) &&
23 !(normalization_can_discard(power->base(), facts, domain, context) &&
24 normalization_proved(detail::query_nonzero_value(power->base(), facts, Domain::Complex, context)))) {
25 return exponent;
26 }
27 base = power->base();
28 return detail::make_node<NumberNode>(integer);
29}

55 std::shared_ptr<const SymbolicNode> finish(
56 const FactsQuery& facts, Domain domain, ComputationContext& context, bool expanding,
57 detail::RewriteBudget* budget) const {
58 if (budget) { budget->require_nodes(1); }
59 if (constant_->is_zero() && factors_defined(facts, domain, context)) { return constant_; }
60 std::vector<std::shared_ptr<const SymbolicNode>> operands;
61 std::size_t nodes = 0;
62 if (!constant_->is_one()) {
63 normalization_append(budget, nodes, operands, constant_);
64 }
65 std::vector<std::shared_ptr<const SymbolicNode>> variables;
66 std::size_t variable_nodes = 0;
67 collect_variables(facts, domain, budget, variables, variable_nodes, context);
68 std::sort(variables.begin(), variables.end(), normalization_factor_less);
69 if (budget) {
70 (void)budget->append_size(nodes, variable_nodes);
71 }
72 operands.insert(operands.end(), variables.begin(), variables.end());
73 if (expanding && operands.size() > 1 &&
74 std::dynamic_pointer_cast<const NumberNode>(operands.back())) {
75 std::rotate(operands.begin(), operands.end() - 1, operands.end());
76 }
77 return make_normalized_multiply_node(operands, budget);
78 }

101 }

132std::shared_ptr<const SymbolicNode> NormalizationVisitor::collect_product(
133 const std::vector<std::shared_ptr<const SymbolicNode>>& operands) {
134 ProductAccumulator accumulated;
135 for (const auto& operand : operands) {
136 if (accumulated.add_number(operand)) {
137 continue;
138 }
139 auto base = operand;
140 auto exponent = collectable_power_exponent(operand, base, facts_, domain_, context_);
141 if (base == operand) {
142 const auto power = std::dynamic_pointer_cast<const PowerNode>(operand);
143 if (power && can_combine_nested_power(*power, exponent)) {
144 base = power->base();
145 exponent = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
146 }
147 }
148 accumulated.add_base(operand, base, exponent, rewrite_budget());
149 }
150 return accumulated.finish(facts_, domain_, context_, false, rewrite_budget());
151}

173std::shared_ptr<const SymbolicNode> NormalizationVisitor::expand_complex_product(
174 const std::shared_ptr<const SymbolicNode>& lhs,
175 const std::shared_ptr<const SymbolicNode>& rhs) {
176 const auto left = std::dynamic_pointer_cast<const ComplexNode>(lhs);
177 const auto right = std::dynamic_pointer_cast<const ComplexNode>(rhs);
178 if (!left && !right) {
179 return nullptr;
180 }
181 std::shared_ptr<const SymbolicNode> real;
182 std::shared_ptr<const SymbolicNode> imag;
183 if (left && right) {
184 auto ac = expand_product(left->real(), right->real());
185 auto bd = expand_product(left->imag(), right->imag());
186 auto ad = expand_product(left->real(), right->imag());
187 auto bc = expand_product(left->imag(), right->real());
188 auto negative_bd = expand_product(detail::make_node<NumberNode>(BigInt(-1)), bd);
189 real = SymbolicFactory::create_add({ac, negative_bd}, rewrite_budget());
190 imag = SymbolicFactory::create_add({ad, bc}, rewrite_budget());
191 } else if (left) {
192 real = expand_product(left->real(), rhs);
193 imag = expand_product(left->imag(), rhs);
194 } else {
195 real = expand_product(lhs, right->real());
196 imag = expand_product(lhs, right->imag());
197 }
198 NormalizationVisitor visitor(context_, facts_, domain_, rewrite_budget());
199 real->accept(visitor);
200 auto normalized_real = visitor.get_result();
201 imag->accept(visitor);
202 normalization_check_children(rewrite_budget(), 1, normalized_real, visitor.get_result());
203 return SymbolicFactory::create_complex(normalized_real, visitor.get_result());
204}

206std::shared_ptr<const SymbolicNode> NormalizationVisitor::expand_product(
207 const std::shared_ptr<const SymbolicNode>& lhs,
208 const std::shared_ptr<const SymbolicNode>& rhs) {
209 detail::RewriteScope scope(rewrite_budget());

211 return complex;

214 const auto right = std::dynamic_pointer_cast<const AddNode>(rhs);

216 return distribute_product(lhs, rhs);
217 }

221std::shared_ptr<const SymbolicNode> NormalizationVisitor::distribute_product(
222 const std::shared_ptr<const SymbolicNode>& lhs,
223 const std::shared_ptr<const SymbolicNode>& rhs) {
224 const auto left = std::dynamic_pointer_cast<const AddNode>(lhs);
225 const auto right = std::dynamic_pointer_cast<const AddNode>(rhs);
226 std::size_t left_count = 1, right_count = 1;
227 if (left) {
228 left_count = left->operands().size();
229 }
230 if (right) {
231 right_count = right->operands().size();
232 }
233 normalization_check_count(rewrite_budget(), left_count, right_count, false);
234 std::vector<std::shared_ptr<const SymbolicNode>> terms;
235 std::size_t nodes = 0;
236 for (std::size_t i = 0; i < left_count; ++i) {
237 const auto& a = left ? left->operands()[i] : lhs;
238 for (std::size_t j = 0; j < right_count; ++j) {
239 const auto& b = right ? right->operands()[j] : rhs;
240 normalization_append(rewrite_budget(), nodes, terms, expand_product(a, b));
241 }
242 }
243 normalization_check_arithmetic<AddNode>(rewrite_budget(), terms, nodes);
244 return detail::make_node<AddNode>(terms);
245}

247std::shared_ptr<const SymbolicNode> NormalizationVisitor::collect_expanded_product(
248 const std::shared_ptr<const SymbolicNode>& lhs,
249 const std::shared_ptr<const SymbolicNode>& rhs) {
250 ProductAccumulator accumulated;
251 const auto process = [&](const std::shared_ptr<const SymbolicNode>& factor) {
252 if (accumulated.add_number(factor)) {
253 return;
254 }
255 auto base = factor;
256 auto exponent = collectable_power_exponent(factor, base, facts_, domain_, context_);
257 accumulated.add_base(factor, base, exponent, rewrite_budget());
258 };
259 for (const auto* node : {&lhs, &rhs}) {
260 const auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(*node);
261 if (!multiply) {
262 process(*node);
263 continue;
264 }
265 for (const auto& operand : multiply->operands()) {
266 process(operand);
267 }
268 }
269 return accumulated.finish(facts_, domain_, context_, true, rewrite_budget());
270}
