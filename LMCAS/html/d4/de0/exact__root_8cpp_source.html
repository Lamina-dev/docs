18double conversion_ulp(double value) {
19 if (!std::isfinite(value)) {
20 return std::numeric_limits<double>::infinity();
21 }
22 const double upward = std::nextafter(
23 value, std::numeric_limits<double>::infinity());
24 const double downward = std::nextafter(
25 value, -std::numeric_limits<double>::infinity());
26 double spacing = 0.0;
27 if (std::isfinite(upward)) {
28 spacing = std::max(spacing, std::abs(upward - value));
29 }
30 if (std::isfinite(downward)) {
31 spacing = std::max(spacing, std::abs(value - downward));
32 }
33 return spacing;
34}

36Result<ApproxReal> approximate_real_isolation(
37 ExactRealAlgebraic isolation,
38 const NumericEvaluationOptions& options,
39 ComputationContext& context,
40 const std::string& operation) {
41 auto refined = refine_exact_real_algebraic_to_tolerance(
42 isolation, options.absolute_tolerance, options.relative_tolerance,
43 context, operation);
44 if (!refined) return Result<ApproxReal>::failure(refined.error());
45 const double lower = isolation.lower.to_double();
46 const double upper = isolation.upper.to_double();
47 const double value = lower + (upper - lower) * 0.5;
48 const double error = std::abs(upper - lower) * 0.5 +
49 conversion_ulp(lower) + conversion_ulp(upper) +
50 conversion_ulp(value);
51 if (!std::isfinite(value) || !std::isfinite(error)) {
52 return Result<ApproxReal>::failure(
53 CasErrc::NumericFailure,
54 "certified root enclosure could not be represented as finite double",
55 operation);
56 }
57 return Result<ApproxReal>::success(
58 ApproxReal{value, error, NumericStatus::Finite});
59}

61ApproxReal approximate_component(
62 const Rational& lower,
63 const Rational& upper) {
64 const double lower_value = lower.to_double();
65 const double upper_value = upper.to_double();
66 const double value =
67 lower_value + (upper_value - lower_value) * 0.5;
68 const double error = std::abs(upper_value - lower_value) * 0.5 +
69 conversion_ulp(lower_value) + conversion_ulp(upper_value) +
70 conversion_ulp(value);
71 return ApproxReal{value, error, NumericStatus::Finite};
72}

77Result<ExactRootId> make_exact_root_id(
78 Polynomial<Rational> polynomial,
79 std::size_t index,
80 ComputationContext& context,
81 const std::string& operation) {
82 auto access = context.consume_steps(1, operation);
83 if (!access) return Result<ExactRootId>::failure(access.error());
84 try {
85 polynomial.trim();
86 if (polynomial.degree() <= 0) {
87 return Result<ExactRootId>::failure(
88 CasErrc::InvalidArgument,
89 "RootOf requires a nonconstant exact rational polynomial",
90 operation);
91 }
92 polynomial = polynomial.square_free_part().make_monic();
93 polynomial.variable_name = "_root";
94 if (polynomial.degree() <= 0 ||
95 index >= static_cast<std::size_t>(polynomial.degree())) {
96 return Result<ExactRootId>::failure(
97 CasErrc::InvalidArgument,
98 "RootOf index exceeds the number of distinct roots",
99 operation);
100 }
101 return Result<ExactRootId>::success(
102 ExactRootId{std::move(polynomial), index});
103 } catch (const std::bad_alloc&) {
104 return Result<ExactRootId>::failure(
105 CasErrc::ResourceLimit,
106 "RootOf canonicalization allocation failed", operation);
107 } catch (const std::exception& error) {
108 return Result<ExactRootId>::failure(
109 CasErrc::InternalInvariant, error.what(), operation);
110 }
111}

113Result<ApproxReal> evaluate_root_real(
114 const ExactRootId& root,
115 const NumericEvaluationOptions& options,
116 ComputationContext& context) {
117 constexpr const char* operation = "rootof.evaluate_real";
118 if (!valid_options(options)) {
119 return Result<ApproxReal>::failure(
120 CasErrc::InvalidArgument,
121 "root evaluation tolerances must be finite and positive",
122 operation);
123 }
124 auto isolated = isolate_exact_root(root, context, operation);

128 return Result<ApproxReal>::failure(
129 CasErrc::DomainError,
130 "selected RootOf is not real",
131 operation);
132 }
133 return approximate_real_isolation(
134 std::move(real->value), options, context, operation);

137Result<ApproxComplex> evaluate_root_complex(
138 const ExactRootId& root,
139 const NumericEvaluationOptions& options,
140 ComputationContext& context) {
141 constexpr const char* operation = "rootof.evaluate_complex";
142 if (!valid_options(options)) {
143 return Result<ApproxComplex>::failure(
144 CasErrc::InvalidArgument,
145 "root evaluation tolerances must be finite and positive",
146 operation);
147 }
148 auto isolated = isolate_exact_root(root, context, operation);
149 if (!isolated) return Result<ApproxComplex>::failure(isolated.error());
150 if (auto* real = std::get_if<RealIsolation>(&isolated.value())) {
151 auto approximation = approximate_real_isolation(
152 std::move(real->value), options, context, operation);
153 if (!approximation) {
154 return Result<ApproxComplex>::failure(approximation.error());
155 }
156 return Result<ApproxComplex>::success(ApproxComplex{
157 approximation.value(),
158 ApproxReal{0.0, 0.0, NumericStatus::Finite}});
159 }
160 auto complex = std::get<ComplexIsolation>(std::move(isolated.value()));
161 auto refined = refine_complex_isolation(
162 root.polynomial, complex, options, context, operation);
163 if (!refined) return Result<ApproxComplex>::failure(refined.error());
164 auto real_part = approximate_component(
165 complex.real_lower, complex.real_upper);
166 auto imaginary_part = approximate_component(
167 complex.imaginary_lower, complex.imaginary_upper);
168 if (!std::isfinite(real_part.value) ||
169 !std::isfinite(real_part.absolute_error) ||
170 !std::isfinite(imaginary_part.value) ||
171 !std::isfinite(imaginary_part.absolute_error)) {
172 return Result<ApproxComplex>::failure(
173 CasErrc::NumericFailure,
174 "certified complex root enclosure could not be represented as finite double",
175 operation);
176 }
177 return Result<ApproxComplex>::success(
178 ApproxComplex{real_part, imaginary_part});
179}
