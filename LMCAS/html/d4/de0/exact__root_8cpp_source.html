18double conversion_ulp(double value) {
19 const double upward = std::nextafter(
20 value, std::numeric_limits<double>::infinity());
21 const double downward = std::nextafter(
22 value, -std::numeric_limits<double>::infinity());
23 return std::max(std::abs(upward - value), std::abs(value - downward));
24}

26Result<ApproxReal> approximate_real_isolation(
27 ExactRealAlgebraic isolation,
28 const NumericEvaluationOptions& options,
29 ComputationContext& context,
30 const std::string& operation) {
31 auto refined = refine_exact_real_algebraic_to_tolerance(
32 isolation, options.absolute_tolerance, options.relative_tolerance,
33 context, operation);
34 if (!refined) return Result<ApproxReal>::failure(refined.error());
35 const double lower = isolation.lower.to_double();
36 const double upper = isolation.upper.to_double();
37 const double value = lower + (upper - lower) * 0.5;
38 const double error = std::abs(upper - lower) * 0.5 +
39 conversion_ulp(lower) + conversion_ulp(upper) +
40 conversion_ulp(value);
41 if (!std::isfinite(value) || !std::isfinite(error)) {
42 return Result<ApproxReal>::failure(
43 CasErrc::NumericFailure,
44 "certified root enclosure could not be represented as finite double",
45 operation);
46 }
47 return Result<ApproxReal>::success(
48 ApproxReal{value, error, NumericStatus::Finite});
49}

51ApproxReal approximate_component(
52 const Rational& lower,
53 const Rational& upper) {
54 const double lower_value = lower.to_double();
55 const double upper_value = upper.to_double();
56 const double value =
57 lower_value + (upper_value - lower_value) * 0.5;
58 const double error = std::abs(upper_value - lower_value) * 0.5 +
59 conversion_ulp(lower_value) + conversion_ulp(upper_value) +
60 conversion_ulp(value);
61 return ApproxReal{value, error, NumericStatus::Finite};
62}

67Result<ExactRootId> make_exact_root_id(
68 Polynomial<Rational> polynomial,
69 std::size_t index,
70 ComputationContext& context,
71 const std::string& operation) {
72 auto access = context.consume_steps(1, operation);
73 if (!access) return Result<ExactRootId>::failure(access.error());
74 try {
75 polynomial.trim();
76 if (polynomial.degree() <= 0) {
77 return Result<ExactRootId>::failure(
78 CasErrc::InvalidArgument,
79 "RootOf requires a nonconstant exact rational polynomial",
80 operation);
81 }
82 polynomial = polynomial.square_free_part().make_monic();
83 polynomial.variable_name = "_root";
84 if (polynomial.degree() <= 0 ||
85 index >= static_cast<std::size_t>(polynomial.degree())) {
86 return Result<ExactRootId>::failure(
87 CasErrc::InvalidArgument,
88 "RootOf index exceeds the number of distinct roots",
89 operation);
90 }
91 return Result<ExactRootId>::success(
92 ExactRootId{std::move(polynomial), index});
93 } catch (const std::bad_alloc&) {
94 return Result<ExactRootId>::failure(
95 CasErrc::ResourceLimit,
96 "RootOf canonicalization allocation failed", operation);
97 } catch (const std::exception& error) {
98 return Result<ExactRootId>::failure(
99 CasErrc::InternalInvariant, error.what(), operation);
100 }
101}

103Result<ApproxReal> evaluate_root_real(
104 const ExactRootId& root,
105 const NumericEvaluationOptions& options,

122 }

127Result<ApproxComplex> evaluate_root_complex(
128 const ExactRootId& root,
129 const NumericEvaluationOptions& options,
130 ComputationContext& context) {
131 constexpr const char* operation = "rootof.evaluate_complex";
132 if (!valid_options(options)) {
133 return Result<ApproxComplex>::failure(
134 CasErrc::InvalidArgument,
135 "root evaluation tolerances must be finite and positive",
136 operation);
137 }
138 auto isolated = isolate_exact_root(root, context, operation);
139 if (!isolated) return Result<ApproxComplex>::failure(isolated.error());
140 if (auto* real = std::get_if<RealIsolation>(&isolated.value())) {
141 auto approximation = approximate_real_isolation(
142 std::move(real->value), options, context, operation);
143 if (!approximation) {
144 return Result<ApproxComplex>::failure(approximation.error());
145 }
146 return Result<ApproxComplex>::success(ApproxComplex{
147 approximation.value(),
148 ApproxReal{0.0, 0.0, NumericStatus::Finite}});
149 }
150 auto complex = std::get<ComplexIsolation>(std::move(isolated.value()));
151 auto refined = refine_complex_isolation(
152 root.polynomial, complex, options, context, operation);
153 if (!refined) return Result<ApproxComplex>::failure(refined.error());
154 auto real_part = approximate_component(
155 complex.real_lower, complex.real_upper);
156 auto imaginary_part = approximate_component(
157 complex.imaginary_lower, complex.imaginary_upper);
158 if (!std::isfinite(real_part.value) ||
159 !std::isfinite(real_part.absolute_error) ||
160 !std::isfinite(imaginary_part.value) ||
161 !std::isfinite(imaginary_part.absolute_error)) {
162 return Result<ApproxComplex>::failure(
163 CasErrc::NumericFailure,
164 "certified complex root enclosure could not be represented as finite double",
165 operation);
166 }
167 return Result<ApproxComplex>::success(
168 ApproxComplex{real_part, imaginary_part});
