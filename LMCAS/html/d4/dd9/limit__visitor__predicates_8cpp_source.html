33std::optional<int> LimitVisitor::assumed_node_sign(
34 const std::shared_ptr<const SymbolicNode>& node) const {
35 std::optional<detail::AssumptionFacts> assumed;
36 if (assumption_ctx_) { assumed.emplace(*assumption_ctx_); }
37 const auto& facts = assumed ? static_cast<const FactsQuery&>(*assumed) : detail::no_facts();
38 auto nonzero = detail::query_nonzero_value(node, facts, Domain::Real, *context_);
39 if (!nonzero) { throw nonzero.error(); }
40 if (nonzero.value() == Tribool::False) { return 0; }
41 auto positive = detail::query_positive_value(node, facts, *context_);
42 if (!positive) { throw positive.error(); }
43 if (positive.value() == Tribool::True) { return 1; }
44 auto nonnegative = detail::query_nonnegative_value(node, facts, *context_);
45 if (!nonnegative) { throw nonnegative.error(); }
46 if (nonnegative.value() == Tribool::False) {
47 auto real = detail::query_real_value(node, facts, *context_);
48 if (!real) { throw real.error(); }
49 if (real.value() == Tribool::True) { return -1; }
50 }
51 return std::nullopt;
52}

60bool LimitVisitor::is_inf(const std::shared_ptr<const SymbolicNode>& node) const {
61 if (!node) { return false; }
62 if (auto f = std::dynamic_pointer_cast<const FunctionNode>(node))
63 { return f->type() == FunctionNode::FuncType::Infinity; }
64 if (auto product = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
65 return is_infinite_product(*product);
66 }

74bool LimitVisitor::is_neg_inf(const std::shared_ptr<const SymbolicNode>& node) const {
75 if (auto product = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
76 if (!is_inf(node)) { return false; }
77 int sign = 1;
78 for (const auto& operand : product->operands())
79 { sign *= is_inf(operand) ? (is_neg_inf(operand) ? -1 : 1) : *get_node_sign(operand); }
80 return sign < 0;
81 }
82 if (auto sum = std::dynamic_pointer_cast<const AddNode>(node)) {
83 if (!is_inf(node)) { return false; }
84 for (const auto& operand : sum->operands()) { if (is_inf(operand)) { return is_neg_inf(operand); } }
85 }
86 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
87 BigInt integer;
88 auto number = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
89 return number && try_get_integer_value(number, integer) && integer.is_odd() && is_neg_inf(power->base());
90 }
91 return false;
92}

94std::optional<int> LimitVisitor::get_node_sign(
95 const std::shared_ptr<const SymbolicNode>& node) const {
96 if (!node) {
97 return std::nullopt;
98 }
99 if (node->is_zero()) {
100 return 0;
101 }
102 if (auto number = std::dynamic_pointer_cast<const NumberNode>(node)) {
103 if (std::holds_alternative<double>(number->value())) {
104 if (!std::isfinite(std::get<double>(number->value()))) { return std::nullopt; }
105 return get_sign(node);
106 }
107 if (std::holds_alternative<BigInt>(number->value())) {
108 return get_sign(node);
109 }
110 if (std::holds_alternative<Rational>(number->value())) {
111 return get_sign(node);
112 }
113 }
114 if (is_inf(node)) {
115 return is_neg_inf(node) ? -1 : 1;
116 }
117 return assumed_node_sign(node);
118}

159bool LimitVisitor::is_bounded(const std::shared_ptr<const SymbolicNode>& node) const {
160 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(node)) {
161 return (function->type() == FunctionNode::FuncType::Sin ||
162 function->type() == FunctionNode::FuncType::Cos ||
163 function->type() == FunctionNode::FuncType::ArcTan) && defined_near_point(node);
