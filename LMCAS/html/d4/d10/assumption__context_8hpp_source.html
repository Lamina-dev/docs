34/**
35 * @brief Scoped assumption context with push/pop management.
36 *
37 * Maintains a stack of scopes, each containing a PropertyStore and RelationStore.
38 * Queries use read-through semantics: searching from the top scope down to root,
39 * returning the first scope that has a declaration for the queried symbol.
40 * Child scope declarations shadow parent declarations without modifying them.
41 * Popping a scope restores all query results to their pre-push values.
42 */

317/**
318 * @brief Declaration of an assumption to apply within a with_assumptions block.
319 *
320 * Represents a single domain, sign, or relational assumption that can be
321 * applied to an AssumptionContext scope.
322 */
323class AssumptionDecl {
324public:
325 /// The type of assumption being declared.
326 enum class Type { Domain, Sign, Relation };
327
328 /// Construct a domain assumption declaration.
329 static AssumptionDecl make_domain(const std::string& sym, Domain d) {
330 return AssumptionDecl(DomainDeclaration{sym, d});
331 }
332
333 /// Construct a sign assumption declaration.
334 static AssumptionDecl make_sign(const std::string& sym, Sign s) {
335 return AssumptionDecl(SignDeclaration{sym, s});
336 }
337
338 /// Construct a relational assumption declaration.
339 static AssumptionDecl make_relation(const SymbolicExpr& rel) {
340 return AssumptionDecl(rel);
341 }
342
343 Type type() const noexcept {
344 if (std::holds_alternative<DomainDeclaration>(payload_)) return Type::Domain;
345 if (std::holds_alternative<SignDeclaration>(payload_)) return Type::Sign;
346 return Type::Relation;
347 }
348
349 const std::string& symbol() const {
350 if (const auto* domain = std::get_if<DomainDeclaration>(&payload_)) {
351 return domain->symbol;
352 }
353 if (const auto* sign = std::get_if<SignDeclaration>(&payload_)) {
354 return sign->symbol;
355 }
356 throw std::logic_error("relational assumption has no symbol field");
357 }
358
359 Domain domain() const {
360 return std::get<DomainDeclaration>(payload_).domain;
361 }
362
363 Sign sign() const {
364 return std::get<SignDeclaration>(payload_).sign;
365 }
366
367 const SymbolicExpr& relation() const {
368 return std::get<SymbolicExpr>(payload_);
369 }
370
371private:
372 struct DomainDeclaration {
373 std::string symbol;
374 Domain domain;
375 };
376
377 struct SignDeclaration {
378 std::string symbol;
379 Sign sign;
380 };
381
382 using Payload = std::variant<DomainDeclaration, SignDeclaration, SymbolicExpr>;
383
384 explicit AssumptionDecl(DomainDeclaration declaration)
385 : payload_(std::move(declaration)) {}
386 explicit AssumptionDecl(SignDeclaration declaration)
387 : payload_(std::move(declaration)) {}
388 explicit AssumptionDecl(SymbolicExpr relation)
389 : payload_(std::move(relation)) {}
390
391 Payload payload_;
392};

394inline AssumptionVoidResult apply_assumption_decl_checked(
395 AssumptionContext& ctx,
396 const AssumptionDecl& decl) {
397 switch (decl.type()) {
398 case AssumptionDecl::Type::Domain:
399 return ctx.assume_domain_checked(decl.symbol(), decl.domain());
400 case AssumptionDecl::Type::Sign:
401 return ctx.assume_sign_checked(decl.symbol(), decl.sign());
402 case AssumptionDecl::Type::Relation:
403 return ctx.assume_checked(decl.relation());
404 }
405 return AssumptionVoidResult::failure(
406 CasErrc::InternalInvariant, "unknown assumption declaration type", "with_assumptions");
407}

409/**
410 * @brief Execute a callable within a temporary assumption scope.
411 *
412 * Pushes a new scope on the context, applies all declarations from the
413 * provided set, invokes the callable, pops the scope, and returns the result.
414 * Exception-safe: the scope is popped even if the callable throws.
415 *
416 * @tparam F Callable type (must be invocable with no arguments)
417 * @param ctx AssumptionContext to operate on
418 * @param decls Set of assumption declarations to apply in the new scope
419 * @param callable Function to invoke under the temporary assumptions
420 * @return The result of invoking callable()
421 *
422 */
423template<typename F>
424auto with_assumptions(AssumptionContext& ctx,
425 std::initializer_list<AssumptionDecl> decls,
426 F&& callable)
427 -> std::enable_if_t<!std::is_void_v<decltype(callable())>, decltype(callable())>
428{
429 ctx.push();
430 try {
431 for (const auto& decl : decls) {
432 switch (decl.type()) {
433 case AssumptionDecl::Type::Domain:
434 ctx.assume_domain(decl.symbol(), decl.domain());
435 break;
436 case AssumptionDecl::Type::Sign:
437 ctx.assume_sign(decl.symbol(), decl.sign());
438 break;
439 case AssumptionDecl::Type::Relation:
440 ctx.assume(decl.relation());
441 break;
442 }
443 }
444 auto result = callable();
445 ctx.pop();
446 return result;
447 } catch (...) {
448 ctx.pop();
449 throw;
450 }
451}

453/**
454 * @brief Applies assumptions for the duration of a non-void callable.
455 *
456 * Returns `CasError` on declaration or callable failure. The scope is always
457 * removed before returning.
458 */
459template<typename F>
460auto with_assumptions_checked(AssumptionContext& ctx,
461 const std::vector<AssumptionDecl>& decls,
462 F&& callable)
463 -> std::enable_if_t<!std::is_void_v<decltype(callable())>, Result<decltype(callable())>>
464{
465 using ReturnT = decltype(callable());
466 ctx.push();
467 try {
468 for (const auto& decl : decls) {
469 auto applied = apply_assumption_decl_checked(ctx, decl);
470 if (!applied.has_value()) {
471 ctx.pop();
472 return Result<ReturnT>::failure(applied.error());
473 }
474 }
475 auto result = callable();
476 ctx.pop();
477 return Result<ReturnT>::success(std::move(result));
478 } catch (const std::bad_alloc&) {
479 ctx.pop();
480 return Result<ReturnT>::failure(
481 CasErrc::ResourceLimit, "with_assumptions allocation failed", "with_assumptions");
482 } catch (const std::exception& ex) {
483 ctx.pop();
484 return Result<ReturnT>::failure(
485 CasErrc::InternalInvariant, ex.what(), "with_assumptions");
486 }
487}
