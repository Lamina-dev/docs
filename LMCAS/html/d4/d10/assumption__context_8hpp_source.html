35 * @brief Scoped assumption context with push/pop management.
36 *
37 * Maintains a stack of scopes, each containing a PropertyStore and RelationStore.
38 * Queries use read-through semantics: searching from the top scope down to root,
39 * 查询按子作用域到父作用域的顺序定位首个符号声明.
40 * 子作用域声明覆盖查询结果,同时保留父作用域状态;弹出作用域后恢复先前结果.

314 * @brief Declaration of an assumption to apply within a with_assumptions block.
315 *
316 * Represents a single domain, sign, or relational assumption that can be
317 * applied to an AssumptionContext scope.
318 */
319class AssumptionDecl {
320public:
321 /// The type of assumption being declared.
322 enum class Type { Domain, Sign, Relation };
323
324 /// Construct a domain assumption declaration.
325 static AssumptionDecl make_domain(const std::string& sym, Domain d) {
326 return AssumptionDecl(DomainDeclaration{sym, d});
327 }
328
329 /// Construct a sign assumption declaration.
330 static AssumptionDecl make_sign(const std::string& sym, Sign s) {
331 return AssumptionDecl(SignDeclaration{sym, s});
332 }
333
334 /// Construct a relational assumption declaration.
335 static AssumptionDecl make_relation(const SymbolicExpr& rel) {
336 return AssumptionDecl(rel);
337 }
338
339 Type type() const noexcept {
340 if (std::holds_alternative<DomainDeclaration>(payload_)) return Type::Domain;
341 if (std::holds_alternative<SignDeclaration>(payload_)) return Type::Sign;
342 return Type::Relation;
343 }
344
345 const std::string& symbol() const {
346 if (const auto* domain = std::get_if<DomainDeclaration>(&payload_)) {
347 return domain->symbol;
348 }
349 if (const auto* sign = std::get_if<SignDeclaration>(&payload_)) {
350 return sign->symbol;
351 }
352 throw std::logic_error("relational assumption has no symbol field");
353 }
354
355 Domain domain() const {
356 return std::get<DomainDeclaration>(payload_).domain;
357 }
358
359 Sign sign() const {
360 return std::get<SignDeclaration>(payload_).sign;
361 }
362
363 const SymbolicExpr& relation() const {
364 return std::get<SymbolicExpr>(payload_);
365 }

367private:
368 struct DomainDeclaration {
369 std::string symbol;
370 Domain domain;
371 };
372
373 struct SignDeclaration {
374 std::string symbol;
375 Sign sign;
376 };

378 using Payload = std::variant<DomainDeclaration, SignDeclaration, SymbolicExpr>;

380 explicit AssumptionDecl(DomainDeclaration declaration)
381 : payload_(std::move(declaration)) {}
382 explicit AssumptionDecl(SignDeclaration declaration)
383 : payload_(std::move(declaration)) {}
384 explicit AssumptionDecl(SymbolicExpr relation)
385 : payload_(std::move(relation)) {}

387 Payload payload_;

390inline AssumptionVoidResult apply_assumption_decl_checked(
391 AssumptionContext& ctx,
392 const AssumptionDecl& decl) {
393 switch (decl.type()) {
394 case AssumptionDecl::Type::Domain:
395 return ctx.assume_domain_checked(decl.symbol(), decl.domain());
396 case AssumptionDecl::Type::Sign:
397 return ctx.assume_sign_checked(decl.symbol(), decl.sign());
398 case AssumptionDecl::Type::Relation:
399 return ctx.assume_checked(decl.relation());
400 }
401 return AssumptionVoidResult::failure(
402 CasErrc::InternalInvariant, "unknown assumption declaration type", "with_assumptions");
403}

405/**
406 * @brief Execute a callable within a temporary assumption scope.
407 *
408 * Declaration and callable failures are returned as `CasError`. The temporary
409 * scope is removed before every return.
410 */
411template<typename F>
412auto with_assumptions(AssumptionContext& ctx,
413 const std::vector<AssumptionDecl>& decls,
414 F&& callable)
415 -> std::enable_if_t<!std::is_void_v<decltype(callable())>, Result<decltype(callable())>>
416{
417 using ReturnT = decltype(callable());
418 ctx.push();
419 try {
420 for (const auto& decl : decls) {
421 auto applied = apply_assumption_decl_checked(ctx, decl);
422 if (!applied) {
423 auto popped = ctx.pop();
424 if (!popped) return Result<ReturnT>::failure(popped.error());
425 return Result<ReturnT>::failure(applied.error());
426 }
427 }
428 auto result = callable();
429 auto popped = ctx.pop();
430 if (!popped) return Result<ReturnT>::failure(popped.error());
431 return Result<ReturnT>::success(std::move(result));
432 } catch (const detail::ResultPropagation& ex) {
433 auto popped = ctx.pop();
434 if (!popped) return Result<ReturnT>::failure(popped.error());
435 return Result<ReturnT>::failure(ex.error());
436 } catch (const std::bad_alloc&) {
437 auto popped = ctx.pop();
438 if (!popped) return Result<ReturnT>::failure(popped.error());
439 return Result<ReturnT>::failure(
440 CasErrc::ResourceLimit, "with_assumptions allocation failed", "with_assumptions");
441 } catch (const std::exception& ex) {
442 auto popped = ctx.pop();
443 if (!popped) return Result<ReturnT>::failure(popped.error());
444 return Result<ReturnT>::failure(
445 CasErrc::InternalInvariant, ex.what(), "with_assumptions");
446 }
447}

449template<typename F>
450auto with_assumptions(AssumptionContext& ctx,
451 const std::vector<AssumptionDecl>& decls,
452 F&& callable)
453 -> std::enable_if_t<std::is_void_v<decltype(callable())>, AssumptionVoidResult>
454{

468 return AssumptionVoidResult::success();

476 return AssumptionVoidResult::failure(
