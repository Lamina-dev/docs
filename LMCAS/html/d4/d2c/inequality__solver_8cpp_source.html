37Result<IntervalUnion> InequalitySolver::solve_inequality_checked(
38 const std::shared_ptr<SymbolicExpr>& expr,
39 InequalityType type,
40 const std::string& variable,
41 ComputationContext& context) {
42 if (!expr || !LMCAS::detail::node(expr)) {
43 return Result<IntervalUnion>::failure(
44 CasErrc::InvalidArgument, "inequality expression cannot be null",
45 kCheckedInequalityOperation);
46 }
47 if (variable.empty()) {
48 return Result<IntervalUnion>::failure(
49 CasErrc::InvalidArgument, "inequality variable cannot be empty",
50 kCheckedInequalityOperation);
51 }
52
53 try {
54 auto recognized = recognize_rational_polynomial(*expr, variable, context);
55 if (!recognized) {
56 return Result<IntervalUnion>::failure(recognized.error());
57 }
58 if (!recognized.value()) {
59 return Result<IntervalUnion>::failure(
60 CasErrc::Inconclusive,
61 "expression is not an exact rational polynomial in the requested variable",
62 kCheckedInequalityOperation);
63 }
64 if (recognized.value()->degree() == 2 &&
65 recognized.value()->coeffs.size() >= 3) {
66 return solve_exact_quadratic_inequality(
67 *recognized.value(), type, context);
68 }
69 if (recognized.value()->degree() <= 1) {
70 return solve_exact_affine_inequality(
71 *recognized.value(), type, context);
72 }
73 return solve_exact_polynomial_inequality(
74 *recognized.value(), type, context);
75 } catch (const std::bad_alloc&) {
76 return Result<IntervalUnion>::failure(
77 CasErrc::ResourceLimit, "inequality allocation failed",
78 kCheckedInequalityOperation);
79 } catch (const std::exception& error) {
80 return Result<IntervalUnion>::failure(
81 CasErrc::InternalInvariant, error.what(), kCheckedInequalityOperation);
82 }

93Result<IntervalUnion> InequalitySolver::solve_inequalities_checked(
94 const std::vector<std::pair<std::shared_ptr<SymbolicExpr>,
95 InequalityType>>& inequalities,
96 const std::string& variable,
97 ComputationContext& context) {
98 if (variable.empty()) {
99 return Result<IntervalUnion>::failure(
100 CasErrc::InvalidArgument, "inequality variable cannot be empty",
101 "solve_inequalities_checked");
102 }
103
104 auto initial_step = context.consume_steps(1, "solve_inequalities_checked");
105 if (!initial_step) {
106 return Result<IntervalUnion>::failure(initial_step.error());
107 }
108 if (inequalities.empty()) {
109 return Result<IntervalUnion>::success(IntervalUnion::entire_line());
110 }
111
112 std::optional<IntervalUnion> aggregate;
113 for (const auto& inequality : inequalities) {
114 auto solved = solve_inequality_checked(
115 inequality.first, inequality.second, variable, context);
116 if (!solved) {
117 return Result<IntervalUnion>::failure(solved.error());
118 }
119 if (!aggregate) {
120 aggregate = std::move(solved.value());
121 continue;
122 }
123 auto intersection = aggregate->intersect_checked(solved.value(), context);
124 if (!intersection) {
125 return Result<IntervalUnion>::failure(intersection.error());
126 }
127 aggregate = std::move(intersection.value());
128 if (aggregate->is_empty()) {
129 break;
130 }
131 }
132 return Result<IntervalUnion>::success(std::move(*aggregate));
133}

145static IntervalUnion solve_constant_inequality(
146 const Polynomial<SymbolicPolyCoeff>& poly, InequalityType type) {
147 auto lc = poly.lead_coeff().val;
148 if (!lc) {
149 return IntervalUnion::empty();
150 }
151 int sign = exact_numeric_sign(lc);
152 if (sign == 0 && !lc->is_zero()) {
153 auto simplified = lc->simplify();
154 if (!simplified || !simplified->is_zero()) {
155 return IntervalUnion::empty();
156 }
157 }
158
159 bool satisfies = false;
160 switch (type) {
161 case InequalityType::GreaterThan: satisfies = (sign > 0); break;
162 case InequalityType::GreaterEqual: satisfies = (sign >= 0); break;
163 case InequalityType::LessThan: satisfies = (sign < 0); break;
164 case InequalityType::LessEqual: satisfies = (sign <= 0); break;
165 default:
166 return IntervalUnion::empty();
167 }
168 return satisfies ? IntervalUnion::entire_line() : IntervalUnion::empty();
169}

171IntervalUnion InequalitySolver::solve_inequality(
172 const std::shared_ptr<SymbolicExpr>& expr,
173 InequalityType type,
174 const std::string& variable) {
175
176 if (!expr) {
177 return IntervalUnion::empty();
178 }
179
180 auto converted = symbolic_to_poly<SymbolicPolyCoeff>(expr, variable);
181 if (!converted) throw std::invalid_argument(converted.error().message);
182 const auto& poly = converted.value();
183 if (poly.is_zero()) {
184 if (type == InequalityType::GreaterEqual || type == InequalityType::LessEqual) {
185 return IntervalUnion::entire_line();
186 }
187 return IntervalUnion::empty();
188 }
189
190 if (poly.degree() <= 0) {
191 return solve_constant_inequality(poly, type);
192 }
193
194 {
195 auto poly_rat = symbolic_to_poly<Rational>(expr, variable);
196 if (!poly_rat) throw std::invalid_argument(poly_rat.error().message);
197
198 }
199
200 auto roots_with_mult = find_roots_with_multiplicity(expr, variable);
201
202 std::sort(roots_with_mult.begin(), roots_with_mult.end(),
203 [](const auto& a, const auto& b) {
204 return root_less_than(a.first, b.first);
205 });
206
207 std::vector<std::pair<std::shared_ptr<SymbolicExpr>, int>> unique_roots;
208 for (const auto& [root, mult] : roots_with_mult) {
209 if (!unique_roots.empty() && roots_equal(unique_roots.back().first, root)) {
210
211 unique_roots.back().second = std::max(unique_roots.back().second, mult);
212 } else {
213 unique_roots.push_back({root, mult});
214 }
215 }
216
217 std::vector<std::shared_ptr<SymbolicExpr>> roots;
218 std::vector<int> multiplicities;
219 for (const auto& [root, mult] : unique_roots) {
220 roots.push_back(root);
221 multiplicities.push_back(mult);
222 }
223
224 auto chart = build_sign_chart(expr, variable, roots, multiplicities);
225
226 return select_intervals(chart, type, roots, multiplicities);
227}

230IntervalUnion InequalitySolver::solve_inequalities(
231 const std::vector<std::pair<std::shared_ptr<SymbolicExpr>,
232 InequalityType>>& inequalities,
233 const std::string& variable) {
234
235 if (inequalities.empty()) {
236 return IntervalUnion::entire_line();
237 }
238
239 IntervalUnion result = solve_inequality(inequalities[0].first, inequalities[0].second, variable);
240
241 for (size_t i = 1; i < inequalities.size(); ++i) {
242 auto solution = solve_inequality(inequalities[i].first, inequalities[i].second, variable);
243 result = result.intersect(solution);
244 if (result.is_empty()) {
245 break;
246 }
247 }
248
249 return result;
250}
