24int exact_numeric_sign(const std::shared_ptr<SymbolicExpr>& expr) {
25 if (!expr || !LMCAS::detail::node(expr)) return 0;
26 auto simplified = expr->simplify();
27 if (!simplified || !LMCAS::detail::node(simplified)) return 0;
28 auto num = std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(simplified));
29 if (!num) return 0;
30 if (std::holds_alternative<BigInt>(num->value())) {
31 const auto& value = std::get<BigInt>(num->value());
32 if (value.is_zero()) return 0;
33 return value.IsNegative() ? -1 : 1;
34 }
35 if (std::holds_alternative<Rational>(num->value())) {
36 const auto& value = std::get<Rational>(num->value());
37 if (value.get_numerator().is_zero()) return 0;
38 return value.get_numerator().IsNegative() ? -1 : 1;
39 }
40 const auto value = std::get<lmmc_real_t>(num->value());
41 if (!std::isfinite(value) || value == 0.0) return 0;
42 return value < 0 ? -1 : 1;
43}

45int determine_leading_sign(const Polynomial<SymbolicPolyCoeff>& poly) {
46 if (poly.is_zero()) return 0;
47 auto lc = poly.lead_coeff().val;
48 if (!lc) return 1;
49 auto simplified = lc->simplify();
50 if (!simplified) return 1;
51
52 if (auto val = try_checked_numeric_constant(*simplified)) {
53 if (*val > 0) return 1;
54 if (*val < 0) return -1;
55 }
56
57 if (auto num = std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(simplified))) {
58 if (std::holds_alternative<BigInt>(num->value())) {
59 return std::get<BigInt>(num->value()).IsNegative() ? -1 : 1;
60 }
61 if (std::holds_alternative<Rational>(num->value())) {
62 return std::get<Rational>(num->value()).get_numerator().IsNegative() ? -1 : 1;
63 }
64 if (std::holds_alternative<lmmc_real_t>(num->value())) {
65 return std::get<lmmc_real_t>(num->value()) < 0 ? -1 : 1;
66 }
67 }
68 return 1;
69}

71static std::shared_ptr<SymbolicExpr> snap_verified_integer_root(
72 const Polynomial<Rational>& poly,
73 const std::shared_ptr<SymbolicExpr>& root) {
74 if (!root || poly.is_zero()) return root;
75 auto numeric = try_checked_numeric_constant(*root);
76 if (!numeric || !std::isfinite(*numeric)) return root;
77
78 double rounded = std::round(*numeric);
79 if (*numeric != rounded) return root;
80 if (rounded < static_cast<double>(std::numeric_limits<long long>::min()) ||
81 rounded > static_cast<double>(std::numeric_limits<long long>::max())) {
82 return root;
83 }
84
85 Rational candidate(BigInt(static_cast<long long>(rounded)));
86 if (poly.eval(candidate) == Rational(0)) {
87 return SymbolicExpr::number(candidate);
88 }
89 return root;
90}

92std::vector<std::pair<std::shared_ptr<SymbolicExpr>, int>> find_roots_with_multiplicity(
93 const std::shared_ptr<SymbolicExpr>& expr,
94 const std::string& variable) {
95
96 std::vector<std::pair<std::shared_ptr<SymbolicExpr>, int>> result;
97
98 auto poly_rat = symbolic_to_poly<Rational>(expr, variable);
99 if (poly_rat.is_zero() || poly_rat.degree() <= 0) {
100 return result;
101 }
102
103 auto factors = square_free_factorization(poly_rat);
104 if (factors.empty()) {
105
106 auto poly_spc = symbolic_to_poly<SymbolicPolyCoeff>(expr, variable);
107 if (!poly_spc.is_zero() && poly_spc.degree() >= 1) {
108 auto roots = solve_by_factoring(poly_spc, variable);
109 for (const auto& root : roots) {
110 if (!root) continue;
111 auto verified_root = snap_verified_integer_root(poly_rat, root);
112 if (try_checked_numeric_constant(*verified_root)) {
113 result.push_back({verified_root, 1});
114 }
115 }
116 }
117 return result;
118 }
119
120 for (const auto& [factor, mult] : factors) {
121 if (factor.degree() <= 0) continue;
122
123 if (factor.degree() == 1) {
124
125 Rational a = factor.coeffs[1];
126 Rational b = factor.coeffs[0];
127 if (a != Rational(0)) {
128 Rational root_val = Rational(0) - b / a;
129 auto root_expr = SymbolicExpr::number(root_val);
130 if (try_checked_numeric_constant(*root_expr)) {
131 result.push_back({root_expr, mult});
132 }
133 }
134 continue;
135 }
136
137 if (factor.degree() == 2) {
138 Rational a = factor.coeffs[2];
139 Rational b = factor.coeffs[1];
140 Rational c = factor.coeffs[0];
141
142 Rational disc = b * b - Rational(4) * a * c;
143 if (disc < Rational(0)) continue;
144 double disc_val = disc.to_double();
145
146 double a_val = a.to_double();
147 double b_val = b.to_double();
148 double sqrt_disc = std::sqrt(disc_val);
149
150 double r1;
151 double r2;
152 if (sqrt_disc == 0.0) {
153 r1 = -b_val / (2.0 * a_val);
154 r2 = r1;
155 } else {
156 const double q =
157 -0.5 * b_val - 0.5 * std::copysign(sqrt_disc, b_val);
158 r1 = q / a_val;
159 r2 = c.to_double() / q;
160 }
161
162 if (std::isfinite(r1)) {
163 auto root = snap_verified_integer_root(factor, SymbolicExpr::number(r1));
164 result.push_back({root, mult});
165 }
166 if (std::isfinite(r2) && r2 != r1) {
167 auto root = snap_verified_integer_root(factor, SymbolicExpr::number(r2));
168 result.push_back({root, mult});
169 }
170 continue;
171 }
172
173 std::vector<SymbolicPolyCoeff> spc_coeffs;
174 for (int i = 0; i <= factor.degree(); ++i) {
175 spc_coeffs.push_back(SymbolicPolyCoeff(SymbolicExpr::number(factor.coeffs[i])));
176 }
177 Polynomial<SymbolicPolyCoeff> factor_spc(spc_coeffs, variable);
178
179 auto factor_roots = solve_by_factoring(factor_spc, variable);
180 for (const auto& root : factor_roots) {
181 if (!root) continue;
182 auto verified_root = snap_verified_integer_root(factor, root);
183 if (try_checked_numeric_constant(*verified_root)) {
184 result.push_back({verified_root, mult});
185 }
186 }
187 }
188
189 return result;
190}

192bool root_less_than(const std::shared_ptr<SymbolicExpr>& a,
193 const std::shared_ptr<SymbolicExpr>& b) {
194 if (!a || !b) return false;
195 auto va = try_checked_numeric_constant(*a);
196 auto vb = try_checked_numeric_constant(*b);
197 if (va && vb) {
198 return *va < *vb;
199 }
200
201 /// 参数根的顺序通过 (a - b) 的符号推导;
202 /// 二次公式的根差可化简为 +/-sqrt(disc)/a.
203 auto diff = SymbolicExpr::add(a, SymbolicExpr::multiply(b, SymbolicExpr::number(-1)))->simplify();
204 if (auto vd = try_checked_numeric_constant(*diff)) {
205 return *vd < 0;
206 }
207
208 /// 尝试判断差值表达式的符号结构:
209 /// 如果差值形如 k * sqrt(...) / denom,判断各因子的符号.
210 /// 这覆盖了二次公式根差 = sqrt(delta)/a 的情形.
211 auto try_sign_of_node = [](const std::shared_ptr<const SymbolicNode>& node) -> int {
212 if (!node) return 0;
213
214 if (auto num_node = std::dynamic_pointer_cast<const NumberNode>(node)) {
215 if (std::holds_alternative<BigInt>(num_node->value())) {
216 auto& v = std::get<BigInt>(num_node->value());
217 if (v.is_zero()) return 0;
218 return v.IsNegative() ? -1 : 1;
219 }
220 if (std::holds_alternative<Rational>(num_node->value())) {
221 auto& v = std::get<Rational>(num_node->value());
222 if (v.get_numerator().is_zero()) return 0;
223 return v.get_numerator().IsNegative() ? -1 : 1;
224 }
225 if (std::holds_alternative<lmmc_real_t>(num_node->value())) {
226 auto v = std::get<lmmc_real_t>(num_node->value());
227 if (v == 0.0) return 0;
228 return v < 0 ? -1 : 1;
229 }
230 }
231
232 /// sqrt(...) 非负(假设参数使判别式非负)
233 if (auto fn = std::dynamic_pointer_cast<const FunctionNode>(node)) {
234 if (fn->type() == FunctionNode::FuncType::Sqrt) return 1;
235 }
236
237 /// x^(1/2) 或 x^0.5 也是平方根,非负
238 if (auto pw = std::dynamic_pointer_cast<const PowerNode>(node)) {
239 auto exp_expr = LMCAS::detail::make_expression_ptr(pw->exponent());
240 if (auto ev = try_checked_numeric_constant(*exp_expr)) {
241 if (*ev > 0 && *ev < 1.0) {
242 /// base^(正分数) >= 0(假设 base 为判别式等非负量)
243 return 1;
244 }
245 }
246
247 /// 对于整数指数,判断底数符号
248 auto base_expr = LMCAS::detail::make_expression_ptr(pw->base());
249 auto bv_checked = try_checked_numeric_constant(*base_expr);
250 auto ev_checked = try_checked_numeric_constant(*exp_expr);
251 if (bv_checked && ev_checked) {
252 double bv = *bv_checked;
253 double ev = *ev_checked;
254 int ei = static_cast<int>(ev);
255 if (std::abs(ev - ei) < 1e-10) {
256 if (bv > 0) return 1;
257 if (bv < 0) return (ei % 2 == 0) ? 1 : -1;
258 }
259 }
260 }
261
262 return 0;
263 };
264
265 /// 对乘积节点,各因子符号之积
266 auto try_sign_of_expr = [&try_sign_of_node](const std::shared_ptr<SymbolicExpr>& expr) -> int {
267 if (!expr || !LMCAS::detail::node(expr)) return 0;
268
269 /// 直接节点
270 int s = try_sign_of_node(LMCAS::detail::node(expr));
271 if (s != 0) return s;
272
273 /// 乘积:各因子符号之积
274 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(LMCAS::detail::node(expr))) {
275 int sign = 1;
276 for (const auto& op : mul->operands()) {
277 int os = try_sign_of_node(op);
278 if (os == 0) return 0;
279 sign *= os;
280 }
281 return sign;
282 }
283
284 return 0;
285 };
286
287 int diff_sign = try_sign_of_expr(diff);
288 if (diff_sign < 0) return true;
289 if (diff_sign > 0) return false;
290
291 return false;
292}

294bool roots_equal(const std::shared_ptr<SymbolicExpr>& a,
295 const std::shared_ptr<SymbolicExpr>& b) {
296 if (!a || !b) return false;

299 if (numeric_a && numeric_b) {

301 }
302 auto difference = SymbolicExpr::add(

304 SymbolicExpr::multiply(SymbolicExpr::number(-1), b));
305 auto simplified = difference ? difference->simplify() : nullptr;
306 return simplified && simplified->is_zero();

317std::vector<std::shared_ptr<SymbolicExpr>> solve_symbolic_poly(
318 const Polynomial<SymbolicPolyCoeff>& poly,
319 const std::string& variable) {
320
321 if (poly.is_zero() || poly.degree() < 1) return {};
322
323 int deg = poly.degree();
324 auto get_coeff = [&](int d) -> std::shared_ptr<SymbolicExpr> {
325 if (d < 0 || d > deg) return SymbolicExpr::number(0);
326 return poly.coeffs[d].val ? poly.coeffs[d].val : SymbolicExpr::number(0);
327 };
328
329 if (deg == 1) {
330
331 auto a = get_coeff(1);
332 auto b = get_coeff(0);
333 auto neg_b = SymbolicExpr::multiply(b, SymbolicExpr::number(-1));
334 auto root = SymbolicExpr::divide(neg_b, a)->simplify();
335 return { root };
336 }
337
338 auto results = solve_by_factoring(poly, variable);
339 return results;
340}
