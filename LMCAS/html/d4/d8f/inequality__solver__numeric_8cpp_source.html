23int exact_numeric_sign(const std::shared_ptr<SymbolicExpr>& expr) {
24 if (!expr || !LMCAS::detail::node(expr)) {
25 return 0;
26 }
27 auto simplified = expr->simplify();
28 if (!simplified || !LMCAS::detail::node(simplified)) {
29 return 0;
30 }
31 auto num = std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(simplified));
32 if (!num) {
33 return 0;
34 }
35 if (std::holds_alternative<BigInt>(num->value())) {
36 const auto& value = std::get<BigInt>(num->value());
37 if (value.is_zero()) {
38 return 0;
39 }
40 return value.is_negative() ? -1 : 1;
41 }
42 if (std::holds_alternative<Rational>(num->value())) {
43 const auto& value = std::get<Rational>(num->value());
44 if (value.get_numerator().is_zero()) {
45 return 0;
46 }
47 return value.get_numerator().is_negative() ? -1 : 1;
48 }
49 const auto value = std::get<lmmc_real_t>(num->value());
50 if (!std::isfinite(value) || value == 0.0) {
51 return 0;
52 }
53 return value < 0 ? -1 : 1;
54}

56int determine_leading_sign(const Polynomial<SymbolicPolyCoeff>& poly) {
57 if (poly.is_zero()) {
58 return 0;
59 }
60 auto lc = poly.lead_coeff().val;
61 if (!lc) {
62 return 1;
63 }
64 auto simplified = lc->simplify();
65 if (!simplified) {
66 return 1;
67 }
68
69 if (auto val = try_checked_numeric_constant(*simplified)) {
70 if (*val > 0) {
71 return 1;
72 }
73 if (*val < 0) {
74 return -1;
75 }
76 }
77
78 if (auto num = std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(simplified))) {
79 if (std::holds_alternative<BigInt>(num->value())) {
80 return std::get<BigInt>(num->value()).is_negative() ? -1 : 1;
81 }
82 if (std::holds_alternative<Rational>(num->value())) {
83 return std::get<Rational>(num->value()).get_numerator().is_negative() ? -1 : 1;
84 }
85 if (std::holds_alternative<lmmc_real_t>(num->value())) {
86 return std::get<lmmc_real_t>(num->value()) < 0 ? -1 : 1;
87 }
88 }
89 return 1;

92static std::shared_ptr<SymbolicExpr> snap_verified_integer_root(
93 const Polynomial<Rational>& poly,
94 const std::shared_ptr<SymbolicExpr>& root) {
95 if (!root || poly.is_zero()) {
96 return root;
97 }
98 auto numeric = try_checked_numeric_constant(*root);
99 if (!numeric) {
100 return root;
101 }
102
103 const Rational candidate = Rational::from_double(*numeric);
104 if (!candidate.is_integer()) {
105 return root;
106 }
107 if (poly.eval(candidate) == Rational(0)) {
108 return SymbolicExpr::number(candidate);
109 }
110 return root;
111}

131static void append_affine_root(RootMultiplicityList& result,
132 const Polynomial<Rational>& factor, int mult) {
133 Rational a = factor.coeffs[1];
134 Rational b = factor.coeffs[0];

136 Rational root_val = Rational(0) - b / a;

144static void append_quadratic_roots(RootMultiplicityList& result,
145 const Polynomial<Rational>& factor, int mult) {
146 Rational a = factor.coeffs[2];
147 Rational b = factor.coeffs[1];
148 Rational c = factor.coeffs[0];
149
150 Rational disc = b * b - Rational(4) * a * c;
151 if (disc < Rational(0)) {
152 return;
153 }
154 double disc_val = disc.to_double();
155
156 double a_val = a.to_double();
157 double b_val = b.to_double();
158 double sqrt_disc = std::sqrt(disc_val);
159
160 double r1;
161 double r2;
162 if (sqrt_disc == 0.0) {
163 r1 = -b_val / (2.0 * a_val);
164 r2 = r1;
165 } else {
166 const double q =
167 -0.5 * b_val - 0.5 * std::copysign(sqrt_disc, b_val);
168 r1 = q / a_val;
169 r2 = c.to_double() / q;
170 }
171
172 if (std::isfinite(r1)) {
173 auto root = snap_verified_integer_root(factor, SymbolicExpr::number(r1));
174 result.push_back({root, mult});
175 }
176 if (std::isfinite(r2) && r2 != r1) {
177 auto root = snap_verified_integer_root(factor, SymbolicExpr::number(r2));
178 result.push_back({root, mult});
179 }
180}

195std::vector<std::pair<std::shared_ptr<SymbolicExpr>, int>> find_roots_with_multiplicity(
196 const std::shared_ptr<SymbolicExpr>& expr,
197 const std::string& variable) {
198
199 std::vector<std::pair<std::shared_ptr<SymbolicExpr>, int>> result;
200
201 auto converted = symbolic_to_poly<Rational>(expr, variable);
202 if (!converted) throw std::invalid_argument(converted.error().message);
203 const auto& poly_rat = converted.value();
204 if (poly_rat.is_zero() || poly_rat.degree() <= 0) {
205 return result;
206 }
207
208 auto factors = square_free_factorization(poly_rat);
209 if (factors.empty()) {
210 auto symbolic = symbolic_to_poly<SymbolicPolyCoeff>(expr, variable);
211 if (!symbolic) throw std::invalid_argument(symbolic.error().message);
212 const auto& poly_spc = symbolic.value();
213 if (!poly_spc.is_zero() && poly_spc.degree() >= 1) {
214 auto roots = solve_by_factoring(poly_spc, variable);
215 append_verified_roots(result, poly_rat, roots, 1);
216 }
217 return result;
218 }
219 for (const auto& [factor, mult] : factors) {
220 if (factor.degree() <= 0) {
221 continue;
222 }
223 if (factor.degree() == 1) {
224 append_affine_root(result, factor, mult);
225 } else if (factor.degree() == 2) {
226 append_quadratic_roots(result, factor, mult);
227 } else {
228 append_factor_roots(result, factor, mult, variable);
229 }
230 }
231 return result;
232}

247std::vector<std::shared_ptr<SymbolicExpr>> solve_symbolic_poly(
248 const Polynomial<SymbolicPolyCoeff>& poly,
249 const std::string& variable) {
250
251 if (poly.is_zero() || poly.degree() < 1) {
252 return {};
253 }
254
255 int deg = poly.degree();
256 auto get_coeff = [&](int d) -> std::shared_ptr<SymbolicExpr> {
257 if (d < 0 || d > deg) {
258 return SymbolicExpr::number(0);
259 }
260 return poly.coeffs[d].val ? poly.coeffs[d].val : SymbolicExpr::number(0);
261 };
262
263 if (deg == 1) {
264
265 auto a = get_coeff(1);
266 auto b = get_coeff(0);
267 auto neg_b = SymbolicExpr::multiply(b, SymbolicExpr::number(-1));
268 auto root = SymbolicExpr::divide(neg_b, a)->simplify();
269 return { root };
270 }
271
272 auto results = solve_by_factoring(poly, variable);
273 return results;
274}
