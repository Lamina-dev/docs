24int exact_numeric_sign(const std::shared_ptr<SymbolicExpr>& expr) {
25 if (!expr || !lamina::detail::node(expr)) return 0;
26 auto simplified = expr->simplify();
27 if (!simplified || !lamina::detail::node(simplified)) return 0;
28 auto num = std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(simplified));
29 if (!num) return 0;
30 if (std::holds_alternative<BigInt>(num->value())) {
31 const auto& value = std::get<BigInt>(num->value());
32 if (value.is_zero()) return 0;
33 return value.IsNegative() ? -1 : 1;
34 }
35 if (std::holds_alternative<Rational>(num->value())) {
36 const auto& value = std::get<Rational>(num->value());
37 if (value.get_numerator().is_zero()) return 0;
38 return value.get_numerator().IsNegative() ? -1 : 1;
39 }
40 const auto value = std::get<lmmc_real_t>(num->value());
41 if (!std::isfinite(value) || value == 0.0) return 0;
42 return value < 0 ? -1 : 1;
43}

45int determine_leading_sign(const Polynomial<SymbolicPolyCoeff>& poly) {
46 if (poly.is_zero()) return 0;
47 auto lc = poly.lead_coeff().val;
48 if (!lc) return 1;
49 auto simplified = lc->simplify();
50 if (!simplified) return 1;
51
52 if (auto val = try_checked_numeric_constant(*simplified)) {
53 if (*val > 0) return 1;
54 if (*val < 0) return -1;
55 }
56
57 if (auto num = std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(simplified))) {
58 if (std::holds_alternative<BigInt>(num->value())) {
59 return std::get<BigInt>(num->value()).IsNegative() ? -1 : 1;
60 }
61 if (std::holds_alternative<Rational>(num->value())) {
62 return std::get<Rational>(num->value()).get_numerator().IsNegative() ? -1 : 1;
63 }
64 if (std::holds_alternative<lmmc_real_t>(num->value())) {
65 return std::get<lmmc_real_t>(num->value()) < 0 ? -1 : 1;
66 }
67 }
68 return 1;
69}

71static std::shared_ptr<SymbolicExpr> snap_verified_integer_root(
72 const Polynomial<Rational>& poly,
73 const std::shared_ptr<SymbolicExpr>& root) {
74 if (!root || poly.is_zero()) return root;
75 auto numeric = try_checked_numeric_constant(*root);
76 if (!numeric || !std::isfinite(*numeric)) return root;
77
78 double rounded = std::round(*numeric);
79 if (std::abs(*numeric - rounded) > 1e-8) return root;
80 if (rounded < static_cast<double>(std::numeric_limits<long long>::min()) ||
81 rounded > static_cast<double>(std::numeric_limits<long long>::max())) {
82 return root;
83 }
84
85 Rational candidate(BigInt(static_cast<long long>(rounded)));
86 if (poly.eval(candidate) == Rational(0)) {
87 return SymbolicExpr::number(candidate);
88 }
89 return root;
90}

92std::vector<std::pair<std::shared_ptr<SymbolicExpr>, int>> find_roots_with_multiplicity(
93 const std::shared_ptr<SymbolicExpr>& expr,
94 const std::string& variable) {
95
96 std::vector<std::pair<std::shared_ptr<SymbolicExpr>, int>> result;
97
98 auto poly_rat = symbolic_to_poly<Rational>(expr, variable);
99 if (poly_rat.is_zero() || poly_rat.degree() <= 0) {
100 return result;
101 }
102
103 auto factors = square_free_factorization(poly_rat);
104 if (factors.empty()) {
105
106 auto poly_spc = symbolic_to_poly<SymbolicPolyCoeff>(expr, variable);
107 if (!poly_spc.is_zero() && poly_spc.degree() >= 1) {
108 auto roots = solve_by_factoring(poly_spc, variable);
109 for (const auto& root : roots) {
110 if (!root) continue;
111 auto verified_root = snap_verified_integer_root(poly_rat, root);
112 if (try_checked_numeric_constant(*verified_root)) {
113 result.push_back({verified_root, 1});
114 }
115 }
116 }
117 return result;
118 }
119
120 for (const auto& [factor, mult] : factors) {
121 if (factor.degree() <= 0) continue;
122
123 if (factor.degree() == 1) {
124
125 Rational a = factor.coeffs[1];
126 Rational b = factor.coeffs[0];
127 if (a != Rational(0)) {
128 Rational root_val = Rational(0) - b / a;
129 auto root_expr = SymbolicExpr::number(root_val);
130 if (try_checked_numeric_constant(*root_expr)) {
131 result.push_back({root_expr, mult});
132 }
133 }
134 continue;
135 }
136
137 if (factor.degree() == 2) {
138 Rational a = factor.coeffs[2];
139 Rational b = factor.coeffs[1];
140 Rational c = factor.coeffs[0];
141
142 Rational disc = b * b - Rational(4) * a * c;
143 double disc_val = disc.to_double();
144 if (disc_val < -1e-10) continue;
145 if (disc_val < 0) disc_val = 0;
146
147 double a_val = a.to_double();
148 double b_val = b.to_double();
149 double sqrt_disc = std::sqrt(disc_val);
150
151 double r1 = (-b_val + sqrt_disc) / (2.0 * a_val);
152 double r2 = (-b_val - sqrt_disc) / (2.0 * a_val);
153
154 if (std::isfinite(r1)) {
155 auto root = snap_verified_integer_root(factor, SymbolicExpr::number(r1));
156 result.push_back({root, mult});
157 }
158 if (std::isfinite(r2) && std::abs(r1 - r2) > 1e-10) {
159 auto root = snap_verified_integer_root(factor, SymbolicExpr::number(r2));
160 result.push_back({root, mult});
161 }
162 continue;
163 }
164
165 std::vector<SymbolicPolyCoeff> spc_coeffs;
166 for (int i = 0; i <= factor.degree(); ++i) {
167 spc_coeffs.push_back(SymbolicPolyCoeff(SymbolicExpr::number(factor.coeffs[i])));
168 }
169 Polynomial<SymbolicPolyCoeff> factor_spc(spc_coeffs, variable);
170
171 auto factor_roots = solve_by_factoring(factor_spc, variable);
172 for (const auto& root : factor_roots) {
173 if (!root) continue;
174 auto verified_root = snap_verified_integer_root(factor, root);
175 if (try_checked_numeric_constant(*verified_root)) {
176 result.push_back({verified_root, mult});
177 }
178 }
179 }
180
181 return result;
182}

184bool root_less_than(const std::shared_ptr<SymbolicExpr>& a,
185 const std::shared_ptr<SymbolicExpr>& b) {
186 if (!a || !b) return false;
187 auto va = try_checked_numeric_constant(*a);
188 auto vb = try_checked_numeric_constant(*b);
189 if (va && vb) {
190 return *va < *vb;
191 }
192
193 /// 参数根的顺序通过 (a - b) 的符号推导;
194 /// 二次公式的根差可化简为 +/-sqrt(disc)/a.
195 auto diff = SymbolicExpr::add(a, SymbolicExpr::multiply(b, SymbolicExpr::number(-1)))->simplify();
196 if (auto vd = try_checked_numeric_constant(*diff)) {
197 return *vd < 0;
198 }
199
200 /// 尝试判断差值表达式的符号结构:
201 /// 如果差值形如 k * sqrt(...) / denom,判断各因子的符号.
202 /// 这覆盖了二次公式根差 = sqrt(delta)/a 的情形.
203 auto try_sign_of_node = [](const std::shared_ptr<const SymbolicNode>& node) -> int {
204 if (!node) return 0;
205
206 if (auto num_node = std::dynamic_pointer_cast<const NumberNode>(node)) {
207 if (std::holds_alternative<BigInt>(num_node->value())) {
208 auto& v = std::get<BigInt>(num_node->value());
209 if (v.is_zero()) return 0;
210 return v.IsNegative() ? -1 : 1;
211 }
212 if (std::holds_alternative<Rational>(num_node->value())) {
213 auto& v = std::get<Rational>(num_node->value());
214 if (v.get_numerator().is_zero()) return 0;
215 return v.get_numerator().IsNegative() ? -1 : 1;
216 }
217 if (std::holds_alternative<lmmc_real_t>(num_node->value())) {
218 auto v = std::get<lmmc_real_t>(num_node->value());
219 if (v == 0.0) return 0;
220 return v < 0 ? -1 : 1;
221 }
222 }
223
224 /// sqrt(...) 非负(假设参数使判别式非负)
225 if (auto fn = std::dynamic_pointer_cast<const FunctionNode>(node)) {
226 if (fn->type() == FunctionNode::FuncType::Sqrt) return 1;
227 }
228
229 /// x^(1/2) 或 x^0.5 也是平方根,非负
230 if (auto pw = std::dynamic_pointer_cast<const PowerNode>(node)) {
231 auto exp_expr = lamina::detail::make_expression_ptr(pw->exponent());
232 if (auto ev = try_checked_numeric_constant(*exp_expr)) {
233 if (*ev > 0 && *ev < 1.0) {
234 /// base^(正分数) >= 0(假设 base 为判别式等非负量)
235 return 1;
236 }
237 }
238
239 /// 对于整数指数,判断底数符号
240 auto base_expr = lamina::detail::make_expression_ptr(pw->base());
241 auto bv_checked = try_checked_numeric_constant(*base_expr);
242 auto ev_checked = try_checked_numeric_constant(*exp_expr);
243 if (bv_checked && ev_checked) {
244 double bv = *bv_checked;
245 double ev = *ev_checked;
246 int ei = static_cast<int>(ev);
247 if (std::abs(ev - ei) < 1e-10) {
248 if (bv > 0) return 1;
249 if (bv < 0) return (ei % 2 == 0) ? 1 : -1;
250 }
251 }
252 }
253
254 return 0;
255 };
256
257 /// 对乘积节点,各因子符号之积
258 auto try_sign_of_expr = [&try_sign_of_node](const std::shared_ptr<SymbolicExpr>& expr) -> int {
259 if (!expr || !lamina::detail::node(expr)) return 0;
260
261 /// 直接节点
262 int s = try_sign_of_node(lamina::detail::node(expr));
263 if (s != 0) return s;
264
265 /// 乘积:各因子符号之积
266 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(expr))) {
267 int sign = 1;
268 for (const auto& op : mul->operands()) {
269 int os = try_sign_of_node(op);
270 if (os == 0) return 0;
271 sign *= os;
272 }
273 return sign;
274 }
275
276 return 0;
277 };
278
279 int diff_sign = try_sign_of_expr(diff);
280 if (diff_sign < 0) return true;
281 if (diff_sign > 0) return false;
282
283 return false;
284}

302std::vector<std::shared_ptr<SymbolicExpr>> solve_symbolic_poly(
303 const Polynomial<SymbolicPolyCoeff>& poly,
304 const std::string& variable) {
305
306 if (poly.is_zero() || poly.degree() < 1) return {};
307
308 int deg = poly.degree();
309 auto get_coeff = [&](int d) -> std::shared_ptr<SymbolicExpr> {
310 if (d < 0 || d > deg) return SymbolicExpr::number(0);
311 return poly.coeffs[d].val ? poly.coeffs[d].val : SymbolicExpr::number(0);
312 };
313
314 if (deg == 1) {
315
316 auto a = get_coeff(1);
317 auto b = get_coeff(0);
318 auto neg_b = SymbolicExpr::multiply(b, SymbolicExpr::number(-1));
319 auto root = SymbolicExpr::divide(neg_b, a)->simplify();
320 return { root };
321 }
322
323 auto results = solve_by_factoring(poly, variable);
324 return results;
325}
