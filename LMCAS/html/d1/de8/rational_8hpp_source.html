22class Rational {
23private:
24 static constexpr std::size_t max_literal_bytes = 1'048'576;
25 static constexpr std::size_t max_decimal_scale = 1'000'000;
26
27 BigInt numerator;
28 BigInt denominator;
29
30 static BigInt pow10(std::size_t exponent) {
31 if (exponent > max_decimal_scale) {
32 throw std::length_error("Rational decimal scale exceeds safety limit");
33 }
34 if (exponent == 0) return BigInt(1);
35 return BigInt(10).power(static_cast<unsigned long>(exponent));
36 }
37
38 static bool power_leq(BigInt base, unsigned long exponent,
39 const BigInt& limit) {
40 BigInt result(1);
41 while (exponent != 0) {
42 if ((exponent & 1UL) != 0) {
43 result *= base;
44 if (result > limit) return false;
45 }
46 exponent >>= 1UL;
47 if (exponent != 0) {
48 base *= base;
49 if (base > limit) base = limit + BigInt(1);
50 }
51 }
52 return result <= limit;
53 }
54
55 static BigInt integer_nth_root(const BigInt& value, unsigned long degree) {
56 if (degree == 0) throw std::domain_error("Zeroth root is undefined");
57 if (value < BigInt(0)) throw std::domain_error("Root input must be non-negative");
58 if (value <= BigInt(1) || degree == 1) return value;
59
60 BigInt low(0);
61 BigInt high(1);
62 while (power_leq(high, degree, value)) high *= BigInt(2);
63
64 while (high - low > BigInt(1)) {
65 BigInt mid = (low + high) / BigInt(2);
66 if (power_leq(mid, degree, value)) low = std::move(mid);
67 else high = std::move(mid);
68 }
69 return low;
70 }
71
72 void simplify() {
73 if (!denominator) {
74 throw std::runtime_error("Denominator cannot be zero");
75 }
76
77 if (denominator.IsNegative()) {
78 numerator = -numerator;
79 denominator = denominator.Abs();
80 }
81
82 BigInt g = BigInt::gcd(numerator, denominator);
83 if (g != 0 && g != 1) {
84 numerator = numerator / g;
85 denominator = denominator / g;
86 }
87 }

124 /**
125 * @brief 从字符串构造有理数，支持小数、科学计数法、循环小数
126 * @param num 数字字符串
127 */
128 explicit Rational(const std::string& num) : Rational() {
129 if (num.empty()) throw std::invalid_argument("Rational literal is empty");
130 if (num.size() > max_literal_bytes) {
131 throw std::length_error("Rational literal exceeds safety limit");
132 }
133
134 std::size_t pos = 0;
135 bool negative = false;
136 if (num[pos] == '+' || num[pos] == '-') {
137 negative = num[pos] == '-';
138 if (++pos == num.size()) {
139 throw std::invalid_argument("Rational literal has no digits");
140 }
141 }
142
143 auto read_digits = [&](std::string& output) {
144 while (pos < num.size() &&
145 std::isdigit(static_cast<unsigned char>(num[pos]))) {
146 output.push_back(num[pos++]);
147 }
148 };
149
150 std::string integer_digits;
151 std::string fractional_digits;
152 std::string repeating_digits;
153 read_digits(integer_digits);
154
155 bool has_decimal_point = false;
156 if (pos < num.size() && num[pos] == '.') {
157 has_decimal_point = true;
158 ++pos;
159 read_digits(fractional_digits);
160 }
161
162 if (pos < num.size() && num[pos] == '(') {
163 if (!has_decimal_point) {
164 throw std::invalid_argument("Repeating decimal section requires a decimal point");
165 }
166 ++pos;
167 read_digits(repeating_digits);
168 if (repeating_digits.empty() || pos >= num.size() || num[pos] != ')') {
169 throw std::invalid_argument("Invalid repeating decimal section");
170 }
171 ++pos;
172 }
173
174 if (integer_digits.empty() && fractional_digits.empty()) {
175 throw std::invalid_argument("Rational literal has no digits");
176 }
177
178 long long exponent = 0;
179 if (pos < num.size() && (num[pos] == 'e' || num[pos] == 'E')) {
180 ++pos;
181 bool exponent_negative = false;
182 if (pos < num.size() && (num[pos] == '+' || num[pos] == '-')) {
183 exponent_negative = num[pos] == '-';
184 ++pos;
185 }
186 if (pos == num.size() ||
187 !std::isdigit(static_cast<unsigned char>(num[pos]))) {
188 throw std::invalid_argument("Rational exponent has no digits");
189 }
190 while (pos < num.size() &&
191 std::isdigit(static_cast<unsigned char>(num[pos]))) {
192 const int digit = num[pos++] - '0';
193 if (exponent > static_cast<long long>(max_decimal_scale / 10) ||
194 exponent * 10 + digit > static_cast<long long>(max_decimal_scale)) {
195 throw std::length_error("Rational exponent exceeds safety limit");
196 }
197 exponent = exponent * 10 + digit;
198 }
199 if (exponent_negative) exponent = -exponent;
200 }
201
202 if (pos != num.size()) {
203 throw std::invalid_argument("Invalid character in Rational literal");
204 }
205
206 if (integer_digits.empty()) integer_digits = "0";
207 const std::string non_repeating = integer_digits + fractional_digits;
208 numerator = BigInt(non_repeating);
209 denominator = pow10(fractional_digits.size());
210
211 if (!repeating_digits.empty()) {
212 const BigInt repeat_scale = pow10(repeating_digits.size());
213 const BigInt repeat_factor = repeat_scale - BigInt(1);
214 numerator = numerator * repeat_factor + BigInt(repeating_digits);
215 denominator *= repeat_factor;
216 }
217
218 if (exponent > 0) {
219 numerator *= pow10(static_cast<std::size_t>(exponent));
220 } else if (exponent < 0) {
221 denominator *= pow10(static_cast<std::size_t>(-exponent));
222 }
223 if (negative) numerator = -numerator;
224 simplify();
225 }

229 /**
230 * @brief 从 double 构造有理数
231 * @param value 浮点数值
232 * @return 对应的精确有理数
233 */
234 static Rational from_double(double value) {
235 if (!std::isfinite(value)) {
236 throw std::invalid_argument("Cannot construct Rational from NaN or infinity");
237 }
238 if (value == 0.0) {
239 return Rational();
240 }
241 if (std::floor(value) == value) {
242 // value 是整数级浮点数；std::to_string(double) 会输出 "123.000000"，
243 // 直接喂给 BigInt 会因含小数点被拒绝。先把整数部分提取成字符串。
244 std::ostringstream iss;
245 iss << std::fixed << std::setprecision(0) << value;
246 return Rational(BigInt(iss.str()));
247 }
248 std::ostringstream oss;
249 oss << std::scientific << std::setprecision(15) << value;
250 std::string str = oss.str();
251
252 size_t e_pos = str.find('e');
253 std::string base = str.substr(0, e_pos);
254 bool negative = false;
255 if (base[0] == '-') {
256 negative = true;
257 base.erase(0, 1);
258 }
259 base.erase(1, 1);
260 while (base.back() == '0') {
261 base.pop_back();
262 }
263 int exponent = std::stoi(str.substr(e_pos + 1));
264
265 size_t decimal_places = std::max(0, static_cast<int>(base.length()) - exponent - 1);
266
267 BigInt num(base);
268 if (negative) num = -num;
269 BigInt den("1" + std::string(decimal_places, '0'));
270
271 Rational r(num, den);
272 r.simplify();
273 return r;
274 }

314 /**
315 * @brief 转换为十进制小数字符串（自动检测循环节）
316 * @return 小数字符串，循环部分用括号标记
317 */
318 inline std::string to_float_string() {
319 if (numerator % denominator == BigInt(0)) return (numerator / denominator).ToString();
320 std::string re;
321 const BigInt ten(10);
322 const BigInt zero(0);
323 if ((numerator < zero) ^ (denominator < zero)) re += '-';
324 BigInt num = numerator.Abs(), den = denominator.Abs();
325 re += (num / den).ToString();
326 num %= den;
327 re += '.';
328
329 std::map<BigInt, size_t> num_index;
330
331 while (num != zero && num_index.find(num) == num_index.end()) {
332 num_index[num] = re.size();
333 re += '0';
334 num *= ten;
335 while (num >= den) {
336 num -= den;
337 re.back()++;
338 }
339 }
340 if (num_index.find(num) != num_index.end()) {
341 re.insert(re.begin() + num_index[num], '(');
342 re += ")...";
343 }
344 return re;
345
346 }

348 /**
349 * @brief 转换为指定精度的十进制小数字符串
350 * @param n 小数位数（正数为小数点后位数，负数为整数部分截断位数）
351 * @return 定精度小数字符串
352 */
353 inline std::string to_float_string(int64_t n) const{
354 if (is_zero()) return "0";
355 if (n == 0) return (numerator / denominator).ToString();
356 if (n > 0) {
357 BigInt pow10n(10);
358 pow10n = pow10n.power(BigInt(static_cast<long long>(n)));
359 BigInt scaled = numerator * pow10n / denominator;
360 bool negative = scaled < BigInt(0);
361 if (negative) scaled = scaled * BigInt(-1);
362 std::string re = scaled.ToString();
363 if ((int64_t)re.size() <= n) re.insert(re.begin(), n - re.size() + 1, '0');
364 re.insert(re.end() - n,'.');
365 while (re.back() == '0') re.pop_back();
366 if (re.back() == '.') re.pop_back();
367 if (negative) re.insert(re.begin(), '-');
368 return re;
369 } else {
370 BigInt pow10n(10);
371 pow10n = pow10n.power(BigInt(static_cast<long long>(n)).Abs());
372 std::string re = (numerator / (denominator * pow10n)).ToString();
373 if (re.size() == 1 && re[0] == '0') return re;
374 for (; n < 0; n++) re.push_back('0');
375 return re;
376 }
377 }

390 inline void floor_without_sim(const int64_t& n) {
391 if (n == 0) return;
392 if (n > 0) {
393 BigInt pow10n(10);
394 pow10n = pow10n.power(BigInt(static_cast<long long>(n)));

397 denominator = pow10n;
398 } else {

401 denominator *= pow10n;
402 numerator /= denominator;
403 numerator *= pow10n;
404 denominator = BigInt(1);
405 }

410 /**
411 * @brief 牛顿迭代法计算平方根（就地修改）
412 * @param n 精度位数
413 */
414 inline void sqrt_self(int64_t n) {
415 if (n < 0) throw std::invalid_argument("Square-root precision must be non-negative");
416 if (numerator < BigInt(0)) throw std::domain_error("Square root of a negative Rational");
417 if (numerator == BigInt(0)) {
418 denominator = BigInt(1);
419 return;
420 }
421 if (static_cast<std::size_t>(n) > max_decimal_scale / 2) {
422 throw std::length_error("Square-root precision exceeds safety limit");
423 }
424
425 BigInt numerator_root = numerator.sqrt();
426 BigInt denominator_root = denominator.sqrt();
427 if (numerator_root * numerator_root == numerator &&
428 denominator_root * denominator_root == denominator) {
429 numerator = std::move(numerator_root);
430 denominator = std::move(denominator_root);
431 simplify();
432 return;
433 }
434
435 const BigInt scale = pow10(static_cast<std::size_t>(n));
436 const BigInt scaled = numerator * scale * scale / denominator;
437 numerator = scaled.sqrt();
438 denominator = scale;
439 simplify();
440 }

453 /**
454 * @brief 牛顿迭代法计算 n 次方根（就地修改）
455 * @param radical 根次数
456 * @param n 精度位数
457 */
458 inline void radicand_self(const BigInt& radical, int64_t n) {
459 if (radical <= BigInt(0)) throw std::domain_error("Root degree must be positive");
460 if (radical > BigInt(1024)) throw std::length_error("Root degree exceeds safety limit");
461 if (n < 0) throw std::invalid_argument("Root precision must be non-negative");
462
463 const unsigned long degree = static_cast<unsigned long>(radical.to_int());
464 const bool negative = numerator < BigInt(0);
465 if (negative && (degree % 2UL) == 0) {
466 throw std::domain_error("Even root of a negative Rational");
467 }
468 if (static_cast<std::size_t>(n) > max_decimal_scale / degree) {
469 throw std::length_error("Root precision exceeds safety limit");
470 }
471
472 const BigInt scale = pow10(static_cast<std::size_t>(n));
473 const BigInt scaled = numerator.Abs() * scale.power(degree) / denominator;
474 numerator = integer_nth_root(scaled, degree);
475 if (negative) numerator = -numerator;
476 denominator = scale;
477 simplify();
478 }

504 /**
505 * @brief 转换为浮点数。
506 *
507 * 精确值可按 binary64 round-to-nearest 转为有限值时返回有限结果，
508 * 包括正负最大有限端点及其半 ULP 内邻域；
509 * 超出有限舍入范围或非零值下溢为零时抛出范围异常。
510 * @return 对应的 double 值
511 */
512 lmmc_real_t to_double() const {
513 if (is_zero()) return 0.0;
514 const std::string num_text = numerator.Abs().ToString();
515 const std::string den_text = denominator.ToString();
516 const std::size_t num_digits = std::min<std::size_t>(18, num_text.size());
517 const std::size_t den_digits = std::min<std::size_t>(18, den_text.size());
518 const long double num_head = std::stold(num_text.substr(0, num_digits));
519 const long double den_head = std::stold(den_text.substr(0, den_digits));
520 const long long decimal_shift =
521 static_cast<long long>(num_text.size()) - static_cast<long long>(num_digits) -
522 static_cast<long long>(den_text.size()) + static_cast<long long>(den_digits);
523 long double value = (num_head / den_head) *
524 std::pow(10.0L, static_cast<long double>(decimal_shift));
525 if (numerator.IsNegative()) value = -value;
526 const lmmc_real_t converted = static_cast<lmmc_real_t>(value);
527 if (!std::isfinite(converted)) {
528 /*
529 * A decimal-head approximation can round slightly above the
530 * binary64 endpoint even when the exact rational still rounds to
531 * DBL_MAX. Resolve that boundary case with exact integer
532 * arithmetic. Values below max + 0.5 ULP round to a finite value.
533 */
534 const BigInt max_finite_integer =
535 (BigInt(1) << 1024) - (BigInt(1) << 971);
536 const BigInt finite_rounding_limit =
537 max_finite_integer + (BigInt(1) << 970);
538 if (numerator.Abs() <
539 denominator * finite_rounding_limit) {
540 return std::copysign(
541 std::numeric_limits<lmmc_real_t>::max(),
542 numerator.IsNegative() ? -1.0 : 1.0);
543 }
544 throw std::overflow_error(
545 "Rational cannot be represented as a finite double");
546 }
547 if (converted == 0.0) {
548 throw std::underflow_error("Rational underflow during double conversion");
549 }
550 return converted;
551 }

589 /**
590 * @brief 有理数整数幂
591 * @param exponent 指数（可为负）
592 * @return this^exponent
593 * @throw std::runtime_error 零的负幂或 0^0 时抛出
594 */
595 Rational power(const BigInt& exponent) const {
596 if (exponent < BigInt(0)) {
597
598 if (is_zero()) {
599 throw std::runtime_error("Cannot raise zero to negative power");
600 }
601 BigInt pos_exp = -exponent;
602 return Rational(denominator.power(pos_exp), numerator.power(pos_exp));
603 }
604
605 if (!exponent) {
606 if (is_zero()) {
607 throw std::runtime_error("0^0 is undefined");
608 }
609 return Rational(1);
610 }
611
612 return Rational(numerator.power(exponent), denominator.power(exponent));
613 }
