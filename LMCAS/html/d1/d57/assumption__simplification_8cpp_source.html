13ExpressionResult simplify_expression(const ExprPtr& expression,
14 ComputationContext& context) {
15 return checked_transform_expr(
16 expression, context, "LMCAS.simplify", "simplify",
17 [&context](const SymbolicExpr& value) {
18 RewriteBudget budget(context, context.limits().max_recursion_depth,
19 context.limits().max_ast_nodes, "LMCAS.simplify");
20 budget.measure(detail::node(value));
21 std::optional<AssumptionFacts> facts;
22 if (context.assumptions()) { facts.emplace(*context.assumptions()); }
23 NormalizationVisitor visitor(context, facts ? *facts : no_facts(), Domain::Real, &budget);

26 budget.measure(result);
