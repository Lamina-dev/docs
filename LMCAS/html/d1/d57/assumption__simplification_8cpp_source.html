67ExpressionResult substitute_expression(
68 const ExprPtr& expression,
69 const std::vector<Binding>& replacements,
70 ComputationContext& context) {
71 return assumed_transform(expression, context, "LMCAS.substitute", "substitute",
72 [&](SymbolicNodePtr value, RewriteBudget& budget, const FactsQuery& facts) {
73 budget.measure(value);
74 for (const auto& replacement : replacements) {
75 const auto symbol = std::dynamic_pointer_cast<const VariableNode>(
76 node(replacement.symbol));
77 if (!symbol || symbol->is_constant() || !replacement.value ||
78 !node(replacement.value)) {
79 throw CasError{CasErrc::InvalidArgument, "invalid substitution binding",
80 "LMCAS.substitute"};
81 }
82 value = substitute_node(value, symbol->name(), node(replacement.value),
83 budget, facts);
84 }
85 return value;
86 });

89void AssumptionFacts::collect_names(const AssumptionContext& assumptions,
90 std::set<std::string>& names, RewriteBudget& budget) {
91 const auto include = [&](const SymbolicExpr& expression) {
92 budget.measure(node(expression));
93 auto variables = free_variables(node(expression), &budget);
94 names.merge(variables);
95 };
96 for (const auto& scope : assumptions.scope_stack_) {
97 auto access = budget.context().consume_steps(1, "assumption.names");
98 if (!access) { throw access.error(); }
99 for (auto& symbol : scope.properties.get_all_symbols()) {
100 access = budget.context().consume_steps(1, "assumption.names");
101 if (!access) { throw access.error(); }
102 for (const auto& [variable, period] : scope.properties.get_period_decls(symbol)) {
103 names.insert(variable);
104 include(period);
105 }
106 for (const auto& declaration : scope.properties.get_monotonicity_decls(symbol)) {
107 access = budget.context().consume_steps(1, "assumption.names");
108 if (!access) { throw access.error(); }
109 names.insert(declaration.variable);
110 }
111 names.insert(std::move(symbol));
112 }
113 for (const auto& relation : scope.relations.get_relations()) {
114 include(relation.lhs);
115 include(relation.rhs);
116 }
117 for (const auto& conditional : scope.conditionals) {
118 include(conditional.condition);
119 include(conditional.conclusion);
120 }
121 }
