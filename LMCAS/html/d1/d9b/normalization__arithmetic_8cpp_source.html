6bool sum_term_less(const std::shared_ptr<const SymbolicNode>& left,
7 const std::shared_ptr<const SymbolicNode>& right) {
8 const BigInt a = get_node_degree_helper(left);
9 const BigInt b = get_node_degree_helper(right);
10 if (a != b) {
11 return a > b;
12 }
13 const bool left_number = std::dynamic_pointer_cast<const NumberNode>(left) != nullptr;
14 const bool right_number = std::dynamic_pointer_cast<const NumberNode>(right) != nullptr;
15 if (left_number != right_number) {
16 return right_number;
17 }
18 return left->compare(*right) < 0;
19}

107std::shared_ptr<const NumberNode> NormalizationVisitor::extract_term_coefficient(
108 std::shared_ptr<const SymbolicNode>& term) {
109 auto coefficient = detail::make_node<NumberNode>(BigInt(1));
110 const auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(term);
111 if (!multiply || multiply->operands().empty()) {
112 return coefficient;
113 }
114 const auto& operands = multiply->operands();
115 auto number = std::dynamic_pointer_cast<const NumberNode>(operands.front());
116 std::size_t index = 0;
117 if (!number) {
118 number = std::dynamic_pointer_cast<const NumberNode>(operands.back());
119 index = operands.size() - 1;
120 }
121 if (!number) {
122 return coefficient;
123 }
124 coefficient = number;
125 if (operands.size() == 2) {
126 term = operands[index == 0 ? 1 : 0];
127 return coefficient;
128 }
129 std::vector<std::shared_ptr<const SymbolicNode>> rest;
130 std::size_t nodes = 0;
131 normalization_check_count(rewrite_budget(), operands.size() - 1, 1, false);
132 rest.reserve(operands.size() - 1);
133 for (std::size_t k = 0; k < operands.size(); ++k) {
134 if (k != index) {
135 normalization_append(rewrite_budget(), nodes, rest, operands[k]);
136 }
137 }
138 auto rest_product = make_normalized_multiply_node(rest, rewrite_budget());
139 NormalizationVisitor visitor(context_, facts_, domain_, rewrite_budget());
140 rest_product->accept(visitor);
141 term = visitor.get_result();
142 return coefficient;
143}

145std::shared_ptr<const SymbolicNode> NormalizationVisitor::collect_sum(
146 const std::vector<std::shared_ptr<const SymbolicNode>>& operands) {
147 auto constant = detail::make_node<NumberNode>(BigInt(0));
148 struct TermAccum {
149 std::shared_ptr<const NumberNode> coefficient;
150 std::vector<std::shared_ptr<const SymbolicNode>> originals;
151 std::size_t nodes;
152 };
153 std::map<std::shared_ptr<const SymbolicNode>, TermAccum, NodeCompare> terms;
154 for (const auto& operand : operands) {
155 if (const auto number = std::dynamic_pointer_cast<const NumberNode>(operand)) {
156 constant = add_numbers(constant, number);
157 continue;
158 }
159 auto term = operand;
160 auto coefficient = extract_term_coefficient(term);
161 auto found = terms.find(term);
162 if (found == terms.end()) {
163 const auto nodes = rewrite_budget() ? rewrite_budget()->measure(operand) : 0;
164 terms.emplace(term, TermAccum{coefficient, {operand}, nodes});
165 } else {
166 found->second.coefficient = add_numbers(found->second.coefficient, coefficient);
167 normalization_append(rewrite_budget(), found->second.nodes, found->second.originals, operand);
168 }
169 }
170 std::vector<std::shared_ptr<const SymbolicNode>> collected;
171 std::size_t nodes = 0;
172 if (!constant->is_zero()) {
173 normalization_append(rewrite_budget(), nodes, collected, constant);
174 }
175 for (const auto& [term, accumulated] : terms) {
176 if (!accumulated.coefficient->is_zero()) {
177 normalization_append(rewrite_budget(), nodes, collected,
178 weighted_term(term, accumulated.coefficient, rewrite_budget()));
179 } else if (!normalization_can_discard(term, facts_, domain_, context_)) {
180 for (const auto& original : accumulated.originals) {
181 normalization_append(rewrite_budget(), nodes, collected, original);
182 }
183 }
184 }
185 if (collected.empty()) {
186 return detail::make_node<NumberNode>(BigInt(0));
187 }
188 if (collected.size() == 1) {
189 return collected.front();
190 }
191 normalization_check_arithmetic<AddNode>(rewrite_budget(), collected, nodes);
192 std::sort(collected.begin(), collected.end(), sum_term_less);
193 return detail::make_node<AddNode>(collected);
194}
