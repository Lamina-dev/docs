254 return checked_inference_result<Tribool>(expr, "query_real_checked", context,
255 [&]() -> InferenceTriboolResult {
256 DepthGuard guard(*this, *LMCAS::detail::node(expr));
257 if (guard.should_abort()) {
258 return Tribool::Unknown;
259 }
260 if (auto num = std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(expr))) {
261 return Impl::number_real(*num);

264 if (auto var = std::dynamic_pointer_cast<const VariableNode>(LMCAS::detail::node(expr))) {
265 if (var->is_constant()) return Tribool::True;
266 const auto& props = impl_->ctx;

268 return Tribool::True;
269 }
270 for (Sign sign : {Sign::Positive, Sign::Negative, Sign::Zero,
271 Sign::NonNegative, Sign::NonPositive}) {
272 auto step = context.consume_steps(1, "query_real_checked");
273 if (!step) return InferenceTriboolResult::failure(step.error());
274 if (props.has_sign(var->name(), sign)) { return Tribool::True; }
275 }
276 return Tribool::Unknown;
277 }
278
279 return impl_->composite_real(*this, expr, context);

478/**
479 * @brief 判断 NumberNode 是否为整数。
480 * BigInt 总是整数；Rational 要求分母为 1；double 要求有限且等于其向下取整值。
481 */
482bool is_integer_number(const NumberNode& num) {
483 if (std::holds_alternative<BigInt>(num.value())) {
484 return true;
485 }
486 if (std::holds_alternative<Rational>(num.value())) {
487 return std::get<Rational>(num.value()).is_integer();
488 }
489 double v = std::get<lmmc_real_t>(num.value());
490 return std::isfinite(v) && v == std::floor(v);
491}
492bool is_even_integer_number(const NumberNode& num) {
493 if (!is_integer_number(num)) {
494 return false;
495 }
496
497 if (std::holds_alternative<BigInt>(num.value())) {
498 return std::get<BigInt>(num.value()).is_even();
499 }
500 if (std::holds_alternative<Rational>(num.value())) {
501 BigInt n = std::get<Rational>(num.value()).get_numerator();
502 return n.is_even();
503 }
504 const double v = std::get<lmmc_real_t>(num.value());
505 return std::fmod(v, 2.0) == 0.0;
506}
507bool is_positive_integer_number(const NumberNode& num) {
508 if (!is_integer_number(num)) {
509 return false;
510 }
511
512 if (std::holds_alternative<BigInt>(num.value())) {
513 const auto& b = std::get<BigInt>(num.value());
514 return !b.is_negative() && !(b == BigInt(0));
515 }
516 if (std::holds_alternative<Rational>(num.value())) {
517 BigInt n = std::get<Rational>(num.value()).get_numerator();
518 return !n.is_negative() && !(n == BigInt(0));
519 }
520 double v = std::get<lmmc_real_t>(num.value());
521 return v > 0.0;
522}
