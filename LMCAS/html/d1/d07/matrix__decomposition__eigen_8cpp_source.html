12Result<void> verify_jordan_identity(const ExprPtr& A,
13 const JordanDecomposition& value, const MatrixNode& matrix,
14 ComputationContext& context, const std::string& operation) {
15 auto ap = matrix_multiply_checked(A, value.P, context);
16 if (!ap) { return Result<void>::failure(ap.error()); }
17 auto pj = matrix_multiply_checked(value.P, value.J, context);
18 if (!pj) { return Result<void>::failure(pj.error()); }
19 auto left = std::dynamic_pointer_cast<const MatrixNode>(detail::node(ap.value()));
20 auto right = std::dynamic_pointer_cast<const MatrixNode>(detail::node(pj.value()));
21 if (!left || !right) {
22 return Result<void>::failure(CasErrc::InternalInvariant,
23 "Jordan matrix product did not produce a matrix", operation);
24 }
25 for (std::size_t row = 0; row < matrix.rows(); ++row) {
26 for (std::size_t col = 0; col < matrix.cols(); ++col) {
27 auto difference = SymbolicExpr::add(detail::make_expression_ptr(left->get(row, col)),
28 SymbolicExpr::multiply(SymbolicExpr::number(-1),
29 detail::make_expression_ptr(right->get(row, col))));
30 auto proof = detail::classify_exact_zero(difference, context, operation);
31 if (!proof) { return Result<void>::failure(proof.error()); }
32 if (proof.value() != detail::ZeroProof::Zero) {
33 return Result<void>::failure(proof.value() == detail::ZeroProof::Unknown
34 ? CasErrc::Inconclusive : CasErrc::InternalInvariant,
35 "Jordan intertwining identity is not proved at (" +
36 std::to_string(row) + "," + std::to_string(col) +
37 "): " + difference->to_string(), operation);
38 }
39 }
40 }
41 return Result<void>::success();
42}

50Result<void> verify_jordan_decomposition(const ExprPtr& A,
51 const JordanDecomposition& value, ComputationContext& context) {
52 const std::string operation = "jordan_form.verify";
53 auto a = std::dynamic_pointer_cast<const MatrixNode>(detail::node(A));
54 auto p = std::dynamic_pointer_cast<const MatrixNode>(detail::node(value.P));
55 auto j = std::dynamic_pointer_cast<const MatrixNode>(detail::node(value.J));
56 if (!a || !p || !j || !jordan_dimensions_match(*a, *p, *j)) {
57 return Result<void>::failure(CasErrc::InternalInvariant,
58 "Jordan certificate has invalid dimensions", operation);
59 }
60 auto rank = matrix_rank_checked(value.P, context);
61 if (!rank) { return Result<void>::failure(rank.error()); }
62 if (rank.value() != a->rows()) {
63 return Result<void>::failure(CasErrc::InternalInvariant,
64 "Jordan change of basis is singular", operation);
65 }
66 return verify_jordan_identity(A, value, *a, context, operation);
67}

76JordanDecompositionResult diagonalization_checked(const ExprPtr& A,
77 ComputationContext& context) {
78 auto matrix = validate_decomposition_matrix(A, true, context, "matrix.diagonalize");
79 if (!matrix) { return JordanDecompositionResult::failure(matrix.error()); }
80 if (auto diagonal = exact_diagonal_entries(matrix.value())) {
81 return certified_jordan(A, {rectangular_diagonal_expr(matrix.value()->rows(),
82 matrix.value()->cols(), *diagonal), identity_matrix_expr(matrix.value()->rows())}, context);
83 }
84 auto spaces = detail::matrix_eigenspaces_checked(A, context);
85 if (!spaces) { return JordanDecompositionResult::failure(spaces.error()); }
86 const auto n = matrix.value()->rows();
87 std::size_t count = 0;
88 for (const auto& space : spaces.value()) { count += space.second.size(); }
89 if (count != n) {
90 return JordanDecompositionResult::failure(CasErrc::Inconclusive,
91 "ordinary eigenspaces do not span the matrix", "matrix.diagonalize");
92 }
93 std::vector<std::vector<ExprPtr>> p(n, std::vector<ExprPtr>(n));
94 std::vector<std::vector<ExprPtr>> j(n,
95 std::vector<ExprPtr>(n, SymbolicExpr::number(0)));
96 std::size_t column = 0;
97 for (const auto& space : spaces.value()) {
98 for (const auto& vector : space.second) {
99 if (vector.size() != n) {
100 return JordanDecompositionResult::failure(CasErrc::InternalInvariant,
101 "eigenspace vector has invalid dimensions", "matrix.diagonalize");
102 }
103 for (std::size_t row = 0; row < n; ++row) { p[row][column] = vector[row]; }
104 j[column][column] = space.first;
105 ++column;
106 }
107 }
108 return certified_jordan(A,
109 {SymbolicExpr::matrix(j), SymbolicExpr::matrix(p)}, context);
110}

112struct QuadraticSigns {
113 bool has_pos = false;
114 bool has_neg = false;
115 bool any_zero = false;
116 void include(int proved_sign) {
117 if (proved_sign > 0) has_pos = true;
118 else if (proved_sign < 0) has_neg = true;
119 else any_zero = true;

121 std::string classification() const {
122 if (has_pos && has_neg) { return "indefinite"; }
123 if (has_pos) { return any_zero ? "positive_semidefinite" : "positive_definite"; }
124 if (has_neg) { return any_zero ? "negative_semidefinite" : "negative_definite"; }
125 return any_zero ? "positive_semidefinite" : "unknown";

129std::optional<int> symbolic_quadratic_sign(const ExprPtr& value,
130 const detail::AssumptionFacts& facts, ComputationContext& context) {
131 auto real = detail::query_real_value(detail::node(value), facts, context);
132 if (!real || real.value() != Tribool::True) { return std::nullopt; }
133 auto zero = detail::classify_exact_zero(value, context, "matrix.quadratic_sign");
134 if (!zero) { return std::nullopt; }
135 if (zero.value() == detail::ZeroProof::Zero) { return 0; }
136 auto positive = detail::query_positive_value(detail::node(value), facts, context);
137 if (!positive) { return std::nullopt; }
138 if (positive.value() == Tribool::True) { return 1; }
139 auto negative_value = SymbolicExpr::multiply(SymbolicExpr::number(-1), value);
140 auto negative = detail::query_positive_value(detail::node(negative_value), facts, context);
141 if (!negative || negative.value() != Tribool::True) { return std::nullopt; }
142 return -1;
143}

145std::optional<int> quadratic_eigenvalue_sign(const ExprPtr& eigenvalue,
146 const detail::AssumptionFacts& facts, ComputationContext& context) {
147 auto value = eigenvalue->simplify();
148 if (auto rational = exact_rational_expr(value)) {
149 return *rational > Rational(0) ? 1 : (*rational < Rational(0) ? -1 : 0);
150 }
151 if (const auto* number = dynamic_cast<const NumberNode*>(detail::node(value).get())) {
152 if (const auto* approximate = std::get_if<lmmc_real_t>(&number->value())) {
153 if (!std::isfinite(*approximate)) { return std::nullopt; }
154 return *approximate > 0 ? 1 : (*approximate < 0 ? -1 : 0);
155 }
156 }
157 return symbolic_quadratic_sign(value, facts, context);
158}

168JordanDecompositionResult jordan_form_checked(const ExprPtr& A,
169 ComputationContext& context) {
170 const std::string operation = "jordan_form";
171 auto matrix = validate_decomposition_matrix(A, true, context, operation);
172 if (!matrix) { return JordanDecompositionResult::failure(matrix.error()); }
173 auto budget = context.consume_steps(matrix.value()->rows() * matrix.value()->cols() * 48 + 32,
174 operation);
175 if (!budget) { return JordanDecompositionResult::failure(budget.error()); }
176 try {
177 if (auto diagonal = exact_diagonal_entries(matrix.value())) {
178 return certified_jordan(A, {rectangular_diagonal_expr(
179 matrix.value()->rows(), matrix.value()->cols(), *diagonal),
180 identity_matrix_expr(matrix.value()->rows())}, context);
181 }
182 auto exact = exact_rational_matrix(matrix.value());
183 if (exact && is_defective_upper_pair(*exact)) {
184 auto zero = SymbolicExpr::number(0);
185 auto one = SymbolicExpr::number(1);
186 auto lambda = exact_number_expr((*exact)[0][0]);
187 auto j = SymbolicExpr::matrix({{lambda, one}, {zero, lambda}});
188 auto p = SymbolicExpr::matrix({{exact_number_expr((*exact)[0][1]), zero}, {zero, one}});
189 return certified_jordan(A, {std::move(j), std::move(p)}, context);
190 }
191 return diagonalization_checked(A, context);
192 } catch (const std::bad_alloc&) {
193 return JordanDecompositionResult::failure(CasErrc::ResourceLimit,
194 "allocation failed while calculating Jordan form", operation);
195 } catch (const std::exception& ex) {
196 return JordanDecompositionResult::failure(CasErrc::InternalInvariant, ex.what(), operation);
197 }
198}

205ExprPtr matrix_exp(const ExprPtr& A) {
206 ComputationContext context;
207 auto decomposition = diagonalization_checked(A, context);
208 if (!decomposition) return SymbolicExpr::exp(A);
209 auto inverse = matrix_inverse_checked(decomposition.value().P, context);
210 if (!inverse) return SymbolicExpr::exp(A);
211 auto diagonal = std::dynamic_pointer_cast<const MatrixNode>(detail::node(decomposition.value().J));
212 const auto n = diagonal->rows();
213 std::vector<std::vector<ExprPtr>> values(n, std::vector<ExprPtr>(n, SymbolicExpr::number(0)));
214 for (std::size_t i = 0; i < n; ++i)
215 values[i][i] = SymbolicExpr::exp(detail::make_expression_ptr(diagonal->get(i, i)));
216 return SymbolicExpr::multiply(decomposition.value().P,
217 SymbolicExpr::multiply(SymbolicExpr::matrix(values), inverse.value()))->simplify();
218}

220ExprPtr matrix_log(const ExprPtr& A) {
221 ComputationContext context;
222 auto decomposition = diagonalization_checked(A, context);
223 if (!decomposition) return nullptr;
224 auto inverse = matrix_inverse_checked(decomposition.value().P, context);
225 if (!inverse) return nullptr;
226 auto diagonal = std::dynamic_pointer_cast<const MatrixNode>(detail::node(decomposition.value().J));
227 const auto n = diagonal->rows();
228 AssumptionContext assumptions;
229 detail::AssumptionFacts facts(assumptions);
230 std::vector<std::vector<ExprPtr>> values(n, std::vector<ExprPtr>(n, SymbolicExpr::number(0)));
231 for (std::size_t i = 0; i < n; ++i) {
232 auto positive = detail::query_positive_value(diagonal->get(i, i), facts, context);
233 if (!positive || positive.value() != Tribool::True) return nullptr;
234 values[i][i] = SymbolicExpr::ln(detail::make_expression_ptr(diagonal->get(i, i)));
235 }
236 return SymbolicExpr::multiply(decomposition.value().P,
237 SymbolicExpr::multiply(SymbolicExpr::matrix(values), inverse.value()))->simplify();
238}

240std::string classify_quadratic_form(const ExprPtr& A) {
241 ComputationContext context;
242 auto eigenvalues = matrix_eigenvalues_checked(A, context);
243 if (!eigenvalues || eigenvalues.value().empty()) { return "unknown"; }
244 AssumptionContext assumptions;
245 detail::AssumptionFacts facts(assumptions);
246 QuadraticSigns signs;
247 for (const auto& eigenvalue : eigenvalues.value()) {
248 auto sign = quadratic_eigenvalue_sign(eigenvalue, facts, context);
249 if (!sign) { return "unknown"; }
250 signs.include(*sign);
251 }
252 return signs.classification();
253}
