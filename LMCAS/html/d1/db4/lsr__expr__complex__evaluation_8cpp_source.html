62Result<ApproxComplex> add_complex(const ApproxComplex& lhs,
63 const ApproxComplex& rhs) {
64 auto result = checked_complex(lhs.real.value + rhs.real.value,
65 lhs.imag.value + rhs.imag.value,
66 kEvalComplexOperation);
67 if (!result) return result;
68 result.value().real.absolute_error +=
69 lhs.real.absolute_error + rhs.real.absolute_error;
70 result.value().imag.absolute_error +=
71 lhs.imag.absolute_error + rhs.imag.absolute_error;
72 return result;
73}

75Result<ApproxComplex> multiply_complex(const ApproxComplex& lhs,
76 const ApproxComplex& rhs) {
77 const double a = lhs.real.value;
78 const double b = lhs.imag.value;
79 const double c = rhs.real.value;
80 const double d = rhs.imag.value;
81 auto result = checked_complex(a * c - b * d, a * d + b * c,
82 kEvalComplexOperation);
83 if (!result) return result;
84 result.value().real.absolute_error +=
85 std::abs(c) * lhs.real.absolute_error +
86 std::abs(a) * rhs.real.absolute_error +
87 std::abs(d) * lhs.imag.absolute_error +
88 std::abs(b) * rhs.imag.absolute_error;
89 result.value().imag.absolute_error +=
90 std::abs(d) * lhs.real.absolute_error +
91 std::abs(a) * rhs.imag.absolute_error +
92 std::abs(c) * lhs.imag.absolute_error +
93 std::abs(b) * rhs.real.absolute_error;
94 return result;
95}

97Result<ApproxComplex> divide_complex(const ApproxComplex& lhs,
98 const ApproxComplex& rhs) {
99 const double c = rhs.real.value;
100 const double d = rhs.imag.value;
101 const double denom = c * c + d * d;
102 if (denom == 0.0 || !std::isfinite(denom)) {
103 return complex_failure(CasErrc::DomainError,
104 "complex division by zero or overflow",
105 kEvalComplexOperation);
106 }
107 return checked_complex((lhs.real.value * c + lhs.imag.value * d) / denom,
108 (lhs.imag.value * c - lhs.real.value * d) / denom,
109 kEvalComplexOperation);
110}

116Result<ApproxComplex> evaluate_complex_node(
117 const std::shared_ptr<const SymbolicNode>& node,
118 const NumericBindings& bindings,
119 ComputationContext& context) {
120 auto step = context.consume_steps(1, kEvalComplexOperation);
121 if (!step) return Result<ApproxComplex>::failure(step.error());
122 if (!node) {
123 return complex_failure(CasErrc::InvalidArgument,
124 "expression contains a null node",
125 kEvalComplexOperation);
126 }
127
128 if (auto complex_node = std::dynamic_pointer_cast<const ComplexNode>(node)) {
129 auto real = evaluate_numeric(
130 *lamina::detail::make_expression_ptr(complex_node->real()),
131 bindings, context);
132 if (!real) return eval_complex_failure(real.error());
133 auto imag = evaluate_numeric(
134 *lamina::detail::make_expression_ptr(complex_node->imag()),
135 bindings, context);
136 if (!imag) return eval_complex_failure(imag.error());
137 if (!real.value().is_finite() || !imag.value().is_finite()) {
138 return complex_failure(CasErrc::NumericFailure,
139 "complex components must be finite",
140 kEvalComplexOperation);
141 }
142 return Result<ApproxComplex>::success(
143 ApproxComplex{real.value(), imag.value()});
144 }
145
146 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(node)) {
147 if (is_imaginary_unit_name(variable->name())) {
148 return Result<ApproxComplex>::success(approx_complex(0.0, 1.0));
149 }
150 }
151
152 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
153 auto sum = Result<ApproxComplex>::success(approx_complex(0.0, 0.0));
154 for (const auto& operand : add->operands()) {
155 auto term = evaluate_complex_node(operand, bindings, context);
156 if (!term) return term;
157 sum = add_complex(sum.value(), term.value());
158 if (!sum) return sum;
159 }
160 return sum;
161 }
162
163 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
164 auto product = Result<ApproxComplex>::success(approx_complex(1.0, 0.0));
165 for (const auto& operand : multiply->operands()) {
166 auto factor = evaluate_complex_node(operand, bindings, context);
167 if (!factor) return factor;
168 product = multiply_complex(product.value(), factor.value());
169 if (!product) return product;
170 }
171 return product;
172 }
173
174 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
175 auto base = evaluate_complex_node(power->base(), bindings, context);
176 if (!base) return base;
177 auto exponent = evaluate_numeric(
178 *lamina::detail::make_expression_ptr(power->exponent()),
179 bindings, context);
180 if (!exponent) return eval_complex_failure(exponent.error());
181 if (!exponent.value().is_finite() ||
182 !std::isfinite(exponent.value().value)) {
183 return complex_failure(CasErrc::NumericFailure,
184 "complex power exponent must be finite",
185 kEvalComplexOperation);
186 }
187 const double exponent_value = exponent.value().value;
188 if (!is_integer_double(exponent_value) ||
189 std::abs(exponent_value) > 64.0) {
190 return complex_failure(CasErrc::UnsupportedExpression,
191 "complex evaluation only supports integer powers with |n| <= 64",
192 kEvalComplexOperation);
193 }
194 int exponent_int = static_cast<int>(exponent_value);
195 auto result = Result<ApproxComplex>::success(approx_complex(1.0, 0.0));
196 for (int i = 0; i < std::abs(exponent_int); ++i) {
197 result = multiply_complex(result.value(), base.value());
198 if (!result) return result;
199 }
200 if (exponent_int < 0) {
201 result = divide_complex(approx_complex(1.0, 0.0), result.value());
202 }
203 return result;
204 }
205
206 return real_to_complex(
207 evaluate_numeric(*lamina::detail::make_expression_ptr(node),
208 bindings, context));
209}

234Result<ApproxComplex> eval_complex(const SymbolicExpr& expression,
235 const NumericBindings& bindings,
236 ComputationContext& context) {
237 try {
238 if (!lamina::detail::node(expression)) {
239 return complex_failure(CasErrc::InvalidArgument,
240 "cannot evaluate an empty expression as complex",
241 kEvalComplexOperation);
242 }
243 return evaluate_complex_node(lamina::detail::node(expression),
244 bindings, context);
245 } catch (const std::bad_alloc&) {
246 return complex_failure(CasErrc::ResourceLimit,
247 "complex evaluation allocation failed",
248 kEvalComplexOperation);
249 } catch (const std::exception& error) {
250 return complex_failure(CasErrc::InternalInvariant, error.what(),
251 kEvalComplexOperation);
252 }
253}
