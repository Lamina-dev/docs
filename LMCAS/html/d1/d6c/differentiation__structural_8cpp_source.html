98 if (!node.is_definite()) {

110 if (node.variable() != var) {
111 node.body()->accept(*this);

116 }
117

119 LMCAS::detail::expression_from_node(node.body()));
120 auto upper_expression = LMCAS::detail::make_expression_ptr(
121 LMCAS::detail::expression_from_node(node.upper()));

132 }

140 SymbolicFactory::create_number(BigInt(-1)),
141 LMCAS::detail::node(at_lower),
