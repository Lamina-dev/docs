48/** @brief 重写规则，定义模式到替换表达式的变换 */
49struct LAMINA_API Rule {

56 /// Assumption-aware condition: receives bindings and the assumption context.

59
60 Rule(SymbolicExpr p, SymbolicExpr r, std::unordered_set<std::string> w,
61 std::function<bool(const MatchMap&)> c = nullptr)

63
64 Rule(SymbolicExpr p, SymbolicExpr r, std::unordered_set<std::string> w,
