35Result<ExprPtr> determinant_cofactor(
36 const ExactMatrixData& matrix,
37 ComputationContext& context,
38 const std::string& operation) {
39 auto budget = context.consume_steps(1, operation);
40 if (!budget) return Result<ExprPtr>::failure(budget.error());
41 if (matrix.rows == 1) {
42 return Result<ExprPtr>::success(matrix.entries[0]);
43 }
44 auto sum = SymbolicExpr::number(0);
45 for (std::size_t column = 0; column < matrix.cols; ++column) {
46 if (auto coefficient = exact_rational(matrix.at(0, column));
47 coefficient && *coefficient == Rational(0)) {
48 continue;
49 }
50 ExactMatrixData minor{
51 matrix.rows - 1, matrix.cols - 1, {}};
52 minor.entries.reserve(minor.rows * minor.cols);
53 for (std::size_t row = 1; row < matrix.rows; ++row) {
54 for (std::size_t source = 0; source < matrix.cols; ++source) {
55 if (source != column) {
56 minor.entries.push_back(matrix.at(row, source));
57 }
58 }
59 }
60 auto minor_determinant = determinant_cofactor(
61 minor, context, operation);
62 if (!minor_determinant) {
63 return Result<ExprPtr>::failure(minor_determinant.error());
64 }
65 auto term = exact_multiply(
66 matrix.at(0, column), minor_determinant.value());
67 if (column % 2 != 0) term = exact_negate(term);
68 sum = exact_add(sum, term);
69 }
70 return Result<ExprPtr>::success(simplify_expr(sum));
71}

73Result<ExprPtr> symbolic_determinant_value(
74 const ExactMatrixData& input, ComputationContext& context,
75 const std::string& operation) {
76 ExactMatrixData matrix = input;
77 const std::size_t n = matrix.rows;
78 if (n == 1) return Result<ExprPtr>::success(matrix.entries[0]);
79 auto previous = SymbolicExpr::number(1);
80 std::size_t swaps = 0;
81 for (std::size_t column = 0; column + 1 < n; ++column) {
82 auto pivot = choose_pivot(matrix, column, column, context, operation);
83 if (!pivot) {
84 if (pivot.error().code == CasErrc::Inconclusive) {
85 return determinant_cofactor(input, context, operation);
86 }
87 return Result<ExprPtr>::failure(pivot.error());
88 }
89 if (!pivot.value()) {
90 return Result<ExprPtr>::success(SymbolicExpr::number(0));
91 }
92 if (*pivot.value() != column) {
93 swap_rows(matrix, *pivot.value(), column);
94 ++swaps;
95 }
96 auto pivot_value = matrix.at(column, column);
97 for (std::size_t row = column + 1; row < n; ++row) {
98 for (std::size_t target = column + 1; target < n; ++target) {
99 auto numerator = exact_subtract(
100 exact_multiply(matrix.at(row, target), pivot_value),
101 exact_multiply(matrix.at(row, column), matrix.at(column, target)));
102 matrix.at(row, target) = exact_divide(numerator, previous);
103 }
104 matrix.at(row, column) = SymbolicExpr::number(0);
105 }
106 previous = pivot_value;
107 }
108 auto determinant = matrix.at(n - 1, n - 1);
109 if (swaps % 2 != 0) determinant = exact_negate(determinant);
110 return Result<ExprPtr>::success(simplify_expr(determinant));
111}

115Result<ExprPtr> determinant_exact(
116 const ExactMatrixData& input,
117 ComputationContext& context,
118 const std::string& operation) {
119 auto valid = validate_matrix(input, operation);
120 if (!valid) return Result<ExprPtr>::failure(valid.error());
121 if (input.rows != input.cols) {
122 return Result<ExprPtr>::failure(
123 CasErrc::InvalidArgument,
124 "determinant requires a square matrix", operation);
125 }
126 for (const auto& entry : input.entries) {
127 if (contains_approximate_number(entry)) {
128 return Result<ExprPtr>::failure(
129 CasErrc::Inconclusive,
130 "exact determinant rejects approximate entries", operation);
131 }
132 }
133 bool all_rational = true;
134 for (const auto& entry : input.entries) {
135 all_rational = all_rational && exact_rational(entry).has_value();
136 }
137 if (all_rational) return rational_determinant(input, context, operation);
138
139 try {
140 return symbolic_determinant_value(input, context, operation);
141 } catch (const std::bad_alloc&) {
142 return Result<ExprPtr>::failure(
143 CasErrc::ResourceLimit,
144 "exact determinant allocation failed", operation);
145 } catch (const std::exception& error) {
146 return Result<ExprPtr>::failure(
147 CasErrc::InternalInvariant, error.what(), operation);
148 }
