53void make_monic(Poly& polynomial) {
54 Rational leading = polynomial.lead_coeff();
55 if (leading == Rational(0) || leading == Rational(1)) return;
56 Rational inverse = Rational(1) / leading;
57 Poly monic(polynomial.num_vars);
58 for (const auto& [monomial, coefficient] : polynomial.terms) {
59 monic.add_term(monomial, coefficient * inverse);

65void buchberger(std::vector<Poly>& basis, size_t num_vars) {
66 initialize_sugar(basis);
67 auto pairs = initial_pairs(basis);
68 std::set<std::pair<size_t, size_t>> processed_pairs;
69 while (!pairs.empty()) {
70 auto [i, j, sugar_degree] = pairs.top();
71 pairs.pop();
72 auto canonical_pair = i < j ? std::make_pair(i, j) : std::make_pair(j, i);
73 processed_pairs.insert(canonical_pair);
74 if (coprime_leading_monomials(basis[i], basis[j])) continue;
75 if (chain_criterion(basis, i, j, processed_pairs)) continue;
