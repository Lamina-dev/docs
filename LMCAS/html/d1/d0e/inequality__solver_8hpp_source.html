24/** @brief 参数化不等式的分段解集 */
25struct LAMINA_API PiecewiseIntervalResult {
26 /** @brief 单个分段：条件 + 对应解集 */
27 struct Case {
28 std::shared_ptr<SymbolicExpr> condition; ///< 参数满足的条件
29 IntervalUnion solution; ///< 该条件下的解集
30 };
31 std::vector<Case> cases; ///< 所有分段
32
33 /**
34 * @brief 判断结果是否为空
35 * @return 无分段时返回 true
36 */
37 bool is_empty() const { return cases.empty(); }
38
39 /**
40 * @brief 判断是否只有一个分段
41 * @return 恰好一个分段时返回 true
42 */
43 bool is_single() const { return cases.size() == 1; }
44
45 /**
46 * @brief 获取唯一分段的解集
47 *
48 * 仅当 is_single() 为真时才返回该分段；多分段意味着解依赖参数取值，
49 * 此时返回空集而非默默丢弃其它分段。
50 * @return 单一分段时返回该分段解集，否则返回空集
51 */
52 IntervalUnion single_solution() const {
53 if (!is_single()) return IntervalUnion::empty();
54 return cases[0].solution;
55 }
56};
