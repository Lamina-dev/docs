168lmmc_real_t Rational::to_double() const {
169 if (is_zero()) return 0.0;
170 const std::string num_text = numerator.abs().to_string();
171 const std::string den_text = denominator.to_string();
172 const std::size_t num_digits = std::min<std::size_t>(18, num_text.size());
173 const std::size_t den_digits = std::min<std::size_t>(18, den_text.size());
174 const long double num_head = std::stold(num_text.substr(0, num_digits));
175 const long double den_head = std::stold(den_text.substr(0, den_digits));
176 const std::size_t num_tail = num_text.size() - num_digits;
177 const std::size_t den_tail = den_text.size() - den_digits;
178 const bool negative_shift = num_tail < den_tail;
179 const std::size_t shift = negative_shift ? den_tail - num_tail : num_tail - den_tail;
180 if (shift > static_cast<std::uint64_t>(std::numeric_limits<std::int64_t>::max())) {
181 if (negative_shift) {
182 throw std::underflow_error("Rational underflow during double conversion");
183 }
184 throw std::overflow_error("Rational cannot be represented as a finite double");
185 }
186 const std::int64_t decimal_shift = negative_shift ?
187 -static_cast<std::int64_t>(shift) : static_cast<std::int64_t>(shift);
188 long double value = (num_head / den_head) *
189 std::pow(10.0L, static_cast<long double>(decimal_shift));
190 if (numerator.is_negative()) value = -value;
191 const lmmc_real_t converted = static_cast<lmmc_real_t>(value);
192 if (!std::isfinite(converted)) {
193 /**
194 * @brief 用精确整数运算判定 binary64 的有限值舍入边界。
195 * 十进制前缀近似可能越过上界，而精确有理数仍舍入为 DBL_MAX；
196 * 绝对值小于 DBL_MAX + 0.5 ULP 时仍舍入为有限值。
197 */
198 const BigInt max_finite_integer =
199 (BigInt(1) << 1024) - (BigInt(1) << 971);
200 const BigInt finite_rounding_limit =
201 max_finite_integer + (BigInt(1) << 970);
202 if (numerator.abs() <
203 denominator * finite_rounding_limit) {
204 return std::copysign(
205 std::numeric_limits<lmmc_real_t>::max(),
206 numerator.is_negative() ? -1.0 : 1.0);
207 }
208 throw std::overflow_error(
209 "Rational cannot be represented as a finite double");
210 }
211 if (converted == 0.0) {
212 throw std::underflow_error("Rational underflow during double conversion");
213 }
214 return converted;
215}
