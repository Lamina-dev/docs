26inline int infinity_sign(const SymbolicExpr& expression) {
27 const auto& node = detail::node(expression);
28 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(node)) {
29 return function->type() == FunctionNode::FuncType::Infinity ? 1 : 0;
30 }
31 auto product = std::dynamic_pointer_cast<const MultiplyNode>(node);
32 if (!product) return 0;
33
34 bool has_infinity = false;
35 int sign = 1;
36 for (const auto& operand : product->operands()) {
37 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(operand)) {
38 has_infinity = has_infinity ||
39 function->type() == FunctionNode::FuncType::Infinity;
40 }
41 if (auto number = std::dynamic_pointer_cast<const NumberNode>(operand)) {
42 if (std::holds_alternative<BigInt>(number->value()) &&
43 std::get<BigInt>(number->value()) < BigInt(0)) {
44 sign = -sign;
45 } else if (std::holds_alternative<Rational>(number->value()) &&
46 std::get<Rational>(number->value()) < Rational(0)) {
47 sign = -sign;
48 } else if (std::holds_alternative<lmmc_real_t>(number->value()) &&
49 std::get<lmmc_real_t>(number->value()) < 0) {
50 sign = -sign;
51 }
52 }
53 }
54 return has_infinity ? sign : 0;
55}

57template <typename T>
58Result<T> checked_inference_result(
59 const SymbolicExpr& expr,
60 const std::string& operation,
61 const std::function<Result<T>()>& query) {
62 if (!detail::node(expr)) {
63 return Result<T>::failure(
64 CasErrc::InvalidArgument,
65 "inference expression must not be null", operation);
66 }
67 try {
68 return query();
69 } catch (const std::bad_alloc&) {
70 return Result<T>::failure(
71 CasErrc::ResourceLimit,
72 "inference query allocation failed", operation);
73 } catch (const std::exception& error) {
74 return Result<T>::failure(
75 CasErrc::InternalInvariant, error.what(), operation);
76 }
77}
