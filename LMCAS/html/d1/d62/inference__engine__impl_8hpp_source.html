27inline bool is_negative_infinity_coefficient(const NumberNode& number) {
28 if (std::holds_alternative<BigInt>(number.value())) {
29 return std::get<BigInt>(number.value()) < BigInt(0);
30 }
31 if (std::holds_alternative<Rational>(number.value())) {
32 return std::get<Rational>(number.value()) < Rational(0);
33 }
34 if (std::holds_alternative<lmmc_real_t>(number.value())) {
35 return std::get<lmmc_real_t>(number.value()) < 0;
36 }
37 return false;
38}

40inline int infinity_sign(const SymbolicExpr& expression, ComputationContext& context) {
41 const auto& node = detail::node(expression);
42 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(node)) {
43 return function->type() == FunctionNode::FuncType::Infinity ? 1 : 0;
44 }
45 auto product = std::dynamic_pointer_cast<const MultiplyNode>(node);
46 if (!product) {
47 return 0;
48 }
49
50 bool has_infinity = false;
51 int sign = 1;
52 for (const auto& operand : product->operands()) {
53 auto step = context.consume_steps(1, "inference.infinity");
54 if (!step) throw step.error();
55 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(operand)) {
56 has_infinity = has_infinity ||
57 function->type() == FunctionNode::FuncType::Infinity;
58 }
59 if (auto number = std::dynamic_pointer_cast<const NumberNode>(operand)) {
60 if (is_negative_infinity_coefficient(*number)) {
61 sign = -sign;
62 }
63 }
64 }
65 return has_infinity ? sign : 0;
66}

68template <typename T>
69Result<T> checked_inference_result(
70 const SymbolicExpr& expr,
71 const std::string& operation,
72 ComputationContext& context,
73 const std::function<Result<T>()>& query) {
74 if (!detail::node(expr)) {
75 return Result<T>::failure(
76 CasErrc::InvalidArgument,
77 "inference expression must not be null", operation);
78 }
79 try {
80 auto entered = context.enter_recursion(operation);
81 if (!entered) return Result<T>::failure(entered.error());
82 struct Leave {
83 ComputationContext& context;
84 ~Leave() { context.leave_recursion(); }
85 } leave{context};
86 return query();
87 } catch (const CasError& error) {
88 return Result<T>::failure(error);
89 } catch (const std::bad_alloc&) {
90 return Result<T>::failure(
91 CasErrc::ResourceLimit,
92 "inference query allocation failed", operation);
93 } catch (const std::exception& error) {
94 return Result<T>::failure(
95 CasErrc::InternalInvariant, error.what(), operation);
96 }
97}
