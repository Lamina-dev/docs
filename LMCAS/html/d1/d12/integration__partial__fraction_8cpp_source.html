30std::shared_ptr<SymbolicExpr> integrate_numeric_quadratic(
31 const Polynomial<SymbolicPolyCoeff>& Q, const std::string& var,
32 ComputationContext& computation) {
33 SymbolicExpr c_expr = *(Q.coeffs[0].val);
34 SymbolicExpr b_expr = *(Q.coeffs[1].val);
35 SymbolicExpr a_expr = *(Q.coeffs[2].val);
36
37 if (!a_expr.is_number() || !b_expr.is_number() || !c_expr.is_number()) {
38 return nullptr;
39 }
40
41 auto a_checked = try_checked_numeric_constant(a_expr, computation);
42 auto b_checked = try_checked_numeric_constant(b_expr, computation);
43 auto c_checked = try_checked_numeric_constant(c_expr, computation);
44 if (!a_checked || !b_checked || !c_checked) {
45 return nullptr;
46 }
47
48 double a = *a_checked;
49 double b = *b_checked;
50 double c = *c_checked;
51
52 int eq_a;
53 lmmc_double_nearly_equal_tol(a, 0.0, 1e-9, 1e-9, &eq_a);
54 if (eq_a) { return nullptr; }
55
56 double delta = b * b - 4 * a * c;
57 int eq_delta;
58 lmmc_double_nearly_equal_tol(delta, 0.0, 1e-9, 1e-9, &eq_delta);
59
60 if (!eq_delta && delta > 0) {
61 double sqrt_delta = std::sqrt(delta);
62 auto scalar = SymbolicExpr::number(1.0 / sqrt_delta);
63 auto two_a = SymbolicExpr::number(2.0 * a);
64 auto b_num = SymbolicExpr::number(b);
65 auto two_a_x = SymbolicExpr::multiply(detail::make_expression_ptr(*two_a), SymbolicExpr::variable(var));
66 auto two_a_x_plus_b = SymbolicExpr::add(detail::make_expression_ptr(*two_a_x), detail::make_expression_ptr(*b_num));
67 auto term1_arg = sym_sub(*two_a_x_plus_b, *SymbolicExpr::number(sqrt_delta));
68 auto term2_arg = SymbolicExpr::add(detail::make_expression_ptr(*two_a_x_plus_b), SymbolicExpr::number(sqrt_delta));
69 auto term1 = SymbolicExpr::ln(detail::make_expression_ptr(*term1_arg));
70 auto term2 = SymbolicExpr::ln(detail::make_expression_ptr(*term2_arg));
71 return SymbolicExpr::multiply(scalar, sym_sub(*term1, *term2));
72 } else if (!eq_delta && delta < 0) {
73 double sqrt_neg_delta = std::sqrt(-delta);
74 auto scalar = SymbolicExpr::number(2.0 / sqrt_neg_delta);
75 auto two_a = SymbolicExpr::number(2.0 * a);
76 auto b_num = SymbolicExpr::number(b);
77 auto num = SymbolicExpr::add(
78 SymbolicExpr::multiply(detail::make_expression_ptr(*two_a), SymbolicExpr::variable(var)),
79 detail::make_expression_ptr(*b_num));
80 auto inner = SymbolicExpr::divide(detail::make_expression_ptr(*num), SymbolicExpr::number(sqrt_neg_delta));
81 return SymbolicExpr::multiply(scalar, make_arctan(inner));
82 }
83 return nullptr;
84}
