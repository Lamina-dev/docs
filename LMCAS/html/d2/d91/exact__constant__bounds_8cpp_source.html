10 * @brief 用交错部分和夹逼 atan(x)。
11 *
12 * |x|≤1/2 时相邻项绝对值递减；相邻交错部分和夹住 atan(x)，差值恰为首个省略项。
13 */
14Bounds small_atan(const Rational& x, std::size_t precision,
15 const ExactBoundArithmetic& arithmetic) {
16 const Rational target(BigInt(1), arithmetic.dyadic_scale(precision));
17 if (x.is_zero()) return {Rational(0), Rational(0)};
18 if (x.get_numerator().is_negative()) {
19 auto bounds = small_atan(-x, precision, arithmetic);
20 return {-bounds.second, -bounds.first};
21 }
22 const Rational square = arithmetic.mul(x, x);
23 Rational power = x;
24 Rational sum = x;
25 BigInt odd(1);
26 bool subtract = true;
27 for (;;) {
28 arithmetic.step();
29 arithmetic.growth(odd.bit_length(), 2);
30 odd += BigInt(2);
31 power = arithmetic.mul(power, square);
32 const Rational omitted = arithmetic.div(power, Rational(odd));
33 const Rational adjacent = subtract ? arithmetic.sub(sum, omitted)
34 : arithmetic.add(sum, omitted);
35 if (!arithmetic.less(target, omitted)) {
36 return subtract ? Bounds{adjacent, sum} : Bounds{sum, adjacent};
37 }
38 sum = adjacent;
39 subtract = !subtract;
40 }
41}

43Bounds pi_bounds(std::size_t precision, const ExactBoundArithmetic& arithmetic) {
44 /**
45 * @brief 以小参数 atan 区间构造 π 界。
46 *
47 * 每个 atan 区间宽度≤2^-(p+5)，组合宽度≤(16+4)*2^-(p+5)<2^-p。
48 */
49 const auto internal = arithmetic.growth(precision, 5);
50 auto fifth = small_atan(Rational(1, 5), internal, arithmetic);
51 auto two_thirty_ninth = small_atan(Rational(1, 239), internal, arithmetic);
52 return {arithmetic.sub(arithmetic.mul(Rational(16), fifth.first),
53 arithmetic.mul(Rational(4), two_thirty_ninth.second)),
54 arithmetic.sub(arithmetic.mul(Rational(16), fifth.second),
55 arithmetic.mul(Rational(4), two_thirty_ninth.first))};
56}

58Bounds atan_bounds(const Rational& x, std::size_t precision,
59 const ExactBoundArithmetic& arithmetic) {
60 arithmetic.step();
61 if (x.get_numerator().is_negative()) {
62 auto bounds = atan_bounds(-x, precision, arithmetic);
63 return {-bounds.second, -bounds.first};
64 }
65 if (!arithmetic.less(Rational(1, 2), x))
66 return small_atan(x, precision, arithmetic);
67 const auto internal = arithmetic.growth(precision, 2);
68 auto pi = pi_bounds(internal, arithmetic);
69 if (arithmetic.less(Rational(1), x)) {
70 auto reciprocal = atan_bounds(arithmetic.div(Rational(1), x), internal, arithmetic);
71 return {arithmetic.sub(arithmetic.div(pi.first, Rational(2)), reciprocal.second),
72 arithmetic.sub(arithmetic.div(pi.second, Rational(2)), reciprocal.first)};
73 }
74 const Rational transformed = arithmetic.div(arithmetic.sub(x, Rational(1)),
75 arithmetic.add(x, Rational(1)));
76 auto correction = small_atan(transformed, internal, arithmetic);
77 return {arithmetic.add(arithmetic.div(pi.first, Rational(4)), correction.first),
78 arithmetic.add(arithmetic.div(pi.second, Rational(4)), correction.second)};
79}
