18Result<std::shared_ptr<const MatrixNode>> validate_metric_matrix(
19 const std::shared_ptr<SymbolicExpr>& metric,
20 const std::vector<std::string>* coords,
21 ComputationContext& context,
22 const std::string& operation)
23{
24 auto step = context.consume_steps(1, operation);
25 if (!step) return Result<std::shared_ptr<const MatrixNode>>::failure(step.error());
26 if (!metric || !lamina::detail::node(metric)) {
27 return Result<std::shared_ptr<const MatrixNode>>::failure(
28 CasErrc::InvalidArgument,
29 "metric tensor cannot be null",
30 operation);
31 }
32 auto matrix = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(metric));
33 if (!matrix) {
34 return Result<std::shared_ptr<const MatrixNode>>::failure(
35 CasErrc::InvalidArgument,
36 "metric tensor must be a matrix",
37 operation);
38 }
39 if (matrix->rows() != matrix->cols()) {
40 return Result<std::shared_ptr<const MatrixNode>>::failure(
41 CasErrc::InvalidArgument,
42 "metric tensor must be square",
43 operation);
44 }
45 if (coords) {
46 if (coords->empty()) {
47 return Result<std::shared_ptr<const MatrixNode>>::failure(
48 CasErrc::InvalidArgument,
49 "coordinate list cannot be empty",
50 operation);
51 }
52 if (coords->size() != matrix->rows()) {
53 return Result<std::shared_ptr<const MatrixNode>>::failure(
54 CasErrc::InvalidArgument,
55 "coordinate count must match metric dimension",
56 operation);
57 }
58 for (const auto& coord : *coords) {
59 if (coord.empty()) {
60 return Result<std::shared_ptr<const MatrixNode>>::failure(
61 CasErrc::InvalidArgument,
62 "coordinate names cannot be empty",
63 operation);
64 }
65 }
66 }
67 return Result<std::shared_ptr<const MatrixNode>>::success(matrix);
68}

115Result<std::size_t> bounded_binomial(std::size_t n,
116 std::size_t k,
117 std::size_t limit,
118 const std::string& operation)
119{
120 if (k > n) return Result<std::size_t>::success(0);
121 k = std::min(k, n - k);
122 std::size_t value = 1;
123 for (std::size_t i = 1; i <= k; ++i) {
124 std::size_t numerator = n - k + i;
125 std::size_t denominator = i;
126 const auto fraction_gcd = std::gcd(numerator, denominator);
127 numerator /= fraction_gcd;
128 denominator /= fraction_gcd;
129 const auto value_gcd = std::gcd(value, denominator);
130 value /= value_gcd;
131 denominator /= value_gcd;
132 if (denominator != 1) {
133 return Result<std::size_t>::failure(
134 CasErrc::InternalInvariant,
135 "binomial coefficient reduction was not exact",
136 operation);
137 }
138 if (numerator != 0 && value > limit / numerator) {
139 return Result<std::size_t>::failure(
140 CasErrc::ResourceLimit,
141 "differential-form component count exceeds the expansion budget",
142 operation);
143 }
144 value *= numerator;
145 }
146 if (value > limit) {
147 return Result<std::size_t>::failure(
148 CasErrc::ResourceLimit,
149 "differential-form component count exceeds the expansion budget",
150 operation);
151 }
152 return Result<std::size_t>::success(value);
153}

155std::vector<std::vector<std::size_t>> index_combinations(std::size_t n,
156 std::size_t k,
157 std::size_t count)
158{
159 std::vector<std::vector<std::size_t>> combinations;
160 combinations.reserve(count);
161 if (k > n) return combinations;
162 if (k == 0) {
163 combinations.emplace_back();
164 return combinations;
165 }
166
167 std::vector<std::size_t> current(k);
168 std::iota(current.begin(), current.end(), 0);
169 while (true) {
170 combinations.push_back(current);
171 std::size_t position = k;
172 while (position > 0) {
173 --position;
174 if (current[position] < n - k + position) break;
175 }
176 if (position == 0 && current[0] == n - k) break;
177 ++current[position];
178 for (std::size_t i = position + 1; i < k; ++i) {
179 current[i] = current[i - 1] + 1;
180 }
181 }
182 return combinations;
183}

280static std::shared_ptr<SymbolicExpr> christoffel_first_kind_impl(
281 const std::shared_ptr<SymbolicExpr>& g_ij,
282 const std::vector<std::string>& coords,
283 int k, int i, int j) {
284
285 auto mat = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(g_ij));
286 if (!mat) return SymbolicExpr::number(0);
287
288 auto g_jk = lamina::detail::make_expression_ptr(mat->get(j, k));
289 auto g_ik = lamina::detail::make_expression_ptr(mat->get(i, k));
290 auto g_ij_comp = lamina::detail::make_expression_ptr(mat->get(i, j));
291
292 auto term1 = g_jk->differentiate(coords[i]);
293 auto term2 = g_ik->differentiate(coords[j]);
294 auto term3 = g_ij_comp->differentiate(coords[k]);
295
296 auto sum12 = SymbolicExpr::add(term1, term2);
297 auto diff = SymbolicExpr::add(sum12, SymbolicExpr::multiply(term3, SymbolicExpr::number(-1)));
298
299 return SymbolicExpr::multiply(
300 SymbolicExpr::number(Rational(BigInt(1), BigInt(2))),
301 diff)->simplify();
302}

355static std::shared_ptr<SymbolicExpr> christoffel_second_kind_impl(
356 const std::shared_ptr<SymbolicExpr>& g_ij,
357 const std::shared_ptr<SymbolicExpr>& g_up_ij,
358 const std::vector<std::string>& coords,
359 int k, int i, int j) {
360
361 auto mat_up = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(g_up_ij));
362 if (!mat_up) return SymbolicExpr::number(0);
363
364 size_t dim = coords.size();
365 auto result = SymbolicExpr::number(0);
366
367 for (size_t m = 0; m < dim; m++) {
368 auto g_up_km = lamina::detail::make_expression_ptr(mat_up->get(k, m));
369 auto gamma_first = christoffel_first_kind_impl(g_ij, coords, m, i, j);
370 auto term = SymbolicExpr::multiply(g_up_km, gamma_first);
371 result = SymbolicExpr::add(result, term);
372 }
373
374 return result->simplify();
375}

377static std::shared_ptr<SymbolicExpr> riemann_curvature_tensor_with_inverse(
378 const std::shared_ptr<SymbolicExpr>& g_ij,
379 const std::shared_ptr<SymbolicExpr>& g_up_ij,
380 const std::vector<std::string>& coords,
381 int rho, int sigma, int mu, int nu) {
382 auto gamma_rho_nu_sigma =
383 christoffel_second_kind_impl(g_ij, g_up_ij, coords, rho, nu, sigma);
384 auto gamma_rho_mu_sigma =
385 christoffel_second_kind_impl(g_ij, g_up_ij, coords, rho, mu, sigma);
386
387 auto term1 = gamma_rho_nu_sigma->differentiate(coords[mu]);
388 auto term2 = gamma_rho_mu_sigma->differentiate(coords[nu]);
389 auto result = SymbolicExpr::add(
390 term1,
391 SymbolicExpr::multiply(term2, SymbolicExpr::number(-1)));
392
393 for (std::size_t lambda = 0; lambda < coords.size(); ++lambda) {
394 auto gamma_rho_mu_lambda =
395 christoffel_second_kind_impl(g_ij, g_up_ij, coords, rho, mu, lambda);
396 auto gamma_lambda_nu_sigma =
397 christoffel_second_kind_impl(g_ij, g_up_ij, coords, lambda, nu, sigma);
398 auto gamma_rho_nu_lambda =
399 christoffel_second_kind_impl(g_ij, g_up_ij, coords, rho, nu, lambda);
400 auto gamma_lambda_mu_sigma =
401 christoffel_second_kind_impl(g_ij, g_up_ij, coords, lambda, mu, sigma);
402 auto product_difference = SymbolicExpr::add(
403 SymbolicExpr::multiply(
404 gamma_rho_mu_lambda, gamma_lambda_nu_sigma),
405 SymbolicExpr::multiply(
406 SymbolicExpr::multiply(
407 gamma_rho_nu_lambda, gamma_lambda_mu_sigma),
408 SymbolicExpr::number(-1)));
409 result = SymbolicExpr::add(result, product_difference);
410 }
411 return result->simplify();
412}

471 const std::vector<std::shared_ptr<SymbolicExpr>>& X,
472 const std::vector<std::string>& vars,

496 if (!result || !lamina::detail::node(result)) {

502 return DifferentialGeometryExprResult::success(result);
503 } catch (const std::bad_alloc&) {
504 return DifferentialGeometryExprResult::failure(CasErrc::ResourceLimit,

524static std::shared_ptr<SymbolicExpr> lie_derivative_impl(
525 const std::shared_ptr<SymbolicExpr>& f,
526 const std::vector<std::shared_ptr<SymbolicExpr>>& X,
527 const std::vector<std::string>& vars,
528 int order) {
529 if (!f || X.size() != vars.size()) return SymbolicExpr::number(0);
530 auto result = f;
531 for (int it = 0; it < order; ++it) {
532 auto acc = SymbolicExpr::number(0);
533 for (size_t i = 0; i < vars.size(); ++i) {
534 auto partial = result->differentiate(vars[i]);
535 acc = SymbolicExpr::add(acc, SymbolicExpr::multiply(X[i], partial));
536 }
537 result = acc->simplify();
538 }
539 return result;
540}

542DifferentialGeometryVectorResult exterior_derivative_checked(
543 const std::vector<std::shared_ptr<SymbolicExpr>>& form_coeffs,
544 int degree,
545 const std::vector<std::string>& vars,
546 ComputationContext& context)
547{
548 const std::string operation = "exterior_derivative";
549 auto step = context.consume_steps(1, operation);
550 if (!step) return DifferentialGeometryVectorResult::failure(step.error());
551 if (vars.empty()) {
552 return DifferentialGeometryVectorResult::failure(CasErrc::InvalidArgument,
553 "variable list cannot be empty",
554 operation);
555 }
556 for (std::size_t i = 0; i < vars.size(); ++i) {
557 if (vars[i].empty()) {
558 return DifferentialGeometryVectorResult::failure(CasErrc::InvalidArgument,
559 "variable names cannot be empty",
560 operation);
561 }
562 for (std::size_t j = 0; j < i; ++j) {
563 if (vars[i] == vars[j]) {
564 return DifferentialGeometryVectorResult::failure(
565 CasErrc::InvalidArgument,
566 "coordinate variables must be unique",
567 operation);
568 }
569 }
570 }
571 if (degree < 0 || static_cast<std::size_t>(degree) > vars.size()) {
572 return DifferentialGeometryVectorResult::failure(CasErrc::InvalidArgument,
573 "form degree must be between zero and the coordinate dimension",
574 operation);
575 }
576 const auto input_count = bounded_binomial(vars.size(),
577 static_cast<std::size_t>(degree),
578 context.limits().max_expansion_terms,
579 operation);
580 if (!input_count) return DifferentialGeometryVectorResult::failure(input_count.error());
581 if (form_coeffs.size() != input_count.value()) {
582 return DifferentialGeometryVectorResult::failure(CasErrc::InvalidArgument,
583 "form coefficient count does not match degree and dimension",
584 operation);
585 }
586 for (const auto& coeff : form_coeffs) {
587 if (!coeff || !lamina::detail::node(coeff)) {
588 return DifferentialGeometryVectorResult::failure(CasErrc::InvalidArgument,
589 "form coefficients cannot be null",
590 operation);
591 }
592 }
593 const auto output_count = bounded_binomial(vars.size(),
594 static_cast<std::size_t>(degree) + 1,
595 context.limits().max_expansion_terms,
596 operation);
597 if (!output_count) return DifferentialGeometryVectorResult::failure(output_count.error());
598 try {
599 const auto input_indices = index_combinations(vars.size(),
600 static_cast<std::size_t>(degree),
601 input_count.value());
602 const auto output_indices = index_combinations(vars.size(),
603 static_cast<std::size_t>(degree) + 1,
604 output_count.value());
605 std::map<std::vector<std::size_t>, std::size_t> coefficient_index;
606 for (std::size_t i = 0; i < input_indices.size(); ++i) {
607 coefficient_index.emplace(input_indices[i], i);
608 }
609
610 std::vector<std::shared_ptr<SymbolicExpr>> result;
611 result.reserve(output_count.value());
612 for (const auto& output_index : output_indices) {
613 auto component = SymbolicExpr::number(0);
614 for (std::size_t removed = 0; removed < output_index.size(); ++removed) {
615 auto term_budget = context.consume_steps(4, operation);
616 if (!term_budget) {
617 return DifferentialGeometryVectorResult::failure(term_budget.error());
618 }
619 auto node_budget = context.reserve_nodes(4, operation);
620 if (!node_budget) {
621 return DifferentialGeometryVectorResult::failure(node_budget.error());
622 }
623
624 auto source_index = output_index;
625 const std::size_t derivative_variable = source_index[removed];
626 source_index.erase(source_index.begin() + static_cast<std::ptrdiff_t>(removed));
627 const auto source = coefficient_index.find(source_index);
628 if (source == coefficient_index.end()) {
629 return DifferentialGeometryVectorResult::failure(
630 CasErrc::InternalInvariant,

633 }

637 return DifferentialGeometryVectorResult::failure(
638 CasErrc::InternalInvariant,
639 "exterior derivative produced a null partial derivative",
640 operation);
641 }
642 if (removed % 2 != 0) {
643 partial = SymbolicExpr::multiply(SymbolicExpr::number(-1), partial);
644 }
645 component = SymbolicExpr::add(component, partial);
646 }
647 component = component->simplify();
648 if (!component || !lamina::detail::node(component)) {
649 return DifferentialGeometryVectorResult::failure(
650 CasErrc::InternalInvariant,
651 "exterior derivative produced a null component",
652 operation);
653 }
654 result.push_back(std::move(component));
655 }
656 return DifferentialGeometryVectorResult::success(std::move(result));
657 } catch (const std::bad_alloc&) {
658 return DifferentialGeometryVectorResult::failure(CasErrc::ResourceLimit,
659 "allocation failed while calculating exterior derivative",
660 operation);
661 } catch (const std::exception& ex) {
662 return DifferentialGeometryVectorResult::failure(CasErrc::InternalInvariant,
663 ex.what(),
664 operation);
665 }
