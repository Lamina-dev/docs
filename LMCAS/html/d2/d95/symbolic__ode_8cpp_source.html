44std::shared_ptr<SymbolicExpr> solve_separable_ode(
45 std::shared_ptr<SymbolicExpr> rhs,
46 const std::string& x,
47 const std::string& y,
48 const AssumptionContext* ctx
49) {
50
51 auto inv_y = SymbolicExpr::divide(SymbolicExpr::number(1), rhs);
52 auto int_y = inv_y->integrate(y);
53 auto int_x = SymbolicExpr::number(1)->integrate(x);
54
55 auto result = SymbolicExpr::add(int_y, SymbolicExpr::multiply(SymbolicExpr::number(-1), int_x));
56
57 // When the dependent variable is known Positive, prefer the positive
58 // solution branch by wrapping in abs() (which simplifies to identity for
59 // positive expressions, signaling downstream that only positive values apply).
60 if (dep_var_is_positive(y, ctx)) {
61 result = make_abs(result);
62 }
63
64 return result;
65}

75) {

97std::shared_ptr<SymbolicExpr> solve_linear2_ode(
98 double a, double b, double c,
99 std::shared_ptr<SymbolicExpr> fx,
100 const std::string& x,
101 const std::string& y,
102 const AssumptionContext* ctx
103) {
104
105 // Determine whether the leading coefficient 'a' is known NonZero via
106 // assumptions. If so, skip the zero-coefficient degenerate case check.
107 bool a_known_nonzero = false;
108 if (ctx) {
109 auto a_expr = SymbolicExpr::number(a);
110 a_known_nonzero = expr_is_nonzero(a_expr, ctx);
111 }
112
113 if (!a_known_nonzero) {
114 // Guard a == 0: equation degenerates to first-order (or constant) form.
115 int a_is_zero = 0;
116 lmmc_double_nearly_equal_tol(a, 0.0, 1e-12, 1e-12, &a_is_zero);
117 if (a_is_zero) {
118 int b_is_zero = 0;
119 lmmc_double_nearly_equal_tol(b, 0.0, 1e-12, 1e-12, &b_is_zero);
120 if (b_is_zero) {
121 // a == 0 and b == 0: not a proper second-order ODE.
122 throw std::invalid_argument(
123 "solve_linear2_ode: leading and first-derivative coefficients are both zero");
124 }
125 // a == 0, b != 0: degenerates to b*y' + c*y = f(x), i.e. y' + (c/b)*y = f/b.
126 auto Px = SymbolicExpr::number(c / b);
127 auto Qx = fx->is_zero()
128 ? SymbolicExpr::number(0)
129 : SymbolicExpr::divide(fx, SymbolicExpr::number(b));
130 return solve_linear1_ode(Px, Qx, x, y, ctx);
131 }
132 }
133
134 double D = b*b - 4*a*c;
135 auto C1 = SymbolicExpr::variable("C1");
136 auto C2 = SymbolicExpr::variable("C2");
137 std::shared_ptr<SymbolicExpr> yh;
138 int eq;
139 lmmc_double_nearly_equal_tol(D, 0.0, 1e-12, 1e-12, &eq);
140 if (!eq && D > 0) {
141 double r1 = (-b + std::sqrt(D)) / (2*a);
142 double r2 = (-b - std::sqrt(D)) / (2*a);
143 yh = SymbolicExpr::add(
144 SymbolicExpr::multiply(C1, SymbolicExpr::exp(SymbolicExpr::multiply(SymbolicExpr::number(r1), SymbolicExpr::variable(x)))),
145 SymbolicExpr::multiply(C2, SymbolicExpr::exp(SymbolicExpr::multiply(SymbolicExpr::number(r2), SymbolicExpr::variable(x))))
146 );
147 } else if (eq) {
148 double r = -b / (2*a);
149 yh = SymbolicExpr::add(
150 SymbolicExpr::multiply(C1, SymbolicExpr::exp(SymbolicExpr::multiply(SymbolicExpr::number(r), SymbolicExpr::variable(x)))),
151 SymbolicExpr::multiply(C2, SymbolicExpr::multiply(SymbolicExpr::variable(x), SymbolicExpr::exp(SymbolicExpr::multiply(SymbolicExpr::number(r), SymbolicExpr::variable(x)))))
152 );
153 } else {
154 double real = -b / (2*a);
155 double imag = std::sqrt(-D) / (2*a);
156
157 auto exp_part = SymbolicExpr::exp(SymbolicExpr::multiply(SymbolicExpr::number(real), SymbolicExpr::variable(x)));
158 auto cos_part = SymbolicExpr::cos(SymbolicExpr::multiply(SymbolicExpr::number(imag), SymbolicExpr::variable(x)));
159 auto sin_part = SymbolicExpr::sin(SymbolicExpr::multiply(SymbolicExpr::number(imag), SymbolicExpr::variable(x)));
160 yh = SymbolicExpr::multiply(exp_part, SymbolicExpr::add(SymbolicExpr::multiply(C1, cos_part), SymbolicExpr::multiply(C2, sin_part)));
161 }
162
163 if (!fx->is_zero()) {
164 // Particular-solution computation for non-homogeneous case is not yet
165 // within the supported domain. Fail loudly instead of returning
166 // the homogeneous solution disguised as the full general solution.
167 throw std::logic_error(
168 "solve_linear2_ode: non-homogeneous case is outside the current support domain");
169 }
170
171 // When the dependent variable is known Positive, prefer positive branch.
172 if (dep_var_is_positive(y, ctx)) {
173 yh = make_abs(yh);
174 }
175
176 return yh;
177}

179Result<std::shared_ptr<SymbolicExpr>> solve_linear2_ode_checked(
180 double a, double b, double c,
181 std::shared_ptr<SymbolicExpr> fx,
182 const std::string& x,
183 const std::string& y,
184 ComputationContext& context,
185 const AssumptionContext* ctx
186) {
187 auto step = context.consume_steps(1, kSolveLinear2OdeOperation);
188 if (!step) return Result<std::shared_ptr<SymbolicExpr>>::failure(step.error());
189
190 if (!fx || !lamina::detail::node(fx)) {
191 return Result<std::shared_ptr<SymbolicExpr>>::failure(
192 CasErrc::InvalidArgument,
193 "forcing expression must not be null",
194 kSolveLinear2OdeOperation);
195 }
196 if (x.empty() || y.empty()) {
197 return Result<std::shared_ptr<SymbolicExpr>>::failure(
198 CasErrc::InvalidArgument,
199 "ODE variables must not be empty",
200 kSolveLinear2OdeOperation);
201 }
202
203 try {
204 if (!fx->is_zero()) {
205 return Result<std::shared_ptr<SymbolicExpr>>::failure(
206 CasErrc::Inconclusive,
207 "non-homogeneous second-order constant-coefficient ODEs are outside the current support domain",
208 kSolveLinear2OdeOperation);
209 }
210
211 return Result<std::shared_ptr<SymbolicExpr>>::success(
212 solve_linear2_ode(a, b, c, std::move(fx), x, y, ctx));
213 } catch (const std::invalid_argument& ex) {
214 return Result<std::shared_ptr<SymbolicExpr>>::failure(
215 CasErrc::InvalidArgument, ex.what(), kSolveLinear2OdeOperation);
216 } catch (const std::bad_alloc&) {
217 return Result<std::shared_ptr<SymbolicExpr>>::failure(
218 CasErrc::ResourceLimit,
219 "ODE solving allocation failed",
220 kSolveLinear2OdeOperation);
221 } catch (const std::logic_error& ex) {
222 return Result<std::shared_ptr<SymbolicExpr>>::failure(
223 CasErrc::Inconclusive, ex.what(), kSolveLinear2OdeOperation);
224 } catch (const std::exception& ex) {
225 return Result<std::shared_ptr<SymbolicExpr>>::failure(
226 CasErrc::InternalInvariant, ex.what(), kSolveLinear2OdeOperation);
227 }
228}
