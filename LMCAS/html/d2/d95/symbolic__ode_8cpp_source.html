22/// Check if the dependent variable is known Positive in the given context.
23static bool dep_var_is_positive(const std::string& y, const AssumptionContext* ctx) {
24 if (!ctx) return false;
25 auto y_expr = SymbolicExpr::variable(y);
26 auto positive = ctx->is_positive(*y_expr);
27 return positive && positive.value() == Tribool::True;
28}

40std::shared_ptr<SymbolicExpr> solve_separable_ode(
41 std::shared_ptr<SymbolicExpr> rhs,
42 const std::string& x,
43 const std::string& y,
44 const AssumptionContext* ctx
45) {
46
47 auto inv_y = SymbolicExpr::divide(SymbolicExpr::number(1), rhs);
48 auto int_y = inv_y->integrate(y);
49 auto int_x = SymbolicExpr::number(1)->integrate(x);
50
51 auto result = SymbolicExpr::add(int_y, SymbolicExpr::multiply(SymbolicExpr::number(-1), int_x));
52
53 // When the dependent variable is known Positive, prefer the positive
54 // solution branch by wrapping in abs() (which simplifies to identity for
55 // positive expressions, signaling downstream that only positive values apply).
56 if (dep_var_is_positive(y, ctx)) {
57 result = make_abs(result);
58 }
59
60 return result;
61}

66 std::shared_ptr<SymbolicExpr> Px,
67 std::shared_ptr<SymbolicExpr> Qx,
68 const std::string& x,
69 const std::string& y,
70 const AssumptionContext* ctx
71) {
72

75

77 auto intQmu = Qmu->integrate(x);

85 result = make_abs(result);

93static Result<std::shared_ptr<SymbolicExpr>>
94solve_linear2_homogeneous(
95 double a, double b, double c,
96 const std::string& x,
97 const std::string& y,
98 const AssumptionContext* ctx)
99{
100 if (!std::isfinite(a) || !std::isfinite(b) || !std::isfinite(c)) {
101 return Result<std::shared_ptr<SymbolicExpr>>::failure(
102 CasErrc::InvalidArgument,
103 "solve_linear2_ode: coefficients must be finite",
104 kSolveLinear2OdeOperation);
105 }
106 if (a == 0.0) {
107 if (b == 0.0) {
108 return Result<std::shared_ptr<SymbolicExpr>>::failure(
109 CasErrc::InvalidArgument,
110 "solve_linear2_ode: leading and first-derivative coefficients are both zero",
111 kSolveLinear2OdeOperation);
112 }
113 const double normalized_c = c / b;
114 if (!std::isfinite(normalized_c)) {
115 return Result<std::shared_ptr<SymbolicExpr>>::failure(
116 CasErrc::NumericFailure,
117 "solve_linear2_ode: degenerate coefficient normalization is non-finite",
118 kSolveLinear2OdeOperation);
119 }
120 auto solution = solve_linear1_ode(
121 SymbolicExpr::number(normalized_c),
122 SymbolicExpr::number(0), x, y, ctx);
123 return Result<std::shared_ptr<SymbolicExpr>>::success(
124 std::move(solution));
125 }
126
127 auto roots = ode_root_detail::find_characteristic_roots(
128 {a, b, c}, kSolveLinear2OdeOperation);
129 if (!roots) {
130 return Result<std::shared_ptr<SymbolicExpr>>::failure(roots.error());
131 }
132
133 auto variable = SymbolicExpr::variable(x);
134 auto solution = SymbolicExpr::number(0);
135 int constant_index = 1;
136 for (const auto& root : roots.value()) {
137 auto exponential = SymbolicExpr::exp(
138 SymbolicExpr::multiply(
139 SymbolicExpr::number(root.real_part), variable));
140 if (root.is_complex) {
141 auto argument = SymbolicExpr::multiply(
142 SymbolicExpr::number(root.imag_part), variable);
143 auto cosine_basis = SymbolicExpr::multiply(
144 exponential, SymbolicExpr::cos(argument));
145 auto sine_basis = SymbolicExpr::multiply(
146 exponential, SymbolicExpr::sin(argument));
147 auto cosine_term = SymbolicExpr::multiply(
148 SymbolicExpr::variable(
149 "C" + std::to_string(constant_index++)),
150 cosine_basis);
151 auto sine_term = SymbolicExpr::multiply(
152 SymbolicExpr::variable(
153 "C" + std::to_string(constant_index++)),
154 sine_basis);
155 solution = SymbolicExpr::add(
156 solution,
157 SymbolicExpr::add(cosine_term, sine_term));
158 continue;
159 }
160 for (int power = 0; power < root.multiplicity; ++power) {
161 auto basis = exponential;
162 if (power > 0) {
163 basis = SymbolicExpr::multiply(
164 SymbolicExpr::power(
165 variable, SymbolicExpr::number(power)),
166 exponential);
167 }
168 solution = SymbolicExpr::add(
169 solution,
170 SymbolicExpr::multiply(
171 SymbolicExpr::variable(
172 "C" + std::to_string(constant_index++)),
173 basis));
174 }
175 }
176 solution = solution->simplify();
177 if (dep_var_is_positive(y, ctx)) solution = make_abs(solution);
178 return Result<std::shared_ptr<SymbolicExpr>>::success(
179 std::move(solution));
180}

182std::shared_ptr<SymbolicExpr> solve_linear2_ode(
183 double a, double b, double c,
184 std::shared_ptr<SymbolicExpr> fx,
185 const std::string& x,
186 const std::string& y,
187 const AssumptionContext* ctx
188) {
189 if (!fx || !LMCAS::detail::node(fx)) {
190 throw std::invalid_argument(
191 "solve_linear2_ode: forcing expression must not be null");
192 }
193 if (!fx->is_zero()) {
194 throw std::logic_error(
195 "solve_linear2_ode: non-homogeneous case is outside the current support domain");
196 }
197 auto solution = solve_linear2_homogeneous(a, b, c, x, y, ctx);
198 if (!solution) {
199 if (solution.error().code == CasErrc::InvalidArgument) {
200 throw std::invalid_argument(solution.error().message);
201 }
202 if (solution.error().code == CasErrc::NumericFailure) {
203 throw std::overflow_error(solution.error().message);
204 }
205 throw std::logic_error(solution.error().message);
206 }
207 return std::move(solution.value());
208}

210Result<std::shared_ptr<SymbolicExpr>> solve_linear2_ode_checked(
211 double a, double b, double c,
212 std::shared_ptr<SymbolicExpr> fx,
213 const std::string& x,
214 const std::string& y,
215 ComputationContext& context,
216 const AssumptionContext* ctx
217) {
218 auto step = context.consume_steps(1, kSolveLinear2OdeOperation);
219 if (!step) return Result<std::shared_ptr<SymbolicExpr>>::failure(step.error());
220
221 if (!fx || !LMCAS::detail::node(fx)) {
222 return Result<std::shared_ptr<SymbolicExpr>>::failure(
223 CasErrc::InvalidArgument,
224 "forcing expression must not be null",
225 kSolveLinear2OdeOperation);
226 }
227 if (x.empty() || y.empty()) {
228 return Result<std::shared_ptr<SymbolicExpr>>::failure(
229 CasErrc::InvalidArgument,
230 "ODE variables must not be empty",
231 kSolveLinear2OdeOperation);
232 }
233
234 try {
235 if (!fx->is_zero()) {
236 return Result<std::shared_ptr<SymbolicExpr>>::failure(
237 CasErrc::Inconclusive,
238 "non-homogeneous second-order constant-coefficient ODEs are outside the current support domain",
239 kSolveLinear2OdeOperation);
240 }
241
242 return solve_linear2_homogeneous(a, b, c, x, y, ctx);
243 } catch (const std::invalid_argument& ex) {
244 return Result<std::shared_ptr<SymbolicExpr>>::failure(
245 CasErrc::InvalidArgument, ex.what(), kSolveLinear2OdeOperation);
246 } catch (const std::bad_alloc&) {
247 return Result<std::shared_ptr<SymbolicExpr>>::failure(
248 CasErrc::ResourceLimit,
249 "ODE solving allocation failed",
250 kSolveLinear2OdeOperation);
251 } catch (const std::logic_error& ex) {
252 return Result<std::shared_ptr<SymbolicExpr>>::failure(
253 CasErrc::Inconclusive, ex.what(), kSolveLinear2OdeOperation);
254 } catch (const std::exception& ex) {
255 return Result<std::shared_ptr<SymbolicExpr>>::failure(
256 CasErrc::InternalInvariant, ex.what(), kSolveLinear2OdeOperation);
257 }
258}
