37Result<OptionalRationalPolynomial> recognize_sum(
38 const AddNode& add, const std::string& variable, ComputationContext& context) {
39 Polynomial<Rational> result(variable);
40 for (const auto& operand : add.operands()) {
41 auto child = recognize_node(operand, variable, context);
42 if (!child) return child;
43 if (!child.value()) {
44 return Result<OptionalRationalPolynomial>::success(std::nullopt);
45 }
46 result = result + *child.value();
47 if (result.coeffs.size() > context.limits().max_expansion_terms) {
48 return Result<OptionalRationalPolynomial>::failure(
49 CasErrc::ResourceLimit, "polynomial term budget exhausted", kOperation);
50 }
51 }

55Result<OptionalRationalPolynomial> recognize_product(
56 const MultiplyNode& multiply, const std::string& variable, ComputationContext& context) {
57 Polynomial<Rational> result({Rational(1)}, variable);
58 for (const auto& operand : multiply.operands()) {
59 auto child = recognize_node(operand, variable, context);
60 if (!child) return child;
61 if (!child.value()) {
62 return Result<OptionalRationalPolynomial>::success(std::nullopt);
63 }
64 const std::size_t projected_terms = result.is_zero() || child.value()->is_zero()
65 ? 0
66 : result.coeffs.size() + child.value()->coeffs.size() - 1;

68 return Result<OptionalRationalPolynomial>::failure(
69 CasErrc::ResourceLimit,
70 "polynomial expansion term budget exhausted",
71 kOperation);

74 }

78Result<OptionalRationalPolynomial> recognize_power(
79 const PowerNode& power, const std::string& variable, ComputationContext& context) {
80 auto exponent_node = std::dynamic_pointer_cast<const NumberNode>(power.exponent());
81 if (!exponent_node || std::holds_alternative<lmmc_real_t>(exponent_node->value())) {
82 return Result<OptionalRationalPolynomial>::success(std::nullopt);
83 }
84
85 BigInt exponent;
86 if (std::holds_alternative<BigInt>(exponent_node->value())) {
87 exponent = std::get<BigInt>(exponent_node->value());
88 } else {
89 const Rational& rational = std::get<Rational>(exponent_node->value());
90 if (!rational.is_integer()) {
91 return Result<OptionalRationalPolynomial>::success(std::nullopt);
92 }
93 exponent = rational.to_bigint();
94 }
95
96 auto exponent_value = exponent.try_to_int64();
97 if (!exponent_value || *exponent_value <= 0) {
98 return Result<OptionalRationalPolynomial>::success(std::nullopt);
99 }
100 if (static_cast<std::uint64_t>(*exponent_value) >=
101 context.limits().max_expansion_terms) {
102 return Result<OptionalRationalPolynomial>::failure(
103 CasErrc::ResourceLimit,
104 "polynomial exponent exceeds the expansion term budget",
105 kOperation);
106 }
107
108 auto base = recognize_node(power.base(), variable, context);
109 if (!base) return base;
110 if (!base.value()) {
111 return Result<OptionalRationalPolynomial>::success(std::nullopt);
112 }
113
114 Polynomial<Rational> result({Rational(1)}, variable);
115 for (std::int64_t i = 0; i < *exponent_value; ++i) {
116 auto step = context.consume_steps(1, kOperation);
117 if (!step) return Result<OptionalRationalPolynomial>::failure(step.error());
118 const std::size_t projected_terms = result.is_zero() || base.value()->is_zero()
119 ? 0
120 : result.coeffs.size() + base.value()->coeffs.size() - 1;
121 if (projected_terms > context.limits().max_expansion_terms) {
122 return Result<OptionalRationalPolynomial>::failure(
123 CasErrc::ResourceLimit,
124 "polynomial expansion term budget exhausted",
125 kOperation);
126 }
127 result = result * *base.value();
128 }
129 return Result<OptionalRationalPolynomial>::success(std::move(result));
130}

132Result<OptionalRationalPolynomial> recognize_node(
133 const std::shared_ptr<const SymbolicNode>& node,
134 const std::string& variable,
135 ComputationContext& context) {
136 auto entered = context.enter_recursion(kOperation);
137 if (!entered) return Result<OptionalRationalPolynomial>::failure(entered.error());
138 RecursionScope scope(context);
139
140 if (!node) {
141 return Result<OptionalRationalPolynomial>::failure(
142 CasErrc::InternalInvariant,
143 "polynomial recognition encountered a null AST node",
144 kOperation);
145 }
146
147 if (auto number = std::dynamic_pointer_cast<const NumberNode>(node)) {
148 return recognize_number(*number, variable);
149 }
150 if (auto symbol = std::dynamic_pointer_cast<const VariableNode>(node)) {
151 if (symbol->is_constant() || symbol->name() != variable) {
152 return Result<OptionalRationalPolynomial>::success(std::nullopt);
153 }
154 return Result<OptionalRationalPolynomial>::success(
155 Polynomial<Rational>({Rational(0), Rational(1)}, variable));
156 }
157 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
158 return recognize_sum(*add, variable, context);
159 }
160 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
161 return recognize_product(*multiply, variable, context);
162 }
163 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
164 return recognize_power(*power, variable, context);
165 }
166 return Result<OptionalRationalPolynomial>::success(std::nullopt);
167}
