27/**
28 * @brief 计算二项式系数 C(n, k).
29 *
30 * 使用乘法公式 C(n, k) = n! / (k! * (n-k)!) 逐步计算;
31 * BigInt 承载全部中间结果,k = min(k, n-k) 缩短迭代路径.
32 *
33 * @param[in] n 上标(非负整数)
34 * @param[in] k 下标(0 <= k <= n)
35 * @return C(n, k) 的精确值
36 * @internal
37 */
38static BigInt hl_binomial(int n, int k) {
39 if (k < 0 || k > n) return BigInt(0);
40 if (k == 0 || k == n) return BigInt(1);
41
42 /// 利用对称性 C(n, k) = C(n, n-k)
43 if (k > n - k) {
44 k = n - k;
45 }
46
47 BigInt result(1);
48 for (int i = 0; i < k; ++i) {
49 result = result * BigInt(n - i);
50 result = result / BigInt(i + 1);
51 }
52 return result;
53}

55/**
56 * @brief 计算多项式系数的 L2 范数的平方.
57 *
58 * 即 ||f||_2^2 = sum(a_i^2),其中 a_i 为多项式各项系数.
59 *
60 * @param[in] poly 整系数多项式
61 * @return 系数平方和(BigInt 精确值)
62 * @internal
63 */
64static BigInt hl_l2_norm_squared(const Polynomial<BigInt>& poly) {
65 BigInt sum(0);
66 for (const auto& c : poly.coeffs) {
67 sum = sum + c * c;
68 }
69 return sum;
70}

72/**
73 * @brief 计算多项式的 Mignotte 界.
74 *
75 * 对于次数为 n 的多项式 f(x),其任意因子的系数绝对值上界为:
76 * B = C(n, floor(n/2)) * ||f||_2
77 *
78 * 其中 C(n, k) 为二项式系数,||f||_2 为系数向量的 L2 范数.
79 *
80 * 实现使用 B^2 = C(n, floor(n/2))^2 * ||f||_2^2 在整数域内计算,
81 * 最后由 BigInt::sqrt() 求向上取整的整数平方根.
82 *
83 * @param[in] poly 整系数多项式(非零,次数 >= 1)
84 * @return Mignotte 界 B
85 *
86 * @see Mignotte, M. "An inequality about factors of polynomials."
87 * Mathematics of Computation, 28(128), 1974.
88 * @internal
89 */
90static BigInt hl_mignotte_bound(const Polynomial<BigInt>& poly) {
91 int n = poly.degree();
92 if (n <= 0) return BigInt(1);
93
94 /// C(n, floor(n/2))
95 BigInt binom = hl_binomial(n, n / 2);
96
97 /// ||f||_2^2
98 BigInt norm_sq = hl_l2_norm_squared(poly);
99
100 /// B^2 = binom^2 * norm_sq
101 BigInt b_squared = binom * binom * norm_sq;
102
103 /// B = ceil(sqrt(B^2))
104 /// BigInt::sqrt() 返回 floor(sqrt(x)),需要检查是否精确
105 BigInt b = b_squared.sqrt();
106 if (b * b < b_squared) {
107 b = b + BigInt(1);
108 }
109
110 return b;
111}

113/**
114 * @brief 计算 Hensel 提升所需的高度 k.
115 *
116 * 提升高度 k 需满足 p^k > 2 * B * |lc(f)|,其中:
117 * - B 为 Mignotte 界
118 * - lc(f) 为多项式首项系数
119 * - p 为选定的素数
120 *
121 * 通过连续乘以 p 直至超过阈值确定 k,使界计算保持在整数域内.
122 *
123 * @param[in] poly 整系数多项式(非零,次数 >= 1)
124 * @param[in] prime 选定的素数 p
125 * @return 所需的提升高度 k(正整数)
126 * @internal
127 */
128static int hl_compute_lift_height(const Polynomial<BigInt>& poly, int64_t prime) {
129 if (poly.is_zero() || poly.degree() <= 0) return 1;
130
131 BigInt B = hl_mignotte_bound(poly);
132 BigInt lc = poly.lead_coeff().Abs();
133
134 /// 阈值 threshold = 2 * B * |lc(f)|
135 BigInt threshold = BigInt(2) * B * lc;
136
137 /// 计算最小 k 使得 p^k > threshold
138 BigInt p_power(1);
139 BigInt big_p(static_cast<long long>(prime));
140 int k = 0;
141
142 while (p_power <= threshold) {
143 p_power = p_power * big_p;
144 k++;
145 }
146
147 /// 至少提升 1 次
148 if (k < 1) k = 1;
149
150 return k;
151}

153/**
154 * @brief 将 BigInt 系数归约到对称表示 [-m/2, m/2).
155 *
156 * 对给定系数 c 和模数 m,计算 c mod m 并映射到 [-m/2, m/2) 区间.
157 * 这确保提升后的系数保持最小绝对值表示.
158 *
159 * @param[in] c 待归约的系数
160 * @param[in] m 模数(正整数)
161 * @return 对称表示下的归约值
162 * @internal
163 */
164static BigInt hl_symmetric_mod(const BigInt& c, const BigInt& m) {
165 if (m.is_zero()) return c;
166
167 BigInt r = c % m;
168 /// 确保 r 在 [0, m) 范围内
169 if (r.IsNegative()) {
170 r = r + m;
171 }
172
173 /// 映射到 [-m/2, m/2)
174 BigInt half_m = m / BigInt(2);
175 if (r > half_m) {
176 r = r - m;
177 }
178 return r;
179}

181/**
182 * @brief 对多项式系数向量执行模归约(对称表示).
183 *
184 * 将向量中每个系数归约到 [-m/2, m/2) 区间,并去除高次零系数.
185 *
186 * @param[in,out] poly 系数向量(就地修改)
187 * @param[in] m 模数
188 * @internal
189 */
190static void hl_reduce_coeffs(std::vector<BigInt>& poly, const BigInt& m) {
191 for (auto& c : poly) {
192 c = hl_symmetric_mod(c, m);
193 }
194 /// 去除高次零系数
195 while (!poly.empty() && poly.back().is_zero()) {
196 poly.pop_back();
197 }
198}

200/**
201 * @brief 多项式乘法(BigInt 系数),结果模 m 归约.
202 *
203 * 计算两个系数向量表示的多项式之积,所有系数归约到对称表示 [-m/2, m/2).
204 *
205 * @param[in] a 第一个多项式的系数向量
206 * @param[in] b 第二个多项式的系数向量
207 * @param[in] m 模数
208 * @return 乘积多项式的系数向量
209 * @internal
210 */
211static std::vector<BigInt> hl_poly_mul_mod(const std::vector<BigInt>& a,
212 const std::vector<BigInt>& b,
213 const BigInt& m) {
214 if (a.empty() || b.empty()) return {};
215
216 size_t n = a.size() + b.size() - 1;
217 std::vector<BigInt> result(n, BigInt(0));
218
219 for (size_t i = 0; i < a.size(); ++i) {
220 if (a[i].is_zero()) continue;
221 for (size_t j = 0; j < b.size(); ++j) {
222 if (b[j].is_zero()) continue;
223 result[i + j] = result[i + j] + a[i] * b[j];
224 }
225 }
226
227 hl_reduce_coeffs(result, m);
228 return result;
229}

231/**
232 * @brief 多项式减法(BigInt 系数),结果模 m 归约.
233 *
234 * @param[in] a 被减数多项式系数向量
235 * @param[in] b 减数多项式系数向量
236 * @param[in] m 模数
237 * @return 差多项式的系数向量
238 * @internal
239 */
240static std::vector<BigInt> hl_poly_sub_mod(const std::vector<BigInt>& a,
241 const std::vector<BigInt>& b,
242 const BigInt& m) {
243 size_t n = std::max(a.size(), b.size());
244 std::vector<BigInt> result(n, BigInt(0));
245
246 for (size_t i = 0; i < n; ++i) {
247 BigInt ai = (i < a.size()) ? a[i] : BigInt(0);
248 BigInt bi = (i < b.size()) ? b[i] : BigInt(0);
249 result[i] = ai - bi;
250 }
251
252 hl_reduce_coeffs(result, m);
253 return result;
254}

256/**
257 * @brief 多项式加法(BigInt 系数),结果模 m 归约.
258 *
259 * @param[in] a 第一个多项式系数向量
260 * @param[in] b 第二个多项式系数向量
261 * @param[in] m 模数
262 * @return 和多项式的系数向量
263 * @internal
264 */
265static std::vector<BigInt> hl_poly_add_mod(const std::vector<BigInt>& a,
266 const std::vector<BigInt>& b,
267 const BigInt& m) {
268 size_t n = std::max(a.size(), b.size());
269 std::vector<BigInt> result(n, BigInt(0));

281/**
282 * @brief 计算 BigInt 在模 m 下的乘法逆元.
283 *
284 * 使用扩展欧几里得算法求 a 在 Z/mZ 中的逆元.
285 * 要求 gcd(a, m) = 1.
286 *
287 * @param[in] a 待求逆的整数
288 * @param[in] m 模数
289 * @return a 的模逆元,归约到 [0, m)
290 * @throw std::domain_error 若 a 不可逆
291 * @internal
292 */
293static BigInt hl_mod_inverse(const BigInt& a, const BigInt& m) {
294 /// 扩展欧几里得:求 s 使得 a*s == 1 (mod m)
295 BigInt r0 = m, r1 = a % m;
296 if (r1.IsNegative()) r1 = r1 + m;
297
298 BigInt s0(0), s1(1);
299
300 while (!r1.is_zero()) {
301 BigInt q = r0 / r1;
302 BigInt r_new = r0 - q * r1;
303 BigInt s_new = s0 - q * s1;
304
305 r0 = r1;
306 r1 = r_new;
307 s0 = s1;
308 s1 = s_new;
309 }
310
311 /// r0 应为 gcd(a, m) = 1
312 if (r0 != BigInt(1) && r0 != BigInt(-1)) {
313 throw std::domain_error("hl_mod_inverse: element not invertible");
314 }
315
316 /// 若 gcd 为 -1,调整符号
317 if (r0.IsNegative()) {
318 s0 = s0.negate();
319 }
320
321 BigInt result = s0 % m;
322 if (result.IsNegative()) result = result + m;
323 return result;
324}

326/**
327 * @brief 多项式带余除法(BigInt 系数),模 m 下运算.
328 *
329 * 计算 a(x) = q(x) * b(x) + r(x),其中 deg(r) < deg(b).
330 * 要求 b 的首项系数在 Z/mZ 中可逆.
331 * 所有系数归约到对称表示.
332 *
333 * @param[in] a 被除数多项式系数向量
334 * @param[in] b 除数多项式系数向量
335 * @param[in] m 模数
336 * @return pair(商, 余数) 的系数向量
337 * @internal
338 */
339static std::pair<std::vector<BigInt>, std::vector<BigInt>>
340hl_poly_divmod(const std::vector<BigInt>& a,
341 const std::vector<BigInt>& b,
342 const BigInt& m) {
343 if (b.empty()) {
344 throw std::runtime_error("hl_poly_divmod: division by zero polynomial");
345 }
346
347 int deg_a = static_cast<int>(a.size()) - 1;
348 int deg_b = static_cast<int>(b.size()) - 1;
349
350 if (deg_a < deg_b) {
351 std::vector<BigInt> r = a;
352 hl_reduce_coeffs(r, m);
353 return {{}, r};
354 }
355
356 /// 首项系数的逆元
357 BigInt lc_inv = hl_mod_inverse(b.back(), m);
358
359 std::vector<BigInt> remainder = a;
360 std::vector<BigInt> quotient(deg_a - deg_b + 1, BigInt(0));
361
362 for (int i = deg_a; i >= deg_b; --i) {
363 BigInt coeff = hl_symmetric_mod(remainder[i], m);
364 if (coeff.is_zero()) continue;
365
366 BigInt factor = hl_symmetric_mod(coeff * lc_inv, m);
367 quotient[i - deg_b] = factor;
368
369 for (int j = 0; j <= deg_b; ++j) {
370 remainder[i - deg_b + j] = remainder[i - deg_b + j] - factor * b[j];
371 remainder[i - deg_b + j] = hl_symmetric_mod(remainder[i - deg_b + j], m);
372 }
373 }
374
375 /// 余数截断到 deg < deg_b
376 std::vector<BigInt> rem(remainder.begin(), remainder.begin() + deg_b);
377 hl_reduce_coeffs(quotient, m);
378 hl_reduce_coeffs(rem, m);
379 return {quotient, rem};
380}

382static bool hl_is_prime(int64_t value) {
383 if (value < 2) return false;
384 if (value == 2) return true;
385 if (value % 2 == 0) return false;
386 for (int64_t divisor = 3; divisor <= value / divisor; divisor += 2) {
387 if (value % divisor == 0) return false;
388 }
389 return true;
390}

392static bool hl_coeff_vectors_equal(std::vector<BigInt> left,
393 std::vector<BigInt> right) {
394 while (!left.empty() && left.back().is_zero()) left.pop_back();
395 while (!right.empty() && right.back().is_zero()) right.pop_back();
396 const std::size_t size = std::max(left.size(), right.size());
397 left.resize(size, BigInt(0));
398 right.resize(size, BigInt(0));
399 return left == right;
400}

402static Result<void> hl_validate_lift_inputs(
403 const Polynomial<BigInt>& poly,
404 const std::vector<std::vector<BigInt>>& factor_vecs,
405 int64_t prime) {
406 constexpr const char* operation = "hensel_lift";
407 if (!hl_is_prime(prime)) {
408 return Result<void>::failure(CasErrc::InvalidArgument,
409 "Hensel lifting requires a prime modulus",
410 operation);
411 }
412 if (poly.is_zero() || poly.degree() <= 0) {
413 return Result<void>::failure(CasErrc::InvalidArgument,
414 "Hensel lifting requires a non-constant polynomial",
415 operation);
416 }
417 if (factor_vecs.empty()) {
418 return Result<void>::failure(CasErrc::InvalidArgument,
419 "Hensel lifting requires at least one mod-p factor",
420 operation);
421 }
422 for (const auto& factor : factor_vecs) {
423 if (factor.empty()) {
424 return Result<void>::failure(CasErrc::InvalidArgument,
425 "Hensel lifting factor cannot be zero",
426 operation);
427 }
428 }
429
430 const BigInt modulus(static_cast<long long>(prime));
431 std::vector<BigInt> product{BigInt(1)};
432 for (const auto& factor : factor_vecs) {
433 product = hl_poly_mul_mod(product, factor, modulus);
434 }
435 std::vector<BigInt> reduced_f = poly.coeffs;
436 hl_reduce_coeffs(reduced_f, modulus);
437 if (!hl_coeff_vectors_equal(product, reduced_f)) {
438 return Result<void>::failure(CasErrc::InvalidArgument,
439 "mod-p factors do not multiply to the input polynomial",
440 operation);
441 }
442 return Result<void>::success();
443}

447/**
448 * @brief 执行一步二次 Hensel 提升:mod m -> mod m^2.
449 *
450 * 给定 f == g*h (mod m) 且 s*g + t*h == 1 (mod m),
451 * 计算 g', h', s', t' 使得 f == g'*h' (mod m^2) 且 s'*g' + t'*h' == 1 (mod m^2).
452 *
453 * 算法步骤:
454 * 1. 计算误差 e = f - g*h(所有系数应被 m 整除)
455 * 2. 计算修正项:(q, r) = divmod(s*e, h) over Z/m^2Z
456 * - h' = h + r (mod m^2)
457 * - g' = g + t*e + q*g (mod m^2)
458 * 3. 更新 Bezout 系数:
459 * - b = s*g' + t*h' - 1 (mod m^2)
460 * - (c, d) = divmod(s*b, h') over Z/m^2Z
461 * - s' = s - d (mod m^2)
462 * - t' = t - t*b - c*g' (mod m^2)
463 *
464 * @param[in] f 原始多项式系数向量
465 * @param[in] current 当前提升状态(g, h, s, t, modulus=m)
466 * @return 提升后的状态(g', h', s', t', modulus=m^2)
467 *
468 * @pre f == g*h (mod m)
469 * @pre s*g + t*h == 1 (mod m)
470 * @pre deg(s) < deg(h), deg(t) < deg(g)
471 *
472 * @see Zassenhaus, H. "On Hensel factorization, I."
473 * Journal of Number Theory, 1(3), 1969.
474 * @internal
475 */
476HenselLiftPair hl_two_factor_lift(
477 const std::vector<BigInt>& f,
478 const HenselLiftPair& current) {
479
480 const auto& g = current.g;
481 const auto& h = current.h;
482 const auto& s = current.s;
483 const auto& t = current.t;
484 const BigInt& m = current.modulus;
485
486 /// 新模数 m^2 = m * m
487 BigInt m2 = m * m;
488
489 /// 步骤 1:计算误差 e = f - g*h (mod m^2)
490 std::vector<BigInt> gh = hl_poly_mul_mod(g, h, m2);
491 std::vector<BigInt> e = hl_poly_sub_mod(f, gh, m2);
492
493 /// 步骤 2:计算修正项
494 /// se = s * e (mod m^2)
495 std::vector<BigInt> se = hl_poly_mul_mod(s, e, m2);
496
497 /// (q, r) = divmod(se, h) over Z/m^2Z,其中 deg(r) < deg(h)
498 auto [q, r] = hl_poly_divmod(se, h, m2);
499
500 /// g' = g + t*e + q*g (mod m^2)
501 std::vector<BigInt> te = hl_poly_mul_mod(t, e, m2);
502 std::vector<BigInt> qg = hl_poly_mul_mod(q, g, m2);
503 std::vector<BigInt> g_new = hl_poly_add_mod(g, hl_poly_add_mod(te, qg, m2), m2);
504 hl_reduce_coeffs(g_new, m2);
505
506 /// h' = h + r (mod m^2)
507 std::vector<BigInt> h_new = hl_poly_add_mod(h, r, m2);
508 hl_reduce_coeffs(h_new, m2);
509
510 /// 步骤 3:更新 Bezout 系数
511 /// b = s*g' + t*h' - 1 (mod m^2)
512 std::vector<BigInt> sg_new = hl_poly_mul_mod(s, g_new, m2);
513 std::vector<BigInt> th_new = hl_poly_mul_mod(t, h_new, m2);
514 std::vector<BigInt> sg_plus_th = hl_poly_add_mod(sg_new, th_new, m2);
515
516 /// 减去常数 1
517 std::vector<BigInt> one_poly = {BigInt(1)};
518 std::vector<BigInt> b = hl_poly_sub_mod(sg_plus_th, one_poly, m2);
519
520 /// (c, d) = divmod(s*b, h') over Z/m^2Z
521 std::vector<BigInt> sb = hl_poly_mul_mod(s, b, m2);
522 auto [c, d] = hl_poly_divmod(sb, h_new, m2);
523
524 /// s' = s - d (mod m^2)
525 std::vector<BigInt> s_new = hl_poly_sub_mod(s, d, m2);
526 hl_reduce_coeffs(s_new, m2);
527
528 /// t' = t - t*b - c*g' (mod m^2)
529 std::vector<BigInt> tb = hl_poly_mul_mod(t, b, m2);
530 std::vector<BigInt> cg_new = hl_poly_mul_mod(c, g_new, m2);
531 std::vector<BigInt> t_new = hl_poly_sub_mod(t, hl_poly_add_mod(tb, cg_new, m2), m2);
532 hl_reduce_coeffs(t_new, m2);
533
534 return HenselLiftPair{g_new, h_new, s_new, t_new, m2};
535}

537/**
538 * @brief 计算两个多项式在模 m 下的扩展 GCD 的 Bezout 系数.
539 *
540 * 给定互素多项式 g, h(mod m),求 s, t 使得 s*g + t*h == 1 (mod m),
541 * 且 deg(s) < deg(h), deg(t) < deg(g).
542 *
543 * 使用扩展欧几里得算法在 Z/mZ[x] 上运算.
544 *
545 * @param[in] g 第一个多项式系数向量
546 * @param[in] h 第二个多项式系数向量
547 * @param[in] m 模数
548 * @return pair(s, t) 满足 s*g + t*h == 1 (mod m)
549 * @internal
550 */
551static std::pair<std::vector<BigInt>, std::vector<BigInt>>
552hl_extended_gcd_poly(const std::vector<BigInt>& g,
553 const std::vector<BigInt>& h,
554 const BigInt& m) {
555 /// 扩展欧几里得算法:r0 = g, r1 = h
556 /// s0*g + t0*h = r0, s1*g + t1*h = r1
557 std::vector<BigInt> r0 = g, r1 = h;
558 std::vector<BigInt> s0 = {BigInt(1)}, s1 = {}; // s0=1, s1=0
559 std::vector<BigInt> t0 = {}, t1 = {BigInt(1)}; // t0=0, t1=1
560
561 while (!r1.empty()) {
562 auto [q, r] = hl_poly_divmod(r0, r1, m);
563
564 /// r_new = r0 - q*r1
565 std::vector<BigInt> r_new = r;
566
567 /// s_new = s0 - q*s1
568 std::vector<BigInt> qs1 = hl_poly_mul_mod(q, s1, m);
569 std::vector<BigInt> s_new = hl_poly_sub_mod(s0, qs1, m);
570
571 /// t_new = t0 - q*t1
572 std::vector<BigInt> qt1 = hl_poly_mul_mod(q, t1, m);
573 std::vector<BigInt> t_new = hl_poly_sub_mod(t0, qt1, m);
574
575 r0 = r1; r1 = r_new;
576 s0 = s1; s1 = s_new;
577 t0 = t1; t1 = t_new;
578 }
579
580 /// r0 = gcd,应为常数(可逆元)
581 /// 归一化使 gcd = 1
582 if (!r0.empty() && r0[0] != BigInt(1)) {
583 BigInt inv = hl_mod_inverse(r0[0], m);
584 for (auto& c : s0) {
585 c = hl_symmetric_mod(c * inv, m);
586 }
587 for (auto& c : t0) {
588 c = hl_symmetric_mod(c * inv, m);
589 }
590 }
591
592 /// 去除高次零系数
593 while (!s0.empty() && s0.back().is_zero()) s0.pop_back();
594 while (!t0.empty() && t0.back().is_zero()) t0.pop_back();
595
596 return {s0, t0};
597}

599/**
600 * @brief 多因子 Hensel 提升:将 k 个模 p 因子同时提升到 mod p^target_k.
601 *
602 * 使用顺序剥离策略:
603 * 1. 将 f 分解为 g_1 和 h_1 = g_2*g₃*...*gₖ
604 * 2. 对 (g_1, h_1) 执行二因子提升
605 * 3. 递归对 h_1 继续剥离下一个因子
606 *
607 * 每步提升通过反复调用 hl_two_factor_lift 实现二次提升(mod p -> p^2 -> p^4 -> ...),
608 * 直到模数达到或超过 p^target_k.
609 *
610 * @param[in] f 原始多项式系数向量(升幂排列)
611 * @param[in] factors 模 p 下的因子列表(各因子系数向量)
612 * @param[in] prime 素数 p
613 * @param[in] target_k 目标提升高度(提升到 mod p^target_k)
614 * @return 提升后的因子列表(各因子系数在对称表示下)
615 *
616 * @pre f == factors[0] * factors[1] * ... * factors[k-1] (mod p)
617 * @pre 各因子两两互素 (mod p)
618 *
619 * @see Zassenhaus, H. "On Hensel factorization, I."
620 * Journal of Number Theory, 1(3), 1969.
621 * @internal
622 */
623static std::vector<std::vector<BigInt>> hl_multi_factor_lift(
624 const std::vector<BigInt>& f,
625 const std::vector<std::vector<BigInt>>& factors,
626 int64_t prime,
627 int target_k) {
628
629 if (factors.empty()) return {};
630 if (factors.size() == 1) {
631 /// 单因子:直接归约到目标模数
632 BigInt target_mod(1);
633 BigInt big_p(static_cast<long long>(prime));
634 for (int i = 0; i < target_k; ++i) {
635 target_mod = target_mod * big_p;
636 }
637 std::vector<BigInt> result = f;
638 hl_reduce_coeffs(result, target_mod);
639 return {result};
640 }
641
642 BigInt big_p(static_cast<long long>(prime));
643 BigInt initial_mod(static_cast<long long>(prime));
644
645 /// 计算目标模数 p^target_k
646 BigInt target_mod(1);
647 for (int i = 0; i < target_k; ++i) {
648 target_mod = target_mod * big_p;
649 }
650
651 /// 顺序剥离策略:逐个提升因子
652 std::vector<std::vector<BigInt>> lifted_factors;
653 lifted_factors.reserve(factors.size());
654
655 /// 当前待分解的多项式(初始为 f)
656 std::vector<BigInt> remaining = f;
657 hl_reduce_coeffs(remaining, target_mod);
658
659 /// 剩余因子列表
660 std::vector<std::vector<BigInt>> remaining_factors = factors;
661
662 for (size_t i = 0; i < factors.size() - 1; ++i) {
663 /// 取出第一个因子 g
664 std::vector<BigInt> g = remaining_factors[0];
665
666 /// 计算 h = 剩余因子的乘积 (mod p)
667 std::vector<BigInt> h = remaining_factors[1];
668 for (size_t j = 2; j < remaining_factors.size(); ++j) {
669 h = hl_poly_mul_mod(h, remaining_factors[j], initial_mod);
670 }
671
672 /// 计算 Bezout 系数 s, t 使得 s*g + t*h == 1 (mod p)
673 auto [s, t] = hl_extended_gcd_poly(g, h, initial_mod);
674
675 /// 构造初始提升状态
676 HenselLiftPair state{g, h, s, t, initial_mod};
677
678 /// 反复二次提升直到模数 >= target_mod
679 while (state.modulus < target_mod) {
680 state = hl_two_factor_lift(remaining, state);
681 }
682
683 /// 提取提升后的 g(已提升到 target_mod)
684 std::vector<BigInt> lifted_g = state.g;
685 hl_reduce_coeffs(lifted_g, target_mod);
686 lifted_factors.push_back(lifted_g);
687
688 /// 更新 remaining 为提升后的 h
689 remaining = state.h;
690 hl_reduce_coeffs(remaining, target_mod);
691
692 /// 更新 remaining_factors:去掉第一个,后续因子保持不变
693 remaining_factors.erase(remaining_factors.begin());
694 }
695
696 /// 最后一个因子就是 remaining
697 hl_reduce_coeffs(remaining, target_mod);
698 lifted_factors.push_back(remaining);
699
700 return lifted_factors;
701}

704/**
705 * @brief Hensel 提升:将模 p 因子提升到 mod p^k.
706 *
707 * 将 Berlekamp 分解得到的模 p 不可约因子通过二次 Hensel 提升
708 * 升至 mod p^k,其中 k 由 Mignotte 界确定.
709 *
710 * 流程:
711 * 1. 将 Polynomial<ModInt> 因子转换为 vector<BigInt> 内部表示
712 * 2. 调用 hl_multi_factor_lift 执行多因子提升
713 * 3. 将结果转换回 Polynomial<BigInt>
714 *
715 * @param[in] poly 整系数多项式
716 * @param[in] mod_factors 模 p 下的不可约因子
717 * @param[in] prime 素数 p
718 * @param[in] lift_bound 提升次数上界 k(若为 0 则自动计算)
719 * @return 提升后的整系数因子列表(系数在对称表示 [-p^k/2, p^k/2] 下)
720 */
721HenselLiftResult hensel_lift_checked(
722 const Polynomial<BigInt>& poly,
723 const std::vector<Polynomial<ModInt>>& mod_factors,
724 int64_t prime,
725 int lift_bound) {
726 constexpr const char* operation = "hensel_lift";
727
728 /// 将原始多项式转换为系数向量
729 std::vector<BigInt> f_vec = poly.coeffs;
730
731 /// 将 Polynomial<ModInt> 因子转换为 vector<BigInt> 表示
732 std::vector<std::vector<BigInt>> factor_vecs;
733 factor_vecs.reserve(mod_factors.size());
734 for (const auto& mf : mod_factors) {
735 std::vector<BigInt> fv;
736 fv.reserve(mf.coeffs.size());
737 for (const auto& c : mf.coeffs) {
738 fv.push_back(BigInt(static_cast<long long>(c.value())));
739 }
740 factor_vecs.push_back(std::move(fv));
741 }
742
743 auto valid = hl_validate_lift_inputs(poly, factor_vecs, prime);
744 if (!valid) return HenselLiftResult::failure(valid.error());
745
746 /// 若调用方未指定提升界,则自动计算
747 int k = lift_bound;
748 if (k <= 0) {
749 k = hl_compute_lift_height(poly, prime);
750 }
751
752 /// 执行多因子 Hensel 提升
753 std::vector<std::vector<BigInt>> lifted_vecs;
754 try {
755 lifted_vecs = hl_multi_factor_lift(f_vec, factor_vecs, prime, k);
756 } catch (const std::exception& ex) {
757 return HenselLiftResult::failure(CasErrc::InternalInvariant,
758 ex.what(),
759 operation);
760 }
761
762 /// 将结果转换回 Polynomial<BigInt>
763 std::vector<Polynomial<BigInt>> result;
764 result.reserve(lifted_vecs.size());
765 for (auto& lv : lifted_vecs) {
766 result.emplace_back(std::move(lv), poly.variable_name);
767 }
768
769 return HenselLiftResult::success(std::move(result));
770}
