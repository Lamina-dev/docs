9InferenceTriboolResult InferenceEngine::query_periodic_checked(const SymbolicExpr& expr) const {
10 return checked_inference_result<Tribool>(expr, "query_periodic_checked",
11 [&]() {
12 // FunctionNode: sin, cos, tan are periodic
13 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(expr))) {
14 switch (func->type()) {
15 case FunctionNode::FuncType::Sin:
16 case FunctionNode::FuncType::Cos:
17 case FunctionNode::FuncType::Tan:
18 return Tribool::True;
19 case FunctionNode::FuncType::ArcTan: {
20 if (func->arguments().size() != 1) {
21 return Tribool::Unknown;
22 }
23 auto argument = lamina::detail::expression_from_node(
24 func->arguments()[0]);
25 if (auto variable =
26 std::dynamic_pointer_cast<const VariableNode>(
27 func->arguments()[0])) {
28 const auto& properties =
29 impl_->ctx.current_properties();
30 return properties.is_periodic(variable->name())
31 ? Tribool::True
32 : Tribool::False;
33 }
34 auto periodic = query_periodic_checked(argument);
35 if (periodic &&
36 periodic.value() == Tribool::True) {
37 return Tribool::True;
38 }
39 return Tribool::Unknown;
40 }
41 default:
42 break;
43 }
44 }
45
46 // VariableNode: check PropertyStore for declared periodicity
47 if (auto var = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(expr))) {
48 const auto& props = impl_->ctx.current_properties();
49 if (props.is_periodic(var->name())) return Tribool::True;
50 }
51 return Tribool::Unknown;
52 });
53}

56InferencePeriodResult InferenceEngine::infer_period_checked(const SymbolicExpr& expr) const {
57 return checked_inference_result<std::optional<SymbolicExpr>>(
58 expr, "infer_period_checked", [&]() -> std::optional<SymbolicExpr> {
59 // FunctionNode: known periods for trig functions
60 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(expr))) {
61 switch (func->type()) {
62 case FunctionNode::FuncType::Sin:
63 case FunctionNode::FuncType::Cos: {
64 // Period = 2*pi
65 auto two = lamina::detail::make_node<NumberNode>(static_cast<lmmc_real_t>(2.0));
66 auto pi_val = lamina::detail::make_node<NumberNode>(static_cast<lmmc_real_t>(LMMC_CONST_PI));
67 auto two_pi = lamina::detail::make_node<MultiplyNode>(
68 std::vector<std::shared_ptr<const SymbolicNode>>{two, pi_val});
69 auto period = lamina::detail::expression_from_node(two_pi);
70 return period;
71 }
72 case FunctionNode::FuncType::Tan: {
73 // Period = pi
74 auto pi_val = lamina::detail::make_node<NumberNode>(static_cast<lmmc_real_t>(LMMC_CONST_PI));
75 auto period = lamina::detail::expression_from_node(pi_val);
76 return period;
77 }
78 default:
79 break;
80 }
81 }
82
83 // VariableNode: check PropertyStore for declared period
84 if (auto var = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(expr))) {
85 const auto& props = impl_->ctx.current_properties();
86 auto stored_period = props.get_period(var->name());
87 if (stored_period.has_value() && *stored_period) {
88 return *(*stored_period); // dereference optional<shared_ptr<SymbolicExpr>>
89 }
90 }
91 return std::nullopt;
92 });
93}

97/**
98 * @brief Helper: check if a MultiplyNode represents negation (multiplication by -1).
99 * Returns the inner expression if it's a negation, nullptr otherwise.
100 */
101static std::shared_ptr<const SymbolicNode> detect_negation(const std::shared_ptr<const SymbolicNode>& node) {
102 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node);
103 if (!mul || mul->operands().size() != 2) return nullptr;
104
105 // Check if one operand is -1
106 for (size_t i = 0; i < 2; ++i) {
107 auto num = std::dynamic_pointer_cast<const NumberNode>(mul->operands()[i]);
108 if (!num) continue;
109
110 bool is_neg_one = false;
111 if (std::holds_alternative<BigInt>(num->value())) {
112 is_neg_one = (std::get<BigInt>(num->value()) == BigInt(-1));
113 } else if (std::holds_alternative<Rational>(num->value())) {
114 is_neg_one = (std::get<Rational>(num->value()) == Rational(-1));
115 } else if (std::holds_alternative<lmmc_real_t>(num->value())) {
116 is_neg_one = (std::get<lmmc_real_t>(num->value()) == -1.0);
117 }
118
119 if (is_neg_one) {
120 return mul->operands()[1 - i]; // return the other operand
121 }
122 }
123 return nullptr;
124}

126Monotonicity InferenceEngine::infer_monotonicity(const SymbolicExpr& expr,
127 const std::string& var,
128 const Interval& interval) const {
129 if (!lamina::detail::node(expr)) return Monotonicity::Unknown;
130
131 // FunctionNode: auto-infer for known functions
132 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(expr))) {
133 // Check that the function argument is the queried variable
134 if (func->arguments().empty()) return Monotonicity::Unknown;
135 auto arg_var = std::dynamic_pointer_cast<const VariableNode>(func->arguments()[0]);
136 if (!arg_var || arg_var->name() != var) return Monotonicity::Unknown;
137
138 switch (func->type()) {
139 case FunctionNode::FuncType::Exp:
140 // exp is strictly increasing on all of ℝ
141 return Monotonicity::Increasing;
142
143 case FunctionNode::FuncType::ArcTan:
144 // atan 在整个实数域严格递增.
145 return Monotonicity::Increasing;
146
147 case FunctionNode::FuncType::Ln: {
148 // ln is strictly increasing on ℝ⁺ (positive reals)
149 // Check if the interval is within positive reals
150 // If the lower bound is > 0 (or open at 0), ln is increasing
151 auto lo = endpoint_to_double(interval.lower);
152 if (lo.has_value() && *lo > 0.0) {
153 return Monotonicity::Increasing;
154 }
155 // If lower is open at 0, also increasing
156 if (lo.has_value() && *lo == 0.0 && interval.lower.is_open) {
157 return Monotonicity::Increasing;
158 }
159 // If interval is (0, +inf) or similar positive interval
160 if (interval.lower.is_neg_infinity) {
161 return Monotonicity::Unknown; // ln not defined on negative reals
162 }
163 return Monotonicity::Unknown;
164 }
165
166 default:
167 break;
168 }
169 }
170
171 // Negation: multiply by -1 reverses monotonicity
172 if (auto inner = detect_negation(lamina::detail::node(expr))) {
173 auto inner_expr = lamina::detail::expression_from_node(inner);
174 Monotonicity inner_mono = infer_monotonicity(inner_expr, var, interval);
175 switch (inner_mono) {
176 case Monotonicity::Increasing: return Monotonicity::Decreasing;
177 case Monotonicity::Decreasing: return Monotonicity::Increasing;
178 case Monotonicity::NonDecreasing: return Monotonicity::NonIncreasing;
179 case Monotonicity::NonIncreasing: return Monotonicity::NonDecreasing;
180 default: return Monotonicity::Unknown;
181 }
182 }
183
184 // VariableNode: check PropertyStore for declared monotonicity
185 if (auto var_node = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(expr))) {
186 const auto& props = impl_->ctx.current_properties();
187 return detail::propagate_result(
188 props.get_monotonicity(var_node->name(), var, interval));
189 }
190
191 return Monotonicity::Unknown;
192}

196/**
197 * @brief Helper: collect positive integer exponents that appear in PowerNode
198 * expressions within the RelationStore involving the given variable names.
199 */
200static std::vector<int> collect_power_exponents(const RelationStore& store,
201 const std::string& lhs_name,
202 const std::string& rhs_name) {
203 std::unordered_set<int> exponents;
204
205 auto check_node = [&](const std::shared_ptr<const SymbolicNode>& node) {
206 if (!node) return;
207 auto pow_node = std::dynamic_pointer_cast<const PowerNode>(node);
208 if (!pow_node) return;
209
210 // Check if the base is one of our variables
211 auto base_var = std::dynamic_pointer_cast<const VariableNode>(pow_node->base());
212 if (!base_var) return;
213 if (base_var->name() != lhs_name && base_var->name() != rhs_name) return;
214
215 // Check if the exponent is a positive integer
216 auto exp_num = std::dynamic_pointer_cast<const NumberNode>(pow_node->exponent());
217 if (!exp_num) return;
218 if (!is_positive_integer_number(*exp_num)) return;
219
220 // Extract the integer value
221 int n = 0;
222 if (std::holds_alternative<BigInt>(exp_num->value())) {
223 const auto& b = std::get<BigInt>(exp_num->value());
224 // Only consider small exponents to avoid explosion
225 if (b > BigInt(100)) return;
226 n = static_cast<int>(b.IsNegative() ? 0 : b.is_zero() ? 0 : std::stoi(b.to_string()));
227 } else if (std::holds_alternative<Rational>(exp_num->value())) {
228 BigInt num_val = std::get<Rational>(exp_num->value()).get_numerator();
229 if (num_val > BigInt(100)) return;
230 n = static_cast<int>(std::stoi(num_val.to_string()));
231 } else {
232 double v = std::get<lmmc_real_t>(exp_num->value());
233 if (v <= 0.0 || v > 100.0 || v != std::floor(v)) return;
234 n = static_cast<int>(v);
235 }
236
237 if (n > 0) {
238 exponents.insert(n);
239 }
240 };
241
242 // Scan all relations in the store for PowerNode expressions
243 for (const auto& rel : store.get_relations()) {
244 check_node(lamina::detail::node(rel.lhs));
245 check_node(lamina::detail::node(rel.rhs));
246 }
247
248 return std::vector<int>(exponents.begin(), exponents.end());
249}

256Result<void> InferenceEngine::apply_monotonicity_rules_checked(
257 const Relation& rel,
258 RelationStore& store,
259 PropertyStore& prop_store,
260 int depth) {
261 // Stop recursion at maximum depth
262 if (depth >= MAX_MONOTONICITY_DEPTH) return Result<void>::success();
263
264 // Only apply to GT (greater-than) relations
265 if (rel.op != RelationalNode::Op::GT) return Result<void>::success();
266
267 // Extract LHS and RHS — both must be single VariableNodes
268 auto lhs_var = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(rel.lhs));
269 auto rhs_var = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(rel.rhs));
270 if (!lhs_var || !rhs_var) return Result<void>::success();
271
272 const std::string& x_name = lhs_var->name();
273 const std::string& y_name = rhs_var->name();
274
275 // Helper to add a deduced relation and recursively apply monotonicity rules
276 auto add_deduced = [&](const SymbolicExpr& new_lhs,
277 const SymbolicExpr& new_rhs) -> Result<void> {
278 // Don't add if already present
279 if (store.has_relation(new_lhs, new_rhs, RelationalNode::Op::GT)) {
280 return Result<void>::success();
281 }
282
283 auto inserted = store.add_relation_checked(
284 new_lhs, new_rhs, RelationalNode::Op::GT, prop_store);
285 if (!inserted.has_value()) {
286 return Result<void>::failure(inserted.error());
287 }
288
289 // Recursively apply monotonicity rules to the newly added relation
290 Relation new_rel{new_lhs, new_rhs, RelationalNode::Op::GT};
291 return apply_monotonicity_rules_checked(new_rel, store, prop_store, depth + 1);
292 };
293
294 // Check domain assumptions using the AssumptionContext (read-through all scopes)
295 bool both_positive = impl_->ctx.has_sign(x_name, Sign::Positive) &&
296 impl_->ctx.has_sign(y_name, Sign::Positive);
297 bool both_real = impl_->ctx.has_domain(x_name, Domain::Real) &&
298 impl_->ctx.has_domain(y_name, Domain::Real);
299 bool both_nonnegative = impl_->ctx.has_sign(x_name, Sign::NonNegative) &&
300 impl_->ctx.has_sign(y_name, Sign::NonNegative);
301
302 if (both_positive) {
303 auto ln_x_node = lamina::detail::make_node<FunctionNode>(
304 FunctionNode::FuncType::Ln,
305 std::vector<std::shared_ptr<const SymbolicNode>>{lhs_var->clone()});
306 auto ln_y_node = lamina::detail::make_node<FunctionNode>(
307 FunctionNode::FuncType::Ln,
308 std::vector<std::shared_ptr<const SymbolicNode>>{rhs_var->clone()});
309
310 auto ln_x = lamina::detail::expression_from_node(ln_x_node);
311 auto ln_y = lamina::detail::expression_from_node(ln_y_node);
312 auto deduced = add_deduced(ln_x, ln_y);
313 if (!deduced.has_value()) return deduced;
314 }
315
316 if (both_positive) {
317 auto sqrt_x_node = lamina::detail::make_node<FunctionNode>(
318 FunctionNode::FuncType::Sqrt,
319 std::vector<std::shared_ptr<const SymbolicNode>>{lhs_var->clone()});
320 auto sqrt_y_node = lamina::detail::make_node<FunctionNode>(
321 FunctionNode::FuncType::Sqrt,
322 std::vector<std::shared_ptr<const SymbolicNode>>{rhs_var->clone()});
323
324 auto sqrt_x = lamina::detail::expression_from_node(sqrt_x_node);
325 auto sqrt_y = lamina::detail::expression_from_node(sqrt_y_node);
326 auto deduced = add_deduced(sqrt_x, sqrt_y);
327 if (!deduced.has_value()) return deduced;
328 }
329
330 if (both_real) {
331 auto exp_x_node = lamina::detail::make_node<FunctionNode>(
332 FunctionNode::FuncType::Exp,
333 std::vector<std::shared_ptr<const SymbolicNode>>{lhs_var->clone()});
334 auto exp_y_node = lamina::detail::make_node<FunctionNode>(
335 FunctionNode::FuncType::Exp,
336 std::vector<std::shared_ptr<const SymbolicNode>>{rhs_var->clone()});
337
338 auto exp_x = lamina::detail::expression_from_node(exp_x_node);
339 auto exp_y = lamina::detail::expression_from_node(exp_y_node);
340 auto deduced = add_deduced(exp_x, exp_y);
341 if (!deduced.has_value()) return deduced;
342 }
343
344 if (both_nonnegative) {
345 std::vector<int> exponents = collect_power_exponents(store, x_name, y_name);
346
347 for (int n : exponents) {
348 auto exp_node = lamina::detail::make_node<NumberNode>(BigInt(n));
349 auto pow_x_node = lamina::detail::make_node<PowerNode>(lhs_var->clone(), exp_node->clone());
350 auto pow_y_node = lamina::detail::make_node<PowerNode>(rhs_var->clone(), exp_node->clone());
351
352 auto pow_x = lamina::detail::expression_from_node(pow_x_node);
353 auto pow_y = lamina::detail::expression_from_node(pow_y_node);
354 auto deduced = add_deduced(pow_x, pow_y);
355 if (!deduced.has_value()) return deduced;
356 }
357 }
358
359 return Result<void>::success();
360}
