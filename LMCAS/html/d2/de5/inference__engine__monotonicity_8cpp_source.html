12InferenceTriboolResult InferenceEngine::query_periodic_checked(const SymbolicExpr& expr) const {
13 return checked_inference_result<Tribool>(expr, "query_periodic_checked",
14 [&]() {
15 // FunctionNode: sin, cos, tan are periodic
16 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(expr))) {
17 switch (func->type()) {
18 case FunctionNode::FuncType::Sin:
19 case FunctionNode::FuncType::Cos:
20 case FunctionNode::FuncType::Tan:
21 return Tribool::True;
22 default:
23 break;
24 }
25 }
26
27 // VariableNode: check PropertyStore for declared periodicity
28 if (auto var = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(expr))) {
29 const auto& props = impl_->ctx.current_properties();
30 if (props.is_periodic(var->name())) return Tribool::True;
31 }
32 return Tribool::Unknown;
33 });

40InferencePeriodResult InferenceEngine::infer_period_checked(const SymbolicExpr& expr) const {
41 return checked_inference_result<std::optional<SymbolicExpr>>(
42 expr, "infer_period_checked", [&]() -> std::optional<SymbolicExpr> {
43 // FunctionNode: known periods for trig functions
44 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(expr))) {
45 switch (func->type()) {
46 case FunctionNode::FuncType::Sin:
47 case FunctionNode::FuncType::Cos: {
48 // Period = 2*pi
49 auto two = lamina::detail::make_node<NumberNode>(static_cast<lmmc_real_t>(2.0));
50 auto pi_val = lamina::detail::make_node<NumberNode>(static_cast<lmmc_real_t>(LMMC_CONST_PI));
51 auto two_pi = lamina::detail::make_node<MultiplyNode>(
52 std::vector<std::shared_ptr<const SymbolicNode>>{two, pi_val});
53 auto period = lamina::detail::expression_from_node(two_pi);
54 return period;
55 }
56 case FunctionNode::FuncType::Tan: {
57 // Period = pi
58 auto pi_val = lamina::detail::make_node<NumberNode>(static_cast<lmmc_real_t>(LMMC_CONST_PI));
59 auto period = lamina::detail::expression_from_node(pi_val);
60 return period;
61 }
62 default:
63 break;
64 }
65 }
66
67 // VariableNode: check PropertyStore for declared period
68 if (auto var = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(expr))) {
69 const auto& props = impl_->ctx.current_properties();
70 auto stored_period = props.get_period(var->name());
71 if (stored_period.has_value() && *stored_period) {
72 return *(*stored_period); // dereference optional<shared_ptr<SymbolicExpr>>
73 }
74 }
75 return std::nullopt;
76 });
77}

81/**
82 * @brief Helper: check if a MultiplyNode represents negation (multiplication by -1).
83 * Returns the inner expression if it's a negation, nullptr otherwise.
84 */
85static std::shared_ptr<const SymbolicNode> detect_negation(const std::shared_ptr<const SymbolicNode>& node) {
86 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node);
87 if (!mul || mul->operands().size() != 2) return nullptr;
88
89 // Check if one operand is -1
90 for (size_t i = 0; i < 2; ++i) {
91 auto num = std::dynamic_pointer_cast<const NumberNode>(mul->operands()[i]);
92 if (!num) continue;
93
94 bool is_neg_one = false;
95 if (std::holds_alternative<BigInt>(num->value())) {
96 is_neg_one = (std::get<BigInt>(num->value()) == BigInt(-1));
97 } else if (std::holds_alternative<Rational>(num->value())) {
98 is_neg_one = (std::get<Rational>(num->value()) == Rational(-1));
99 } else if (std::holds_alternative<lmmc_real_t>(num->value())) {
100 is_neg_one = (std::get<lmmc_real_t>(num->value()) == -1.0);
101 }
102
103 if (is_neg_one) {
104 return mul->operands()[1 - i]; // return the other operand
105 }
106 }
107 return nullptr;
108}

110Monotonicity InferenceEngine::infer_monotonicity(const SymbolicExpr& expr,
111 const std::string& var,
112 const Interval& interval) const {
113 if (!lamina::detail::node(expr)) return Monotonicity::Unknown;
114
115 // FunctionNode: auto-infer for known functions
116 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(expr))) {
117 // Check that the function argument is the queried variable
118 if (func->arguments().empty()) return Monotonicity::Unknown;
119 auto arg_var = std::dynamic_pointer_cast<const VariableNode>(func->arguments()[0]);
120 if (!arg_var || arg_var->name() != var) return Monotonicity::Unknown;
121
122 switch (func->type()) {
123 case FunctionNode::FuncType::Exp:
124 // exp is strictly increasing on all of ℝ
125 return Monotonicity::Increasing;
126
127 case FunctionNode::FuncType::Ln: {
128 // ln is strictly increasing on ℝ⁺ (positive reals)
129 // Check if the interval is within positive reals
130 // If the lower bound is > 0 (or open at 0), ln is increasing
131 auto lo = endpoint_to_double(interval.lower);
132 if (lo.has_value() && *lo > 0.0) {
133 return Monotonicity::Increasing;
134 }
135 // If lower is open at 0, also increasing
136 if (lo.has_value() && *lo == 0.0 && interval.lower.is_open) {
137 return Monotonicity::Increasing;
138 }
139 // If interval is (0, +inf) or similar positive interval
140 if (interval.lower.is_neg_infinity) {
141 return Monotonicity::Unknown; // ln not defined on negative reals
142 }
143 return Monotonicity::Unknown;
144 }
145
146 default:
147 break;
148 }
149 }
150
151 // Negation: multiply by -1 reverses monotonicity
152 if (auto inner = detect_negation(lamina::detail::node(expr))) {
153 auto inner_expr = lamina::detail::expression_from_node(inner);
154 Monotonicity inner_mono = infer_monotonicity(inner_expr, var, interval);
155 switch (inner_mono) {
156 case Monotonicity::Increasing: return Monotonicity::Decreasing;
157 case Monotonicity::Decreasing: return Monotonicity::Increasing;
158 case Monotonicity::NonDecreasing: return Monotonicity::NonIncreasing;
159 case Monotonicity::NonIncreasing: return Monotonicity::NonDecreasing;
160 default: return Monotonicity::Unknown;
161 }
162 }
163
164 // VariableNode: check PropertyStore for declared monotonicity
165 if (auto var_node = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(expr))) {
166 const auto& props = impl_->ctx.current_properties();
167 return props.get_monotonicity(var_node->name(), var, interval);
168 }
169
170 return Monotonicity::Unknown;
171}

175/**
176 * @brief Helper: collect positive integer exponents that appear in PowerNode
177 * expressions within the RelationStore involving the given variable names.
178 */
179static std::vector<int> collect_power_exponents(const RelationStore& store,
180 const std::string& lhs_name,
181 const std::string& rhs_name) {
182 std::unordered_set<int> exponents;
183
184 auto check_node = [&](const std::shared_ptr<const SymbolicNode>& node) {
185 if (!node) return;
186 auto pow_node = std::dynamic_pointer_cast<const PowerNode>(node);
187 if (!pow_node) return;
188
189 // Check if the base is one of our variables
190 auto base_var = std::dynamic_pointer_cast<const VariableNode>(pow_node->base());
191 if (!base_var) return;
192 if (base_var->name() != lhs_name && base_var->name() != rhs_name) return;
193
194 // Check if the exponent is a positive integer
195 auto exp_num = std::dynamic_pointer_cast<const NumberNode>(pow_node->exponent());
196 if (!exp_num) return;
197 if (!is_positive_integer_number(*exp_num)) return;
198
199 // Extract the integer value
200 int n = 0;
201 if (std::holds_alternative<BigInt>(exp_num->value())) {
202 const auto& b = std::get<BigInt>(exp_num->value());
203 // Only consider small exponents to avoid explosion
204 if (b > BigInt(100)) return;
205 n = static_cast<int>(b.IsNegative() ? 0 : b.is_zero() ? 0 : std::stoi(b.to_string()));
206 } else if (std::holds_alternative<Rational>(exp_num->value())) {
207 BigInt num_val = std::get<Rational>(exp_num->value()).get_numerator();
208 if (num_val > BigInt(100)) return;
209 n = static_cast<int>(std::stoi(num_val.to_string()));
210 } else {
211 double v = std::get<lmmc_real_t>(exp_num->value());
212 if (v <= 0.0 || v > 100.0 || v != std::floor(v)) return;
213 n = static_cast<int>(v);
214 }
215
216 if (n > 0) {
217 exponents.insert(n);
218 }
219 };
220
221 // Scan all relations in the store for PowerNode expressions
222 for (const auto& rel : store.get_relations()) {
223 check_node(lamina::detail::node(rel.lhs));
224 check_node(lamina::detail::node(rel.rhs));
225 }
226
227 return std::vector<int>(exponents.begin(), exponents.end());
228}

235Result<void> InferenceEngine::apply_monotonicity_rules_checked(
236 const Relation& rel,
237 RelationStore& store,
238 PropertyStore& prop_store,
239 int depth) {
240 // Stop recursion at maximum depth
241 if (depth >= MAX_MONOTONICITY_DEPTH) return Result<void>::success();
242
243 // Only apply to GT (greater-than) relations
244 if (rel.op != RelationalNode::Op::GT) return Result<void>::success();
245
246 // Extract LHS and RHS — both must be single VariableNodes
247 auto lhs_var = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(rel.lhs));
248 auto rhs_var = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(rel.rhs));
249 if (!lhs_var || !rhs_var) return Result<void>::success();
250
251 const std::string& x_name = lhs_var->name();
252 const std::string& y_name = rhs_var->name();
253
254 // Helper to add a deduced relation and recursively apply monotonicity rules
255 auto add_deduced = [&](const SymbolicExpr& new_lhs,
256 const SymbolicExpr& new_rhs) -> Result<void> {
257 // Don't add if already present
258 if (store.has_relation(new_lhs, new_rhs, RelationalNode::Op::GT)) {
259 return Result<void>::success();
260 }
261
262 auto inserted = store.add_relation_checked(
263 new_lhs, new_rhs, RelationalNode::Op::GT, prop_store);
264 if (!inserted.has_value()) {
265 return Result<void>::failure(inserted.error());
266 }
267
268 // Recursively apply monotonicity rules to the newly added relation
269 Relation new_rel{new_lhs, new_rhs, RelationalNode::Op::GT};
270 return apply_monotonicity_rules_checked(new_rel, store, prop_store, depth + 1);
271 };
272
273 // Check domain assumptions using the AssumptionContext (read-through all scopes)
274 bool both_positive = impl_->ctx.has_sign(x_name, Sign::Positive) &&
275 impl_->ctx.has_sign(y_name, Sign::Positive);
276 bool both_real = impl_->ctx.has_domain(x_name, Domain::Real) &&
277 impl_->ctx.has_domain(y_name, Domain::Real);
278 bool both_nonnegative = impl_->ctx.has_sign(x_name, Sign::NonNegative) &&
279 impl_->ctx.has_sign(y_name, Sign::NonNegative);
280
281 if (both_positive) {
282 auto ln_x_node = lamina::detail::make_node<FunctionNode>(
283 FunctionNode::FuncType::Ln,
284 std::vector<std::shared_ptr<const SymbolicNode>>{lhs_var->clone()});
285 auto ln_y_node = lamina::detail::make_node<FunctionNode>(
286 FunctionNode::FuncType::Ln,
287 std::vector<std::shared_ptr<const SymbolicNode>>{rhs_var->clone()});
288
289 auto ln_x = lamina::detail::expression_from_node(ln_x_node);
290 auto ln_y = lamina::detail::expression_from_node(ln_y_node);
291 auto deduced = add_deduced(ln_x, ln_y);
292 if (!deduced.has_value()) return deduced;
293 }
294
295 if (both_positive) {
296 auto sqrt_x_node = lamina::detail::make_node<FunctionNode>(
297 FunctionNode::FuncType::Sqrt,
298 std::vector<std::shared_ptr<const SymbolicNode>>{lhs_var->clone()});
299 auto sqrt_y_node = lamina::detail::make_node<FunctionNode>(
300 FunctionNode::FuncType::Sqrt,
301 std::vector<std::shared_ptr<const SymbolicNode>>{rhs_var->clone()});
302
303 auto sqrt_x = lamina::detail::expression_from_node(sqrt_x_node);
304 auto sqrt_y = lamina::detail::expression_from_node(sqrt_y_node);
305 auto deduced = add_deduced(sqrt_x, sqrt_y);
306 if (!deduced.has_value()) return deduced;
307 }
308
309 if (both_real) {
310 auto exp_x_node = lamina::detail::make_node<FunctionNode>(
311 FunctionNode::FuncType::Exp,
312 std::vector<std::shared_ptr<const SymbolicNode>>{lhs_var->clone()});
313 auto exp_y_node = lamina::detail::make_node<FunctionNode>(
314 FunctionNode::FuncType::Exp,
315 std::vector<std::shared_ptr<const SymbolicNode>>{rhs_var->clone()});
316
317 auto exp_x = lamina::detail::expression_from_node(exp_x_node);
318 auto exp_y = lamina::detail::expression_from_node(exp_y_node);
319 auto deduced = add_deduced(exp_x, exp_y);
320 if (!deduced.has_value()) return deduced;
321 }
322
323 if (both_nonnegative) {
324 std::vector<int> exponents = collect_power_exponents(store, x_name, y_name);
325
326 for (int n : exponents) {
327 auto exp_node = lamina::detail::make_node<NumberNode>(BigInt(n));
328 auto pow_x_node = lamina::detail::make_node<PowerNode>(lhs_var->clone(), exp_node->clone());
329 auto pow_y_node = lamina::detail::make_node<PowerNode>(rhs_var->clone(), exp_node->clone());
330
331 auto pow_x = lamina::detail::expression_from_node(pow_x_node);
332 auto pow_y = lamina::detail::expression_from_node(pow_y_node);
333 auto deduced = add_deduced(pow_x, pow_y);
334 if (!deduced.has_value()) return deduced;
335 }
336 }
337
338 return Result<void>::success();
339}
