80static ODESolutionResult solve_variation_of_parameters_core(
81 const std::shared_ptr<SymbolicExpr>& y1,
82 const std::shared_ptr<SymbolicExpr>& y2,
83 const std::shared_ptr<SymbolicExpr>& g,
84 const std::string& x,
85 ComputationContext& context)
86{
87 ODESolution result;
88 result.method_used = ODEType::HigherOrder_ConstCoeff;
89 result.constants = {};
90
91 if (!y1 || !y2 || !g) {
92 result.general_solution = nullptr;
93 return result;
94 }
95
96 /// 计算 y₁' 和 y₂'
97 auto y1_prime = y1->differentiate(x);
98 auto y2_prime = y2->differentiate(x);
99
100 if (!y1_prime || !y2_prime) {
101 result.general_solution = nullptr;
102 return result;
103 }
104
105 /// 计算 Wronskian: W = y₁·y₂' - y₂·y₁'
106 auto term1 = SymbolicExpr::multiply(y1, y2_prime);
107 auto term2 = SymbolicExpr::multiply(y2, y1_prime);
108 auto W = SymbolicExpr::add(term1,
109 SymbolicExpr::multiply(SymbolicExpr::number(-1), term2));
110 W = W->simplify();
111 EqvOptions trig_options;
112 trig_options.profile = EqvProfile::TrigBasic;
113 auto unit_wronskian = check_equivalent(
114 W, SymbolicExpr::number(1), context, trig_options);
115 if (!unit_wronskian) {
116 return ODESolutionResult::failure(unit_wronskian.error());
117 }
118 if (std::holds_alternative<ProvedZeroResidual>(
119 unit_wronskian.value())) {
120 W = SymbolicExpr::number(1);
121 }
122
123 if (!W || W->is_zero()) {
124 result.general_solution = nullptr;
125 return result;
126 }
127
128 /// 计算 u₁' = -y₂·g(x)/W
129 auto u1_prime = SymbolicExpr::divide(
130 SymbolicExpr::multiply(SymbolicExpr::number(-1),
131 SymbolicExpr::multiply(y2, g)),
132 W);
133 u1_prime = u1_prime->simplify();
134
135 /// 计算 u₂' = y₁·g(x)/W
136 auto u2_prime = SymbolicExpr::divide(
137 SymbolicExpr::multiply(y1, g),
138 W);
139 u2_prime = u2_prime->simplify();
140
141 Integrator integrator;
142 auto u1_value =
143 integrator.integrate_checked(*u1_prime, x, context);
144 if (!u1_value) {
145 return ODESolutionResult::failure(u1_value.error());
146 }
147 auto u2_value =
148 integrator.integrate_checked(*u2_prime, x, context);
149 if (!u2_value) {
150 return ODESolutionResult::failure(u2_value.error());
151 }
152 auto u1 = std::make_shared<SymbolicExpr>(
153 std::move(u1_value.value()));
154 auto u2 = std::make_shared<SymbolicExpr>(
155 std::move(u2_value.value()));
156
157 /// 特解: y_p = u₁·y₁ + u₂·y₂
158 auto y_p = SymbolicExpr::add(
159 SymbolicExpr::multiply(u1, y1),
160 SymbolicExpr::multiply(u2, y2));
161 y_p = y_p->simplify();
162
163 result.general_solution = y_p;
164 return result;
165}
