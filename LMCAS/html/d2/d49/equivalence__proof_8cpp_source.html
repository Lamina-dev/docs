10bool collect_polynomial_variable_names(
11 const std::shared_ptr<const SymbolicNode>& node,
12 std::set<std::string>& variables, detail::RewriteBudget& budget) {
13 if (!node) { return true; }
14 detail::RewriteScope scope(&budget);
15 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(node)) {
16 if (variable->is_constant()) return false;
17 variables.insert(variable->name());
18 return true;
19 }
20 if (std::dynamic_pointer_cast<const NumberNode>(node)) {
21 return true;
22 }
23 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
24 for (const auto& operand : add->operands()) {
25 if (!collect_polynomial_variable_names(operand, variables, budget)) {
26 return false;
27 }
28 }
29 return true;
30 }
31 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
32 for (const auto& operand : multiply->operands()) {
33 if (!collect_polynomial_variable_names(operand, variables, budget)) {
34 return false;
35 }
36 }
37 return true;
38 }
39 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
40 return collect_polynomial_variable_names(power->base(), variables, budget) &&
41 collect_polynomial_variable_names(power->exponent(), variables, budget);
42 }
43 if (auto complex_node = std::dynamic_pointer_cast<const ComplexNode>(node)) {
44 return collect_polynomial_variable_names(complex_node->real(), variables, budget) &&
45 collect_polynomial_variable_names(complex_node->imag(), variables, budget);
46 }
47 return false;
48}

68 }
69
70 auto step = context.consume_steps(4, kEquivalentOperation);
71 if (!step) { return Result<std::optional<bool>>::failure(step.error()); }
72
73 std::set<std::string> variables;
74 if (!collect_polynomial_variable_names(
75 LMCAS::detail::node(difference), variables, budget)) {
76 return Result<std::optional<bool>>::success(std::nullopt);
77 }
78 if (variables.size() > 1) {

80 context, options.budget.max_rewrite_depth, budget.max_nodes(),
81 kEquivalentOperation);

83 return Result<std::optional<bool>>::success(
84 expanded->is_zero() ? std::optional<bool>{true} : std::nullopt);
85 }
86 const std::string variable = variables.empty() ? "x" : *variables.begin();

88 if (!recognized) {

90 }
91 if (!recognized.value()) {
92 return Result<std::optional<bool>>::success(std::nullopt);
93 }
