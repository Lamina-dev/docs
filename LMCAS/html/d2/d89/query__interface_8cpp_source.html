1/**
2 * @file query_interface.cpp
3 * @brief Implementation of the QueryInterface class.
4 *
5 * The QueryInterface is the single public entry point for property queries.
6 * It handles special cases (null, NaN, Infinity, Matrix, Relational, Logical)
7 * and delegates to the InferenceEngine for all other node types.
8 *
9 * Query results are cached by (expression_hash, property_type) to avoid
10 * redundant inference. The cache is invalidated on any AssumptionContext mutation.
11 */

37QueryTriboolResult checked_query_result(
38 const SymbolicExpr& expr,
39 const std::string& operation,
40 const std::function<Tribool()>& query) {
41 if (!lamina::detail::node(expr)) {
42 return QueryTriboolResult::failure(
43 CasErrc::InvalidArgument, "query expression must not be null", operation);
44 }
45 try {
46 return QueryTriboolResult::success(query());
47 } catch (const std::bad_alloc&) {
48 return QueryTriboolResult::failure(
49 CasErrc::ResourceLimit, "query allocation failed", operation);
50 } catch (const std::exception& ex) {
51 return QueryTriboolResult::failure(CasErrc::InternalInvariant, ex.what(), operation);
52 }
53}

92
93QueryTriboolResult QueryInterface::cached_query_checked(
94 const SymbolicExpr& expr,
95 PropType prop,
96 const std::string& operation,
97 const std::function<Tribool()>& compute) const {
98 // Null root cannot be hashed — skip cache
99 if (!lamina::detail::node(expr)) {
100 return QueryTriboolResult::failure(
101 CasErrc::InvalidArgument, "query expression must not be null", operation);
102 }
103
104 // Check if the context has been mutated since we last validated the cache.
105 // If so, invalidate and update our observed generation.
106 uint64_t current_gen = ctx_.cache_generation();
107 if (current_gen != observed_generation_) {
108 cache_.clear();
109 observed_generation_ = current_gen;
110 }
111
112 CacheKey key{lamina::detail::node(expr)->hash(), prop};
113 auto it = cache_.find(key);
114 if (it != cache_.end()) {
115 for (const auto& entry : it->second) {
116 if (lamina::detail::node(entry.expression)->equals(*lamina::detail::node(expr))) {
117 return QueryTriboolResult::success(entry.result);
118 }
119 }
120 }
121
122 Tribool result = compute();
123 cache_[key].push_back(CacheEntry{expr, result});
124 return QueryTriboolResult::success(result);
125}

129bool QueryInterface::is_unhandled_type(const SymbolicExpr& expression) const {
130 const auto& node = lamina::detail::node(expression);
131 if (!node) return true;
132 if (std::dynamic_pointer_cast<const MatrixNode>(node)) return true;
133 if (std::dynamic_pointer_cast<const RelationalNode>(node)) return true;
134 if (std::dynamic_pointer_cast<const LogicalNode>(node)) return true;
135 return false;
136}

138bool QueryInterface::is_nan_number(const SymbolicExpr& expression) const {
139 const auto node = std::dynamic_pointer_cast<const NumberNode>(
140 lamina::detail::node(expression));
141 if (node && std::holds_alternative<lmmc_real_t>(node->value())) {
142 lmmc_real_t v = std::get<lmmc_real_t>(node->value());
143 return std::isnan(v);
144 }
145 return false;
146}

148bool QueryInterface::is_infinity_node(const SymbolicExpr& expression) const {
149 const auto& node = lamina::detail::node(expression);
150 // Direct infinity: FunctionNode with FuncType::Infinity
151 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(node)) {
152 return func->type() == FunctionNode::FuncType::Infinity;
153 }
154 // Negative infinity: MultiplyNode(-1, Infinity)
155 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
156 if (mul->operands().size() == 2) {
157 for (const auto& op : mul->operands()) {
158 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(op)) {
159 if (func->type() == FunctionNode::FuncType::Infinity) {
160 return true;
161 }
162 }
163 }
164 }
165 }
166 return false;
167}

169int QueryInterface::get_infinity_sign(const SymbolicExpr& expression) const {
170 const auto& node = lamina::detail::node(expression);
171 // Direct infinity node → positive infinity
172 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(node)) {
173 if (func->type() == FunctionNode::FuncType::Infinity) {
174 return +1;
175 }
176 }
177 // Negative infinity: MultiplyNode containing -1 and Infinity
178 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
179 bool has_infinity = false;
180 bool has_neg_one = false;
181 for (const auto& op : mul->operands()) {
182 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(op)) {
183 if (func->type() == FunctionNode::FuncType::Infinity) {
184 has_infinity = true;
185 }
186 }
187 if (auto num = std::dynamic_pointer_cast<const NumberNode>(op)) {
188 // Check if it's -1
189 if (std::holds_alternative<BigInt>(num->value())) {
190 if (std::get<BigInt>(num->value()) == BigInt(-1)) {
191 has_neg_one = true;
192 }
193 } else if (std::holds_alternative<Rational>(num->value())) {
194 if (std::get<Rational>(num->value()) == Rational(-1)) {
195 has_neg_one = true;
196 }
197 } else if (std::holds_alternative<lmmc_real_t>(num->value())) {
198 lmmc_real_t v = std::get<lmmc_real_t>(num->value());
199 if (v == -1.0) {
200 has_neg_one = true;
201 }
202 }
203 }
204 }
205 if (has_infinity && has_neg_one) return -1;
206 if (has_infinity) return +1; // Infinity multiplied by something else positive
207 }
208 return 0; // indeterminate
209}

237 });
238 if (!result) {
239 throw std::runtime_error(result.error().message);
240 }

635
636std::vector<QueryInterface::ConditionSet> QueryInterface::query_conditions_impl(
637 const SymbolicExpr& expr, Sign target) const {
638 if (!lamina::detail::node(expr)) {
639 return {};
640 }
641 if (auto var = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(expr))) {
642 ConditionSet cs;
643 cs.sign_conditions.emplace_back(var->name(), target);
644 return {cs};
645 }
646 if (auto add = std::dynamic_pointer_cast<const AddNode>(lamina::detail::node(expr))) {
647 if (add->operands().size() == 2) {
648 std::shared_ptr<const SymbolicNode> pos_operand = nullptr;
649 std::shared_ptr<const SymbolicNode> neg_operand = nullptr; // the term being subtracted
650
651 for (size_t i = 0; i < 2; ++i) {
652 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(add->operands()[i]);
653 if (mul && mul->operands().size() == 2) {
654 for (size_t j = 0; j < 2; ++j) {
655 auto num = std::dynamic_pointer_cast<const NumberNode>(mul->operands()[j]);
656 if (num) {
657 bool is_neg_one = false;
658 if (std::holds_alternative<BigInt>(num->value())) {
659 is_neg_one = (std::get<BigInt>(num->value()) == BigInt(-1));
660 } else if (std::holds_alternative<Rational>(num->value())) {
661 is_neg_one = (std::get<Rational>(num->value()) == Rational(-1));
662 } else if (std::holds_alternative<lmmc_real_t>(num->value())) {
663 is_neg_one = (std::get<lmmc_real_t>(num->value()) == -1.0);
664 }
665 if (is_neg_one) {
666 neg_operand = mul->operands()[1 - j];
667 pos_operand = add->operands()[1 - i];
668 break;
669 }
670 }
671 }
672 }
673 if (neg_operand) break;
674 }
675 if (pos_operand && neg_operand) {
676 auto lhs_var = std::dynamic_pointer_cast<const VariableNode>(pos_operand);
677 auto rhs_var = std::dynamic_pointer_cast<const VariableNode>(neg_operand);
678
679 if (lhs_var && rhs_var && target == Sign::Positive) {
680 ConditionSet cs1;
681 cs1.sign_conditions.emplace_back(lhs_var->name(), Sign::Positive);
682 cs1.sign_conditions.emplace_back(rhs_var->name(), Sign::Negative);
683 ConditionSet cs2;
684 auto lhs_expr = lamina::detail::expression_from_node(pos_operand);
685 auto rhs_expr = lamina::detail::expression_from_node(neg_operand);
686 Relation rel{lhs_expr, rhs_expr, RelationalNode::Op::GT};
687 cs2.relational_conditions.push_back(rel);
688 cs2.sign_conditions.emplace_back(rhs_var->name(), Sign::NonNegative);
689
690 return {cs1, cs2};
691 } else if (lhs_var && rhs_var && target == Sign::Negative) {
692 ConditionSet cs1;
693 cs1.sign_conditions.emplace_back(lhs_var->name(), Sign::Negative);
694 cs1.sign_conditions.emplace_back(rhs_var->name(), Sign::Positive);
695
696 ConditionSet cs2;
697 auto lhs_expr = lamina::detail::expression_from_node(pos_operand);
698 auto rhs_expr = lamina::detail::expression_from_node(neg_operand);
699 Relation rel{rhs_expr, lhs_expr, RelationalNode::Op::GT};
700 cs2.relational_conditions.push_back(rel);
701 cs2.sign_conditions.emplace_back(lhs_var->name(), Sign::NonNegative);
702
703 return {cs1, cs2};
704 } else if (lhs_var && rhs_var && target == Sign::NonNegative) {
705 ConditionSet cs1;
706 cs1.sign_conditions.emplace_back(lhs_var->name(), Sign::NonNegative);
707 cs1.sign_conditions.emplace_back(rhs_var->name(), Sign::NonPositive);
708
709 ConditionSet cs2;
710 auto lhs_expr = lamina::detail::expression_from_node(pos_operand);
711 auto rhs_expr = lamina::detail::expression_from_node(neg_operand);
712 Relation rel{lhs_expr, rhs_expr, RelationalNode::Op::GEQ};
713 cs2.relational_conditions.push_back(rel);
714
715 return {cs1, cs2};
716 }
717 }
718 }
719 if (target == Sign::Positive || target == Sign::NonNegative) {
720 bool all_variables = true;
721 ConditionSet cs;
722 for (const auto& op : add->operands()) {
723 if (auto var = std::dynamic_pointer_cast<const VariableNode>(op)) {
724 cs.sign_conditions.emplace_back(var->name(), target);
725 } else {
726 all_variables = false;
727 break;
728 }
729 }
730 if (all_variables && !cs.sign_conditions.empty()) {
731 return {cs};
732 }
733 }
734 return {};
735 }
736 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(expr))) {
737 if (mul->operands().size() == 2) {
738 std::shared_ptr<const SymbolicNode> numerator = nullptr;
739 std::shared_ptr<const SymbolicNode> denominator = nullptr;
740
741 for (size_t i = 0; i < 2; ++i) {
742 auto pow = std::dynamic_pointer_cast<const PowerNode>(mul->operands()[i]);
743 if (pow) {
744 auto exp_num = std::dynamic_pointer_cast<const NumberNode>(pow->exponent());
745 if (exp_num) {
746 bool is_neg_one = false;
747 if (std::holds_alternative<BigInt>(exp_num->value())) {
748 is_neg_one = (std::get<BigInt>(exp_num->value()) == BigInt(-1));
749 } else if (std::holds_alternative<Rational>(exp_num->value())) {
750 is_neg_one = (std::get<Rational>(exp_num->value()) == Rational(-1));
751 } else if (std::holds_alternative<lmmc_real_t>(exp_num->value())) {
752 is_neg_one = (std::get<lmmc_real_t>(exp_num->value()) == -1.0);
753 }
754 if (is_neg_one) {
755 denominator = pow->base();
756 numerator = mul->operands()[1 - i];
757 break;
758 }
759 }
760 }
761 }
762 if (numerator && denominator) {
763 auto num_var = std::dynamic_pointer_cast<const VariableNode>(numerator);
764 auto den_var = std::dynamic_pointer_cast<const VariableNode>(denominator);
765
766 if (num_var && den_var) {
767 if (target == Sign::Positive) {
768 // x/y > 0: both positive OR both negative
769 ConditionSet cs1;
770 cs1.sign_conditions.emplace_back(num_var->name(), Sign::Positive);
771 cs1.sign_conditions.emplace_back(den_var->name(), Sign::Positive);
772
773 ConditionSet cs2;
774 cs2.sign_conditions.emplace_back(num_var->name(), Sign::Negative);
775 cs2.sign_conditions.emplace_back(den_var->name(), Sign::Negative);
776
777 return {cs1, cs2};
778 } else if (target == Sign::Negative) {
779 // x/y < 0: one positive, one negative
780 ConditionSet cs1;
781 cs1.sign_conditions.emplace_back(num_var->name(), Sign::Positive);
782 cs1.sign_conditions.emplace_back(den_var->name(), Sign::Negative);
783
784 ConditionSet cs2;
785 cs2.sign_conditions.emplace_back(num_var->name(), Sign::Negative);
786 cs2.sign_conditions.emplace_back(den_var->name(), Sign::Positive);
787
788 return {cs1, cs2};
789 }
790 }
791 return {};
792 }
793 }
794 if (target == Sign::Positive) {
795 bool all_variables = true;
796 ConditionSet cs;
797 for (const auto& op : mul->operands()) {
798 if (auto var = std::dynamic_pointer_cast<const VariableNode>(op)) {
799 cs.sign_conditions.emplace_back(var->name(), Sign::Positive);
800 } else {
801 all_variables = false;
802 break;
803 }
804 }
805 if (all_variables && !cs.sign_conditions.empty()) {
806 return {cs};
807 }
808 }
809
810 return {};
811 }
812 return {};
813}
