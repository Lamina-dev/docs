18/**
19 * @brief 多元 Hensel 提升（单变量步）
20 *
21 * 将一元因子逐变量提升，从 f(x₁, a₂, ..., aₙ) 的因子恢复到包含 lift_var 的因子。
22 * 每次引入一个辅助变量，计算修正项直到达到次数上界。
23 *
24 * 算法（参考 Geddes, Czapor, Labahn §15.6）：
25 * 1. 将一元因子嵌入到完整变量集中
26 * 2. 对 k = 1, ..., degree_bound：
27 * a. 计算当前因子乘积（截断到 lift_var 次数 ≤ k）
28 * b. 计算误差 E = poly - product（提取 lift_var 次数恰为 k 的部分）
29 * c. 若误差为零则继续
30 * d. 用丢番图求解器分配误差到各因子
31 * e. 将修正项加到各因子上
32 * f. 验证：product of lifted factors ≡ poly mod (lift_var - eval_point)^(k+1)
33 * 3. 返回提升后的因子
34 *
35 * @param[in] poly 原始多元多项式
36 * @param[in] univariate_factors 一元分解得到的因子列表
37 * @param[in] lift_var 当前提升的变量名
38 * @param[in] eval_point 该变量的求值点
39 * @param[in] degree_bound 提升次数上界
40 * @return 提升后的多元因子列表
41 *
42 * @see Geddes, Czapor, Labahn. "Algorithms for Computer Algebra." §15.6.
43 */
44std::vector<MultiPoly> multivariate_hensel_lift(
45 const MultiPoly& poly,
46 const std::vector<Polynomial<Rational>>& univariate_factors,
47 const std::string& lift_var,
48 const Rational& eval_point,
49 int degree_bound)
50{
51 int r = static_cast<int>(univariate_factors.size());
52 if (r == 0) return {};
53 if (r == 1) {
54 return {poly};
55 }
56 if (degree_bound <= 0) {
57 /// 无需提升：将一元因子嵌入完整变量集返回
58 const auto& vars = poly.variables();
59 std::vector<MultiPoly> result;
60 result.reserve(r);
61 for (const auto& uf : univariate_factors) {
62 MultiPoly mp = MultiPoly::from_univariate(uf, uf.variable_name);
63 if (mp.variables() != vars && !vars.empty()) {
64 std::string uni_var = uf.variable_name;
65 int uni_idx = -1;
66 for (size_t i = 0; i < vars.size(); ++i) {
67 if (vars[i] == uni_var) { uni_idx = static_cast<int>(i); break; }
68 }
69 if (uni_idx >= 0) {
70 std::vector<MultiPoly::Term> new_terms;
71 for (const auto& term : mp.terms()) {
72 Monomial full_mono(vars.size(), 0);
73 if (!term.first.empty()) full_mono[uni_idx] = term.first[0];
74 new_terms.emplace_back(std::move(full_mono), term.second);
75 }
76 mp = MultiPoly(std::move(new_terms), vars);
77 }
78 }
79 result.push_back(std::move(mp));
80 }
81 return result;
82 }
83
84 const auto& vars = poly.variables();
85
86 /// 确定主变量（一元因子的变量）和提升变量在完整变量集中的位置
87 std::string main_var = univariate_factors[0].variable_name;
88 int main_var_idx = -1;
89 int lift_var_idx = -1;
90 for (size_t i = 0; i < vars.size(); ++i) {
91 if (vars[i] == main_var) main_var_idx = static_cast<int>(i);
92 if (vars[i] == lift_var) lift_var_idx = static_cast<int>(i);
93 }
94
95 /// 将一元因子嵌入到完整变量集中
96 std::vector<MultiPoly> factors;
97 factors.reserve(r);
98 for (const auto& uf : univariate_factors) {
99 MultiPoly mp = MultiPoly::from_univariate(uf, main_var);
100 if (mp.variables().size() != vars.size() && !vars.empty()) {
101 std::vector<MultiPoly::Term> new_terms;
102 for (const auto& term : mp.terms()) {
103 Monomial full_mono(vars.size(), 0);
104 if (main_var_idx >= 0 && !term.first.empty()) {
105 full_mono[main_var_idx] = term.first[0];
106 }
107 new_terms.emplace_back(std::move(full_mono), term.second);
108 }
109 mp = MultiPoly(std::move(new_terms), vars);
110 }
111 factors.push_back(std::move(mp));
112 }
113
114 /// 若 lift_var 不在变量集中，无法提升，直接返回嵌入后的因子
115 if (lift_var_idx < 0) {
116 return factors;
117 }
118
119 /// 迭代提升：对 k = 1, ..., degree_bound
120 /// 当 eval_point != 0 时，需要在 (lift_var - eval_point) 的幂次展开中工作
121 /// 预计算 (lift_var - eval_point) 的幂次用于修正项构造
122 std::vector<MultiPoly> shift_powers;
123 shift_powers.reserve(degree_bound + 1);
124 shift_powers.push_back(MultiPoly(Rational(1), vars)); // (lift_var - eval_point)^0 = 1
125 if (lift_var_idx >= 0) {
126 Monomial var_mono(vars.size(), 0);
127 var_mono[lift_var_idx] = 1;
128 std::vector<MultiPoly::Term> lin_terms;
129 lin_terms.emplace_back(var_mono, Rational(1));
130 if (!eval_point.is_zero()) {
131 Monomial const_mono(vars.size(), 0);
132 lin_terms.emplace_back(const_mono, -eval_point);
133 }
134 MultiPoly shift_lin(std::move(lin_terms), vars);
135 shift_powers.push_back(shift_lin);
136 for (int k = 2; k <= degree_bound; ++k) {
137 shift_powers.push_back(shift_powers[k - 1] * shift_lin);
138 }
139 } else {
140 for (int k = 1; k <= degree_bound; ++k) {
141 shift_powers.push_back(MultiPoly(Rational(1), vars));
142 }
143 }
144
145 for (int k = 1; k <= degree_bound; ++k) {
146 /// 计算当前因子乘积，截断到 lift_var 次数 ≤ k
147 MultiPoly product = factors[0];
148 for (int i = 1; i < r; ++i) {
149 product = product * factors[i];
150 product = truncate_mod_var(product, lift_var, k + 1);
151 }
152
153 /// 计算误差 E = poly_trunc - product（仅比较 lift_var 次数 ≤ k 的部分）
154 MultiPoly poly_trunc = truncate_mod_var(poly, lift_var, k + 1);
155 MultiPoly error = poly_trunc - product;
156
157 if (error.is_zero()) continue;
158
159 /// 提取误差的 k 阶 Taylor 系数（关于 lift_var - eval_point）
160 /// 方法：将 error 除以 (lift_var - eval_point)^k，然后在 lift_var = eval_point 处求值
161 /// 对于 eval_point = 0，这等价于提取 lift_var^k 的系数
162 MultiPoly error_k;
163 if (eval_point.is_zero()) {
164 /// 简化路径：直接提取 lift_var 指数恰为 k 的项
165 std::vector<MultiPoly::Term> error_k_terms;
166 for (const auto& term : error.terms()) {
167 const Monomial& mono = term.first;
168 int exp = (lift_var_idx >= 0 && static_cast<size_t>(lift_var_idx) < mono.size())
169 ? mono[lift_var_idx] : 0;
170 if (exp == k) {
171 Monomial reduced = mono;
172 reduced[lift_var_idx] = 0;
173 error_k_terms.emplace_back(std::move(reduced), term.second);
174 }
175 }
176 if (error_k_terms.empty()) continue;
177 error_k = MultiPoly(std::move(error_k_terms), vars);
178 } else {
179 /// 非零求值点：计算 error / (lift_var - eval_point)^k 在 lift_var = eval_point 处的值
180 /// 等价于对 error 关于 lift_var 求 k 次导数后除以 k! 再在 eval_point 处求值
181 /// 实现：逐次除以 (lift_var - eval_point) 并求值
182 MultiPoly remainder = error;
183 for (int j = 0; j < k; ++j) {
184 /// 除以 (lift_var - eval_point)：先在 eval_point 处求值确认余数为零
185 /// 然后执行多项式除法
186 MultiPoly eval_check = remainder.eval(lift_var, eval_point);
187 if (!eval_check.is_zero()) {
188 /// 低阶误差未完全消除——不应发生，但防御性跳过
189 remainder = MultiPoly(Rational(0), vars);
190 break;
191 }
192 /// 执行除法：remainder / (lift_var - eval_point)
193 try {
194 remainder = remainder.exact_div(shift_powers[1]);
195 } catch (...) {
196 remainder = MultiPoly(Rational(0), vars);
197 break;
198 }
199 }
200 /// 在 eval_point 处求值得到 error_k
201 error_k = remainder.eval(lift_var, eval_point);
202 if (error_k.is_zero()) continue;
203 }
204
205 /// 求解丢番图方程：找到修正项 δ₁, ..., δᵣ
206 /// 使得 Σ δᵢ * cofactor_i ≡ error_k，其中 cofactor_i = ∏_{j≠i} fⱼ
207 /// 因子在 lift_var = eval_point 处求值得到一元形式
208 std::vector<MultiPoly> cofactors;
209 cofactors.reserve(r);
210 for (int i = 0; i < r; ++i) {
211 MultiPoly cof(Rational(1), vars);
212 for (int j = 0; j < r; ++j) {
213 if (j == i) continue;
214 MultiPoly fj_eval = factors[j].eval(lift_var, eval_point);
215 cof = cof * fj_eval;
216 }
217 cofactors.push_back(cof);
218 }
219
220 std::vector<MultiPoly> corrections = multivariate_diophantine(
221 cofactors, error_k, lift_var, eval_point, k + 1);
222
223 if (corrections.size() != static_cast<size_t>(r)) continue;
224
225 /// 更新因子：fᵢ = fᵢ + δᵢ * (lift_var - eval_point)^k
226 for (int i = 0; i < r; ++i) {
227 if (corrections[i].is_zero()) continue;
228
229 /// 构造 δᵢ * (lift_var - eval_point)^k
230 MultiPoly correction_poly = corrections[i] * shift_powers[k];
231 factors[i] = factors[i] + correction_poly;
232 }
233
234 /// 验证：product of lifted factors ≡ poly mod (lift_var - eval_point)^(k+1)
235 MultiPoly verify_product = factors[0];
236 for (int i = 1; i < r; ++i) {
237 verify_product = verify_product * factors[i];
238 verify_product = truncate_mod_var(verify_product, lift_var, k + 1);
239 }
240 MultiPoly verify_error = poly_trunc - verify_product;
241 if (!verify_error.is_zero()) {
242 bool residual_ok = true;
243 for (const auto& term : verify_error.terms()) {
244 const Monomial& mono = term.first;
245 int exp = (lift_var_idx >= 0 &&
246 static_cast<size_t>(lift_var_idx) < mono.size())
247 ? mono[lift_var_idx] : 0;
248 if (exp <= k) {
249 residual_ok = false;
250 break;
251 }
252 }
253 if (!residual_ok) {
254 break;
255 }
256 }
257 }
258
259 return factors;
260}
261/**
262 * @internal
263 * @brief 一元多项式扩展 GCD（有理数域上）
264 *
265 * 给定 a, b ∈ Q[x]，计算 gcd(a, b) 及 Bezout 系数 s, t，
266 * 使得 s*a + t*b = gcd(a, b)。
267 *
268 * @param[in] a 第一个多项式
269 * @param[in] b 第二个多项式
270 * @param[out] s Bezout 系数 s
271 * @param[out] t Bezout 系数 t
272 * @return gcd(a, b)
273 */
274static Polynomial<Rational> extended_gcd_poly(
275 const Polynomial<Rational>& a,
276 const Polynomial<Rational>& b,
277 Polynomial<Rational>& s,
278 Polynomial<Rational>& t)
279{
280 std::string var = a.variable_name;
281
282 /// r0 = a, r1 = b
283 /// s0*a + t0*b = r0
284 /// s1*a + t1*b = r1
285 Polynomial<Rational> r0 = a, r1 = b;
286 Polynomial<Rational> s0({Rational(1)}, var), s1(var); // s0=1, s1=0
287 Polynomial<Rational> t0(var), t1({Rational(1)}, var); // t0=0, t1=1
288
289 while (!r1.is_zero()) {
290 auto [q, r] = r0.div_mod(r1);
291
292 Polynomial<Rational> r_new = r;
293 Polynomial<Rational> s_new = s0 - q * s1;
294 Polynomial<Rational> t_new = t0 - q * t1;
295
296 r0 = r1; r1 = r_new;
297 s0 = s1; s1 = s_new;
298 t0 = t1; t1 = t_new;
299 }
300
301 /// 归一化使 gcd 为首一多项式
302 if (!r0.is_zero()) {
303 Rational lc = r0.lead_coeff();
304 if (lc != Rational(1)) {
305 Rational inv = Rational(1) / lc;
306 for (auto& c : r0.coeffs) c = c * inv;
307 for (auto& c : s0.coeffs) c = c * inv;
308 for (auto& c : t0.coeffs) c = c * inv;
309 }
310 }
311
312 s = s0;
313 t = t0;
314 return r0;
315}

317/**
318 * @internal
319 * @brief 将 MultiPoly 截断为关于指定变量次数 < degree_bound 的部分
320 *
321 * 保留所有项中指定变量指数严格小于 degree_bound 的项。
322 *
323 * @param[in] poly 输入多项式
324 * @param[in] var 变量名
325 * @param[in] degree_bound 次数上界（保留 < degree_bound 的项）
326 * @return 截断后的多项式
327 */
328static MultiPoly truncate_mod_var(const MultiPoly& poly, const std::string& var,
329 int degree_bound)
330{
331 if (poly.is_zero()) return poly;
332
333 const auto& vars = poly.variables();
334 int var_idx = -1;
335 for (size_t i = 0; i < vars.size(); ++i) {
336 if (vars[i] == var) { var_idx = static_cast<int>(i); break; }
337 }
338 /// 若变量不在列表中，多项式不含该变量，无需截断
339 if (var_idx < 0) return poly;
340
341 std::vector<MultiPoly::Term> result_terms;
342 for (const auto& term : poly.terms()) {
343 const Monomial& mono = term.first;
344 int exp = (static_cast<size_t>(var_idx) < mono.size()) ? mono[var_idx] : 0;
345 if (exp < degree_bound) {
346 result_terms.push_back(term);
347 }
348 }
349
350 if (result_terms.empty()) return MultiPoly(Rational(0), vars);
351 return MultiPoly(std::move(result_terms), vars);
352}

354/**
355 * @brief 多元丢番图方程求解器
356 *
357 * 求解 s₁·f₁ + s₂·f₂ + ... + sᵣ·fᵣ ≡ target (mod (var - eval_point)^degree_bound)，
358 * 其中各 fᵢ 两两互素。用于 Hensel 提升过程中计算修正项。
359 *
360 * 算法：
361 * 1. 二因子情形（r=2）：使用扩展 GCD 求 Bezout 系数
362 * 2. 一般情形（r>2）：递归归约——令 g = f₂·...·fᵣ，
363 * 先解 s₁·f₁ + t·g ≡ target，再递归解 s₂·f₂ + ... + sᵣ·fᵣ ≡ t
364 * 3. 对每个解截断为关于 var 的次数 < degree_bound
365 *
366 * @param[in] factors 互素因子列表 [f₁, ..., fᵣ]
367 * @param[in] target 目标多项式 c
368 * @param[in] var 变量名
369 * @param[in] eval_point 求值点
370 * @param[in] degree_bound 次数上界
371 * @return 解多项式列表 [s₁, s₂, ..., sᵣ]
372 *
373 * @see Geddes, Czapor, Labahn. "Algorithms for Computer Algebra." §15.5.
374 */
375std::vector<MultiPoly> multivariate_diophantine(
376 const std::vector<MultiPoly>& factors,
377 const MultiPoly& target,
378 const std::string& var,
379 const Rational& eval_point,
380 int degree_bound)
381{
382 int r = static_cast<int>(factors.size());
383 if (r == 0) return {};
384 if (r == 1) {
385 /// 单因子情形：s₁ = target / f₁（精确除法后截断）
386 try {
387 MultiPoly s1 = target.exact_div(factors[0]);
388 s1 = truncate_mod_var(s1, var, degree_bound);
389 return {s1};
390 } catch (...) {
391 /// 若不整除，返回 target 本身（退化情形）
392 return {truncate_mod_var(target, var, degree_bound)};
393 }
394 }
395
396 const auto& vars = factors[0].variables();
397
398 if (r == 2) {
399 /// 二因子情形：使用扩展 GCD
400 /// 将 MultiPoly 因子转换为一元 Polynomial<Rational> 进行 GCD 计算
401 /// 因为在 Hensel 提升过程中，因子在求值点处本质为一元多项式
402
403 /// 先对因子在 eval_point 处求值（去除 var 维度），得到一元多项式
404 /// 但实际上 factors 可能已经是关于某个主变量的一元多项式
405 /// 策略：尝试直接转换为一元多项式；若失败则在 var 处求值后转换
406
407 Polynomial<Rational> f1_uni, f2_uni;
408 bool converted = false;
409
410 /// 尝试直接转换
411 try {
412 f1_uni = factors[0].to_univariate();
413 f2_uni = factors[1].to_univariate();
414 converted = true;
415 } catch (...) {
416 /// 若不是一元的，在 var 处求值后再转换
417 try {
418 MultiPoly f1_eval = factors[0].eval(var, eval_point);
419 MultiPoly f2_eval = factors[1].eval(var, eval_point);
420 f1_uni = f1_eval.to_univariate();
421 f2_uni = f2_eval.to_univariate();
422 converted = true;
423 } catch (...) {
424 converted = false;
425 }
426 }
427
428 if (!converted) {
429 /// 无法转换为一元多项式，返回零解作为退化处理
430 return {MultiPoly(Rational(0), vars), MultiPoly(Rational(0), vars)};
431 }
432
433 /// 计算扩展 GCD：s*f1 + t*f2 = gcd(f1, f2)
434 Polynomial<Rational> s_coeff, t_coeff;
435 Polynomial<Rational> g = extended_gcd_poly(f1_uni, f2_uni, s_coeff, t_coeff);
436
437 /// 将 target 也转换为一元多项式
438 Polynomial<Rational> target_uni;
439 try {
440 target_uni = target.to_univariate();
441 } catch (...) {
442 try {
443 MultiPoly target_eval = target.eval(var, eval_point);
444 target_uni = target_eval.to_univariate();
445 } catch (...) {
446 /// target 为常数
447 if (target.is_constant()) {
448 Rational c = target.is_zero() ? Rational(0) : target.terms()[0].second;
449 target_uni = Polynomial<Rational>({c}, f1_uni.variable_name);
450 } else {
451 return {MultiPoly(Rational(0), vars), MultiPoly(Rational(0), vars)};
452 }
453 }
454 }
455
456 /// gcd 应为 1（因子互素），但处理一般情形
457 /// s₁ = s * (target / gcd), s₂ = t * (target / gcd)
458 Polynomial<Rational> scale;
459 if (g.is_zero() || g.degree() < 0) {
460 /// 退化情形
461 return {MultiPoly(Rational(0), vars), MultiPoly(Rational(0), vars)};
462 }
463
464 auto [quotient, remainder] = target_uni.div_mod(g);
465 if (!remainder.is_zero()) {
466 /// target 不能被 gcd 整除——不应发生于互素因子
467 /// 退化处理：返回零
468 return {MultiPoly(Rational(0), vars), MultiPoly(Rational(0), vars)};
469 }
470
471 /// s₁ = s * quotient, s₂ = t * quotient
472 Polynomial<Rational> s1_uni = s_coeff * quotient;
473 Polynomial<Rational> s2_uni = t_coeff * quotient;
474
475 /// 对 s₁ 取模 f₂，对 s₂ 取模 f₁，确保次数约束
476 /// s₁*f₁ + s₂*f₂ = target，且 deg(s₁) < deg(f₂), deg(s₂) < deg(f₁)
477 if (!f2_uni.is_zero() && s1_uni.degree() >= f2_uni.degree()) {
478 auto [q1, r1] = s1_uni.div_mod(f2_uni);
479 s1_uni = r1;
480 /// 调整 s₂：s₂ = s₂ + q1 * f₁
481 s2_uni = s2_uni + q1 * f1_uni;
482 }
483
484 /// 确定用于 from_univariate 的变量名
485 std::string uni_var = f1_uni.variable_name;
486
487 /// 转换回 MultiPoly
488 MultiPoly s1_mp = MultiPoly::from_univariate(s1_uni, uni_var);
489 MultiPoly s2_mp = MultiPoly::from_univariate(s2_uni, uni_var);
490
491 /// 若变量集不匹配，嵌入到完整变量集
492 if (s1_mp.variables() != vars && !vars.empty()) {
493 /// 找到 uni_var 在 vars 中的位置
494 int uni_idx = -1;
495 for (size_t i = 0; i < vars.size(); ++i) {
496 if (vars[i] == uni_var) { uni_idx = static_cast<int>(i); break; }
497 }
498 if (uni_idx >= 0) {
499 std::vector<MultiPoly::Term> new_terms1, new_terms2;
500 for (const auto& term : s1_mp.terms()) {
501 Monomial full_mono(vars.size(), 0);
502 if (!term.first.empty()) full_mono[uni_idx] = term.first[0];
503 new_terms1.emplace_back(std::move(full_mono), term.second);
504 }
505 for (const auto& term : s2_mp.terms()) {
506 Monomial full_mono(vars.size(), 0);
507 if (!term.first.empty()) full_mono[uni_idx] = term.first[0];
508 new_terms2.emplace_back(std::move(full_mono), term.second);
509 }
510 s1_mp = MultiPoly(std::move(new_terms1), vars);
511 s2_mp = MultiPoly(std::move(new_terms2), vars);
512 }
513 }
514
515 /// 截断为 degree < degree_bound
516 s1_mp = truncate_mod_var(s1_mp, var, degree_bound);
517 s2_mp = truncate_mod_var(s2_mp, var, degree_bound);
518
519 return {s1_mp, s2_mp};
520 }
521
522 /// 一般情形（r > 2）：递归归约
523 /// 令 g = f₂ * f₃ * ... * fᵣ
524 MultiPoly g = factors[1];
525 for (int i = 2; i < r; ++i) {
526 g = g * factors[i];
527 }
528
529 /// 解二因子方程：s₁*f₁ + t*g ≡ target
530 std::vector<MultiPoly> two_factors = {factors[0], g};
531 std::vector<MultiPoly> two_solution = multivariate_diophantine(
532 two_factors, target, var, eval_point, degree_bound);
533
534 if (two_solution.size() != 2) {
535 /// 退化情形
536 std::vector<MultiPoly> result(r, MultiPoly(Rational(0), vars));
537 return result;
538 }
539
540 /// s₁ 已确定
541 MultiPoly s1 = two_solution[0];
542 MultiPoly t_poly = two_solution[1];
543
544 /// 递归解：s₂*f₂ + ... + sᵣ*fᵣ ≡ t_poly * g
545 /// 因为原方程为 s₁*f₁ + t*g = target，
546 /// 我们需要 s₂*f₂ + ... + sᵣ*fᵣ = t*g = target - s₁*f₁
547 MultiPoly remaining_target = t_poly * g;
548 remaining_target = truncate_mod_var(remaining_target, var, degree_bound);
549
550 std::vector<MultiPoly> remaining_factors(factors.begin() + 1, factors.end());
551 std::vector<MultiPoly> remaining_solution = multivariate_diophantine(
552 remaining_factors, remaining_target, var, eval_point, degree_bound);
553
554 /// 组装完整解
555 std::vector<MultiPoly> result;
556 result.push_back(std::move(s1));
557 for (auto& sol : remaining_solution) {
558 result.push_back(std::move(sol));
559 }
560
561 return result;
562}
