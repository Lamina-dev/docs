20// FlsFree can invoke callbacks for other threads. Only thread exit or the
21// unloading thread may release its lease; live foreign threads are reclaimed
22// by process exit. Explicit unload requires all other CAS threads to be joined.
23void CALLBACK release_lmmc_lifecycle(void* owner) {
24 if (reinterpret_cast<std::uintptr_t>(owner) == GetCurrentThreadId() &&
25 lmmc_deinit() != LMMC_STATUS_OK) {
26 std::terminate();
27 }
