23void gaussian_eliminate(std::vector<std::vector<std::shared_ptr<SymbolicExpr>>>& A, size_t m, size_t n, std::vector<size_t>& pivot_col_for_row, int& sign) {
24 size_t pivot_row = 0;
25 sign = 1;
26 pivot_col_for_row.assign(m, (size_t)-1);
27
28 for (size_t col = 0; col < n && pivot_row < m; ++col) {
29 size_t max_row = pivot_row;
30 bool found_pivot = false;
31 while (max_row < m) {
32 A[max_row][col] = A[max_row][col]->simplify();
33 if (!A[max_row][col]->is_zero()) {
34 found_pivot = true;
35 break;
36 }
37 max_row++;
38 }
39
40 if (!found_pivot) continue;
41
42 if (pivot_row != max_row) {
43 std::swap(A[pivot_row], A[max_row]);
44 sign = -sign;
45 }
46
47 pivot_col_for_row[pivot_row] = col;
48 auto pivot = A[pivot_row][col];
49 auto pivot_inv = SymbolicExpr::power(pivot, SymbolicExpr::number(-1));
50
51 for (size_t k = col; k < A[pivot_row].size(); ++k) {
52 A[pivot_row][k] = SymbolicExpr::multiply(A[pivot_row][k], pivot_inv)->simplify();
53 }
54
55 for (size_t i = 0; i < m; ++i) {
56 if (i != pivot_row) {
57 auto factor = A[i][col];
58 if (!factor->is_zero()) {
59 auto neg_factor = SymbolicExpr::multiply(factor, SymbolicExpr::number(-1));
60 for (size_t k = col; k < A[i].size(); ++k) {
61 auto term = SymbolicExpr::multiply(neg_factor, A[pivot_row][k]);
62 A[i][k] = SymbolicExpr::add(A[i][k], term)->simplify();
63 }
64 }
65 }
66 }
67 pivot_row++;
68 }

75static bool get_integer_value(const std::shared_ptr<const SymbolicNode>& node, long long& value) {
76 auto num = std::dynamic_pointer_cast<const NumberNode>(node);
77 if (!num) return false;
78 if (std::holds_alternative<BigInt>(num->value())) {
79 value = std::get<BigInt>(num->value()).to_int();
80 return true;
81 }
82 if (std::holds_alternative<Rational>(num->value())) {
83 const auto& r = std::get<Rational>(num->value());
84 if (!r.is_integer()) return false;
85 value = r.to_BigInt().to_int();
86 return true;
87 }
88 if (std::holds_alternative<lmmc_real_t>(num->value())) {
89 lmmc_real_t d = std::get<lmmc_real_t>(num->value());
90 int eq;
91 lmmc_double_nearly_equal_tol(d, std::round(d), 1e-12, 1e-12, &eq);
92 if (!eq) return false;
93 value = static_cast<long long>(std::llround(d));
94 return true;
95 }
96 return false;
97}

149 const std::shared_ptr<const SymbolicNode>& term,
150 const std::vector<std::shared_ptr<const SymbolicNode>>& den_factors
151) {

153 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(term)) {
154 factors.insert(factors.end(), mul->operands().begin(), mul->operands().end());
155 } else if (term) {
156 factors.push_back(term);
157 }
158 factors.insert(factors.end(), den_factors.begin(), den_factors.end());
159
160 std::shared_ptr<const NumberNode> const_acc =
161 lamina::detail::make_node<NumberNode>(BigInt(1));
162 std::map<std::shared_ptr<const SymbolicNode>, std::shared_ptr<const NumberNode>, NodeLess> bases;
163
164 for (const auto& op : factors) {
165 if (!op) continue;
166 if (auto num = std::dynamic_pointer_cast<const NumberNode>(op)) {
167 const_acc = multiply_number_nodes(const_acc, num);
168 continue;
169 }
170
171 std::shared_ptr<const SymbolicNode> base = op;
172 std::shared_ptr<const NumberNode> exp = lamina::detail::make_node<NumberNode>(BigInt(1));
173 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(op)) {
174 base = pow->base();
175 if (auto e_num = std::dynamic_pointer_cast<const NumberNode>(pow->exponent())) {
176 exp = e_num;
177 }
178 }
179
180 auto it = bases.find(base);
181 if (it == bases.end()) {
182 bases[base] = exp;
183 } else {
184 bases[base] = add_number_nodes(it->second, exp);
185 }
186 }
187
188 std::vector<std::shared_ptr<const SymbolicNode>> final_ops;
189 if (!const_acc->is_one()) final_ops.push_back(const_acc);
190
191 for (const auto& [base, exp] : bases) {
192 if (exp->is_zero()) continue;
193 if (exp->is_one()) final_ops.push_back(base);
194 else final_ops.push_back(lamina::detail::make_node<PowerNode>(base, exp));
195 }
196
197 if (final_ops.empty()) return SymbolicExpr::number(1);
198 if (final_ops.size() == 1) return lamina::detail::make_expression_ptr(final_ops[0]);
199 return lamina::detail::make_expression_ptr(lamina::detail::make_node<MultiplyNode>(final_ops));
200}

202bool solver_detail::is_polynomial_node(const std::shared_ptr<const SymbolicNode>& node) {
203 if (!node) return false;
204 if (std::dynamic_pointer_cast<const NumberNode>(node)) return true;
205 if (std::dynamic_pointer_cast<const VariableNode>(node)) return true;
206
207 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
208 for (const auto& op : add->operands()) {
209 if (!is_polynomial_node(op)) return false;
210 }
211 return true;
212 }
213
214 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
215 for (const auto& op : mul->operands()) {
216 if (!is_polynomial_node(op)) return false;
217 }
218 return true;
219 }
220
221 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
222 long long exp = 0;
223 if (!get_integer_value(pow->exponent(), exp)) return false;
224 if (exp < 0) return false;
225 return is_polynomial_node(pow->base());
226 }
227
228 return false;
229}

241 const std::shared_ptr<const SymbolicNode>& node,
242 std::vector<std::shared_ptr<const SymbolicNode>>& den_factors,
243 std::vector<std::shared_ptr<SymbolicExpr>>& den_constraints
244) {
245 if (!node) return false;
246
247 if (std::dynamic_pointer_cast<const NumberNode>(node) || std::dynamic_pointer_cast<const VariableNode>(node)) {
248 return true;
249 }
250
251 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
252 for (const auto& op : add->operands()) {
253 if (!collect_denominator_factors(op, den_factors, den_constraints)) return false;
254 }
255 return true;
256 }
257
258 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
259 for (const auto& op : mul->operands()) {
260 if (!collect_denominator_factors(op, den_factors, den_constraints)) return false;
261 }
262 return true;
263 }
264
265 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
266 long long exp = 0;
267 if (!get_integer_value(pow->exponent(), exp)) return false;
268
269 if (exp < 0) {
270 if (!is_polynomial_node(pow->base())) return false;
271 long long k = -exp;
272 if (k == 1) {
273 den_factors.push_back(pow->base());
274 } else {
275 den_factors.push_back(SymbolicFactory::create_power(pow->base(), SymbolicFactory::create_number(BigInt(k))));
276 }
277 den_constraints.push_back(lamina::detail::make_expression_ptr(pow->base()));
278 return true;
279 }
280
281 if (!collect_denominator_factors(pow->base(), den_factors, den_constraints)) return false;
282 return true;
283 }
284
285 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(node)) {
286 for (const auto& arg : func->arguments()) {
287 if (!collect_denominator_factors(arg, den_factors, den_constraints)) return false;
288 }
289 return true;
290 }
291
292 return false;
293}

325std::map<std::string, SymbolicExpr> Solver::solve_linear_system(
326 const std::vector<SymbolicExpr>& equations_in,
327 const std::vector<std::string>& variables)
328{
329
330 size_t num_vars = variables.size();
331 size_t num_eqs = equations_in.size();
332
333 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> matrix(num_eqs, std::vector<std::shared_ptr<SymbolicExpr>>(num_vars + 1));
334
335 for (size_t i = 0; i < num_eqs; ++i) {
336 SymbolicExpr current_eq = equations_in[i];
337
338 SymbolicExpr constant_part = current_eq;
339
340 for (size_t j = 0; j < num_vars; ++j) {
341 auto [coeff, remainder] = isolate_linear_coeff(constant_part, variables[j]);
342 matrix[i][j] = lamina::detail::make_expression_ptr(coeff);
343
344 constant_part = remainder;
345 }
346
347 std::vector<std::shared_ptr<const SymbolicNode>> ops;
348 ops.push_back(SymbolicFactory::create_number(BigInt(-1)));
349 ops.push_back(lamina::detail::node(constant_part));
350 auto neg_const = SymbolicFactory::create_multiply(ops);
351 matrix[i][num_vars] = lamina::detail::make_expression_ptr(neg_const);
352 }
353
354 std::vector<size_t> pivot_col_for_row;
355 int sign;
356 gaussian_eliminate(matrix, num_eqs, num_vars, pivot_col_for_row, sign);
357
358 std::map<std::string, SymbolicExpr> solution;
359 for (size_t i = 0; i < num_eqs; ++i) {
360 size_t pcol = pivot_col_for_row[i];
361 if (pcol != (size_t)-1) {
362 SymbolicExpr val = *matrix[i][num_vars];
363 for (size_t j = pcol + 1; j < num_vars; ++j) {
364 SymbolicExpr c = *matrix[i][j];
365 if (!lamina::detail::node(c)->is_zero()) {
366
367 std::vector<std::shared_ptr<const SymbolicNode>> mops;
368 mops.push_back(lamina::detail::node(c));
369 mops.push_back(SymbolicFactory::create_variable(variables[j]));
370 auto term = SymbolicFactory::create_multiply(mops);
371
372 std::vector<std::shared_ptr<const SymbolicNode>> nops;
373 nops.push_back(SymbolicFactory::create_number(BigInt(-1)));
374 nops.push_back(term);
375
376 std::vector<std::shared_ptr<const SymbolicNode>> aops;
377 aops.push_back(lamina::detail::node(val));
378 aops.push_back(SymbolicFactory::create_multiply(nops));
379 val = lamina::detail::expression_from_node(
380 SymbolicFactory::create_add(aops));
381 }
382 }
383 solution.insert_or_assign(variables[pcol], std::move(val));
384 }
385 }
386
387 return solution;
388}
