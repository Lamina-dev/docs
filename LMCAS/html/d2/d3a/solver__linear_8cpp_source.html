29static bool get_integer_value(const std::shared_ptr<const SymbolicNode>& node, long long& value) {
30 auto num = std::dynamic_pointer_cast<const NumberNode>(node);
31 if (!num) return false;
32 if (std::holds_alternative<BigInt>(num->value())) {
33 value = std::get<BigInt>(num->value()).to_int();
34 return true;
35 }
36 if (std::holds_alternative<Rational>(num->value())) {
37 const auto& r = std::get<Rational>(num->value());
38 if (!r.is_integer()) return false;
39 value = r.to_BigInt().to_int();
40 return true;
41 }
42 if (std::holds_alternative<lmmc_real_t>(num->value())) {
43 lmmc_real_t d = std::get<lmmc_real_t>(num->value());
44 int eq;
45 lmmc_double_nearly_equal_tol(d, std::round(d), 1e-12, 1e-12, &eq);
46 if (!eq) return false;
47 value = static_cast<long long>(std::llround(d));
48 return true;
49 }
50 return false;
51}

102std::shared_ptr<SymbolicExpr> solver_detail::multiply_no_expand(
103 const std::shared_ptr<const SymbolicNode>& term,
104 const std::vector<std::shared_ptr<const SymbolicNode>>& den_factors
105) {
106 std::vector<std::shared_ptr<const SymbolicNode>> factors;
107 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(term)) {
108 factors.insert(factors.end(), mul->operands().begin(), mul->operands().end());
109 } else if (term) {
110 factors.push_back(term);
111 }
112 factors.insert(factors.end(), den_factors.begin(), den_factors.end());
113
114 std::shared_ptr<const NumberNode> const_acc =
115 lamina::detail::make_node<NumberNode>(BigInt(1));
116 std::map<std::shared_ptr<const SymbolicNode>, std::shared_ptr<const NumberNode>, NodeLess> bases;
117
118 for (const auto& op : factors) {
119 if (!op) continue;
120 if (auto num = std::dynamic_pointer_cast<const NumberNode>(op)) {
121 const_acc = multiply_number_nodes(const_acc, num);
122 continue;
123 }
124
125 std::shared_ptr<const SymbolicNode> base = op;
126 std::shared_ptr<const NumberNode> exp = lamina::detail::make_node<NumberNode>(BigInt(1));
127 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(op)) {
128 base = pow->base();
129 if (auto e_num = std::dynamic_pointer_cast<const NumberNode>(pow->exponent())) {
130 exp = e_num;
131 }
132 }
133
134 auto it = bases.find(base);
135 if (it == bases.end()) {
136 bases[base] = exp;
137 } else {
138 bases[base] = add_number_nodes(it->second, exp);
139 }
140 }
141
142 std::vector<std::shared_ptr<const SymbolicNode>> final_ops;
143 if (!const_acc->is_one()) final_ops.push_back(const_acc);
144
145 for (const auto& [base, exp] : bases) {
146 if (exp->is_zero()) continue;
147 if (exp->is_one()) final_ops.push_back(base);
148 else final_ops.push_back(lamina::detail::make_node<PowerNode>(base, exp));
149 }
150
151 if (final_ops.empty()) return SymbolicExpr::number(1);
152 if (final_ops.size() == 1) return lamina::detail::make_expression_ptr(final_ops[0]);
153 return lamina::detail::make_expression_ptr(lamina::detail::make_node<MultiplyNode>(final_ops));
154}

156bool solver_detail::is_polynomial_node(const std::shared_ptr<const SymbolicNode>& node) {
157 if (!node) return false;
158 if (std::dynamic_pointer_cast<const NumberNode>(node)) return true;
159 if (std::dynamic_pointer_cast<const VariableNode>(node)) return true;
160
161 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
162 for (const auto& op : add->operands()) {
163 if (!is_polynomial_node(op)) return false;
164 }
165 return true;
166 }
167
168 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
169 for (const auto& op : mul->operands()) {
170 if (!is_polynomial_node(op)) return false;
171 }
172 return true;
173 }
174
175 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
176 long long exp = 0;
177 if (!get_integer_value(pow->exponent(), exp)) return false;
178 if (exp < 0) return false;
179 return is_polynomial_node(pow->base());
180 }
181
182 return false;
183}

194bool solver_detail::collect_denominator_factors(
195 const std::shared_ptr<const SymbolicNode>& node,
196 std::vector<std::shared_ptr<const SymbolicNode>>& den_factors,
197 std::vector<std::shared_ptr<SymbolicExpr>>& den_constraints
198) {
199 if (!node) return false;
200
201 if (std::dynamic_pointer_cast<const NumberNode>(node) || std::dynamic_pointer_cast<const VariableNode>(node)) {
202 return true;
203 }
204
205 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
206 for (const auto& op : add->operands()) {
207 if (!collect_denominator_factors(op, den_factors, den_constraints)) return false;
208 }
209 return true;
210 }
211
212 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
213 for (const auto& op : mul->operands()) {
214 if (!collect_denominator_factors(op, den_factors, den_constraints)) return false;
215 }
216 return true;
217 }
218
219 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
220 long long exp = 0;
221 if (!get_integer_value(pow->exponent(), exp)) return false;
222
223 if (exp < 0) {
224 if (!is_polynomial_node(pow->base())) return false;
225 long long k = -exp;
226 if (k == 1) {
227 den_factors.push_back(pow->base());
228 } else {
229 den_factors.push_back(SymbolicFactory::create_power(pow->base(), SymbolicFactory::create_number(BigInt(k))));
230 }
231 den_constraints.push_back(lamina::detail::make_expression_ptr(pow->base()));
232 return true;
233 }
234
235 if (!collect_denominator_factors(pow->base(), den_factors, den_constraints)) return false;
236 return true;
237 }
238
239 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(node)) {
240 for (const auto& arg : func->arguments()) {
241 if (!collect_denominator_factors(arg, den_factors, den_constraints)) return false;
242 }
243 return true;
244 }
245
246 return false;
247}

249static std::pair<SymbolicExpr, SymbolicExpr> isolate_linear_coeff(const SymbolicExpr& expr, const std::string& var) {
250
251 auto expr_ptr = to_ptr(expr);
252 auto A_ptr = expr_ptr->differentiate(var);
253
254 std::vector<std::shared_ptr<const SymbolicNode>> mops;
255 mops.push_back(lamina::detail::node(A_ptr));
256 mops.push_back(SymbolicFactory::create_variable(var));
257 auto term_Ax = SymbolicFactory::create_multiply(mops);
258
259 std::vector<std::shared_ptr<const SymbolicNode>> nops;
260 nops.push_back(SymbolicFactory::create_number(BigInt(-1)));
261 nops.push_back(term_Ax);
262 auto neg_term = SymbolicFactory::create_multiply(nops);
263
264 std::vector<std::shared_ptr<const SymbolicNode>> aops;
265 aops.push_back(lamina::detail::node(expr));
266 aops.push_back(neg_term);
267 auto B_node = SymbolicFactory::create_add(aops);
268
269 auto B_expr = lamina::detail::expression_from_node(B_node);
270 auto B_simp = to_ptr(B_expr)->simplify();
271
272 auto A_expr = lamina::detail::expression_from_node(lamina::detail::node(A_ptr));
273 auto A_simp = to_ptr(A_expr)->simplify();
274
275 return {lamina::detail::expression_from_node(lamina::detail::node(A_simp)),
276 lamina::detail::expression_from_node(lamina::detail::node(B_simp))};
277}

279std::map<std::string, SymbolicExpr> Solver::solve_linear_system(
280 const std::vector<SymbolicExpr>& equations_in,
281 const std::vector<std::string>& variables)
282{
283
284 size_t num_vars = variables.size();
285 size_t num_eqs = equations_in.size();
286
287 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> matrix(num_eqs, std::vector<std::shared_ptr<SymbolicExpr>>(num_vars + 1));
288
289 for (size_t i = 0; i < num_eqs; ++i) {
290 SymbolicExpr current_eq = equations_in[i];
291
292 SymbolicExpr constant_part = current_eq;
293
294 for (size_t j = 0; j < num_vars; ++j) {
295 auto [coeff, remainder] = isolate_linear_coeff(constant_part, variables[j]);
296 matrix[i][j] = lamina::detail::make_expression_ptr(coeff);
297
298 constant_part = remainder;
299 }
300
301 std::vector<std::shared_ptr<const SymbolicNode>> ops;
302 ops.push_back(SymbolicFactory::create_number(BigInt(-1)));
303 ops.push_back(lamina::detail::node(constant_part));
304 auto neg_const = SymbolicFactory::create_multiply(ops);
305 matrix[i][num_vars] = lamina::detail::make_expression_ptr(neg_const);
306 }
307
308 detail::ExactMatrixData augmented{
309 num_eqs, num_vars + 1, {}};
310 augmented.entries.reserve(num_eqs * (num_vars + 1));
311 for (const auto& row : matrix) {
312 augmented.entries.insert(
313 augmented.entries.end(), row.begin(), row.end());
314 }
315 ComputationContext context;
316 auto solved = detail::solve_linear_exact(
317 std::move(augmented), num_vars, context,
318 "solve_linear_system");
319 if (!solved ||
320 std::holds_alternative<detail::InconsistentLinearSolution>(
321 solved.value())) {
322 return {};
323 }
324
325 std::vector<std::shared_ptr<SymbolicExpr>> values;
326 if (auto* unique =
327 std::get_if<detail::UniqueLinearSolution>(&solved.value())) {
328 values = std::move(unique->values);
329 } else {
330 auto& parametric =
331 std::get<detail::ParametricLinearSolution>(solved.value());
332 values = parametric.particular;
333 for (std::size_t parameter = 0;
334 parameter < parametric.free_columns.size(); ++parameter) {
335 auto free_variable = SymbolicExpr::variable(
336 variables[parametric.free_columns[parameter]]);
337 for (std::size_t column = 0; column < num_vars; ++column) {
338 auto term = SymbolicExpr::multiply(
339 parametric.nullspace_basis[parameter][column],
340 free_variable);
341 values[column] = SymbolicExpr::add(
342 values[column], term)->simplify();
343 }
344 }
345 }
346
347 std::map<std::string, SymbolicExpr> solution;
348 for (std::size_t column = 0; column < num_vars; ++column) {
349 solution.insert_or_assign(
350 variables[column], *values[column]);
351 }
352 return solution;
353}
