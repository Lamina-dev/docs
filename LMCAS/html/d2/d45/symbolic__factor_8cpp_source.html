24static std::optional<int> polynomial_exponent(
25 const std::shared_ptr<const SymbolicNode>& node) {
26 auto number = std::dynamic_pointer_cast<const NumberNode>(node);
27 if (!number) { return std::nullopt; }
28 BigInt exponent;
29 if (std::holds_alternative<BigInt>(number->value())) {
30 exponent = std::get<BigInt>(number->value());
31 } else if (std::holds_alternative<Rational>(number->value())) {
32 const auto& rational = std::get<Rational>(number->value());
33 if (!rational.is_integer()) { return std::nullopt; }
34 exponent = rational.to_bigint();
35 } else {
36 const auto value = std::get<lmmc_real_t>(number->value());
37 if (!std::isfinite(value) || value != std::floor(value)) { return std::nullopt; }
38 exponent = Rational::from_double(value).to_bigint();
39 }
40 if (exponent.is_negative() || exponent > BigInt(100)) { return std::nullopt; }
41 return static_cast<int>(exponent.try_to_int64().value());
42}

44static bool is_poly_expr_node(const std::shared_ptr<const SymbolicNode>& node) {
45 if (!node) { return false; }
46 if (std::dynamic_pointer_cast<const NumberNode>(node)) { return true; }
47 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(node)) {
48 return !variable->is_constant();
49 }
50
51 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
52 for (const auto& op : add->operands()) {
53 if (!is_poly_expr_node(op)) { return false; }
54 }
55 return true;
56 }
57
58 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
59 for (const auto& op : mul->operands()) {
60 if (!is_poly_expr_node(op)) { return false; }
61 }
62 return true;
63 }
64
65 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
66 return polynomial_exponent(pow->exponent()).has_value() &&
67 is_poly_expr_node(pow->base());
68 }
69
70 return false;
71}

83static LMCAS::MultiPoly symbolic_node_to_multipoly(
84 const std::shared_ptr<const SymbolicNode>& node,
85 const std::vector<std::string>& vars)
86{
87 if (!node) { return LMCAS::MultiPoly(Rational(0), vars); }
88
89 if (auto num = std::dynamic_pointer_cast<const NumberNode>(node)) {
90 return LMCAS::MultiPoly(factor_number_coefficient(*num), vars);
91 }
92
93 if (auto var_node = std::dynamic_pointer_cast<const VariableNode>(node)) {
94 LMCAS::Monomial mono(vars.size(), 0);
95 for (size_t i = 0; i < vars.size(); ++i) {
96 if (vars[i] == var_node->name()) {
97 mono[i] = 1;
98 break;
99 }
100 }
101 std::vector<LMCAS::MultiPoly::Term> terms;
102 terms.push_back({mono, Rational(1)});
103 return LMCAS::MultiPoly(std::move(terms), vars);
104 }
105
106 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
107 LMCAS::MultiPoly result(Rational(0), vars);
108 for (const auto& op : add->operands()) {
109 result = result + symbolic_node_to_multipoly(op, vars);
110 }
111 return result;
112 }
113
114 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
115 LMCAS::MultiPoly result(Rational(1), vars);
116 for (const auto& op : mul->operands()) {
117 result = result * symbolic_node_to_multipoly(op, vars);
118 }
119 return result;
120 }
121
122 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
123 auto base_poly = symbolic_node_to_multipoly(pow->base(), vars);
124 const int exp_val = polynomial_exponent(pow->exponent()).value();
125 if (exp_val == 0) { return LMCAS::MultiPoly(Rational(1), vars); }
126 LMCAS::MultiPoly result(Rational(1), vars);
127 for (int i = 0; i < exp_val; ++i) {
128 result = result * base_poly;
129 }
130 return result;
131 }
132
133 return LMCAS::MultiPoly(Rational(0), vars);
134}

136static std::shared_ptr<SymbolicExpr> multipoly_to_symbolic(const LMCAS::MultiPoly& poly) {
137 if (poly.is_zero()) { return SymbolicExpr::number(0); }
138
139 const auto& vars = poly.variables();
140 const auto& terms = poly.terms();
141
142 std::vector<std::shared_ptr<SymbolicExpr>> term_exprs;
143 term_exprs.reserve(terms.size());
144
145 for (const auto& [mono, coeff] : terms) {
146 term_exprs.push_back(
147 detail::multipoly_term_to_expression(mono, coeff, vars));
148 }
149
150 if (term_exprs.empty()) { return SymbolicExpr::number(0); }
151 if (term_exprs.size() == 1) { return term_exprs[0]->simplify(); }
152
153 auto result = term_exprs[0];
154 for (size_t i = 1; i < term_exprs.size(); ++i) {
155 result = SymbolicExpr::add(result, term_exprs[i]);
156 }
157 return result->simplify();
158}

160static std::optional<ExpressionResult> factor_numeric_content(
161 const std::shared_ptr<SymbolicExpr>& simp,
162 const std::shared_ptr<const AddNode>& add_node, ComputationContext& context) {
163 auto content_result =
164 LMCAS::symbolic_polynomial_content(*simp, context);
165 if (!content_result) {
166 if (content_result.error().code == CasErrc::UnsupportedExpression) {
167 return std::nullopt;
168 }
169 return LMCAS::ExpressionResult::failure(
170 content_result.error());
171 }
172 const Rational content_value = content_result.value();
173 if (content_value != Rational(0) && content_value != Rational(1)) {
174 auto content = content_value.is_integer()
175 ? SymbolicExpr::number(content_value.to_bigint())
176 : SymbolicExpr::number(content_value);
177 const Rational inverse_value = Rational(1) / content_value;
178 auto inverse_content = inverse_value.is_integer()
179 ? SymbolicExpr::number(inverse_value.to_bigint())
180 : SymbolicExpr::number(inverse_value);
181 std::vector<std::shared_ptr<const SymbolicNode>> primitive_terms;
182 primitive_terms.reserve(add_node->operands().size());
183 for (const auto& operand : add_node->operands()) {
184 auto term = LMCAS::detail::make_expression_ptr(operand);
185 auto primitive_term = SymbolicExpr::multiply(term, inverse_content)->simplify();
186 primitive_terms.push_back(LMCAS::detail::node(primitive_term));
187 }
188 auto primitive_sum = LMCAS::detail::make_expression_ptr(
189 LMCAS::detail::make_node<AddNode>(primitive_terms));
190 auto factored_primitive =
191 primitive_sum->factor_checked(context);
192 if (!factored_primitive) { return factored_primitive; }
193 return SymbolicExpr::multiply(
194 content, factored_primitive.value())->simplify();
195 }
196 return std::nullopt;
197}

199static std::optional<ExpressionResult> factor_common_polynomial(
200 const std::shared_ptr<const AddNode>& add_node, ComputationContext& context) {
201 std::shared_ptr<SymbolicExpr> common = nullptr;
202 for (const auto& op : add_node->operands()) {
203 auto expr_op = LMCAS::detail::make_expression_ptr(op);
204 if (!common) {
205 common = expr_op;
206 } else {
207 auto gcd = LMCAS::symbolic_polynomial_gcd(*common, *expr_op, context);
208 if (!gcd) {
209 if (gcd.error().code == CasErrc::UnsupportedExpression) {
210 return std::nullopt;
211 }
212 return gcd;
213 }
214 common = std::move(gcd.value());
215 }
216 }
217
218 if (common && !common->is_one() && !common->is_zero()) {
219 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
220 for (const auto& op : add_node->operands()) {
221 auto term = LMCAS::detail::make_expression_ptr(op);
222
223 auto inv_common = SymbolicExpr::power(common, SymbolicExpr::number(-1));
224 auto quot = SymbolicExpr::multiply(term, inv_common);
225 quot = quot->simplify();
226 new_ops.push_back(LMCAS::detail::node(quot));
227 }
228 auto new_sum = LMCAS::detail::make_expression_ptr(LMCAS::detail::make_node<AddNode>(new_ops));
229
230 auto factored_sum = new_sum->factor_checked(context);
231 if (!factored_sum) { return factored_sum; }
232
233 std::vector<std::shared_ptr<const SymbolicNode>> final_ops = {
234 LMCAS::detail::node(common),
235 LMCAS::detail::node(factored_sum.value())};
236 return LMCAS::detail::make_expression_ptr(LMCAS::detail::make_node<MultiplyNode>(final_ops));
237 }
238 return std::nullopt;
239}

241static std::vector<std::shared_ptr<const SymbolicNode>> rational_root_factors(
242 const std::vector<Rational>& roots, const Rational& leading,
243 const std::shared_ptr<SymbolicExpr>& x_node) {
244 std::vector<std::shared_ptr<const SymbolicNode>> factors;
245
246 if (!(leading == Rational(1))) {
247 factors.push_back(LMCAS::detail::node(SymbolicExpr::number(leading)));
248 }
249
250 for (const auto& r : roots) {
251 std::shared_ptr<SymbolicExpr> linear_factor;
252 if (r == Rational(0)) {
253 linear_factor = x_node;
254 } else {
255 auto neg_r = SymbolicExpr::number(Rational(0) - r);
256 linear_factor = SymbolicExpr::add(x_node, neg_r)->simplify();
257 }
258 factors.push_back(LMCAS::detail::node(linear_factor));
259 }
260 return factors;
261}

264 const std::vector<Rational>& roots, const Rational& leading,
265 const std::string& var) {
266 LMCAS::Polynomial<Rational> factored_product({leading}, var);
267 for (const auto& r : roots) {

269 factored_product = factored_product * lin;
270 }

280static RationalRootReconstruction reconstruct_rational_roots(
281 const Polynomial<Rational>& polynomial,
282 const std::vector<Rational>& roots, const std::string& variable) {
283 auto symbol =
284 detail::make_expression_ptr(detail::make_node<VariableNode>(variable));
285 const auto leading = polynomial.lead_coeff();
286 auto factors = rational_root_factors(roots, leading, symbol);
287 auto product = rational_root_product(roots, leading, variable);
288 auto [quotient, remainder] = polynomial.div_mod(product);
289 return {
290 std::move(factors), std::move(quotient), std::move(remainder)};
291}

293static std::optional<ExpressionResult> factor_rational_univariate(
294 const std::shared_ptr<SymbolicExpr>& simp, const std::string& var,
295 ComputationContext& context) {
296 try {
297 auto converted = symbolic_to_poly<Rational>(simp, var);
298 if (!converted) {
299 if (converted.error().code == CasErrc::UnsupportedExpression) return std::nullopt;
300 return ExpressionResult::failure(converted.error());
301 }
302 const auto& poly = converted.value();
303 if (poly.degree() < 2) { return std::nullopt; }
304 auto roots = find_rational_roots(poly);
305 if (roots.empty()) { return std::nullopt; }
306 auto reconstruction =
307 reconstruct_rational_roots(poly, roots, var);
308 if (!reconstruction.remainder.is_zero()) return std::nullopt;
309 auto& factors = reconstruction.factors;
310 auto& quotient = reconstruction.quotient;
311 if (quotient.degree() >= 1) {
312 auto q_expr = poly_to_symbolic(quotient)->simplify();
313 if (quotient.degree() >= 2) {
314 auto q_factored = q_expr->factor_checked(context);
315 if (!q_factored) { return q_factored; }
316 factors.push_back(detail::node(q_factored.value()));
317 } else {
318 factors.push_back(detail::node(q_expr));
319 }
320 }
321 if (factors.size() > 1) {
322 return detail::make_expression_ptr(detail::make_node<MultiplyNode>(factors));
323 }
324 if (factors.size() == 1) { return detail::make_expression_ptr(factors[0]); }
325 } catch (const std::invalid_argument&) {
326 return std::nullopt;
327 } catch (const std::out_of_range&) {
328 return std::nullopt;
329 }
330 return std::nullopt;

377static std::shared_ptr<SymbolicExpr> multivariate_factor_expression(
378 const MultiFactorResult& result) {
379 std::vector<std::shared_ptr<const SymbolicNode>> factor_nodes;
380
381 if (!(result.constant == Rational(1))) {
382 factor_nodes.push_back(
383 LMCAS::detail::node(SymbolicExpr::number(result.constant)));
384 }
385
386 for (size_t i = 0; i < result.factors.size(); ++i) {
387 auto factor_expr = multipoly_to_symbolic(result.factors[i]);
388 if (!factor_expr || factor_expr->is_zero()) { continue; }
389
390 int mult = (i < result.multiplicities.size())
391 ? result.multiplicities[i] : 1;
392
393 if (mult == 1) {
394 factor_nodes.push_back(LMCAS::detail::node(factor_expr));
395 } else {
396 auto pow_expr = SymbolicExpr::power(
397 factor_expr, SymbolicExpr::number(mult));
398 factor_nodes.push_back(LMCAS::detail::node(pow_expr));
399 }
400 }
401
402 if (factor_nodes.size() > 1) {
403 return LMCAS::detail::make_expression_ptr(
404 LMCAS::detail::make_node<MultiplyNode>(std::move(factor_nodes)));
405 } else if (factor_nodes.size() == 1) {
406 return LMCAS::detail::make_expression_ptr(factor_nodes[0]);
407 }
408 return nullptr;
409}

501 const std::shared_ptr<SymbolicExpr>& simp,
502 ComputationContext& context) {
503 const auto transform_variables = LMCAS::free_variables(LMCAS::detail::node(simp));
504 if (transform_variables.empty()) { return simp; }

526LMCAS::ExpressionResult SymbolicExpr::factor_impl(
527 LMCAS::ComputationContext& context) const {
528 auto simp = simplify();
529 if (!simp || !detail::node(simp)) { return simp; }
530 auto add_node = std::dynamic_pointer_cast<const AddNode>(detail::node(simp));
531 if (!add_node) { return factor_transcendental_expression(simp, context); }
532
533 auto content = factor_numeric_content(simp, add_node, context);
534 if (content) { return std::move(*content); }
535 auto common = factor_common_polynomial(add_node, context);
536 if (common) { return std::move(*common); }
537
538 auto variables = free_variables(detail::node(simp));
539 if (variables.size() == 1) {
540 const auto& variable = *variables.begin();
541 auto rational = factor_rational_univariate(simp, variable, context);
542 if (rational) { return std::move(*rational); }
543 auto quadratic = factor_symbolic_quadratic(simp, variable, context);
544 if (quadratic) { return std::move(*quadratic); }
545 }
546 if (variables.size() > 1) {
547 auto multivariate = factor_multivariate_expression(simp, variables, context);
548 if (multivariate) { return std::move(*multivariate); }
549 for (const auto& variable : variables) {
550 auto factored = factor_variable_rational_roots(simp, variable);
551 if (factored) { return std::move(*factored); }
552 }
553 }
554 return factor_transcendental_expression(simp, context);
555}

559{
