20/**
21 * @internal
22 * @brief 判断符号节点是否为多项式表达式(不含超越函数,负指数等)
23 */
24static bool is_poly_expr_node(const std::shared_ptr<const SymbolicNode>& node) {
25 if (!node) return false;
26 if (std::dynamic_pointer_cast<const NumberNode>(node)) return true;
27 if (std::dynamic_pointer_cast<const VariableNode>(node)) return true;
28
29 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
30 for (const auto& op : add->operands()) {
31 if (!is_poly_expr_node(op)) return false;
32 }
33 return true;
34 }
35
36 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
37 for (const auto& op : mul->operands()) {
38 if (!is_poly_expr_node(op)) return false;
39 }
40 return true;
41 }
42
43 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
44 if (auto exp_num = std::dynamic_pointer_cast<const NumberNode>(pow->exponent())) {
45 long long exp_val = 0;
46 if (std::holds_alternative<BigInt>(exp_num->value())) {
47 auto bi = std::get<BigInt>(exp_num->value());
48 if (bi.IsNegative()) return false;
49 exp_val = bi.to_int();
50 } else if (std::holds_alternative<Rational>(exp_num->value())) {
51 auto r = std::get<Rational>(exp_num->value());
52 if (!r.is_integer()) return false;
53 auto bi = r.to_BigInt();
54 if (bi.IsNegative()) return false;
55 exp_val = bi.to_int();
56 } else if (std::holds_alternative<lmmc_real_t>(exp_num->value())) {
57 lmmc_real_t d = std::get<lmmc_real_t>(exp_num->value());
58 if (!std::isfinite(d) || d < 0 || d != std::floor(d)) return false;
59 exp_val = static_cast<long long>(d);
60 } else {
61 return false;
62 }
63 if (exp_val < 0 || exp_val > 100) return false;
64 return is_poly_expr_node(pow->base());
65 }
66 return false;
67 }
68
69 return false;
70}

74/**
75 * @internal
76 * @brief 递归将符号节点转换为 MultiPoly
77 *
78 * 假设节点已通过 is_poly_expr_node 验证为多项式.
79 * @param[in] node 符号节点
80 * @param[in] vars 变量名列表(确定单项式各分量的含义)
81 * @return 对应的 MultiPoly
82 */
83static lamina::MultiPoly symbolic_node_to_multipoly(
84 const std::shared_ptr<const SymbolicNode>& node,
85 const std::vector<std::string>& vars)
86{
87 if (!node) return lamina::MultiPoly(Rational(0), vars);
88
89 if (auto num = std::dynamic_pointer_cast<const NumberNode>(node)) {
90 Rational coeff(0);
91 if (std::holds_alternative<BigInt>(num->value())) {
92 coeff = Rational(std::get<BigInt>(num->value()));
93 } else if (std::holds_alternative<Rational>(num->value())) {
94 coeff = std::get<Rational>(num->value());
95 } else if (std::holds_alternative<lmmc_real_t>(num->value())) {
96 coeff = Rational::from_double(std::get<lmmc_real_t>(num->value()));
97 }
98 return lamina::MultiPoly(coeff, vars);
99 }
100
101 if (auto var_node = std::dynamic_pointer_cast<const VariableNode>(node)) {
102 lamina::Monomial mono(vars.size(), 0);
103 for (size_t i = 0; i < vars.size(); ++i) {
104 if (vars[i] == var_node->name()) {
105 mono[i] = 1;
106 break;
107 }
108 }
109 std::vector<lamina::MultiPoly::Term> terms;
110 terms.push_back({mono, Rational(1)});
111 return lamina::MultiPoly(std::move(terms), vars);
112 }
113
114 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
115 lamina::MultiPoly result(Rational(0), vars);
116 for (const auto& op : add->operands()) {
117 result = result + symbolic_node_to_multipoly(op, vars);
118 }
119 return result;
120 }
121
122 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
123 lamina::MultiPoly result(Rational(1), vars);
124 for (const auto& op : mul->operands()) {
125 result = result * symbolic_node_to_multipoly(op, vars);
126 }
127 return result;
128 }
129
130 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
131 auto base_poly = symbolic_node_to_multipoly(pow->base(), vars);
132 int exp_val = 0;
133 if (auto exp_num = std::dynamic_pointer_cast<const NumberNode>(pow->exponent())) {
134 if (std::holds_alternative<BigInt>(exp_num->value())) {
135 exp_val = std::get<BigInt>(exp_num->value()).to_int();
136 } else if (std::holds_alternative<Rational>(exp_num->value())) {
137 exp_val = std::get<Rational>(exp_num->value()).to_BigInt().to_int();
138 } else if (std::holds_alternative<lmmc_real_t>(exp_num->value())) {
139 exp_val = static_cast<int>(std::get<lmmc_real_t>(exp_num->value()));
140 }
141 }
142 if (exp_val == 0) return lamina::MultiPoly(Rational(1), vars);
143 lamina::MultiPoly result(Rational(1), vars);
144 for (int i = 0; i < exp_val; ++i) {
145 result = result * base_poly;
146 }
147 return result;
148 }
149
150 return lamina::MultiPoly(Rational(0), vars);
151}

153/**
154 * @internal
155 * @brief 将 MultiPoly 转换为符号表达式
156 * @param[in] poly 多元多项式
157 * @return 对应的符号表达式
158 */
159static std::shared_ptr<SymbolicExpr> multipoly_to_symbolic(const lamina::MultiPoly& poly) {
160 if (poly.is_zero()) return SymbolicExpr::number(0);
161
162 const auto& vars = poly.variables();
163 const auto& terms = poly.terms();
164
165 std::vector<std::shared_ptr<SymbolicExpr>> term_exprs;
166 term_exprs.reserve(terms.size());
167
168 for (const auto& [mono, coeff] : terms) {
169 std::vector<std::shared_ptr<SymbolicExpr>> factors;
170
171 /// 系数部分
172 if (!(coeff == Rational(1)) || lamina::total_degree(mono) == 0) {
173 if (coeff == Rational(-1) && lamina::total_degree(mono) > 0) {
174 factors.push_back(SymbolicExpr::number(-1));
175 } else {
176 factors.push_back(SymbolicExpr::number(coeff));
177 }
178 }
179
180 /// 变量部分
181 for (size_t i = 0; i < vars.size() && i < mono.size(); ++i) {
182 if (mono[i] == 0) continue;
183 auto var_expr = lamina::detail::make_expression_ptr(
184 lamina::detail::make_node<VariableNode>(vars[i]));
185 if (mono[i] == 1) {
186 factors.push_back(var_expr);
187 } else {
188 factors.push_back(SymbolicExpr::power(var_expr,
189 SymbolicExpr::number(mono[i])));
190 }
191 }
192
193 std::shared_ptr<SymbolicExpr> term_expr;
194 if (factors.empty()) {
195 term_expr = SymbolicExpr::number(1);
196 } else if (factors.size() == 1) {
197 term_expr = factors[0];
198 } else {
199 auto result = factors[0];
200 for (size_t i = 1; i < factors.size(); ++i) {
201 result = SymbolicExpr::multiply(result, factors[i]);
202 }
203 term_expr = result;
204 }
205 term_exprs.push_back(term_expr);
206 }
207
208 if (term_exprs.empty()) return SymbolicExpr::number(0);
209 if (term_exprs.size() == 1) return term_exprs[0]->simplify();
210
211 auto result = term_exprs[0];
212 for (size_t i = 1; i < term_exprs.size(); ++i) {
213 result = SymbolicExpr::add(result, term_exprs[i]);
214 }
215 return result->simplify();
216}

218std::shared_ptr<SymbolicExpr> SymbolicExpr::factor_impl(
219 lamina::ComputationContext& context) const {
220 auto simp = simplify();
221 if (!simp || !lamina::detail::node(simp)) return simp;
222
223 if (auto add_node = std::dynamic_pointer_cast<const AddNode>(lamina::detail::node(simp))) {
224
225 const Rational content_value = lamina::detail::propagate_result(
226 lamina::symbolic_polynomial_content(*simp, context));
227 if (content_value != Rational(0) && content_value != Rational(1)) {
228 auto content = content_value.is_integer()
229 ? number(content_value.to_BigInt())
230 : number(content_value);
231 const Rational inverse_value = Rational(1) / content_value;
232 auto inverse_content = inverse_value.is_integer()
233 ? number(inverse_value.to_BigInt())
234 : number(inverse_value);
235 std::vector<std::shared_ptr<const SymbolicNode>> primitive_terms;
236 primitive_terms.reserve(add_node->operands().size());
237 for (const auto& operand : add_node->operands()) {
238 auto term = lamina::detail::make_expression_ptr(operand);
239 auto primitive_term = multiply(term, inverse_content)->simplify();
240 primitive_terms.push_back(lamina::detail::node(primitive_term));
241 }
242 auto primitive_sum = lamina::detail::make_expression_ptr(
243 lamina::detail::make_node<AddNode>(primitive_terms));
244 auto factored_primitive = lamina::detail::propagate_result(
245 primitive_sum->factor_checked(context));
246 return multiply(content, factored_primitive)->simplify();
247 }
248
249 /// 步骤 1:提取各项的公因式(GCD)
250 std::shared_ptr<SymbolicExpr> common = nullptr;
251 for (const auto& op : add_node->operands()) {
252 auto expr_op = lamina::detail::make_expression_ptr(op);
253 if (!common) common = expr_op;
254 else {
255 common = lamina::detail::propagate_result(
256 lamina::symbolic_polynomial_gcd(
257 *common, *expr_op, context));
258 }
259 }
260
261 if (common && !common->is_one() && !common->is_zero()) {
262 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
263 for (const auto& op : add_node->operands()) {
264 auto term = lamina::detail::make_expression_ptr(op);
265
266 auto inv_common = power(common, number(-1));
267 auto quot = multiply(term, inv_common);
268 quot = quot->simplify();
269 new_ops.push_back(lamina::detail::node(quot));
270 }
271 auto new_sum = lamina::detail::make_expression_ptr(lamina::detail::make_node<AddNode>(new_ops));
272
273 /// 递归分解余下的和式
274 auto factored_sum = lamina::detail::propagate_result(
275 new_sum->factor_checked(context));
276
277 std::vector<std::shared_ptr<const SymbolicNode>> final_ops = {lamina::detail::node(common), lamina::detail::node(factored_sum)};
278 return lamina::detail::make_expression_ptr(lamina::detail::make_node<MultiplyNode>(final_ops));
279 }
280
281 /// 步骤 2:一元多项式分解(支持任意次数)
282 auto factor_variables = lamina::free_variables(lamina::detail::node(simp));
283 if (factor_variables.size() == 1) {
284 std::string var = *factor_variables.begin();
285 try {
286 auto poly = lamina::symbolic_to_poly<Rational>(simp, var);
287 int deg = poly.degree();
288
289 if (deg >= 2) {
290 /// 使用有理根定理逐步分解
291 auto roots = lamina::find_rational_roots(poly);
292
293 if (!roots.empty()) {
294 auto x_node = lamina::detail::make_expression_ptr(lamina::detail::make_node<VariableNode>(var));
295 auto leading = poly.lead_coeff();
296
297 std::vector<std::shared_ptr<const SymbolicNode>> factors;
298
299 /// 首项系数
300 if (!(leading == Rational(1))) {
301 factors.push_back(lamina::detail::node(number(leading)));
302 }
303
304 /// 从根构建线性因子 (x - r)
305 for (const auto& r : roots) {
306 std::shared_ptr<SymbolicExpr> linear_factor;
307 if (r == Rational(0)) {
308 linear_factor = x_node;
309 } else {
310 auto neg_r = number(Rational(0) - r);
311 linear_factor = SymbolicExpr::add(x_node, neg_r)->simplify();
312 }
313 factors.push_back(lamina::detail::node(linear_factor));
314 }
315
316 /// 计算余下的不可约因子:原多项式 / 已分解因子的乘积
317 lamina::Polynomial<Rational> factored_product({leading}, var);
318 for (const auto& r : roots) {
319 lamina::Polynomial<Rational> lin({Rational(0) - r, Rational(1)}, var);
320 factored_product = factored_product * lin;
321 }
322
323 auto [quotient, remainder] = poly.div_mod(factored_product);
324
325 if (remainder.is_zero() && quotient.degree() >= 1) {
326 /// 余下的不可约因子
327 if (quotient.degree() == 1 && quotient.lead_coeff() == Rational(1)) {
328 /// 一次因子直接加入
329 auto q_expr = lamina::poly_to_symbolic(quotient)->simplify();
330 factors.push_back(lamina::detail::node(q_expr));
331 } else if (quotient.degree() >= 2) {
332 /// 尝试递归分解余下部分
333 auto q_expr = lamina::poly_to_symbolic(quotient)->simplify();
334 auto q_factored = lamina::detail::propagate_result(
335 q_expr->factor_checked(context));
336 factors.push_back(lamina::detail::node(q_factored));
337 } else {
338 auto q_expr = lamina::poly_to_symbolic(quotient)->simplify();
339 factors.push_back(lamina::detail::node(q_expr));
340 }
341 } else if (!remainder.is_zero()) {
342 /// 除法有余数 -> 回退到原始方法
343 goto try_solve_quadratic;
344 }
345
346 if (factors.size() > 1) {
347 return lamina::detail::make_expression_ptr(lamina::detail::make_node<MultiplyNode>(factors));
348 } else if (factors.size() == 1) {
349 return lamina::detail::make_expression_ptr(factors[0]);
350 }
351 }
352 }
353 } catch (const std::invalid_argument&) {
354 } catch (const std::out_of_range&) {
355 }
356
357 try_solve_quadratic:
358 /// 后备:二次多项式通过求解方程分解
359 try {
360 auto poly = lamina::symbolic_to_poly<lamina::SymbolicPolyCoeff>(
361 simp, *factor_variables.begin());
362 if (poly.degree() == 2) {
363 std::string var = *factor_variables.begin();
364 auto solutions =
365 lamina::detail::propagate_result(
366 lamina::solve_finite_checked(simp, var, context));
367 if (solutions.size() == 2) {
368 auto leading = poly.coeffs[2].val;
369 if (!leading) leading = number(1);
370 leading = leading->simplify();
371
372 auto x_node = lamina::detail::make_expression_ptr(lamina::detail::make_node<VariableNode>(var));
373
374 auto t1 = SymbolicExpr::add(x_node, multiply(solutions[0], number(-1)))->simplify();
375 auto t2 = SymbolicExpr::add(x_node, multiply(solutions[1], number(-1)))->simplify();
376
377 std::vector<std::shared_ptr<const SymbolicNode>> factors;
378 if (!leading->is_one()) factors.push_back(lamina::detail::node(leading));
379 factors.push_back(lamina::detail::node(t1));
380 factors.push_back(lamina::detail::node(t2));
381
382 return lamina::detail::make_expression_ptr(lamina::detail::make_node<MultiplyNode>(factors));
383 }
384 }
385 } catch (const std::invalid_argument&) {
386 } catch (const std::out_of_range&) {
387 }
388 }
389
390 /// 步骤 3:多元多项式 - 先尝试 factor_multivariate,再回退逐变量分解
391 if (factor_variables.size() > 1) {
392 /// 3a: 检测是否为多项式表达式,若是则使用 MultiPoly 路径
393 if (is_poly_expr_node(lamina::detail::node(simp))) {
394 try {
395 std::vector<std::string> var_list(
396 factor_variables.begin(), factor_variables.end());
397 auto mpoly = symbolic_node_to_multipoly(lamina::detail::node(simp), var_list);
398
399 if (!mpoly.is_zero() && !mpoly.is_constant()) {
400 const auto& checked =
401 lamina::detail::propagate_result(
402 lamina::factor_multivariate_checked(
403 mpoly, context));
404 if (checked.completeness !=
405 lamina::Completeness::Complete) {
406 return simp;
407 }
408 const auto& result = checked.value;
409
410 /// 检查分解是否产生了多个因子
411 if (!result.factors.empty()) {
412 std::vector<std::shared_ptr<const SymbolicNode>> factor_nodes;
413
414 /// 常数因子
415 if (!(result.constant == Rational(1))) {
416 factor_nodes.push_back(
417 lamina::detail::node(SymbolicExpr::number(result.constant)));
418 }
419
420 /// 各不可约因子
421 for (size_t i = 0; i < result.factors.size(); ++i) {
422 auto factor_expr = multipoly_to_symbolic(result.factors[i]);
423 if (!factor_expr || factor_expr->is_zero()) continue;
424
425 int mult = (i < result.multiplicities.size())
426 ? result.multiplicities[i] : 1;
427
428 if (mult == 1) {
429 factor_nodes.push_back(lamina::detail::node(factor_expr));
430 } else {
431 auto pow_expr = SymbolicExpr::power(
432 factor_expr, SymbolicExpr::number(mult));
433 factor_nodes.push_back(lamina::detail::node(pow_expr));
434 }
435 }
436
437 if (factor_nodes.size() > 1) {
438 return lamina::detail::make_expression_ptr(
439 lamina::detail::make_node<MultiplyNode>(std::move(factor_nodes)));
440 } else if (factor_nodes.size() == 1) {
441 return lamina::detail::make_expression_ptr(factor_nodes[0]);
442 }
443 }
444 }
445 } catch (const std::invalid_argument&) {
446 /// MultiPoly conversion failed; use the variable fallback.
447 } catch (const std::out_of_range&) {
448 /// MultiPoly conversion failed; use the variable fallback.
449 }
450 }
451
452 /// 3b: 回退 - 逐变量尝试有理根分解
453 for (const auto& var : factor_variables) {
454 try {
455 auto poly = lamina::symbolic_to_poly<lamina::SymbolicPolyCoeff>(simp, var);
456 if (poly.degree() >= 2) {
457 /// 尝试用 Rational 系数做分解
458 auto poly_r = lamina::symbolic_to_poly<Rational>(simp, var);
459 auto roots = lamina::find_rational_roots(poly_r);
460 if (!roots.empty()) {
461 auto x_node = lamina::detail::make_expression_ptr(lamina::detail::make_node<VariableNode>(var));
462 std::vector<std::shared_ptr<const SymbolicNode>> factors;
463
464 auto leading = poly_r.lead_coeff();
465 if (!(leading == Rational(1))) {
466 factors.push_back(lamina::detail::node(number(leading)));
467 }
468
469 for (const auto& r : roots) {
470 std::shared_ptr<SymbolicExpr> linear_factor;
471 if (r == Rational(0)) {
472 linear_factor = x_node;
473 } else {
474 auto neg_r = number(Rational(0) - r);
475 linear_factor = SymbolicExpr::add(x_node, neg_r)->simplify();
476 }
477 factors.push_back(lamina::detail::node(linear_factor));
478 }
479
480 /// 计算余下因子
481 lamina::Polynomial<Rational> factored_product({leading}, var);
482 for (const auto& r : roots) {
483 lamina::Polynomial<Rational> lin({Rational(0) - r, Rational(1)}, var);
484 factored_product = factored_product * lin;
485 }
486 auto [quotient, remainder] = poly_r.div_mod(factored_product);
487 if (remainder.is_zero() && quotient.degree() >= 1) {
488 auto q_expr = lamina::poly_to_symbolic(quotient)->simplify();
489 factors.push_back(lamina::detail::node(q_expr));
490 } else if (remainder.is_zero() && !quotient.is_zero() && quotient.degree() == 0) {
491 /// 常数商 -> 已完全分解
492 if (!(quotient.coeffs[0] == Rational(1))) {
493 factors.push_back(
494 lamina::detail::node(number(quotient.coeffs[0])));
495 }
496 }
497
498 if (factors.size() > 1) {
499 return lamina::detail::make_expression_ptr(lamina::detail::make_node<MultiplyNode>(factors));
500 }
501 }
502 }
503 } catch (const std::invalid_argument&) {
504 continue;
505 } catch (const std::out_of_range&) {
506 continue;
507 }
508 }
509 }
510 }
511
512 /// 标准多项式分解未匹配时,后备路径检测超越函数并尝试超越因式分解.
513 {
514 const auto transform_variables =
515 lamina::free_variables(lamina::detail::node(simp));
516 std::string target_var;
517 if (!transform_variables.empty()) {
518 target_var = *transform_variables.begin();
519 }
520
521 if (!target_var.empty()) {
522 try {
523 auto trans_factors = lamina::factor_transcendental(simp, target_var);
524 if (trans_factors.size() > 1) {
525 /// 超越分解成功:组装乘积表达式
526 std::vector<std::shared_ptr<const SymbolicNode>> factor_nodes;
527 factor_nodes.reserve(trans_factors.size());
528 for (const auto& f : trans_factors) {
529 if (f && lamina::detail::node(f)) {
530 factor_nodes.push_back(lamina::detail::node(f));
531 }
532 }
533 if (factor_nodes.size() > 1) {
534 return lamina::detail::make_expression_ptr(
535 lamina::detail::make_node<MultiplyNode>(std::move(factor_nodes)));
536 }
537 }
538 } catch (const std::invalid_argument&) {
539 /// Transcendental conversion did not apply.
540 } catch (const std::out_of_range&) {
541 /// Transcendental conversion did not apply.
542 }
543 }
544 }
545
546 return simp;
547}

549lamina::ExpressionResult SymbolicExpr::factor_checked(
550 lamina::ComputationContext& context) const
551{
552 constexpr const char* operation = "factor";
553 auto budget = context.consume_steps(1, operation);
554 if (!budget) return lamina::ExpressionResult::failure(budget.error());
555 try {
556 return lamina::ExpressionResult::success(factor_impl(context));
557 } catch (const lamina::detail::ResultPropagation& propagation) {
558 return lamina::ExpressionResult::failure(propagation.error());
559 } catch (const std::bad_alloc&) {
560 return lamina::ExpressionResult::failure(
561 lamina::CasErrc::ResourceLimit,
562 "allocation failed while factoring expression", operation);
563 } catch (const std::exception& ex) {
564 return lamina::ExpressionResult::failure(
565 lamina::CasErrc::InternalInvariant, ex.what(), operation);
566 }
567}

570{

575std::shared_ptr<SymbolicExpr> SymbolicExpr::cancel() const {
576 auto simp = simplify();
577 if (!simp || !lamina::detail::node(simp)) return simp;
578
579 /// 辅助 lambda:从乘积节点中分离分子因子和分母因子.
580 /// 分母因子 = 含负指数的 PowerNode.
581 auto separate_num_den = [](const std::shared_ptr<const SymbolicNode>& node,
582 std::vector<std::shared_ptr<const SymbolicNode>>& num_out,
583 std::vector<std::shared_ptr<const SymbolicNode>>& den_out) {
584 auto classify = [&](const std::shared_ptr<const SymbolicNode>& factor) {
585 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(factor)) {
586 if (auto exp_num = std::dynamic_pointer_cast<const NumberNode>(pow->exponent())) {
587 bool is_negative = false;
588 if (std::holds_alternative<BigInt>(exp_num->value())) {
589 is_negative = std::get<BigInt>(exp_num->value()).IsNegative();
590 } else if (std::holds_alternative<lmmc_real_t>(exp_num->value())) {
591 is_negative = std::get<lmmc_real_t>(exp_num->value()) < 0;
592 } else if (std::holds_alternative<Rational>(exp_num->value())) {
593 is_negative = std::get<Rational>(exp_num->value()).get_numerator().IsNegative();
594 }
595 if (is_negative) {
596 /// 取绝对值指数
597 std::shared_ptr<const SymbolicNode> pos_exp;
598 if (std::holds_alternative<BigInt>(exp_num->value())) {
599 pos_exp = lamina::detail::make_node<NumberNode>(BigInt(0) - std::get<BigInt>(exp_num->value()));
600 } else if (std::holds_alternative<lmmc_real_t>(exp_num->value())) {
601 pos_exp = lamina::detail::make_node<NumberNode>(-std::get<lmmc_real_t>(exp_num->value()));
602 } else {
603 auto r = std::get<Rational>(exp_num->value());
604 pos_exp = lamina::detail::make_node<NumberNode>(Rational(BigInt(0) - r.get_numerator(), r.get_denominator()));
605 }
606 auto pos_exp_num = std::dynamic_pointer_cast<const NumberNode>(pos_exp);
607 if (pos_exp_num && pos_exp_num->is_one()) {
608 den_out.push_back(pow->base());
609 } else {
610 den_out.push_back(lamina::detail::make_node<PowerNode>(pow->base(), pos_exp));
611 }
612 return;
613 }
614 }
615 }
616 num_out.push_back(factor);
617 };
618
619 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
620 for (const auto& op : mul->operands()) {
621 classify(op);
622 }
623 } else {
624 classify(node);
625 }
626 };

628 /// 辅助 lambda:从因子列表构建乘积表达式
629 auto build_product = [](const std::vector<std::shared_ptr<const SymbolicNode>>& factors) -> std::shared_ptr<SymbolicExpr> {
630 if (factors.empty()) return SymbolicExpr::number(1);
631 if (factors.size() == 1) return lamina::detail::make_expression_ptr(factors[0]);
632 return lamina::detail::make_expression_ptr(lamina::detail::make_node<MultiplyNode>(factors));
633 };

635 /// 策略 1:表达式是 AddNode,各项可能含公共分母因子.
636 /// 例如 simplify 后 (x^2-1)/(x-1) 变为 -1*(x-1)^-1 + x^2*(x-1)^-1
637 /// 需要提取公共分母,重组为 (分子之和)/分母 再做 GCD 约分.
638 if (auto add_node = std::dynamic_pointer_cast<const AddNode>(lamina::detail::node(simp))) {
639 /// 对每个加法项分离分子/分母
640 struct TermInfo {
641 std::shared_ptr<SymbolicExpr> numerator;
642 std::shared_ptr<SymbolicExpr> denominator;
643 };
644 std::vector<TermInfo> terms;
645 bool has_denominator = false;
646
647 for (const auto& term : add_node->operands()) {
648 std::vector<std::shared_ptr<const SymbolicNode>> t_num, t_den;
649 separate_num_den(term, t_num, t_den);
650 auto num_expr = build_product(t_num)->simplify();
651 auto den_expr = build_product(t_den)->simplify();
652 if (!den_expr->is_one()) has_denominator = true;
653 terms.push_back({num_expr, den_expr});
654 }
655
656 if (has_denominator) {
657 /// 计算公共分母(所有项分母的 LCM,简化处理:乘积)
658 /// 对于常见情况(所有项分母相同),直接提取.
659 /// 检查是否所有分母相同
660 bool all_same_den = true;
661 auto first_den_str = terms[0].denominator->to_string();
662 for (size_t i = 1; i < terms.size(); ++i) {
663 if (terms[i].denominator->to_string() != first_den_str) {
664 all_same_den = false;
665 break;
666 }
667 }
668
669 std::shared_ptr<SymbolicExpr> combined_num;
670 std::shared_ptr<SymbolicExpr> combined_den;
671
672 if (all_same_den) {
673 /// 所有项分母相同:分子直接相加
674 combined_den = terms[0].denominator;
675 std::vector<std::shared_ptr<const SymbolicNode>> num_ops;
676 for (const auto& t : terms) {
677 if (t.numerator && lamina::detail::node(t.numerator)) {
678 num_ops.push_back(lamina::detail::node(t.numerator));
679 }
680 }
681 if (num_ops.empty()) return SymbolicExpr::number(0);
682 if (num_ops.size() == 1) combined_num = lamina::detail::make_expression_ptr(num_ops[0]);
683 else combined_num = lamina::detail::make_expression_ptr(lamina::detail::make_node<AddNode>(num_ops));
684 combined_num = combined_num->simplify();
685 } else {
686 /// 分母不同:通分(乘以其他项的分母)
687 /// 计算总分母 = 所有不同分母的乘积
688 std::shared_ptr<SymbolicExpr> total_den = SymbolicExpr::number(1);
689 /// 收集不同的分母
690 std::vector<std::shared_ptr<SymbolicExpr>> unique_dens;
691 for (const auto& t : terms) {
692 if (!t.denominator->is_one()) {
693 bool found = false;
694 for (const auto& ud : unique_dens) {
695 if (ud->to_string() == t.denominator->to_string()) {
696 found = true;
697 break;
698 }
699 }
700 if (!found) {
701 unique_dens.push_back(t.denominator);
702 }
703 }
704 }
705 for (const auto& ud : unique_dens) {
706 total_den = SymbolicExpr::multiply(total_den, ud)->simplify();
707 }
708 combined_den = total_den;
709
710 /// 每项分子乘以 (总分母 / 该项分母)
711 std::vector<std::shared_ptr<const SymbolicNode>> num_ops;
712 for (const auto& t : terms) {
713 auto factor = divide(total_den, t.denominator)->simplify();
714 auto adjusted_num = SymbolicExpr::multiply(t.numerator, factor)->simplify();
715 if (adjusted_num && lamina::detail::node(adjusted_num)) {
716 num_ops.push_back(lamina::detail::node(adjusted_num));
717 }
718 }
719 if (num_ops.empty()) return SymbolicExpr::number(0);
720 if (num_ops.size() == 1) combined_num = lamina::detail::make_expression_ptr(num_ops[0]);
721 else combined_num = lamina::detail::make_expression_ptr(lamina::detail::make_node<AddNode>(num_ops));
722 combined_num = combined_num->expand()->simplify();
723 }
724
725 /// 对 combined_num / combined_den 做多项式 GCD 约分
726 combined_den = combined_den->expand()->simplify();
727
728 auto rational_variables =
729 lamina::free_variables(lamina::detail::node(combined_num));
730 const auto denominator_variables =
731 lamina::free_variables(lamina::detail::node(combined_den));
732 rational_variables.insert(denominator_variables.begin(),
733 denominator_variables.end());
734
735 auto cur_num = combined_num;
736 auto cur_den = combined_den;
737
738 for (const auto& var : rational_variables) {
739 try {
740 auto num_expanded = cur_num->expand()->simplify();
741 auto den_expanded = cur_den->expand()->simplify();
742
743 /// 使用 SymbolicPolyCoeff 保留其他变量作为符号系数
744 auto poly_num = lamina::symbolic_to_poly<lamina::SymbolicPolyCoeff>(num_expanded, var);
745 auto poly_den = lamina::symbolic_to_poly<lamina::SymbolicPolyCoeff>(den_expanded, var);
746
747 if (poly_num.is_zero()) return SymbolicExpr::number(0);
748 if (poly_den.is_zero()) return simp;
749
750 /// 尝试用 Rational 系数做 GCD(纯数值系数时更可靠)
751 auto poly_num_r = lamina::symbolic_to_poly<Rational>(num_expanded, var);
752 auto poly_den_r = lamina::symbolic_to_poly<Rational>(den_expanded, var);
753
754 /// 检查 Rational 转换是否丢失了信息(多元情况)
755 bool rational_ok = true;
756 auto reconstructed_num = lamina::poly_to_symbolic(poly_num_r)->expand()->simplify();
757 auto reconstructed_den = lamina::poly_to_symbolic(poly_den_r)->expand()->simplify();
758 auto diff_num = SymbolicExpr::add(num_expanded, SymbolicExpr::multiply(SymbolicExpr::number(-1), reconstructed_num))->simplify();
759 auto diff_den = SymbolicExpr::add(den_expanded, SymbolicExpr::multiply(SymbolicExpr::number(-1), reconstructed_den))->simplify();
760 if (!diff_num->is_zero() || !diff_den->is_zero()) {
761 rational_ok = false;
762 }
763
764 if (rational_ok) {
765 auto g = lamina::Polynomial<Rational>::gcd(poly_num_r, poly_den_r);
766 if (g.degree() >= 1) {
767 auto [q_num, r_num] = poly_num_r.div_mod(g);
768 auto [q_den, r_den] = poly_den_r.div_mod(g);
769 cur_num = lamina::poly_to_symbolic(q_num)->simplify();
770 cur_den = lamina::poly_to_symbolic(q_den)->simplify();
771 }
772 } else {
773 /// 多元情况:使用 SymbolicPolyCoeff 做 GCD
774 auto g = lamina::Polynomial<lamina::SymbolicPolyCoeff>::gcd(poly_num, poly_den);
775 if (g.degree() >= 1) {
776 auto [q_num, r_num] = poly_num.div_mod(g);
777 auto [q_den, r_den] = poly_den.div_mod(g);
778 cur_num = lamina::poly_to_symbolic(q_num)->simplify();
779 cur_den = lamina::poly_to_symbolic(q_den)->simplify();
780 }
781 }
782 } catch (const std::invalid_argument&) {
783 continue;
784 } catch (const std::out_of_range&) {
785 continue;
786 } catch (const std::runtime_error&) {
787 continue;
788 }
789 }
790
791 if (cur_den->is_one()) return cur_num;
792
793 /// 分母为 -1 时取负
794 if (lamina::detail::node(cur_den)) {
795 auto diff = SymbolicExpr::add(lamina::detail::make_expression_ptr(lamina::detail::node(cur_den)), SymbolicExpr::number(-1))->simplify();
796 if (diff->is_zero()) {
797 return SymbolicExpr::multiply(SymbolicExpr::number(-1), cur_num)->simplify();
798 }
799 }
800
801 return divide(cur_num, cur_den)->simplify();
802 }
803 }

805 /// 策略 2:表达式是 MultiplyNode,直接分离分子/分母.
806 std::vector<std::shared_ptr<const SymbolicNode>> num_factors;
807 std::vector<std::shared_ptr<const SymbolicNode>> den_factors;
808 separate_num_den(lamina::detail::node(simp), num_factors, den_factors);
