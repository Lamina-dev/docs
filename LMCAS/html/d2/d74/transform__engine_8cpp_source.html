22static bool te_contains_var(const std::shared_ptr<const SymbolicNode>& node,
23 const std::string& var) {
24 if (!node) return false;
25 if (auto vn = std::dynamic_pointer_cast<const VariableNode>(node))
26 return vn->name() == var;
27 if (auto an = std::dynamic_pointer_cast<const AddNode>(node)) {
28 for (const auto& op : an->operands())
29 if (te_contains_var(op, var)) return true;
30 return false;
31 }
32 if (auto mn = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
33 for (const auto& op : mn->operands())
34 if (te_contains_var(op, var)) return true;
35 return false;
36 }
37 if (auto pn = std::dynamic_pointer_cast<const PowerNode>(node))
38 return te_contains_var(pn->base(), var) ||
39 te_contains_var(pn->exponent(), var);
40 if (auto fn = std::dynamic_pointer_cast<const FunctionNode>(node)) {
41 for (const auto& arg : fn->arguments())
42 if (te_contains_var(arg, var)) return true;
43 return false;
44 }
45 if (auto tn = std::dynamic_pointer_cast<const TransformNode>(node))
46 return te_contains_var(tn->body(), var);
47 return false;
48}

55 const std::string& input_var,

155static std::pair<std::shared_ptr<SymbolicExpr>,std::shared_ptr<SymbolicExpr>>
156te_split_coeff(const std::shared_ptr<SymbolicExpr>& e, const std::string& v) {
157 if (!e || !lamina::detail::node(e) || !te_depends_on(e, v)) return {e ? e : SymbolicExpr::number(1), SymbolicExpr::number(1)};
158 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(e));
159 if (!mul) return {SymbolicExpr::number(1), e};
160 std::vector<std::shared_ptr<const SymbolicNode>> cp, vp;
161 for (auto& op : mul->operands()) {
162 if (te_contains_var(op, v)) vp.push_back(op); else cp.push_back(op);
163 }
164 if (cp.empty()) return {SymbolicExpr::number(1), e};
165 if (vp.empty()) return {e, SymbolicExpr::number(1)};
166 auto c = (cp.size()==1) ? lamina::detail::make_expression_ptr(cp[0])
167 : lamina::detail::make_expression_ptr(lamina::detail::make_node<MultiplyNode>(cp));
168 auto b = (vp.size()==1) ? lamina::detail::make_expression_ptr(vp[0])
169 : lamina::detail::make_expression_ptr(lamina::detail::make_node<MultiplyNode>(vp));
170 return {c, b};
171}

173static bool te_is_power_of_var(const std::shared_ptr<SymbolicExpr>& e, const std::string& v, int& n) {
174 if (!e || !lamina::detail::node(e)) return false;
175 auto vn = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(e));
176 if (vn && vn->name() == v) { n = 1; return true; }
177 auto pw = std::dynamic_pointer_cast<const PowerNode>(lamina::detail::node(e));
178 if (!pw) return false;
179 auto bv = std::dynamic_pointer_cast<const VariableNode>(pw->base());
180 if (!bv || bv->name() != v) return false;
181 auto en = std::dynamic_pointer_cast<const NumberNode>(pw->exponent());
182 if (!en) return false;
183 double val = te_numval(pw->exponent());
184 int iv = static_cast<int>(val);
185 if (std::abs(val - iv) < 1e-12 && iv >= 0) { n = iv; return true; }
186 return false;
187}
188static bool te_is_trig(const std::shared_ptr<SymbolicExpr>& e, const std::string& v,
189 bool& is_sin, std::shared_ptr<SymbolicExpr>& freq) {
190 if (!e || !lamina::detail::node(e)) return false;
191 auto fn = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(e));
192 if (!fn || fn->arguments().empty()) return false;
193 if (fn->type() != FunctionNode::FuncType::Sin && fn->type() != FunctionNode::FuncType::Cos) return false;
194 is_sin = (fn->type() == FunctionNode::FuncType::Sin);
195 auto arg = fn->arguments()[0];
196 auto ml = std::dynamic_pointer_cast<const MultiplyNode>(arg);
197 if (ml && ml->operands().size() == 2) {
198 for (size_t i = 0; i < 2; ++i) {
199 auto vn = std::dynamic_pointer_cast<const VariableNode>(ml->operands()[i]);
200 if (vn && vn->name() == v && !te_contains_var(ml->operands()[1-i], v)) {
201 freq = lamina::detail::make_expression_ptr(ml->operands()[1-i]); return true;
202 }
203 }
204 }
205 auto vn = std::dynamic_pointer_cast<const VariableNode>(arg);
206 if (vn && vn->name() == v) { freq = SymbolicExpr::number(1); return true; }
207 return false;
208}

231static bool te_extract_exp(const std::shared_ptr<SymbolicExpr>& e, const std::string& t,
232 std::shared_ptr<SymbolicExpr>& a_out, std::shared_ptr<SymbolicExpr>& rem_out) {
233 if (!e || !lamina::detail::node(e)) return false;
234 auto fn = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(e));
235 if (fn && fn->type() == FunctionNode::FuncType::Exp && !fn->arguments().empty()) {
236 auto arg = fn->arguments()[0];
237 auto ml = std::dynamic_pointer_cast<const MultiplyNode>(arg);
238 if (ml && ml->operands().size() == 2) {
239 for (size_t i = 0; i < 2; ++i) {
240 auto vn = std::dynamic_pointer_cast<const VariableNode>(ml->operands()[i]);
241 if (vn && vn->name() == t && !te_contains_var(ml->operands()[1-i], t)) {
242 a_out = lamina::detail::make_expression_ptr(ml->operands()[1-i]);
243 rem_out = SymbolicExpr::number(1); return true;
244 }
245 }
246 }
247 auto vn = std::dynamic_pointer_cast<const VariableNode>(arg);
248 if (vn && vn->name() == t) { a_out = SymbolicExpr::number(1); rem_out = SymbolicExpr::number(1); return true; }
249 }
250 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(e));
251 if (mul) {
252 for (size_t i = 0; i < mul->operands().size(); ++i) {
253 auto fac = lamina::detail::make_expression_ptr(mul->operands()[i]);
254 std::shared_ptr<SymbolicExpr> at, rt;
255 if (te_extract_exp(fac, t, at, rt)) {
256 std::vector<std::shared_ptr<const SymbolicNode>> rest;
257 for (size_t j = 0; j < mul->operands().size(); ++j)
258 if (j != i) rest.push_back(mul->operands()[j]);
259 if (rest.empty()) rem_out = SymbolicExpr::number(1);
260 else if (rest.size() == 1) rem_out = lamina::detail::make_expression_ptr(rest[0]);
261 else rem_out = lamina::detail::make_expression_ptr(lamina::detail::make_node<MultiplyNode>(std::move(rest)));
262 a_out = at; return true;
263 }
264 }
265 }
266 return false;
267}

273static std::shared_ptr<SymbolicExpr> te_laplace_lookup(
274 const std::shared_ptr<SymbolicExpr>& f, const std::string& t, const std::string& s) {
275 if (!f || !lamina::detail::node(f)) return nullptr;
276 auto sv = SymbolicExpr::variable(s);
277 if (!te_depends_on(f, t)) return SymbolicExpr::divide(f, sv);
278 int n = 0;
279 if (te_is_power_of_var(f, t, n)) {
280 return SymbolicExpr::divide(SymbolicExpr::number(static_cast<long long>(te_factorial(n))),
281 SymbolicExpr::power(sv, SymbolicExpr::number(n + 1)));
282 }
283 bool is_sin = false; std::shared_ptr<SymbolicExpr> freq;
284 if (te_is_trig(f, t, is_sin, freq)) {
285 auto den = SymbolicExpr::add(SymbolicExpr::power(sv, SymbolicExpr::number(2)), SymbolicExpr::power(freq, SymbolicExpr::number(2)));
286 return is_sin ? SymbolicExpr::divide(freq, den) : SymbolicExpr::divide(sv, den);
287 }
288 bool is_sinh = false; std::shared_ptr<SymbolicExpr> hf;
289 if (te_is_hyp(f, t, is_sinh, hf)) {
290 auto den = SymbolicExpr::add(SymbolicExpr::power(sv, SymbolicExpr::number(2)), SymbolicExpr::multiply(SymbolicExpr::number(-1), SymbolicExpr::power(hf, SymbolicExpr::number(2))));
291 return is_sinh ? SymbolicExpr::divide(hf, den) : SymbolicExpr::divide(sv, den);
292 }
293 std::shared_ptr<SymbolicExpr> a, rem;
294 if (te_extract_exp(f, t, a, rem) && !te_depends_on(rem, t))
295 return SymbolicExpr::divide(rem, SymbolicExpr::add(sv, SymbolicExpr::multiply(SymbolicExpr::number(-1), a)));
296 return nullptr;
297}

299static std::vector<std::shared_ptr<SymbolicExpr>> te_laplace_roc(
300 const std::shared_ptr<SymbolicExpr>& f,
301 const std::string& t,
302 const std::string& s) {
303 if (!f || !lamina::detail::node(f)) return {};
304 auto sv = SymbolicExpr::variable(s);
305 if (!te_depends_on(f, t)) {
306 auto condition = te_gt_condition(sv, SymbolicExpr::number(0));
307 return condition ? std::vector<std::shared_ptr<SymbolicExpr>>{condition}
308 : std::vector<std::shared_ptr<SymbolicExpr>>{};
309 }
310 int n = 0;
311 if (te_is_power_of_var(f, t, n)) {
312 auto condition = te_gt_condition(sv, SymbolicExpr::number(0));
313 return condition ? std::vector<std::shared_ptr<SymbolicExpr>>{condition}
314 : std::vector<std::shared_ptr<SymbolicExpr>>{};
315 }
316 bool is_sin = false;
317 std::shared_ptr<SymbolicExpr> freq;
318 if (te_is_trig(f, t, is_sin, freq)) {
319 auto condition = te_gt_condition(sv, SymbolicExpr::number(0));
320 return condition ? std::vector<std::shared_ptr<SymbolicExpr>>{condition}
321 : std::vector<std::shared_ptr<SymbolicExpr>>{};
322 }
323 std::shared_ptr<SymbolicExpr> a;
324 std::shared_ptr<SymbolicExpr> rem;
325 if (te_extract_exp(f, t, a, rem) && !te_depends_on(rem, t)) {
326 auto condition = te_gt_condition(sv, a);
327 return condition ? std::vector<std::shared_ptr<SymbolicExpr>>{condition}
328 : std::vector<std::shared_ptr<SymbolicExpr>>{};
329 }
330 return {};
331}

333std::shared_ptr<SymbolicExpr> laplace_transform(
334 const std::shared_ptr<SymbolicExpr>& f, const std::string& t, const std::string& s) {
335 if (!f || !lamina::detail::node(f)) return nullptr;
336 if (te_is_zero(f)) return SymbolicExpr::number(0);
337 if (!te_depends_on(f, t)) return SymbolicExpr::divide(f, SymbolicExpr::variable(s));
338 auto add = std::dynamic_pointer_cast<const AddNode>(lamina::detail::node(f));
339 if (add) {
340 std::shared_ptr<SymbolicExpr> result;
341 for (auto& op : add->operands()) {
342 auto lt = laplace_transform(lamina::detail::make_expression_ptr(op), t, s);
343 if (!lt) return te_unevaluated_laplace(f, t, s);
344 result = result ? SymbolicExpr::add(result, lt) : lt;
345 }
346 return result;
347 }
348 auto [coeff, body] = te_split_coeff(f, t);
349 if (!coeff->is_one()) {
350 auto lt_body = laplace_transform(body, t, s);
351 if (lt_body) return SymbolicExpr::multiply(coeff, lt_body);
352 }
353 std::shared_ptr<SymbolicExpr> a_shift, remainder;
354 if (te_extract_exp(f, t, a_shift, remainder)) {
355 if (te_depends_on(remainder, t)) {
356 auto lt_rem = laplace_transform(remainder, t, s);
357 if (lt_rem) return lt_rem->substitute(s, SymbolicExpr::add(SymbolicExpr::variable(s), SymbolicExpr::multiply(SymbolicExpr::number(-1), a_shift)));
358 } else {
359 return SymbolicExpr::divide(remainder, SymbolicExpr::add(SymbolicExpr::variable(s), SymbolicExpr::multiply(SymbolicExpr::number(-1), a_shift)));
360 }
361 }
362 auto lookup = te_laplace_lookup(f, t, s);
363 if (lookup) return lookup;
364 return te_unevaluated_laplace(f, t, s);
365}

401static std::shared_ptr<SymbolicExpr> te_inv_power(
402 const std::shared_ptr<SymbolicExpr>& F, const std::string& s, const std::string& t) {
403 if (!F || !lamina::detail::node(F)) return nullptr;
404 auto tv = SymbolicExpr::variable(t);
405 auto pw = std::dynamic_pointer_cast<const PowerNode>(lamina::detail::node(F));
406 if (!pw) return nullptr;
407 auto en = std::dynamic_pointer_cast<const NumberNode>(pw->exponent());
408 if (!en) return nullptr;
409 double ev = te_numval(pw->exponent());
410 if (ev >= 0) return nullptr;
411 int n = static_cast<int>(-ev);
412 if (std::abs(ev + n) > 1e-12 || n < 1) return nullptr;
413 auto make_exp = [&](const std::shared_ptr<SymbolicExpr>& a) {
414 auto arg = SymbolicExpr::multiply(a, tv);
415 return lamina::detail::make_expression_ptr(lamina::detail::make_node<FunctionNode>(
416 FunctionNode::FuncType::Exp,
417 std::vector<std::shared_ptr<const SymbolicNode>>{lamina::detail::node(arg)}));
418 };
419 auto bv = std::dynamic_pointer_cast<const VariableNode>(pw->base());
420 if (bv && bv->name() == s) {
421 if (n == 1) return SymbolicExpr::number(1);
422 return SymbolicExpr::divide(SymbolicExpr::power(tv, SymbolicExpr::number(n-1)),
423 SymbolicExpr::number(static_cast<long long>(te_factorial(n-1))));
424 }
425 auto ba = std::dynamic_pointer_cast<const AddNode>(pw->base());
426 if (ba && ba->operands().size() == 2) {
427 std::shared_ptr<SymbolicExpr> pole;
428 for (size_t i = 0; i < 2; ++i) {
429 auto vn = std::dynamic_pointer_cast<const VariableNode>(ba->operands()[i]);
430 if (vn && vn->name() == s) {
431 auto c = lamina::detail::make_expression_ptr(ba->operands()[1-i]);
432 pole = SymbolicExpr::multiply(SymbolicExpr::number(-1), c); break;
433 }
434 }
435 if (pole) {
436 if (n == 1) {
437 if (te_is_zero(pole)) return SymbolicExpr::number(1);
438 return make_exp(pole);
439 }
440 auto tp = SymbolicExpr::power(tv, SymbolicExpr::number(n-1));
441 auto fv = SymbolicExpr::number(static_cast<long long>(te_factorial(n-1)));
442 if (te_is_zero(pole)) return SymbolicExpr::divide(tp, fv);
443 return SymbolicExpr::divide(SymbolicExpr::multiply(tp, make_exp(pole)), fv);
444 }
445 }
446 return nullptr;
447}

449static std::shared_ptr<SymbolicExpr> te_inv_product(
450 const std::shared_ptr<SymbolicExpr>& F, const std::string& s, const std::string& t) {
451 if (!F || !lamina::detail::node(F)) return nullptr;
452 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(F));
453 if (!mul || mul->operands().size() < 2) return nullptr;
454 std::vector<std::shared_ptr<const SymbolicNode>> num_parts;
455 std::shared_ptr<const SymbolicNode> den_part = nullptr;
456 for (auto& op : mul->operands()) {
457 auto pw = std::dynamic_pointer_cast<const PowerNode>(op);
458 if (pw && !den_part) {
459 double ev = te_numval(pw->exponent());
460 if (ev < 0 && te_contains_var(pw->base(), s)) { den_part = op; continue; }
461 }
462 num_parts.push_back(op);
463 }
464 if (!den_part) return nullptr;
465 std::shared_ptr<SymbolicExpr> num_expr;
466 if (num_parts.empty()) num_expr = SymbolicExpr::number(1);
467 else if (num_parts.size() == 1) num_expr = lamina::detail::make_expression_ptr(num_parts[0]);
468 else num_expr = lamina::detail::make_expression_ptr(lamina::detail::make_node<MultiplyNode>(num_parts));
469 if (!te_depends_on(num_expr, s)) {
470 auto inv = te_inv_power(lamina::detail::make_expression_ptr(den_part), s, t);
471 if (inv) return SymbolicExpr::multiply(num_expr, inv);
472 }
473 return nullptr;
474}

476std::shared_ptr<SymbolicExpr> inverse_laplace(
477 const std::shared_ptr<SymbolicExpr>& F, const std::string& s, const std::string& t) {
478 if (!F || !lamina::detail::node(F)) return nullptr;
479 if (te_is_zero(F)) return SymbolicExpr::number(0);
480 auto add = std::dynamic_pointer_cast<const AddNode>(lamina::detail::node(F));
481 if (add) {

485 if (!ilt) return te_unevaluated_inv_laplace(F, s, t);
486 result = result ? SymbolicExpr::add(result, ilt) : ilt;
487 }
488 return result;
489 }
490 auto [coeff, body] = te_split_coeff(F, s);
491 if (!coeff->is_one()) {
492 auto ilt_body = inverse_laplace(body, s, t);
493 if (ilt_body) return SymbolicExpr::multiply(coeff, ilt_body);
494 }
495 auto pw_res = te_inv_power(F, s, t);
496 if (pw_res) return pw_res;
497 auto prod_res = te_inv_product(F, s, t);
498 if (prod_res) return prod_res;
499 return te_unevaluated_inv_laplace(F, s, t);
500}

533/**
534 * @internal
535 * @brief 检测 e^(-a*t^2) 高斯形式并提取系数 a > 0。
536 */
537static bool te_is_gaussian(const std::shared_ptr<SymbolicExpr>& f, const std::string& t,
538 double& a_out) {
539 if (!f || !lamina::detail::node(f)) return false;
540 auto fn = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(f));
541 if (!fn || fn->type() != FunctionNode::FuncType::Exp || fn->arguments().empty()) return false;
542 auto arg = fn->arguments()[0];
543 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(arg);
544 if (!mul) return false;
545 double coeff_val = 1.0;
546 bool has_t_sq = false;
547 for (auto& op : mul->operands()) {
548 auto nd = std::dynamic_pointer_cast<const NumberNode>(op);
549 if (nd) { coeff_val *= te_numval(op); continue; }
550 auto pw = std::dynamic_pointer_cast<const PowerNode>(op);
551 if (pw) {
552 auto bv = std::dynamic_pointer_cast<const VariableNode>(pw->base());
553 double ev = te_numval(pw->exponent());
554 if (bv && bv->name() == t && std::abs(ev - 2.0) < 1e-12) {
555 has_t_sq = true; continue;
556 }
557 }
558 return false; // 含有其他因子
559 }
560 if (has_t_sq && coeff_val < 0) {
561 a_out = -coeff_val;
562 return true;
563 }
564 return false;
565}

567/**
568 * @internal
569 * @brief 检测 e^(-a*|t|) 形式并提取系数 a（a 可为符号或数值正数）。
570 */
571static bool te_is_abs_exp(const std::shared_ptr<SymbolicExpr>& f, const std::string& t,
572 std::shared_ptr<SymbolicExpr>& a_out) {
573 if (!f || !lamina::detail::node(f)) return false;
574 auto fn = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(f));
575 if (!fn || fn->type() != FunctionNode::FuncType::Exp || fn->arguments().empty()) return false;
576 auto arg = fn->arguments()[0];
577 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(arg);
578 if (!mul) return false;
579 std::shared_ptr<SymbolicExpr> a_candidate;
580 bool has_abs_t = false;
581 bool has_neg = false;
582 for (auto& op : mul->operands()) {
583 auto fn2 = std::dynamic_pointer_cast<const FunctionNode>(op);
584 if (fn2 && fn2->type() == FunctionNode::FuncType::Abs && fn2->arguments().size() == 1) {
585 auto av = std::dynamic_pointer_cast<const VariableNode>(fn2->arguments()[0]);
586 if (av && av->name() == t) { has_abs_t = true; continue; }
587 }
588 auto nd = std::dynamic_pointer_cast<const NumberNode>(op);
589 if (nd) {
590 double v = te_numval(op);
591 if (v < 0) {
592 has_neg = true;
593 a_candidate = SymbolicExpr::number(-v);
594 } else {
595 a_candidate = lamina::detail::make_expression_ptr(op);
596 }
597 continue;
598 }
599 /// 允许符号变量作为系数（如 -a）
600 auto vn = std::dynamic_pointer_cast<const VariableNode>(op);
601 if (vn) {
602 /// 假设符号变量为正 a，前面有负号处理
603 a_candidate = lamina::detail::make_expression_ptr(op);
604 continue;
605 }
606 return false;
607 }
608 if (has_abs_t && has_neg && a_candidate) {
609 a_out = a_candidate;
610 return true;
611 }
612 return false;
613}

615std::shared_ptr<SymbolicExpr> fourier_transform(
616 const std::shared_ptr<SymbolicExpr>& f, const std::string& t, const std::string& omega) {
617 if (!f || !lamina::detail::node(f)) return nullptr;
618 auto wv = SymbolicExpr::variable(omega);
619
620 /// 不依赖 t：无有限 Fourier 变换（δ函数），返回未求值
621 if (!te_depends_on(f, t)) {
622 return lamina::detail::make_expression_ptr(lamina::detail::make_node<TransformNode>(
623 TransformNode::TransformType::Fourier, lamina::detail::node(f)->clone(), t, omega));
624 }
625
626 /// 线性性：处理加法 F{af+bg} = aF{f} + bF{g}
627 auto add_node = std::dynamic_pointer_cast<const AddNode>(lamina::detail::node(f));
628 if (add_node) {
629 std::shared_ptr<SymbolicExpr> result;
630 for (auto& op : add_node->operands()) {
631 auto ft = fourier_transform(lamina::detail::make_expression_ptr(op), t, omega);
632 if (!ft) {
633 return lamina::detail::make_expression_ptr(lamina::detail::make_node<TransformNode>(
634 TransformNode::TransformType::Fourier, lamina::detail::node(f)->clone(), t, omega));
635 }
636 result = result ? SymbolicExpr::add(result, ft) : ft;
637 }
638 return result;
639 }
640
641 /// 分离常数系数: c * body
642 auto [coeff, body] = te_split_coeff(f, t);
643 if (!coeff->is_one()) {
644 auto ft_body = fourier_transform(body, t, omega);
645 if (ft_body) return SymbolicExpr::multiply(coeff, ft_body);
646 }
647
648 /// F{e^(-a*t^2)} = sqrt(π/a) * e^(-ω²/(4a))
649 {
650 double a_val = 0.0;
651 if (te_is_gaussian(f, t, a_val)) {
652 double sqrt_pi_over_a = std::sqrt(3.14159265358979323846 / a_val);
653 auto coeff_expr = SymbolicExpr::number(sqrt_pi_over_a);
654 auto neg_w_sq_over_4a = SymbolicExpr::multiply(
655 SymbolicExpr::number(-1.0 / (4.0 * a_val)),
656 SymbolicExpr::power(wv, SymbolicExpr::number(2)));
657 auto exp_part = SymbolicExpr::exp(neg_w_sq_over_4a);
658 return SymbolicExpr::multiply(coeff_expr, exp_part);
659 }
660 }
661
662 /// F{e^(-a|t|)} = 2a/(a² + ω²)
663 {
664 std::shared_ptr<SymbolicExpr> a_expr;
665 if (te_is_abs_exp(f, t, a_expr)) {
666 auto a_sq = SymbolicExpr::power(a_expr, SymbolicExpr::number(2));
667 auto w_sq = SymbolicExpr::power(wv, SymbolicExpr::number(2));
668 auto den = SymbolicExpr::add(a_sq, w_sq);
669 auto num_expr = SymbolicExpr::multiply(SymbolicExpr::number(2), a_expr);
670 return SymbolicExpr::divide(num_expr, den);
671 }
672 }
673
674 /// F{e^(-a*t)} (因果指数，a > 0): 1/(a + iω)
675 {
676 /// 提取指数 e^(arg)，其中 arg 关于 t 线性：arg = c * t（c 不依赖 t）。
677 auto fn = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(f));
678 if (fn && fn->type() == FunctionNode::FuncType::Exp && fn->arguments().size() == 1) {
679 auto arg = lamina::detail::make_expression_ptr(fn->arguments()[0]);
680 /// c = d(arg)/dt；若 arg 关于 t 线性则 c 不依赖 t。
681 auto c = arg->differentiate(t);
682 if (c && !te_depends_on(c, t)) {
683 /// arg - c*t 应为 0（纯线性，无常数项）才是标准因果指数。
684 auto linear = SymbolicExpr::multiply(c, SymbolicExpr::variable(t));
685 auto residual = SymbolicExpr::add(arg,
686 SymbolicExpr::multiply(SymbolicExpr::number(-1), linear))->simplify();
687 if (lamina::detail::node(residual) && lamina::detail::node(residual)->is_zero()) {
688 /// 衰减率 a = -c（要求 a>0，即 c 为负）。ℱ = 1/(a + iω)。
689 auto a = SymbolicExpr::multiply(SymbolicExpr::number(-1), c)->simplify();
690 auto i_unit = lamina::detail::make_expression_ptr(
691 SymbolicFactory::create_complex(
692 lamina::detail::node(SymbolicExpr::number(0)),
693 lamina::detail::node(SymbolicExpr::number(1))));
694 auto iw = SymbolicExpr::multiply(i_unit, wv);
695 auto denom = SymbolicExpr::add(a, iw);
696 return SymbolicExpr::divide(SymbolicExpr::number(1), denom)->simplify();
697 }
698 }
699 }
700 }
701
702 /// 未知形式：返回未求值 TransformNode
703 return lamina::detail::make_expression_ptr(lamina::detail::make_node<TransformNode>(
704 TransformNode::TransformType::Fourier, lamina::detail::node(f)->clone(), t, omega));
705}

761 }

766 if (te_is_gaussian(F, omega, a_val)) {

770 SymbolicExpr::number(-1.0 / (4.0 * a_val)),

773 return SymbolicExpr::multiply(coeff_expr, exp_part);
774 }
775 }

778 return lamina::detail::make_expression_ptr(lamina::detail::make_node<TransformNode>(
779 TransformNode::TransformType::InverseFourier, lamina::detail::node(F)->clone(), omega, t));
780}

813std::shared_ptr<SymbolicExpr> convolve(
814 const std::shared_ptr<SymbolicExpr>& f, const std::shared_ptr<SymbolicExpr>& g,
815 const std::string& var) {
816 if (!f || !lamina::detail::node(f)) return nullptr;
817 if (!g || !lamina::detail::node(g)) return nullptr;
818 /// (f * g)(t) = ∫_{-∞}^{∞} f(τ)g(t-τ)dτ
819 /// 使用辅助变量 tau 避免命名冲突
820 std::string tau = "__conv_tau__";
821 auto tau_var = SymbolicExpr::variable(tau);
822 auto t_var = SymbolicExpr::variable(var);
823
824 auto f_tau = f->substitute(var, tau_var);
825 if (!f_tau) return nullptr;
826
827 auto t_minus_tau = SymbolicExpr::add(t_var,
828 SymbolicExpr::multiply(SymbolicExpr::number(-1), tau_var));
829 auto g_shifted = g->substitute(var, t_minus_tau);
830 if (!g_shifted) return nullptr;
831
832 auto integrand = SymbolicExpr::multiply(f_tau, g_shifted)->simplify();
833 std::vector<std::shared_ptr<const SymbolicNode>> arguments{
834 lamina::detail::node(integrand),
835 lamina::detail::node(tau_var),
836 lamina::detail::node(SymbolicExpr::infinity(-1)),
837 lamina::detail::node(SymbolicExpr::infinity(1)),
838 };
839 return lamina::detail::make_expression_ptr(
840 lamina::detail::make_node<FunctionNode>(
841 FunctionNode::FuncType::Calculus_Integral,
842 std::move(arguments)));
843}

892 bool& is_sin_out, std::shared_ptr<SymbolicExpr>& omega_out) {

901 for (size_t i = 0; i < 2; ++i) {

903 auto other = lamina::detail::make_expression_ptr(ml->operands()[1 - i]);
904 if (vn && vn->name() == n && !te_depends_on(other, n)) {
905 omega_out = other;
906 return true;
907 }
908 }
909 }
910 auto vn = std::dynamic_pointer_cast<const VariableNode>(arg);
911 if (vn && vn->name() == n) {
912 omega_out = SymbolicExpr::number(1);
913 return true;
914 }
915 return false;

942static std::vector<std::shared_ptr<SymbolicExpr>> zt_roc(
943 const std::shared_ptr<SymbolicExpr>& f_n,
944 const std::string& n,
945 const std::string& z) {
946 if (!f_n || !lamina::detail::node(f_n)) return {};
947
948 auto add_node = std::dynamic_pointer_cast<const AddNode>(lamina::detail::node(f_n));
949 if (add_node) {
950 std::vector<std::shared_ptr<SymbolicExpr>> combined;
951 for (const auto& op : add_node->operands()) {
952 auto term_roc = zt_roc(lamina::detail::make_expression_ptr(op), n, z);
953 combined.insert(combined.end(), term_roc.begin(), term_roc.end());
954 }
955 return combined;
956 }
957
958 if (!te_depends_on(f_n, n)) return zt_unit_roc(z);
959
960 auto [coeff, body] = te_split_coeff(f_n, n);
961 if (!coeff->is_one()) return zt_roc(body, n, z);
962
963 auto vn = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(f_n));
964 if (vn && vn->name() == n) return zt_unit_roc(z);
965
966 auto pw = std::dynamic_pointer_cast<const PowerNode>(lamina::detail::node(f_n));
967 if (pw) {
968 auto bv = std::dynamic_pointer_cast<const VariableNode>(pw->base());
969 double ev = te_numval(pw->exponent());
970 if (bv && bv->name() == n && std::abs(ev - 2.0) < 1e-12) {
971 return zt_unit_roc(z);
972 }
973 }
974
975 std::shared_ptr<SymbolicExpr> base_expr;
976 if (zt_is_exp_seq(f_n, n, base_expr)) return zt_base_roc(base_expr, z);
977
978 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(f_n));
979 if (mul && mul->operands().size() == 2) {
980 for (size_t i = 0; i < 2; ++i) {
981 auto term_n = std::dynamic_pointer_cast<const VariableNode>(mul->operands()[i]);
982 auto other = lamina::detail::make_expression_ptr(mul->operands()[1 - i]);
983 if (term_n && term_n->name() == n &&
984 zt_is_exp_seq(other, n, base_expr)) {
985 return zt_base_roc(base_expr, z);
986 }
987 }
988 }
989
990 bool is_sin = false;
991 std::shared_ptr<SymbolicExpr> omega_expr;
992 if (zt_is_trig_seq(f_n, n, is_sin, omega_expr)) return zt_unit_roc(z);
993
994 return {};
995}

997std::shared_ptr<SymbolicExpr> z_transform(
998 const std::shared_ptr<SymbolicExpr>& f_n, const std::string& n, const std::string& z) {
999 if (!f_n || !lamina::detail::node(f_n)) return nullptr;
1000 auto zv = SymbolicExpr::variable(z);
1001
1002 /// 线性性：处理加法
1003 auto add_node = std::dynamic_pointer_cast<const AddNode>(lamina::detail::node(f_n));
1004 if (add_node) {
1005 std::shared_ptr<SymbolicExpr> result;
1006 for (auto& op : add_node->operands()) {
1007 auto zt = z_transform(lamina::detail::make_expression_ptr(op), n, z);
1008 if (!zt) {

1050 SymbolicExpr::number(3));

1055 }
1056 }

1058 /// Z{a^n} = z/(z-a)
1059 {

1062 auto den = SymbolicExpr::add(zv,
1063 SymbolicExpr::multiply(SymbolicExpr::number(-1), base_expr));
1064 return SymbolicExpr::divide(zv, den)->simplify();
1065 }
1066 }

1068 /// Z{n·a^n} = a·z/(z-a)^2
1069 {
1070 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(f_n));
1071 if (mul && mul->operands().size() == 2) {
1072 for (size_t i = 0; i < 2; ++i) {
1073 auto vn = std::dynamic_pointer_cast<const VariableNode>(mul->operands()[i]);
1074 auto other = lamina::detail::make_expression_ptr(mul->operands()[1 - i]);
1075 if (vn && vn->name() == n) {
1076 std::shared_ptr<SymbolicExpr> base_expr;
1077 if (zt_is_exp_seq(other, n, base_expr)) {
1078 auto den = SymbolicExpr::power(
1079 SymbolicExpr::add(zv,
1080 SymbolicExpr::multiply(SymbolicExpr::number(-1), base_expr)),
1081 SymbolicExpr::number(2));
1082 auto num_expr = SymbolicExpr::multiply(base_expr, zv);
1083 return SymbolicExpr::divide(num_expr, den)->simplify();
1084 }
1085 }
1086 }
1087 }
1088 }
1089
1090 /// Z{sin(w·n)} = z·sin(w)/(z^2 - 2z·cos(w) + 1)
1091 /// Z{cos(w·n)} = z·(z-cos(w))/(z^2 - 2z·cos(w) + 1)
1092 {
1093 bool is_sin_s = false;
1094 std::shared_ptr<SymbolicExpr> omega_expr;
1095 if (zt_is_trig_seq(f_n, n, is_sin_s, omega_expr)) {
1096 auto cos_w = SymbolicExpr::cos(omega_expr);
1097 auto z_sq = SymbolicExpr::power(zv, SymbolicExpr::number(2));
1098 auto two_z_cos_w = SymbolicExpr::multiply(
1099 SymbolicExpr::multiply(SymbolicExpr::number(2), zv), cos_w);
1100 auto den = SymbolicExpr::add(
1101 SymbolicExpr::add(z_sq,
1102 SymbolicExpr::multiply(SymbolicExpr::number(-1), two_z_cos_w)),
1103 SymbolicExpr::number(1));
1104 if (is_sin_s) {
1105 auto num_expr = SymbolicExpr::multiply(zv, SymbolicExpr::sin(omega_expr));
1106 return SymbolicExpr::divide(num_expr, den)->simplify();
1107 } else {
1108 auto z_minus_cos = SymbolicExpr::add(zv,
1109 SymbolicExpr::multiply(SymbolicExpr::number(-1), cos_w));
1110 auto num_expr = SymbolicExpr::multiply(zv, z_minus_cos);
1111 return SymbolicExpr::divide(num_expr, den)->simplify();
1112 }
1113 }
1114 }
1115
1116 /// 未知形式：返回未求值节点
1117 return lamina::detail::make_expression_ptr(lamina::detail::make_node<TransformNode>(
1118 TransformNode::TransformType::ZTransform, lamina::detail::node(f_n)->clone(), n, z));
1119}
