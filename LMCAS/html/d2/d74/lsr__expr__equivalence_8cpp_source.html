37bool exact_integer_node(const std::shared_ptr<const SymbolicNode>& node,
38 int expected) {
39 auto number = std::dynamic_pointer_cast<const NumberNode>(node);
40 if (!number) return false;
41 if (std::holds_alternative<BigInt>(number->value())) {
42 return std::get<BigInt>(number->value()) == BigInt(expected);
43 }
44 if (std::holds_alternative<Rational>(number->value())) {
45 return std::get<Rational>(number->value()) == Rational(expected);
46 }
47 return false;
48}

50bool trig_square_argument(const std::shared_ptr<const SymbolicNode>& node,
51 FunctionNode::FuncType type,
52 std::shared_ptr<const SymbolicNode>& argument) {
53 auto power = std::dynamic_pointer_cast<const PowerNode>(node);
54 if (!power || !exact_integer_node(power->exponent(), 2)) return false;
55 auto function = std::dynamic_pointer_cast<const FunctionNode>(power->base());
56 if (!function || function->type() != type ||
57 function->arguments().size() != 1) {
58 return false;
59 }
60 argument = function->arguments()[0];
61 return true;
62}

64std::optional<ExprPtr> unwrap_trig_negated_argument(
65 const std::shared_ptr<const SymbolicNode>& node) {
66 auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node);
67 if (!multiply) return std::nullopt;
68
69 bool found_negative_one = false;
70 std::vector<std::shared_ptr<const SymbolicNode>> remaining;
71 remaining.reserve(multiply->operands().size());
72 for (const auto& operand : multiply->operands()) {
73 if (!found_negative_one && exact_integer_node(operand, -1)) {
74 found_negative_one = true;
75 continue;
76 }
77 remaining.push_back(operand);
78 }
79 if (!found_negative_one || remaining.empty()) return std::nullopt;
80 if (remaining.size() == 1) {
81 return lamina::detail::make_expression_ptr(remaining.front());
82 }
83 return lamina::detail::make_expression_ptr(
84 SymbolicFactory::create_multiply(std::move(remaining)))->simplify();
85}

87ExprPtr rewrite_trig_basic_identity(const std::shared_ptr<const SymbolicNode>& node) {
88 if (!node) return nullptr;
89
90 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
91 std::vector<std::shared_ptr<const SymbolicNode>> rewritten;
92 rewritten.reserve(add->operands().size());
93 for (const auto& operand : add->operands()) {
94 auto child = rewrite_trig_basic_identity(operand);
95 rewritten.push_back(lamina::detail::node(child));
96 }
97
98 std::vector<bool> used(rewritten.size(), false);
99 std::vector<std::shared_ptr<const SymbolicNode>> result_nodes;
100 for (std::size_t i = 0; i < rewritten.size(); ++i) {
101 if (used[i]) continue;
102 std::shared_ptr<const SymbolicNode> sin_arg;
103 std::shared_ptr<const SymbolicNode> cos_arg;
104 const bool is_sin_square = trig_square_argument(
105 rewritten[i], FunctionNode::FuncType::Sin, sin_arg);
106 const bool is_cos_square = trig_square_argument(
107 rewritten[i], FunctionNode::FuncType::Cos, cos_arg);
108 bool matched = false;
109 for (std::size_t j = i + 1; j < rewritten.size(); ++j) {
110 if (used[j]) continue;
111 std::shared_ptr<const SymbolicNode> other_arg;
112 if (is_sin_square &&
113 trig_square_argument(rewritten[j],
114 FunctionNode::FuncType::Cos,
115 other_arg) &&
116 sin_arg->equals(*other_arg)) {
117 matched = true;
118 } else if (is_cos_square &&
119 trig_square_argument(rewritten[j],
120 FunctionNode::FuncType::Sin,
121 other_arg) &&
122 cos_arg->equals(*other_arg)) {
123 matched = true;
124 }
125 if (matched) {
126 used[i] = true;
127 used[j] = true;
128 result_nodes.push_back(
129 lamina::detail::node(SymbolicExpr::number(1)));
130 break;
131 }
132 }
133 if (!matched && !used[i]) {
134 result_nodes.push_back(rewritten[i]);
135 }
136 }
137
138 if (result_nodes.empty()) {
139 return SymbolicExpr::number(0);
140 }
141 return lamina::detail::make_expression_ptr(
142 SymbolicFactory::create_add(std::move(result_nodes)))->simplify();
143 }
144
145 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
146 std::vector<std::shared_ptr<const SymbolicNode>> operands;
147 operands.reserve(multiply->operands().size());
148 for (const auto& operand : multiply->operands()) {
149 auto child = rewrite_trig_basic_identity(operand);
150 operands.push_back(lamina::detail::node(child));
151 }
152 return lamina::detail::make_expression_ptr(
153 SymbolicFactory::create_multiply(std::move(operands)))->simplify();
154 }
155
156 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
157 auto base = rewrite_trig_basic_identity(power->base());
158 auto exponent = rewrite_trig_basic_identity(power->exponent());
159 return SymbolicExpr::power(base, exponent)->simplify();
160 }
161
162 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(node)) {
163 std::vector<std::shared_ptr<const SymbolicNode>> args;
164 args.reserve(function->arguments().size());
165 for (const auto& argument : function->arguments()) {
166 auto child = rewrite_trig_basic_identity(argument);
167 args.push_back(lamina::detail::node(child));
168 }
169
170 if (args.size() == 1) {
171 auto positive_arg = unwrap_trig_negated_argument(args[0]);
172 if (positive_arg && *positive_arg) {
173 if (function->type() == FunctionNode::FuncType::Sin) {
174 return SymbolicExpr::multiply(
175 SymbolicExpr::number(-1),
176 SymbolicExpr::sin(*positive_arg))->simplify();
177 }
178 if (function->type() == FunctionNode::FuncType::Cos) {
179 return SymbolicExpr::cos(*positive_arg)->simplify();
180 }
181 }
182 }
183
184 return lamina::detail::make_expression_ptr(
185 lamina::detail::make_node<FunctionNode>(function->type(),
186 std::move(args)))->simplify();
187 }
188
189 if (auto complex_node = std::dynamic_pointer_cast<const ComplexNode>(node)) {
190 auto real_part = rewrite_trig_basic_identity(complex_node->real());
191 auto imag_part = rewrite_trig_basic_identity(complex_node->imag());
192 auto value = complex(real_part, imag_part);
193 if (!value) throw detail::ResultPropagation(value.error());
194 return value.value()->simplify();
195 }
196
197 return lamina::detail::make_expression_ptr(node);
198}

200ExprPtr rewrite_exp_log_basic_identity(
201 const std::shared_ptr<const SymbolicNode>& node,
202 const AssumptionContext* assumptions) {
203 if (!node) return nullptr;
204
205 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
206 std::vector<std::shared_ptr<const SymbolicNode>> operands;
207 operands.reserve(add->operands().size());
208 for (const auto& operand : add->operands()) {
209 auto child = rewrite_exp_log_basic_identity(operand, assumptions);
210 operands.push_back(lamina::detail::node(child));
211 }
212 return lamina::detail::make_expression_ptr(
213 SymbolicFactory::create_add(std::move(operands)))->simplify();
214 }
215
216 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
217 std::vector<std::shared_ptr<const SymbolicNode>> operands;
218 operands.reserve(multiply->operands().size());
219 for (const auto& operand : multiply->operands()) {
220 auto child = rewrite_exp_log_basic_identity(operand, assumptions);
221 operands.push_back(lamina::detail::node(child));
222 }
223 return lamina::detail::make_expression_ptr(
224 SymbolicFactory::create_multiply(std::move(operands)))->simplify();
225 }
226
227 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
228 auto base = rewrite_exp_log_basic_identity(power->base(), assumptions);
229 auto exponent = rewrite_exp_log_basic_identity(power->exponent(), assumptions);
230 return SymbolicExpr::power(base, exponent)->simplify();
231 }
232
233 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(node)) {
234 std::vector<std::shared_ptr<const SymbolicNode>> args;
235 args.reserve(function->arguments().size());
236 for (const auto& argument : function->arguments()) {
237 auto child = rewrite_exp_log_basic_identity(argument, assumptions);
238 args.push_back(lamina::detail::node(child));
239 }
240
241 if (function->type() == FunctionNode::FuncType::Exp &&
242 args.size() == 1 && exact_integer_node(args[0], 0)) {
243 return SymbolicExpr::number(1);
244 }
245 if (function->type() == FunctionNode::FuncType::Ln &&
246 args.size() == 1 && exact_integer_node(args[0], 1)) {
247 return SymbolicExpr::number(0);
248 }
249 if (function->type() == FunctionNode::FuncType::Exp &&
250 args.size() == 1) {
251 auto inner_ln = std::dynamic_pointer_cast<const FunctionNode>(args[0]);
252 if (inner_ln && inner_ln->type() == FunctionNode::FuncType::Ln &&
253 inner_ln->arguments().size() == 1) {
254 auto ln_arg = lamina::detail::make_expression_ptr(
255 inner_ln->arguments()[0]);
256 const bool known_positive = inner_ln->arguments()[0]->is_positive() ||
257 (assumptions &&
258 detail::propagate_result(assumptions->is_positive(*ln_arg)) ==
259 Tribool::True);
260 if (known_positive) {
261 return ln_arg->simplify();
262 }
263 }
264 }
265
266 return lamina::detail::make_expression_ptr(
267 lamina::detail::make_node<FunctionNode>(function->type(),
268 std::move(args)))->simplify();
269 }
270
271 if (auto complex_node = std::dynamic_pointer_cast<const ComplexNode>(node)) {
272 auto real_part = rewrite_exp_log_basic_identity(complex_node->real(),
273 assumptions);
274 auto imag_part = rewrite_exp_log_basic_identity(complex_node->imag(),
275 assumptions);
276 auto value = complex(real_part, imag_part);
277 if (!value) throw detail::ResultPropagation(value.error());
278 return value.value()->simplify();
279 }
280
281 return lamina::detail::make_expression_ptr(node);
282}

284void collect_variable_names(
285 const std::shared_ptr<const SymbolicNode>& node,
286 std::set<std::string>& variables) {
287 if (!node) return;
288 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(node)) {
289 variables.insert(variable->name());
290 return;
291 }
292 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
293 for (const auto& operand : add->operands()) {
294 collect_variable_names(operand, variables);
295 }
296 return;
297 }
298 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
299 for (const auto& operand : multiply->operands()) {
300 collect_variable_names(operand, variables);
301 }
302 return;
303 }
304 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
305 collect_variable_names(power->base(), variables);
306 collect_variable_names(power->exponent(), variables);
307 return;
308 }
309 if (auto complex_node = std::dynamic_pointer_cast<const ComplexNode>(node)) {
310 collect_variable_names(complex_node->real(), variables);
311 collect_variable_names(complex_node->imag(), variables);
312 }

315Result<std::optional<bool>> prove_rational_polynomial_equivalence(
316 const ExprPtr& difference,
317 ComputationContext& context,
318 const EqvOptions& options) {
319 if (!difference) {
320 return Result<std::optional<bool>>::failure(
321 CasErrc::InternalInvariant,
322 "equivalence difference is null",
323 kEquivalentOperation);
324 }
325
326 if (options.budget.max_rewrite_steps < 4) {
327 return Result<std::optional<bool>>::failure(
328 CasErrc::ResourceLimit,
329 "equivalence rewrite budget exhausted before polynomial normalization",
330 kEquivalentOperation);
331 }
332
333 auto step = context.consume_steps(4, kEquivalentOperation);
334 if (!step) return Result<std::optional<bool>>::failure(step.error());
335
336 auto expanded = difference->expand();
337 if (!expanded || !lamina::detail::node(expanded)) {
338 return Result<std::optional<bool>>::failure(
339 CasErrc::InternalInvariant,
340 "equivalence expansion returned null",
341 kEquivalentOperation);
342 }
343
344 std::set<std::string> variables;
345 collect_variable_names(lamina::detail::node(expanded), variables);
346 if (variables.size() > 1) {
347 return Result<std::optional<bool>>::success(std::nullopt);
348 }
349 const std::string variable = variables.empty() ? "x" : *variables.begin();
350
351 auto recognized = recognize_rational_polynomial(*expanded, variable, context);
352 if (!recognized) {
353 return Result<std::optional<bool>>::failure(recognized.error());
354 }
355 if (!recognized.value()) {
356 return Result<std::optional<bool>>::success(std::nullopt);
357 }
358 return Result<std::optional<bool>>::success(recognized.value()->is_zero());
359}

361ExprPtr canonicalize_lsr_complex_product(const SymbolicExpr& expression) {
362 const auto& node = lamina::detail::node(expression);
363
364 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(node)) {
365 if (is_imaginary_unit_name(variable->name())) {
366 auto unit = imaginary_unit();
367 if (!unit) throw detail::ResultPropagation(unit.error());
368 return unit.value();
369 }
370 return lamina::detail::make_expression_ptr(node);
371 }
372
373 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
374 std::vector<std::shared_ptr<const SymbolicNode>> operands;
375 operands.reserve(add->operands().size());
376 for (const auto& operand : add->operands()) {
377 auto canonical_operand = canonicalize_lsr_complex_product(
378 *lamina::detail::make_expression_ptr(operand));
379 operands.push_back(lamina::detail::node(canonical_operand));
380 }
381 return lamina::detail::make_expression_ptr(
382 SymbolicFactory::create_add(std::move(operands)))->simplify();
383 }
384
385 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
386 auto exponent = exact_small_integer_node(power->exponent(), 0, 16);
387 if (exponent) {
388 auto canonical_base = canonicalize_lsr_complex_product(
389 *lamina::detail::make_expression_ptr(power->base()));
390 if (std::dynamic_pointer_cast<const ComplexNode>(
391 lamina::detail::node(canonical_base))) {
392 auto result = SymbolicExpr::number(1);
393 for (int i = 0; i < *exponent; ++i) {
394 result = canonicalize_lsr_complex_product(
395 *SymbolicExpr::multiply(result, canonical_base));
396 }
397 return result->simplify();
398 }
399 }
400 return lamina::detail::make_expression_ptr(node);
401 }
402
403 if (auto complex_node = std::dynamic_pointer_cast<const ComplexNode>(node)) {
404 auto real_part = canonicalize_lsr_complex_product(
405 *lamina::detail::make_expression_ptr(complex_node->real()));
406 auto imag_part = canonicalize_lsr_complex_product(
407 *lamina::detail::make_expression_ptr(complex_node->imag()));
408 auto value = complex(real_part, imag_part);
409 if (!value) throw detail::ResultPropagation(value.error());
410 return value.value()->simplify();
411 }
412
413 auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node);
414 if (!multiply) {
415 return lamina::detail::make_expression_ptr(node);
416 }
417
418 bool saw_complex = false;
419 auto real = SymbolicExpr::number(1);
420 auto imag = SymbolicExpr::number(0);
421
422 for (const auto& operand : multiply->operands()) {
423 auto canonical_operand = canonicalize_lsr_complex_product(
424 *lamina::detail::make_expression_ptr(operand));
425 const auto& operand_node = lamina::detail::node(canonical_operand);
426 ExprPtr factor_real;
427 ExprPtr factor_imag;
428 if (auto complex_operand = std::dynamic_pointer_cast<const ComplexNode>(
429 operand_node)) {
430 saw_complex = true;
431 factor_real = lamina::detail::make_expression_ptr(complex_operand->real());
432 factor_imag = lamina::detail::make_expression_ptr(complex_operand->imag());
433 } else {
434 factor_real = canonical_operand;
435 factor_imag = SymbolicExpr::number(0);
436 }
437
438 auto ac = SymbolicExpr::multiply(real, factor_real);
439 auto bd = SymbolicExpr::multiply(imag, factor_imag);
440 auto ad = SymbolicExpr::multiply(real, factor_imag);
441 auto bc = SymbolicExpr::multiply(imag, factor_real);
442 auto next_real = SymbolicExpr::add(
443 ac, SymbolicExpr::multiply(SymbolicExpr::number(-1), bd))->simplify();
444 auto next_imag = SymbolicExpr::add(ad, bc)->simplify();
445 real = std::move(next_real);
446 imag = std::move(next_imag);
447 }
448
449 if (!saw_complex) {
450 return lamina::detail::make_expression_ptr(node);
451 }
452 auto result = complex(real, imag);
453 if (!result) {
454 throw detail::ResultPropagation(result.error());
455 }
456 return result.value()->simplify();
457}

461Result<EqvProfile> eqv_profile_from_name(const std::string& name) {
462 if (name == "Core") {
463 return Result<EqvProfile>::success(EqvProfile::Core);

507Result<bool> equivalent_core(const SymbolicExpr& lhs,
508 const SymbolicExpr& rhs,
509 ComputationContext& context,
510 const EqvOptions& options) {
511 auto options_valid = validate_eqv_options(options);
512 if (!options_valid) return Result<bool>::failure(options_valid.error());
513
514 auto step = context.consume_steps(1, kEquivalentOperation);
515 if (!step) return Result<bool>::failure(step.error());
516 try {
517 auto lhs_dimension = dimension_of(lhs);
518 if (!lhs_dimension) return Result<bool>::failure(lhs_dimension.error());
519 auto rhs_dimension = dimension_of(rhs);
520 if (!rhs_dimension) return Result<bool>::failure(rhs_dimension.error());
521 if (lhs_dimension.value() != rhs_dimension.value()) {
522 return Result<bool>::success(false);
523 }
524 auto lhs_ptr = std::make_shared<SymbolicExpr>(lhs);
525 auto rhs_ptr = std::make_shared<SymbolicExpr>(rhs);
526 if (std::dynamic_pointer_cast<const QuantityNode>(lamina::detail::node(lhs))) {
527 auto stripped = strip_unit(lhs_ptr, UnitStripMode::BaseValue, context);
528 if (!stripped) return Result<bool>::failure(stripped.error());
529 lhs_ptr = stripped.value();
530 }
531 if (std::dynamic_pointer_cast<const QuantityNode>(lamina::detail::node(rhs))) {
532 auto stripped = strip_unit(rhs_ptr, UnitStripMode::BaseValue, context);
533 if (!stripped) return Result<bool>::failure(stripped.error());
534 rhs_ptr = stripped.value();
535 }
536 auto canonical_lhs = canonicalize_lsr_complex_product(*lhs_ptr);
537 auto canonical_rhs = canonicalize_lsr_complex_product(*rhs_ptr);
538 if (structurally_equal(*canonical_lhs, *canonical_rhs)) {
539 return Result<bool>::success(true);
540 }
541 auto difference = SymbolicExpr::add(
542 canonical_lhs,
543 SymbolicExpr::multiply(SymbolicExpr::number(-1),
544 canonical_rhs));
545 if (!difference) {
546 return Result<bool>::failure(CasErrc::InternalInvariant,
547 "equivalence difference construction failed",
548 kEquivalentOperation);
549 }
550 if (difference->simplify()->is_zero()) {
551 return Result<bool>::success(true);
552 }
553
554 auto polynomial_proof = prove_rational_polynomial_equivalence(
555 difference, context, options);
556 if (!polynomial_proof) {
557 return Result<bool>::failure(polynomial_proof.error());
558 }
559 if (polynomial_proof.value()) {
560 return Result<bool>::success(*polynomial_proof.value());
561 }
562 if (options.profile == EqvProfile::TrigBasic) {
563 if (options.budget.max_rewrite_steps < 8) {
564 return Result<bool>::failure(
565 CasErrc::ResourceLimit,
566 "equivalence rewrite budget exhausted before Trig-Basic normalization",
567 kEquivalentOperation);
568 }
569 auto trig_step = context.consume_steps(8, kEquivalentOperation);
570 if (!trig_step) return Result<bool>::failure(trig_step.error());
571 auto trig_lhs = rewrite_trig_basic_identity(lamina::detail::node(lhs));
572 auto trig_rhs = rewrite_trig_basic_identity(lamina::detail::node(rhs));
573 EqvOptions core_options = options;
574 core_options.profile = EqvProfile::Core;
575 return equivalent_core(*trig_lhs, *trig_rhs, context,
576 core_options);
577 }
578 if (options.profile == EqvProfile::ExpLogBasic) {
579 if (options.budget.max_rewrite_steps < 8) {
580 return Result<bool>::failure(
581 CasErrc::ResourceLimit,
582 "equivalence rewrite budget exhausted before ExpLog-Basic normalization",
583 kEquivalentOperation);
584 }
585 auto exp_log_step = context.consume_steps(8, kEquivalentOperation);
586 if (!exp_log_step) return Result<bool>::failure(exp_log_step.error());
587 auto exp_log_lhs = rewrite_exp_log_basic_identity(
588 lamina::detail::node(lhs), context.assumptions().get());
589 auto exp_log_rhs = rewrite_exp_log_basic_identity(
590 lamina::detail::node(rhs), context.assumptions().get());
591 EqvOptions core_options = options;
592 core_options.profile = EqvProfile::Core;
593 return equivalent_core(*exp_log_lhs, *exp_log_rhs, context,
594 core_options);
595 }
596 return Result<bool>::success(false);
597 } catch (const detail::ResultPropagation& propagation) {
598 return Result<bool>::failure(propagation.error());
599 } catch (const std::bad_alloc&) {
600 return Result<bool>::failure(CasErrc::ResourceLimit,
601 "equivalence check allocation failed",
602 kEquivalentOperation);
603 } catch (const std::exception& error) {
604 return Result<bool>::failure(CasErrc::Inconclusive, error.what(),
605 kEquivalentOperation);
606 }
607}

615Result<bool> equivalent(const SymbolicExpr& lhs,
616 const SymbolicExpr& rhs,
617 ComputationContext& context,
618 const EqvOptions& options) {
619 auto checked = equivalent_core(lhs, rhs, context, options);
620 if (checked) return checked;
621
622 const auto code = checked.error().code;
623 if (code == CasErrc::ResourceLimit ||
624 code == CasErrc::Inconclusive ||
625 code == CasErrc::UnsupportedExpression) {
626 (void)context.add_diagnostic(
627 Diagnostic{DiagnosticSeverity::Warning,
628 kEquivalentOperation,
629 checked.error().message});
