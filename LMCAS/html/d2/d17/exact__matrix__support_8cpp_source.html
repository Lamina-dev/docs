10Result<void> validate_matrix(
11 const ExactMatrixData& matrix,
12 const std::string& operation) {
13 if (matrix.rows == 0 || matrix.cols == 0) {
14 return Result<void>::failure(
15 CasErrc::InvalidArgument,
16 "exact matrix dimensions must be non-zero", operation);
17 }
18 if (matrix.rows > std::numeric_limits<std::size_t>::max() / matrix.cols ||
19 matrix.entries.size() != matrix.rows * matrix.cols) {
20 return Result<void>::failure(
21 CasErrc::InvalidArgument,
22 "exact matrix storage does not match its dimensions", operation);
23 }
24 for (const auto& entry : matrix.entries) {

26 return Result<void>::failure(
27 CasErrc::InvalidArgument,
28 "exact matrix entries cannot be null", operation);
29 }
30 }
31 return Result<void>::success();

95Result<std::optional<std::size_t>> choose_pivot(
96 const ExactMatrixData& matrix,
97 std::size_t first_row,
98 std::size_t column,
99 ComputationContext& context,
100 const std::string& operation) {
101 bool saw_unknown = false;
102 for (std::size_t row = first_row; row < matrix.rows; ++row) {
103 auto proof = classify_exact_zero(matrix.at(row, column), context, operation);
104 if (!proof) {
105 return Result<std::optional<std::size_t>>::failure(proof.error());
106 }
107 if (proof.value() == ZeroProof::NonZero) {
108 return Result<std::optional<std::size_t>>::success(row);
109 }
110 saw_unknown = saw_unknown || proof.value() == ZeroProof::Unknown;
111 }
112 if (saw_unknown) {
113 return Result<std::optional<std::size_t>>::failure(
114 CasErrc::Inconclusive,
115 "exact pivot is not provably zero or non-zero", operation);
116 }
117 return Result<std::optional<std::size_t>>::success(std::nullopt);

142std::optional<ZeroProof> classify_numeric_sqrt(const ExprPtr& simplified) {
143 auto function =
144 std::dynamic_pointer_cast<const FunctionNode>(node(simplified));
145 if (!function) {
146 return std::nullopt;
147 }
148 if (function->type() != FunctionNode::FuncType::Sqrt ||
149 function->arguments().size() != 1) {
150 return std::nullopt;
151 }
152 auto argument = exact_rational(
153 make_expression_ptr(function->arguments()[0]));
154 if (!argument || *argument < Rational(0)) {
155 return std::nullopt;
156 }

160std::optional<ZeroProof> classify_numeric_value(const ExprPtr& expression) {
161 if (auto value = exact_rational(expression)) {
162 return *value == Rational(0) ? ZeroProof::Zero : ZeroProof::NonZero;
163 }
164 if (auto proof = classify_numeric_power(expression)) {
165 return proof;
166 }
167 if (auto proof = classify_numeric_sqrt(expression)) {
168 return proof;
169 }
170 if (auto root = std::dynamic_pointer_cast<const RootOfNode>(node(expression));
171 root && !root->exact_id().polynomial.coeffs.empty() &&
172 root->exact_id().polynomial.coeffs[0] != Rational(0)) {
173 return ZeroProof::NonZero;
174 }
175 return std::nullopt;

179Result<ZeroProof> classify_exact_zero(
180 const ExprPtr& expression,
181 ComputationContext& context,
182 const std::string& operation) {
183 auto access = context.consume_steps(1, operation);
184 if (!access) {
185 return Result<ZeroProof>::failure(access.error());
186 }
187 if (!expression || !node(expression)) {
188 return Result<ZeroProof>::failure(
189 CasErrc::InvalidArgument,
190 "zero classification requires an expression", operation);
191 }
192 try {
193 if (contains_approximate_number(expression)) {
194 return Result<ZeroProof>::success(ZeroProof::Unknown);
195 }
196 auto simplified = simplify_expr(expression);
197 if (auto proof = classify_numeric_value(simplified)) {
198 return Result<ZeroProof>::success(*proof);
199 }
200 if (!context.assumptions()) {
201 return Result<ZeroProof>::success(ZeroProof::Unknown);
202 }
203 auto assumption = context.assumptions()->is_nonzero_checked(*simplified, context);
204 if (!assumption) {
205 return Result<ZeroProof>::failure(assumption.error());
206 }
207 if (assumption.value() == Tribool::True) {
208 return Result<ZeroProof>::success(ZeroProof::NonZero);
209 }
210 if (assumption.value() == Tribool::False) {
211 return Result<ZeroProof>::success(ZeroProof::Zero);
212 }
213 return Result<ZeroProof>::success(ZeroProof::Unknown);
214 } catch (const std::bad_alloc&) {
215 return Result<ZeroProof>::failure(
216 CasErrc::ResourceLimit,
217 "zero classification allocation failed", operation);
218 } catch (const std::exception& error) {
219 return Result<ZeroProof>::failure(
220 CasErrc::InternalInvariant, error.what(), operation);
221 }
222}
