53void NormalizationVisitor::visit(const LogicalNode& node) {
54 /// Implication: A ⇒ B = ¬A ∨ B
55 if (node.op() == LogicalNode::Op::Implies) {
56 auto not_left = LMCAS::detail::make_node<LogicalNode>(node.left(), nullptr, LogicalNode::Op::Not);
57 auto or_node = LMCAS::detail::make_node<LogicalNode>(not_left, node.right(), LogicalNode::Op::Or);
58 or_node->accept(*this);
59 return;
60 }
61
62 /// NOT handling: De Morgan's laws and double negation
63 if (node.op() == LogicalNode::Op::Not) {
64 /// Normalize the operand first
65 std::shared_ptr<const SymbolicNode> new_left = nullptr;
66 if (node.left()) {
67 node.left()->accept(*this);
68 new_left = result;
69 }
70 if (!new_left) new_left = node.left();
71
72 /// Double negation: ¬(¬A) = A
73 if (auto inner_logical = std::dynamic_pointer_cast<const LogicalNode>(new_left)) {
74 if (inner_logical->op() == LogicalNode::Op::Not) {
75 result = inner_logical->left();
76 return;
77 }
78 /// De Morgan's law: ¬(A∧B) = ¬A∨¬B
79 if (inner_logical->op() == LogicalNode::Op::And) {
80 auto not_a = LMCAS::detail::make_node<LogicalNode>(inner_logical->left(), nullptr, LogicalNode::Op::Not);
81 auto not_b = LMCAS::detail::make_node<LogicalNode>(inner_logical->right(), nullptr, LogicalNode::Op::Not);
82 auto or_node = LMCAS::detail::make_node<LogicalNode>(not_a, not_b, LogicalNode::Op::Or);
83 or_node->accept(*this);
84 return;
85 }
86 /// De Morgan's law: ¬(A∨B) = ¬A∧¬B
87 if (inner_logical->op() == LogicalNode::Op::Or) {
88 auto not_a = LMCAS::detail::make_node<LogicalNode>(inner_logical->left(), nullptr, LogicalNode::Op::Not);
89 auto not_b = LMCAS::detail::make_node<LogicalNode>(inner_logical->right(), nullptr, LogicalNode::Op::Not);
90 auto and_node = LMCAS::detail::make_node<LogicalNode>(not_a, not_b, LogicalNode::Op::And);
91 and_node->accept(*this);
92 return;
93 }
94 }
95
96 result = LMCAS::detail::make_node<LogicalNode>(new_left, nullptr, LogicalNode::Op::Not);
97 return;
98 }
99
100 /// And / Or: normalize operands
101 std::shared_ptr<const SymbolicNode> new_left = nullptr;
102 std::shared_ptr<const SymbolicNode> new_right = nullptr;
103
104 if (node.left()) {
105 node.left()->accept(*this);
106 new_left = result;
107 }
108 if (node.right()) {
109 node.right()->accept(*this);
110 new_right = result;
111 }
112
113 if (!new_left) new_left = node.left();
114 if (!new_right) new_right = node.right();
115
116 result = LMCAS::detail::make_node<LogicalNode>(new_left, new_right, node.op());
117 }

155 auto lo_n = std::dynamic_pointer_cast<const NumberNode>(new_lower);

163 return;
164 }

168 std::vector<std::shared_ptr<const SymbolicNode>> terms;
169 for (std::int64_t kk = *lo;; ++kk) {

173 term->accept(inner);
174 terms.push_back(inner.get_result());
175 if (kk == *hi) break;
176 }
177 if (terms.empty()) { result = LMCAS::detail::make_node<NumberNode>(BigInt(0)); return; }

179 sum_node->accept(*this);
