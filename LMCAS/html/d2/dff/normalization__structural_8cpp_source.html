72void NormalizationVisitor::visit(const LogicalNode& node) {
73 /// Implication: A ⇒ B = ¬A ∨ B
74 if (node.op() == LogicalNode::Op::Implies) {
75 normalization_check_children(rewrite_budget(), 1, node.left());
76 auto not_left = LMCAS::detail::make_node<LogicalNode>(node.left(), nullptr, LogicalNode::Op::Not);
77 normalization_check_children(rewrite_budget(), 1, not_left, node.right());
78 auto or_node = LMCAS::detail::make_node<LogicalNode>(not_left, node.right(), LogicalNode::Op::Or);
79 or_node->accept(*this);
80 return;
81 }
82
83 /// NOT handling: De Morgan's laws and double negation
84 if (node.op() == LogicalNode::Op::Not) {
85 /// Normalize the operand first
86 std::shared_ptr<const SymbolicNode> new_left = nullptr;
87 if (node.left()) {
88 node.left()->accept(*this);
89 new_left = result;
90 }
91 if (!new_left) new_left = node.left();
92
93 /// Double negation: ¬(¬A) = A
94 if (auto inner_logical = std::dynamic_pointer_cast<const LogicalNode>(new_left)) {
95 if (inner_logical->op() == LogicalNode::Op::Not) {
96 set_result(inner_logical->left());
97 return;
98 }
99 /// De Morgan's law: ¬(A∧B) = ¬A∨¬B
100 if (inner_logical->op() == LogicalNode::Op::And) {
101 normalization_check_children(rewrite_budget(), 1, inner_logical->left());
102 auto not_a = LMCAS::detail::make_node<LogicalNode>(inner_logical->left(), nullptr, LogicalNode::Op::Not);
103 normalization_check_children(rewrite_budget(), 1, inner_logical->right());
104 auto not_b = LMCAS::detail::make_node<LogicalNode>(inner_logical->right(), nullptr, LogicalNode::Op::Not);
105 normalization_check_children(rewrite_budget(), 1, not_a, not_b);
106 auto or_node = LMCAS::detail::make_node<LogicalNode>(not_a, not_b, LogicalNode::Op::Or);
107 or_node->accept(*this);
108 return;
109 }
110 /// De Morgan's law: ¬(A∨B) = ¬A∧¬B
111 if (inner_logical->op() == LogicalNode::Op::Or) {
112 normalization_check_children(rewrite_budget(), 1, inner_logical->left());
113 auto not_a = LMCAS::detail::make_node<LogicalNode>(inner_logical->left(), nullptr, LogicalNode::Op::Not);
114 normalization_check_children(rewrite_budget(), 1, inner_logical->right());
115 auto not_b = LMCAS::detail::make_node<LogicalNode>(inner_logical->right(), nullptr, LogicalNode::Op::Not);
116 normalization_check_children(rewrite_budget(), 1, not_a, not_b);
117 auto and_node = LMCAS::detail::make_node<LogicalNode>(not_a, not_b, LogicalNode::Op::And);
118 and_node->accept(*this);
119 return;
120 }
121 }
122
123 normalization_check_children(rewrite_budget(), 1, new_left);
124 set_result(LMCAS::detail::make_node<LogicalNode>(new_left, nullptr, LogicalNode::Op::Not));
125 return;
126 }
127
128 /// And / Or: normalize operands
129 std::shared_ptr<const SymbolicNode> new_left = nullptr;
130 std::shared_ptr<const SymbolicNode> new_right = nullptr;
131
132 if (node.left()) {
133 node.left()->accept(*this);
134 new_left = result;
135 }
136 if (node.right()) {
137 node.right()->accept(*this);
138 new_right = result;
139 }
140
141 if (!new_left) new_left = node.left();
142 if (!new_right) new_right = node.right();
143
144 normalization_check_children(rewrite_budget(), 1, new_left, new_right);
145 set_result(LMCAS::detail::make_node<LogicalNode>(new_left, new_right, node.op()));
146 }

190 auto lo_n = std::dynamic_pointer_cast<const NumberNode>(new_lower);
191 auto hi_n = std::dynamic_pointer_cast<const NumberNode>(new_upper);
192 BigInt lo;
193 BigInt hi;
194 if (try_get_integer_value(lo_n, lo) && try_get_integer_value(hi_n, hi)) {
195 if (hi < lo) {
196 set_result(LMCAS::detail::make_node<NumberNode>(BigInt(0)));
197 return;
198 }
199 const BigInt span = hi - lo;
200 if (span < BigInt(1000)) {
201 const auto count = span.try_to_uint64();
202 std::vector<std::shared_ptr<const SymbolicNode>> terms;
203 std::size_t nodes = 0;
204 normalization_check_count(rewrite_budget(), static_cast<std::size_t>(*count) + 1, 1, false);
205 terms.reserve(static_cast<std::size_t>(*count) + 1);
206 BigInt index = lo;
207 for (std::uint64_t offset = 0; offset <= *count; ++offset, index += BigInt(1)) {
208 auto kval = LMCAS::detail::make_node<NumberNode>(index);
209 auto term = norm_subst_index(new_body, node.index_var(), kval, rewrite_budget());
210 normalization_check_children(rewrite_budget(), 0, term);
211 NormalizationVisitor inner(context_, facts_, domain_, rewrite_budget());
212 term->accept(inner);
213 normalization_append(rewrite_budget(), nodes, terms, inner.get_result());
214 }
215 if (terms.empty()) { set_result(LMCAS::detail::make_node<NumberNode>(BigInt(0))); return; }
216 normalization_check_arithmetic<AddNode>(rewrite_budget(), terms, nodes);
217 auto sum_node = LMCAS::detail::make_node<AddNode>(terms);
218 sum_node->accept(*this);
219 return;
220 }
221 }
222
223 normalization_check_children(rewrite_budget(), 1, new_body, new_lower, new_upper);
224 set_result(LMCAS::detail::make_node<SummationNode>(new_body, node.index_var(), new_lower, new_upper));
