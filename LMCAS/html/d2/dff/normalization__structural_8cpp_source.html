49void NormalizationVisitor::visit(const LogicalNode& node) {
50 /// Implication: A ⇒ B = ¬A ∨ B
51 if (node.op() == LogicalNode::Op::Implies) {
52 auto not_left = lamina::detail::make_node<LogicalNode>(node.left(), nullptr, LogicalNode::Op::Not);
53 auto or_node = lamina::detail::make_node<LogicalNode>(not_left, node.right(), LogicalNode::Op::Or);
54 or_node->accept(*this);
55 return;
56 }
57
58 /// NOT handling: De Morgan's laws and double negation
59 if (node.op() == LogicalNode::Op::Not) {
60 /// Normalize the operand first
61 std::shared_ptr<const SymbolicNode> new_left = nullptr;
62 if (node.left()) {
63 node.left()->accept(*this);
64 new_left = result;
65 }
66 if (!new_left) new_left = node.left();
67
68 /// Double negation: ¬(¬A) = A
69 if (auto inner_logical = std::dynamic_pointer_cast<const LogicalNode>(new_left)) {
70 if (inner_logical->op() == LogicalNode::Op::Not) {
71 result = inner_logical->left();
72 return;
73 }
74 /// De Morgan's law: ¬(A∧B) = ¬A∨¬B
75 if (inner_logical->op() == LogicalNode::Op::And) {
76 auto not_a = lamina::detail::make_node<LogicalNode>(inner_logical->left(), nullptr, LogicalNode::Op::Not);
77 auto not_b = lamina::detail::make_node<LogicalNode>(inner_logical->right(), nullptr, LogicalNode::Op::Not);
78 auto or_node = lamina::detail::make_node<LogicalNode>(not_a, not_b, LogicalNode::Op::Or);
79 or_node->accept(*this);
80 return;
81 }
82 /// De Morgan's law: ¬(A∨B) = ¬A∧¬B
83 if (inner_logical->op() == LogicalNode::Op::Or) {
84 auto not_a = lamina::detail::make_node<LogicalNode>(inner_logical->left(), nullptr, LogicalNode::Op::Not);
85 auto not_b = lamina::detail::make_node<LogicalNode>(inner_logical->right(), nullptr, LogicalNode::Op::Not);
86 auto and_node = lamina::detail::make_node<LogicalNode>(not_a, not_b, LogicalNode::Op::And);
87 and_node->accept(*this);
88 return;
89 }
90 }
91
92 result = lamina::detail::make_node<LogicalNode>(new_left, nullptr, LogicalNode::Op::Not);
93 return;
94 }
95
96 /// And / Or: normalize operands
97 std::shared_ptr<const SymbolicNode> new_left = nullptr;
98 std::shared_ptr<const SymbolicNode> new_right = nullptr;
99
100 if (node.left()) {
101 node.left()->accept(*this);
102 new_left = result;
103 }
104 if (node.right()) {
105 node.right()->accept(*this);
106 new_right = result;
107 }
108
109 if (!new_left) new_left = node.left();
110 if (!new_right) new_right = node.right();
111
112 result = lamina::detail::make_node<LogicalNode>(new_left, new_right, node.op());
113 }
