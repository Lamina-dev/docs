9static std::optional<NumericQuarticDepression> depress_numeric_quartic(
10 double av, double bv, double cv, double dv, double ev) {
11
12 double shift_val = bv / (4.0 * av);
13 double p_val = (8.0*av*cv - 3.0*bv*bv) / (8.0*av*av);
14 double q_val = (bv*bv*bv - 4.0*av*bv*cv + 8.0*av*av*dv) / (8.0*av*av*av);
15 double r_val = (-3.0*bv*bv*bv*bv + 256.0*av*av*av*ev - 64.0*av*av*bv*dv + 16.0*av*bv*bv*cv) / (256.0*av*av*av*av);
16 if (!std::isfinite(shift_val) || !std::isfinite(p_val) ||
17 !std::isfinite(q_val) || !std::isfinite(r_val)) {
18 return std::nullopt;
19 }
20 return NumericQuarticDepression{p_val, q_val, r_val, shift_val};
21}

47static std::optional<double> select_numeric_resolvent(
48 const std::vector<std::shared_ptr<SymbolicExpr>>& cubic_roots) {
49 double m_val = 0.0;
50 bool found_m = false;
51 for (const auto& root : cubic_roots) {
52 auto maybe_val = finite_numeric_value(root);
53 if (maybe_val && *maybe_val > 0.0 &&
54 (!found_m || *maybe_val > m_val)) {
55 m_val = *maybe_val;
56 found_m = true;
57 }
58 }
59 if (!found_m) {
60 for (const auto& root : cubic_roots) {
61 auto maybe_val = finite_numeric_value(root);
62 if (maybe_val && *maybe_val != 0.0 &&
63 (!found_m ||
64 std::abs(*maybe_val) > std::abs(m_val))) {
65 m_val = *maybe_val;
66 found_m = true;
67 }
68 }
69 }
70 if (!found_m) {
71 return std::nullopt;
72 }
73 return m_val;
74}

97static std::vector<std::shared_ptr<SymbolicExpr>> negative_resolvent_roots(
98 double m_val, double p_normalized, double q_normalized,
99 double root_scale, double shift_val) {
100 auto shift_expr = SymbolicExpr::number(shift_val);
101 auto scale_expr = SymbolicExpr::number(root_scale);
102 auto p_expr = SymbolicExpr::number(p_normalized);
103 auto q_expr = SymbolicExpr::number(q_normalized);
104 auto s_expr = SymbolicExpr::sqrt(
105 SymbolicExpr::number(2.0 * m_val));
106 auto m_plus_p_half = SymbolicExpr::number(
107 m_val + p_normalized / 2.0);
108 auto q_over_2s = SymbolicExpr::divide(
109 q_expr, SymbolicExpr::multiply(num(2), s_expr));
110 auto quad1_c_expr =
111 sub(m_plus_p_half, q_over_2s)->simplify();
112 auto quad2_c_expr =
113 SymbolicExpr::add(
114 m_plus_p_half, q_over_2s)->simplify();
115 auto y_roots1 = solve_quadratic_internal(
116 num(1), s_expr, quad1_c_expr);
117 auto y_roots2 = solve_quadratic_internal(
118 num(1), negate(s_expr), quad2_c_expr);
119
120 std::vector<std::shared_ptr<SymbolicExpr>> results;
121 for (const auto& y : y_roots1) {
122 auto scaled_y =
123 SymbolicExpr::multiply(scale_expr, y)->simplify();
124 results.push_back(
125 sub(scaled_y, shift_expr)->simplify());
126 }
127 for (const auto& y : y_roots2) {
128 auto scaled_y =
129 SymbolicExpr::multiply(scale_expr, y)->simplify();
130 results.push_back(
131 sub(scaled_y, shift_expr)->simplify());
132 }
133 return results;
134}

136static void append_resolvent_quadratic(
137 std::vector<std::shared_ptr<SymbolicExpr>>& results,
138 double linear, double constant, double root_scale, double shift_val) {
139 const double discriminant = linear * linear - 4.0 * constant;
140 if (discriminant >= 0.0) {
141 const double square_root = std::sqrt(discriminant);
142 const double first = (-linear + square_root) / 2.0;
143 const double second = (-linear - square_root) / 2.0;
144 results.push_back(SymbolicExpr::number(first * root_scale - shift_val));
145 results.push_back(SymbolicExpr::number(second * root_scale - shift_val));
146 } else {
147 const double real = -linear * root_scale / 2.0 - shift_val;
148 const double imaginary = std::sqrt(-discriminant) * root_scale / 2.0;
149 auto unit = SymbolicExpr::sqrt(num(-1));
150 results.push_back(SymbolicExpr::add(SymbolicExpr::number(real),
151 SymbolicExpr::multiply(SymbolicExpr::number(imaginary), unit))->simplify());
152 results.push_back(sub(SymbolicExpr::number(real),
153 SymbolicExpr::multiply(SymbolicExpr::number(imaginary), unit))->simplify());
154 }
155}

157static std::vector<std::shared_ptr<SymbolicExpr>> positive_resolvent_roots(
158 double m_val, double p_normalized, double q_normalized,
159 double root_scale, double shift_val) {
160 const double s_val = std::sqrt(2.0 * m_val);
161 const double quad1_c_val =
162 m_val + p_normalized / 2.0 -
163 q_normalized / (2.0 * s_val);
164 const double quad2_c_val =
165 m_val + p_normalized / 2.0 +
166 q_normalized / (2.0 * s_val);
167 std::vector<std::shared_ptr<SymbolicExpr>> results;
168 append_resolvent_quadratic(results, s_val, quad1_c_val, root_scale, shift_val);
169 append_resolvent_quadratic(results, -s_val, quad2_c_val, root_scale, shift_val);
170 return results;
171}

173std::vector<std::shared_ptr<SymbolicExpr>> numeric_quartic_roots(
174 const NumericQuarticDepression& depression, const std::string& var) {
175 const auto& [p_val, q_val, r_val, shift_val] = depression;
176 if (q_val == 0.0) {
177 return shifted_biquadratic_roots(p_val, r_val, shift_val);
178 }
179 const double root_scale = std::max({
180 std::sqrt(std::abs(p_val)),
181 std::cbrt(std::abs(q_val)),
182 std::sqrt(std::sqrt(std::abs(r_val)))
183 });
184 const double p_normalized =
185 (p_val / root_scale) / root_scale;
186 const double q_normalized =
187 ((q_val / root_scale) / root_scale) / root_scale;
188 const double r_normalized =
189 (((r_val / root_scale) / root_scale) / root_scale) / root_scale;
190 if (q_normalized == 0.0) {
191 return {};
192 }
193 auto cubic_roots = solve_cubic(
194 SymbolicExpr::number(8.0),
195 SymbolicExpr::number(8.0 * p_normalized),
196 SymbolicExpr::number(
197 2.0 * p_normalized * p_normalized -
198 8.0 * r_normalized),
199 SymbolicExpr::number(-(q_normalized * q_normalized)),
200 var);
201 auto selected = select_numeric_resolvent(cubic_roots);
202 if (!selected || *selected == 0.0) {
203 return {};
204 }
205 if (*selected < 0.0) {
206 return negative_resolvent_roots(
207 *selected, p_normalized, q_normalized, root_scale, shift_val);
208 }
209 return positive_resolvent_roots(
210 *selected, p_normalized, q_normalized, root_scale, shift_val);
211}
