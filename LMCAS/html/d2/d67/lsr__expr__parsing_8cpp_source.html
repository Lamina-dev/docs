20 ExprResult parse() {
21 skip_space();
22 if (eof()) {
23 return fail("empty expression");
24 }
25 auto result = parse_logical_or();
26 if (!result) return result;
27 skip_space();
28 if (!eof()) {
29 return fail("unexpected token '" + std::string(1, peek()) + "'");
30 }
31 return result;
32 }

37
38 bool eof() const noexcept { return pos_ >= source_.size(); }
39
40 char peek() const noexcept {
41 return eof() ? '\0' : source_[pos_];
42 }

44 bool match(char c) {
45 skip_space();
46 if (peek() != c) return false;
47 ++pos_;
48 return true;
49 }

51 bool match_text(const char* text) {
52 skip_space();
53 const std::size_t start = pos_;
54 for (const char* p = text; *p; ++p) {
55 if (pos_ >= source_.size() || source_[pos_] != *p) {
56 pos_ = start;
57 return false;
58 }
59 ++pos_;
60 }
61 return true;
62 }
63
64 bool match_keyword(const char* text) {
65 skip_space();
66 const std::size_t start = pos_;
67 if (!match_text(text)) return false;
68 if (is_ident_continue(static_cast<unsigned char>(peek()))) {
69 pos_ = start;
70 return false;
71 }
72 return true;
73 }

75 void skip_space() noexcept {
76 while (!eof() &&
77 std::isspace(static_cast<unsigned char>(source_[pos_]))) {

104 ExprResult parse_logical_or() {
105 auto lhs = parse_logical_and();
106 if (!lhs) return lhs;
107 while (match_keyword("or")) {
108 auto rhs = parse_logical_and();
109 if (!rhs) return rhs;
110 lhs = logical(lhs.value(), rhs.value(), LogicalNode::Op::Or);
111 if (!lhs) return lhs;
112 }
113 return lhs;
114 }

116 ExprResult parse_logical_and() {
117 auto lhs = parse_logical_not();
118 if (!lhs) return lhs;
119 while (match_keyword("and")) {
120 auto rhs = parse_logical_not();
121 if (!rhs) return rhs;
122 lhs = logical(lhs.value(), rhs.value(), LogicalNode::Op::And);
123 if (!lhs) return lhs;
124 }
125 return lhs;
126 }

128 ExprResult parse_logical_not() {
129 if (match_keyword("not") || match('!')) {
130 auto operand = parse_logical_not();
131 if (!operand) return operand;
132 return logical(operand.value(), nullptr, LogicalNode::Op::Not);
133 }
134 return parse_membership();
135 }

137 ExprResult parse_membership() {
138 auto lhs = parse_equality();
139 if (!lhs) return lhs;
140 if (match_keyword("in")) {
141 auto rhs = parse_equality();
142 if (!rhs) return rhs;
143 return membership(lhs.value(), rhs.value(), false);
144 }
145 const auto save = pos_;
146 if (match_keyword("not")) {
147 if (match_keyword("in")) {
148 auto rhs = parse_equality();
149 if (!rhs) return rhs;
150 return membership(lhs.value(), rhs.value(), true);
151 }
152 pos_ = save;
153 }
154 return lhs;
155 }

157 ExprResult parse_equality() {
158 auto lhs = parse_relational();
159 if (!lhs) return lhs;
160 while (true) {
161 RelationOp op;
162 if (match_text("==")) {
163 op = RelationOp::EQ;
164 } else if (match_text("!=")) {
165 op = RelationOp::NEQ;
166 } else {
167 return lhs;
168 }
169 auto rhs = parse_relational();
170 if (!rhs) return rhs;
171 lhs = relational(lhs.value(), rhs.value(), op);
172 if (!lhs) return lhs;
173 }
174 }

176 ExprResult parse_relational() {
177 auto lhs = parse_additive();
178 if (!lhs) return lhs;
179 while (true) {
180 RelationOp op;
181 if (match_text("<=")) {
182 op = RelationOp::LEQ;
183 } else if (match_text(">=")) {
184 op = RelationOp::GEQ;
185 } else if (match('<')) {
186 op = RelationOp::LT;
187 } else if (match('>')) {
188 op = RelationOp::GT;
189 } else {
190 return lhs;
191 }
192 auto rhs = parse_additive();
193 if (!rhs) return rhs;
194 lhs = relational(lhs.value(), rhs.value(), op);
195 if (!lhs) return lhs;
196 }
197 }

199 ExprResult parse_additive() {
200 auto lhs = parse_multiplicative();
201 if (!lhs) return lhs;
202 while (true) {
203 if (match('+')) {
204 auto rhs = parse_multiplicative();
205 if (!rhs) return rhs;
206 lhs = add(lhs.value(), rhs.value());
207 } else if (match('-')) {
208 auto rhs = parse_multiplicative();
209 if (!rhs) return rhs;
210 lhs = sub(lhs.value(), rhs.value());
211 } else {
212 return lhs;
213 }
214 if (!lhs) return lhs;
215 }

218 ExprResult parse_multiplicative() {
219 auto lhs = parse_power();
220 if (!lhs) return lhs;
221 while (true) {
222 if (match_text("**")) {
223 return fail("operator '**' is not supported; use '^'");
224 } else if (match('*')) {
225 auto rhs = parse_power();
226 if (!rhs) return rhs;
227 lhs = mul(lhs.value(), rhs.value());
228 } else if (match('/')) {
229 auto rhs = parse_power();
230 if (!rhs) return rhs;
231 lhs = div(lhs.value(), rhs.value());
232 } else {
233 return lhs;
234 }
235 if (!lhs) return lhs;
236 }

239 ExprResult parse_power() {
240 auto base = parse_unary();
241 if (!base) return base;
242 if (match_text("**")) {
243 return fail("operator '**' is not supported; use '^'");
244 }
245 if (match('^')) {
246 auto exponent = parse_power();
247 if (!exponent) return exponent;
248 return pow(base.value(), exponent.value());
249 }
250 return base;

253 ExprResult parse_unary() {
254 if (match('+')) return parse_unary();
255 if (match('-')) {
256 auto operand = parse_unary();
257 if (!operand) return operand;
258 return neg(operand.value());
259 }
260 return parse_primary();

263 ExprResult parse_primary() {
264 skip_space();
265 if (match('(')) {
266 auto inner = parse_logical_or();
267 if (!inner) return inner;
268 if (match(',')) {
269 auto rhs = parse_logical_or();
270 if (!rhs) return rhs;
271 if (!match(')')) return fail("expected ')' after interval");
272 return interval(inner.value(), rhs.value(), false, false);
273 }
274 if (!match(')')) return fail("expected ')'");
275 return inner;
276 }
277 if (match('{')) {
278 return parse_set_literal();
279 }
280 if (match('[')) {
281 return parse_interval_literal(true);
282 }
283 if (std::isdigit(static_cast<unsigned char>(peek())) || peek() == '.') {
284 return parse_number();
285 }
286 if (is_ident_start(static_cast<unsigned char>(peek()))) {
287 return parse_identifier_or_call();
288 }
289 return fail("expected expression");
290 }

292 ExprResult parse_set_literal() {
293 std::vector<ExprPtr> elements;
294 if (!match('}')) {
295 while (true) {
296 auto element = parse_logical_or();
297 if (!element) return element;
298 elements.push_back(element.value());
299 if (match('}')) break;
300 if (!match(',')) return fail("expected ',' or '}' in set literal");
301 }
302 }
303 return finite_set(elements);
304 }

306 ExprResult parse_interval_literal(bool lower_closed) {
307 auto lower = parse_logical_or();
308 if (!lower) return lower;
309 if (!match(',')) return fail("expected ',' in interval literal");
310 auto upper = parse_logical_or();
311 if (!upper) return upper;
312 bool upper_closed;
313 if (match(']')) {
314 upper_closed = true;
315 } else if (match(')')) {
316 upper_closed = false;
317 } else {
318 return fail("expected ']' or ')' after interval literal");
319 }
320 return interval(lower.value(), upper.value(), lower_closed, upper_closed);
321 }

323 ExprResult parse_number() {
324 skip_space();
325 const std::size_t start = pos_;
326 bool saw_digit = false;
327 while (std::isdigit(static_cast<unsigned char>(peek()))) {
328 saw_digit = true;
329 ++pos_;
330 }
331 if (peek() == '.') {
332 ++pos_;
333 while (std::isdigit(static_cast<unsigned char>(peek()))) {
334 saw_digit = true;
335 ++pos_;
336 }
337 }
338 if (!saw_digit) return fail("invalid number literal");
339 if (peek() == 'e' || peek() == 'E') {
340 ++pos_;
341 if (peek() == '+' || peek() == '-') ++pos_;
342 bool saw_exp_digit = false;
343 while (std::isdigit(static_cast<unsigned char>(peek()))) {
344 saw_exp_digit = true;
345 ++pos_;
346 }
347 if (!saw_exp_digit) return fail("invalid number exponent");
348 }
349 try {
350 return rational(Rational(source_.substr(start, pos_ - start)));
351 } catch (const std::bad_alloc&) {
352 return expression_failure(CasErrc::ResourceLimit,
353 "number literal allocation failed",
354 kParseOperation);
355 } catch (const std::exception& error) {
356 return expression_failure(CasErrc::ParseError, error.what(),
357 kParseOperation);
358 }

361 std::string parse_identifier() {
362 skip_space();
363 const std::size_t start = pos_;
364 if (!is_ident_start(static_cast<unsigned char>(peek()))) return {};
365 ++pos_;
366 while (is_ident_continue(static_cast<unsigned char>(peek()))) ++pos_;
367 return source_.substr(start, pos_ - start);

370 ExprResult parse_identifier_or_call() {
371 auto name = parse_identifier();
372 if (name.empty()) return fail("expected identifier");
373 skip_space();
374 if (match('(')) {
375 std::vector<ExprPtr> arguments;
376 bool closed = match(')');
377 if (!closed) {
378 do {
379 auto argument = parse_logical_or();
380 if (!argument) return argument;
381 arguments.push_back(argument.value());
382 if (match(')')) {
383 closed = true;
384 break;
385 }
386 } while (match(','));
387 }
388 if (!closed) return fail("expected ')'");
389 return apply_function(name, arguments);
390 }
391 if (name == "pi" || name == "π") return pi();
392 if (name == "e") return e();
393 if (name == "phi") return phi();
394 if (is_imaginary_unit_name(name)) return imaginary_unit();
395 return sym(name);
396 }

549 ExprResult apply_function(const std::string& name,
550 const std::vector<ExprPtr>& arguments) {
551 if (name == "sin" && arguments.size() == 1) return sin(arguments[0]);
552 if (name == "cos" && arguments.size() == 1) return cos(arguments[0]);
553 if (name == "tan" && arguments.size() == 1) return tan(arguments[0]);
554 if (name == "asin" && arguments.size() == 1) return asin(arguments[0]);
555 if (name == "acos" && arguments.size() == 1) return acos(arguments[0]);
556 if (name == "atan" && arguments.size() == 1) return atan(arguments[0]);
557 if (name == "sqrt" && arguments.size() == 1) return sqrt(arguments[0]);
558 if (name == "exp" && arguments.size() == 1) return exp(arguments[0]);
559 if ((name == "ln" || name == "log") && arguments.size() == 1) return log(arguments[0]);
560 if (name == "log" && arguments.size() == 2) {
561 try {
562 auto result = SymbolicExpr::log(arguments[0], arguments[1]);
563 return ExprResult::success(std::move(result));
564 } catch (const std::bad_alloc&) {
565 return expression_failure(CasErrc::ResourceLimit,
566 "log expression allocation failed",
567 kParseOperation);
568 } catch (const std::exception& error) {
569 return expression_failure(CasErrc::InvalidArgument, error.what(),
570 kParseOperation);
571 }
572 }
573 if (name == "atan2" && arguments.size() == 2) {
574 try {
575 auto result = SymbolicExpr::atan2(arguments[0], arguments[1]);
576 return ExprResult::success(std::move(result));
577 } catch (const std::bad_alloc&) {
578 return expression_failure(CasErrc::ResourceLimit,
579 "atan2 expression allocation failed",
580 kParseOperation);
581 } catch (const std::exception& error) {
582 return expression_failure(CasErrc::InvalidArgument, error.what(),
583 kParseOperation);
584 }
585 }
586 if (name == "log10" && arguments.size() == 1) return log10(arguments[0]);
587 if (name == "floor" && arguments.size() == 1) return floor(arguments[0]);
588 if (name == "ceil" && arguments.size() == 1) return ceil(arguments[0]);
589 if (name == "round" && arguments.size() == 1) return round(arguments[0]);
590 if (name == "abs" && arguments.size() == 1) return abs(arguments[0]);
591 if (name == "real" && arguments.size() == 1) return real(arguments[0]);
592 if (name == "imag" && arguments.size() == 1) return imag(arguments[0]);
593 if (name == "conj" && arguments.size() == 1) return conj(arguments[0]);
594 if ((name == "Integral" || name == "integral") &&
595 (arguments.size() == 2 || arguments.size() == 4)) {
596 auto variable = std::dynamic_pointer_cast<const VariableNode>(
597 lamina::detail::node(arguments[1]));
598 if (!variable) {
599 return fail("Integral variable must be a symbol");
600 }
601 try {
602 auto node = lamina::detail::make_node<IntegralNode>(
603 lamina::detail::node(arguments[0]), variable->name(),
604 arguments.size() == 4
605 ? lamina::detail::node(arguments[2])
606 : nullptr,
607 arguments.size() == 4
608 ? lamina::detail::node(arguments[3])
609 : nullptr);
610 return ExprResult::success(
611 lamina::detail::make_expression_ptr(std::move(node)));
612 } catch (const std::bad_alloc&) {
613 return expression_failure(
614 CasErrc::ResourceLimit,
615 "Integral expression allocation failed",
616 kParseOperation);
617 } catch (const std::exception& error) {
618 return expression_failure(
619 CasErrc::InvalidArgument, error.what(), kParseOperation);
620 }
621 }
622 if ((name == "limit" || name == "Limit") && arguments.size() == 4) {
623 auto variable = std::dynamic_pointer_cast<const VariableNode>(
624 lamina::detail::node(arguments[1]));
625 auto direction = std::dynamic_pointer_cast<const VariableNode>(
626 lamina::detail::node(arguments[3]));
627 if (!variable || !direction) {
628 return fail("limit variable and direction must be symbols");
629 }
630 LimitDirection parsed_direction;
631 if (direction->name() == "both") {
632 parsed_direction = LimitDirection::Both;
633 } else if (direction->name() == "below" ||
634 direction->name() == "left" ||
635 direction->name() == "from_below") {
636 parsed_direction = LimitDirection::FromBelow;
637 } else if (direction->name() == "above" ||
638 direction->name() == "right" ||
639 direction->name() == "from_above") {
640 parsed_direction = LimitDirection::FromAbove;
641 } else {
642 return fail("invalid limit direction");
643 }
644 return ExprResult::success(lamina::detail::make_expression_ptr(
645 lamina::detail::make_node<LimitNode>(
646 lamina::detail::node(arguments[0]), variable->name(),
647 lamina::detail::node(arguments[2]), parsed_direction)));
648 }
649 if ((name == "rootof" || name == "RootOf") &&
650 arguments.size() == 3) {
651 auto variable = std::dynamic_pointer_cast<const VariableNode>(
652 lamina::detail::node(arguments[1]));
653 auto index = std::dynamic_pointer_cast<const NumberNode>(
654 lamina::detail::node(arguments[2]));
655 if (!variable || !index ||
656 std::holds_alternative<lmmc_real_t>(index->value())) {
657 return fail("RootOf variable and index are invalid");
658 }
659 BigInt index_value;
660 if (std::holds_alternative<BigInt>(index->value())) {
661 index_value = std::get<BigInt>(index->value());
662 } else {
663 const auto& rational = std::get<Rational>(index->value());
664 if (!rational.is_integer()) {
665 return fail("RootOf index must be an integer");
666 }
667 index_value = rational.to_BigInt();
668 }
669 const auto converted = index_value.try_to_int64();
670 if (!converted || *converted < 0) {
671 return fail("RootOf index is negative or too large");
672 }
673 auto root = make_rootof_checked(
674 arguments[0], variable->name(),
675 static_cast<std::size_t>(*converted));
676 if (!root) return ExprResult::failure(root.error());
677 return ExprResult::success(std::move(root.value()));
678 }
679 if (name == "max" && !arguments.empty()) {
680 return function_node(name, arguments, FunctionNode::FuncType::Max);
681 }
682 if (name == "min" && !arguments.empty()) {
683 return function_node(name, arguments, FunctionNode::FuncType::Min);
684 }
685 if (name == "clamp" && arguments.size() == 3) {
686 return clamp(arguments[0], arguments[1], arguments[2]);
687 }
688 if (name == "sin" || name == "cos" || name == "tan" ||
689 name == "asin" || name == "acos" || name == "atan" ||
690 name == "sqrt" || name == "exp" || name == "ln" ||
691 name == "log" || name == "log10" || name == "floor" ||
692 name == "ceil" || name == "round" || name == "abs" ||
693 name == "real" || name == "imag" || name == "conj" ||
694 name == "atan2" || name == "clamp" ||
695 name == "Integral" || name == "integral" ||
696 name == "Limit" || name == "limit" ||
697 name == "RootOf" || name == "rootof") {
698 return fail("invalid argument count for function '" + name + "'");
699 }
700 try {
701 std::vector<std::shared_ptr<const SymbolicNode>> nodes;
702 nodes.reserve(arguments.size());
703 for (const auto& argument : arguments) {
