128std::shared_ptr<const SymbolicNode> NormalizationVisitor::normalize_matrix_product(
129 const std::vector<std::shared_ptr<const SymbolicNode>>& operands) {
130 std::vector<std::shared_ptr<const SymbolicNode>> nonscalar;
131 std::size_t nonscalar_nodes = 0;
132 auto scalar = detail::make_node<NumberNode>(BigInt(1));
133 for (const auto& operand : operands) {
134 if (const auto number = std::dynamic_pointer_cast<const NumberNode>(operand)) {
135 scalar = multiply_numbers(scalar, number);
136 } else {
137 normalization_append(rewrite_budget(), nonscalar_nodes, nonscalar, operand);
138 }
139 }
140 std::vector<std::shared_ptr<const SymbolicNode>> fused;
141 std::size_t fused_nodes = 0;
142 std::size_t last_nodes = 0;
143 if (!nonscalar.empty()) {
144 normalization_append(rewrite_budget(), fused_nodes, fused, nonscalar.front());
145 last_nodes = fused_nodes;
146 }
147 for (std::size_t i = 1; i < nonscalar.size(); ++i) {
148 if (auto product = fuse_matrix_product(fused.back(), nonscalar[i])) {
149 if (rewrite_budget()) {
150 const auto replacement_nodes = rewrite_budget()->measure(product);
151 fused_nodes = rewrite_budget()->append_size(fused_nodes - last_nodes, replacement_nodes);
152 last_nodes = replacement_nodes;
153 }
154 fused.back() = std::move(product);
155 } else {
156 const auto previous_nodes = fused_nodes;
157 normalization_append(rewrite_budget(), fused_nodes, fused, nonscalar[i]);
158 last_nodes = fused_nodes - previous_nodes;
159 }
160 }
161 if (fused.size() == 1 && !scalar->is_one()) {
162 if (const auto matrix = std::dynamic_pointer_cast<const MatrixNode>(fused.front())) {
163 return scale_matrix(*matrix, scalar);
164 }
165 }
166 if (!scalar->is_one()) {
167 if (rewrite_budget()) { (void)rewrite_budget()->append_size(fused_nodes, 1); }
168 fused.insert(fused.begin(), scalar);
169 }
170 return make_normalized_multiply_node(fused, rewrite_budget());
171}
