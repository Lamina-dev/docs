87/**
88 * @brief Solve an equation with assumption-based domain filtering.
89 *
90 * Calls the existing SymbolicExpr::solve() to compute all solutions, then
91 * applies domain filtering based on the variable's declared domain in the
92 * AssumptionContext. Solutions that violate the domain constraint are excluded.
93 *
94 * Domain filtering rules:
95 * - Real: exclude solutions containing sqrt(-1) or non-real numeric values
96 * - PositiveInt: exclude non-integer or <= 0 solutions
97 * - NonNegative (Natural): exclude solutions evaluating to < 0
98 *
99 * @param equation The equation to solve (equal to zero, or RelationalNode)
100 * @param variable The variable name to solve for
101 * @param ctx Optional assumption context; if nullptr, all solutions returned unfiltered
102 * @return Filtered solutions preserving original order
103 */
104LAMINA_API std::vector<std::shared_ptr<SymbolicExpr>> solve_with_assumptions(
105 const std::shared_ptr<SymbolicExpr>& equation,
106 const std::string& variable,
107 const AssumptionContext* ctx = nullptr);
