32static void collect_transcendental_subexprs(
33 const std::shared_ptr<const SymbolicNode>& node,
34 const std::string& var,
35 std::vector<std::shared_ptr<SymbolicExpr>>& candidates) {
36
37 if (!node) {
38 return;
39 }
40
41 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(node)) {
42 if (function_candidate(*func, var)) {
43 candidates.push_back(LMCAS::detail::make_expression_ptr(node));
44 }
45
46 for (auto& arg : func->arguments()) {
47 collect_transcendental_subexprs(arg, var, candidates);
48 }
49 return;
50 }
51
52 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
53 if (power_candidate(*pow, var)) {
54 candidates.push_back(LMCAS::detail::make_expression_ptr(pow->base()));
55 }
56
57 collect_transcendental_subexprs(pow->base(), var, candidates);
58 collect_transcendental_subexprs(pow->exponent(), var, candidates);
59 return;
60 }
61
62 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
63 for (auto& op : add->operands()) {
64 collect_transcendental_subexprs(op, var, candidates);
65 }
66 return;
67 }
68
69 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
70 for (auto& op : mul->operands()) {
71 collect_transcendental_subexprs(op, var, candidates);
72 }
73 return;
74 }

77static void deduplicate_candidates(std::vector<std::shared_ptr<SymbolicExpr>>& candidates) {
78 std::vector<std::shared_ptr<SymbolicExpr>> unique;
79 std::vector<std::string> seen;
80 for (auto& c : candidates) {
81 std::string s = c->to_string();
82 bool found = false;
83 for (auto& existing : seen) {
84 if (existing == s) { found = true; break; }
85 }
86 if (!found) {
87 seen.push_back(s);
88 unique.push_back(c);
89 }
90 }
91 candidates = std::move(unique);
92}

94static BigInt product_exp_multiplier(const MultiplyNode& product, const std::string& var) {
95 std::shared_ptr<const NumberNode> number;
96 bool has_variable = false;
97 bool has_other = false;
98 for (const auto& operand : product.operands()) {
99 if (auto numeric = std::dynamic_pointer_cast<const NumberNode>(operand)) {
100 if (!number) {
101 number = numeric;
102 } else {
103 has_other = true;
104 }
105 } else if (auto variable = std::dynamic_pointer_cast<const VariableNode>(operand)) {
106 if (!variable->is_constant() && variable->name() == var) {
107 has_variable = true;
108 } else {
109 has_other = true;
110 }
111 } else {
112 has_other = true;
113 }
114 }
115 BigInt value;
116 if (has_variable && number && !has_other &&
117 try_get_integer_value(number, value) && value > 0) {
118 return value;
119 }
120 return 0;
121}

123static BigInt extract_exp_multiplier(const std::shared_ptr<const SymbolicNode>& node, const std::string& var) {
124 auto func = std::dynamic_pointer_cast<const FunctionNode>(node);
125 if (!func || func->type() != FunctionNode::FuncType::Exp || func->arguments().size() != 1) {
126 return 0;
127 }
128
129 auto arg = func->arguments()[0];
130
131 if (auto v = std::dynamic_pointer_cast<const VariableNode>(arg)) {
132 if (!v->is_constant() && v->name() == var) {
133 return 1;
134 }
135 }
136
137 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(arg)) {
138 return product_exp_multiplier(*mul, var);
139 }
140
141 return 0;

144static std::shared_ptr<const SymbolicNode> rewrite_exp_as_u_power(
145 const std::shared_ptr<const SymbolicNode>& node,
146 const std::string& var,
147 const std::string& u_var) {
148
149 if (!node) {
150 return node;
151 }
152
153 const BigInt k = extract_exp_multiplier(node, var);
154 if (k > 0) {
155 auto u_node = LMCAS::detail::make_node<VariableNode>(u_var);
156 if (k == 1) {
157 return u_node;
158 }
159 return SymbolicFactory::create_power(u_node, LMCAS::detail::make_node<NumberNode>(k));
160 }
161
162 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
163 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
164 new_ops.reserve(add->operands().size());
165 for (auto& op : add->operands()) {
166 new_ops.push_back(rewrite_exp_as_u_power(op, var, u_var));
167 }
168 return SymbolicFactory::create_add(std::move(new_ops));
169 }
170
171 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
172 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
173 new_ops.reserve(mul->operands().size());
174 for (auto& op : mul->operands()) {
175 new_ops.push_back(rewrite_exp_as_u_power(op, var, u_var));
176 }
177 return SymbolicFactory::create_multiply(std::move(new_ops));
178 }
179
180 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
181 auto new_base = rewrite_exp_as_u_power(pow->base(), var, u_var);
182 auto new_exp = rewrite_exp_as_u_power(pow->exponent(), var, u_var);
183 return SymbolicFactory::create_power(new_base, new_exp);
184 }
185
186 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(node)) {
187 std::vector<std::shared_ptr<const SymbolicNode>> new_args;
188 new_args.reserve(func->arguments().size());
189 for (auto& arg : func->arguments()) {
190 new_args.push_back(rewrite_exp_as_u_power(arg, var, u_var));
191 }
192 return LMCAS::detail::make_node<FunctionNode>(func->type(), std::move(new_args));
193 }
194
195 return node;
196}

322Result<std::optional<SubstitutionResult>> detect_substitution(
323 const std::shared_ptr<SymbolicExpr>& expr,
324 const std::string& var) {
325
326 if (!expr || !LMCAS::detail::node(expr)) {
327 return std::optional<SubstitutionResult>{};
328 }
329
330 if (!expression_depends_on_variable(LMCAS::detail::node(expr), var)) {
331 return std::optional<SubstitutionResult>{};
332 }
333
334 const std::string u_var = "_u_subst";
335
336 std::vector<std::shared_ptr<SymbolicExpr>> candidates;
337 collect_transcendental_subexprs(LMCAS::detail::node(expr), var, candidates);
338 deduplicate_candidates(candidates);
339
340 for (auto& h : candidates) {
341
342 auto redundant = redundant_variable_candidate(h, expr, var);
343 if (!redundant) {
344 return Result<std::optional<SubstitutionResult>>::failure(redundant.error());
345 }
346 if (redundant.value()) {
347 continue;
348 }
349
350
351
352 SubstNodeVisitor visitor;
353 visitor.target = LMCAS::detail::node(h);
354 visitor.u_name = u_var;
355
356 auto substituted_node = visitor.replace(LMCAS::detail::node(expr));
357 auto substituted_expr = LMCAS::detail::make_expression_ptr(substituted_node);
358 substituted_expr = substituted_expr->simplify();
359
360 auto polynomial = is_substitution_polynomial(substituted_expr, var, u_var);
361 if (!polynomial) {
362 return Result<std::optional<SubstitutionResult>>::failure(polynomial.error());
363 }
364 if (polynomial.value()) {
365 return std::optional<SubstitutionResult>{{h, substituted_expr, u_var}};
366 }
367 }
368
369 if (has_exp_k_var_terms(LMCAS::detail::node(expr), var)) {
370 auto rewritten_node = rewrite_exp_as_u_power(LMCAS::detail::node(expr), var, u_var);
371 auto rewritten_expr = LMCAS::detail::make_expression_ptr(rewritten_node);
372 rewritten_expr = rewritten_expr->simplify();
373
374 auto polynomial = is_substitution_polynomial(rewritten_expr, var, u_var);
375 if (!polynomial) {
376 return Result<std::optional<SubstitutionResult>>::failure(polynomial.error());
377 }
378 if (polynomial.value()) {
379 auto exp_x = SymbolicExpr::exp(SymbolicExpr::variable(var));
380 return std::optional<SubstitutionResult>{{exp_x, rewritten_expr, u_var}};
381 }
382 }
383
384 return std::optional<SubstitutionResult>{};
385}
