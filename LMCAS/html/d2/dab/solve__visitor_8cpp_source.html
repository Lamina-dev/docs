93static SolveResult conditional_linear_polynomial(
94 const Polynomial<SymbolicPolyCoeff>& polynomial,
95 const ExprPtr& leading, const std::string& variable) {
96 auto constant = get_coeff(polynomial, 0);
97 auto zero = SymbolicExpr::number(0);
98 auto generic = polynomial_logic(
99 polynomial_relation(leading, zero, RelationOp::NEQ),
100 polynomial_relation(
101 SymbolicExpr::multiply(
102 leading, SymbolicExpr::variable(variable)),
103 SymbolicExpr::multiply(
104 SymbolicExpr::number(-1), constant),
105 RelationOp::EQ),
106 LogicalNode::Op::And);
107 auto degenerate = polynomial_logic(
108 polynomial_relation(leading, zero, RelationOp::EQ),
109 polynomial_relation(constant, zero, RelationOp::EQ),
110 LogicalNode::Op::And);
111 return SolutionSet{ConditionalSolutions{ConditionSet{
112 variable,
113 polynomial_logic(generic, degenerate, LogicalNode::Op::Or),
114 {}}}};
115}

117static SolveResult solve_symbolic_quadratic(
118 const Polynomial<SymbolicPolyCoeff>& polynomial,
119 const ExprPtr& leading, const FactsQuery& facts,
120 ComputationContext& context) {
121 auto linear = get_coeff(polynomial, 1);
122 auto constant = get_coeff(polynomial, 0);
123 auto discriminant = SymbolicExpr::add(
124 SymbolicExpr::power(linear, SymbolicExpr::number(2)),
125 SymbolicExpr::multiply(
126 SymbolicExpr::number(-4),
127 SymbolicExpr::multiply(leading, constant)))->simplify();
128 auto nonzero = detail::query_nonzero_value(
129 detail::node(discriminant), facts, Domain::Complex, context);
130 if (!nonzero) {
131 return SolveResult::failure(nonzero.error());
132 }
133
134 auto radical = detail::make_expression_ptr(
135 detail::make_node<FunctionNode>(
136 FunctionNode::FuncType::Sqrt,
137 std::vector<std::shared_ptr<const SymbolicNode>>{
138 detail::node(discriminant)}));
139 auto negative_linear =
140 SymbolicExpr::multiply(SymbolicExpr::number(-1), linear);
141 auto denominator =
142 SymbolicExpr::multiply(SymbolicExpr::number(2), leading);
143 auto positive = SymbolicExpr::divide(
144 SymbolicExpr::add(negative_linear, radical),
145 denominator)->simplify();
146 auto negative = SymbolicExpr::divide(
147 SymbolicExpr::add(
148 negative_linear,
149 SymbolicExpr::multiply(
150 SymbolicExpr::number(-1), radical)),
151 denominator)->simplify();
152 auto repeated =
153 SymbolicExpr::divide(negative_linear, denominator)->simplify();
154
155 if (nonzero.value() == Tribool::False) {
156 return SolutionSet{FiniteSolutions{{
157 FiniteSolution{repeated, 2, {}}}}};
158 }
159 if (nonzero.value() == Tribool::True) {
160 return SolutionSet{FiniteSolutions{{
161 FiniteSolution{positive, 1, {}},
162 FiniteSolution{negative, 1, {}}}}};
163 }
164
165 auto zero = SymbolicExpr::number(0);
166 auto distinct =
167 polynomial_relation(discriminant, zero, RelationOp::NEQ);
168 auto coincident =
169 polynomial_relation(discriminant, zero, RelationOp::EQ);
170 return SolutionSet{FiniteSolutions{{
171 FiniteSolution{positive, 1, {distinct}},
172 FiniteSolution{negative, 1, {distinct}},
173 FiniteSolution{positive, 2, {coincident}}}}};
174}

176static SolveResult solve_symbolic_polynomial_set(
177 const Polynomial<SymbolicPolyCoeff>& polynomial,
178 const ExprPtr& original, const std::string& variable,
179 ComputationContext& context, const SolveOptions& options) {
180 auto valid = validate_symbolic_polynomial_input(original, options);
181 if (!valid) {
182 return SolveResult::failure(valid.error());
183 }
184
185 std::optional<detail::AssumptionFacts> adapter;
186 if (context.assumptions()) {
187 adapter.emplace(*context.assumptions());
188 }
189 const FactsQuery& facts = adapter
190 ? static_cast<const FactsQuery&>(*adapter)
191 : detail::no_facts();
192 int degree = polynomial.degree();
193 while (degree >= 0) {
194 auto leading = get_coeff(polynomial, degree)->simplify();
195 auto nonzero = detail::query_nonzero_value(
196 detail::node(leading), facts, Domain::Complex, context);
197 if (!nonzero) {
198 return SolveResult::failure(nonzero.error());
199 }
200 if (nonzero.value() == Tribool::False) {
201 --degree;
202 continue;
203 }
204 if (nonzero.value() == Tribool::Unknown || degree >= 3) {
205 return degree == 1
206 ? conditional_linear_polynomial(
207 polynomial, leading, variable)
208 : conditional_equation(original, variable);
209 }
210 if (degree == 0) {
211 return SolutionSet{EmptySolutions{}};
212 }
213 if (degree == 2) {
214 return solve_symbolic_quadratic(
215 polynomial, leading, facts, context);
216 }
217 auto value = SymbolicExpr::divide(
218 SymbolicExpr::multiply(
219 SymbolicExpr::number(-1),
220 get_coeff(polynomial, 0)),
221 leading)->simplify();
222 return SolutionSet{FiniteSolutions{{
223 FiniteSolution{value, 1, {}}}}};
224 }
225 return SolutionSet{UniversalSolutions{}};
226}

228static SolveResult solve_product_factor(const detail::SymbolicNodePtr& node,
229 const std::string& variable, ComputationContext& context, const SolveOptions& options,
230 std::size_t& multiplicity) {
231 auto factor = detail::make_expression_ptr(node);
232 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
233 BigInt exponent;
234 if (try_get_integer_value(
235 std::dynamic_pointer_cast<const NumberNode>(power->exponent()), exponent)) {
236 if (exponent <= BigInt(0)) { return SolutionSet{EmptySolutions{}}; }
237 if (exponent > BigInt(static_cast<std::int64_t>(std::numeric_limits<int>::max()))) {
238 return SolveResult::failure(CasErrc::ResourceLimit, "root multiplicity overflow", "solve.product");
239 }
240 multiplicity = static_cast<std::size_t>(*exponent.try_to_int64());
241 factor = detail::make_expression_ptr(power->base());
242 }
243 }
244 return solve_equation(factor, variable, context, options);
245}

247static Result<void> append_product_roots(FiniteSolutions& values, std::size_t multiplicity,
248 std::vector<FiniteSolution>& finite) {
249 for (auto& value : values.values) {
250 if (value.multiplicity > std::numeric_limits<std::size_t>::max() / multiplicity) {
251 return Result<void>::failure(CasErrc::ResourceLimit, "root multiplicity overflow", "solve.product");
252 }
253 value.multiplicity *= multiplicity;
254 auto existing = std::find_if(finite.begin(), finite.end(), [&](const FiniteSolution& candidate) {
255 return detail::node(candidate.value)->equals(*detail::node(value.value)) &&
256 candidate.conditions.empty() && value.conditions.empty();
257 });
258 if (existing == finite.end()) { finite.push_back(std::move(value)); }
259 else {
260 if (existing->multiplicity > std::numeric_limits<std::size_t>::max() - value.multiplicity) {
261 return Result<void>::failure(CasErrc::ResourceLimit, "root multiplicity overflow", "solve.product");
262 }
263 existing->multiplicity += value.multiplicity;
264 }
265 }
266 return Result<void>::success();
267}

269static void append_product_families(ParametricSolutions& values,
270 const std::set<std::string>& original_names, std::vector<ParametricSolution>& families) {
271 for (auto& family : values.values) {
272 auto occupied = original_names;
273 const auto value_names = detail::all_variable_names(detail::node(family.value));
274 occupied.insert(value_names.begin(), value_names.end());
275 for (auto& parameter : family.integer_parameters) {
276 if (!original_names.count(parameter)) { continue; }
277 std::string fresh = "_k";
278 for (std::size_t suffix = 1; occupied.count(fresh); ++suffix) {
279 fresh = "_k" + std::to_string(suffix);
280 }
281 auto symbol = SymbolicExpr::variable(fresh);
282 family.value = substitute_raw(family.value, parameter, symbol);
283 for (auto& condition : family.conditions) {
284 condition = substitute_raw(condition, parameter, symbol);
285 }
286 parameter = fresh;
287 occupied.insert(fresh);
288 }
289 families.push_back(std::move(family));
290 }
291}

293static std::optional<SolveResult> solve_product_set(const ExprPtr& expression,
294 const std::string& variable, ComputationContext& context, const SolveOptions& options) {
295 auto product = std::dynamic_pointer_cast<const MultiplyNode>(detail::node(expression));
296 if (!product) { return std::nullopt; }
297 std::vector<FiniteSolution> finite;
298 std::vector<ParametricSolution> families;
299 const auto original_names = detail::all_variable_names(detail::node(expression));
300 for (const auto& node : product->operands()) {
301 std::size_t multiplicity = 1;
302 auto solved = solve_product_factor(node, variable, context, options, multiplicity);
303 if (!solved) {
304 if (solved.error().code == CasErrc::Inconclusive) { return conditional_equation(expression, variable); }
305 return SolveResult::failure(solved.error());
306 }
307 if (std::holds_alternative<EmptySolutions>(solved.value())) { continue; }
308 if (std::holds_alternative<UniversalSolutions>(solved.value())) { return conditional_equation(expression, variable); }
309 if (auto values = std::get_if<FiniteSolutions>(&solved.value())) {
310 auto appended = append_product_roots(*values, multiplicity, finite);
311 if (!appended) { return SolveResult::failure(appended.error()); }
312 } else if (auto values = std::get_if<ParametricSolutions>(&solved.value())) {
313 append_product_families(*values, original_names, families);
314 } else { return conditional_equation(expression, variable); }
315 }
316 if (!finite.empty() && !families.empty()) { return conditional_equation(expression, variable); }
317 if (!families.empty()) { return SolveResult::success(SolutionSet{ParametricSolutions{std::move(families)}}); }
318 if (!finite.empty()) { return SolveResult::success(SolutionSet{FiniteSolutions{std::move(finite)}}); }
319 return SolveResult::success(SolutionSet{EmptySolutions{}});
320}

322static SolveResult collect_exact_solutions(
323 std::vector<std::shared_ptr<SymbolicExpr>> roots, ComputationContext& context) {
324 constexpr const char* operation = "solve_equation";
325 std::vector<FiniteSolution> solutions;
326 for (auto& root : roots) {
327 auto root_step = context.consume_steps(1, operation);
328 if (!root_step) {
329 return SolveResult::failure(root_step.error());
330 }
331 if (!root || !LMCAS::detail::node(root)) {
332 return SolveResult::failure(
333 CasErrc::InternalInvariant,
334 "polynomial solver produced a null root", operation);
335 }
336 auto existing = std::find_if(
337 solutions.begin(), solutions.end(),
338 [&](const FiniteSolution& solution) {
339 return LMCAS::detail::node(solution.value)->equals(
340 *LMCAS::detail::node(root));
341 });
342 if (existing != solutions.end()) {
343 ++existing->multiplicity;
344 } else {
345 solutions.push_back(
346 FiniteSolution{std::move(root), 1, {}});
347 }
348 }
349 return SolveResult::success(
350 SolutionSet{FiniteSolutions{std::move(solutions)}});
351}

353static SolveVectorResult certified_quartic_roots(const Polynomial<Rational>& remainder,
354 const ExprPtr& expression, const std::string& var, ComputationContext& context) {
355 auto candidates = solve_quartic(
356 SymbolicExpr::number(remainder.coeffs[4]),
357 SymbolicExpr::number(remainder.coeffs[3]),
358 SymbolicExpr::number(remainder.coeffs[2]),
359 SymbolicExpr::number(remainder.coeffs[1]),
360 SymbolicExpr::number(remainder.coeffs[0]), var);
361 bool certified = candidates.size() == 4;
362 for (std::size_t i = 0; certified && i < candidates.size(); ++i) {
363 if (contains_inexact_number(detail::node(candidates[i]))) {
364 certified = false;
365 break;
366 }
367 auto proof = check_zero_residual(substitute_raw(expression, var, candidates[i]), context);
368 if (!proof) { return SolveVectorResult::failure(proof.error()); }
369 certified = std::holds_alternative<ProvedZeroResidual>(proof.value());
370 for (std::size_t j = 0; certified && j < i; ++j) {
371 auto equal = detail::compare_pointwise_values(
372 candidates[i], candidates[j], context, Domain::Complex);
373 if (!equal) { return SolveVectorResult::failure(equal.error()); }
374 certified = equal.value() == Tribool::False;
375 }
376 }
377 if (certified) { return candidates; }
378 return std::vector<ExprPtr>{};
379}

381static SolveVectorResult solve_higher_exact_roots(const Polynomial<Rational>& remainder,
382 bool bounded_candidates, const std::string& var, ComputationContext& context) {
383 auto expression = poly_to_symbolic(remainder);
384 std::vector<ExprPtr> represented_roots;
385 if (remainder.degree() == 4 && bounded_candidates) {

387 if (!candidates) { return SolveVectorResult::failure(candidates.error()); }

394 represented_roots.push_back(std::move(root.value()));
395 }
396 }
397 return represented_roots;

400static SolveVectorResult solve_exact_factor(Polynomial<Rational> remainder,
401 const std::string& var, ComputationContext& context) {
402 std::vector<ExprPtr> factor_roots;
403 const bool bounded_candidates =
404 polynomial_solver_detail::bounded_rational_root_search(remainder);
405 if (bounded_candidates) {
406 for (const auto& root : find_rational_roots(remainder)) {
407 Polynomial<Rational> divisor({-root, Rational(1)}, var);
408 auto division = remainder.div_mod(divisor);
409 if (!division.second.is_zero()) { continue; }
410 factor_roots.push_back(SymbolicExpr::number(root));
411 remainder = std::move(division.first);
412 }
413 }
414 if (remainder.degree() == 1) {
415 factor_roots.push_back(SymbolicExpr::number(-remainder.coeffs[0] / remainder.coeffs[1]));
416 } else if (remainder.degree() == 2) {
417 auto quadratic = detail::real_quadratic_roots_checked(
418 SymbolicExpr::number(remainder.coeffs[2]),
419 SymbolicExpr::number(remainder.coeffs[1]),
420 SymbolicExpr::number(remainder.coeffs[0]), context);
421 if (!quadratic) { return SolveVectorResult::failure(quadratic.error()); }
422 for (const auto& root : quadratic.value()) {
423 factor_roots.push_back(detail::make_expression_ptr(
424 SymbolicFactory::create_complex(detail::node(root.real), detail::node(root.imag))));
425 }
426 } else if (remainder.degree() > 2) {
427 auto represented_roots = solve_higher_exact_roots(remainder, bounded_candidates, var, context);
428 if (!represented_roots) { return SolveVectorResult::failure(represented_roots.error()); }
429 factor_roots.insert(factor_roots.end(), represented_roots.value().begin(), represented_roots.value().end());
430 }
431 return factor_roots;
432}

434static SolveResult solve_exact_polynomial_set(const Polynomial<Rational>& exact,
435 const std::string& var,
436 ComputationContext& context) {
437 if (exact.is_zero()) { return SolutionSet{UniversalSolutions{}}; }
438 if (exact.degree() == 0) { return SolutionSet{EmptySolutions{}}; }
439 std::vector<ExprPtr> roots;
440 for (auto& factor : square_free_factorization(exact)) {
441 auto step = context.consume_steps(1, "solve.exact");
442 if (!step) { return SolveResult::failure(step.error()); }
443 auto solved = solve_exact_factor(factor.first, var, context);
444 if (!solved) { return SolveResult::failure(solved.error()); }
445 const auto& factor_roots = solved.value();
446 if (factor_roots.size() != static_cast<std::size_t>(factor.first.degree()))
447 { return SolveResult::failure(CasErrc::InternalInvariant,
448 "square-free factor root count mismatch", "solve.exact"); }
449 for (int copy = 0; copy < factor.second; ++copy)
450 { roots.insert(roots.end(), factor_roots.begin(), factor_roots.end()); }
451 }
452 if (roots.size() != static_cast<std::size_t>(exact.degree()))
453 { return SolveResult::failure(CasErrc::InternalInvariant,
454 "polynomial root multiplicities do not cover its degree", "solve.exact"); }
455 return collect_exact_solutions(std::move(roots), context);
456}

458static SolveResult dispatch_equation(const ExprPtr& expr, const ExprPtr& f_expr,
459 const std::string& var, ComputationContext& context, const SolveOptions& opts) {
460 auto polynomial = recognize_rational_polynomial(*f_expr, var, context);
461 if (!polynomial) { return SolveResult::failure(polynomial.error()); }
462 if (polynomial.value()) {
463 auto solved = solve_exact_polynomial_set(*polynomial.value(), var, context);
464 if (!solved) { return solved; }
465 return detail::finalize_solution_set(expr, var, std::move(solved.value()), context, Domain::Complex, opts);
466 }
467 auto symbolic = symbolic_to_poly<SymbolicPolyCoeff>(f_expr, var);
468 if (symbolic) {
469 auto solved = solve_symbolic_polynomial_set(symbolic.value(), f_expr, var, context, opts);
470 if (!solved) { return solved; }
471 return detail::finalize_solution_set(expr, var, std::move(solved.value()), context, Domain::Complex, opts);
472 }
473 if (symbolic.error().code != CasErrc::UnsupportedExpression) {
474 return SolveResult::failure(symbolic.error());
475 }
476 if (auto product = solve_product_set(f_expr, var, context, opts)) {
477 if (!*product) { return std::move(*product); }
478 return detail::finalize_solution_set(expr, var, std::move(product->value()), context, Domain::Real, opts);
479 }
480 auto solved = solve_set_core(f_expr, var, context, opts);
481 if (!solved) { return solved; }
482 return detail::finalize_solution_set(expr, var, std::move(solved.value()), context, Domain::Real, opts);
483}

485LMCAS::SolveResult solve_equation(
486 const std::shared_ptr<SymbolicExpr>& expr,
487 const std::string& var,
488 ComputationContext& context,
489 const SolveOptions& opts) {
490 constexpr const char* operation = "solve_equation";
491 if (!expr || !LMCAS::detail::node(expr)) {
492 return SolveResult::failure(
493 CasErrc::InvalidArgument, "equation expression cannot be null", operation);
494 }
495 if (var.empty()) {
496 return SolveResult::failure(
497 CasErrc::InvalidArgument, "solve variable cannot be empty", operation);
498 }
499
500 auto step = context.consume_steps(1, operation);
501 if (!step) {
502 return SolveResult::failure(step.error());
503 }
504
505 std::shared_ptr<SymbolicExpr> f_expr = expr;
506 if (auto relation = std::dynamic_pointer_cast<const RelationalNode>(LMCAS::detail::node(expr))) {
507 if (relation->op() != RelationalNode::Op::EQ) {
508 return SolveResult::failure(
509 CasErrc::InvalidArgument,
510 "solve_equation accepts equations, not inequalities",
511 operation);
512 }
513 auto left = LMCAS::detail::make_expression_ptr(relation->left());
514 auto right = LMCAS::detail::make_expression_ptr(relation->right());
515 f_expr = SymbolicExpr::add(
516 left, SymbolicExpr::multiply(SymbolicExpr::number(-1), right));
517 }
518
519 return dispatch_equation(expr, f_expr, var, context, opts);
520}

531static std::optional<SolveVectorResult> solve_finite_mixed(
532 const std::shared_ptr<SymbolicExpr>& f_expr, const std::string& var,
533 ComputationContext& context, const SolveOptions& opts) {
534 constexpr const char* operation = "solve_dispatch_vector";
535 if (!contains_transcendental_of_var(f_expr, var)) {
536 return std::nullopt;
537 }
538 auto substitution = detect_trans_substitutions(f_expr, var);
539 if (!substitution.mappings.empty() && is_polynomial_after_substitution(substitution)) {
540 return std::nullopt;
541 }
542 if (!opts.allow_numeric) {
543 return SolveVectorResult::failure(
544 CasErrc::Inconclusive,
545 "mixed transcendental equation is outside the finite-vector support domain",
546 operation);
547 }
548 auto mixed = solve_mixed_transcendental_checked(f_expr, var, opts, context);
549 if (!mixed) {
550 return SolveVectorResult::failure(mixed.error());
551 }
552 if (mixed.value().completeness == Completeness::Complete) {
553 return SolveVectorResult::success(std::move(mixed.value().value));
554 }
555 return SolveVectorResult::failure(
556 CasErrc::Inconclusive,
557 mixed.value().reason.empty() ? "mixed transcendental search is incomplete" : mixed.value().reason,
558 operation);
559}

596 }

598 std::shared_ptr<SymbolicExpr> f_expr = simplified;
599 if (auto rel = std::dynamic_pointer_cast<const RelationalNode>(LMCAS::detail::node(simplified))) {
600 if (rel->op() == RelationalNode::Op::EQ) {
601 auto left = LMCAS::detail::make_expression_ptr(rel->left());
602 auto right = LMCAS::detail::make_expression_ptr(rel->right());
603 f_expr = SymbolicExpr::add(
604 left, SymbolicExpr::multiply(right, SymbolicExpr::number(-1)));
605 }
606 }
607
608 auto polynomial = symbolic_to_poly<SymbolicPolyCoeff>(f_expr, var);
609 if (polynomial) {
610 return solve_symbolic_polynomial_set(polynomial.value(), expr, var, context, opts);
611 }
612 if (polynomial.error().code != CasErrc::UnsupportedExpression) {
613 return SolveResult::failure(polynomial.error());
614 }
615
616 auto transcendental = solve_transcendental(f_expr, var, context, opts);
617 if (transcendental) return transcendental;
618 if (transcendental.error().code != CasErrc::Inconclusive) {
619 return SolveResult::failure(transcendental.error());
620 }
621
622 if (auto mixed = solve_finite_mixed(f_expr, var, context, opts)) {
623 return finite_values_to_set(std::move(*mixed));
624 }
625
626
627 return SolveResult::failure(CasErrc::Inconclusive,
628 "expression is outside the supported finite exact solve domain", operation);
629}

631LMCAS::FiniteSolveResult solve_finite_checked(
632 const std::shared_ptr<SymbolicExpr>& expr,
633 const std::string& var,
634 ComputationContext& context,
635 const SolveOptions& opts) {
636 constexpr const char* operation = "solve_finite_projection";
637 auto solved = solve_equation(expr, var, context, opts);
638 if (!solved) {
639 return SolveVectorResult::failure(solved.error());
640 }
641 auto solution_set = std::move(solved.value());
642 if (std::holds_alternative<EmptySolutions>(solution_set)) {
643 return SolveVectorResult::success({});
644 }
645 auto* finite = std::get_if<FiniteSolutions>(&solution_set);
646 if (!finite) {
647 return SolveVectorResult::failure(
648 CasErrc::Inconclusive,
649 "solution set is not finitely enumerable", operation);
650 }
651 std::size_t value_count = 0;
652 for (const auto& solution : finite->values) {
653 if (!solution.conditions.empty()) {
654 return SolveVectorResult::failure(CasErrc::Inconclusive,
655 "Conditional roots cannot be projected to an unconditional vector", operation);
656 }
657 value_count += solution.multiplicity;
658 }
659 std::vector<std::shared_ptr<SymbolicExpr>> values;
660 values.reserve(value_count);
661 for (const auto& solution : finite->values) {
662 for (std::size_t copy = 0; copy < solution.multiplicity; ++copy) {
663 values.push_back(solution.value);
664 }
665 }
666 return SolveVectorResult::success(std::move(values));
667}

681SymbolicExpr::solve_system(
682 const std::vector<std::shared_ptr<SymbolicExpr>>& equations,
683 const std::vector<std::string>& vars) {
684 std::vector<SymbolicExpr> values;
685 values.reserve(equations.size());
686 for (const auto& equation : equations) {
687 if (!equation) {
688 return {};
689 }
690 if (auto relation =
691 std::dynamic_pointer_cast<const RelationalNode>(
692 LMCAS::detail::node(equation))) {
693 if (relation->op() != RelationalNode::Op::EQ) {
694 return {};
695 }
696 auto normalized = SymbolicExpr::add(
697 LMCAS::detail::make_expression_ptr(relation->left()),
698 SymbolicExpr::multiply(
699 SymbolicExpr::number(-1),
700 LMCAS::detail::make_expression_ptr(relation->right())));
701 values.push_back(*normalized->simplify());
702 } else {
703 values.push_back(*equation);
704 }
705 }
706 auto solved = LMCAS::Solver::solve_linear_system(values, vars);
707 if (solved.empty()) {
708 return {};
709 }
710 std::map<std::string, std::shared_ptr<SymbolicExpr>> projected;
711 for (auto& [variable, value] : solved) {
712 projected.emplace(
713 variable, std::make_shared<SymbolicExpr>(std::move(value)));
714 }
715 return {std::move(projected)};
716}

727 if (!mat) {
728 return mat;
729 }
730 auto m_node = std::dynamic_pointer_cast<const MatrixNode>(LMCAS::detail::node(mat));
731 if (!m_node) {
732 return mat;
733 }
734 size_t r = m_node->rows();
735 size_t c = m_node->cols();
736
737 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> new_data(c, std::vector<std::shared_ptr<SymbolicExpr>>(r));
738
739 for(size_t i=0; i<r; ++i) {
740 for(size_t j=0; j<c; ++j) {
741 new_data[j][i] = LMCAS::detail::make_expression_ptr(m_node->get(i,j));
742 }
743 }
744
745 return SymbolicExpr::matrix(new_data);
