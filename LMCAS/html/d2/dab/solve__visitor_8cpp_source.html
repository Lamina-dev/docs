25static bool solve_is_minus_one(const std::shared_ptr<const NumberNode>& number) {
26 if (!number) return false;
27 if (std::holds_alternative<BigInt>(number->value())) {
28 return std::get<BigInt>(number->value()) == BigInt(-1);

36static bool solve_contains_rootof_node(const std::shared_ptr<const SymbolicNode>& node) {
37 if (!node) return false;
38 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(node)) {
39 if (function->type() == FunctionNode::FuncType::RootOf) return true;
40 for (const auto& arg : function->arguments()) {
41 if (solve_contains_rootof_node(arg)) return true;
42 }
43 return false;
44 }
45 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
46 for (const auto& operand : add->operands()) {
47 if (solve_contains_rootof_node(operand)) return true;
48 }
49 return false;
50 }
51 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
52 for (const auto& operand : multiply->operands()) {
53 if (solve_contains_rootof_node(operand)) return true;
54 }
55 return false;
56 }
57 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
58 return solve_contains_rootof_node(power->base()) ||
59 solve_contains_rootof_node(power->exponent());
60 }
61 if (auto relation = std::dynamic_pointer_cast<const RelationalNode>(node)) {
62 return solve_contains_rootof_node(relation->left()) ||
63 solve_contains_rootof_node(relation->right());
64 }
65 return false;
66}

83static std::shared_ptr<const SymbolicNode> solve_simplify_with_nonzero_pivots(
84 const std::shared_ptr<const SymbolicNode>& node) {
85 if (!node) return node;
86
87 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
88 std::vector<std::shared_ptr<const SymbolicNode>> operands;
89 operands.reserve(add->operands().size());
90 for (const auto& operand : add->operands()) {
91 operands.push_back(solve_simplify_with_nonzero_pivots(operand));
92 }
93 return lamina::detail::make_node<AddNode>(std::move(operands));
94 }
95
96 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
97 return lamina::detail::make_node<PowerNode>(
98 solve_simplify_with_nonzero_pivots(power->base()),
99 solve_simplify_with_nonzero_pivots(power->exponent()));
100 }
101
102 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
103 std::vector<std::shared_ptr<const SymbolicNode>> operands;
104 operands.reserve(multiply->operands().size());
105 for (const auto& operand : multiply->operands()) {
106 auto simplified = solve_simplify_with_nonzero_pivots(operand);
107 if (auto nested = std::dynamic_pointer_cast<const MultiplyNode>(simplified)) {
108 operands.insert(operands.end(), nested->operands().begin(), nested->operands().end());
109 } else {
110 operands.push_back(simplified);
111 }
112 }
113
114 std::vector<bool> consumed(operands.size(), false);
115 for (size_t i = 0; i < operands.size(); ++i) {
116 if (consumed[i]) continue;
117 auto reciprocal = std::dynamic_pointer_cast<const PowerNode>(operands[i]);
118 auto exponent = reciprocal ? std::dynamic_pointer_cast<const NumberNode>(reciprocal->exponent()) : nullptr;
119 if (!reciprocal || !solve_is_minus_one(exponent)) continue;
120
121 for (size_t j = 0; j < operands.size(); ++j) {
122 if (i == j || consumed[j]) continue;
123 if (reciprocal->base()->compare(*operands[j]) == 0) {
124 consumed[i] = true;
125 consumed[j] = true;
126 break;
127 }
128 }
129 }
130
131 std::vector<std::shared_ptr<const SymbolicNode>> remaining;
132 for (size_t i = 0; i < operands.size(); ++i) {
133 if (!consumed[i]) remaining.push_back(operands[i]);
134 }
135 if (remaining.empty()) return lamina::detail::make_node<NumberNode>(BigInt(1));
136 if (remaining.size() == 1) return remaining.front();
137 return lamina::detail::make_node<MultiplyNode>(std::move(remaining));
138 }
139
140 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(node)) {
141 std::vector<std::shared_ptr<const SymbolicNode>> args;
142 args.reserve(function->arguments().size());
143 for (const auto& arg : function->arguments()) {
144 args.push_back(solve_simplify_with_nonzero_pivots(arg));
145 }
146 return lamina::detail::make_node<FunctionNode>(function->type(), std::move(args));
147 }
148
149 return node;
150}

159static std::vector<std::shared_ptr<SymbolicExpr>> solve_polynomial_closed(
160 const Polynomial<SymbolicPolyCoeff>& poly,
161 const std::string& var) {
162
163 int deg = poly.degree();
164 if (deg <= 0) return {};
165
166 if (deg == 1) {
167 auto a = get_coeff(poly, 1);
168 auto b = get_coeff(poly, 0);
169 auto neg_b = SymbolicExpr::multiply(b, SymbolicExpr::number(-1));
170 auto result = SymbolicExpr::divide(neg_b, a);
171 return { result->simplify() };
172 }
173
174 if (deg == 2) {
175 auto a = get_coeff(poly, 2);
176 auto b = get_coeff(poly, 1);
177 auto c = get_coeff(poly, 0);
178
179 auto b2 = SymbolicExpr::power(b, SymbolicExpr::number(2));
180 auto ac4 = SymbolicExpr::multiply(SymbolicExpr::number(4), SymbolicExpr::multiply(a, c));
181 auto delta = SymbolicExpr::add(b2, SymbolicExpr::multiply(ac4, SymbolicExpr::number(-1)));
182
183 auto sqrt_delta = SymbolicExpr::power(delta, SymbolicExpr::number(0.5));
184 auto neg_b = SymbolicExpr::multiply(b, SymbolicExpr::number(-1));
185 auto two_a = SymbolicExpr::multiply(SymbolicExpr::number(2), a);
186
187 auto numerator1 = SymbolicExpr::add(neg_b, sqrt_delta);
188 auto numerator2 = SymbolicExpr::add(neg_b, SymbolicExpr::multiply(sqrt_delta, SymbolicExpr::number(-1)));
189
190 auto x1 = SymbolicExpr::divide(numerator1, two_a);
191 auto x2 = SymbolicExpr::divide(numerator2, two_a);
192
193 return { x1->simplify(), x2->simplify() };
194 }
195
196 if (deg == 3) {
197 auto a = get_coeff(poly, 3);
198 auto b = get_coeff(poly, 2);
199 auto c = get_coeff(poly, 1);
200 auto d = get_coeff(poly, 0);
201 return solve_cubic(a, b, c, d, var);
202 }
203
204 if (deg == 4) {
205 auto a = get_coeff(poly, 4);
206 auto b = get_coeff(poly, 3);
207 auto c = get_coeff(poly, 2);
208 auto d = get_coeff(poly, 1);
209 auto e = get_coeff(poly, 0);
210 return solve_quartic(a, b, c, d, e, var);
211 }
212
213 return {};
214}

216lamina::SolveResult lamina::solve_equation(
217 const std::shared_ptr<SymbolicExpr>& expr,
218 const std::string& var,
219 ComputationContext& context,
220 const SolveOptions& opts) {
221 constexpr const char* operation = "solve_dispatch";
222 if (!expr || !lamina::detail::node(expr)) {
223 return SolveResult::failure(
224 CasErrc::InvalidArgument, "equation expression cannot be null", operation);
225 }
226 if (var.empty()) {
227 return SolveResult::failure(
228 CasErrc::InvalidArgument, "solve variable cannot be empty", operation);
229 }
230
231 auto step = context.consume_steps(1, operation);
232 if (!step) return SolveResult::failure(step.error());
233
234 std::shared_ptr<SymbolicExpr> f_expr = expr;
235 if (auto relation = std::dynamic_pointer_cast<const RelationalNode>(lamina::detail::node(expr))) {
236 if (relation->op() != RelationalNode::Op::EQ) {
237 return SolveResult::failure(
238 CasErrc::InvalidArgument,
239 "solve_equation accepts equations, not inequalities",
240 operation);
241 }
242 auto left = lamina::detail::make_expression_ptr(relation->left());
243 auto right = lamina::detail::make_expression_ptr(relation->right());
244 f_expr = SymbolicExpr::add(
245 left, SymbolicExpr::multiply(SymbolicExpr::number(-1), right));
246 }
247
248 auto polynomial = recognize_rational_polynomial(*f_expr, var, context);
249 if (!polynomial) return SolveResult::failure(polynomial.error());
250 if (polynomial.value()) {
251 const auto& exact = *polynomial.value();
252 if (exact.is_zero()) {
253 return SolveResult::success(SolutionSet::universal());
254 }
255 if (exact.degree() == 0) {
256 return SolveResult::success(SolutionSet::empty());
257 }
258
259 std::vector<FiniteSolution> solutions;
260 solutions.reserve(static_cast<std::size_t>(exact.degree()));
261 for (int index = 0; index < exact.degree(); ++index) {
262 auto root_step = context.consume_steps(1, operation);
263 if (!root_step) return SolveResult::failure(root_step.error());
264 solutions.push_back(FiniteSolution{
265 SymbolicExpr::root_of(f_expr, var, index), 1, {}});
266 }
267 return SolveResult::success(SolutionSet::finite(std::move(solutions)));
268 }
269
270 if (!opts.allow_numeric) {
271 return SolveResult::success(SolutionSet::inconclusive(
272 "expression is outside the exact rational-polynomial support domain"));
273 }
274
275 auto numeric = solve_numeric_checked(f_expr, var, context, opts);
276 if (!numeric) return SolveResult::failure(numeric.error());
277 if (numeric.value().empty()) {
278 return SolveResult::success(SolutionSet::inconclusive(
279 "numeric iteration produced no verified candidate"));
280 }
281
282 std::vector<FiniteSolution> solutions;
283 solutions.reserve(numeric.value().size());
284 for (const auto& root : numeric.value()) {
285 solutions.push_back(FiniteSolution{
286 SymbolicExpr::number(root.value), 1, {}});
287 }
288 return SolveResult::success(SolutionSet::finite(std::move(solutions)));
289}

299lamina::SolveVectorResult lamina::solve_dispatch_vector_checked(
300 const std::shared_ptr<SymbolicExpr>& expr,
301 const std::string& var,
302 ComputationContext& context,
303 const SolveOptions& opts) {
304 constexpr const char* operation = "solve_dispatch_vector";
305 if (!expr || !lamina::detail::node(expr)) {
306 return SolveVectorResult::failure(CasErrc::InvalidArgument,
307 "expression cannot be null",
308 operation);
309 }
310 if (var.empty()) {
311 return SolveVectorResult::failure(CasErrc::InvalidArgument,
312 "solve variable cannot be empty",
313 operation);
314 }
315
316 auto simplified = expr->simplify();
317 if (!simplified || !lamina::detail::node(simplified)) {
318 return SolveVectorResult::failure(CasErrc::InvalidArgument,
319 "expression simplification failed",
320 operation);
321 }
322
323 std::shared_ptr<SymbolicExpr> f_expr = simplified;
324 if (auto rel = std::dynamic_pointer_cast<const RelationalNode>(lamina::detail::node(simplified))) {
325 if (rel->op() == RelationalNode::Op::EQ) {
326 auto left = lamina::detail::make_expression_ptr(rel->left());
327 auto right = lamina::detail::make_expression_ptr(rel->right());
328 f_expr = SymbolicExpr::add(
329 left, SymbolicExpr::multiply(right, SymbolicExpr::number(-1)));
330 }
331 }
332
333 auto polynomial = symbolic_to_poly<SymbolicPolyCoeff>(f_expr, var);
334 if (!polynomial.is_zero() && polynomial.degree() >= 1) {
335 if (polynomial.degree() <= 4) {
336 auto results = solve_polynomial_closed(polynomial, var);
337 results = solve_filter_rootof_results(std::move(results), opts);
338 if (!results.empty()) return SolveVectorResult::success(std::move(results));
339 }
340
341 if (polynomial.degree() > 4) {
342 auto results = solve_by_factoring(polynomial, var);
343 results = solve_filter_rootof_results(std::move(results), opts);
344 if (!results.empty()) return SolveVectorResult::success(std::move(results));
345
346 if (opts.return_rootof) {
347 return SolveVectorResult::success(make_rootof_solutions(polynomial, var));
348 }
349 }
350 }
351
352 auto transcendental_results = solve_transcendental(f_expr, var);
353 if (!transcendental_results.empty()) {
354 return SolveVectorResult::success(std::move(transcendental_results));
355 }
356
357 if (contains_transcendental_of_var(f_expr, var)) {
358 auto substitution = detect_trans_substitutions(f_expr, var);
359 if (substitution.mappings.empty() || !is_polynomial_after_substitution(substitution)) {
360 if (opts.allow_numeric) {
361 auto mixed_results = solve_mixed_transcendental(f_expr, var, opts);
362 if (!mixed_results.empty()) {
363 return SolveVectorResult::success(std::move(mixed_results));
364 }
365 return SolveVectorResult::failure(
366 CasErrc::Inconclusive,
367 "mixed transcendental numeric search produced no verified vector result",
368 operation);
369 }
370 return SolveVectorResult::failure(
371 CasErrc::Inconclusive,
372 "mixed transcendental equation is outside the finite-vector support domain",
373 operation);
374 }
375 }
376
377 if (opts.allow_numeric) {
378 auto numeric_roots = solve_numeric_checked(f_expr, var, context, opts);
379 if (!numeric_roots) return SolveVectorResult::failure(numeric_roots.error());
380 if (!numeric_roots.value().empty()) {
381 std::vector<std::shared_ptr<SymbolicExpr>> results;
382 results.reserve(numeric_roots.value().size());
383 for (const auto& root : numeric_roots.value()) {
384 results.push_back(SymbolicExpr::number(root.value));
385 }
386 return SolveVectorResult::success(std::move(results));
387 }
388 }
389
390 return SolveVectorResult::success({});
391}

411std::vector<std::shared_ptr<SymbolicExpr>> SymbolicExpr::solve(std::shared_ptr<SymbolicExpr> eq, const std::string& var_name) {
412 if (!eq) return {};
413 auto simplified_eq = eq->simplify();
414
415 if (auto rel = std::dynamic_pointer_cast<const RelationalNode>(lamina::detail::node(simplified_eq))) {
416 if (rel->op() != RelationalNode::Op::EQ) {
417 auto L = lamina::detail::make_expression_ptr(rel->left());
418 auto R = lamina::detail::make_expression_ptr(rel->right());
419 auto diff = SymbolicExpr::add(L, SymbolicExpr::multiply(R, SymbolicExpr::number(-1)));
420
421 auto poly = symbolic_to_poly<SymbolicPolyCoeff>(diff, var_name);
422
423 if (!poly.is_zero()) {
424 int max_deg = poly.degree();
425 if (max_deg == 1) {
426 auto a = get_coeff(poly, 1);
427 auto b = get_coeff(poly, 0);
428
429 auto neg_b = SymbolicExpr::multiply(b, SymbolicExpr::number(-1));
430 auto a_inv = SymbolicExpr::power(a, SymbolicExpr::number(-1));
431
432 bool flip = false;
433
434 try {
435 auto a_simp = a->simplify();
436 if (auto num_a = std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(a_simp))) {
437 if (std::holds_alternative<double>(num_a->value())) {
438 if (std::get<double>(num_a->value()) < 0) flip = true;
439 } else if (std::holds_alternative<BigInt>(num_a->value())) {
440 if (std::get<BigInt>(num_a->value()).IsNegative()) flip = true;
441 } else if (std::holds_alternative<Rational>(num_a->value())) {
442 if (std::get<Rational>(num_a->value()).get_numerator().IsNegative()) flip = true;
443 }
444 }
445 } catch(...) {}
446
447 RelationalNode::Op new_op = rel->op();
448 if (flip) {
449 switch(rel->op()) {
450 case RelationalNode::Op::LT: new_op = RelationalNode::Op::GT; break;
451 case RelationalNode::Op::GT: new_op = RelationalNode::Op::LT; break;
452 case RelationalNode::Op::LEQ: new_op = RelationalNode::Op::GEQ; break;
453 case RelationalNode::Op::GEQ: new_op = RelationalNode::Op::LEQ; break;
454 default: break;
455 }
456 }
457
458 auto rhs = SymbolicExpr::multiply(neg_b, a_inv)->simplify();
459 auto var_node = lamina::detail::make_node<VariableNode>(var_name);
460 auto res_node = lamina::detail::make_node<RelationalNode>(var_node, lamina::detail::node(rhs), new_op);
461
462 return { lamina::detail::make_expression_ptr(res_node) };
463 }
464 }
465 return {};
466 }
467 }
468
469 return lamina::solve_dispatch(simplified_eq, var_name, lamina::SolveOptions{});
470}

518std::shared_ptr<SymbolicExpr> SymbolicExpr::determinant(const std::shared_ptr<SymbolicExpr>& mat) {
519 if (!mat) return SymbolicExpr::number(0);
520 auto mat_node = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(mat));
521 if (!mat_node || mat_node->rows() != mat_node->cols()) return SymbolicExpr::number(0);
522
523 size_t n = mat_node->rows();
524 if (n == 1) return lamina::detail::make_expression_ptr(mat_node->get(0,0));
525 if (n == 2) {
526 auto a = lamina::detail::make_expression_ptr(mat_node->get(0,0));
527 auto b = lamina::detail::make_expression_ptr(mat_node->get(0,1));
528 auto c = lamina::detail::make_expression_ptr(mat_node->get(1,0));
529 auto d = lamina::detail::make_expression_ptr(mat_node->get(1,1));
530 return SymbolicExpr::add(SymbolicExpr::multiply(a,d), SymbolicExpr::multiply(SymbolicExpr::multiply(b,c), SymbolicExpr::number(-1)))->simplify();
531 }
532
533 if (n > 3) {
534 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> A(n, std::vector<std::shared_ptr<SymbolicExpr>>(n));
535 for(size_t i=0; i<n; ++i) {
536 for(size_t j=0; j<n; ++j) {
537 A[i][j] = lamina::detail::make_expression_ptr(mat_node->get(i,j));
538 }
539 }
540
541 int sign = 1;
542 auto det = SymbolicExpr::number(1);
543
544 for (size_t col = 0; col < n; ++col) {
545 size_t pivot_row = col;
546 while (pivot_row < n && A[pivot_row][col]->is_zero()) {
547 pivot_row++;
548 }
549 if (pivot_row == n) return SymbolicExpr::number(0);
550
551 if (pivot_row != col) {
552 std::swap(A[col], A[pivot_row]);
553 sign = -sign;
554 }
555
556 auto pivot = A[col][col];
557 det = SymbolicExpr::multiply(det, pivot)->simplify();
558 auto pivot_inv = SymbolicExpr::power(pivot, SymbolicExpr::number(-1));
559
560 for (size_t r = col + 1; r < n; ++r) {
561 auto factor = A[r][col];
562 if (!factor->is_zero()) {
563 auto mult = SymbolicExpr::multiply(factor, pivot_inv)->simplify();
564 auto neg_mult = SymbolicExpr::multiply(mult, SymbolicExpr::number(-1));
565 for (size_t c = col + 1; c < n; ++c) {
566 auto term = SymbolicExpr::multiply(neg_mult, A[col][c]);
567 A[r][c] = SymbolicExpr::add(A[r][c], term)->simplify();
568 }
569 A[r][col] = SymbolicExpr::number(0);
570 }
571 }
572 }
573
574 if (sign == -1) {
575 det = SymbolicExpr::multiply(det, SymbolicExpr::number(-1))->simplify();
576 }
577 return det;
578 }
579
580 std::vector<std::shared_ptr<SymbolicExpr>> terms;
581 for(size_t c=0; c<n; ++c) {
582 auto elem = lamina::detail::make_expression_ptr(mat_node->get(0,c));
583 if (elem->is_zero()) continue;
584
585 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> minor_data;
586 for(size_t r=1; r<n; ++r) {
587 std::vector<std::shared_ptr<SymbolicExpr>> row;
588 for(size_t k=0; k<n; ++k) {
589 if (k == c) continue;
590 row.push_back(lamina::detail::make_expression_ptr(mat_node->get(r,k)));
591 }
592 minor_data.push_back(row);
593 }
594 auto minor_mat = SymbolicExpr::matrix(minor_data);
595 auto minor_det = SymbolicExpr::determinant(minor_mat);
596
597 auto term = SymbolicExpr::multiply(elem, minor_det);
598 if (c % 2 == 1) term = SymbolicExpr::multiply(term, SymbolicExpr::number(-1));
599 terms.push_back(term);
600 }
601
602 if (terms.empty()) return SymbolicExpr::number(0);
603 auto result = terms[0];
604 for(size_t k=1; k<terms.size(); ++k) result = SymbolicExpr::add(result, terms[k]);
605 return result->simplify();
606}

655std::vector<std::pair<std::shared_ptr<SymbolicExpr>, std::vector<std::shared_ptr<SymbolicExpr>>>> SymbolicExpr::eigenvectors(const std::shared_ptr<SymbolicExpr>& mat) {
656 auto evals_expr = eigenvalues(mat);
657
658 std::vector<std::pair<std::shared_ptr<SymbolicExpr>, std::vector<std::shared_ptr<SymbolicExpr>>>> result;
659
660 auto mat_node = evals_expr ? std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(evals_expr)) : nullptr;
661 if (!mat_node) {
662 return {};
663 }
664
665 size_t num_evals = mat_node->cols();
666
667 auto A_node = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(mat));
668 if (!A_node) return {};
669 size_t n = A_node->rows();
670
671 for(size_t i=0; i<num_evals; ++i) {
672 auto lambda_node = mat_node->get(0, i);
673 auto lambda = lamina::detail::make_expression_ptr(lambda_node);
674
675 std::vector<std::shared_ptr<SymbolicExpr>> equations;
676 std::vector<std::string> vars;
677 for(size_t k=0; k<n; ++k) vars.push_back("v" + std::to_string(k));
678
679 for(size_t i=0; i<n; ++i) {
680 std::vector<std::shared_ptr<SymbolicExpr>> terms;
681 for(size_t j=0; j<n; ++j) {
682 auto a_ij = lamina::detail::make_expression_ptr(A_node->get(i,j));
683 std::shared_ptr<SymbolicExpr> coeff;
684 if (i == j) {
685 coeff = SymbolicExpr::add(a_ij, SymbolicExpr::multiply(lambda, SymbolicExpr::number(-1)));
686 } else {
687 coeff = a_ij;
688 }
689
690 auto var = SymbolicExpr::variable(vars[j]);
691 terms.push_back(SymbolicExpr::multiply(coeff, var));
692 }
693 auto row_eq = terms[0];
694 for(size_t k=1; k<terms.size(); ++k) row_eq = SymbolicExpr::add(row_eq, terms[k]);
695 equations.push_back(row_eq->simplify());
696 }
697
698 auto sols = solve_system(equations, vars);
699
700 std::vector<std::shared_ptr<SymbolicExpr>> eigenvec;
701 bool sols_usable = !sols.empty();
702 if (sols_usable) {
703 for(const auto& v : vars) {
704 auto it = sols[0].find(v);
705 if (it == sols[0].end()) { sols_usable = false; break; }
706 eigenvec.push_back(it->second);
707 }
708 }
709
710 bool is_non_zero = false;
711 if (sols_usable) {
712 for(auto& x : eigenvec) if(!x->is_zero()) is_non_zero = true;
713 }
714
715 if (sols_usable && is_non_zero) {
716 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> col_vec_data;
717 for(auto& val : eigenvec) col_vec_data.push_back({val});
718 result.push_back({lambda, {SymbolicExpr::matrix(col_vec_data)}});
719 } else {
720 if (n > 1) {
721 bool found_vec = false;
722 for(int free_var_idx = n-1; free_var_idx >= 0 && !found_vec; --free_var_idx) {
723 std::vector<std::shared_ptr<SymbolicExpr>> sub_eqs;
724 std::vector<std::string> sub_vars;
725
726 for(size_t k=0; k<n; ++k) {
727 auto eq_sub = equations[k]->substitute(vars[free_var_idx], SymbolicExpr::number(1))->simplify();
728 if (!eq_sub->is_zero()) {
729 sub_eqs.push_back(eq_sub);
730 }
731 }
732
733 for(size_t k=0; k<n; ++k) {
734 if (k != (size_t)free_var_idx) sub_vars.push_back(vars[k]);
735 }
736
737 auto sub_sols = solve_system(sub_eqs, sub_vars);
738 if (!sub_sols.empty()) {
739 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> col_vec_data(n);
740 bool ok = true;
741 for(size_t k=0; k<n; ++k) {
742 if (k == (size_t)free_var_idx) { col_vec_data[k] = {SymbolicExpr::number(1)}; continue; }
743 auto it = sub_sols[0].find(vars[k]);
744 if (it == sub_sols[0].end()) { ok = false; break; }
745 col_vec_data[k] = {it->second};
746 }
747 if (ok) {
748 auto vec_expr = SymbolicExpr::matrix(col_vec_data);
749 result.push_back({lambda, {vec_expr}});
750 found_vec = true;
751 }
752 }
753 }
754 }
755 }
756 }
757
758 return result;
759}

779std::shared_ptr<SymbolicExpr> SymbolicExpr::rref(const std::shared_ptr<SymbolicExpr>& mat) {
780 if (!mat) return mat;
781 auto m_node = std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(mat));
782 if (!m_node) return mat;
783 size_t rows = m_node->rows();
784 size_t cols = m_node->cols();
785
786 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> data(rows, std::vector<std::shared_ptr<SymbolicExpr>>(cols));
787 for(size_t i=0; i<rows; ++i) {
788 for(size_t j=0; j<cols; ++j) {
789 data[i][j] = lamina::detail::make_expression_ptr(m_node->get(i,j));
790 }
791 }
792
793 size_t lead = 0;
794 for (size_t r = 0; r < rows; ++r) {
795 if (cols <= lead) break;
796
797 size_t i = r;
798 while (true) {
799 if (i >= rows) {
800 i = r;
801 lead++;
802 if (cols == lead) goto end_loops;
803 continue;
804 }
805 auto val = data[i][lead]->simplify();
806 if (!val->get_number_value_is_zero()) {
807 break;
808 }
809 i++;
810 }
811
812 std::swap(data[i], data[r]);
813
814 auto pivot = data[r][lead];
815 auto pivot_inv = SymbolicExpr::power(pivot, SymbolicExpr::number(-1));
816
817 for (size_t j = 0; j < cols; ++j) {
818 data[r][j] = SymbolicExpr::multiply(data[r][j], pivot_inv)->simplify();
819 }
820
821 for (size_t k = 0; k < rows; ++k) {
822 if (k != r) {
823 auto factor = data[k][lead];
824 if (factor->simplify()->get_number_value_is_zero()) continue;
825
826 auto neg_factor = SymbolicExpr::multiply(factor, SymbolicExpr::number(-1));
827 for (size_t j = 0; j < cols; ++j) {
828 auto term = SymbolicExpr::multiply(neg_factor, data[r][j]);
829 data[k][j] = SymbolicExpr::add(data[k][j], term)->simplify();
830 }
831 }
832 }
833 lead++;
834 }
835 end_loops:;
836
837 return SymbolicExpr::matrix(data);
838}

840std::shared_ptr<SymbolicExpr> SymbolicExpr::inverse(const std::shared_ptr<SymbolicExpr>& mat) {
841 if (!mat) return nullptr;

843 if (!m_node) return nullptr;

845 if (n != m_node->cols()) return nullptr;
846

866 if (!lamina::detail::node(diag)->is_one()) return nullptr;

874 }

876 return SymbolicExpr::matrix(inv_data);
