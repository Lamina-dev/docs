35inline std::optional<int> exact_small_integer_node(
36 const std::shared_ptr<const SymbolicNode>& node,
37 int min_value,
38 int max_value) {
39 auto number = std::dynamic_pointer_cast<const NumberNode>(node);
40 if (!number) return std::nullopt;
41
42 BigInt value;
43 if (std::holds_alternative<BigInt>(number->value())) {
44 value = std::get<BigInt>(number->value());
45 } else if (std::holds_alternative<Rational>(number->value())) {
46 const Rational& rational = std::get<Rational>(number->value());
47 if (!rational.is_integer()) return std::nullopt;
48 value = rational.to_BigInt();
49 } else {
50 return std::nullopt;
51 }
52
53 if (value < BigInt(min_value) || value > BigInt(max_value)) {
54 return std::nullopt;
55 }
56 return value.to_int();
57}
