33static std::shared_ptr<SymbolicExpr> te_inv_power(
34 const std::shared_ptr<SymbolicExpr>& F, const std::string& s,
35 const std::string& t, ComputationContext& context) {
36 if (!F || !LMCAS::detail::node(F)) {
37 return nullptr;
38 }
39 auto tv = SymbolicExpr::variable(t);
40 auto pw = std::dynamic_pointer_cast<const PowerNode>(LMCAS::detail::node(F));
41 if (!pw) {
42 return nullptr;
43 }
44 BigInt exponent;
45 if (!try_get_integer_value(std::dynamic_pointer_cast<const NumberNode>(pw->exponent()), exponent) ||
46 !exponent.is_negative()) {
47 return nullptr;
48 }
49 const BigInt n = -exponent;
50 auto make_exp = [&](const std::shared_ptr<SymbolicExpr>& a) {
51 auto arg = SymbolicExpr::multiply(a, tv);
52 return LMCAS::detail::make_expression_ptr(LMCAS::detail::make_node<FunctionNode>(
53 FunctionNode::FuncType::Exp,
54 std::vector<std::shared_ptr<const SymbolicNode>>{LMCAS::detail::node(arg)}));
55 };
56 auto bv = std::dynamic_pointer_cast<const VariableNode>(pw->base());
57 if (bv && !bv->is_constant() && bv->name() == s) {
58 if (n == BigInt(1)) {
59 return SymbolicExpr::number(1);
60 }
61 return SymbolicExpr::divide(SymbolicExpr::power(tv, SymbolicExpr::number(n - BigInt(1))),
62 SymbolicExpr::number(te_factorial(n - BigInt(1), context)));
63 }
64 auto pole = te_inverse_pole(pw->base(), s);
65 if (pole) {
66 if (n == BigInt(1)) {
67 if (te_is_zero(pole)) {
68 return SymbolicExpr::number(1);
69 }
70 return make_exp(pole);
71 }
72 auto tp = SymbolicExpr::power(tv, SymbolicExpr::number(n - BigInt(1)));
73 auto fv = SymbolicExpr::number(te_factorial(n - BigInt(1), context));
74 if (te_is_zero(pole)) {
75 return SymbolicExpr::divide(tp, fv);
76 }
77 return SymbolicExpr::divide(SymbolicExpr::multiply(tp, make_exp(pole)), fv);
78 }
79 return nullptr;
80}

91static std::shared_ptr<SymbolicExpr> te_inv_product(
92 const std::shared_ptr<SymbolicExpr>& F, const std::string& s,
93 const std::string& t, ComputationContext& context) {
94 if (!F || !LMCAS::detail::node(F)) {
95 return nullptr;
96 }
97 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(LMCAS::detail::node(F));
98 if (!mul || mul->operands().size() < 2) {
99 return nullptr;
100 }
101 std::vector<std::shared_ptr<const SymbolicNode>> num_parts;
102 std::shared_ptr<const SymbolicNode> den_part = nullptr;
103 for (const auto& op : mul->operands()) {
104 if (!den_part && te_is_denominator_factor(op, s)) {
105 den_part = op;
106 continue;
107 }
108 num_parts.push_back(op);
109 }
110 if (!den_part) {
111 return nullptr;
112 }
113 std::shared_ptr<SymbolicExpr> num_expr;
114 if (num_parts.empty()) {
115 num_expr = SymbolicExpr::number(1);
116 } else if (num_parts.size() == 1) {
117 num_expr = LMCAS::detail::make_expression_ptr(num_parts[0]);
118 } else {
119 num_expr = LMCAS::detail::make_expression_ptr(LMCAS::detail::make_node<MultiplyNode>(num_parts));
120 }
121 if (!te_depends_on(num_expr, s)) {
122 auto inv = te_inv_power(LMCAS::detail::make_expression_ptr(den_part), s, t, context);
123 if (inv) {
124 return SymbolicExpr::multiply(num_expr, inv);
125 }
126 }
127 return nullptr;
128}

130static std::shared_ptr<SymbolicExpr> inverse_laplace_core(
131 const std::shared_ptr<SymbolicExpr>& F, const std::string& s,
132 const std::string& t, ComputationContext& context) {
133 auto step = context.consume_steps(1, "inverse_laplace.recursive");
134 if (!step) {
135 return nullptr;
136 }
137 if (!F || !LMCAS::detail::node(F)) {
138 return nullptr;
139 }
140 if (te_is_zero(F)) {
141 return SymbolicExpr::number(0);
142 }
143 auto add = std::dynamic_pointer_cast<const AddNode>(LMCAS::detail::node(F));
144 if (add) {
145 std::shared_ptr<SymbolicExpr> result;
146 for (auto& op : add->operands()) {
147 auto ilt = inverse_laplace_core(
148 LMCAS::detail::make_expression_ptr(op), s, t, context);
149 if (!ilt) {
150 return te_unevaluated_inv_laplace(F, s, t);
151 }
152 result = result ? SymbolicExpr::add(result, ilt) : ilt;
153 }
154 return result;
155 }
156 auto [coeff, body] = te_split_coeff(F, s);
157 if (!coeff->is_one()) {
158 auto ilt_body = inverse_laplace_core(body, s, t, context);
159 if (ilt_body) {
160 return SymbolicExpr::multiply(coeff, ilt_body);
161 }
162 }
163 auto pw_res = te_inv_power(F, s, t, context);
164 if (pw_res) {
165 return pw_res;
166 }
167 auto prod_res = te_inv_product(F, s, t, context);
168 if (prod_res) {
169 return prod_res;
170 }
171 return te_unevaluated_inv_laplace(F, s, t);
172}
