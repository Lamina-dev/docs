46static Result<OdeTaylorCoefficients> ordinary_taylor_coefficients(
47 const std::shared_ptr<SymbolicExpr>& p,
48 const std::shared_ptr<SymbolicExpr>& q,
49 const std::shared_ptr<SymbolicExpr>& x0,
50 const std::string& x, int order) {
51 std::vector<double> p_coeffs(order + 2, 0.0);
52 std::vector<double> q_coeffs(order + 2, 0.0);
53
54 auto p_current = p;
55 auto q_current = q;
56 double factorial = 1.0;
57 for (int k = 0; k <= order + 1; ++k) {
58 if (k > 0) {
59 factorial *= k;
60 }
61 auto p_val_expr = p_current->substitute(x, x0)->simplify();
62 auto q_val_expr = q_current->substitute(x, x0)->simplify();
63 double pv = try_eval_double(p_val_expr);
64 double qv = try_eval_double(q_val_expr);
65 if (!std::isfinite(pv) || !std::isfinite(qv)) {
66 return Result<OdeTaylorCoefficients>::failure(frobenius_unsupported_numeric());
67 }
68 p_coeffs[k] = pv / factorial;
69 q_coeffs[k] = qv / factorial;
70 p_current = p_current->differentiate(x);
71 q_current = q_current->differentiate(x);
72 if (!p_current) {
73 p_current = SymbolicExpr::number(0);
74 }
75 if (!q_current) {
76 q_current = SymbolicExpr::number(0);
77 }
78 }
79
80 return OdeTaylorCoefficients{std::move(p_coeffs), std::move(q_coeffs)};
81}

83static Result<std::vector<double>> ordinary_series_recurrence(
84 const OdeTaylorCoefficients& coefficients, int order) {
85 const auto& p_coeffs = coefficients.p;
86 const auto& q_coeffs = coefficients.q;
87 std::vector<double> a(order + 1, 0.0);
88 a[0] = 1.0;
89 for (int n_idx = 0; n_idx + 2 <= order; ++n_idx) {
90 double sum = 0.0;
91 for (int k = 0; k <= n_idx; ++k) {
92 sum += (k + 1) * a[k + 1] * p_coeffs[n_idx - k];
93 sum += a[k] * q_coeffs[n_idx - k];
94 }
95 double denom = static_cast<double>((n_idx + 2) * (n_idx + 1));
96 a[n_idx + 2] = -sum / denom;
97 if (!std::isfinite(a[n_idx + 2])) {
98 return Result<std::vector<double>>::failure(frobenius_arithmetic_failure());
99 }
100 }
101
102 return a;
103}

105static std::shared_ptr<SymbolicExpr> assemble_frobenius_polynomial(
106 const std::vector<double>& a,
107 const std::shared_ptr<SymbolicExpr>& x_minus_x0) {
108 const int order = static_cast<int>(a.size()) - 1;
109 auto series_sol = SymbolicExpr::number(0);
110 for (int k = 0; k <= order; ++k) {
111 if (std::abs(a[k]) < 1e-15) {
112 continue;
113 }
114 auto coeff = SymbolicExpr::number(a[k]);
115 auto power_term = (k == 0) ? SymbolicExpr::number(1)
116 : SymbolicExpr::power(x_minus_x0, SymbolicExpr::number(k));
117 series_sol = SymbolicExpr::add(series_sol,
118 SymbolicExpr::multiply(coeff, power_term));
119 }
120
121 return series_sol;
122}

124static Result<std::pair<double, double>> frobenius_indicial_roots(
125 double P0, double Q0) {
126 double ind_b = P0 - 1.0;
127 double ind_c = Q0;
128 double ind_D = ind_b * ind_b - 4.0 * ind_c;
129 if (!std::isfinite(ind_D)) {
130 return Result<std::pair<double, double>>::failure(frobenius_arithmetic_failure());
131 }
132
133 double r1, r2;
134 int eq;
135 lmmc_double_nearly_equal_tol(ind_D, 0.0, 1e-12, 1e-12, &eq);
136 if (!eq && ind_D > 0) {
137 r1 = (-ind_b + std::sqrt(ind_D)) / 2.0;
138 r2 = (-ind_b - std::sqrt(ind_D)) / 2.0;
139 } else if (eq) {
140 r1 = -ind_b / 2.0;
141 r2 = r1;
142 } else {
143 r1 = -ind_b / 2.0;
144 r2 = r1;
145 }
146 if (r1 < r2) {
147 std::swap(r1, r2);
148 }
149
150 return std::make_pair(r1, r2);
151}

170static Result<OdeTaylorCoefficients> singular_taylor_coefficients(
171 const std::shared_ptr<SymbolicExpr>& xp_expr,
172 const std::shared_ptr<SymbolicExpr>& x2q_expr,
173 const std::shared_ptr<SymbolicExpr>& x0,
174 const std::string& x, int order, double P0, double Q0,
175 ComputationContext& context) {
176 std::vector<double> pn_coeffs(order + 1, 0.0);
177 std::vector<double> qn_coeffs(order + 1, 0.0);
178
179 auto xp_current = xp_expr;
180 auto x2q_current = x2q_expr;
181 double fact = 1.0;
182 for (int k = 0; k <= order; ++k) {
183 if (k > 0) {
184 fact *= k;
185 }
186 double pv, qv;
187 if (k == 0) {
188 pv = P0;
189 qv = Q0;
190 } else {
191 auto pv_expr = xp_current->substitute(x, x0)->simplify();
192 auto qv_expr = x2q_current->substitute(x, x0)->simplify();
193 pv = try_eval_double(pv_expr);
194 qv = try_eval_double(qv_expr);
195
196 auto p_limit = frobenius_coefficient_limit(xp_current, x, x0, pv, context);
197 if (!p_limit) {
198 return Result<OdeTaylorCoefficients>::failure(p_limit.error());
199 }
200 pv = p_limit.value();
201 auto q_limit = frobenius_coefficient_limit(x2q_current, x, x0, qv, context);
202 if (!q_limit) {
203 return Result<OdeTaylorCoefficients>::failure(q_limit.error());
204 }
205 qv = q_limit.value();
206 }
207 if (!std::isfinite(pv) || !std::isfinite(qv)) {
208 return Result<OdeTaylorCoefficients>::failure(frobenius_unsupported_numeric());
209 }
210 pn_coeffs[k] = pv / fact;
211 qn_coeffs[k] = qv / fact;
212
213 xp_current = xp_current->differentiate(x);
214 x2q_current = x2q_current->differentiate(x);
215 if (!xp_current) {
216 xp_current = SymbolicExpr::number(0);
217 }
218 if (!x2q_current) {
219 x2q_current = SymbolicExpr::number(0);
220 }
221 }
222
223 return OdeTaylorCoefficients{std::move(pn_coeffs), std::move(qn_coeffs)};
224}

226static Result<std::vector<double>> singular_series_recurrence(
227 const OdeTaylorCoefficients& coefficients,
228 double P0, double Q0, double r1, int order) {
229 const auto& pn_coeffs = coefficients.p;
230 const auto& qn_coeffs = coefficients.q;
231 /**
232 * @brief 递推 a_n = -1/F(r1+n) * ∑_{k=0}^{n-1} [(r1+k)·p_{n-k} + q_{n-k}]·a_k。
233 * @note F(s) = s(s-1) + P₀·s + Q₀。
234 */
235 auto indicial_poly = [&](double s) -> double {
236 return s * (s - 1.0) + P0 * s + Q0;
237 };
238
239 std::vector<double> a(order + 1, 0.0);
240 a[0] = 1.0;
241
242 for (int n_idx = 1; n_idx <= order; ++n_idx) {
243 double F_val = indicial_poly(r1 + n_idx);
244 if (!std::isfinite(F_val)) {
245 return Result<std::vector<double>>::failure(frobenius_arithmetic_failure());
246 }
247 if (std::abs(F_val) < 1e-15) {
248 a[n_idx] = 0.0;
249 continue;
250 }
251 double sum = 0.0;
252 for (int k = 0; k < n_idx; ++k) {
253 int idx = n_idx - k;
254 if (idx > order) {
255 continue;
256 }
257 double p_term = (idx < static_cast<int>(pn_coeffs.size())) ? pn_coeffs[idx] : 0.0;
258 double q_term = (idx < static_cast<int>(qn_coeffs.size())) ? qn_coeffs[idx] : 0.0;
259 sum += ((r1 + k) * p_term + q_term) * a[k];
260 }
261 a[n_idx] = -sum / F_val;
262 if (!std::isfinite(a[n_idx])) {
263 return Result<std::vector<double>>::failure(frobenius_arithmetic_failure());
264 }
265 }
266
267 return a;
268}

270static FrobeniusSolutionResult ordinary_frobenius_solution(
271 const std::shared_ptr<SymbolicExpr>& p,
272 const std::shared_ptr<SymbolicExpr>& q,
273 const std::shared_ptr<SymbolicExpr>& x0,
274 const std::string& x, const std::shared_ptr<SymbolicExpr>& x_minus_x0,
275 FrobeniusSolution result) {
276 auto coefficients = ordinary_taylor_coefficients(p, q, x0, x, result.truncation_order);
277 if (!coefficients) {
278 return FrobeniusSolutionResult::failure(coefficients.error());
279 }
280 auto recurrence = ordinary_series_recurrence(coefficients.value(), result.truncation_order);
281 if (!recurrence) {
282 return FrobeniusSolutionResult::failure(recurrence.error());
283 }
284 result.series_solution = assemble_frobenius_polynomial(
285 recurrence.value(), x_minus_x0)->simplify();
286 result.indicial_roots = {0.0};
287 return result;

290static FrobeniusSolutionResult singular_frobenius_solution(
291 const std::shared_ptr<SymbolicExpr>& p,
292 const std::shared_ptr<SymbolicExpr>& q,
293 const std::shared_ptr<SymbolicExpr>& x0,
294 const std::string& x, const std::shared_ptr<SymbolicExpr>& x_minus_x0,
295 FrobeniusSolution result, ComputationContext& context) {
296 auto xp_expr = SymbolicExpr::multiply(x_minus_x0, p)->simplify();
297 auto x2q_expr = SymbolicExpr::multiply(
298 SymbolicExpr::power(x_minus_x0, SymbolicExpr::number(2)), q)->simplify();
299
300 auto P0_result = limit_expression_checked(
301 xp_expr, x, x0, LimitDirection::Both, context);
302 if (!P0_result) {
303 return FrobeniusSolutionResult::failure(P0_result.error());
304 }
305 auto Q0_result = limit_expression_checked(
306 x2q_expr, x, x0, LimitDirection::Both, context);
307 if (!Q0_result) {
308 return FrobeniusSolutionResult::failure(Q0_result.error());
309 }
310 auto P0_expr = std::move(P0_result.value());
311 auto Q0_expr = std::move(Q0_result.value());
312
313 double P0 = P0_expr
314 ? try_eval_double(P0_expr)
315 : std::numeric_limits<double>::quiet_NaN();
316 double Q0 = Q0_expr
317 ? try_eval_double(Q0_expr)
318 : std::numeric_limits<double>::quiet_NaN();
319 if (!std::isfinite(P0) || !std::isfinite(Q0)) {
320 return FrobeniusSolutionResult::failure(frobenius_unsupported_numeric());
321 }
322 /**
323 * @brief 在原定义域内确认去心极限有限后约分，提取可去奇点解析延拓的 Taylor 系数。
324 * @note 原乘积未经约分的导数仍含缺失点，不能直接用于系数计算。
325 */
326 xp_expr = xp_expr->cancel()->simplify();
327 x2q_expr = x2q_expr->cancel()->simplify();
328
329 auto roots = frobenius_indicial_roots(P0, Q0);
330 if (!roots) {
331 return FrobeniusSolutionResult::failure(roots.error());
332 }
333 const double r1 = roots.value().first;
334 result.indicial_roots = {r1, roots.value().second};
335 auto coefficients = singular_taylor_coefficients(
336 xp_expr, x2q_expr, x0, x, result.truncation_order, P0, Q0, context);
337 if (!coefficients) {
338 return FrobeniusSolutionResult::failure(coefficients.error());
339 }
340 auto recurrence = singular_series_recurrence(
341 coefficients.value(), P0, Q0, r1, result.truncation_order);
342 if (!recurrence) {
343 return FrobeniusSolutionResult::failure(recurrence.error());
344 }
345 auto power_prefix = SymbolicExpr::power(x_minus_x0, SymbolicExpr::number(r1));
346 auto series_part = assemble_frobenius_polynomial(recurrence.value(), x_minus_x0);
347 result.series_solution = SymbolicExpr::multiply(power_prefix, series_part)->simplify();
348 return result;
349}

351ODESingularityResult classify_singular_point_checked(
352 const std::shared_ptr<SymbolicExpr>& p,
353 const std::shared_ptr<SymbolicExpr>& q,
354 const std::shared_ptr<SymbolicExpr>& x0,
355 const std::string& x,
356 ComputationContext& context)
357{
358 const std::string operation = "classify_singular_point";
359 auto valid = validate_ode_two_expr_point(p, q, x0, x, context, operation);
360 if (!valid) {
361 return ODESingularityResult::failure(valid.error());
362 }
363 try {
364 auto p_at_x0 = p->substitute(x, x0)->simplify();
365 auto q_at_x0 = q->substitute(x, x0)->simplify();
366 const double p_val = try_eval_double(p_at_x0);
367 const double q_val = try_eval_double(q_at_x0);
368 if (std::isfinite(p_val) && std::isfinite(q_val)) {
369 return ODESingularityResult::success(
370 ODESingularityType::Ordinary);
371 }
372 if ((!std::isfinite(p_val) &&
373 !free_variables(detail::node(p_at_x0)).empty()) ||
374 (!std::isfinite(q_val) &&
375 !free_variables(detail::node(q_at_x0)).empty())) {
376 return ODESingularityResult::failure(
377 CasErrc::Inconclusive,
378 "singularity classification requires numeric coefficient values",
379 operation);
380 }
381
382 auto x_var = SymbolicExpr::variable(x);
383 auto x_minus_x0 = SymbolicExpr::add(
384 x_var, SymbolicExpr::multiply(SymbolicExpr::number(-1), x0));
385 auto xp = SymbolicExpr::multiply(x_minus_x0, p)->simplify();
386 auto x2q = SymbolicExpr::multiply(
387 SymbolicExpr::power(x_minus_x0, SymbolicExpr::number(2)),
388 q)->simplify();
389 auto xp_limit = limit_checked(xp, x, x0, LimitDirection::Both, context);
390 if (!xp_limit) return ODESingularityResult::failure(xp_limit.error());
391 auto x2q_limit = limit_checked(x2q, x, x0, LimitDirection::Both, context);
392 if (!x2q_limit) return ODESingularityResult::failure(x2q_limit.error());
393 return ODESingularityResult::success(
394 std::holds_alternative<FiniteLimit>(xp_limit.value().value) &&
395 std::holds_alternative<FiniteLimit>(x2q_limit.value().value)
396 ? ODESingularityType::RegularSingular
397 : ODESingularityType::IrregularSingular);
398 } catch (const std::bad_alloc&) {
399 return ODESingularityResult::failure(
400 CasErrc::ResourceLimit,
401 "allocation failed while classifying ODE singularity",
402 operation);
403 } catch (const std::exception& ex) {
404 return ODESingularityResult::failure(
405 CasErrc::InternalInvariant, ex.what(), operation);
406 }
407}

420static Result<void> validate_frobenius_regular_singular_domain(
421 const std::shared_ptr<SymbolicExpr>& p,
422 const std::shared_ptr<SymbolicExpr>& q,
423 const std::shared_ptr<SymbolicExpr>& x0,
424 const std::string& x,
425 ComputationContext& context,
426 const std::string& operation)
427{
428 auto x_var = SymbolicExpr::variable(x);
429 auto x_minus_x0 = SymbolicExpr::add(x_var,
430 SymbolicExpr::multiply(SymbolicExpr::number(-1), x0));
431 auto xp_expr = SymbolicExpr::multiply(x_minus_x0, p)->simplify();
432 auto x2q_expr = SymbolicExpr::multiply(
433 SymbolicExpr::power(x_minus_x0, SymbolicExpr::number(2)), q)->simplify();
434
435 auto P0_result = limit_expression_checked(
436 xp_expr, x, x0, LimitDirection::Both, context);
437 if (!P0_result) {
438 return Result<void>::failure(P0_result.error());
439 }
440 auto Q0_result = limit_expression_checked(
441 x2q_expr, x, x0, LimitDirection::Both, context);
442 if (!Q0_result) {
443 return Result<void>::failure(Q0_result.error());
444 }
445 const auto& P0_expr = P0_result.value();
446 const auto& Q0_expr = Q0_result.value();
447 double P0 = P0_expr
448 ? try_eval_double(P0_expr)
449 : std::numeric_limits<double>::quiet_NaN();
450 double Q0 = Q0_expr
451 ? try_eval_double(Q0_expr)
452 : std::numeric_limits<double>::quiet_NaN();
453 if (!std::isfinite(P0) || !std::isfinite(Q0)) {
454 return Result<void>::failure(
455 CasErrc::Inconclusive,
456 "Frobenius regular-singular coefficients are outside the checked numeric support domain",
457 operation);
458 }
459
460 double discriminant = (P0 - 1.0) * (P0 - 1.0) - 4.0 * Q0;
461 int zero_discriminant;
462 lmmc_double_nearly_equal_tol(discriminant, 0.0, 1e-12, 1e-12,
463 &zero_discriminant);
464 if (discriminant < 0.0 && !zero_discriminant) {
465 return Result<void>::failure(
466 CasErrc::Inconclusive,
467 "Frobenius checked API currently supports real indicial roots only",
468 operation);
469 }
470
471 return Result<void>::success();
472}

474FrobeniusSolutionResult solve_frobenius_checked(
475 const std::shared_ptr<SymbolicExpr>& p,
476 const std::shared_ptr<SymbolicExpr>& q,
477 const std::shared_ptr<SymbolicExpr>& x0,
478 const std::string& x,
479 int order,
480 ComputationContext& context)
481{
482 const std::string operation = "solve_frobenius";
483 auto valid = validate_ode_two_expr_point(p, q, x0, x, context, operation);
484 if (!valid) {
485 return FrobeniusSolutionResult::failure(valid.error());
486 }
487 if (order < 0 || order > 64) {
488 return FrobeniusSolutionResult::failure(
489 CasErrc::InvalidArgument,
490 "Frobenius truncation order must be between 0 and 64",
491 operation);
492 }
493 auto budget = context.consume_steps(static_cast<std::size_t>(order + 1) * 24 + 24,
494 operation);
495 if (!budget) {
496 return FrobeniusSolutionResult::failure(budget.error());
497 }
498
499 try {
500 auto classified =
501 classify_singular_point_checked(p, q, x0, x, context);
502 if (!classified) {
503 return FrobeniusSolutionResult::failure(classified.error());
504 }
505 const auto point_type = classified.value();
506 if (point_type == ODESingularityType::IrregularSingular) {
507 return FrobeniusSolutionResult::failure(
508 CasErrc::Inconclusive,
509 "Frobenius checked API does not support irregular singular points",
510 operation);
511 }
512 if (point_type == ODESingularityType::RegularSingular) {
513 auto regular_domain = validate_frobenius_regular_singular_domain(
514 p, q, x0, x, context, operation);
515 if (!regular_domain) {
516 return FrobeniusSolutionResult::failure(regular_domain.error());
517 }
518 }
519
520 auto solved =
521 solve_frobenius_impl(
522 p, q, x0, x, order, point_type, context);
523 if (!solved) {
524 return solved;
525 }
526 auto solution = std::move(solved.value());
527 if (!solution.series_solution ||
528 !LMCAS::detail::node(solution.series_solution)) {
529 return FrobeniusSolutionResult::failure(
530 CasErrc::Inconclusive,
531 "Frobenius solver produced no series in the supported domain",
532 operation);
533 }
534 if (solution.point_type != point_type) {
535 return FrobeniusSolutionResult::failure(
536 CasErrc::InternalInvariant,
537 "Frobenius solver reported an unexpected singularity type",
538 operation);
539 }
540 return FrobeniusSolutionResult::success(std::move(solution));
541 } catch (const std::bad_alloc&) {
542 return FrobeniusSolutionResult::failure(
543 CasErrc::ResourceLimit,
544 "allocation failed while solving Frobenius series",
545 operation);
546 } catch (const std::exception& ex) {
547 return FrobeniusSolutionResult::failure(
548 CasErrc::InternalInvariant,
549 ex.what(),
550 operation);
551 }
552}

565static FrobeniusSolutionResult solve_frobenius_impl(
566 const std::shared_ptr<SymbolicExpr>& p,
567 const std::shared_ptr<SymbolicExpr>& q,
568 const std::shared_ptr<SymbolicExpr>& x0,
569 const std::string& x,
570 int order,
571 ODESingularityType point_type,
572 ComputationContext& context)
573{
574 FrobeniusSolution result;
575 result.truncation_order = order;
576 result.point_type = point_type;
577 auto x_var = SymbolicExpr::variable(x);
578 auto x_minus_x0 = SymbolicExpr::add(x_var,
579 SymbolicExpr::multiply(SymbolicExpr::number(-1), x0));
580
581 if (result.point_type == ODESingularityType::IrregularSingular) {
582 result.series_solution = nullptr;
583 return result;
584 }
585
586 if (result.point_type == ODESingularityType::Ordinary) {
587 return ordinary_frobenius_solution(p, q, x0, x, x_minus_x0, std::move(result));
588 }
589 return singular_frobenius_solution(
590 p, q, x0, x, x_minus_x0, std::move(result), context);
591}
