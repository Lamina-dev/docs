70static ODESolution solve_homogeneous_ode_impl(
71 const std::shared_ptr<SymbolicExpr>& rhs,
72 const std::string& x,
73 const std::string& y)
74{
75 ODESolution result;
76 result.method_used = ODEType::Homogeneous;
77 result.constants = {"C"};
78
79 /// 齐次方程 y' = f(y/x)
80 /// 令 v = y/x,则 y = v*x,y' = v + x*v'
81 /// 代入: v + x*v' = f(v)
82 /// 即: x*v' = f(v) - v
83 /// 分离变量: dv/(f(v) - v) = dx/x
84
85 /// 将 rhs 中的 y 替换为 v*x,得到 f 关于 v 和 x 的表达式
86 /// 由于齐次性,f(x, vx) = f(1, v)(令 t=1/x 缩放)
87 /// 所以直接用 x=1, y=v 代入得到 f(v)
88 std::string v_name = "v";
89 auto v_var = SymbolicExpr::variable(v_name);
90 auto x_var = SymbolicExpr::variable(x);
91
92 /// f(v) = rhs(1, v):将 x=1, y=v 代入
93 auto f_v = rhs->substitute(x, SymbolicExpr::number(1));
94 f_v = f_v->substitute(y, v_var);
95 f_v = f_v->simplify();
96
97 /// 分离变量方程: dv/(f(v) - v) = dx/x
98 /// 积分: integral dv/(f(v) - v) = integral dx/x = ln|x| + C
99 auto f_minus_v = SymbolicExpr::add(f_v,
100 SymbolicExpr::multiply(SymbolicExpr::number(-1), v_var));
101 f_minus_v = f_minus_v->simplify();
102
103 /// 计算 integral 1/(f(v) - v) dv
104 auto integrand = SymbolicExpr::divide(SymbolicExpr::number(1), f_minus_v);
105 auto lhs_integral = integrand->integrate(v_name);
106
107 /// 右端: ln(x)
108 auto rhs_integral = SymbolicExpr::ln(x_var);
109
110 /// 解为: lhs_integral = ln(x) + C
111 /// 即: lhs_integral - ln(x) = C
112 /// 回代 v = y/x
113 auto v_replacement = SymbolicExpr::divide(
114 SymbolicExpr::variable(y), x_var);
115
116 auto solution = lhs_integral->substitute(v_name, v_replacement);
117 solution = SymbolicExpr::add(solution,
118 SymbolicExpr::multiply(SymbolicExpr::number(-1), rhs_integral));
119 solution = solution->simplify();
120
121 /// 返回隐式解形式: F(x,y) = C
122 result.general_solution = solution;
123 return result;
124}

174static ODESolution solve_bernoulli_ode_impl(
175 const std::shared_ptr<SymbolicExpr>& P,
176 const std::shared_ptr<SymbolicExpr>& Q,
177 int n,
178 const std::string& x,
179 const std::string&)
180{
181 ODESolution result;
182 result.method_used = ODEType::Bernoulli;
183 result.constants = {"C"};
184
185 /// Bernoulli 方程: y' + P(x)*y = Q(x)*y^n (n != 0, 1)
186 /// 令 v = y^(1-n),则 v' = (1-n)*y^(-n)*y'
187 /// 从原方程: y' = -P(x)*y + Q(x)*y^n
188 /// 两边乘以 (1-n)*y^(-n):
189 /// (1-n)*y^(-n)*y' = (1-n)*(-P(x)*y^(1-n) + Q(x))
190 /// v' = (1-n)*(-P(x)*v + Q(x))
191 /// v' + (1-n)*P(x)*v = (1-n)*Q(x)
192 /// 这是关于 v 的一阶线性 ODE
193
194 int one_minus_n = 1 - n;
195 auto coeff = SymbolicExpr::number(one_minus_n);
196
197 /// 线性 ODE 的系数: P_linear = (1-n)*P(x), Q_linear = (1-n)*Q(x)
198 auto P_linear = SymbolicExpr::multiply(coeff, P)->simplify();
199 auto Q_linear = SymbolicExpr::multiply(coeff, Q)->simplify();
200
201 /// 用积分因子法求解线性 ODE: v' + P_linear*v = Q_linear
202 /// 积分因子 mu = exp(integralP_linear dx)
203 auto intP = P_linear->integrate(x);
204 auto mu = SymbolicExpr::exp(intP);
205
206 /// v = (1/mu) * (integral Q_linear * mu dx + C)
207 auto Q_mu = SymbolicExpr::multiply(Q_linear, mu)->simplify();
208 auto int_Q_mu = Q_mu->integrate(x);
209
210 auto C_const = SymbolicExpr::variable("C");
211 auto numerator = SymbolicExpr::add(int_Q_mu, C_const);
212 auto v_solution = SymbolicExpr::divide(numerator, mu)->simplify();
213
214 /// 回代: v = y^(1-n),所以 y = v^(1/(1-n))
215 /// y^(1-n) = v_solution
216 /// y = v_solution^(1/(1-n))
217 auto exponent = SymbolicExpr::divide(
218 SymbolicExpr::number(1),
219 SymbolicExpr::number(one_minus_n));
220 auto y_solution = SymbolicExpr::power(v_solution, exponent)->simplify();
221
222 result.general_solution = y_solution;
223 return result;
224}

227std::shared_ptr<SymbolicExpr> find_integrating_factor(
228 const std::shared_ptr<SymbolicExpr>& M,
229 const std::shared_ptr<SymbolicExpr>& N,
230 const std::string& x,
231 const std::string& y)
232{
233 if (!M || !N) return nullptr;
234
235 /// 计算 partialM/partialy - partialN/partialx
236 auto dM_dy = M->differentiate(y);
237 auto dN_dx = N->differentiate(x);
238 if (!dM_dy || !dN_dx) return nullptr;
239
240 auto diff = SymbolicExpr::add(dM_dy,
241 SymbolicExpr::multiply(SymbolicExpr::number(-1), dN_dx));
242 diff = diff->simplify();
243
244 if (diff->is_zero()) {
245 /// 已经恰当,积分因子为 1
246 return SymbolicExpr::number(1);
247 }
248
249 /// 尝试 mu = mu(x): (partialM/partialy - partialN/partialx) / N 仅依赖 x
250 if (N && !N->is_zero()) {
251 auto ratio_x = SymbolicExpr::divide(diff, N)->simplify();
252 if (!expression_depends_on_variable(lamina::detail::node(ratio_x), y)) {
253 /// mu(x) = exp(integral ratio_x dx)
254 auto int_ratio = ratio_x->integrate(x);
255 return SymbolicExpr::exp(int_ratio);
256 }
257 }
258
259 /// 尝试 mu = mu(y): (partialN/partialx - partialM/partialy) / M 仅依赖 y
260 if (M && !M->is_zero()) {
261 auto neg_diff = SymbolicExpr::multiply(SymbolicExpr::number(-1), diff)->simplify();
262 auto ratio_y = SymbolicExpr::divide(neg_diff, M)->simplify();
263 if (!expression_depends_on_variable(lamina::detail::node(ratio_y), x)) {
264 /// mu(y) = exp(integral ratio_y dy)
265 auto int_ratio = ratio_y->integrate(y);
266 return SymbolicExpr::exp(int_ratio);
267 }
268 }
269
270 return nullptr;
271}

314static ODESolution solve_exact_ode_impl(
315 const std::shared_ptr<SymbolicExpr>& M,
316 const std::shared_ptr<SymbolicExpr>& N,
317 const std::string& x,
318 const std::string& y)
319{
320 ODESolution result;
321 result.method_used = ODEType::Exact;
322 result.constants = {"C"};
323
324 auto M_eff = M;
325 auto N_eff = N;
326
327 /// 检查是否恰当,若不恰当则尝试寻找积分因子
328 if (!is_exact_ode(M, N, x, y)) {
329 auto mu = find_integrating_factor(M, N, x, y);
330 if (!mu) {
331 /// 积分因子搜索未决时以空 general_solution 表示.
332 result.general_solution = nullptr;
333 return result;
334 }
335 /// 乘以积分因子
336 M_eff = SymbolicExpr::multiply(mu, M)->simplify();
337 N_eff = SymbolicExpr::multiply(mu, N)->simplify();
338 }
339
340 /// 恰当方程: partialF/partialx = M, partialF/partialy = N
341 /// F(x,y) = integral M dx + g(y)
342 /// 对 F 关于 y 求导: partialF/partialy = partial/partialy(integral M dx) + g'(y) = N
343 /// 所以 g'(y) = N - partial/partialy(integral M dx)
344
345 /// 步骤 1: 计算 integral M dx(将 y 视为常数)
346 auto F_partial = M_eff->integrate(x);
347
348 /// 步骤 2: 对 F_partial 关于 y 求导
349 auto dF_dy = F_partial->differentiate(y);
350 dF_dy = dF_dy ? dF_dy->simplify() : SymbolicExpr::number(0);
351
352 /// 步骤 3: g'(y) = N - dF_dy
353 auto g_prime = SymbolicExpr::add(N_eff,
354 SymbolicExpr::multiply(SymbolicExpr::number(-1), dF_dy));
355 g_prime = g_prime->simplify();
356
357 /// 步骤 4: g(y) = integral g'(y) dy
358 auto g_y = g_prime->integrate(y);
359
360 /// 步骤 5: F(x,y) = F_partial + g(y)
361 auto F_total = SymbolicExpr::add(F_partial, g_y)->simplify();
362
363 /// 解为 F(x,y) = C
364 result.general_solution = F_total;
365 return result;
366}
