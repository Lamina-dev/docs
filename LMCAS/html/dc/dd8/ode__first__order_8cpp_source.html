68static ODESolution solve_homogeneous_ode_impl(
69 const std::shared_ptr<SymbolicExpr>& rhs,
70 const std::string& x,
71 const std::string& y)
72{
73 ODESolution result;
74 result.method_used = ODEType::Homogeneous;
75 result.constants = {"C"};
76
77 /// 齐次方程 y' = f(y/x)
78 /// 令 v = y/x,则 y = v*x,y' = v + x*v'
79 /// 代入: v + x*v' = f(v)
80 /// 即: x*v' = f(v) - v
81 /// 分离变量: dv/(f(v) - v) = dx/x
82
83 /// 将 rhs 中的 y 替换为 v*x,得到 f 关于 v 和 x 的表达式
84 /// 由于齐次性,f(x, vx) = f(1, v)(令 t=1/x 缩放)
85 /// 所以直接用 x=1, y=v 代入得到 f(v)
86 std::string v_name = "v";
87 auto v_var = SymbolicExpr::variable(v_name);
88 auto x_var = SymbolicExpr::variable(x);
89
90 /// f(v) = rhs(1, v):将 x=1, y=v 代入
91 auto f_v = rhs->substitute(x, SymbolicExpr::number(1));
92 f_v = f_v->substitute(y, v_var);
93 f_v = f_v->simplify();
94
95 /// 分离变量方程: dv/(f(v) - v) = dx/x
96 /// 积分: integral dv/(f(v) - v) = integral dx/x = ln|x| + C
97 auto f_minus_v = SymbolicExpr::add(f_v,
98 SymbolicExpr::multiply(SymbolicExpr::number(-1), v_var));
99 f_minus_v = f_minus_v->simplify();
100
101 /// 计算 integral 1/(f(v) - v) dv
102 auto integrand = SymbolicExpr::divide(SymbolicExpr::number(1), f_minus_v);
103 auto lhs_integral = integrand->integrate(v_name);
104
105 /// 右端: ln(x)
106 auto rhs_integral = SymbolicExpr::ln(x_var);
107
108 /// 解为: lhs_integral = ln(x) + C
109 /// 即: lhs_integral - ln(x) = C
110 /// 回代 v = y/x
111 auto v_replacement = SymbolicExpr::divide(
112 SymbolicExpr::variable(y), x_var);
113
114 auto solution = lhs_integral->substitute(v_name, v_replacement);
115 solution = SymbolicExpr::add(solution,
116 SymbolicExpr::multiply(SymbolicExpr::number(-1), rhs_integral));
117 solution = solution->simplify();
118
119 /// 返回隐式解形式: F(x,y) = C
120 result.general_solution = solution;
121 return result;
122}

170static ODESolution solve_bernoulli_ode_impl(
171 const std::shared_ptr<SymbolicExpr>& P,
172 const std::shared_ptr<SymbolicExpr>& Q,
173 int n,
174 const std::string& x,
175 const std::string&)
176{
177 ODESolution result;
178 result.method_used = ODEType::Bernoulli;
179 result.constants = {"C"};
180
181 /// Bernoulli 方程: y' + P(x)*y = Q(x)*y^n (n != 0, 1)
182 /// 令 v = y^(1-n),则 v' = (1-n)*y^(-n)*y'
183 /// 从原方程: y' = -P(x)*y + Q(x)*y^n
184 /// 两边乘以 (1-n)*y^(-n):
185 /// (1-n)*y^(-n)*y' = (1-n)*(-P(x)*y^(1-n) + Q(x))
186 /// v' = (1-n)*(-P(x)*v + Q(x))
187 /// v' + (1-n)*P(x)*v = (1-n)*Q(x)
188 /// 这是关于 v 的一阶线性 ODE
189
190 int one_minus_n = 1 - n;
191 auto coeff = SymbolicExpr::number(one_minus_n);
192
193 /// 线性 ODE 的系数: P_linear = (1-n)*P(x), Q_linear = (1-n)*Q(x)
194 auto P_linear = SymbolicExpr::multiply(coeff, P)->simplify();
195 auto Q_linear = SymbolicExpr::multiply(coeff, Q)->simplify();
196
197 /// 用积分因子法求解线性 ODE: v' + P_linear*v = Q_linear
198 /// 积分因子 mu = exp(integralP_linear dx)
199 auto intP = P_linear->integrate(x);
200 auto mu = SymbolicExpr::exp(intP);
201
202 /// v = (1/mu) * (integral Q_linear * mu dx + C)
203 auto Q_mu = SymbolicExpr::multiply(Q_linear, mu)->simplify();
204 auto int_Q_mu = Q_mu->integrate(x);
205
206 auto C_const = SymbolicExpr::variable("C");
207 auto numerator = SymbolicExpr::add(int_Q_mu, C_const);
208 auto v_solution = SymbolicExpr::divide(numerator, mu)->simplify();
209
210 /// 回代: v = y^(1-n),所以 y = v^(1/(1-n))
211 /// y^(1-n) = v_solution
212 /// y = v_solution^(1/(1-n))
213 auto exponent = SymbolicExpr::divide(
214 SymbolicExpr::number(1),
215 SymbolicExpr::number(one_minus_n));
216 auto y_solution = SymbolicExpr::power(v_solution, exponent)->simplify();
217
218 result.general_solution = y_solution;
219 return result;
220}

223std::shared_ptr<SymbolicExpr> find_integrating_factor(
224 const std::shared_ptr<SymbolicExpr>& M,
225 const std::shared_ptr<SymbolicExpr>& N,
226 const std::string& x,
227 const std::string& y)
228{
229 if (!M || !N) return nullptr;
230
231 /// 计算 partialM/partialy - partialN/partialx
232 auto dM_dy = M->differentiate(y);
233 auto dN_dx = N->differentiate(x);
234 if (!dM_dy || !dN_dx) return nullptr;
235
236 auto diff = SymbolicExpr::add(dM_dy,
237 SymbolicExpr::multiply(SymbolicExpr::number(-1), dN_dx));
238 diff = diff->simplify();
239
240 if (diff->is_zero()) {
241 /// 已经恰当,积分因子为 1
242 return SymbolicExpr::number(1);
243 }
244
245 /// 尝试 mu = mu(x): (partialM/partialy - partialN/partialx) / N 仅依赖 x
246 if (N && !N->is_zero()) {
247 auto ratio_x = SymbolicExpr::divide(diff, N)->simplify();
248 if (!expression_depends_on_variable(LMCAS::detail::node(ratio_x), y)) {
249 /// mu(x) = exp(integral ratio_x dx)
250 auto int_ratio = ratio_x->integrate(x);
251 return SymbolicExpr::exp(int_ratio);
252 }
253 }
254
255 /// 尝试 mu = mu(y): (partialN/partialx - partialM/partialy) / M 仅依赖 y
256 if (M && !M->is_zero()) {
257 auto neg_diff = SymbolicExpr::multiply(SymbolicExpr::number(-1), diff)->simplify();
258 auto ratio_y = SymbolicExpr::divide(neg_diff, M)->simplify();
259 if (!expression_depends_on_variable(LMCAS::detail::node(ratio_y), x)) {
260 /// mu(y) = exp(integral ratio_y dy)
261 auto int_ratio = ratio_y->integrate(y);
262 return SymbolicExpr::exp(int_ratio);
263 }
264 }
265
266 return nullptr;
267}

308static ODESolution solve_exact_ode_impl(
309 const std::shared_ptr<SymbolicExpr>& M,
310 const std::shared_ptr<SymbolicExpr>& N,
311 const std::string& x,
312 const std::string& y)
313{
314 ODESolution result;
315 result.method_used = ODEType::Exact;
316 result.constants = {"C"};
317
318 auto M_eff = M;
319 auto N_eff = N;
320
321 /// 检查是否恰当,若不恰当则尝试寻找积分因子
322 if (!is_exact_ode(M, N, x, y)) {
323 auto mu = find_integrating_factor(M, N, x, y);
324 if (!mu) {
325 /// 积分因子搜索未决时以空 general_solution 表示.
326 result.general_solution = nullptr;
327 return result;
328 }
329 /// 乘以积分因子
330 M_eff = SymbolicExpr::multiply(mu, M)->simplify();
331 N_eff = SymbolicExpr::multiply(mu, N)->simplify();
332 }
333
334 /// 恰当方程: partialF/partialx = M, partialF/partialy = N
335 /// F(x,y) = integral M dx + g(y)
336 /// 对 F 关于 y 求导: partialF/partialy = partial/partialy(integral M dx) + g'(y) = N
337 /// 所以 g'(y) = N - partial/partialy(integral M dx)
338
339 /// 步骤 1: 计算 integral M dx(将 y 视为常数)
340 auto F_partial = M_eff->integrate(x);
341
342 /// 步骤 2: 对 F_partial 关于 y 求导
343 auto dF_dy = F_partial->differentiate(y);
344 dF_dy = dF_dy ? dF_dy->simplify() : SymbolicExpr::number(0);
345
346 /// 步骤 3: g'(y) = N - dF_dy
347 auto g_prime = SymbolicExpr::add(N_eff,
348 SymbolicExpr::multiply(SymbolicExpr::number(-1), dF_dy));
349 g_prime = g_prime->simplify();
350
351 /// 步骤 4: g(y) = integral g'(y) dy
352 auto g_y = g_prime->integrate(y);
353
354 /// 步骤 5: F(x,y) = F_partial + g(y)
355 auto F_total = SymbolicExpr::add(F_partial, g_y)->simplify();
356
357 /// 解为 F(x,y) = C
358 result.general_solution = F_total;
359 return result;
360}
