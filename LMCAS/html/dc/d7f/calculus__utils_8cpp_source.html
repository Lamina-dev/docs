95bool calculus_utils_is_infinity(const std::shared_ptr<SymbolicExpr>& expr)
96{
97 if (!expr || !LMCAS::detail::node(expr)) {
98 return false;
99 }
100
101 if (auto f = std::dynamic_pointer_cast<const FunctionNode>(LMCAS::detail::node(expr))) {
102 return f->type() == FunctionNode::FuncType::Infinity;
103 }
104 if (auto m = std::dynamic_pointer_cast<const MultiplyNode>(LMCAS::detail::node(expr))) {
105 for (auto& op : m->operands()) {
106 if (is_infinity_node(op)) {
107 return true;
108 }
109 }
110 }
111 return false;
112}

144std::shared_ptr<SymbolicExpr> log_differentiate(
145 const std::shared_ptr<SymbolicExpr>& f, const std::string& var)
146{
147 if (!f || !LMCAS::detail::node(f)) {
148 return nullptr;
149 }
150
151 /// 计算 ln(f)
152 auto ln_f = SymbolicExpr::ln(f);
153
154 /// 对 ln(f) 求导 - 化简器会自动应用对数规则
155 /// (ln(a*b) = ln(a)+ln(b), ln(a^n) = n*ln(a))
156 auto ln_f_simplified = ln_f->simplify();
157 auto d_ln_f = ln_f_simplified->differentiate(var);
158
159 /// 结果 = f * d/dx[ln(f)]
160 auto result = SymbolicExpr::multiply(f, d_ln_f);
161 return result->simplify();
162}

175std::vector<std::pair<std::shared_ptr<SymbolicExpr>, std::string>> total_differential(
176 const std::shared_ptr<SymbolicExpr>& f, const std::vector<std::string>& vars)
177{
178 std::vector<std::pair<std::shared_ptr<SymbolicExpr>, std::string>> result;
179 if (!f || !LMCAS::detail::node(f)) {
180 return result;
181 }
182
183 result.reserve(vars.size());
184 for (const auto& v : vars) {
185 auto partial = f->differentiate(v);
186 result.emplace_back(partial, v);
187 }
188 return result;
189}
