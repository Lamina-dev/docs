108 return f->type() == FunctionNode::FuncType::Infinity;

111 for (auto& op : m->operands()) {
112 if (auto f = std::dynamic_pointer_cast<const FunctionNode>(op)) {
113 if (f->type() == FunctionNode::FuncType::Infinity) return true;
114 }
115 }
116 }
117 return false;

133static bool calculus_utils_contains_rootof(const std::shared_ptr<const SymbolicNode>& node)
134{
135 if (!node) return false;
136
137 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(node)) {
138 if (function->type() == FunctionNode::FuncType::RootOf) return true;
139 for (const auto& argument : function->arguments()) {
140 if (calculus_utils_contains_rootof(argument)) return true;
141 }
142 return false;
143 }
144 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
145 for (const auto& operand : add->operands()) {
146 if (calculus_utils_contains_rootof(operand)) return true;
147 }
148 return false;
149 }
150 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
151 for (const auto& operand : multiply->operands()) {
152 if (calculus_utils_contains_rootof(operand)) return true;
153 }
154 return false;
155 }
156 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
157 return calculus_utils_contains_rootof(power->base()) ||
158 calculus_utils_contains_rootof(power->exponent());
159 }
160 return false;
161}

252static CalculusRealCandidateStatus calculus_utils_extract_real_candidate(
253 const std::shared_ptr<SymbolicExpr>& candidate,
254 std::shared_ptr<SymbolicExpr>& real_candidate)
255{
256 real_candidate.reset();
257 auto simplified_candidate = rootof_simplify(candidate);
258 if (!simplified_candidate || !lamina::detail::node(simplified_candidate)) {
259 return CalculusRealCandidateStatus::Inconclusive;
260 }
261 if (calculus_utils_contains_rootof(lamina::detail::node(simplified_candidate))) {
262 return CalculusRealCandidateStatus::Inconclusive;
263 }
264 if (calculus_utils_contains_nonreal_sqrt(lamina::detail::node(simplified_candidate))) {
265 return CalculusRealCandidateStatus::NonReal;
266 }
267
268 if (auto complex = std::dynamic_pointer_cast<const ComplexNode>(
269 lamina::detail::node(simplified_candidate))) {
270 auto imag = lamina::detail::make_expression_ptr(complex->imag())->simplify();
271 if (!imag || !lamina::detail::node(imag)) {
272 return CalculusRealCandidateStatus::Inconclusive;
273 }
274 if (imag->is_zero()) {
275 auto real = lamina::detail::make_expression_ptr(complex->real())->simplify();
276 if (!real || !lamina::detail::node(real)) {
277 return CalculusRealCandidateStatus::Inconclusive;
278 }
279 real_candidate = real;
280 return CalculusRealCandidateStatus::Real;
281 }
282 return CalculusRealCandidateStatus::NonReal;
283 }
284
285 if (calculus_utils_contains_complex(lamina::detail::node(simplified_candidate))) {
286 return CalculusRealCandidateStatus::Inconclusive;
287 }
288
289 real_candidate = simplified_candidate;
290 return CalculusRealCandidateStatus::Real;
291}

293/**
294 * @internal
295 * @brief 从表达式中提取分母部分。
296 *
297 * 对于 PowerNode 且指数为负，base 即为分母。
298 * 对于 MultiplyNode，查找指数为负的因子作为分母。
299 */
300static std::shared_ptr<SymbolicExpr> calculus_utils_extract_denominator(
301 const std::shared_ptr<SymbolicExpr>& expr)
302{
303 if (!expr || !lamina::detail::node(expr)) return nullptr;
304
305 /// 检查是否为 PowerNode 且指数为负（如 x^(-1) 表示 1/x）
306 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(lamina::detail::node(expr))) {
307 if (auto num = std::dynamic_pointer_cast<const NumberNode>(pow->exponent())) {
308 double e = 0;
309 if (std::holds_alternative<lmmc_real_t>(num->value()))
310 e = std::get<lmmc_real_t>(num->value());
311 else if (std::holds_alternative<BigInt>(num->value()))
312 e = std::get<BigInt>(num->value()).to_double();
313 else if (std::holds_alternative<Rational>(num->value()))
314 e = std::get<Rational>(num->value()).to_double();
315
316 if (e < 0) {
317 if (e == -1.0) {
318 return lamina::detail::make_expression_ptr(pow->base());
319 }
320 auto pos_exp = lamina::detail::make_node<NumberNode>(BigInt(static_cast<int>(-e)));
321 auto den_node = lamina::detail::make_node<PowerNode>(pow->base(), pos_exp);
322 return lamina::detail::make_expression_ptr(den_node);
323 }
324 }
325 }
326
327 /// 检查 MultiplyNode 中是否有负指数因子
328 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(expr))) {
329 std::vector<std::shared_ptr<const SymbolicNode>> den_factors;
330
331 for (auto& op : mul->operands()) {
332 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(op)) {
333 if (auto num = std::dynamic_pointer_cast<const NumberNode>(pow->exponent())) {
334 double e = 0;
335 if (std::holds_alternative<lmmc_real_t>(num->value()))
336 e = std::get<lmmc_real_t>(num->value());
337 else if (std::holds_alternative<BigInt>(num->value()))
338 e = std::get<BigInt>(num->value()).to_double();
339 else if (std::holds_alternative<Rational>(num->value()))
340 e = std::get<Rational>(num->value()).to_double();
341
342 if (e < 0) {
343 if (e == -1.0) {
344 den_factors.push_back(pow->base());
345 } else {
346 auto pos_exp = lamina::detail::make_node<NumberNode>(BigInt(static_cast<int>(-e)));
347 den_factors.push_back(lamina::detail::make_node<PowerNode>(pow->base(), pos_exp));
348 }
349 }
350 }
351 }
352 }
353
354 if (!den_factors.empty()) {
355 std::shared_ptr<const SymbolicNode> den_node;
356 if (den_factors.size() == 1) {
357 den_node = den_factors[0];
358 } else {
359 den_node = lamina::detail::make_node<MultiplyNode>(den_factors);
360 }
361 return lamina::detail::make_expression_ptr(den_node);
362 }
363 }
364
365 return nullptr;
366}

388static bool calculus_utils_split_rational(
389 const std::shared_ptr<SymbolicExpr>& expr,
390 std::shared_ptr<SymbolicExpr>& numerator,
391 std::shared_ptr<SymbolicExpr>& denominator)
392{
393 if (!expr || !lamina::detail::node(expr)) return false;
394
395 std::vector<std::shared_ptr<const SymbolicNode>> num_factors;
396 std::vector<std::shared_ptr<const SymbolicNode>> den_factors;
397
398 auto collect_factor = [&](const std::shared_ptr<const SymbolicNode>& factor) {
399 if (auto power = std::dynamic_pointer_cast<const PowerNode>(factor)) {
400 auto exponent = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
401 if (calculus_utils_is_minus_one(exponent)) {
402 den_factors.push_back(power->base());
403 return;
404 }
405 }
406 num_factors.push_back(factor);
407 };
408
409 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(expr))) {
410 for (const auto& factor : multiply->operands()) collect_factor(factor);
411 } else {
412 collect_factor(lamina::detail::node(expr));
413 }
414
415 if (den_factors.empty()) return false;
416 numerator = calculus_utils_product(num_factors);
417 denominator = calculus_utils_product(den_factors);
418 return numerator && denominator;
419}

421static bool calculus_utils_oblique_from_rational(
422 const std::shared_ptr<SymbolicExpr>& f,
423 const std::string& var,
424 std::shared_ptr<SymbolicExpr>& slope,
425 std::shared_ptr<SymbolicExpr>& intercept)
426{
427 std::shared_ptr<SymbolicExpr> numerator;
428 std::shared_ptr<SymbolicExpr> denominator;
429 if (!calculus_utils_split_rational(f, numerator, denominator)) return false;
430
431 try {
432 auto num_poly = symbolic_to_poly<Rational>(numerator->expand()->simplify(), var);
433 auto den_poly = symbolic_to_poly<Rational>(denominator->expand()->simplify(), var);
434 if (den_poly.is_zero()) return false;
435 auto [quotient, remainder] = num_poly.div_mod(den_poly);
436 (void)remainder;
437 if (quotient.degree() != 1 || quotient.coeffs.size() < 2) return false;
438 intercept = SymbolicExpr::number(quotient.coeffs[0])->simplify();
439 slope = SymbolicExpr::number(quotient.coeffs[1])->simplify();
440 return !slope->is_zero();
441 } catch (...) {
442 return false;
443 }
444}

447ContinuityType continuity_at(
448 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
449 const std::shared_ptr<SymbolicExpr>& point)
450{
451 if (!f || !point) return ContinuityType::Essential;
452
453 /// 计算左极限
454 auto left_lim = f->limit(var, point, "-");
455 /// 计算右极限
456 auto right_lim = f->limit(var, point, "+");
457 /// 计算函数值（直接代入）
458 auto func_val = f->substitute(var, point);
459
460 /// 简化结果
461 if (left_lim) left_lim = left_lim->simplify();
462 if (right_lim) right_lim = right_lim->simplify();
463 if (func_val) func_val = func_val->simplify();
464
465 /// 检查极限是否存在（不为无穷）
466 bool left_exists = left_lim && !calculus_utils_is_infinity(left_lim);
467 bool right_exists = right_lim && !calculus_utils_is_infinity(right_lim);
468
469 /// 如果任一侧极限不存在或为无穷 → 本性间断点
470 if (!left_exists || !right_exists) {
471 return ContinuityType::Essential;
472 }
473
474 /// 检查左极限是否等于右极限
475 bool limits_equal = calculus_utils_expr_equal(left_lim, right_lim);
476
477 if (!limits_equal) {
478 /// 左极限 ≠ 右极限 → 跳跃间断点
479 return ContinuityType::Jump;
480 }
481
482 /// 左极限 = 右极限，检查是否等于函数值
483 bool val_exists = func_val && !calculus_utils_is_infinity(func_val);
484
485 if (!val_exists || !calculus_utils_expr_equal(left_lim, func_val)) {
486 /// 极限存在但不等于函数值（或函数无定义） → 可去间断点
487 return ContinuityType::Removable;
488 }
489
490 /// 左极限 = 右极限 = 函数值 → 连续
491 return ContinuityType::Continuous;
492}

495AsymptoteAnalysisResult asymptotes_checked(
496 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
497 ComputationContext& context)
498{
499 AsymptoteResult result;
500 const std::string operation = "asymptotes";
501 auto input = calculus_utils_validate_expr(f, var, context, operation);
502 if (!input) return AsymptoteAnalysisResult::failure(input.error());
503
504 auto pos_inf = SymbolicExpr::infinity(1);
505 auto neg_inf = SymbolicExpr::infinity(-1);
506
507 /// 提取分母，求解分母 = 0 的点
508 auto denominator = calculus_utils_extract_denominator(f);
509 if (denominator) {
510 auto solved_zeros = solve_equation(denominator, var, context, SolveOptions{});
511 if (!solved_zeros) return AsymptoteAnalysisResult::failure(solved_zeros.error());
512
513 const auto& zero_set = solved_zeros.value();
514 if (zero_set.kind() != SolutionSet::Kind::Empty &&
515 zero_set.kind() != SolutionSet::Kind::Finite) {
516 return AsymptoteAnalysisResult::failure(
517 CasErrc::Inconclusive,
518 "asymptote denominator zeros are outside the finite exact support domain",
519 operation);
520 }
521
522 std::vector<std::shared_ptr<SymbolicExpr>> zeros;
523 if (zero_set.kind() == SolutionSet::Kind::Finite) {
524 zeros.reserve(zero_set.finite_solutions().size());
525 for (const auto& solution : zero_set.finite_solutions()) {
526 if (!solution.value || !lamina::detail::node(solution.value)) {
527 return AsymptoteAnalysisResult::failure(
528 CasErrc::InternalInvariant,
529 "checked solver returned a null vertical-asymptote candidate",
530 operation);
531 }
532 auto candidate = rootof_simplify(solution.value);
533 if (!candidate || !lamina::detail::node(candidate)) {
534 return AsymptoteAnalysisResult::failure(
535 CasErrc::InternalInvariant,
536 "checked solver returned a malformed vertical-asymptote candidate",
537 operation);
538 }
539 if (calculus_utils_contains_rootof(lamina::detail::node(candidate))) {
540 return AsymptoteAnalysisResult::failure(
541 CasErrc::Inconclusive,
542 "vertical-asymptote candidate could not be reduced to an exact explicit point",
543 operation);
544 }
545 zeros.push_back(candidate);
546 }
547 }
548
549 for (auto& z : zeros) {
550 if (!z || calculus_utils_is_infinity(z)) continue;
551
552 /// 验证该点处极限为 ±∞
553 auto lim_at_z = f->limit(var, z);
554 if (calculus_utils_is_infinity(lim_at_z)) {
555 result.vertical.push_back(z);
556 } else {
557 /// 尝试单侧极限
558 auto lim_right = f->limit(var, z, "+");
559 auto lim_left = f->limit(var, z, "-");
560 if (calculus_utils_is_infinity(lim_right) || calculus_utils_is_infinity(lim_left)) {
561 result.vertical.push_back(z);
562 }
563 }
564 }
565 }
566
567 /// 计算 x→+∞ 和 x→-∞ 的极限
568 auto lim_pos = f->limit(var, pos_inf);
569 auto lim_neg = f->limit(var, neg_inf);
570
571 if (lim_pos) lim_pos = lim_pos->simplify();
572 if (lim_neg) lim_neg = lim_neg->simplify();
573
574 bool has_horiz_pos = lim_pos && !calculus_utils_is_infinity(lim_pos);
575 bool has_horiz_neg = lim_neg && !calculus_utils_is_infinity(lim_neg);
576
577 if (has_horiz_pos) {
578 result.horizontal.push_back(lim_pos);
579 }
580 if (has_horiz_neg) {
581 /// 避免重复添加相同的水平渐近线
582 bool duplicate = has_horiz_pos && calculus_utils_expr_equal(lim_pos, lim_neg);
583 if (!duplicate) {
584 result.horizontal.push_back(lim_neg);
585 }
586 }
587
588 /// 仅当对应方向无水平渐近线时才检查斜渐近线
589 auto x_expr = SymbolicExpr::variable(var);
590
591 if (!has_horiz_pos && !has_horiz_neg) {
592 std::shared_ptr<SymbolicExpr> slope;
593 std::shared_ptr<SymbolicExpr> intercept;
594 if (calculus_utils_oblique_from_rational(f, var, slope, intercept)) {
595 result.oblique.emplace_back(slope, intercept);
596 return AsymptoteAnalysisResult::success(std::move(result));
597 }
598 }
599
600 if (!has_horiz_pos) {
601 /// 计算 slope = lim(f/x) as x→+∞
602 auto f_over_x = SymbolicExpr::multiply(f, SymbolicExpr::power(x_expr, SymbolicExpr::number(-1)));
603 auto slope_pos = f_over_x->limit(var, pos_inf);
604 if (slope_pos) slope_pos = slope_pos->simplify();
605
606 if (slope_pos && !calculus_utils_is_infinity(slope_pos) && !slope_pos->is_zero()) {
607 /// 计算 intercept = lim(f - slope*x) as x→+∞
608 auto slope_times_x = SymbolicExpr::multiply(slope_pos, x_expr);
609 auto f_minus_mx = SymbolicExpr::add(f, SymbolicExpr::multiply(SymbolicExpr::number(-1), slope_times_x));
610 auto intercept_pos = f_minus_mx->limit(var, pos_inf);
611 if (intercept_pos) intercept_pos = intercept_pos->simplify();
612
613 if (intercept_pos && !calculus_utils_is_infinity(intercept_pos)) {
614 result.oblique.emplace_back(slope_pos, intercept_pos);
615 }
616 }
617 }
618
619 if (!has_horiz_neg) {
620 /// 计算 slope = lim(f/x) as x→-∞
621 auto f_over_x = SymbolicExpr::multiply(f, SymbolicExpr::power(x_expr, SymbolicExpr::number(-1)));
622 auto slope_neg = f_over_x->limit(var, neg_inf);
623 if (slope_neg) slope_neg = slope_neg->simplify();
624
625 if (slope_neg && !calculus_utils_is_infinity(slope_neg) && !slope_neg->is_zero()) {
626 /// 计算 intercept = lim(f - slope*x) as x→-∞
627 auto slope_times_x = SymbolicExpr::multiply(slope_neg, x_expr);
628 auto f_minus_mx = SymbolicExpr::add(f, SymbolicExpr::multiply(SymbolicExpr::number(-1), slope_times_x));
629 auto intercept_neg = f_minus_mx->limit(var, neg_inf);
630 if (intercept_neg) intercept_neg = intercept_neg->simplify();
631
632 if (intercept_neg && !calculus_utils_is_infinity(intercept_neg)) {
633 /// 避免重复：如果 +∞ 方向已有相同斜渐近线则跳过
634 bool duplicate = false;
635 for (auto& [s, i] : result.oblique) {
636 if (calculus_utils_expr_equal(s, slope_neg) &&
637 calculus_utils_expr_equal(i, intercept_neg)) {
638 duplicate = true;
639 break;
640 }
641 }
642 if (!duplicate) {
643 result.oblique.emplace_back(slope_neg, intercept_neg);
644 }
645 }
646 }
647 }
648
649 return AsymptoteAnalysisResult::success(std::move(result));
650}

713ExpressionResult inverse_derivative_checked(
714 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
715 const std::shared_ptr<SymbolicExpr>& point,
716 ComputationContext& context)
717{
718 const std::string operation = "inverse_derivative";
719 auto input = calculus_utils_validate_expr_target(f, var, point, context, operation);
720 if (!input) return ExpressionResult::failure(input.error());
721 auto step = context.consume_steps(4, operation);
722 if (!step) return ExpressionResult::failure(step.error());
723
724 try {
725 auto eq = SymbolicExpr::add(
726 f, SymbolicExpr::multiply(SymbolicExpr::number(-1), point));
727 auto simplified_eq = eq ? eq->simplify() : nullptr;
728 if (simplified_eq && lamina::detail::node(simplified_eq)) eq = simplified_eq;
729
730 std::vector<std::shared_ptr<SymbolicExpr>> candidates;
731 auto inverse = inverse_function_checked(f, var, point, context);
732 if (inverse) {
733 candidates = inverse.value();
734 } else {
735 return ExpressionResult::failure(inverse.error());
736 }
737
738 std::vector<std::shared_ptr<SymbolicExpr>> real_candidates;
739 real_candidates.reserve(candidates.size());
740 for (const auto& candidate : candidates) {
741 std::shared_ptr<SymbolicExpr> real_candidate;
742 auto status = calculus_utils_extract_real_candidate(candidate, real_candidate);
743 if (status == CalculusRealCandidateStatus::Inconclusive) {
744 return ExpressionResult::failure(
745 CasErrc::Inconclusive,
746 "inverse derivative candidate reality could not be proven",
747 operation);
748 }
749 if (status == CalculusRealCandidateStatus::NonReal) {
750 continue;
751 }
752
753 auto residual = eq->substitute(var, real_candidate);
754 auto simplified_residual = residual ? residual->simplify() : nullptr;
755 if (!simplified_residual || !lamina::detail::node(simplified_residual)) {
756 return ExpressionResult::failure(
757 CasErrc::InternalInvariant,
758 "inverse derivative candidate verification produced a null residual",
759 operation);
760 }
761 if (!simplified_residual->is_zero()) {
762 continue;
763 }
764
765 bool duplicate = false;
766 for (const auto& existing : real_candidates) {
767 if (existing && lamina::detail::node(existing) &&
768 lamina::detail::node(existing)->equals(*lamina::detail::node(real_candidate))) {
769 duplicate = true;
770 break;
771 }
772 }
773 if (!duplicate) real_candidates.push_back(real_candidate);
774 }
775 candidates = std::move(real_candidates);
776
777 if (candidates.empty()) {
778 return ExpressionResult::failure(
779 CasErrc::DomainError,
780 "inverse point is outside the proven range",
781 operation);
782 }
783 if (candidates.size() != 1) {
784 return ExpressionResult::failure(
785 CasErrc::Inconclusive,
786 "inverse derivative requires a unique inverse branch",
787 operation);
788 }
789
790 auto f_prime = f->differentiate(var);
791 if (!f_prime || !lamina::detail::node(f_prime)) {
792 return ExpressionResult::failure(
793 CasErrc::Inconclusive,
794 "inverse derivative could not construct the derivative",
795 operation);
796 }
797
798 auto f_prime_at_x0 = f_prime->substitute(var, candidates[0]);
799 auto simplified_derivative = f_prime_at_x0 ? f_prime_at_x0->simplify() : nullptr;
800 if (!simplified_derivative || !lamina::detail::node(simplified_derivative)) {
801 return ExpressionResult::failure(
802 CasErrc::InternalInvariant,
803 "inverse derivative substitution produced a null expression",
804 operation);
805 }
806
807 if (simplified_derivative->is_zero()) {
808 return ExpressionResult::failure(
809 CasErrc::DomainError,
810 "inverse derivative is undefined where f' is zero",
811 operation);
812 }
813
814 auto result = SymbolicExpr::divide(SymbolicExpr::number(1), simplified_derivative);
815 auto simplified = result ? result->simplify() : nullptr;
816 if (!simplified || !lamina::detail::node(simplified)) {
817 return ExpressionResult::failure(
818 CasErrc::InternalInvariant,
819 "inverse derivative result construction failed",
820 operation);
821 }
822 return ExpressionResult::success(simplified);
823 } catch (const std::bad_alloc&) {
824 return ExpressionResult::failure(CasErrc::ResourceLimit,
825 "inverse derivative allocation failed",
826 operation);
827 } catch (const std::exception& e) {
828 return ExpressionResult::failure(CasErrc::InternalInvariant,
829 e.what(), operation);
830 }
831}

938static bool calculus_utils_contains_unevaluated_integral(
939 const std::shared_ptr<const SymbolicNode>& node,
940 std::size_t depth = 0)
941{
942 if (!node || depth > 200) return false;
943 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(node)) {
944 if (func->type() == FunctionNode::FuncType::Calculus_Integral) return true;
945 for (const auto& arg : func->arguments()) {
946 if (calculus_utils_contains_unevaluated_integral(arg, depth + 1)) {
947 return true;
948 }
949 }
950 return false;
951 }
952 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
953 for (const auto& op : add->operands()) {
954 if (calculus_utils_contains_unevaluated_integral(op, depth + 1)) {
955 return true;
956 }
957 }
958 return false;
959 }
960 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
961 for (const auto& op : mul->operands()) {
962 if (calculus_utils_contains_unevaluated_integral(op, depth + 1)) {
963 return true;
964 }
965 }
966 return false;
967 }
968 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
969 return calculus_utils_contains_unevaluated_integral(pow->base(), depth + 1) ||
970 calculus_utils_contains_unevaluated_integral(pow->exponent(), depth + 1);
971 }
972 if (auto complex = std::dynamic_pointer_cast<const ComplexNode>(node)) {
973 return calculus_utils_contains_unevaluated_integral(complex->real(), depth + 1) ||
974 calculus_utils_contains_unevaluated_integral(complex->imag(), depth + 1);
975 }
976 return false;
977}

979/// 尝试符号定积分，若结果仍含未求值积分节点则返回 nullptr
980static std::shared_ptr<SymbolicExpr> calculus_utils_try_symbolic_definite(
981 const std::shared_ptr<SymbolicExpr>& integrand,
982 const std::string& var,
983 const std::shared_ptr<SymbolicExpr>& a,
984 const std::shared_ptr<SymbolicExpr>& b)
985{
986 Integrator integrator;
987 SymbolicExpr result = integrator.integrate_def(*integrand, var, *a, *b);
988
989 if (calculus_utils_contains_unevaluated_integral(lamina::detail::node(result))) return nullptr;
990 auto res = lamina::detail::make_expression_ptr(result);
991 auto simplified = res->simplify();
992 if (simplified &&
993 calculus_utils_contains_unevaluated_integral(lamina::detail::node(simplified))) {
994 return nullptr;
995 }
996 return simplified ? simplified : res;
997}

999/// 数值定积分回退（复合 Simpson 法）
1000static std::shared_ptr<SymbolicExpr> calculus_utils_numerical_definite(
1001 const std::shared_ptr<SymbolicExpr>& integrand,
1002 const std::string& var,
1003 const std::shared_ptr<SymbolicExpr>& a,
1004 const std::shared_ptr<SymbolicExpr>& b)
1005{
1006 ComputationContext context;
1007 auto a_eval = evaluate_numeric(*a, NumericBindings{}, context);
1008 auto b_eval = evaluate_numeric(*b, NumericBindings{}, context);
1009 if (!a_eval || !b_eval ||
1010 !a_eval.value().is_finite() || !b_eval.value().is_finite() ||
1011 !std::isfinite(a_eval.value().value) ||
1012 !std::isfinite(b_eval.value().value)) {
1013 return nullptr;
1014 }
1015 double a_val = a_eval.value().value;
1016 double b_val = b_eval.value().value;
1017
1018 int n = 1000;
1019 double h = (b_val - a_val) / n;
1020 if (!std::isfinite(h)) return nullptr;
1021 double sum = 0.0;
1022
1023 for (int i = 0; i <= n; ++i) {
1024 double xi = a_val + i * h;
1025 auto xi_expr = SymbolicExpr::number(xi);
1026 auto fi = integrand->substitute(var, xi_expr);
1027 if (!fi || !lamina::detail::node(fi)) return nullptr;
1028 auto fi_eval = evaluate_numeric(*fi, NumericBindings{}, context);
1029 if (!fi_eval || !fi_eval.value().is_finite() ||
1030 !std::isfinite(fi_eval.value().value)) {
1031 return nullptr;
1032 }
1033 double fi_val = fi_eval.value().value;
1034
1035 double weight = 1.0;
1036 if (i == 0 || i == n) {
1037 weight = 1.0;
1038 } else if (i % 2 == 1) {
1039 weight = 4.0;
1040 } else {
1041 weight = 2.0;
1042 }
1043 sum += weight * fi_val;
1044 if (!std::isfinite(sum)) return nullptr;
1045 }
1046 sum *= h / 3.0;
1047 if (!std::isfinite(sum)) return nullptr;
1048
1049 return SymbolicExpr::number(sum);
1050}

1053static std::shared_ptr<SymbolicExpr> calculus_utils_integrate_with_fallback(
1054 const std::shared_ptr<SymbolicExpr>& integrand,

1067ExpressionResult curvature_checked(
1068 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
1069 ComputationContext& context)
1070{
1071 const std::string operation = "curvature";
1072 auto input = calculus_utils_validate_expr(f, var, context, operation);
1073 if (!input) return ExpressionResult::failure(input.error());
1074
1075 /// f' = df/dvar
1076 auto f_prime = f->differentiate(var);
1077 if (!f_prime || !lamina::detail::node(f_prime)) {
1078 return ExpressionResult::failure(CasErrc::UnsupportedExpression,
1079 "first derivative could not be constructed",
1080 operation);
1081 }
1082 /// f'' = d²f/dvar²
1083 auto f_double_prime = f_prime->differentiate(var);
1084 if (!f_double_prime || !lamina::detail::node(f_double_prime)) {
1085 return ExpressionResult::failure(CasErrc::UnsupportedExpression,
1086 "second derivative could not be constructed",
1087 operation);
1088 }
1089
1090 /// |f''|
1091 auto abs_f_pp = calculus_utils_make_abs(f_double_prime);
1092 if (!abs_f_pp || !lamina::detail::node(abs_f_pp)) {
1093 return ExpressionResult::failure(CasErrc::InternalInvariant,
1094 "absolute value node construction failed",
1095 operation);
1096 }
1097
1098 /// 1 + f'²
1099 auto f_prime_sq = SymbolicExpr::power(f_prime, SymbolicExpr::number(2));
1100 auto one_plus_fp_sq = SymbolicExpr::add(SymbolicExpr::number(1), f_prime_sq);
1101
1102 /// (1 + f'²)^(3/2)
1103 auto three_half = SymbolicExpr::number(Rational(3, 2));
1104 auto denom = SymbolicExpr::power(one_plus_fp_sq, three_half);
1105
1106 /// κ = |f''| / (1 + f'²)^(3/2)
1107 auto result = SymbolicExpr::divide(abs_f_pp, denom);
1108 auto simplified = result->simplify();
1109 return ExpressionResult::success(simplified ? simplified : result);
1110}

1127ExpressionResult curvature_parametric_checked(
1128 const std::shared_ptr<SymbolicExpr>& x_t,
1129 const std::shared_ptr<SymbolicExpr>& y_t, const std::string& t,
1130 ComputationContext& context)
1131{
1132 const std::string operation = "curvature_parametric";
1133 auto input = calculus_utils_validate_two_exprs(x_t, y_t, t, context, operation);
1134 if (!input) return ExpressionResult::failure(input.error());
1135
1136 /// x' = dx/dt, x'' = d²x/dt²
1137 auto x_prime = x_t->differentiate(t);
1138 if (!x_prime || !lamina::detail::node(x_prime)) {
1139 return ExpressionResult::failure(CasErrc::UnsupportedExpression,
1140 "x derivative could not be constructed",
1141 operation);
1142 }
1143 auto x_double_prime = x_prime->differentiate(t);
1144 if (!x_double_prime || !lamina::detail::node(x_double_prime)) {
1145 return ExpressionResult::failure(CasErrc::UnsupportedExpression,
1146 "x second derivative could not be constructed",
1147 operation);
1148 }
1149
1150 /// y' = dy/dt, y'' = d²y/dt²
1151 auto y_prime = y_t->differentiate(t);
1152 if (!y_prime || !lamina::detail::node(y_prime)) {
1153 return ExpressionResult::failure(CasErrc::UnsupportedExpression,
1154 "y derivative could not be constructed",
1155 operation);
1156 }
1157 auto y_double_prime = y_prime->differentiate(t);
1158 if (!y_double_prime || !lamina::detail::node(y_double_prime)) {
1159 return ExpressionResult::failure(CasErrc::UnsupportedExpression,
1160 "y second derivative could not be constructed",
1161 operation);
1162 }
1163
1164 if (x_prime->is_zero() && y_prime->is_zero()) {
1165 return ExpressionResult::failure(CasErrc::DomainError,
1166 "parametric curvature is undefined for zero velocity",
1167 operation);
1168 }
1169
1170 /// x'y'' - y'x''
1171 auto cross_term = SymbolicExpr::add(
1172 SymbolicExpr::multiply(x_prime, y_double_prime),
1173 SymbolicExpr::multiply(
1174 SymbolicExpr::number(-1),
1175 SymbolicExpr::multiply(y_prime, x_double_prime)));
1176
1177 /// |x'y'' - y'x''|
1178 auto abs_cross = calculus_utils_make_abs(cross_term);
1179 if (!abs_cross || !lamina::detail::node(abs_cross)) {
1180 return ExpressionResult::failure(CasErrc::InternalInvariant,
1181 "absolute value node construction failed",
1182 operation);
1183 }
1184
1185 /// x'² + y'²
1186 auto x_prime_sq = SymbolicExpr::power(x_prime, SymbolicExpr::number(2));
1187 auto y_prime_sq = SymbolicExpr::power(y_prime, SymbolicExpr::number(2));
1188 auto sum_sq = SymbolicExpr::add(x_prime_sq, y_prime_sq);
1189
1190 /// (x'² + y'²)^(3/2)
1191 auto three_half = SymbolicExpr::number(Rational(3, 2));
1192 auto denom = SymbolicExpr::power(sum_sq, three_half);
1193
1194 /// κ = |x'y'' - y'x''| / (x'² + y'²)^(3/2)
1195 auto result = SymbolicExpr::divide(abs_cross, denom);
1196 auto simplified = result->simplify();
1197 return ExpressionResult::success(simplified ? simplified : result);
1198}

1275 if (simplified && lamina::detail::node(simplified) && simplified->is_zero()) {
1276 verified.push_back(candidate);
1277 }
1278 }
1279 if (!verified.empty()) {
1280 return SymbolicExprVectorResult::success(std::move(verified));
1281 }
1282 }
1283
1284 return SymbolicExprVectorResult::failure(
1285 CasErrc::Inconclusive,
