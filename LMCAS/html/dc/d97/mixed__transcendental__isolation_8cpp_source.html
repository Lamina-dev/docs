26 bool initialize(const SearchInterval& interval, ComputationContext* context,
27 std::optional<CasError>* failure, bool* complete) {
28 const auto step = (interval.hi - interval.lo) / initial_divisions;
29 std::vector<lmmc_real_t> samples(initial_divisions + 1);
30 for (int i = 0; i <= initial_divisions; ++i) {
31 if (!consume_mixed_step(context, failure, complete)) {
32 return false;
33 }
34 samples[i] = sample(interval.lo + i * step, step * 0.5);
35 if (samples[i] == 0.0) {
36 const auto x = interval.lo + i * step;
37 if (mixed_evaluate_at(expr_, var_, x) == 0.0)
38 result_.push_back({x, x, true});
39 }
40 }
41 for (int i = 0; i < initial_divisions; ++i) {
42 const auto fa = samples[i];
43 const auto fb = samples[i + 1];
44 if (std::isnan(fa) && std::isnan(fb)) {
45 continue;
46 }
47 if (std::isnan(fa) || std::isnan(fb) || fa * fb < 0.0) {
48 work_.push_back({interval.lo + i * step,
49 interval.lo + (i + 1) * step, fa, fb});
50 }
51 }
52 return true;
53 }
54
55 void run(const SolveOptions& opts, ComputationContext* context,
56 std::optional<CasError>* failure, bool* complete) {
57 while (!work_.empty()) {
58 if (!consume_mixed_step(context, failure, complete)) {
59 break;
60 }
61 if (opts.max_roots > 0 &&
62 result_.size() >= static_cast<std::size_t>(opts.max_roots)) {
63 if (complete) {
64 *complete = false;
65 }
66 break;
67 }
68 auto current = work_.back();
69 work_.pop_back();
70 process(current);
71 }
72 }

74 std::vector<IsolatedInterval> finish(int max_roots, bool* complete) {
75 std::sort(result_.begin(), result_.end(),
76 [](const IsolatedInterval& a, const IsolatedInterval& b) {
77 return a.lo < b.lo;
78 });
79 if (max_roots > 0 && result_.size() > static_cast<std::size_t>(max_roots)) {
80 if (complete) {
81 *complete = false;
82 }
83 result_.resize(static_cast<std::size_t>(max_roots));
84 }
85 return std::move(result_);

97 bool resample_endpoints(SampledInterval& interval) {
98 const auto width = interval.hi - interval.lo;
99 if (std::isnan(interval.f_lo)) {
100 interval.f_lo = sample(interval.lo, width * 0.25);
101 if (std::isnan(interval.f_lo)) {
102 if (width > minimum_width * 2.0) {
103 const auto mid = (interval.lo + interval.hi) * 0.5;
104 const auto fm = sample(mid, width * 0.125);
105 work_.push_back({mid, interval.hi, fm, interval.f_hi});
106 }
107 return false;
108 }
109 }
110 if (std::isnan(interval.f_hi)) {
111 interval.f_hi = sample(interval.hi, width * 0.25);
112 if (std::isnan(interval.f_hi)) {
113 if (width > minimum_width * 2.0) {
114 const auto mid = (interval.lo + interval.hi) * 0.5;
115 const auto fm = sample(mid, width * 0.125);
116 work_.push_back({interval.lo, mid, interval.f_lo, fm});
117 }
118 return false;
119 }
120 }
121 return true;
122 }

124 void subdivide(const SampledInterval& interval) {
125 const auto mid = (interval.lo + interval.hi) * 0.5;
126 const auto fm = sample(mid, (interval.hi - interval.lo) * 0.125);
127 if (std::isnan(fm)) {
128 accept(interval, false);
129 return;
130 }
131 if (fm == 0.0 && mixed_evaluate_at(expr_, var_, mid) == 0.0) {
132 result_.push_back({mid, mid, true});
133 }
134 if (interval.f_lo * fm < 0.0) {
135 work_.push_back({interval.lo, mid, interval.f_lo, fm});
136 }
137 if (fm * interval.f_hi < 0.0) {
138 work_.push_back({mid, interval.hi, fm, interval.f_hi});
139 }
140 if (interval.f_lo * fm >= 0.0 && fm * interval.f_hi >= 0.0) {
141 accept(interval, false);
142 }
143 }
144
145 void process(const SampledInterval& current) {
146 auto interval = current;
147 if (!resample_endpoints(interval) || interval.f_lo * interval.f_hi >= 0.0) {
148 return;
149 }
150 const auto width = interval.hi - interval.lo;
151 if (!derivative_) {
152 if (width > minimum_width * 4.0) {
153 subdivide(interval);
154 } else {
155 accept(interval, false);
156 }
157 return;
158 }
159 const auto da = mixed_evaluate_at(derivative_, var_, interval.lo);
160 const auto db = mixed_evaluate_at(derivative_, var_, interval.hi);
161 const bool finite = !std::isnan(da) && !std::isnan(db);
162 const bool monotone = finite && da * db > 0.0;
163 if (width <= minimum_width || monotone) {
164 accept(interval, monotone);
165 return;
166 }
167 if (finite && da * db < 0.0 && width > minimum_width * 2.0) {
168 subdivide(interval);
169 } else {
170 accept(interval, false);
171 }
172 }
