31BerlekampResult berlekamp_factor(
32 const Polynomial<Rational>& poly,
33 int64_t prime) {
34
35 BerlekampResult result;
36
37 /// 零多项式或常数多项式:无因子
38 if (poly.is_zero() || poly.degree() <= 0) {
39 return result;
40 }
41
42 /// 显式和自动模数都必须保持次数且使约化多项式无平方。
43 int64_t p = prime;
44 if (p > 0) {
45 if (!bk_is_suitable_prime(poly, p)) {
46 return result;
47 }
48 } else {
49 p = bk_select_prime(poly);
50 if (p < 0) {
51 return result;
52 }
53 }
54
55 result.prime = p;
56
57 /// 将多项式约化到 F_p
58 std::vector<int64_t> f_coeffs = bk_reduce_to_mod_coeffs(poly, p);
59
60 /// 首一化
61 if (!f_coeffs.empty()) {
62 int64_t lc = f_coeffs.back();
63 if (lc != 1) {
64 int64_t lc_inv = bk_mod_inverse(lc, p);
65 for (auto& c : f_coeffs) {
66 c = (c * lc_inv) % p;
67 }
68 }
69 }
70
71 /// 线性多项式:本身不可约
72 if (f_coeffs.size() <= 2) {
73 result.factors.push_back(bk_to_poly_modint(f_coeffs, p, poly.variable_name));
74 return result;
75 }
76
77 /// 构造 Berlekamp 矩阵 Q
78 auto Q = bk_build_q_matrix(f_coeffs, p);
79
80 /// 计算 (Q - I)^T 的零空间
81 auto null_basis = bk_null_space(Q, p);
82 int null_dim = static_cast<int>(null_basis.size());
83
84 /// 存储零空间信息
85 result.null_space_dim = null_dim;
86 result.null_space_basis = null_basis;
87
88 /// 零空间维度 = 1:多项式在 F_p 上不可约
89 if (null_dim <= 1) {
90 result.factors.push_back(bk_to_poly_modint(f_coeffs, p, poly.variable_name));
91 return result;
92 }
93
94 /// 零空间维度 > 1:多项式可分解,使用基向量分裂
95 auto split = bk_split_factors(f_coeffs, null_basis, null_dim, p);
96 for (const auto& factor_coeffs : split) {
97 result.factors.push_back(bk_to_poly_modint(factor_coeffs, p, poly.variable_name));
98 }
99 return result;
100}
