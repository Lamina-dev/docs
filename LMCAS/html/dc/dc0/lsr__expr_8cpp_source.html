116Result<ApproxComplex> add_complex(const ApproxComplex& lhs,
117 const ApproxComplex& rhs) {
118 auto result = checked_complex(lhs.real.value + rhs.real.value,
119 lhs.imag.value + rhs.imag.value,
120 kEvalComplexOperation);
121 if (!result) return result;
122 result.value().real.absolute_error +=
123 lhs.real.absolute_error + rhs.real.absolute_error;
124 result.value().imag.absolute_error +=

129Result<ApproxComplex> multiply_complex(const ApproxComplex& lhs,
130 const ApproxComplex& rhs) {
131 const double a = lhs.real.value;
132 const double b = lhs.imag.value;
133 const double c = rhs.real.value;
134 const double d = rhs.imag.value;
135 auto result = checked_complex(a * c - b * d, a * d + b * c,
136 kEvalComplexOperation);
137 if (!result) return result;
138 result.value().real.absolute_error +=
139 std::abs(c) * lhs.real.absolute_error +
140 std::abs(a) * rhs.real.absolute_error +
141 std::abs(d) * lhs.imag.absolute_error +
142 std::abs(b) * rhs.imag.absolute_error;
143 result.value().imag.absolute_error +=
144 std::abs(d) * lhs.real.absolute_error +
145 std::abs(a) * rhs.imag.absolute_error +
146 std::abs(c) * lhs.imag.absolute_error +
147 std::abs(b) * rhs.real.absolute_error;
148 return result;
149}

151Result<ApproxComplex> divide_complex(const ApproxComplex& lhs,
152 const ApproxComplex& rhs) {
153 const double c = rhs.real.value;
154 const double d = rhs.imag.value;
155 const double denom = c * c + d * d;
156 if (denom == 0.0 || !std::isfinite(denom)) {
157 return complex_failure(CasErrc::DomainError,
158 "complex division by zero or overflow",
159 kEvalComplexOperation);
160 }
161 return checked_complex((lhs.real.value * c + lhs.imag.value * d) / denom,
162 (lhs.imag.value * c - lhs.real.value * d) / denom,
163 kEvalComplexOperation);
164}

421 ExprResult parse_primary() {
422 skip_space();
423 if (match('(')) {
424 auto inner = parse_logical_or();
425 if (!inner) return inner;
426 if (match(',')) {
427 auto rhs = parse_logical_or();
428 if (!rhs) return rhs;
429 if (!match(')')) return fail("expected ')' after interval");
430 return interval(inner.value(), rhs.value(), false, false);
431 }
432 if (!match(')')) return fail("expected ')'");
433 return inner;
434 }
435 if (match('{')) {
436 return parse_set_literal();
437 }
438 if (match('[')) {
439 return parse_interval_literal(true);

442 return parse_number();
443 }

482 skip_space();
483 const std::size_t start = pos_;
484 bool saw_digit = false;
485 while (std::isdigit(static_cast<unsigned char>(peek()))) {
486 saw_digit = true;
487 ++pos_;
488 }
489 if (peek() == '.') {
490 ++pos_;
491 while (std::isdigit(static_cast<unsigned char>(peek()))) {
492 saw_digit = true;
493 ++pos_;
494 }
495 }
496 if (!saw_digit) return fail("invalid number literal");
497 if (peek() == 'e' || peek() == 'E') {
498 ++pos_;
499 if (peek() == '+' || peek() == '-') ++pos_;

502 saw_exp_digit = true;
503 ++pos_;
504 }
505 if (!saw_exp_digit) return fail("invalid number exponent");
506 }
507 try {
508 return rational(Rational(source_.substr(start, pos_ - start)));
509 } catch (const std::bad_alloc&) {
510 return expression_failure(CasErrc::ResourceLimit,

515 kParseOperation);

707 ExprResult apply_function(const std::string& name,
708 const std::vector<ExprPtr>& arguments) {
709 if (name == "sin" && arguments.size() == 1) return sin(arguments[0]);
710 if (name == "cos" && arguments.size() == 1) return cos(arguments[0]);
711 if (name == "tan" && arguments.size() == 1) return tan(arguments[0]);
712 if (name == "asin" && arguments.size() == 1) return asin(arguments[0]);
713 if (name == "acos" && arguments.size() == 1) return acos(arguments[0]);
714 if (name == "atan" && arguments.size() == 1) return atan(arguments[0]);
715 if (name == "sqrt" && arguments.size() == 1) return sqrt(arguments[0]);
716 if (name == "exp" && arguments.size() == 1) return exp(arguments[0]);

746 if (name == "ceil" && arguments.size() == 1) return ceil(arguments[0]);
747 if (name == "round" && arguments.size() == 1) return round(arguments[0]);
748 if (name == "abs" && arguments.size() == 1) return abs(arguments[0]);
749 if (name == "real" && arguments.size() == 1) return real(arguments[0]);
750 if (name == "imag" && arguments.size() == 1) return imag(arguments[0]);
751 if (name == "conj" && arguments.size() == 1) return conj(arguments[0]);
752 if (name == "max" && !arguments.empty()) {

784 } catch (const std::bad_alloc&) {

816Result<bool> domain_contains_node(
817 NumberDomain domain,
818 const std::shared_ptr<const SymbolicNode>& node) {
819 if (!node) {
820 return bool_failure(CasErrc::InvalidArgument,
821 "domain membership element must not be null",
822 kNumberDomainOperation);
823 }
824 if (domain == NumberDomain::Expressions) {
825 return Result<bool>::success(true);
826 }
827
828 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(node)) {
829 if (is_imaginary_unit_name(variable->name())) {
830 return Result<bool>::success(domain == NumberDomain::Complexes);
831 }
832 }
833
834 if (auto number = std::dynamic_pointer_cast<const NumberNode>(node)) {
835 const auto& value = number->value();
836 if (std::holds_alternative<BigInt>(value)) {
837 return Result<bool>::success(true);
838 }
839 if (std::holds_alternative<Rational>(value)) {
840 const auto& rational = std::get<Rational>(value);
841 switch (domain) {
842 case NumberDomain::Integers:
843 return Result<bool>::success(rational.is_integer());
844 case NumberDomain::Rationals:
845 case NumberDomain::Reals:
846 case NumberDomain::Complexes:
847 case NumberDomain::Expressions:
848 return Result<bool>::success(true);
849 }
850 }
851 const lmmc_real_t real = std::get<lmmc_real_t>(value);
852 if (!std::isfinite(static_cast<double>(real))) {
853 return bool_failure(CasErrc::NumericFailure,
854 "domain membership requires finite numeric literals",
855 kNumberDomainOperation);
856 }
857 switch (domain) {
858 case NumberDomain::Integers:
859 case NumberDomain::Rationals:
860 return Result<bool>::success(false);
861 case NumberDomain::Reals:
862 case NumberDomain::Complexes:
863 case NumberDomain::Expressions:
864 return Result<bool>::success(true);
865 }
866 }
867
868 if (auto complex_node = std::dynamic_pointer_cast<const ComplexNode>(node)) {
869 if (domain == NumberDomain::Complexes) {
870 auto real_part =
871 domain_contains_node(NumberDomain::Reals, complex_node->real());
872 if (!real_part) return real_part;
873 auto imag_part =
874 domain_contains_node(NumberDomain::Reals, complex_node->imag());
875 if (!imag_part) return imag_part;
876 return Result<bool>::success(real_part.value() && imag_part.value());
877 }
878
879 if (!complex_node->imag()->is_zero()) {
880 return Result<bool>::success(false);
881 }
882 return domain_contains_node(domain, complex_node->real());
883 }
884
885 if (domain == NumberDomain::Complexes) {
886 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
887 for (const auto& operand : add->operands()) {
888 auto member = domain_contains_node(domain, operand);
889 if (!member || !member.value()) return member;
890 }
891 return Result<bool>::success(true);
892 }
893
894 if (auto multiply =
895 std::dynamic_pointer_cast<const MultiplyNode>(node)) {
896 for (const auto& operand : multiply->operands()) {
897 auto member = domain_contains_node(domain, operand);
898 if (!member || !member.value()) return member;
899 }
900 return Result<bool>::success(true);
901 }
902
903 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
904 if (!exact_small_integer_node(power->exponent(), 1, 64)) {
905 return bool_failure(
906 CasErrc::Inconclusive,
907 "complex domain membership for powers requires a positive exact integer exponent",
908 kNumberDomainOperation);
909 }
910 auto base_member = domain_contains_node(domain, power->base());
911 if (!base_member) return base_member;
912 return Result<bool>::success(base_member.value());
913 }
914 }
915
916 return bool_failure(CasErrc::Inconclusive,
917 "domain membership is only decidable for numeric and explicit complex expressions",
918 kNumberDomainOperation);
919}

1007ExprPtr rewrite_trig_basic_identity(const std::shared_ptr<const SymbolicNode>& node) {
1008 if (!node) return nullptr;
1009
1010 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
1011 std::vector<std::shared_ptr<const SymbolicNode>> rewritten;
1012 rewritten.reserve(add->operands().size());
1013 for (const auto& operand : add->operands()) {
1014 auto child = rewrite_trig_basic_identity(operand);
1015 rewritten.push_back(lamina::detail::node(child));
1016 }
1017
1018 std::vector<bool> used(rewritten.size(), false);
1019 std::vector<std::shared_ptr<const SymbolicNode>> result_nodes;
1020 for (std::size_t i = 0; i < rewritten.size(); ++i) {
1021 if (used[i]) continue;
1022 std::shared_ptr<const SymbolicNode> sin_arg;
1023 std::shared_ptr<const SymbolicNode> cos_arg;
1024 const bool is_sin_square = trig_square_argument(
1025 rewritten[i], FunctionNode::FuncType::Sin, sin_arg);
1026 const bool is_cos_square = trig_square_argument(
1027 rewritten[i], FunctionNode::FuncType::Cos, cos_arg);
1028 bool matched = false;
1029 for (std::size_t j = i + 1; j < rewritten.size(); ++j) {
1030 if (used[j]) continue;
1031 std::shared_ptr<const SymbolicNode> other_arg;
1032 if (is_sin_square &&
1033 trig_square_argument(rewritten[j],
1034 FunctionNode::FuncType::Cos,
1035 other_arg) &&
1036 sin_arg->equals(*other_arg)) {
1037 matched = true;
1038 } else if (is_cos_square &&
1039 trig_square_argument(rewritten[j],
1040 FunctionNode::FuncType::Sin,
1041 other_arg) &&
1042 cos_arg->equals(*other_arg)) {
1043 matched = true;
1044 }
1045 if (matched) {
1046 used[i] = true;
1047 used[j] = true;
1048 result_nodes.push_back(
1049 lamina::detail::node(SymbolicExpr::number(1)));
1050 break;
1051 }
1052 }
1053 if (!matched && !used[i]) {
1054 result_nodes.push_back(rewritten[i]);
1055 }
1056 }
1057
1058 if (result_nodes.empty()) {
1059 return SymbolicExpr::number(0);
1060 }
1061 return lamina::detail::make_expression_ptr(
1062 SymbolicFactory::create_add(std::move(result_nodes)))->simplify();
1063 }
1064
1065 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
1066 std::vector<std::shared_ptr<const SymbolicNode>> operands;
1067 operands.reserve(multiply->operands().size());
1068 for (const auto& operand : multiply->operands()) {
1069 auto child = rewrite_trig_basic_identity(operand);
1070 operands.push_back(lamina::detail::node(child));
1071 }
1072 return lamina::detail::make_expression_ptr(
1073 SymbolicFactory::create_multiply(std::move(operands)))->simplify();
1074 }
1075
1076 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
1077 auto base = rewrite_trig_basic_identity(power->base());
1078 auto exponent = rewrite_trig_basic_identity(power->exponent());
1079 return SymbolicExpr::power(base, exponent)->simplify();
1080 }
1081
1082 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(node)) {
1083 std::vector<std::shared_ptr<const SymbolicNode>> args;
1084 args.reserve(function->arguments().size());
1085 for (const auto& argument : function->arguments()) {
1086 auto child = rewrite_trig_basic_identity(argument);
1087 args.push_back(lamina::detail::node(child));
1088 }
1089
1090 if (args.size() == 1) {
1091 auto positive_arg = unwrap_trig_negated_argument(args[0]);
1092 if (positive_arg && *positive_arg) {
1093 if (function->type() == FunctionNode::FuncType::Sin) {
1094 return SymbolicExpr::multiply(
1095 SymbolicExpr::number(-1),
1096 SymbolicExpr::sin(*positive_arg))->simplify();
1097 }
1098 if (function->type() == FunctionNode::FuncType::Cos) {
1099 return SymbolicExpr::cos(*positive_arg)->simplify();
1100 }
1101 }
1102 }
1103
1104 return lamina::detail::make_expression_ptr(
1105 lamina::detail::make_node<FunctionNode>(function->type(),
1106 std::move(args)))->simplify();
1107 }
1108
1109 if (auto complex_node = std::dynamic_pointer_cast<const ComplexNode>(node)) {
1110 auto real_part = rewrite_trig_basic_identity(complex_node->real());
1111 auto imag_part = rewrite_trig_basic_identity(complex_node->imag());
1112 auto value = complex(real_part, imag_part);
1113 if (!value) throw std::runtime_error(value.error().message);
1114 return value.value()->simplify();
1115 }
1116
1117 return lamina::detail::make_expression_ptr(node);
1118}

1332 if (variable.empty()) {
1333 return Result<std::optional<ExprSet>>::failure(
1334 CasErrc::InvalidArgument, "solve variable cannot be empty",
1335 kSolveExprSetOperation);
1336 }
1337
1338 auto recognized = recognize_rational_polynomial(*equation, variable, context);
1339 if (!recognized) {
1340 return Result<std::optional<ExprSet>>::failure(recognized.error());
1341 }
1342 if (!recognized.value()) {
1343 return Result<std::optional<ExprSet>>::success(std::nullopt);
1344 }
1345
1346 const Polynomial<Rational>& polynomial = *recognized.value();
1347 const int degree = polynomial.degree();
1348 if (degree < 1 || degree > 2) {
1349 return Result<std::optional<ExprSet>>::success(std::nullopt);
1350 }
1351
1352 std::vector<ExprPtr> roots;
1353 if (degree == 1) {
1354 const Rational b = polynomial_coeff_or_zero(polynomial, 1);
1355 if (b.is_zero()) {
1356 return Result<std::optional<ExprSet>>::success(std::nullopt);
1357 }
1358 const Rational c = polynomial_coeff_or_zero(polynomial, 0);
1359 roots.push_back(rational_expression((-c) / b)->simplify());
1360 } else {
1361 const Rational a = polynomial_coeff_or_zero(polynomial, 2);
1362 const Rational b = polynomial_coeff_or_zero(polynomial, 1);
1363 const Rational c = polynomial_coeff_or_zero(polynomial, 0);
1364 if (a.is_zero()) {
1365 return Result<std::optional<ExprSet>>::success(std::nullopt);
1366 }
1367
1368 const Rational two_a = Rational(2) * a;
1369 const Rational discriminant = b * b - Rational(4) * a * c;
1370 const Rational real_component = (-b) / two_a;
1371
1372 if (discriminant < Rational(0)) {
1373 const Rational positive_discriminant = -discriminant;
1374 const Rational positive_denominator = two_a.abs();
1375 auto imag_magnitude = SymbolicExpr::divide(
1376 sqrt_rational_expression(positive_discriminant),
1377 rational_expression(positive_denominator))->simplify();
1378 auto positive = verified_lsr_complex(
1379 rational_expression(real_component), imag_magnitude);
1380 if (!positive) {
1381 return Result<std::optional<ExprSet>>::failure(positive.error());
1382 }
1383 auto negative_imag = SymbolicExpr::multiply(
1384 SymbolicExpr::number(-1), imag_magnitude)->simplify();
1385 auto negative = verified_lsr_complex(
1386 rational_expression(real_component), negative_imag);
1387 if (!negative) {
1388 return Result<std::optional<ExprSet>>::failure(negative.error());
1389 }
1390 roots.push_back(negative.value());
1391 roots.push_back(positive.value());
1392 } else {
1393 auto sqrt_discriminant = sqrt_rational_expression(discriminant);
1394 auto numerator_left = SymbolicExpr::add(
1395 rational_expression(-b),
1396 SymbolicExpr::multiply(SymbolicExpr::number(-1),
1397 sqrt_discriminant))->simplify();
1398 roots.push_back(SymbolicExpr::divide(
1399 numerator_left, rational_expression(two_a))->simplify());
1400 if (!discriminant.is_zero()) {
1401 auto numerator_right = SymbolicExpr::add(
1402 rational_expression(-b), sqrt_discriminant)->simplify();
1403 roots.push_back(SymbolicExpr::divide(
1404 numerator_right, rational_expression(two_a))->simplify());
1405 }
1406 }
1407 }
1408
1409 auto set = ExprSet::make(std::move(roots));
1410 if (!set) {
1411 return Result<std::optional<ExprSet>>::failure(set.error());
1412 }
1413 return Result<std::optional<ExprSet>>::success(std::move(set.value()));
1414}

1416ExprPtr canonicalize_lsr_complex_product(const SymbolicExpr& expression) {
1417 const auto& node = lamina::detail::node(expression);
1418
1419 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(node)) {
1420 if (is_imaginary_unit_name(variable->name())) {
1421 auto unit = imaginary_unit();
1422 if (!unit) throw std::runtime_error(unit.error().message);
1423 return unit.value();
1424 }
1425 return lamina::detail::make_expression_ptr(node);
1426 }
1427
1428 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
1429 std::vector<std::shared_ptr<const SymbolicNode>> operands;
1430 operands.reserve(add->operands().size());
1431 for (const auto& operand : add->operands()) {
1432 auto canonical_operand = canonicalize_lsr_complex_product(
1433 *lamina::detail::make_expression_ptr(operand));
1434 operands.push_back(lamina::detail::node(canonical_operand));
1435 }
1436 return lamina::detail::make_expression_ptr(
1437 SymbolicFactory::create_add(std::move(operands)))->simplify();
1438 }
1439
1440 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
1441 auto exponent = exact_small_integer_node(power->exponent(), 0, 16);
1442 if (exponent) {
1443 auto canonical_base = canonicalize_lsr_complex_product(
1444 *lamina::detail::make_expression_ptr(power->base()));
1445 if (std::dynamic_pointer_cast<const ComplexNode>(
1446 lamina::detail::node(canonical_base))) {
1447 auto result = SymbolicExpr::number(1);
1448 for (int i = 0; i < *exponent; ++i) {
1449 result = canonicalize_lsr_complex_product(
1450 *SymbolicExpr::multiply(result, canonical_base));
1451 }
1452 return result->simplify();
1453 }
1454 }
1455 return lamina::detail::make_expression_ptr(node);
1456 }
1457
1458 if (auto complex_node = std::dynamic_pointer_cast<const ComplexNode>(node)) {
1459 auto real_part = canonicalize_lsr_complex_product(
1460 *lamina::detail::make_expression_ptr(complex_node->real()));
1461 auto imag_part = canonicalize_lsr_complex_product(
1462 *lamina::detail::make_expression_ptr(complex_node->imag()));
1463 auto value = complex(real_part, imag_part);
1464 if (!value) throw std::runtime_error(value.error().message);
1465 return value.value()->simplify();
1466 }
1467
1468 auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node);
1469 if (!multiply) {
1470 return lamina::detail::make_expression_ptr(node);
1471 }
1472
1473 bool saw_complex = false;
1474 auto real = SymbolicExpr::number(1);
1475 auto imag = SymbolicExpr::number(0);
1476
1477 for (const auto& operand : multiply->operands()) {
1478 auto canonical_operand = canonicalize_lsr_complex_product(
1479 *lamina::detail::make_expression_ptr(operand));
1480 const auto& operand_node = lamina::detail::node(canonical_operand);
1481 ExprPtr factor_real;
1482 ExprPtr factor_imag;
1483 if (auto complex_operand = std::dynamic_pointer_cast<const ComplexNode>(
1484 operand_node)) {
1485 saw_complex = true;
1486 factor_real = lamina::detail::make_expression_ptr(complex_operand->real());
1487 factor_imag = lamina::detail::make_expression_ptr(complex_operand->imag());
1488 } else {
1489 factor_real = canonical_operand;
1490 factor_imag = SymbolicExpr::number(0);
1491 }
1492
1493 auto ac = SymbolicExpr::multiply(real, factor_real);
1494 auto bd = SymbolicExpr::multiply(imag, factor_imag);
1495 auto ad = SymbolicExpr::multiply(real, factor_imag);
1496 auto bc = SymbolicExpr::multiply(imag, factor_real);
1497 auto next_real = SymbolicExpr::add(
1498 ac, SymbolicExpr::multiply(SymbolicExpr::number(-1), bd))->simplify();
1499 auto next_imag = SymbolicExpr::add(ad, bc)->simplify();
1500 real = std::move(next_real);
1501 imag = std::move(next_imag);
1502 }
1503
1504 if (!saw_complex) {
1505 return lamina::detail::make_expression_ptr(node);
1506 }
1507 auto result = complex(real, imag);
1508 if (!result) {
1509 throw std::runtime_error(result.error().message);
1510 }
1511 return result.value()->simplify();
1512}

1514Result<ApproxComplex> evaluate_complex_node(
1515 const std::shared_ptr<const SymbolicNode>& node,
1516 const NumericBindings& bindings,
1517 ComputationContext& context) {
1518 auto step = context.consume_steps(1, kEvalComplexOperation);
1519 if (!step) return Result<ApproxComplex>::failure(step.error());
1520 if (!node) {
1521 return complex_failure(CasErrc::InvalidArgument,
1522 "expression contains a null node",
1523 kEvalComplexOperation);
1524 }
1525
1526 if (auto complex_node = std::dynamic_pointer_cast<const ComplexNode>(node)) {
1527 auto real = evaluate_numeric(
1528 *lamina::detail::make_expression_ptr(complex_node->real()),
1529 bindings, context);
1530 if (!real) return eval_complex_failure(real.error());
1531 auto imag = evaluate_numeric(
1532 *lamina::detail::make_expression_ptr(complex_node->imag()),
1533 bindings, context);
1534 if (!imag) return eval_complex_failure(imag.error());
1535 if (!real.value().is_finite() || !imag.value().is_finite()) {
1536 return complex_failure(CasErrc::NumericFailure,
1537 "complex components must be finite",
1538 kEvalComplexOperation);
1539 }
1540 return Result<ApproxComplex>::success(
1541 ApproxComplex{real.value(), imag.value()});
1542 }
1543
1544 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(node)) {
1545 if (is_imaginary_unit_name(variable->name())) {
1546 return Result<ApproxComplex>::success(approx_complex(0.0, 1.0));
1547 }
1548 }
1549
1550 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
1551 auto sum = Result<ApproxComplex>::success(approx_complex(0.0, 0.0));
1552 for (const auto& operand : add->operands()) {
1553 auto term = evaluate_complex_node(operand, bindings, context);
1554 if (!term) return term;
1555 sum = add_complex(sum.value(), term.value());
1556 if (!sum) return sum;
1557 }
1558 return sum;
1559 }
1560
1561 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
1562 auto product = Result<ApproxComplex>::success(approx_complex(1.0, 0.0));
1563 for (const auto& operand : multiply->operands()) {
1564 auto factor = evaluate_complex_node(operand, bindings, context);
1565 if (!factor) return factor;
1566 product = multiply_complex(product.value(), factor.value());
1567 if (!product) return product;
1568 }
1569 return product;
1570 }
1571
1572 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
1573 auto base = evaluate_complex_node(power->base(), bindings, context);
1574 if (!base) return base;
1575 auto exponent = evaluate_numeric(
1576 *lamina::detail::make_expression_ptr(power->exponent()),
1577 bindings, context);
1578 if (!exponent) return eval_complex_failure(exponent.error());
1579 if (!exponent.value().is_finite() ||
1580 !std::isfinite(exponent.value().value)) {
1581 return complex_failure(CasErrc::NumericFailure,
1582 "complex power exponent must be finite",
1583 kEvalComplexOperation);
1584 }
1585 const double exponent_value = exponent.value().value;
1586 if (!is_integer_double(exponent_value) ||
1587 std::abs(exponent_value) > 64.0) {
1588 return complex_failure(CasErrc::UnsupportedExpression,
1589 "complex evaluation only supports integer powers with |n| <= 64",
1590 kEvalComplexOperation);
1591 }
1592 int exponent_int = static_cast<int>(exponent_value);
1593 auto result = Result<ApproxComplex>::success(approx_complex(1.0, 0.0));
1594 for (int i = 0; i < std::abs(exponent_int); ++i) {
1595 result = multiply_complex(result.value(), base.value());
1596 if (!result) return result;
1597 }
1598 if (exponent_int < 0) {
1599 result = divide_complex(approx_complex(1.0, 0.0), result.value());
1600 }
1601 return result;
1602 }
1603
1604 return real_to_complex(
1605 evaluate_numeric(*lamina::detail::make_expression_ptr(node),
1606 bindings, context));
1607}

1965ExprSetResult solve_expr_set(const ExprPtr& equation,
1966 const std::string& variable,
1967 ComputationContext& context,
1968 const SolveOptions& options) {
1969 auto closed_form = try_lsr_closed_form_rational_poly_roots(
1970 equation, variable, context);
1971 if (!closed_form) {
1972 return ExprSetResult::failure(closed_form.error());
1973 }
1974 if (closed_form.value()) {
1975 return ExprSetResult::success(std::move(*closed_form.value()));
1976 }
1977
1978 auto solved = solve_set(equation, variable, context, options);
1979 if (!solved) {
1980 return ExprSetResult::failure(solved.error());
1981 }
1982
