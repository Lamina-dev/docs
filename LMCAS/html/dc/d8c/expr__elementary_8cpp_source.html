134ExprResult log10(const ExprPtr& expression, ComputationContext& context) {
135 auto valid = require_dimensionless(expression, "log10");
136 if (!valid) { return valid; }
137 auto step = context.consume_steps(1, kMathOperation);
138 if (!step) { return ExprResult::failure(step.error()); }
139 if (!expression || !LMCAS::detail::node(expression)) {
140 return expression_failure(CasErrc::InvalidArgument,
141 "log10 argument cannot be null",
142 kMathOperation);
143 }
144 try {
145 auto numerator = SymbolicExpr::ln(expression);
146 auto denominator = SymbolicExpr::ln(SymbolicExpr::number(10));
147 auto result = SymbolicExpr::divide(numerator, denominator);
148 if (!result || !LMCAS::detail::node(result)) {
149 return expression_failure(CasErrc::InternalInvariant,
150 "log10 expression construction failed",
151 kMathOperation);
152 }
153 return ExprResult::success(std::move(result));

199ExprResult clamp(const ExprPtr& expression,
200 const ExprPtr& lower,
201 const ExprPtr& upper,
202 ComputationContext& context) {
203 auto step = context.consume_steps(1, kMathOperation);
204 if (!step) { return ExprResult::failure(step.error()); }
205 if (!expression || !LMCAS::detail::node(expression) ||
206 !lower || !LMCAS::detail::node(lower) ||
207 !upper || !LMCAS::detail::node(upper)) {
208 return expression_failure(CasErrc::InvalidArgument,
209 "clamp arguments cannot be null",
210 kMathOperation);
211 }
212 try {
213 auto max_node = LMCAS::detail::make_node<FunctionNode>(
214 FunctionNode::FuncType::Max,
215 std::vector<std::shared_ptr<const SymbolicNode>>{
216 LMCAS::detail::node(expression), LMCAS::detail::node(lower)});
217 auto min_node = LMCAS::detail::make_node<FunctionNode>(
218 FunctionNode::FuncType::Min,
219 std::vector<std::shared_ptr<const SymbolicNode>>{
220 std::move(max_node), LMCAS::detail::node(upper)});
221 return ExprResult::success(
222 LMCAS::detail::make_expression_ptr(std::move(min_node)));
223 } catch (const std::bad_alloc&) {
224 return expression_failure(CasErrc::ResourceLimit,
225 "clamp expression allocation failed",
226 kMathOperation);
227 } catch (const std::exception& error) {
228 return expression_failure(CasErrc::InternalInvariant, error.what(),
229 kMathOperation);
230 }
