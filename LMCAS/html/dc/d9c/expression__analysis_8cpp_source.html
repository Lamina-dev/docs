34 void visit_binder(const SymbolicNode& node) {
35 const auto binder = detail::binder_view(node);
36 visit_children(binder->outside_scope);
37 ++bound_[binder->bound_name];
38 visit_child(binder->scoped_body);
39 auto found = bound_.find(binder->bound_name);
40 if (--found->second == 0) bound_.erase(found);
41 }

126class FreeSubstitution final : public detail::SymbolicRewriter {
127public:
128 FreeSubstitution(std::string target, detail::SymbolicNodePtr replacement,
129 std::set<std::string> replacement_free,
130 std::set<std::string> occupied, detail::RewriteBudget* budget)
131 : SymbolicRewriter(budget), target_(std::move(target)),
132 replacement_(std::move(replacement)),
133 replacement_free_(std::move(replacement_free)),
134 occupied_(std::move(occupied)) {}
135
136 void visit(const VariableNode& node) override {
137 set_result(!node.is_constant() && node.name() == target_ ? replacement_ : current());
138 }
139
140 void visit(const SummationNode& node) override {
141 bool changed = false;
142 auto lower = rewrite_child(node.lower_bound(), changed);
143 auto upper = rewrite_child(node.upper_bound(), changed);
144 auto binder = node.index_var();
145 auto body = rewrite_scoped(node.body(), binder, changed);
146 set_result(changed ? detail::make_node<SummationNode>(body, binder, lower, upper)
147 : current());
148 }
149
150 void visit(const ProductNode& node) override {
151 bool changed = false;
152 auto lower = rewrite_child(node.lower_bound(), changed);
153 auto upper = rewrite_child(node.upper_bound(), changed);
154 auto binder = node.index_var();
155 auto body = rewrite_scoped(node.body(), binder, changed);
156 set_result(changed ? detail::make_node<ProductNode>(body, binder, lower, upper)
157 : current());
158 }
159
160 void visit(const IntegralNode& node) override {
161 bool changed = false;
162 auto lower = node.lower() ? rewrite_child(node.lower(), changed) : nullptr;
163 auto upper = node.upper() ? rewrite_child(node.upper(), changed) : nullptr;
164 auto binder = node.variable();
165 auto body = rewrite_scoped(node.body(), binder, changed);
166 set_result(changed ? detail::make_node<IntegralNode>(body, binder, lower, upper)
167 : current());
168 }
169
170 void visit(const TransformNode& node) override {
171 bool changed = false;
172 auto target = rewrite_child(node.target(), changed);
173 auto binder = node.source_var();
174 auto body = rewrite_scoped(node.body(), binder, changed);
175 set_result(changed ? detail::make_node<TransformNode>(node.transform_type(), body,
176 binder, target)
177 : current());
178 }
179
180 void visit(const QuantifierNode& node) override {
181 bool changed = false;
182 auto domain = rewrite_child(node.domain(), changed);
183 auto binder = node.bound_var();
184 auto predicate = rewrite_scoped(node.predicate(), binder, changed);
185 set_result(changed ? detail::make_node<QuantifierNode>(node.quantifier_type(), binder,
186 domain, predicate)
187 : current());
188 }
189
190 void visit(const SetBuilderNode& node) override {
191 bool changed = false;
192 auto domain = rewrite_child(node.domain(), changed);
193 auto binder = node.element_var();
194 auto predicate = rewrite_scoped(node.predicate(), binder, changed);
195 set_result(changed ? detail::make_node<SetBuilderNode>(binder, domain, predicate)
196 : current());
197 }
198
199 void visit(const LimitNode& node) override {
200 bool changed = false;
201 auto point = rewrite_child(node.point(), changed);
202 auto binder = node.variable();
203 auto body = rewrite_scoped(node.body(), binder, changed);
204 set_result(changed ? detail::make_node<LimitNode>(
205 body, binder, point, node.direction())
206 : current());
207 }
208
209 void visit(const RootOfNode&) override { set_result(current()); }
210
211private:
212 detail::SymbolicNodePtr rewrite_scoped(
213 const detail::SymbolicNodePtr& original_body,
214 std::string& binder,
215 bool& changed) {
216 if (binder == target_) return original_body;
217
218 auto body = original_body;
219 if (expression_depends_on_variable(body, target_, rewrite_budget()) &&
220 replacement_free_.find(binder) != replacement_free_.end()) {
221 const auto renamed = fresh_name(binder, occupied_);
222 body = substitute_free(body, binder,
223 SymbolicFactory::create_variable(renamed), rewrite_budget());
224 binder = renamed;
225 changed = true;
226 }
227 return rewrite_child(body, changed);
228 }
229
230 std::string target_;
231 detail::SymbolicNodePtr replacement_;
232 const std::set<std::string> replacement_free_;
233 std::set<std::string> occupied_;
234};

255Result<std::optional<detail::AffineForm>> detail::recognize_affine(
256 const SymbolicExpr& expression, const std::string& variable,
257 ComputationContext& context) {
258 using AffineResult = Result<std::optional<AffineForm>>;
259 auto converted = symbolic_to_poly_checked(expression, variable, context);
260 if (!converted) {
261 if (converted.error().code == CasErrc::UnsupportedExpression) {
262 return std::optional<AffineForm>{};
263 }
264 return AffineResult::failure(converted.error());
265 }
266 const auto& polynomial = converted.value();
267 if (polynomial.degree() > 1) return std::optional<AffineForm>{};
268 return std::optional<AffineForm>{AffineForm{
269 polynomial.coeffs.size() > 1 ? polynomial.coeffs[1].val : SymbolicExpr::number(0),
270 polynomial.coeffs.empty() ? SymbolicExpr::number(0) : polynomial.coeffs[0].val}};
271}

273std::optional<int> exact_small_integer_node(
274 const detail::SymbolicNodePtr& node, int min_value, int max_value) {
275 auto number = std::dynamic_pointer_cast<const NumberNode>(node);
276 if (!number) return std::nullopt;
277
278 BigInt value;
279 if (std::holds_alternative<BigInt>(number->value())) {
280 value = std::get<BigInt>(number->value());
281 } else if (std::holds_alternative<Rational>(number->value())) {
282 const Rational& rational = std::get<Rational>(number->value());
283 if (!rational.is_integer()) return std::nullopt;
284 value = rational.to_bigint();
285 } else {
286 return std::nullopt;
287 }
288
289 const auto exact = value.try_to_int64();
290 if (!exact || *exact < min_value || *exact > max_value) {
291 return std::nullopt;
292 }
293 return static_cast<int>(*exact);
294}

320detail::SymbolicNodePtr substitute_free(
321 const detail::SymbolicNodePtr& expression,
322 const std::string& variable,
323 const detail::SymbolicNodePtr& replacement, detail::RewriteBudget* budget) {
324 if (!expression) return nullptr;
325 if (variable.empty()) throw std::invalid_argument("substitution variable cannot be empty");
326 if (!replacement) throw std::invalid_argument("substitution replacement cannot be null");
327
328 auto replacement_free = free_variables(replacement, budget);
329 auto occupied = all_names(expression, budget);
330 auto replacement_names = all_names(replacement, budget);
331 occupied.insert(replacement_names.begin(), replacement_names.end());
332 occupied.insert(variable);
333
334 FreeSubstitution substitution(variable, replacement,
335 std::move(replacement_free),
336 std::move(occupied), budget);
337 return substitution.rewrite(expression);
338}
