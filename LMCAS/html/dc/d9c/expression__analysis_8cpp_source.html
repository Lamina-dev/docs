122class FreeSubstitution final : public detail::SymbolicRewriter {
123public:
124 FreeSubstitution(std::string target, detail::SymbolicNodePtr replacement,
125 std::set<std::string> replacement_free,
126 std::set<std::string> occupied)
127 : target_(std::move(target)), replacement_(std::move(replacement)),
128 replacement_free_(std::move(replacement_free)),
129 occupied_(std::move(occupied)) {}
130
131 void visit(const VariableNode& node) override {
132 set_result(node.name() == target_ ? replacement_ : current());
133 }
134
135 void visit(const SummationNode& node) override {
136 bool changed = false;
137 auto lower = rewrite_child(node.lower_bound(), changed);
138 auto upper = rewrite_child(node.upper_bound(), changed);
139 auto binder = node.index_var();
140 auto body = rewrite_scoped(node.body(), binder, changed);
141 set_result(changed ? detail::make_node<SummationNode>(body, binder, lower, upper)
142 : current());
143 }
144
145 void visit(const ProductNode& node) override {
146 bool changed = false;
147 auto lower = rewrite_child(node.lower_bound(), changed);
148 auto upper = rewrite_child(node.upper_bound(), changed);
149 auto binder = node.index_var();
150 auto body = rewrite_scoped(node.body(), binder, changed);
151 set_result(changed ? detail::make_node<ProductNode>(body, binder, lower, upper)
152 : current());
153 }
154
155 void visit(const IntegralNode& node) override {
156 bool changed = false;
157 auto lower = node.lower() ? rewrite_child(node.lower(), changed) : nullptr;
158 auto upper = node.upper() ? rewrite_child(node.upper(), changed) : nullptr;
159 auto binder = node.variable();
160 auto body = rewrite_scoped(node.body(), binder, changed);
161 set_result(changed ? detail::make_node<IntegralNode>(body, binder, lower, upper)
162 : current());
163 }
164
165 void visit(const TransformNode& node) override {
166 bool changed = false;
167 auto target = rewrite_child(node.target(), changed);
168 auto binder = node.source_var();
169 auto body = rewrite_scoped(node.body(), binder, changed);
170 set_result(changed ? detail::make_node<TransformNode>(node.transform_type(), body,
171 binder, target)
172 : current());
173 }
174
175 void visit(const QuantifierNode& node) override {
176 bool changed = false;
177 auto domain = rewrite_child(node.domain(), changed);
178 auto binder = node.bound_var();
179 auto predicate = rewrite_scoped(node.predicate(), binder, changed);
180 set_result(changed ? detail::make_node<QuantifierNode>(node.quantifier_type(), binder,
181 domain, predicate)
182 : current());
183 }
184
185 void visit(const SetBuilderNode& node) override {
186 bool changed = false;
187 auto domain = rewrite_child(node.domain(), changed);
188 auto binder = node.element_var();
189 auto predicate = rewrite_scoped(node.predicate(), binder, changed);
190 set_result(changed ? detail::make_node<SetBuilderNode>(binder, domain, predicate)
191 : current());
192 }
193
194 void visit(const LimitNode& node) override {
195 bool changed = false;
196 auto point = rewrite_child(node.point(), changed);
197 auto binder = node.variable();
198 auto body = rewrite_scoped(node.body(), binder, changed);
199 set_result(changed ? detail::make_node<LimitNode>(
200 body, binder, point, node.direction())
201 : current());
202 }
203
204 void visit(const RootOfNode&) override { set_result(current()); }
205
206private:
207 detail::SymbolicNodePtr rewrite_scoped(
208 const detail::SymbolicNodePtr& original_body,
209 std::string& binder,
210 bool& changed) {
211 if (binder == target_) return original_body;
212
213 auto body = original_body;
214 if (expression_depends_on_variable(body, target_) &&
215 replacement_free_.find(binder) != replacement_free_.end()) {
216 const auto renamed = fresh_name(binder, occupied_);
217 body = substitute_free(body, binder,
218 SymbolicFactory::create_variable(renamed));
219 binder = renamed;
220 changed = true;
221 }
222 return rewrite_child(body, changed);
223 }
224
225 std::string target_;
226 detail::SymbolicNodePtr replacement_;
227 const std::set<std::string> replacement_free_;
228 std::set<std::string> occupied_;
229};

247detail::SymbolicNodePtr substitute_free(
248 const detail::SymbolicNodePtr& expression,
249 const std::string& variable,
250 const detail::SymbolicNodePtr& replacement) {
251 if (!expression) return nullptr;
252 if (variable.empty()) throw std::invalid_argument("substitution variable cannot be empty");
253 if (!replacement) throw std::invalid_argument("substitution replacement cannot be null");
254
255 auto replacement_free = free_variables(replacement);
256 auto occupied = all_names(expression);
257 auto replacement_names = all_names(replacement);
258 occupied.insert(replacement_names.begin(), replacement_names.end());
259 occupied.insert(variable);
260
261 FreeSubstitution substitution(variable, replacement,
262 std::move(replacement_free),
263 std::move(occupied));
264 return substitution.rewrite(expression);
265}
