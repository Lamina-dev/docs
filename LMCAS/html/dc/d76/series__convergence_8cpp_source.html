42static bool series_extract_alternating(const std::shared_ptr<const SymbolicNode>& node,
43 const std::string& n,
44 std::shared_ptr<SymbolicExpr>& remainder) {
45 auto pow = std::dynamic_pointer_cast<const PowerNode>(node);
46 if (pow) {
47 auto base_num = std::dynamic_pointer_cast<const NumberNode>(pow->base());
48 auto exp_var = std::dynamic_pointer_cast<const VariableNode>(pow->exponent());
49 if (base_num && exp_var && exp_var->name() == n) {
50 double base_val = 0.0;
51 if (std::holds_alternative<BigInt>(base_num->value()))
52 base_val = std::get<BigInt>(base_num->value()).to_double();
53 else if (std::holds_alternative<Rational>(base_num->value()))
54 base_val = std::get<Rational>(base_num->value()).to_double();
55 else
56 base_val = static_cast<double>(std::get<lmmc_real_t>(base_num->value()));
57 if (std::abs(base_val + 1.0) < 1e-12) {
58 remainder = SymbolicExpr::number(1);
59 return true;
60 }
61 }
62 }
63 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node);
64 if (mul) {
65 for (size_t i = 0; i < mul->operands().size(); ++i) {
66 auto pw = std::dynamic_pointer_cast<const PowerNode>(mul->operands()[i]);
67 if (!pw) continue;
68 auto base_num = std::dynamic_pointer_cast<const NumberNode>(pw->base());
69 auto exp_var = std::dynamic_pointer_cast<const VariableNode>(pw->exponent());
70 if (!base_num || !exp_var || exp_var->name() != n) continue;
71 double base_val = 0.0;
72 if (std::holds_alternative<BigInt>(base_num->value()))
73 base_val = std::get<BigInt>(base_num->value()).to_double();
74 else if (std::holds_alternative<Rational>(base_num->value()))
75 base_val = std::get<Rational>(base_num->value()).to_double();
76 else
77 base_val = static_cast<double>(std::get<lmmc_real_t>(base_num->value()));
78 if (std::abs(base_val + 1.0) < 1e-12) {
79 std::vector<std::shared_ptr<const SymbolicNode>> rest;
80 for (size_t j = 0; j < mul->operands().size(); ++j)
81 if (j != i) rest.push_back(mul->operands()[j]);
82 if (rest.empty()) remainder = SymbolicExpr::number(1);
83 else if (rest.size() == 1) remainder = lamina::detail::make_expression_ptr(rest[0]);
84 else remainder = lamina::detail::make_expression_ptr(lamina::detail::make_node<MultiplyNode>(rest));
85 return true;
86 }
87 }
88 }
89 return false;
90}

92static bool series_detect_trig_oscillation(
93 const std::shared_ptr<const SymbolicNode>& node,
94 const std::string& n,
95 std::shared_ptr<SymbolicExpr>& amplitude,
96 lamina::ComputationContext& context) {
97 auto func = std::dynamic_pointer_cast<const FunctionNode>(node);
98 if (func && (func->type() == FunctionNode::FuncType::Sin || func->type() == FunctionNode::FuncType::Cos)) {
99 if (!func->arguments().empty()) {
100 auto arg_expr = lamina::detail::make_expression_ptr(func->arguments()[0]);
101 auto arg_limit = lamina::detail::propagate_result(
102 lamina::limit_expression_checked(
103 arg_expr, n, SymbolicExpr::infinity(),
104 LimitDirection::Both, context));
105 if (series_is_infinity(arg_limit)) { amplitude = SymbolicExpr::number(1); return true; }
106 }
107 }
108 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node);
109 if (!mul) return false;
110 for (size_t i = 0; i < mul->operands().size(); ++i) {
111 auto f = std::dynamic_pointer_cast<const FunctionNode>(mul->operands()[i]);
112 if (!f || (f->type() != FunctionNode::FuncType::Sin && f->type() != FunctionNode::FuncType::Cos)) continue;
113 if (f->arguments().empty()) continue;
114 auto arg_expr = lamina::detail::make_expression_ptr(f->arguments()[0]);
115 auto arg_limit = lamina::detail::propagate_result(
116 lamina::limit_expression_checked(
117 arg_expr, n, SymbolicExpr::infinity(),
118 LimitDirection::Both, context));
119 if (!series_is_infinity(arg_limit)) continue;
120 std::vector<std::shared_ptr<const SymbolicNode>> rest;
121 for (size_t j = 0; j < mul->operands().size(); ++j)
122 if (j != i) rest.push_back(mul->operands()[j]);
123 if (rest.empty()) amplitude = SymbolicExpr::number(1);
124 else if (rest.size() == 1) amplitude = lamina::detail::make_expression_ptr(rest[0]);
125 else amplitude = lamina::detail::make_expression_ptr(lamina::detail::make_node<MultiplyNode>(rest));
126 return true;
127 }
128 return false;

153 const std::shared_ptr<SymbolicExpr>& general_coefficient,
154 const std::string& index_var,
155 ComputationContext& context) {
156 constexpr const char* operation = "convergence_radius";
157 if (!general_coefficient || !lamina::detail::node(general_coefficient) ||
158 index_var.empty()) {
159 return ExpressionResult::failure(
160 CasErrc::InvalidArgument,
161 "convergence radius requires a coefficient term and index variable",
162 operation);
163 }
164 auto step = context.consume_steps(2, operation);
165 if (!step) return ExpressionResult::failure(step.error());
166 auto node = lamina::detail::node(general_coefficient);
167 if (auto number = std::dynamic_pointer_cast<const NumberNode>(node)) {
168 return ExpressionResult::success(
169 number->is_zero() ? SymbolicExpr::infinity()
170 : SymbolicExpr::number(1));
171 }
172 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
173 auto exponent_variable = std::dynamic_pointer_cast<const VariableNode>(
174 power->exponent());
175 if (exponent_variable && exponent_variable->name() == index_var &&
176 !expression_depends_on_variable(power->base(), index_var)) {
177 auto absolute = lamina::detail::make_expression_ptr(
178 lamina::detail::make_node<FunctionNode>(
179 FunctionNode::FuncType::Abs,
180 std::vector<std::shared_ptr<const SymbolicNode>>{
181 power->base()}));
182 return ExpressionResult::success(
183 SymbolicExpr::divide(SymbolicExpr::number(1), absolute)->simplify());
184 }
185 auto base_variable = std::dynamic_pointer_cast<const VariableNode>(
186 power->base());
187 if (base_variable && base_variable->name() == index_var &&
188 !expression_depends_on_variable(power->exponent(), index_var)) {
189 return ExpressionResult::success(SymbolicExpr::number(1));
190 }
191 }
192 return ExpressionResult::failure(
193 CasErrc::Inconclusive,
194 "coefficient-term limit is outside the supported ratio/root-test domain",
195 operation);
196}

305static ConvergenceInfo convergence_test_impl(
306 const std::shared_ptr<SymbolicExpr>& general_term,
307 const std::string& index_var,
308 ComputationContext& context) {
309 if (!general_term || !lamina::detail::node(general_term)) return {ConvergenceResult::Inconclusive, ""};
310 if (auto power = std::dynamic_pointer_cast<const PowerNode>(lamina::detail::node(general_term))) {
311 auto base_var = std::dynamic_pointer_cast<const VariableNode>(power->base());
312 auto exponent = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
313 if (base_var && base_var->name() == index_var && exponent) {
314 double p = 0.0;
315 if (std::holds_alternative<BigInt>(exponent->value())) {
316 p = std::get<BigInt>(exponent->value()).to_double();
317 } else if (std::holds_alternative<Rational>(exponent->value())) {
318 p = std::get<Rational>(exponent->value()).to_double();
319 } else {
320 p = static_cast<double>(std::get<lmmc_real_t>(exponent->value()));
321 }
322 if (p < -1.0) return {ConvergenceResult::Convergent, "p-series"};
323 if (p >= -1.0 && p < 0.0) return {ConvergenceResult::Divergent, "p-series"};
324 }
325 }
326 /// Laurent 单项式 c*n^e 直接应用 p 级数判据.
327 /// 该路径使倒数幂保持在线性规则内,并跳过 Abs 比值极限的递归化简.
328 if (const auto laurent_power = supported_laurent_integer_power(
329 lamina::detail::node(general_term), index_var)) {
330 if (*laurent_power < -1) {
331 return {ConvergenceResult::Convergent, "p-series"};
332 }
333 if (*laurent_power >= -1 && *laurent_power < 0) {
334 return {ConvergenceResult::Divergent, "p-series"};
335 }
336 }
337 auto n = SymbolicExpr::variable(index_var);
338 auto n1 = SymbolicExpr::add(n, SymbolicExpr::number(1));
339 auto inf = SymbolicExpr::infinity();
340 auto a_n1 = general_term->substitute(index_var, n1);
341 if (a_n1) {
342 a_n1 = a_n1->simplify();
343 auto ratio = SymbolicExpr::divide(a_n1, general_term);
344 if (ratio) {
345 ratio = ratio->simplify();
346 auto abs_r = lamina::detail::make_expression_ptr(lamina::detail::make_node<FunctionNode>(
347 FunctionNode::FuncType::Abs, std::vector<std::shared_ptr<const SymbolicNode>>{lamina::detail::node(ratio)}));
348 abs_r = abs_r->simplify();
349 auto lim = detail::propagate_result(limit_expression_checked(
350 abs_r, index_var, inf, LimitDirection::Both, context));
351 if (lim) {
352 auto ls = lim->simplify();
353 if (ls && series_is_number(ls)) {
354 double v = series_get_double(ls);
355 if (v < 1.0 - 1e-12) return {ConvergenceResult::Convergent, "ratio"};
356 if (v > 1.0 + 1e-12) return {ConvergenceResult::Divergent, "ratio"};
357 }
358 if (ls && ls->is_zero()) return {ConvergenceResult::Convergent, "ratio"};
359 if (series_is_infinity(ls)) return {ConvergenceResult::Divergent, "ratio"};
360 }
361 }
362 }
363 return {ConvergenceResult::Inconclusive, ""};
364}
365ExpressionResult lim_sup_checked(
366 const std::shared_ptr<SymbolicExpr>& a_n, const std::string& n,
367 ComputationContext& context) {
368 constexpr const char* operation = "lim_sup";
369 if (!a_n || !detail::node(a_n) || n.empty()) {

372 "upper limit requires a sequence term and index variable",

376 if (!budget) return ExpressionResult::failure(budget.error());
377 try {
378 std::shared_ptr<SymbolicExpr> remainder;
379 if (series_extract_alternating(detail::node(a_n), n, remainder)) {
380 auto limit = detail::propagate_result(limit_expression_checked(
381 remainder, n, SymbolicExpr::infinity(),
382 LimitDirection::Both, context));

384 if (simplified && !series_is_infinity(simplified)) {
385 return ExpressionResult::success(series_abs(simplified));
386 }

397 amplitude, n, SymbolicExpr::infinity(),
398 LimitDirection::Both, context));
399 auto simplified = limit ? limit->simplify() : nullptr;
400 if (simplified && series_is_number(simplified)) {
401 return ExpressionResult::success(SymbolicExpr::number(

403 }
404 if (simplified) {
405 return ExpressionResult::success(series_abs(simplified));
406 }
407 }

410 a_n, n, SymbolicExpr::infinity(),
411 LimitDirection::Both, context));

413 return ExpressionResult::success(

415 } catch (const detail::ResultPropagation& propagation) {
416 return ExpressionResult::failure(propagation.error());
417 } catch (const std::bad_alloc&) {
