10Result<void> prove_exact_full_column_rank_qr_support(
11 const std::shared_ptr<const MatrixNode>& matrix,
12 ComputationContext& context,
13 const std::string& operation) {
14 auto values = exact_rational_matrix(matrix);
15 if (!values) {
16 return Result<void>::failure(
17 CasErrc::Inconclusive,
18 "QR input is outside the exact rational full-column-rank support domain",
19 operation);
20 }
21 const size_t rows = values->size();
22 const size_t cols = rows == 0 ? 0 : (*values)[0].size();
23 if (rows < cols) {
24 return Result<void>::failure(
25 CasErrc::Inconclusive,
26 "QR decomposition requires at least as many rows as columns in the supported domain",
27 operation);
28 }
29 std::vector<std::vector<Rational>> gram(
30 cols, std::vector<Rational>(cols, Rational(0)));
31 for (size_t i = 0; i < cols; ++i) {
32 for (size_t j = 0; j < cols; ++j) {
33 Rational sum(0);
34 for (size_t row = 0; row < rows; ++row) {
35 sum = sum + (*values)[row][i] * (*values)[row][j];
36 }
37 gram[i][j] = sum;
38 }
39 }
40 for (size_t order = 1; order <= cols; ++order) {
41 std::vector<Rational> leading;
42 leading.reserve(order * order);
43 for (size_t row = 0; row < order; ++row) {
44 for (size_t col = 0; col < order; ++col) {
45 leading.push_back(gram[row][col]);
46 }
47 }
48 auto determinant = detail::rational_determinant_exact(
49 order, std::move(leading), context, operation);
50 if (!determinant) return Result<void>::failure(determinant.error());
51 if (determinant.value() <= Rational(0)) {
52 return Result<void>::failure(
53 CasErrc::Inconclusive,
54 "QR decomposition requires proven full column rank",
55 operation);
56 }
57 }
58 return Result<void>::success();
59}
60static bool qr_decomposition_impl(
61 const std::shared_ptr<SymbolicExpr>& A,
62 std::shared_ptr<SymbolicExpr>& Q,
63 std::shared_ptr<SymbolicExpr>& R) {
64 auto mat = std::dynamic_pointer_cast<const MatrixNode>(LMCAS::detail::node(A));
65 if (!mat) return false;
66 size_t m = mat->rows();
67 size_t n = mat->cols();
68
69 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> Q_grid(m, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
70 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> R_grid(n, std::vector<std::shared_ptr<SymbolicExpr>>(n, SymbolicExpr::number(0)));
71
72 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> A_cols(n, std::vector<std::shared_ptr<SymbolicExpr>>(m));
73 for (size_t j = 0; j < n; j++) {
74 for (size_t i = 0; i < m; i++) A_cols[j][i] = LMCAS::detail::make_expression_ptr(mat->get(i, j));
75 }
76
77 for (size_t j = 0; j < n; j++) {
78 std::vector<std::shared_ptr<SymbolicExpr>> u_j = A_cols[j];
79 for (size_t i = 0; i < j; i++) {
80 auto dot = SymbolicExpr::number(0);
81 for (size_t k = 0; k < m; k++) dot = SymbolicExpr::add(dot, SymbolicExpr::multiply(Q_grid[k][i], A_cols[j][k]));
82 R_grid[i][j] = dot;
83
84 for (size_t k = 0; k < m; k++) {
85 u_j[k] = SymbolicExpr::add(u_j[k], SymbolicExpr::multiply(SymbolicExpr::number(-1), SymbolicExpr::multiply(R_grid[i][j], Q_grid[k][i])));
86 }
87 }
88
89 auto norm_sq = SymbolicExpr::number(0);
90 for (size_t k = 0; k < m; k++) norm_sq = SymbolicExpr::add(norm_sq, SymbolicExpr::multiply(u_j[k], u_j[k]));
91 R_grid[j][j] = SymbolicExpr::sqrt(norm_sq);
92
93 for (size_t k = 0; k < m; k++) Q_grid[k][j] = SymbolicExpr::divide(u_j[k], R_grid[j][j]);
94 }
95
96 Q = SymbolicExpr::matrix(Q_grid);
97 R = SymbolicExpr::matrix(R_grid);
98 return true;
99}

147std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> gram_schmidt(
148 const std::vector<std::vector<std::shared_ptr<SymbolicExpr>>>& vectors,
149 bool normalize) {
150 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> basis;
151 for (const auto& v : vectors) {
152 std::vector<std::shared_ptr<SymbolicExpr>> u = v;
153 for (const auto& b : basis) {
154 auto vb = SymbolicExpr::number(0);
155 auto bb = SymbolicExpr::number(0);
156 for (size_t k = 0; k < u.size(); ++k) {
157 vb = SymbolicExpr::add(vb, SymbolicExpr::multiply(v[k], b[k]));
158 bb = SymbolicExpr::add(bb, SymbolicExpr::multiply(b[k], b[k]));
159 }
160 if (LMCAS::detail::node(bb) && LMCAS::detail::node(bb)->is_zero()) continue;
161 auto coeff = SymbolicExpr::divide(vb, bb);
162 for (size_t k = 0; k < u.size(); ++k) {
163 u[k] = SymbolicExpr::add(u[k],
164 SymbolicExpr::multiply(SymbolicExpr::number(-1),
165 SymbolicExpr::multiply(coeff, b[k])))->simplify();
166 }
167 }
168 auto norm_sq = SymbolicExpr::number(0);
169 for (auto& x : u) norm_sq = SymbolicExpr::add(norm_sq, SymbolicExpr::multiply(x, x));
170 norm_sq = norm_sq->simplify();
171 if (LMCAS::detail::node(norm_sq) && LMCAS::detail::node(norm_sq)->is_zero()) continue;
172 basis.push_back(u);
173 }
174 if (normalize) {
175 for (auto& u : basis) {
176 auto norm_sq = SymbolicExpr::number(0);
177 for (auto& x : u) norm_sq = SymbolicExpr::add(norm_sq, SymbolicExpr::multiply(x, x));
178 auto norm = SymbolicExpr::sqrt(norm_sq);
179 for (auto& x : u) x = SymbolicExpr::divide(x, norm)->simplify();
180 }
181 }
182 return basis;
183}
