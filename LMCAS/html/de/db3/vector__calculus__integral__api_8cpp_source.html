66std::shared_ptr<SymbolicExpr> greens_theorem(
67 const std::shared_ptr<SymbolicExpr>& P,
68 const std::shared_ptr<SymbolicExpr>& Q,
69 const std::vector<std::string>& vars,
70 const std::pair<std::shared_ptr<SymbolicExpr>, std::shared_ptr<SymbolicExpr>>& x_bounds,
71 const std::pair<std::shared_ptr<SymbolicExpr>, std::shared_ptr<SymbolicExpr>>& y_bounds)
72{
73 if (!P || !Q) {
74 throw std::invalid_argument("greens_theorem: P and Q must not be null");
75 }
76 if (vars.size() != 2) {
77 throw std::invalid_argument("greens_theorem: vars must have exactly 2 elements");
78 }
79 if (!x_bounds.first || !x_bounds.second ||
80 !y_bounds.first || !y_bounds.second) {
81 throw std::invalid_argument("greens_theorem: bounds must not be null");
82 }
83
84 const std::string& x_var = vars[0];
85 const std::string& y_var = vars[1];
86
87 /// 计算被积函数: ∂Q/∂x - ∂P/∂y
88 auto dQ_dx = Q->differentiate(x_var);
89 auto dP_dy = P->differentiate(y_var);
90
91 if (!dQ_dx) dQ_dx = SymbolicExpr::number(0);
92 if (!dP_dy) dP_dy = SymbolicExpr::number(0);
93
94 dQ_dx = dQ_dx->simplify();
95 dP_dy = dP_dy->simplify();
96
97 auto integrand = SymbolicExpr::add(
98 dQ_dx,
99 SymbolicExpr::multiply(SymbolicExpr::number(-1), dP_dy));
100 integrand = integrand->simplify();
101
102 Integrator integrator;
103 ComputationContext context;
104 std::vector<IntegrationStep> steps = {
105 {y_var, y_bounds.first, y_bounds.second},
106 {x_var, x_bounds.first, x_bounds.second},
107 };
108 auto result = integrate_multiple_checked(*integrand, steps, integrator, context);
109 if (!result) throw std::runtime_error(result.error().message);
110 return LMCAS::detail::make_expression_ptr(result.value());
111}

337std::shared_ptr<SymbolicExpr> stokes_theorem(
338 const VectorField& F,
339 const std::vector<std::string>& vars,
340 const VectorField& parametrization,
341 const std::string& u, const std::string& v,
342 const std::pair<std::shared_ptr<SymbolicExpr>, std::shared_ptr<SymbolicExpr>>& u_bounds,
343 const std::pair<std::shared_ptr<SymbolicExpr>, std::shared_ptr<SymbolicExpr>>& v_bounds)
344{
345 if (F.size() != 3) {
346 throw std::invalid_argument("stokes_theorem: F must have 3 components");
347 }
348 if (vars.size() != 3) {
349 throw std::invalid_argument("stokes_theorem: vars must have exactly 3 elements");
350 }
351 if (parametrization.size() != 3) {
352 throw std::invalid_argument(
353 "stokes_theorem: parametrization must have 3 components");
354 }
355 if (!u_bounds.first || !u_bounds.second ||
356 !v_bounds.first || !v_bounds.second) {
357 throw std::invalid_argument("stokes_theorem: bounds must not be null");
358 }
359
360 /// 计算旋度 ∇×F
361 auto curl_F = curl(F, vars);
362
363 /// 计算 r_u × r_v（曲面法向量）
364 auto cross = vector_calculus_cross_product_partials(parametrization, u, v);
365
366 /// 将 (∇×F) 中的坐标变量替换为参数化表达式，然后计算 (∇×F)·(r_u × r_v)
367 std::shared_ptr<SymbolicExpr> dot_product = nullptr;
368
369 for (size_t i = 0; i < 3; ++i) {
370 if (!curl_F[i]) {
371 continue;
372 }
373
374 /// 将 (∇×F)ᵢ 中的坐标变量替换为参数化表达式
375 auto curl_i_composed = curl_F[i];
376 for (size_t j = 0; j < 3; ++j) {
377 curl_i_composed = curl_i_composed->substitute(vars[j], parametrization[j]);
378 }
379 curl_i_composed = curl_i_composed->simplify();
380
381 /// (∇×F)ᵢ(r(u,v)) · (r_u × r_v)ᵢ
382 auto term = SymbolicExpr::multiply(curl_i_composed, cross[i]);
383 term = term->simplify();
384
385 if (!dot_product) {
386 dot_product = term;
387 } else {
388 dot_product = SymbolicExpr::add(dot_product, term);
389 dot_product = dot_product->simplify();
390 }
391 }
392
393 if (!dot_product) {
394 return SymbolicExpr::number(0);
395 }
396
397 Integrator integrator;
398 ComputationContext context;
399 std::vector<IntegrationStep> steps = {
400 {v, v_bounds.first, v_bounds.second},
401 {u, u_bounds.first, u_bounds.second},
402 };
403 auto result = integrate_multiple_checked(*dot_product, steps, integrator, context);
404 if (!result) throw std::runtime_error(result.error().message);
405 return LMCAS::detail::make_expression_ptr(result.value());
406}
