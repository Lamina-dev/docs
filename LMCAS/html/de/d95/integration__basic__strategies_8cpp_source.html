5std::shared_ptr<SymbolicExpr> TableLookupStrategy::try_integrate_raw(
6 const SymbolicExpr& expr, const std::string& var, Integrator& ctx,
7 ComputationContext& computation, int) {
8 (void)computation;
9
10 auto all_entries = ctx.table().get_all_sorted();
11
12 for (const auto* entry : all_entries) {
13 MatchMap bindings;
14 if (Matcher::match(entry->pattern, expr, entry->wildcards, bindings)) {
15
16 // The matcher allows partial matches on commutative operations
17 // (Add/Multiply): unmatched operands are bound to __Add_REST__ /
18 // __Mul_REST__ for use by rewrite rules. For integration table
19 // lookup that behavior is wrong - a pattern like (1 + _u^2)^-1
20 // would otherwise match (1 + x^2 + x + x^3)^-1 with _u=x and the
21 // rest x + x^3 silently dropped, producing arctan(x) for an
22 // integrand that has nothing to do with arctan. Require an exact
23 // match (no leftover) so the result genuinely equals the
24 // integrand under this binding.
25 if (bindings.find("__Add_REST__") != bindings.end() ||
26 bindings.find("__Mul_REST__") != bindings.end()) {
27 continue;
28 }
29 const bool complete_binding = std::all_of(
30 entry->wildcards.begin(), entry->wildcards.end(),
31 [&](const std::string& wildcard_name) {
32 return bindings.find(wildcard_name) != bindings.end();
33 });
34 if (!complete_binding) continue;
35
36 if (entry->condition && !entry->condition(bindings, var)) {
37 continue;
38 }
39
40 SymbolicExpr result = Matcher::replace(entry->result, bindings, false);
41 auto simplified = result.simplify();
42 if (simplified && !contains_unevaluated_integral(lamina::detail::node(simplified))) {
43 return simplified;
44 }
45 }
46 }
47 return nullptr;
48}

81std::shared_ptr<SymbolicExpr> SubstitutionStrategy::try_integrate_raw(
82 const SymbolicExpr& expr, const std::string& var, Integrator&,
83 ComputationContext& computation, int) {
84
85 std::vector<std::shared_ptr<const SymbolicNode>> ops;
86 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(expr))) {
87 ops = mul->operands();
88 } else {
89 ops.push_back(lamina::detail::node(expr));
90 }
91
92 for (size_t i = 0; i < ops.size(); ++i) {
93 auto candidate_term = lamina::detail::expression_from_node(ops[i]);
94 std::optional<SymbolicExpr> u;
95
96 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(lamina::detail::node(candidate_term))) {
97 u = lamina::detail::expression_from_node(pow->base());
98 } else if (auto func = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(candidate_term))) {
99 if (!func->arguments().empty()) {
100 u = lamina::detail::expression_from_node(func->arguments()[0]);
101 }
102 }
103
104 if (u && depends_on_integration_variable(*u, var)) {
105 auto d_ptr = u->differentiate(var);
106 if (!d_ptr) continue;
107 auto du = d_ptr->simplify();
108 if (du->is_zero()) continue;
109
110 auto f_u = candidate_term;
111 auto term_times_du = SymbolicExpr::multiply(make_expr_ptr(f_u), make_expr_ptr(*du));
112 auto ratio = SymbolicExpr::divide(make_expr_ptr(expr), term_times_du)->simplify();
113 bool ratio_independent = !depends_on_integration_variable(*ratio, var);
114 if (!ratio_independent && computation.assumptions()) {
115 Tribool du_nonzero = detail::propagate_result(
116 computation.assumptions()->is_nonzero(*du));
117 if (du_nonzero == Tribool::True) {
118 ratio_independent = !depends_on_integration_variable(*ratio, var);
119 }
120 }
121
122 if (ratio_independent) {
123 std::shared_ptr<SymbolicExpr> prim = nullptr;
124
125 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(lamina::detail::node(candidate_term))) {
126 auto n = lamina::detail::expression_from_node(pow->exponent());
127 auto np1 = SymbolicExpr::add(make_expr_ptr(n), SymbolicExpr::number(1))->simplify();
128 if (np1->is_zero()) {
129 prim = SymbolicExpr::ln(make_expr_ptr(*u));
130 } else {
131 prim = SymbolicExpr::divide(
132 SymbolicExpr::power(make_expr_ptr(*u), np1), np1);
133 }
134 } else if (auto func = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(candidate_term))) {
135 if (func->type() == FunctionNode::FuncType::Cos) {
136 prim = SymbolicExpr::sin(make_expr_ptr(*u));
137 } else if (func->type() == FunctionNode::FuncType::Sin) {
138 prim = SymbolicExpr::multiply(SymbolicExpr::number(-1), SymbolicExpr::cos(make_expr_ptr(*u)));
139 } else if (func->type() == FunctionNode::FuncType::Exp) {
140 prim = SymbolicExpr::exp(make_expr_ptr(*u));
141 }
142 }
143
144 if (prim) {
145 return SymbolicExpr::multiply(ratio, prim);
146 }
147 }
148 }
149 }
150 return nullptr;
151}

154bool LinearSubstitutionStrategy::extract_linear_arg(
155 const SymbolicExpr& arg,
156 const std::string& var,
157 std::shared_ptr<SymbolicExpr>& a_out,
158 std::shared_ptr<SymbolicExpr>& b_out) {
159
160 // Argument must actually depend on the integration variable.
161 if (!expression_depends_on_variable(lamina::detail::node(arg), var)) return false;
162
163 Polynomial<SymbolicPolyCoeff> poly;
164 try {
165 poly = symbolic_to_poly<SymbolicPolyCoeff>(make_expr_ptr(arg), var);
166 } catch (const std::invalid_argument&) {
167 return false;
168 } catch (const std::out_of_range&) {
169 return false;
170 } catch (const std::runtime_error&) {
171 return false;
172 }
173
174 // Need to be exactly degree 1 in var: arg = a * var + b with a, b constant.
175 if (poly.degree() != 1) return false;
176 if (poly.coeffs.size() < 2) return false;
177
178 auto a_expr = poly.coeffs[1].val;
179 auto b_expr = poly.coeffs[0].val;
180 if (!a_expr || !b_expr) return false;
181
182 // Coefficients must not depend on the integration variable.
183 if (expression_depends_on_variable(lamina::detail::node(a_expr), var)) return false;
184 if (expression_depends_on_variable(lamina::detail::node(b_expr), var)) return false;
185
186 auto a_simp = a_expr->simplify();
187 auto b_simp = b_expr->simplify();
188 if (!a_simp) a_simp = a_expr;
189 if (!b_simp) b_simp = b_expr;
190
191 // a must be demonstrably non-zero (refuse if simplification yields zero).
192 if (a_simp->is_zero()) return false;
193
194 a_out = a_simp;
195 b_out = b_simp;
196 return true;
197}

199std::shared_ptr<SymbolicExpr> LinearSubstitutionStrategy::try_integrate_raw(
200 const SymbolicExpr& expr, const std::string& var, Integrator& ctx,
201 ComputationContext& computation, int depth) {
202
203 using FT = FunctionNode::FuncType;
204
205 enum class Wrapper { None, Function, PowerOfFunction, PowerOfLinear };
206 Wrapper kind = Wrapper::None;
207
208 std::shared_ptr<const FunctionNode> fn; // outer FunctionNode (Function case)
209 std::shared_ptr<const PowerNode> pn; // outer PowerNode (Power* cases)
210 std::shared_ptr<const FunctionNode> base_fn; // inner FunctionNode of a PowerNode
211 std::optional<SymbolicExpr> arg_expr; // the linear-candidate sub-expression
212
213 if ((fn = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(expr)))) {
214 // Skip wrappers that are not "real" single-argument functions.
215 switch (fn->type()) {
216 case FT::Infinity:
217 case FT::Atan2:
218 case FT::Log:
219 return nullptr;
220 default:
221 break;
222 }
223 if (fn->arguments().size() != 1) return nullptr;
224 arg_expr = lamina::detail::expression_from_node(fn->arguments()[0]);
225 kind = Wrapper::Function;
226 } else if ((pn = std::dynamic_pointer_cast<const PowerNode>(lamina::detail::node(expr)))) {
227 // Exponent must be independent of the integration variable so that
228 // substituting the linear argument back is sound.
229 auto exp_expr = lamina::detail::expression_from_node(pn->exponent());
230 if (expression_depends_on_variable(lamina::detail::node(exp_expr), var)) return nullptr;
231
232 if ((base_fn = std::dynamic_pointer_cast<const FunctionNode>(pn->base()))) {
233 switch (base_fn->type()) {
234 case FT::Infinity:
235 case FT::Atan2:
236 case FT::Log:
237 return nullptr;
238 default:
239 break;
240 }
241 if (base_fn->arguments().size() != 1) return nullptr;
242 arg_expr = lamina::detail::expression_from_node(base_fn->arguments()[0]);
243 kind = Wrapper::PowerOfFunction;
244 } else {
245 // Treat the whole base as the linear-candidate (e.g. (2x+1)^n).
246 arg_expr = lamina::detail::expression_from_node(pn->base());
247 kind = Wrapper::PowerOfLinear;
248 }
249 } else {
250 return nullptr;
251 }
252
253 std::shared_ptr<SymbolicExpr> a_coeff, b_coeff;
254 if (!arg_expr || !extract_linear_arg(*arg_expr, var, a_coeff, b_coeff)) return nullptr;
255
256 // Degenerate case: arg == var (a = 1, b = 0). The plain TableLookup /
257 // PowerRule strategies would have handled this earlier in the chain;
258 // returning nullptr lets the next strategy try.
259 bool a_is_one = a_coeff && a_coeff->is_one();
260 bool b_is_zero = b_coeff && b_coeff->is_zero();
261 if (a_is_one && b_is_zero) return nullptr;
262
263 const std::string dummy_name = "__lin_sub_u__";
264 auto dummy_var = SymbolicExpr::variable(dummy_name);
265
266 std::shared_ptr<SymbolicExpr> test_expr;
267 if (kind == Wrapper::Function) {
268 std::vector<std::shared_ptr<const SymbolicNode>> new_args{lamina::detail::node(dummy_var)};
269 test_expr = lamina::detail::make_expression_ptr(
270 lamina::detail::make_node<FunctionNode>(fn->type(), new_args));
271 } else if (kind == Wrapper::PowerOfFunction) {
272 std::vector<std::shared_ptr<const SymbolicNode>> new_args{lamina::detail::node(dummy_var)};
273 auto new_fn = lamina::detail::make_node<FunctionNode>(base_fn->type(), new_args);
274 test_expr = lamina::detail::make_expression_ptr(
275 lamina::detail::make_node<PowerNode>(new_fn, pn->exponent()));
276 } else { // Wrapper::PowerOfLinear
277 test_expr = lamina::detail::make_expression_ptr(
278 lamina::detail::make_node<PowerNode>(lamina::detail::node(dummy_var), pn->exponent()));
279 }
280 if (!test_expr) return nullptr;
281
282 TableLookupStrategy table_only;
283 auto table_attempt = table_only.try_integrate(
284 *test_expr, dummy_name, ctx, computation, depth);
285 if (!table_attempt) {
286 throw detail::ResultPropagation(table_attempt.error());
287 }
288 auto* table_candidate =
289 std::get_if<IntegrationCandidate>(&table_attempt.value());
290 if (!table_candidate || !table_candidate->expression) return nullptr;
291 auto F_substituted = table_candidate->expression->substitute(
292 dummy_name, make_expr_ptr(*arg_expr));
293 if (!F_substituted) return nullptr;
294
295 auto inv_a = SymbolicExpr::power(a_coeff, SymbolicExpr::number(-1));
296 auto result = SymbolicExpr::multiply(inv_a, F_substituted);
297
298 auto simplified = result->simplify();
299 if (simplified && !contains_unevaluated_integral(lamina::detail::node(simplified))) {
300 return simplified;
301 }
302 return result;
303}

305std::shared_ptr<SymbolicExpr> PartialFractionStrategy::try_integrate_raw(
306 const SymbolicExpr& expr, const std::string& var, Integrator&,
307 ComputationContext& computation, int) {
308
309 std::shared_ptr<SymbolicExpr> den = nullptr;
310
311 if (auto p = std::dynamic_pointer_cast<const PowerNode>(lamina::detail::node(expr))) {
312 double exp_val = 0;
313 bool is_inv = false;
314 if (auto num_node = std::dynamic_pointer_cast<const NumberNode>(p->exponent())) {
315 if (std::holds_alternative<lmmc_real_t>(num_node->value()))
316 exp_val = std::get<lmmc_real_t>(num_node->value());
317 else if (std::holds_alternative<Rational>(num_node->value()))
318 exp_val = std::get<Rational>(num_node->value()).to_double();
319 else if (std::holds_alternative<BigInt>(num_node->value()))
320 exp_val = std::get<BigInt>(num_node->value()).to_double();
321 int eq;
322 lmmc_double_nearly_equal_tol(exp_val, -1.0, 1e-9, 1e-9, &eq);
323 if (eq) is_inv = true;
324 }
325 if (is_inv) {
326 den = make_expr_ptr(lamina::detail::expression_from_node(p->base()));
327 }
328 }
329
330 if (!den) return nullptr;
331
332 try {
333 Polynomial<SymbolicPolyCoeff> Q = symbolic_to_poly<SymbolicPolyCoeff>(den, var);
334
335 if (Q.degree() == 2) {
336 SymbolicExpr c_expr = *(Q.coeffs[0].val);
337 SymbolicExpr b_expr = *(Q.coeffs[1].val);
338 SymbolicExpr a_expr = *(Q.coeffs[2].val);
339
340 if (!a_expr.is_number() || !b_expr.is_number() || !c_expr.is_number()) {
341 return nullptr;
342 }
343
344 auto a_checked = try_checked_numeric_constant(a_expr, computation);
345 auto b_checked = try_checked_numeric_constant(b_expr, computation);
346 auto c_checked = try_checked_numeric_constant(c_expr, computation);
347 if (!a_checked || !b_checked || !c_checked) {
348 return nullptr;
349 }
350
351 double a = *a_checked;
352 double b = *b_checked;
353 double c = *c_checked;
354
355 int eq_a;
356 lmmc_double_nearly_equal_tol(a, 0.0, 1e-9, 1e-9, &eq_a);
357 if (eq_a) return nullptr;
358
359 double delta = b * b - 4 * a * c;
360 int eq_delta;
361 lmmc_double_nearly_equal_tol(delta, 0.0, 1e-9, 1e-9, &eq_delta);
362
363 if (!eq_delta && delta > 0) {
364 double sqrt_delta = std::sqrt(delta);
365 auto scalar = SymbolicExpr::number(1.0 / sqrt_delta);
366 auto two_a = SymbolicExpr::number(2.0 * a);
367 auto b_num = SymbolicExpr::number(b);
368 auto two_a_x = SymbolicExpr::multiply(make_expr_ptr(*two_a), SymbolicExpr::variable(var));
369 auto two_a_x_plus_b = SymbolicExpr::add(make_expr_ptr(*two_a_x), make_expr_ptr(*b_num));
370 auto term1_arg = sym_sub(*two_a_x_plus_b, *SymbolicExpr::number(sqrt_delta));
371 auto term2_arg = SymbolicExpr::add(make_expr_ptr(*two_a_x_plus_b), SymbolicExpr::number(sqrt_delta));
372 auto term1 = SymbolicExpr::ln(make_expr_ptr(*term1_arg));
373 auto term2 = SymbolicExpr::ln(make_expr_ptr(*term2_arg));
374 return SymbolicExpr::multiply(scalar, sym_sub(*term1, *term2));
375 } else if (!eq_delta && delta < 0) {
376 double sqrt_neg_delta = std::sqrt(-delta);
377 auto scalar = SymbolicExpr::number(2.0 / sqrt_neg_delta);
378 auto two_a = SymbolicExpr::number(2.0 * a);
379 auto b_num = SymbolicExpr::number(b);
380 auto num = SymbolicExpr::add(
381 SymbolicExpr::multiply(make_expr_ptr(*two_a), SymbolicExpr::variable(var)),
382 make_expr_ptr(*b_num));
383 auto inner = SymbolicExpr::divide(make_expr_ptr(*num), SymbolicExpr::number(sqrt_neg_delta));
384 return SymbolicExpr::multiply(scalar, make_arctan(inner));
385 }
386 }
387 } catch (const std::invalid_argument&) {
388 } catch (const std::out_of_range&) {
389 } catch (const std::runtime_error&) {
390 }
391
392 return nullptr;
393}

395std::shared_ptr<SymbolicExpr> IBPStrategy::try_integrate_raw(
396 const SymbolicExpr& expr, const std::string& var, Integrator& ctx,
397 ComputationContext& computation, int depth) {
398
399 std::vector<std::shared_ptr<const SymbolicNode>> ops;
400 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(expr))) {
401 ops = mul->operands();
402 } else {
403 ops.push_back(lamina::detail::node(expr));
404 }
405
406 int best_u_idx = -1;
407 int best_score = 100;
408
409 auto get_score = [&](const std::shared_ptr<const SymbolicNode>& node) -> int {
410 auto e = lamina::detail::expression_from_node(node);
411 if (auto fn = std::dynamic_pointer_cast<const FunctionNode>(node)) {
412 if (fn->type() == FunctionNode::FuncType::Ln || fn->type() == FunctionNode::FuncType::Log) return 1;
413 if (fn->type() == FunctionNode::FuncType::ArcSin || fn->type() == FunctionNode::FuncType::ArcTan) return 2;
414 if (fn->type() == FunctionNode::FuncType::Sin || fn->type() == FunctionNode::FuncType::Cos) return 4;
415 if (fn->type() == FunctionNode::FuncType::Exp) return 5;
416 }
417 if (!depends_on_integration_variable(e, var)) return 10;
418 if (std::dynamic_pointer_cast<const VariableNode>(node)) return 3;
419 if (std::dynamic_pointer_cast<const PowerNode>(node)) return 3;
420 return 10;
421 };
422
423 if (ops.size() == 1) {
424 int s = get_score(ops[0]);
425 if (s <= 2) best_u_idx = 0;
426 } else {
427 for (size_t i = 0; i < ops.size(); ++i) {
428 if (!depends_on_integration_variable(lamina::detail::expression_from_node(ops[i]), var)) continue;
429 int s = get_score(ops[i]);
430 if (s < best_score) {
431 best_score = s;
432 best_u_idx = (int)i;
433 }
434 }
435 }
436
437 if (best_u_idx == -1) return nullptr;
438
439 auto u = lamina::detail::expression_from_node(ops[best_u_idx]);
440 std::vector<std::shared_ptr<const SymbolicNode>> dv_ops;
441 for (size_t i = 0; i < ops.size(); ++i) {
442 if ((int)i != best_u_idx) dv_ops.push_back(ops[i]);
443 }
444
445 std::shared_ptr<SymbolicExpr> dv;
446 if (dv_ops.empty()) dv = SymbolicExpr::number(1);
447 else if (dv_ops.size() == 1) {
448 dv = make_expr_ptr(lamina::detail::expression_from_node(dv_ops[0]));
449 }
450 else dv = lamina::detail::make_expression_ptr(lamina::detail::make_node<MultiplyNode>(dv_ops));
451
452 auto v = detail::propagate_result(
453 ctx.integrate_recursive(*dv, var, computation, depth + 1));
454 if (!v || !lamina::detail::node(v)) return nullptr;
455 if (contains_unevaluated_integral(lamina::detail::node(v))) return nullptr;
456
457 auto du_ptr = u.differentiate(var);
458 if (!du_ptr) return nullptr;
459 auto simplified_du = du_ptr->simplify();
460 if (!simplified_du || !lamina::detail::node(simplified_du)) return nullptr;
461 auto du = make_expr_ptr(*simplified_du);
462
463 auto uv = SymbolicExpr::multiply(make_expr_ptr(u), v);
464 auto vdu = SymbolicExpr::multiply(v, du);
465 if (!uv || !vdu) return nullptr;
466 vdu = vdu->cancel()->simplify();
467 auto int_vdu = detail::propagate_result(
468 ctx.integrate_recursive(*vdu, var, computation, depth + 1));
469 if (!int_vdu || !lamina::detail::node(int_vdu) ||
470 contains_unevaluated_integral(lamina::detail::node(int_vdu))) {
471 return nullptr;
472 }
473
474 return sym_sub(*uv, *int_vdu);
475}
