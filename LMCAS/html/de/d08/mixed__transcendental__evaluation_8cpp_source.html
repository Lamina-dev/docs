28lmmc_real_t evaluate_inverse_trig(FunctionNode::FuncType type, lmmc_real_t arg) {
29 using Type = FunctionNode::FuncType;
30 switch (type) {
31 case Type::ArcSin:
32 if (arg < -1.0 || arg > 1.0) { return unsupported_value(); }
33 return std::asin(arg);
34 case Type::ArcCos:
35 if (arg < -1.0 || arg > 1.0) { return unsupported_value(); }
36 return std::acos(arg);
37 case Type::ArcTan:
38 return std::atan(arg);
39 case Type::Sec: {
40 const auto cosine = std::cos(arg);
41 if (std::fabs(cosine) < 1e-15) { return unsupported_value(); }
42 return 1.0 / cosine;
43 }
44 case Type::Csc:
45 case Type::Cot: {
46 const auto sine = std::sin(arg);
47 if (std::fabs(sine) < 1e-15) { return unsupported_value(); }
48 return type == Type::Csc ? 1.0 / sine : std::cos(arg) / sine;
49 }
50 default:
51 return unsupported_value();
52 }
53}

92lmmc_real_t evaluate_function(const FunctionNode& function) {
93 if (function.arguments().size() == 1) {
94 const auto argument = mixed_recursive_eval(function.arguments()[0]);
95 if (std::isnan(argument)) { return argument; }
96 return evaluate_unary(function.type(), argument);
97 }
98 if (function.arguments().size() == 2 &&
99 function.type() == FunctionNode::FuncType::Atan2) {
100 const auto y = mixed_recursive_eval(function.arguments()[0]);
101 const auto x = mixed_recursive_eval(function.arguments()[1]);
102 if (std::isnan(y) || std::isnan(x)) { return unsupported_value(); }
103 return std::atan2(y, x);
104 }
105 return unsupported_value();

158lmmc_real_t mixed_evaluate_at(

161 try {

174lmmc_real_t mixed_evaluate_with_retry(
175 const std::shared_ptr<SymbolicExpr>& expr,
176 const std::string& var, lmmc_real_t x,
177 lmmc_real_t half_width, int max_retries) {
178 auto value = mixed_evaluate_at(expr, var, x);
179 if (!std::isnan(value)) { return value; }
180 auto offset = half_width;
181 for (int i = 0; i < max_retries; ++i) {
182 offset *= 0.5;
183 value = mixed_evaluate_at(expr, var, x + offset);
184 if (!std::isnan(value)) { return value; }
185 value = mixed_evaluate_at(expr, var, x - offset);
186 if (!std::isnan(value)) { return value; }
187 }
188 return unsupported_value();
189}
