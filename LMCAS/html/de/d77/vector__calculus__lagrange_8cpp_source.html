24static VectorCalculusFieldResult lagrange_gradient_strict(
25 const std::shared_ptr<SymbolicExpr>& f,
26 const std::vector<std::string>& vars, const std::string& operation)
27{
28 VectorField gradient;
29 gradient.reserve(vars.size());
30 for (const auto& var : vars) {
31 auto partial = vector_calculus_differentiate_strict(f, var, operation);
32 if (!partial) {
33 return VectorCalculusFieldResult::failure(partial.error());
34 }
35 gradient.push_back(std::move(partial.value()));
36 }
37 return VectorCalculusFieldResult::success(std::move(gradient));
38}

51static std::shared_ptr<SymbolicExpr> lagrange_stationarity_equation(
52 const VectorField& gradient, const std::vector<VectorField>& constraints,
53 const std::vector<std::string>& multipliers, size_t index)
54{
55 auto equation = gradient[index];
56 for (size_t k = 0; k < multipliers.size(); ++k) {
57 auto multiplier = SymbolicExpr::variable(multipliers[k]);
58 auto term = SymbolicExpr::multiply(multiplier, constraints[k][index]);
59 equation = SymbolicExpr::add(equation,
60 SymbolicExpr::multiply(SymbolicExpr::number(-1), term));
61 }
62 return equation;
63}

65static LagrangeResult solve_lagrange_strict(
66 const VectorField& equations, const std::vector<std::string>& all_vars,
67 ComputationContext& context, const std::string& operation)
68{
69 std::vector<std::map<std::string, std::shared_ptr<SymbolicExpr>>> full_solutions;
70 std::vector<SymbolicExpr> poly_eqs;
71 poly_eqs.reserve(equations.size());
72 for (const auto& eq : equations) {
73 if (!eq || !LMCAS::detail::node(eq)) {
74 return LagrangeResult::failure(
75 CasErrc::Inconclusive,
76 "Lagrange equation construction failed in the supported domain",
77 operation);
78 }
79 poly_eqs.push_back(*eq);
80 }
81
82 auto checked_solutions =
83 Solver::solve_polynomial_system_checked(poly_eqs, all_vars, context);
84 if (!checked_solutions) return LagrangeResult::failure(checked_solutions.error());
85 const auto& poly_solutions = checked_solutions.value();
86 for (const auto& sol : poly_solutions) {
87 std::map<std::string, std::shared_ptr<SymbolicExpr>> pt;
88 for (const auto& [name, val] : sol) {
89 pt[name] = LMCAS::detail::make_expression_ptr(val);
90 }
91 full_solutions.push_back(std::move(pt));
92 }
93
94 return LagrangeResult::success(std::move(full_solutions));
95}

97static LagrangeResult verify_lagrange_solutions(
98 const std::vector<std::map<std::string, std::shared_ptr<SymbolicExpr>>>& full_solutions,
99 const VectorField& equations, const std::vector<std::string>& all_vars,
100 const std::vector<std::string>& vars, ComputationContext& context, const std::string& operation)
101{
102 std::vector<std::map<std::string, std::shared_ptr<SymbolicExpr>>> verified;
103 std::set<std::string> var_set(vars.begin(), vars.end());
104 for (const auto& full_solution : full_solutions) {
105 if (!vector_calculus_point_has_vars(full_solution, all_vars)) {
106 return LagrangeResult::failure(
107 CasErrc::Inconclusive,
108 "Lagrange candidate omits one or more variables or multipliers",
109 operation);
110 }
111 for (const auto& eq : equations) {
112 auto residual = eq;
113 for (const auto& [variable, value] : full_solution)
114 residual = residual->substitute(variable, value);
115 auto proof = check_zero_residual(residual, context);
116 if (!proof) return LagrangeResult::failure(proof.error());
117 if (!std::holds_alternative<ProvedZeroResidual>(proof.value())) {
118 return LagrangeResult::failure(
119 CasErrc::Inconclusive,
120 "Lagrange candidate does not verify against the full stationarity system",
121 operation);
122 }
123 }
124
125 std::map<std::string, std::shared_ptr<SymbolicExpr>> filtered;
126 for (const auto& [name, val] : full_solution) {
127 if (var_set.count(name)) {
128 filtered[name] = val;
129 }
130 }
131 if (!vector_calculus_point_has_vars(filtered, vars)) {
132 return LagrangeResult::failure(
133 CasErrc::Inconclusive,
134 "Lagrange candidate omits one or more variables",
135 operation);
136 }
137 verified.push_back(std::move(filtered));
138 }
139
140 return LagrangeResult::success(std::move(verified));
141}

144static LagrangeResult lagrange_multipliers_strict(
145 const std::shared_ptr<SymbolicExpr>& f,
146 const std::vector<std::shared_ptr<SymbolicExpr>>& constraints,
147 const std::vector<std::string>& vars, ComputationContext& context, const std::string& operation)
148{
149 const size_t n = vars.size();
150 const size_t m = constraints.size();
151 auto gradient = lagrange_gradient_strict(f, vars, operation);
152 if (!gradient) {
153 return LagrangeResult::failure(gradient.error());
154 }
155 std::vector<VectorField> grad_constraints;
156 grad_constraints.reserve(m);
157 for (const auto& constraint : constraints) {
158 auto partials = lagrange_gradient_strict(constraint, vars, operation);
159 if (!partials) {
160 return LagrangeResult::failure(partials.error());
161 }
162 grad_constraints.push_back(std::move(partials.value()));
163 }
164 auto lambda_names = lagrange_multiplier_names(m);
165 VectorField equations;
166 equations.reserve(n + m);
167 for (size_t i = 0; i < n; ++i) {
168 auto equation = lagrange_stationarity_equation(
169 gradient.value(), grad_constraints, lambda_names, i);
170 auto checked = vector_calculus_simplify_strict(equation, operation,
171 "Lagrange stationarity equation is outside the supported domain");
172 if (!checked) {
173 return LagrangeResult::failure(checked.error());
174 }
175 equations.push_back(std::move(checked.value()));
176 }
177 for (const auto& constraint : constraints) equations.push_back(constraint);
178 std::vector<std::string> all_vars;
179 all_vars.reserve(n + m);
180 all_vars.insert(all_vars.end(), vars.begin(), vars.end());
181 all_vars.insert(all_vars.end(), lambda_names.begin(), lambda_names.end());
182 auto solutions = solve_lagrange_strict(equations, all_vars, context, operation);
183 if (!solutions) {
184 return solutions;
185 }
186 return verify_lagrange_solutions(solutions.value(), equations, all_vars, vars, context, operation);
187}

189LagrangeResult lagrange_multipliers_checked(
190 const std::shared_ptr<SymbolicExpr>& f,
191 const std::vector<std::shared_ptr<SymbolicExpr>>& constraints,
192 const std::vector<std::string>& vars,
193 ComputationContext& context)
194{
195 const std::string operation = "lagrange_multipliers";
196 auto valid = vector_calculus_validate_expr_vars(f, vars, context, operation);
197 if (!valid) {
198 return LagrangeResult::failure(valid.error());
199 }
200 auto distinct = vector_calculus_validate_distinct_vars(vars, vars.size(), context, operation);
201 if (!distinct) {
202 return LagrangeResult::failure(distinct.error());
203 }
204 if (constraints.empty()) {
205 return LagrangeResult::failure(CasErrc::InvalidArgument,
206 "constraint list cannot be empty",
207 operation);
208 }
209 for (const auto& constraint : constraints) {
210 if (!constraint || !LMCAS::detail::node(constraint)) {
211 return LagrangeResult::failure(CasErrc::InvalidArgument,
212 "constraint expressions cannot be null",
213 operation);
214 }
215 }
216 auto step = context.consume_steps(vars.size() * constraints.size() * 6 +
217 vars.size() * 6 + constraints.size() * 4 + 8,
218 operation);
219 if (!step) {
220 return LagrangeResult::failure(step.error());
221 }
222
223 try {
224 return lagrange_multipliers_strict(f, constraints, vars, context, operation);
225 } catch (const std::bad_alloc&) {
226 return LagrangeResult::failure(CasErrc::ResourceLimit,
227 "Lagrange multiplier allocation failed",
228 operation);
229 } catch (const std::exception& e) {
230 return LagrangeResult::failure(CasErrc::InternalInvariant,
231 e.what(), operation);
232 }
233}
