616 Poly reduce(Poly p, const std::vector<Poly>& G) {
617 Poly r(p.num_vars);
618
619 while (!p.is_zero()) {
620 bool reduced = false;
621 Monomial p_lm = p.LM();
622
623 for (const auto& g : G) {
624 if (divides_mono(g.LM(), p_lm)) {
625
626 Term factor;
627 factor.coeff = p.LC() / g.LC();
628 factor.mono = div_mono(p_lm, g.LM());
629
630 Poly term_g = mul_poly_term(g, factor);
631 p = sub_poly(p, term_g);
632 reduced = true;
633 break;
634 }
635 }
636
637 if (!reduced) {
638 Term lt = p.LT();
639 r.add_term(lt.mono, lt.coeff);
640
641 p.terms.erase(p.terms.begin());
642 }
643 }
644 return r;

665 for (size_t i = 0; i < lm_f.size() && i < lm_g.size(); ++i) {
666 if (lm_f[i] > 0 && lm_g[i] > 0) return false;
667 }
668 return true;
669 }

671 bool chain_criterion(const std::vector<Poly>& G, size_t i, size_t j,
672 const std::set<std::pair<size_t,size_t>>& processed_pairs) {
673 Monomial L = lcm_mono(G[i].LM(), G[j].LM());
674 for (size_t k = 0; k < G.size(); ++k) {
675 if (k == i || k == j) continue;
676 if (G[k].is_zero()) continue;
677 if (divides_mono(G[k].LM(), L)) {
678 auto pair_ik = (i < k) ? std::make_pair(i, k) : std::make_pair(k, i);
679 auto pair_kj = (k < j) ? std::make_pair(k, j) : std::make_pair(j, k);
680 if (processed_pairs.count(pair_ik) && processed_pairs.count(pair_kj)) {
681 return true;
682 }
683 }
684 }
685 return false;
686 }

688 struct SugarPair {
689 size_t i, j;
690 int sugar_degree;
691
692 bool operator>(const SugarPair& other) const {
693 return sugar_degree > other.sugar_degree;
694 }

697 int compute_spoly_sugar(const Poly& f, const Poly& g) {
698 Monomial L = lcm_mono(f.LM(), g.LM());
699
700 int deg_L = 0;
701 for (int e : L) deg_L += e;
702
703 int deg_lm_f = 0;
704 for (int e : f.LM()) deg_lm_f += e;
705
706 int deg_lm_g = 0;
707 for (int e : g.LM()) deg_lm_g += e;
708
709 int sugar_f = f.sugar + deg_L - deg_lm_f;
710 int sugar_g = g.sugar + deg_L - deg_lm_g;
711
712 return std::max(sugar_f, sugar_g);
713 }

717std::vector<SymbolicExpr> Solver::groebner_basis(
718 const std::vector<SymbolicExpr>& polynomials,
719 const std::vector<std::string>& variables)
720{
721
722 PolyContext ctx(variables);
723
724 std::vector<Poly> G;
725 for (const auto& expr : polynomials) {
726 Poly p = to_poly(expr, ctx);
727 if (!p.is_zero()) G.push_back(p);
728 }
729
730 size_t max_vars = ctx.ext_vars.size();
731 for (auto& p : G) {
732 if (p.num_vars < max_vars) {
733 Poly resized(max_vars);
734 for (auto& [m, c] : p.terms) {
735 Monomial extended = m;
736 extended.resize(max_vars, 0);
737 resized.add_term(extended, c);
738 }
739 p = resized;
740 }
741 }
742
743 for (auto& p : G) {
744 if (!p.is_zero()) {
745 int max_deg = 0;
746 for (const auto& [m, c] : p.terms) {
747 int d = 0;
748 for (int e : m) d += e;
749 max_deg = std::max(max_deg, d);
750 }
751 p.sugar = max_deg;
752 }
753 }
754
755 std::priority_queue<SugarPair, std::vector<SugarPair>, std::greater<SugarPair>> pairs;
756 for (size_t i = 0; i < G.size(); ++i) {
757 for (size_t j = i + 1; j < G.size(); ++j) {
758 int sugar = compute_spoly_sugar(G[i], G[j]);
759 pairs.push({i, j, sugar});
760 }
761 }
762
763 std::set<std::pair<size_t, size_t>> processed_pairs;
764
765 while (!pairs.empty()) {
766 auto [i, j, sugar_deg] = pairs.top();
767 pairs.pop();
768
769 auto canonical_pair = (i < j) ? std::make_pair(i, j) : std::make_pair(j, i);
770 processed_pairs.insert(canonical_pair);
771
772 if (coprime_leading_monomials(G[i], G[j])) {
773 continue;
774 }
775
776 if (chain_criterion(G, i, j, processed_pairs)) {
777 continue;
778 }
779
780 Poly S = s_poly(G[i], G[j]);
781
782 if (S.num_vars < max_vars) {
783 Poly resized(max_vars);
784 for (auto& [m, c] : S.terms) {
785 Monomial extended = m;
786 extended.resize(max_vars, 0);
787 resized.add_term(extended, c);
788 }
789 S = resized;
790 }
791
792 Poly r = reduce(S, G);
793
794 if (!r.is_zero()) {
795
796 r.sugar = sugar_deg;
797 size_t new_idx = G.size();
798 G.push_back(r);
799 for (size_t k = 0; k < new_idx; ++k) {
800 int new_sugar = compute_spoly_sugar(G[k], G[new_idx]);
801 pairs.push({k, new_idx, new_sugar});
802 }
803 }
804 }
805
806 std::vector<SymbolicExpr> result;
807 for (const auto& p : G) {
808 result.push_back(from_poly_ext(p, ctx, variables));
809 }
810 return result;
811}

814static PolynomialSystemResult solve_polynomial_system_impl(
815 const std::vector<SymbolicExpr>& equations,
816 const std::vector<std::string>& variables,
817 ComputationContext& context) {
818 std::vector<SymbolicExpr> cleared_equations;
819 std::vector<std::shared_ptr<SymbolicExpr>> denom_constraints;
820 cleared_equations.reserve(equations.size());
821 for (const auto& eq : equations) {
822 if (!LMCAS::detail::node(eq)) return PolynomialSolutions{};
823 std::vector<std::shared_ptr<const SymbolicNode>> den_factors;
824 std::vector<std::shared_ptr<SymbolicExpr>> den_local_constraints;
825 if (!collect_denominator_factors(
826 LMCAS::detail::node(eq), den_factors, den_local_constraints))
827 return PolynomialSolutions{};
828 auto denom_expr = multiply_factors(den_factors);
829 auto cleared = to_ptr(eq);
830 if (!den_factors.empty()) {
831 if (auto add = std::dynamic_pointer_cast<const AddNode>(LMCAS::detail::node(cleared))) {
832 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
833 new_ops.reserve(add->operands().size());
834 for (const auto& op : add->operands()) {
835 auto prod = multiply_no_expand(op, den_factors);
836 new_ops.push_back(LMCAS::detail::node(prod));
837 }
838 cleared = LMCAS::detail::make_expression_ptr(LMCAS::detail::make_node<AddNode>(new_ops));
839 } else {
840 cleared = multiply_no_expand(LMCAS::detail::node(cleared), den_factors);
841 }
842 } else {
843 cleared = cleared->simplify();
844 }
845 if (!cleared || !LMCAS::detail::node(cleared) ||
846 !is_polynomial_node(LMCAS::detail::node(cleared)))
847 return PolynomialSolutions{};
848 cleared_equations.push_back(*cleared);
849 for (const auto& c : den_local_constraints) {
850 denom_constraints.push_back(c);
851 }
852 }
853 if (cleared_equations.size() == 1 && variables.size() == 1) {
854 auto solved = solve_finite_checked(
855 detail::make_expression_ptr(cleared_equations[0]),
856 variables[0], context, SolveOptions{});
857 if (!solved)
858 return PolynomialSystemResult::failure(solved.error());
859 auto roots = std::move(solved.value());
860 PolynomialSolutions single_solutions;
861 for (const auto& r : roots) {
862 single_solutions.push_back({{variables[0], *r}});
863 }
864 if (denom_constraints.empty()) return single_solutions;
865 PolynomialSolutions filtered;
866 filtered.reserve(single_solutions.size());
867 for (const auto& sol : single_solutions) {
868 bool ok = true;
869 for (const auto& den : denom_constraints) {
870 auto sub = den;
871 for (const auto& [name, val] : sol) {
872 sub = sub->substitute(name, LMCAS::detail::make_expression_ptr(val));
873 if (!sub) break;
874 }
875 if (!sub) continue;
876 sub = sub->simplify();
877 if (sub && sub->is_zero()) {
878 ok = false;
879 break;
880 }
881 }
882 if (ok) filtered.push_back(sol);
883 }
884 return filtered;
885 }
886 auto G_basis = Solver::groebner_basis(cleared_equations, variables);
887 std::vector<std::shared_ptr<SymbolicExpr>> basis;
888 basis.reserve(G_basis.size());
889 for (const auto& g : G_basis) {
890 auto g_ptr = LMCAS::detail::make_expression_ptr(g);
891 auto simp = g_ptr->simplify();
892 if (simp && !simp->is_zero()) basis.push_back(simp);
893 }
894 if (variables.empty()) {
895 for (const auto& p : basis) {
896 if (p->is_number() && !p->is_zero())
897 return PolynomialSolutions{};
898 }
899 return PolynomialSolutions{PolynomialSolution{}};
900 }
901 auto substitute_all = [&](const std::shared_ptr<SymbolicExpr>& expr,
902 const PolynomialSolution& subs) {
903 auto res = expr;
904 for (const auto& [name, val] : subs) {
905 res = res->substitute(name, LMCAS::detail::make_expression_ptr(val));
906 if (!res) return std::shared_ptr<SymbolicExpr>(nullptr);
907 }
908 return res->simplify();
909 };
910 auto solve_rec = [&](auto&& self, int var_pos,
911 const PolynomialSolution& partial)
912 -> PolynomialSystemResult {
913 std::vector<std::shared_ptr<SymbolicExpr>> reduced;
914 reduced.reserve(basis.size());
915 for (const auto& p : basis) {
916 auto r = substitute_all(p, partial);
917 if (!r) continue;
918 if (r->is_zero()) continue;
919 bool depends = false;
920 for (int i = 0; i <= var_pos && i < (int)variables.size(); ++i) {
921 if (contains(*r, variables[i])) {
922 depends = true;
923 break;
924 }
925 }
926 if (!depends) {
927 if (r->is_number() && !r->is_zero())
928 return PolynomialSolutions{};
929 continue;
930 }
931 reduced.push_back(r);
932 }
933 if (var_pos < 0) {
934 return PolynomialSolutions{partial};
935 }
936 const auto& curr_var = variables[var_pos];
937 bool curr_var_appears = false;
938 std::shared_ptr<SymbolicExpr> target = nullptr;
939 int best_deg = std::numeric_limits<int>::max();
940 for (const auto& r : reduced) {
941 if (!contains(*r, curr_var)) continue;
942 curr_var_appears = true;
943 bool has_other = false;
944 for (int i = 0; i < var_pos; ++i) {
945 if (contains(*r, variables[i])) {
946 has_other = true;
947 break;
948 }
949 }
950 if (has_other) continue;
951 auto poly = symbolic_to_poly<SymbolicPolyCoeff>(r, curr_var);
952 int deg = poly.degree();
953 if (deg >= 1 && deg < best_deg) {
954 best_deg = deg;
955 target = r;
956 }
957 }
958 if (!curr_var_appears) {
959 auto next_partial = partial;
960 next_partial.insert_or_assign(curr_var, *SymbolicExpr::variable(curr_var));
961 return self(self, var_pos - 1, next_partial);
962 }
963 if (!target) return PolynomialSolutions{};
964 auto solved = solve_finite_checked(
965 target, curr_var, context, SolveOptions{});
966 if (!solved)
967 return PolynomialSystemResult::failure(solved.error());
968 auto roots = std::move(solved.value());
969 if (roots.empty()) return PolynomialSolutions{};
970 PolynomialSolutions results;
971 for (const auto& r : roots) {
972 auto next_partial = partial;
973 next_partial.insert_or_assign(curr_var, *r);
974 auto sub_res = self(self, var_pos - 1, next_partial);
975 if (!sub_res) return sub_res;
976 auto& values = sub_res.value();
977 results.insert(results.end(),
978 std::make_move_iterator(values.begin()),
979 std::make_move_iterator(values.end()));
980 }
981 return results;
982 };
983 PolynomialSolution empty;
984 auto solved_candidates = solve_rec(
985 solve_rec, static_cast<int>(variables.size()) - 1, empty);
986 if (!solved_candidates) return solved_candidates;
987 auto candidates = std::move(solved_candidates.value());
988 if (denom_constraints.empty()) return candidates;
989 PolynomialSolutions filtered;
990 filtered.reserve(candidates.size());
991 for (const auto& sol : candidates) {
992 bool ok = true;
993 for (const auto& den : denom_constraints) {
994 auto sub = den;
995 for (const auto& [name, val] : sol) {
996 sub = sub->substitute(name, LMCAS::detail::make_expression_ptr(val));
997 if (!sub) break;
998 }
999 if (!sub) continue;
1000 sub = sub->simplify();
1001 if (sub && sub->is_zero()) {
1002 ok = false;
1003 break;
1004 }
1005 }
1006 if (ok) filtered.push_back(sol);
1007 }
1008 return filtered;
1009}

1044 const std::vector<SymbolicExpr>& polynomials,
1045 const std::vector<std::string>& variables)
1046{
1047
1048 auto gb_exprs = groebner_basis(polynomials, variables);
1049 if (gb_exprs.empty()) return {};
1050
1051 PolyContext ctx(variables);
1052 std::vector<Poly> G;
1053 for (const auto& expr : gb_exprs) {
1054 Poly p = to_poly(expr, ctx);
1055 if (!p.is_zero()) G.push_back(p);
1056 }
1057
1058 size_t max_vars = ctx.ext_vars.size();
1059 for (auto& p : G) {
1060 if (p.num_vars < max_vars) {
1061 Poly resized(max_vars);
1062 for (auto& [m, c] : p.terms) {
1063 Monomial extended = m;
1064 extended.resize(max_vars, 0);
1065 resized.add_term(extended, c);
1066 }
1067 p = resized;
1068 }
1069 }
1070
1071 std::vector<bool> marked(G.size(), false);
1072 for (size_t i = 0; i < G.size(); ++i) {
1073 if (marked[i]) continue;
1074 for (size_t j = 0; j < G.size(); ++j) {
1075 if (i == j || marked[j]) continue;
1076 if (divides_mono(G[j].LM(), G[i].LM())) {
1077 marked[i] = true;
1078 break;
1079 }
1080 }
1081 }
1082 std::vector<Poly> minimal;
1083 for (size_t i = 0; i < G.size(); ++i) {
1084 if (!marked[i]) minimal.push_back(G[i]);
1085 }
1086
1087 for (size_t i = 0; i < minimal.size(); ++i) {
1088 std::vector<Poly> others;
1089 for (size_t j = 0; j < minimal.size(); ++j) {
1090 if (j != i) others.push_back(minimal[j]);
1091 }
1092 minimal[i] = reduce(minimal[i], others);
1093 }
1094
1095 std::vector<Poly> reduced;
1096 for (auto& p : minimal) {
1097 if (!p.is_zero()) reduced.push_back(p);
1098 }
1099
1100 for (auto& p : reduced) {
1101 Rational lc = p.LC();
1102 if (lc != Rational(0) && lc != Rational(1)) {
1103 Rational inv = Rational(1) / lc;
1104 Poly monic(p.num_vars);
1105 for (auto& [m, c] : p.terms) {
1106 monic.add_term(m, c * inv);
1107 }
1108 p = monic;
1109 }
1110 }
1111
1112 std::vector<SymbolicExpr> result;
1113 for (const auto& p : reduced) {
1114 result.push_back(from_poly_ext(p, ctx, variables));
1115 }
1116 return result;
