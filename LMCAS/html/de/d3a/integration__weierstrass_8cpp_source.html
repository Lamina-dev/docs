7/// 判断表达式是否仅由 var 通过 sin(var)/cos(var)/tan(var) 以及常数、四则、整数幂构成，
8/// 即关于 sin/cos 的有理函数。含有其它依赖 var 的函数（exp/ln/sqrt 等）时返回 false。
9bool weier_is_rational_trig(const std::shared_ptr<const SymbolicNode>& node, const std::string& var) {
10 if (!node) return true;
11 if (auto vn = std::dynamic_pointer_cast<const VariableNode>(node)) {
12 /// 裸 var 不允许（如 x*sin(x) 不是 sin/cos 的有理函数）
13 return vn->name() != var;
14 }
15 if (std::dynamic_pointer_cast<const NumberNode>(node)) return true;
16 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
17 for (auto& op : add->operands()) if (!weier_is_rational_trig(op, var)) return false;
18 return true;
19 }
20 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
21 for (auto& op : mul->operands()) if (!weier_is_rational_trig(op, var)) return false;
22 return true;
23 }
24 if (auto pw = std::dynamic_pointer_cast<const PowerNode>(node)) {
25 /// 指数必须是不依赖 var 的整数常数
26 auto en = std::dynamic_pointer_cast<const NumberNode>(pw->exponent());
27 if (!en) return false;
28 return weier_is_rational_trig(pw->base(), var);
29 }
30 if (auto fn = std::dynamic_pointer_cast<const FunctionNode>(node)) {
31 using FT = FunctionNode::FuncType;
32 if ((fn->type() == FT::Sin || fn->type() == FT::Cos || fn->type() == FT::Tan ||
33 fn->type() == FT::Sec || fn->type() == FT::Csc || fn->type() == FT::Cot) &&
34 fn->arguments().size() == 1) {
35 /// 参数必须恰为 var
36 auto av = std::dynamic_pointer_cast<const VariableNode>(fn->arguments()[0]);
37 if (av && av->name() == var) return true;
38 /// 参数不依赖 var 时也算常数
39 return !expression_depends_on_variable(fn->arguments()[0], var);
40 }
41 /// 其它函数：仅当不依赖 var 才允许
42 for (auto& a : fn->arguments()) if (expression_depends_on_variable(a, var)) return false;
43 return true;
44 }
45 return false;
46}

48/// 是否至少包含一个 sin(var)/cos(var)/tan(var)... 形式（确保确实是三角有理函数）
49bool weier_has_trig_of_var(const std::shared_ptr<const SymbolicNode>& node, const std::string& var) {
50 if (!node) return false;
51 if (auto fn = std::dynamic_pointer_cast<const FunctionNode>(node)) {
52 using FT = FunctionNode::FuncType;
53 if ((fn->type() == FT::Sin || fn->type() == FT::Cos || fn->type() == FT::Tan ||
54 fn->type() == FT::Sec || fn->type() == FT::Csc || fn->type() == FT::Cot) &&
55 fn->arguments().size() == 1) {
56 auto av = std::dynamic_pointer_cast<const VariableNode>(fn->arguments()[0]);
57 if (av && av->name() == var) return true;
58 }
59 for (auto& a : fn->arguments()) if (weier_has_trig_of_var(a, var)) return true;
60 return false;
61 }
62 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
63 for (auto& op : add->operands()) if (weier_has_trig_of_var(op, var)) return true;
64 return false;
65 }
66 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
67 for (auto& op : mul->operands()) if (weier_has_trig_of_var(op, var)) return true;
68 return false;
69 }
70 if (auto pw = std::dynamic_pointer_cast<const PowerNode>(node)) {
71 return weier_has_trig_of_var(pw->base(), var) || weier_has_trig_of_var(pw->exponent(), var);
72 }
73 return false;
74}

76/// 递归替换：将 sin(var)/cos(var)/tan(var)... 替换为关于 t 的有理表达式。
77/// sin = 2t/(1+t²), cos = (1-t²)/(1+t²), tan = 2t/(1-t²)
78std::shared_ptr<const SymbolicNode> weier_replace(const std::shared_ptr<const SymbolicNode>& node,
79 const std::string& var, const std::string& tvar) {
80 if (!node) return node;
81 using FT = FunctionNode::FuncType;
82 if (auto fn = std::dynamic_pointer_cast<const FunctionNode>(node)) {
83 if (fn->arguments().size() == 1) {
84 auto av = std::dynamic_pointer_cast<const VariableNode>(fn->arguments()[0]);
85 bool is_var = av && av->name() == var;
86 if (is_var) {
87 auto t = SymbolicExpr::variable(tvar);
88 auto one = SymbolicExpr::number(1);
89 auto t2 = SymbolicExpr::power(t, SymbolicExpr::number(2));
90 auto onep = SymbolicExpr::add(one, t2); // 1+t²
91 auto onem = SymbolicExpr::add(one, SymbolicExpr::multiply(SymbolicExpr::number(-1), t2)); // 1-t²
92 auto two_t = SymbolicExpr::multiply(SymbolicExpr::number(2), t);
93 switch (fn->type()) {
94 case FT::Sin: return lamina::detail::node(SymbolicExpr::divide(two_t, onep));
95 case FT::Cos: return lamina::detail::node(SymbolicExpr::divide(onem, onep));
96 case FT::Tan: return lamina::detail::node(SymbolicExpr::divide(two_t, onem));
97 case FT::Csc: return lamina::detail::node(SymbolicExpr::divide(onep, two_t));
98 case FT::Sec: return lamina::detail::node(SymbolicExpr::divide(onep, onem));
99 case FT::Cot: return lamina::detail::node(SymbolicExpr::divide(onem, two_t));
100 default: break;
101 }
102 }
103 }
104 /// 其它函数：递归替换参数
105 std::vector<std::shared_ptr<const SymbolicNode>> new_args;
106 for (auto& a : fn->arguments()) new_args.push_back(weier_replace(a, var, tvar));
107 return lamina::detail::make_node<FunctionNode>(fn->type(), new_args);
108 }
109 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
110 std::vector<std::shared_ptr<const SymbolicNode>> ops;
111 for (auto& op : add->operands()) ops.push_back(weier_replace(op, var, tvar));
112 return lamina::detail::make_node<AddNode>(ops);
113 }
114 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
115 std::vector<std::shared_ptr<const SymbolicNode>> ops;
116 for (auto& op : mul->operands()) ops.push_back(weier_replace(op, var, tvar));
117 return lamina::detail::make_node<MultiplyNode>(ops);
118 }
119 if (auto pw = std::dynamic_pointer_cast<const PowerNode>(node)) {
120 return lamina::detail::make_node<PowerNode>(weier_replace(pw->base(), var, tvar),
121 weier_replace(pw->exponent(), var, tvar));
122 }
123 return node->clone();
124}

126/// 将（已做 sin/cos→t 代换的）表达式树递归求值为有理数对 (分子, 分母)，
127/// 分子分母均为关于 t 的多项式表达式。返回 nullopt 表示遇到无法处理的结构。
128typedef std::pair<std::shared_ptr<SymbolicExpr>, std::shared_ptr<SymbolicExpr>> RatPair;
129
130std::optional<RatPair> weier_to_rational(const std::shared_ptr<const SymbolicNode>& node,
131 const std::string& tvar) {
132 auto one = SymbolicExpr::number(1);
133 if (!node) return RatPair{SymbolicExpr::number(0), one};
134
135 if (std::dynamic_pointer_cast<const NumberNode>(node) ||
136 std::dynamic_pointer_cast<const VariableNode>(node)) {
137 return RatPair{lamina::detail::make_expression_ptr(node->clone()), one};
138 }
139 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
140 /// 累加：a/b + c/d = (a*d + c*b)/(b*d)
141 std::shared_ptr<SymbolicExpr> num = SymbolicExpr::number(0);
142 std::shared_ptr<SymbolicExpr> den = one;
143 for (auto& op : add->operands()) {
144 auto r = weier_to_rational(op, tvar);
145 if (!r) return std::nullopt;
146 auto [n2, d2] = *r;
147 auto new_num = SymbolicExpr::add(
148 SymbolicExpr::multiply(num, d2), SymbolicExpr::multiply(n2, den));
149 den = SymbolicExpr::multiply(den, d2)->simplify();
150 num = new_num->simplify();
151 }
152 return RatPair{num, den};
153 }
154 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
155 std::shared_ptr<SymbolicExpr> num = one;
156 std::shared_ptr<SymbolicExpr> den = one;
157 for (auto& op : mul->operands()) {
158 auto r = weier_to_rational(op, tvar);
159 if (!r) return std::nullopt;
160 auto [n2, d2] = *r;
161 num = SymbolicExpr::multiply(num, n2)->simplify();
162 den = SymbolicExpr::multiply(den, d2)->simplify();
163 }
164 return RatPair{num, den};
165 }
166 if (auto pw = std::dynamic_pointer_cast<const PowerNode>(node)) {
167 auto en = std::dynamic_pointer_cast<const NumberNode>(pw->exponent());
168 if (!en) return std::nullopt;
169 long long e;
170 if (std::holds_alternative<BigInt>(en->value())) e = (long long)std::get<BigInt>(en->value()).to_int();
171 else if (std::holds_alternative<lmmc_real_t>(en->value())) {
172 double d = std::get<lmmc_real_t>(en->value());
173 if (d != (long long)d) return std::nullopt;
174 e = (long long)d;
175 } else if (std::holds_alternative<Rational>(en->value())) {
176 double d = std::get<Rational>(en->value()).to_double();
177 if (d != (long long)d) return std::nullopt;
178 e = (long long)d;
179 } else return std::nullopt;
180
181 auto base = weier_to_rational(pw->base(), tvar);
182 if (!base) return std::nullopt;
183 auto [bn, bd] = *base;
184 bool neg = e < 0;
185 long long k = neg ? -e : e;
186 if (k > 32) return std::nullopt; // 防止指数爆炸
187 std::shared_ptr<SymbolicExpr> num = one, den = one;
188 for (long long i = 0; i < k; ++i) {
189 num = SymbolicExpr::multiply(num, bn)->simplify();
190 den = SymbolicExpr::multiply(den, bd)->simplify();
191 }
192 if (neg) std::swap(num, den);
193 return RatPair{num, den};
194 }
195 /// 其它结构（含未代换的函数）无法表示为 t 的有理函数
196 return std::nullopt;
197}

201std::shared_ptr<SymbolicExpr> WeierstrassStrategy::try_integrate(
202 const SymbolicExpr& expr, const std::string& var, Integrator&,
203 ComputationContext& computation, int depth) {
204 if (!lamina::detail::node(expr)) return nullptr;
205
206 /// 必须确实含有 sin/cos(var) 且整体为其有理函数
207 if (!weier_has_trig_of_var(lamina::detail::node(expr), var)) return nullptr;
208 if (!weier_is_rational_trig(lamina::detail::node(expr), var)) return nullptr;
209
210 const std::string tvar = "__weier_t";
211
212 /// 替换 sin/cos -> t 的有理式，并乘以 dx = 2/(1+t²) dt
213 auto replaced = lamina::detail::make_expression_ptr(weier_replace(lamina::detail::node(expr), var, tvar));
214 auto t = SymbolicExpr::variable(tvar);
215 auto t2 = SymbolicExpr::power(t, SymbolicExpr::number(2));
216 auto onep = SymbolicExpr::add(SymbolicExpr::number(1), t2);
217 auto dx = SymbolicExpr::divide(SymbolicExpr::number(2), onep); // 2/(1+t²)
218 auto integrand_raw = SymbolicExpr::multiply(replaced, dx);
219
220 /// 关键：被积函数是 sin/cos 的有理函数，代换后仍是 t 的有理函数，但
221 /// 嵌套分式 simplify() 无法约化。用「有理数对 (分子多项式, 分母多项式)」
222 /// 递归求值整棵表达式树，得到干净的 N(t)/D(t)，再交给有理函数积分。
223 auto rat = weier_to_rational(lamina::detail::node(integrand_raw), tvar);
224 if (!rat) return nullptr; // 出现非多项式结构，放弃
225 auto [num_poly, den_poly] = *rat;
226 if (!den_poly || lamina::detail::node(den_poly)->is_zero()) return nullptr;
227
228 /// 用多项式 GCD 约简 num/den，得到最简有理函数，避免 simplify() 把
229 /// 单一分式重新展开成分式之和（那样 RationalDecomposition 无法识别）。
230 std::shared_ptr<SymbolicExpr> integrand_t;
231 try {
232 Polynomial<Rational> Np = symbolic_to_poly<Rational>(num_poly->expand(), tvar);
233 Polynomial<Rational> Dp = symbolic_to_poly<Rational>(den_poly->expand(), tvar);
234 if (Dp.degree() < 0) { integrand_t = SymbolicExpr::divide(num_poly, den_poly); }
235 else {
236 auto g = Polynomial<Rational>::gcd(Np, Dp);
237 if (g.degree() >= 1) {
238 auto [q1, r1] = Np.div_mod(g);
239 auto [q2, r2] = Dp.div_mod(g);
240 if (r1.is_zero() && r2.is_zero()) { Np = q1; Dp = q2; }
241 }
242 auto np_expr = poly_to_symbolic(Np);
243 auto dp_expr = poly_to_symbolic(Dp);
244 integrand_t = SymbolicExpr::divide(np_expr, dp_expr);
245 }
246 } catch (...) {
247 integrand_t = SymbolicExpr::divide(num_poly, den_poly)->simplify();
248 }
249
250 /// 用有理分解策略直接积分（用独立的 Integrator 实例，避免污染外层
251 /// 积分器的循环检测状态导致结果被错误改写）。
252 Integrator inner;
253 RationalDecompositionStrategy rds;
254 auto integrated = rds.try_integrate(
255 *integrand_t, tvar, inner, computation, depth + 1);
256 if (!integrated) {
257 /// 退化为多项式（如常数 1）时，RationalDecomposition 可能不接受；
258 /// 直接用幂律积分兜底。
259 integrated = inner.integrate_recursive(*integrand_t, tvar, computation, 0);
260 }
261 if (!integrated) return nullptr;
262
263 /// 若结果仍含未求值积分节点，视为失败
264 if (expression_depends_on_variable(lamina::detail::node(integrated), tvar)) {
265 /// 检查是否残留 Calculus_Integral
266 bool has_uneval = false;
267 std::function<void(const std::shared_ptr<const SymbolicNode>&)> scan =
268 [&](const std::shared_ptr<const SymbolicNode>& n) {
269 if (!n) return;
270 if (auto fn = std::dynamic_pointer_cast<const FunctionNode>(n))
271 if (fn->type() == FunctionNode::FuncType::Calculus_Integral) has_uneval = true;
272 if (auto a = std::dynamic_pointer_cast<const AddNode>(n)) for (auto& o : a->operands()) scan(o);
273 if (auto m = std::dynamic_pointer_cast<const MultiplyNode>(n)) for (auto& o : m->operands()) scan(o);
274 if (auto p = std::dynamic_pointer_cast<const PowerNode>(n)) { scan(p->base()); scan(p->exponent()); }
275 if (auto f = std::dynamic_pointer_cast<const FunctionNode>(n)) for (auto& o : f->arguments()) scan(o);
276 };
277 scan(lamina::detail::node(integrated));
278 if (has_uneval) return nullptr;
279 }
280
281 /// 回代 t = tan(x/2)
282 auto half_x = SymbolicExpr::multiply(SymbolicExpr::number(Rational(1, 2)),
283 SymbolicExpr::variable(var));
284 auto tan_half = lamina::detail::make_expression_ptr(lamina::detail::make_node<FunctionNode>(
285 FunctionNode::FuncType::Tan, std::vector<std::shared_ptr<const SymbolicNode>>{lamina::detail::node(half_x)}));
286 auto result = integrated->substitute(tvar, tan_half);
287 if (!result) return nullptr;
288 return result->simplify();
289}
