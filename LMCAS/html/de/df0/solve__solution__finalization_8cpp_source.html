28Result<void> apply_root_representation_policy(ExprPtr& root, const SolveOptions& options) {

65Result<Tribool> logical_condition_truth(const LogicalNode& logic,
66 const FactsQuery& facts, ComputationContext& context) {
67 auto left = solve_condition_truth(detail::make_expression_ptr(logic.left()), facts, context);
68 if (!left) { return left; }
69 if (logic.op() == LogicalNode::Op::Not) { return negate_truth(left.value()); }
70 auto right = solve_condition_truth(detail::make_expression_ptr(logic.right()), facts, context);
71 if (!right) { return right; }
72 auto l = left.value();
73 auto r = right.value();
74 if (logic.op() == LogicalNode::Op::Implies) { l = negate_truth(l); }
75 if (logic.op() == LogicalNode::Op::And) {
76 if (l == Tribool::False || r == Tribool::False) { return Tribool::False; }
77 return l == Tribool::True && r == Tribool::True ? Tribool::True : Tribool::Unknown;
78 }
79 if (l == Tribool::True || r == Tribool::True) { return Tribool::True; }
80 return l == Tribool::False && r == Tribool::False ? Tribool::False : Tribool::Unknown;
81}

83Result<Tribool> equality_condition_truth(const ExprPtr& difference, Tribool nonzero,
84 RelationOp op, ComputationContext& context) {
85 if (nonzero == Tribool::Unknown) {
86 auto residual = check_zero_residual(difference, context);
87 if (!residual) { return Result<Tribool>::failure(residual.error()); }
88 if (std::holds_alternative<ProvedZeroResidual>(residual.value())) { nonzero = Tribool::False; }
89 }
90 if (op == RelationOp::NEQ) { return nonzero; }
91 return negate_truth(nonzero);
92}

94Result<Tribool> relational_condition_truth(const ExprPtr& expression,
95 const RelationalNode& relation, const FactsQuery& facts, ComputationContext& context) {
96 auto defined = detail::query_definedness(detail::node(expression), facts, Domain::Complex, context);
97 if (!defined) { return Result<Tribool>::failure(defined.error()); }
98 if (defined.value() != Tribool::True) { return Tribool::Unknown; }
99 auto difference = equation_residual(expression)->simplify();
100 auto nz = detail::query_nonzero_value(detail::node(difference), facts, Domain::Complex, context);
101 if (!nz) { return nz; }
102 if (relation.op() == RelationOp::EQ || relation.op() == RelationOp::NEQ) {
103 return equality_condition_truth(difference, nz.value(), relation.op(), context);
104 }
105 if (relation.op() == RelationOp::LT || relation.op() == RelationOp::LEQ) {
106 difference = SymbolicExpr::multiply(SymbolicExpr::number(-1), difference)->simplify();
107 }
108 if (relation.op() == RelationOp::GEQ || relation.op() == RelationOp::LEQ) {
109 return detail::query_nonnegative_value(detail::node(difference), facts, context);
110 }
111 return detail::query_positive_value(detail::node(difference), facts, context);
112}
113Result<Tribool> solve_condition_truth(const ExprPtr& expression,
114 const FactsQuery& facts, ComputationContext& context) {
115 auto step = context.consume_steps(1, "solve.conditions");
116 if (!step) { return Result<Tribool>::failure(step.error()); }
117 const auto node = detail::node(expression);
118 if (auto logic = std::dynamic_pointer_cast<const LogicalNode>(node)) {
119 return logical_condition_truth(*logic, facts, context);
120 }
121 auto relation = std::dynamic_pointer_cast<const RelationalNode>(node);
122 if (!relation) { return Tribool::Unknown; }
123 return relational_condition_truth(expression, *relation, facts, context);
124}

134 Result<bool> finish_conditions(std::vector<ExprPtr>& conditions) const {
135 std::vector<ExprPtr> remaining;
136 for (const auto& condition : conditions) {
137 auto truth = solve_condition_truth(condition, facts, context);
138 if (!truth) { return Result<bool>::failure(truth.error()); }
139 if (truth.value() == Tribool::False) { return false; }
140 if (truth.value() == Tribool::Unknown) { append_condition(remaining, condition); }
141 }
142 conditions = std::move(remaining);
143 return true;
144 }

146 Result<bool> finalize_value(const ExprPtr& value, std::vector<ExprPtr>& conditions) const {
147 for (auto& condition : conditions) {
148 condition = substitute_raw(condition, variable, value);
149 }
150 auto bound = substitute_raw(original, variable, value);
151 auto defined = detail::query_definedness(detail::node(bound), facts, domain, context);
152 if (!defined) { return Result<bool>::failure(defined.error()); }
153 if (defined.value() == Tribool::False) { return false; }
154 auto domain_conditions = detail::domain_constraints(detail::node(bound), facts, domain, context);
155 if (!domain_conditions) { return Result<bool>::failure(domain_conditions.error()); }
156 if (!domain_conditions.value()) {
157 return Result<bool>::failure(CasErrc::Inconclusive,
158 "candidate domain cannot be completely expressed", "solve.finalize");
159 }
160 for (const auto& condition : *domain_conditions.value()) { append_condition(conditions, condition); }
161 auto valid = finish_conditions(conditions);
162 if (!valid || !valid.value()) { return valid; }
163 auto residual = check_zero_residual(equation_residual(bound), context);
164 if (!residual) { return Result<bool>::failure(residual.error()); }
165 if (!std::holds_alternative<ProvedZeroResidual>(residual.value())) {
166 auto pointwise_nonzero = detail::query_nonzero_value(detail::node(equation_residual(bound)->simplify()), facts, domain, context);
167 if (!pointwise_nonzero) { return Result<bool>::failure(pointwise_nonzero.error()); }
168 if (pointwise_nonzero.value() == Tribool::True) { return false; }
169 if (pointwise_nonzero.value() == Tribool::Unknown) {
170 append_condition(conditions, equation_predicate(bound));
171 }
172 }
173 return true;
174 }

176 template<class Solutions>
177 SolveResult filter_values(Solutions& solutions) const {
178 Solutions kept;
179 for (auto& solution : solutions.values) {
180 auto valid = finalize_value(solution.value, solution.conditions);
181 if (!valid) { return SolveResult::failure(valid.error()); }
182 if (valid.value()) {
183 auto represented = apply_root_representation_policy(solution.value, options);
184 if (!represented) { return SolveResult::failure(represented.error()); }
185 kept.values.push_back(std::move(solution));
186 }
187 }
188 if (kept.values.empty()) { return SolutionSet{EmptySolutions{}}; }
189 return SolutionSet{std::move(kept)};
190 }

193 SolutionFinalizer(const ExprPtr& original, const std::string& variable,
194 const FactsQuery& facts, ComputationContext& context, Domain domain,
195 const SolveOptions& options)
196 : original(original), variable(variable), facts(facts), context(context),
197 domain(domain), options(options) {}

199 SolveResult finish(SolutionSet solutions) const {
200 if (auto finite = std::get_if<FiniteSolutions>(&solutions)) { return filter_values(*finite); }
201 if (auto parametric = std::get_if<ParametricSolutions>(&solutions)) { return filter_values(*parametric); }
202 auto obligations = detail::domain_constraints(detail::node(original), facts, domain, context);
203 if (!obligations) { return SolveResult::failure(obligations.error()); }
204 if (!obligations.value()) {
205 return SolveResult::failure(CasErrc::Inconclusive,
206 "original equation domain cannot be completely expressed", "solve.finalize");
207 }
208 if (auto conditional = std::get_if<ConditionalSolutions>(&solutions)) {
209 for (const auto& condition : *obligations.value()) {
210 append_condition(conditional->value.conditions, condition);
211 }
212 auto valid = finish_conditions(conditional->value.conditions);
213 if (!valid) { return SolveResult::failure(valid.error()); }
214 if (!valid.value()) { return SolutionSet{EmptySolutions{}}; }
215 return solutions;
216 }
217 auto conditions = std::move(*obligations.value());
218 auto valid = finish_conditions(conditions);
219 if (!valid) { return SolveResult::failure(valid.error()); }
220 if (!valid.value()) { return SolutionSet{EmptySolutions{}}; }
221 if (conditions.empty()) { return solutions; }
222 return SolutionSet{ConditionalSolutions{ConditionSet{variable,
223 equation_predicate(original), std::move(conditions)}}};
224 }
225};

228SolveResult detail::finalize_solution_set(const ExprPtr& original,
229 const std::string& variable, SolutionSet solutions, ComputationContext& context,
230 Domain domain, const SolveOptions& options) {
231 if (std::holds_alternative<EmptySolutions>(solutions)) { return solutions; }
232 AssumptionContext local_facts = context.assumptions() ?
233 *context.assumptions() : AssumptionContext{};
234 if (domain == Domain::Real) {
235 auto declared = local_facts.assume_domain_checked(variable, Domain::Real);
236 if (!declared) { return SolveResult::failure(declared.error()); }
237 }
238 if (const auto* families = std::get_if<ParametricSolutions>(&solutions)) {
239 for (const auto& family : families->values) {
240 for (const auto& parameter : family.integer_parameters) {
241 auto declared = local_facts.assume_domain_checked(parameter, Domain::Integer);
242 if (!declared) { return SolveResult::failure(declared.error()); }
243 }
244 }
245 }
246 detail::AssumptionFacts adapter(local_facts);
247 return SolutionFinalizer(original, variable, adapter, context, domain, options).finish(std::move(solutions));
248}
