18static bool match_commutative_recursive(const std::vector<std::shared_ptr<const SymbolicNode>>& p_ops,
19 const std::vector<std::shared_ptr<const SymbolicNode>>& t_ops,
20 std::vector<bool>& used_t,
21 size_t p_index,
22 const std::unordered_set<std::string>& wildcards,
23 MatchMap& results) {
24
25 if (p_index == p_ops.size()) {
26 return true;
27 }
28
29 for (size_t j = 0; j < t_ops.size(); ++j) {
30 if (!used_t[j]) {
31
32 MatchMap saved_results = results;
33
34 if (match_recursive(p_ops[p_index], t_ops[j], wildcards, results)) {
35 used_t[j] = true;
36 if (match_commutative_recursive(p_ops, t_ops, used_t, p_index + 1, wildcards, results)) {
37 return true;
38 }
39
40 used_t[j] = false;
41 results = saved_results;
42 } else {
43
44 results = saved_results;
45 }
46 }
47 }
48
49 return false;
50}

77template <typename Operation>
78static void record_remainder(const Operation& target, const std::vector<bool>& used,
79 const char* key, MatchMap& results) {
80 std::vector<std::shared_ptr<const SymbolicNode>> rest;
81 for (size_t index = 0; index < used.size(); ++index) {
82 if (!used[index]) {
83 rest.push_back(target.operands()[index]);
84 }
85 }
86 if (rest.empty()) {
87 return;
88 }
89 auto remainder = rest.size() == 1 ? rest.front() : combine_operands<Operation>(std::move(rest));
90 const auto found = results.find(key);
91 if (found == results.end()) {
92 results.emplace(key, detail::expression_from_node(remainder));
93 return;
94 }
95 std::vector<std::shared_ptr<const SymbolicNode>> operands;
96 if (const auto* existing = dynamic_cast<const Operation*>(detail::node(found->second).get())) {
97 operands.insert(operands.end(), existing->operands().begin(), existing->operands().end());
98 } else {
99 operands.push_back(detail::node(found->second));
100 }
101 operands.push_back(remainder);
102 found->second = detail::expression_from_node(combine_operands<Operation>(std::move(operands)));
103}

106static bool match_operation(const Operation& pattern, const Operation& target,
107 const std::unordered_set<std::string>& wildcards,
108 const char* rest_key, MatchMap& results) {
109 if (pattern.operands().size() > target.operands().size()) {
110 return false;
111 }
112 std::vector<bool> used(target.operands().size(), false);
113 if (!match_commutative_recursive(pattern.operands(), target.operands(), used, 0, wildcards, results)) {
114 return false;
115 }
116 record_remainder(target, used, rest_key, results);
117 return true;

120static bool match_function(const FunctionNode& pattern, const FunctionNode& target,
121 const std::unordered_set<std::string>& wildcards, MatchMap& results) {
122 if (pattern.type() != target.type() || pattern.arguments().size() != target.arguments().size()) {
123 return false;
124 }
125 for (size_t index = 0; index < pattern.arguments().size(); ++index) {
126 if (!match_recursive(pattern.arguments()[index], target.arguments()[index], wildcards, results)) {

133static bool match_recursive(const std::shared_ptr<const SymbolicNode>& pattern,
134 const std::shared_ptr<const SymbolicNode>& target,
135 const std::unordered_set<std::string>& wildcards, MatchMap& results) {
136 if (!pattern) {
137 return !target;
138 }
139 if (!target) {
140 return false;
141 }
142 std::string name;
143 if (is_wildcard(pattern, wildcards, name)) {
144 const auto existing = results.find(name);
145 if (existing != results.end()) {
146 return detail::node(existing->second)->equals(*target);
147 }
148 results.emplace(name, detail::expression_from_node(target));
149 return true;
150 }
151 if (pattern->type_priority() != target->type_priority()) {
152 return false;
153 }
154 if (pattern->is_number() || dynamic_cast<const VariableNode*>(pattern.get())) {
155 return pattern->equals(*target);
156 }
157 if (const auto* add = dynamic_cast<const AddNode*>(pattern.get())) {
158 return match_operation(*add, static_cast<const AddNode&>(*target), wildcards, "__Add_REST__", results);
159 }
160 if (const auto* multiply = dynamic_cast<const MultiplyNode*>(pattern.get())) {
161 return match_operation(*multiply, static_cast<const MultiplyNode&>(*target), wildcards, "__Mul_REST__", results);
162 }
163 if (const auto* power = dynamic_cast<const PowerNode*>(pattern.get())) {
164 const auto& other = static_cast<const PowerNode&>(*target);
165 return match_recursive(power->base(), other.base(), wildcards, results) &&
166 match_recursive(power->exponent(), other.exponent(), wildcards, results);
167 }
168 if (const auto* function = dynamic_cast<const FunctionNode*>(pattern.get())) {
169 return match_function(*function, static_cast<const FunctionNode&>(*target), wildcards, results);
170 }
171 return pattern->equals(*target);
172}
