19static bool match_commutative_recursive(const std::vector<std::shared_ptr<const SymbolicNode>>& p_ops,
20 const std::vector<std::shared_ptr<const SymbolicNode>>& t_ops,
21 std::vector<bool>& used_t,
22 size_t p_index,
23 const std::unordered_set<std::string>& wildcards,
24 MatchMap& results) {
25
26 if (p_index == p_ops.size()) {
27 return true;
28 }
29
30 for (size_t j = 0; j < t_ops.size(); ++j) {
31 if (!used_t[j]) {
32
33 MatchMap saved_results = results;
34
35 if (match_recursive(p_ops[p_index], t_ops[j], wildcards, results)) {
36 used_t[j] = true;
37 if (match_commutative_recursive(p_ops, t_ops, used_t, p_index + 1, wildcards, results)) {
38 return true;
39 }
40
41 used_t[j] = false;
42 results = saved_results;
43 } else {
44
45 results = saved_results;
46 }
47 }
48 }
49
50 return false;
51}

68static bool match_recursive(const std::shared_ptr<const SymbolicNode>& p_node,
69 const std::shared_ptr<const SymbolicNode>& t_node,
70 const std::unordered_set<std::string>& wildcards,
71 MatchMap& results) {
72 if (!p_node) return !t_node;
73 if (!t_node) return false;
74
75 std::string wname;
76 if (is_wildcard(p_node, wildcards, wname)) {
77
78 if (const auto existing = results.find(wname); existing != results.end()) {
79 return lamina::detail::node(existing->second)->equals(*t_node);
80 } else {
81 results.emplace(wname, lamina::detail::expression_from_node(t_node));
82 return true;
83 }
84 }
85
86 if (p_node->type_priority() != t_node->type_priority()) return false;
87
88 if (p_node->is_number() || std::dynamic_pointer_cast<const VariableNode>(p_node)) {
89 return p_node->equals(*t_node);
90 }
91
92 if (auto p_add = std::dynamic_pointer_cast<const AddNode>(p_node)) {
93 auto t_add = std::dynamic_pointer_cast<const AddNode>(t_node);
94
95 if (p_add->operands().size() > t_add->operands().size()) return false;
96
97 std::vector<bool> used(t_add->operands().size(), false);
98 if (match_commutative_recursive(p_add->operands(), t_add->operands(), used, 0, wildcards, results)) {
99
100 std::vector<std::shared_ptr<const SymbolicNode>> rest;
101 for (size_t i = 0; i < used.size(); ++i) {
102 if (!used[i]) {
103 rest.push_back(t_add->operands()[i]);
104 }
105 }
106
107 if (!rest.empty()) {
108
109 std::shared_ptr<const SymbolicNode> rest_node;
110 if (rest.size() == 1) rest_node = rest[0];
111 else rest_node = SymbolicFactory::create_add(rest);
112
113 if (const auto found = results.find("__Add_REST__"); found != results.end()) {
114 auto existing = found->second;
115
116 std::vector<std::shared_ptr<const SymbolicNode>> combined_ops;
117 if (auto existing_add = std::dynamic_pointer_cast<const AddNode>(lamina::detail::node(existing))) {
118 combined_ops.insert(combined_ops.end(), existing_add->operands().begin(), existing_add->operands().end());
119 } else {
120 combined_ops.push_back(lamina::detail::node(existing));
121 }
122 combined_ops.push_back(rest_node);
123 found->second = lamina::detail::expression_from_node(
124 SymbolicFactory::create_add(combined_ops));
125 } else {
126 results.emplace("__Add_REST__", lamina::detail::expression_from_node(rest_node));
127 }
128 }
129 return true;
130 }
131 return false;
132 }
133
134 if (auto p_mul = std::dynamic_pointer_cast<const MultiplyNode>(p_node)) {
135 auto t_mul = std::dynamic_pointer_cast<const MultiplyNode>(t_node);
136
137 if (p_mul->operands().size() > t_mul->operands().size()) return false;
138
139 std::vector<bool> used(t_mul->operands().size(), false);
140 if (match_commutative_recursive(p_mul->operands(), t_mul->operands(), used, 0, wildcards, results)) {
141
142 std::vector<std::shared_ptr<const SymbolicNode>> rest;
143 for (size_t i = 0; i < used.size(); ++i) {
144 if (!used[i]) {
145 rest.push_back(t_mul->operands()[i]);
146 }
147 }
148 if (!rest.empty()) {
149 std::shared_ptr<const SymbolicNode> rest_node;
150 if (rest.size() == 1) rest_node = rest[0];
151 else rest_node = SymbolicFactory::create_multiply(rest);
152
153 if (const auto found = results.find("__Mul_REST__"); found != results.end()) {
154 auto existing = found->second;
155 std::vector<std::shared_ptr<const SymbolicNode>> combined_ops;
156 if (auto existing_mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(existing))) {
157 combined_ops.insert(combined_ops.end(), existing_mul->operands().begin(), existing_mul->operands().end());
158 } else {
159 combined_ops.push_back(lamina::detail::node(existing));
160 }
161 combined_ops.push_back(rest_node);
162 found->second = lamina::detail::expression_from_node(
163 SymbolicFactory::create_multiply(combined_ops));
164 } else {
165 results.emplace("__Mul_REST__", lamina::detail::expression_from_node(rest_node));
166 }
167 }
168 return true;
169 }
170 return false;
171 }
172
173 if (auto p_pow = std::dynamic_pointer_cast<const PowerNode>(p_node)) {
174 auto t_pow = std::dynamic_pointer_cast<const PowerNode>(t_node);
175 return match_recursive(p_pow->base(), t_pow->base(), wildcards, results) &&
176 match_recursive(p_pow->exponent(), t_pow->exponent(), wildcards, results);
177 }
178
179 if (auto p_func = std::dynamic_pointer_cast<const FunctionNode>(p_node)) {
180 auto t_func = std::dynamic_pointer_cast<const FunctionNode>(t_node);
181 if (p_func->type() != t_func->type()) return false;
182 if (p_func->arguments().size() != t_func->arguments().size()) return false;
183 for (size_t i = 0; i < p_func->arguments().size(); ++i) {
184 if (!match_recursive(p_func->arguments()[i], t_func->arguments()[i], wildcards, results)) {
185 return false;
186 }
187 }
188 return true;
189 }
190
191 return p_node->equals(*t_node);
192}

202SymbolicExpr Matcher::replace(const SymbolicExpr& template_expr, const MatchMap& bindings, bool use_rest) {
203 if (!lamina::detail::node(template_expr)) return template_expr;
204
205 auto replaced = lamina::detail::node(template_expr);
206 std::set<std::string> occupied = lamina::free_variables(replaced);
207 for (const auto& [name, expression] : bindings) {
208 const auto names = lamina::free_variables(lamina::detail::node(expression));
209 occupied.insert(names.begin(), names.end());
210 }
211
212 std::vector<std::pair<std::string, std::string>> placeholders;
213 placeholders.reserve(bindings.size());
214 std::size_t suffix = 0;
215 for (const auto& [name, expression] : bindings) {
216 (void)expression;
217 std::string placeholder;
218 do {
219 placeholder = "__lmcas_match_binding_" + std::to_string(suffix++);
220 } while (!occupied.insert(placeholder).second);
221 replaced = lamina::substitute_free(
222 replaced, name, SymbolicFactory::create_variable(placeholder));
223 placeholders.emplace_back(name, std::move(placeholder));
224 }
225 for (const auto& [name, placeholder] : placeholders) {
226 replaced = lamina::substitute_free(
227 replaced, placeholder, lamina::detail::node(bindings.at(name)));
228 }
229
230 auto res = lamina::detail::expression_from_node(std::move(replaced));
231 if (use_rest) {
232
233 if (bindings.find("__Add_REST__") != bindings.end()) {
234 auto rest = bindings.at("__Add_REST__");
235
236 std::vector<std::shared_ptr<const SymbolicNode>> ops;
237 if (auto add_node = std::dynamic_pointer_cast<const AddNode>(lamina::detail::node(res))) {
238 ops = add_node->operands();
239 } else {
240 ops.push_back(lamina::detail::node(res));
241 }
242
243 if (auto rest_add = std::dynamic_pointer_cast<const AddNode>(lamina::detail::node(rest))) {
244 ops.insert(ops.end(), rest_add->operands().begin(), rest_add->operands().end());
245 } else {
246 ops.push_back(lamina::detail::node(rest));
247 }
248 res = lamina::detail::expression_from_node(SymbolicFactory::create_add(ops));
249 }
250
251 if (bindings.find("__Mul_REST__") != bindings.end()) {
252 auto rest = bindings.at("__Mul_REST__");
253
254 std::vector<std::shared_ptr<const SymbolicNode>> ops;
255 if (auto mul_node = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(res))) {
256 ops = mul_node->operands();
257 } else {
258 ops.push_back(lamina::detail::node(res));
259 }
260
261 if (auto rest_mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(rest))) {
262 ops.insert(ops.end(), rest_mul->operands().begin(), rest_mul->operands().end());
263 } else {
264 ops.push_back(lamina::detail::node(rest));
265 }
266 res = lamina::detail::expression_from_node(SymbolicFactory::create_multiply(ops));
267 }
268 }
269
270 return res;
271}

289 std::shared_ptr<const SymbolicNode> try_match(std::shared_ptr<const SymbolicNode> node) {
290 auto current_expr = lamina::detail::expression_from_node(node);
291 const auto& rules = engine.get_rules();
292 const AssumptionContext* ctx = context.assumptions().get();
293 for (const auto& rule : rules) {
294 MatchMap bindings;
295
296 if (Matcher::match(rule.pattern, current_expr, rule.wildcards, bindings)) {
297
298 // Evaluate condition: prefer assumption_condition when context is available
299 if (rule.assumption_condition && ctx) {
300 if (!rule.assumption_condition(bindings, ctx)) {
301 continue;
302 }
303 } else if (rule.assumption_condition && !ctx) {
304 // No context available; fall back to plain condition if present
305 if (rule.condition && !rule.condition(bindings)) {
306 continue;
307 }
308 } else if (rule.condition) {
309 if (!rule.condition(bindings)) {
310 continue;
311 }
312 }
313
314 SymbolicExpr new_expr = Matcher::replace(rule.replacement, bindings, true);
315
316 changed = true;
317 return lamina::detail::node(new_expr);
318 }
319 }
320 return node;
321 }

336 void finish_rewrite(std::shared_ptr<const SymbolicNode> node, bool child_changed,
337 bool original_changed) {

341 result = matched;

344 changed = original_changed;

654 body, node.variable(), point, node.direction()),
655 child_changed, original_changed);
656 }
657 void visit(const RootOfNode& node) override {

659 finish_rewrite(node.clone(), false, original_changed);
660 }

663Result<SymbolicExpr> RewriteEngine::apply_step_checked(
664 const SymbolicExpr& expr,
665 ComputationContext& context) const {
666 auto budget = context.consume_steps(1, "rewrite.apply_step");
667 if (!budget) return Result<SymbolicExpr>::failure(budget.error());
668 if (!lamina::detail::node(expr)) {
669 return Result<SymbolicExpr>::failure(
670 CasErrc::InvalidArgument, "rewrite expression must not be null",
671 "rewrite.apply_step");
672 }
673 RewriteVisitor v(*this, context);
674 lamina::detail::node(expr)->accept(v);
675 return Result<SymbolicExpr>::success(
676 lamina::detail::expression_from_node(v.get_result()));
677}

679Result<SymbolicExpr> RewriteEngine::apply_checked(
680 const SymbolicExpr& expr,
681 ComputationContext& context,
682 int max_iterations) const {
683 if (max_iterations < 0) {
684 return Result<SymbolicExpr>::failure(
685 CasErrc::InvalidArgument, "rewrite iteration count must be non-negative",
686 "rewrite.apply");
687 }
688 SymbolicExpr current = expr;
689 for (int i = 0; i < max_iterations; ++i) {
690 auto next_result = apply_step_checked(current, context);
691 if (!next_result) return next_result;
692 SymbolicExpr next = std::move(next_result.value());
693 if (lamina::detail::node(next)->equals(*lamina::detail::node(current))) {
694 return Result<SymbolicExpr>::success(std::move(current));
695 }
696 current = next;
697 }
698 return Result<SymbolicExpr>::success(std::move(current));
699}

701Result<SymbolicExpr> RewriteEngine::apply_step(const SymbolicExpr& expr) const {
702 ComputationContext context;
703 return apply_step_checked(expr, context);
704}

706Result<SymbolicExpr> RewriteEngine::apply(
707 const SymbolicExpr& expr,
708 int max_iterations) const {
709 ComputationContext context;
710 return apply_checked(expr, context, max_iterations);
711}
