17 auto candidate = node;
18 detail::SymbolicNodePtr coefficient =
19 detail::make_node<NumberNode>(BigInt(1));
20 if (const auto multiply =
21 std::dynamic_pointer_cast<const MultiplyNode>(node);
22 multiply && multiply->operands().size() == 2) {
23 const auto first_number =
24 std::dynamic_pointer_cast<const NumberNode>(
25 multiply->operands()[0]);
26 const auto second_number =
27 std::dynamic_pointer_cast<const NumberNode>(
28 multiply->operands()[1]);
29 if (first_number) {
30 coefficient = first_number;
31 candidate = multiply->operands()[1];
32 } else if (second_number) {
33 coefficient = second_number;
34 candidate = multiply->operands()[0];
35 }
36 }
37 const auto power =
38 std::dynamic_pointer_cast<const PowerNode>(candidate);
39 if (!power || !exact_integer_node(power->exponent(), 2)) {

44 if (!function || function->arguments().size() != 1 ||
45 (function->type() != FunctionNode::FuncType::Sin &&
46 function->type() != FunctionNode::FuncType::Cos)) {
47 return std::nullopt;
48 }
49 return TrigSquare{
50 function->type(), function->arguments()[0], coefficient};

53std::optional<ExprPtr> unwrap_trig_negated_argument(
54 const detail::SymbolicNodePtr& node, detail::RewriteBudget& budget) {
55 auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node);
56 if (!multiply) { return std::nullopt; }
57
58 bool found_negative_one = false;
59 std::vector<detail::SymbolicNodePtr> remaining;
60 budget.require_nodes(multiply->operands().size());
61 std::size_t size = 0;
62 remaining.reserve(multiply->operands().size());
63 for (const auto& operand : multiply->operands()) {
64 if (!found_negative_one && exact_integer_node(operand, -1)) {
65 found_negative_one = true;
66 continue;
67 }
68 size = budget.append_size(size, budget.measure(operand));
69 remaining.push_back(operand);
70 }
71 if (!found_negative_one || remaining.empty()) { return std::nullopt; }
72 if (remaining.size() == 1) {
73 return detail::make_expression_ptr(remaining.front());
74 }
75 return detail::make_expression_ptr(
76 SymbolicFactory::create_multiply(std::move(remaining), &budget));

94complementary_square(
95 const std::vector<detail::SymbolicNodePtr>& nodes,
96 const std::vector<bool>& used, std::size_t index) {
97 const auto square = trig_square(nodes[index]);
98 if (!square) { return std::nullopt; }
99 const auto complement =
100 square->type == FunctionNode::FuncType::Sin
101 ? FunctionNode::FuncType::Cos
102 : FunctionNode::FuncType::Sin;
103 for (std::size_t other = index + 1; other < nodes.size(); ++other) {
104 if (used[other]) { continue; }
105 const auto candidate = trig_square(nodes[other]);
106 if (candidate && candidate->type == complement &&
107 square->argument->equals(*candidate->argument) &&
108 square->coefficient->equals(*candidate->coefficient)) {
109 return std::pair<std::size_t, detail::SymbolicNodePtr>{
110 other, square->coefficient};
111 }
112 }
113 return std::nullopt;
114}

116ExprPtr rewrite_trig_sum(const AddNode& add, detail::RewriteBudget& budget) {
117 auto rewritten = rewrite_operands(add.operands(), budget);
118 std::vector<bool> used(rewritten.size(), false);
119 std::vector<detail::SymbolicNodePtr> result_nodes;
120 std::size_t size = 0;
121 for (std::size_t index = 0; index < rewritten.size(); ++index) {
122 if (used[index]) { continue; }
123 auto partner = complementary_square(rewritten, used, index);
124 if (partner) {
125 used[index] = true;
126 used[partner->first] = true;
127 size = budget.append_size(
128 size, budget.measure(partner->second));
129 result_nodes.push_back(partner->second);
130 } else {
131 size = budget.append_size(size, budget.measure(rewritten[index]));
132 result_nodes.push_back(rewritten[index]);
133 }
134 }
135 if (result_nodes.empty()) {
136 budget.require_nodes(1);
137 return SymbolicExpr::number(0);
138 }
139 return detail::make_expression_ptr(
140 SymbolicFactory::create_add(std::move(result_nodes), &budget));

144 auto arguments = rewrite_operands(function.arguments(), budget);

186 return rewrite_trig_function(*function, budget);
