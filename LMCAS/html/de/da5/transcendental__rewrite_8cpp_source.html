16/**
17 * @brief 将因子表达式中的不定元变量替换回原始超越子表达式.
18 *
19 * 对换元后的因子执行逆操作:遍历映射列表,将每个不定元(u0, u1, ...)
20 * 替换为其对应的原始超越表达式(如 sin(x),cos(x) 等).
21 * 利用 SymbolicExpr::substitute() 逐一执行变量替换.
22 *
23 * @param[in] factor_expr 以不定元表示的因子表达式
24 * @param[in] mappings 换元映射列表(indeterminate -> trans_expr)
25 * @return 替换后的符号表达式,以原始变量和超越函数表示
26 * @internal
27 */
28std::shared_ptr<SymbolicExpr> tf_back_substitute(
29 const std::shared_ptr<SymbolicExpr>& factor_expr,
30 const std::vector<TransSubstitution>& mappings) {
31
32 if (!factor_expr || !LMCAS::detail::node(factor_expr)) {
33 return factor_expr;
34 }
35 if (mappings.empty()) {
36 return factor_expr;
37 }
38
39 auto result = factor_expr;
40
41 for (const auto& m : mappings) {
42 if (!m.trans_expr || m.indeterminate.empty()) {
43 continue;
44 }
45
46 /// 仅当表达式依赖该不定元时才执行替换
47 if (expression_depends_on_variable(LMCAS::detail::node(result), m.indeterminate)) {
48 result = result->substitute(m.indeterminate, m.trans_expr);
49 }
50 }
51
52 return result;
53}

56/**
57 * @brief 从 NumberNode 中提取有理数值.
58 *
59 * 将 BigInt,Rational,lmmc_real_t 统一转换为 Rational 表示.
60 * 对于浮点数,仅当其为精确整数时才转换;否则返回失败.
61 *
62 * @param[in] num_node 数值节点
63 * @param[out] out 输出的有理数值
64 * @return 提取成功返回 true
65 * @internal
66 */
67static bool tf_extract_rational(const std::shared_ptr<const NumberNode>& num_node, Rational& out) {
68 if (!num_node) {
69 return false;
70 }
71
72 if (std::holds_alternative<BigInt>(num_node->value())) {
73 out = Rational(std::get<BigInt>(num_node->value()));
74 return true;
75 }
76 if (std::holds_alternative<Rational>(num_node->value())) {
77 out = std::get<Rational>(num_node->value());
78 return true;
79 }
80 if (std::holds_alternative<lmmc_real_t>(num_node->value())) {
81 lmmc_real_t v = std::get<lmmc_real_t>(num_node->value());
82 /// 仅处理精确整数浮点值
83 if (std::isfinite(v) && v == std::floor(v) && std::abs(v) < 1e15) {
84 out = Rational::from_double(v);
85 return true;
86 }
87 return false;
88 }
89 return false;
90}

94class FactorSimplification {
95public:
96 std::vector<std::shared_ptr<SymbolicExpr>> run(
97 const std::vector<std::shared_ptr<SymbolicExpr>>& factors) {
98 for (const auto& factor : factors) {
99 if (!factor || !LMCAS::detail::node(factor)) {
100 continue;
101 }
102 auto simplified = factor->simplify();
103 if (!simplified || !LMCAS::detail::node(simplified)) {
104 simplified = factor;
105 }
106 if (!append(simplified)) {
107 return std::move(result_);
108 }
109 }
110 if (constant_ != Rational(1)) {
111 result_.insert(result_.begin(), SymbolicExpr::number(constant_));
112 }
113 return std::move(result_);
114 }
115
116private:
117 bool append(const std::shared_ptr<SymbolicExpr>& factor) {
118 const auto& node = LMCAS::detail::node(factor);
119 if (auto number = std::dynamic_pointer_cast<const NumberNode>(node)) {
120 return append_number(number, factor);
121 }
122 if (auto product = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
123 append_product(*product, factor);
124 } else {
125 result_.push_back(factor);
126 }
127 return true;
128 }
129
130 bool append_number(const std::shared_ptr<const NumberNode>& number,
131 const std::shared_ptr<SymbolicExpr>& factor) {
132 Rational value;
133 if (!tf_extract_rational(number, value)) {
134 result_.push_back(factor);
135 return true;
136 }
137 if (value == Rational(0)) {
138 result_.clear();
139 result_.push_back(SymbolicExpr::number(0));
140 return false;
141 }
142 constant_ = constant_ * value;
143 return true;
144 }
145
146 void append_mixed_product(
147 const std::vector<std::shared_ptr<const SymbolicNode>>& numeric,
148 std::vector<std::shared_ptr<const SymbolicNode>>& symbolic) {
149 for (const auto& operand : numeric) {
150 auto number = std::dynamic_pointer_cast<const NumberNode>(operand);
151 Rational value;
152 if (tf_extract_rational(number, value)) {
153 constant_ = constant_ * value;
154 } else {
155 symbolic.push_back(operand);
156 }
157 }
158 if (symbolic.size() == 1) {
159 result_.push_back(LMCAS::detail::make_expression_ptr(symbolic[0]));
160 } else {
161 result_.push_back(LMCAS::detail::make_expression_ptr(
162 LMCAS::detail::make_node<MultiplyNode>(std::move(symbolic))));
163 }
164 }
165
166 void accumulate_numeric_product(
167 const std::vector<std::shared_ptr<const SymbolicNode>>& numeric) {
168 Rational product(1);
169 for (const auto& operand : numeric) {
170 auto number = std::dynamic_pointer_cast<const NumberNode>(operand);
171 Rational value;
172 if (tf_extract_rational(number, value)) {
173 product = product * value;
174 }
175 }
176 constant_ = constant_ * product;
177 }
178
179 void append_product(const MultiplyNode& product,
180 const std::shared_ptr<SymbolicExpr>& factor) {
181 std::vector<std::shared_ptr<const SymbolicNode>> numeric;
182 std::vector<std::shared_ptr<const SymbolicNode>> symbolic;
183 for (const auto& operand : product.operands()) {
184 if (std::dynamic_pointer_cast<const NumberNode>(operand)) {
185 numeric.push_back(operand);
186 } else {
187 symbolic.push_back(operand);
188 }
189 }
190 if (!numeric.empty() && !symbolic.empty()) {
191 append_mixed_product(numeric, symbolic);
192 } else if (numeric.empty()) {
193 result_.push_back(factor);
194 } else {
195 accumulate_numeric_product(numeric);
196 }
197 }
198
199 std::vector<std::shared_ptr<SymbolicExpr>> result_;
200 Rational constant_{1};
201};

218/**
219 * @brief 判断换元后的表达式是否对所有不定元和原始变量均为线性.
220 *
221 * 换元表达式对每个不定元(u0, u1, ...)及原始变量的次数均小于等于 1 时,
222 * 该表达式在超越多项式环中为整体元素.
223 * 例如 a*sin(x) + b*x + c 映射为 a*u0 + b*x + c,对 u0 与 x 均为线性.
224 *
225 * @param[in] sub_result 换元结果(含 poly_expr 和 mappings)
226 * @param[in] var 原始目标变量名
227 * @return 表达式对所有变量均为线性返回 true
228 * @internal
229 */
230bool tf_is_linear_irreducible(
231 const TransSubstitutionResult& sub_result,
232 const std::string& var) {
233
234 if (!sub_result.poly_expr || !LMCAS::detail::node(sub_result.poly_expr)) {
235 return false;
236 }
237 if (sub_result.mappings.empty()) {
238 return false;
239 }
240
241 const auto& root = LMCAS::detail::node(sub_result.poly_expr);
242
243 /// 检查每个不定元的次数是否 <= 1
244 for (const auto& m : sub_result.mappings) {
245 int deg = tf_degree_in(root, m.indeterminate);
246 if (deg < 0 || deg > 1) {
247 return false;
248 }
249 }
250
251 /// 检查原始变量的次数是否 <= 1
252 if (expression_depends_on_variable(root, var)) {
253 int deg = tf_degree_in(root, var);
254 if (deg < 0 || deg > 1) {
255 return false;
256 }
257 }
258
259 return true;
260}

262/**
263 * @brief 检测表达式是否已为独立子表达式的乘积形式.
264 *
265 * MultiplyNode 的各操作数直接形成独立因子,沿乘法结构完成分解;
266 * 数值常数单独累积,并在值异于 1 时形成常数因子.
267 *
268 * @param[in] expr 待检测的符号表达式
269 * @return 因子列表;若表达式非乘积形式则返回空向量(表示无快速路径)
270 * @internal
271 */
272std::vector<std::shared_ptr<SymbolicExpr>> tf_detect_multiplicative_structure(
273 const std::shared_ptr<SymbolicExpr>& expr) {
274
275 if (!expr || !LMCAS::detail::node(expr)) {
276 return {};
277 }
278
279 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(LMCAS::detail::node(expr));
280 if (!mul || mul->operands().size() < 2) {
281 return {};
282 }
283
284 std::vector<std::shared_ptr<SymbolicExpr>> factors;
285 Rational constant_acc(1);
286
287 for (const auto& op : mul->operands()) {
288 if (!op) {
289 continue;
290 }
291
292 /// 数值常数单独累积
293 if (op->is_number()) {
294 auto num = std::dynamic_pointer_cast<const NumberNode>(op);
295 if (num) {
296 if (std::holds_alternative<BigInt>(num->value())) {
297 constant_acc = constant_acc * Rational(std::get<BigInt>(num->value()));
298 } else if (std::holds_alternative<Rational>(num->value())) {
299 constant_acc = constant_acc * std::get<Rational>(num->value());
300 } else {
301 /// 浮点数值:作为独立因子保留
302 factors.push_back(LMCAS::detail::make_expression_ptr(op));
303 }
304 }
305 continue;
306 }
307
308 /// 非数值操作数作为独立因子
309 factors.push_back(LMCAS::detail::make_expression_ptr(op));
310 }
311
312 /// 仅当存在至少两个非常数因子(或一个非常数因子加一个非 1 常数)时才视为有效乘积分解
313 if (factors.size() < 2 && (factors.empty() || constant_acc == Rational(1))) {
314 return {};
315 }
316
317 /// 插入累积常数因子(若非 1)
318 if (constant_acc != Rational(1)) {
319 auto const_expr = SymbolicExpr::number(constant_acc);
320 factors.insert(factors.begin(), const_expr);
321 }
322
323 return factors;
324}

339/**
340 * @brief 从乘积项中提取指数函数因子.
341 *
342 * 若节点本身为 exp(f(x)) 形式,直接返回该节点.
343 * 若节点为 MultiplyNode,遍历其操作数寻找 exp(f(x)) 因子.
344 * 仅提取第一个匹配的指数函数因子.
345 *
346 * @param[in] node 待检测的 AST 节点
347 * @param[in] var 目标变量名
348 * @return 找到的 exp 因子节点;未找到返回 nullptr
349 * @internal
350 */
351static std::shared_ptr<const SymbolicNode> tf_extract_exp_factor(
352 const std::shared_ptr<const SymbolicNode>& node,
353 const std::string& var) {
354
355 if (!node) {
356 return nullptr;
357 }
358
359 /// 直接为 exp(f(x)) 形式
360 if (tf_is_variable_exponential(node, var)) {
361 return node;
362 }
363
364 /// 乘积形式:遍历操作数寻找 exp 因子
365 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
366 for (const auto& op : mul->operands()) {
367 if (tf_is_variable_exponential(op, var)) {
368 return op;
369 }
370 }
371 }
372
373 return nullptr;
374}

376/**
377 * @brief 从乘积项中移除指定的指数函数因子,返回剩余部分.
378 *
379 * 若节点本身即为该 exp 因子,返回数值 1.
380 * 若节点为 MultiplyNode,移除匹配的 exp 操作数后重建乘积.
381 *
382 * @param[in] node 原始乘积项节点
383 * @param[in] exp_factor 待移除的 exp 因子节点
384 * @return 移除 exp 因子后的剩余节点
385 * @internal
386 */
387static std::shared_ptr<const SymbolicNode> tf_remove_exp_factor(
388 const std::shared_ptr<const SymbolicNode>& node,
389 const std::shared_ptr<const SymbolicNode>& exp_factor) {
390
391 if (!node || !exp_factor) {
392 return node;
393 }
394
395 /// 节点本身即为 exp 因子
396 if (node->equals(*exp_factor)) {
397 return LMCAS::detail::make_node<NumberNode>(BigInt(1));
398 }
399
400 /// 乘积形式:移除匹配的操作数
401 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
402 std::vector<std::shared_ptr<const SymbolicNode>> remaining_ops;
403 bool removed = false;
404
405 for (const auto& op : mul->operands()) {
406 if (!removed && op->equals(*exp_factor)) {
407 removed = true;
408 continue;
409 }
410 remaining_ops.push_back(op);
411 }
412
413 if (!removed) {
414 return node;
415 }
416
417 if (remaining_ops.empty()) {
418 return LMCAS::detail::make_node<NumberNode>(BigInt(1));
419 }
420 if (remaining_ops.size() == 1) {
421 return remaining_ops[0];
422 }
423 return LMCAS::detail::make_node<MultiplyNode>(std::move(remaining_ops));
424 }
425
426 return node;
427}

429/**
430 * @brief 检测加法表达式中的公共指数因子并执行分离.
431 *
432 * 对于 AddNode 形式的表达式,检查所有加法项是否共享相同的 exp(f(x)) 因子.
433 * 若是,则提取公因子:expr = exp(f(x)) * (t1' + t2' + ... + tn'),
434 * 其中 ti' = ti / exp(f(x)).
435 *
436 * 典型用例:
437 * - exp(x)*x + exp(x) -> [exp(x), x+1]
438 * - exp(x)*x^2 + 2*exp(x)*x + exp(x) -> [exp(x), x^2+2x+1]
439 *
440 * @param[in] expr 待检测的符号表达式
441 * @param[in] var 目标变量名
442 * @return 因子列表 [exp(f(x)), remaining_sum];若无公共 exp 因子则返回空向量
443 * @internal
444 */
445std::vector<std::shared_ptr<SymbolicExpr>> tf_detect_exponential_separation(
446 const std::shared_ptr<SymbolicExpr>& expr,
447 const std::string& var) {
448
449 if (!expr || !LMCAS::detail::node(expr)) {
450 return {};
451 }
452
453 auto add = std::dynamic_pointer_cast<const AddNode>(LMCAS::detail::node(expr));
454 if (!add || add->operands().size() < 2) {
455 return {};
456 }
457
458 /// 从第一个加法项中提取 exp 因子作为候选公因子
459 std::shared_ptr<const SymbolicNode> common_exp = tf_extract_exp_factor(add->operands()[0], var);
460 if (!common_exp) {
461 return {};
462 }
463
464 /// 验证所有加法项均含有相同的 exp 因子
465 for (size_t i = 1; i < add->operands().size(); ++i) {
466 std::shared_ptr<const SymbolicNode> term_exp = tf_extract_exp_factor(add->operands()[i], var);
467 if (!term_exp || !term_exp->equals(*common_exp)) {
468 return {};
469 }
470 }
471
472 /// 所有项共享相同的 exp(f(x)),执行分离
473 /// 构造剩余和:对每个项移除 exp 因子
474 std::vector<std::shared_ptr<const SymbolicNode>> remainder_terms;
475 remainder_terms.reserve(add->operands().size());
476
477 for (const auto& op : add->operands()) {
478 auto remainder = tf_remove_exp_factor(op, common_exp);
479 remainder_terms.push_back(remainder);
480 }
481
482 /// 构造结果
483 auto exp_factor_expr = LMCAS::detail::make_expression_ptr(common_exp);
484
485 std::shared_ptr<SymbolicExpr> sum_expr;
486 if (remainder_terms.size() == 1) {
487 sum_expr = LMCAS::detail::make_expression_ptr(remainder_terms[0]);
488 } else {
489 sum_expr = LMCAS::detail::make_expression_ptr(
490 LMCAS::detail::make_node<AddNode>(std::move(remainder_terms)));
491 }
492
493 /// 化简剩余和
494 auto simplified_sum = sum_expr->simplify();
495 if (simplified_sum && LMCAS::detail::node(simplified_sum)) {
496 sum_expr = simplified_sum;
497 }
498
499 return {exp_factor_expr, sum_expr};
500}

503/**
504 * @brief 判断表达式 AST 中是否包含依赖指定变量的超越函数.
505 *
506 * 递归遍历 AST,若发现任何 FunctionNode 类型为 Sin/Cos/Exp/Ln/Tan
507 * 且其参数依赖 var,则返回 true.
508 *
509 * @param[in] node 当前 AST 节点
510 * @param[in] var 目标变量名
511 * @return 包含超越函数返回 true
512 * @internal
513 */
514static bool any_transcendental(
515 const std::vector<std::shared_ptr<const SymbolicNode>>& nodes,
516 const std::string& var) {
517 for (const auto& node : nodes) {
518 if (tf_contains_transcendental(node, var)) {
519 return true;
520 }
521 }
522 return false;
523}
524
525bool tf_contains_transcendental(
526 const std::shared_ptr<const SymbolicNode>& node,
527 const std::string& var) {
528
529 if (!node) {
530 return false;
531 }
532
533 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(node)) {
534 if (func->arguments().size() == 1 &&
535 tf_is_transcendental_type(func->type()) &&
536 expression_depends_on_variable(func->arguments()[0], var)) {
537 return true;
538 }
539 return any_transcendental(func->arguments(), var);
540 }
541
542 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
543 return any_transcendental(add->operands(), var);
544 }
545
546 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
547 return any_transcendental(mul->operands(), var);
548 }
549 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
550 return tf_contains_transcendental(pow->base(), var) ||
551 tf_contains_transcendental(pow->exponent(), var);
552 }
553
554 return false;
555}
