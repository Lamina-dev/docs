63 const std::shared_ptr<SymbolicExpr>& denominator,

71 critical_points.push_back({root, mult, 0});
72 }

95std::vector<Interval> rational_sign_intervals(
96 const std::vector<CriticalPoint>& critical_points,
97 int combined_leading_sign, int target_sign) {
98 size_t n_cp = critical_points.size();
99 std::vector<int> interval_signs(n_cp + 1);
100
101 interval_signs[n_cp] = combined_leading_sign;
102
103 for (int i = (int)n_cp - 1; i >= 0; --i) {
104 int total_mult = critical_points[i].num_multiplicity + critical_points[i].den_multiplicity;
105 interval_signs[i] = interval_signs[i + 1];
106 if (total_mult % 2 != 0) {
107 interval_signs[i] = -interval_signs[i];
108 }
109 }
110 std::vector<Interval> result_intervals;
111 for (size_t i = 0; i <= n_cp; ++i) {
112 if (interval_signs[i] != target_sign) {
113 continue;
114 }
115 Interval interval;
116 interval.lower = i == 0 ? Endpoint::neg_inf() : Endpoint::open(critical_points[i - 1].value);
117 interval.upper = i == n_cp ? Endpoint::pos_inf() : Endpoint::open(critical_points[i].value);
118 result_intervals.push_back(interval);
119 }
120 return result_intervals;
121}
