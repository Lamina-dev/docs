91std::vector<Interval> rational_sign_intervals(
92 const std::vector<CriticalPoint>& critical_points,
93 int combined_leading_sign, int target_sign) {
94 size_t n_cp = critical_points.size();
95 std::vector<int> interval_signs(n_cp + 1);
96
97 interval_signs[n_cp] = combined_leading_sign;
98
99 for (int i = (int)n_cp - 1; i >= 0; --i) {
100 int total_mult = critical_points[i].num_multiplicity + critical_points[i].den_multiplicity;
101 interval_signs[i] = interval_signs[i + 1];
102 if (total_mult % 2 != 0) {
103 interval_signs[i] = -interval_signs[i];
104 }
105 }
106 std::vector<Interval> result_intervals;
107 for (size_t i = 0; i <= n_cp; ++i) {
108 if (interval_signs[i] != target_sign) {
109 continue;
110 }
111 Interval interval;
112 interval.lower = i == 0 ? Endpoint::neg_inf() : Endpoint::open(critical_points[i - 1].value);
113 interval.upper = i == n_cp ? Endpoint::pos_inf() : Endpoint::open(critical_points[i].value);
114 result_intervals.push_back(interval);
115 }
116 return result_intervals;

169 int combined_leading_sign = num_leading_sign * den_leading_sign;
170 bool want_positive = (type == InequalityType::GreaterThan || type == InequalityType::GreaterEqual);
