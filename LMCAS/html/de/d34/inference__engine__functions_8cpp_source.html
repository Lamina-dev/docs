8InferenceTriboolResult InferenceEngine::Impl::real_definedness(
9 const InferenceEngine& engine, const SymbolicNode& node, ComputationContext& context) {
10 const detail::AssumptionFacts facts(engine);
11 const std::shared_ptr<const SymbolicNode> borrowed(
12 std::shared_ptr<const SymbolicNode>{}, &node);
13 bool violated_real_restriction = false;
14 auto defined = detail::query_definedness(
15 borrowed, facts, Domain::Real, context, &violated_real_restriction);
16 if (defined && defined.value() == Tribool::False) {
17 if (!violated_real_restriction) {
18 auto complex_defined =
19 detail::query_definedness(borrowed, facts, Domain::Complex, context);
20 if (!complex_defined) return InferenceTriboolResult::failure(complex_defined.error());
21 if (complex_defined.value() != Tribool::False) {
22 auto real_value = detail::query_real_value(borrowed, facts, context);
23 if (!real_value) return InferenceTriboolResult::failure(real_value.error());
24 if (real_value.value() == Tribool::Unknown) return Tribool::Unknown;
25 }
26 }
27 return InferenceTriboolResult::failure(
28 CasErrc::DomainError, "expression is undefined in the real domain",
29 "inference.definedness");
30 }
31 return defined;
32}

59InferenceTriboolResult InferenceEngine::Impl::absolute_sign(
60 const InferenceEngine& engine,
61 const SymbolicExpr& arg_expr,
62 Sign target, ComputationContext& context) const {
63 auto arg_real_result = engine.query_domain_of_checked(arg_expr, Domain::Real, context);
64 if (!arg_real_result) {
65 return arg_real_result;
66 }
67 const Tribool arg_real = arg_real_result.value();
68 if (arg_real != Tribool::True) {
69 return InferenceTriboolResult::success(Tribool::Unknown);
70 }
71
72 auto arg_pos_result = engine.query_sign_of_checked(arg_expr, Sign::Positive, context);
73 if (!arg_pos_result) {
74 return arg_pos_result;
75 }
76 const Tribool arg_pos = arg_pos_result.value();
77 auto arg_neg_result = engine.query_sign_of_checked(arg_expr, Sign::Negative, context);
78 if (!arg_neg_result) {
79 return arg_neg_result;
80 }
81 const Tribool arg_neg = arg_neg_result.value();
82 auto arg_nz_result = engine.query_sign_of_checked(arg_expr, Sign::NonZero, context);
83 if (!arg_nz_result) {
84 return arg_nz_result;
85 }
86 const Tribool arg_nz = arg_nz_result.value();
87 bool abs_is_positive = (arg_pos == Tribool::True) ||
88 (arg_neg == Tribool::True) ||
89 (arg_nz == Tribool::True);
90
91 if (abs_is_positive) {
92 return known_sign(1, target);
93 }
94 return nonnegative_sign(target);
95}

97InferenceTriboolResult InferenceEngine::Impl::logarithm_sign(
98 const InferenceEngine&,
99 const SymbolicExpr& arg_expr,
100 Sign target, ComputationContext& context) const {
101 if (const auto* number = dynamic_cast<const NumberNode*>(detail::node(arg_expr).get())) {
102 const int sign = std::visit([](const auto& value) {
103 using T = std::decay_t<decltype(value)>;
104 return value > T(1) ? 1 : value < T(1) ? -1 : 0;
105 }, number->value());
106 return known_sign(sign, target);
107 }
108 auto step = context.consume_steps(1, "inference.logarithm");
109 if (!step) {
110 return InferenceTriboolResult::failure(step.error());
111 }
112 const auto one = detail::expression_from_node(detail::make_node<NumberNode>(BigInt(1)));
113 auto greater = ctx.has_relation_checked(arg_expr, one, RelationOp::GT, context);
114 if (!greater) {
115 return InferenceTriboolResult::failure(greater.error());
116 }
117 if (greater.value()) {
118 return known_sign(1, target);
119 }
120 auto equal = ctx.has_relation_checked(arg_expr, one, RelationOp::EQ, context);
121 if (!equal) {
122 return InferenceTriboolResult::failure(equal.error());
123 }
124 if (equal.value()) {
125 return known_sign(0, target);
126 }
127 auto less = ctx.has_relation_checked(arg_expr, one, RelationOp::LT, context);
128 if (!less) {
129 return InferenceTriboolResult::failure(less.error());
130 }
131 if (less.value()) {
132 return known_sign(-1, target);
133 }
134 return Tribool::Unknown;
135}
