72 const ExprPtr& rhs,
73 ComputationContext& context) {
74 if (!lhs || !rhs) {
75 return expression_failure(CasErrc::InvalidArgument,
76 "eq operands cannot be null", kExprOperation);
77 }
78 auto left_dimension = dimension_of(*lhs);
79 if (!left_dimension) return ExprResult::failure(left_dimension.error());
80 auto right_dimension = dimension_of(*rhs);
81 if (!right_dimension) return ExprResult::failure(right_dimension.error());
82 if (left_dimension.value() != right_dimension.value()) return integer(0);
83 auto left = comparison_value(lhs, context);
84 if (!left) return left;
85 auto right = comparison_value(rhs, context);
86 if (!right) return right;
87 return make_binary_expr(left.value(), right.value(), context, "eq", SymbolicExpr::eq);

349ExprResult abs(const ExprPtr& expression, ComputationContext& context) {
350 auto step = context.consume_steps(1, kAbsOperation);
351 if (!step) return ExprResult::failure(step.error());
352 auto re = real(expression, context);
353 if (!re) return re;
354 auto im = imag(expression, context);
355 if (!im) return im;
356 try {
357 auto re_squared = SymbolicExpr::power(re.value(), SymbolicExpr::number(2));
358 auto im_squared = SymbolicExpr::power(im.value(), SymbolicExpr::number(2));
359 auto sum = SymbolicExpr::add(re_squared, im_squared);
360 auto result = SymbolicExpr::sqrt(sum)->simplify();
361 if (!result || !lamina::detail::node(result)) {
362 return expression_failure(CasErrc::InternalInvariant,
363 "complex absolute value construction failed",
364 kAbsOperation);
365 }
366 return ExprResult::success(std::move(result));
367 } catch (const std::bad_alloc&) {
368 return expression_failure(CasErrc::ResourceLimit,
369 "complex absolute value allocation failed",
370 kAbsOperation);
371 } catch (const std::exception& error) {
372 return expression_failure(CasErrc::InvalidArgument, error.what(),
373 kAbsOperation);
374 }
375}
