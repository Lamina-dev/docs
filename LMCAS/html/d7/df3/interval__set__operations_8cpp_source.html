78 auto left = checked_interval_views(intervals_, context, operation);
79 if (!left) {
80 return Result<IntervalUnion>::failure(left.error());
81 }

83 if (!right) {
84 return Result<IntervalUnion>::failure(right.error());
85 }
86
87 const auto& a_intervals = left.value();
88 const auto& b_intervals = right.value();
89 std::vector<Interval> result;
90 std::size_t i = 0;
91 std::size_t j = 0;
92 while (i < a_intervals.size() && j < b_intervals.size()) {
93 const auto& a = a_intervals[i];
94 const auto& b = b_intervals[j];
95 auto lower_order = compare_comparable(a.lower, b.lower, context);
96 if (!lower_order) {
97 return Result<IntervalUnion>::failure(lower_order.error());
98 }
99 auto upper_order = compare_comparable(a.upper, b.upper, context);
100 if (!upper_order) {
101 return Result<IntervalUnion>::failure(upper_order.error());
102 }
103 auto candidate = intersection_candidate(
104 a, b, lower_order.value(), upper_order.value());
105 auto empty = interval_is_empty_checked(candidate, context);
106 if (!empty) {
107 return Result<IntervalUnion>::failure(
108 empty.error().code, empty.error().message, operation);
109 }
110 if (!empty.value()) {
111 result.push_back(std::move(candidate));
112 }
113
114 if (upper_order.value() < 0) {
115 ++i;
116 } else {
117 ++j;
118 }
119 }
120
121 auto normalized = normalize_intervals_checked(std::move(result), context);
122 if (!normalized) {
123 return Result<IntervalUnion>::failure(
124 normalized.error().code, normalized.error().message, operation);
125 }

163Result<IntervalUnion> IntervalUnion::complement_checked(
164 ComputationContext& context) const {
165 constexpr const char* operation = "interval_union_complement";
166 auto checked = checked_interval_views(intervals_, context, operation);
167 if (!checked) {
168 return Result<IntervalUnion>::failure(checked.error());
169 }
170 const auto& intervals = checked.value();
171
172 if (intervals.empty()) {
173 return Result<IntervalUnion>::success(
174 IntervalUnion::from_checked_normalized({Interval::entire_line()}));
175 }
176 if (intervals.size() == 1 &&
177 intervals[0].interval.lower.is_neg_infinity &&
178 intervals[0].interval.upper.is_pos_infinity) {
179 return Result<IntervalUnion>::success(IntervalUnion::from_checked_normalized({}));
180 }
181
182 std::vector<Interval> result;
183 const Interval& first = intervals.front().interval;
184 if (!first.lower.is_neg_infinity) {
185 result.push_back(Interval{
186 Endpoint::neg_inf(),
187 complement_upper_from_lower(first.lower)
188 });
189 }
190
191 for (std::size_t i = 0; i + 1 < intervals.size(); ++i) {
192 const Interval& current = intervals[i].interval;
193 const Interval& next = intervals[i + 1].interval;
194 result.push_back(Interval{
195 complement_lower_from_upper(current.upper),
196 complement_upper_from_lower(next.lower)
197 });
198 }
199
200 const Interval& last = intervals.back().interval;
201 if (!last.upper.is_pos_infinity) {
202 result.push_back(Interval{
203 complement_lower_from_upper(last.upper),
204 Endpoint::pos_inf()
205 });
206 }
207
208 auto normalized = normalize_intervals_checked(std::move(result), context);
209 if (!normalized) {

211 }
