127/**
128 * @brief 二因子二次 Hensel 提升的状态结构.
129 *
130 * 存储一对因子 g, h 及其 Bezout 系数 s, t,满足:
131 * - f == g * h (mod modulus)
132 * - s * g + t * h == 1 (mod modulus)
133 *
134 * 系数按升幂存储:coeffs[i] 对应 x^i 的系数.
135 */
136struct HenselLiftPair {
137 std::vector<BigInt> g; ///< 第一个因子系数向量
138 std::vector<BigInt> h; ///< 第二个因子系数向量
139 std::vector<BigInt> s; ///< g 的 Bezout 系数向量
140 std::vector<BigInt> t; ///< h 的 Bezout 系数向量
141 BigInt modulus; ///< 当前模数
142};
143
144/**
145 * @brief 执行一步二次 Hensel 提升:mod m -> mod m^2.
146 *
147 * 给定 f == g*h (mod m) 且 s*g + t*h == 1 (mod m),
148 * 计算 g', h', s', t' 使得 f == g'*h' (mod m^2) 且 s'*g' + t'*h' == 1 (mod m^2).
149 *
150 * @param[in] f 原始多项式系数向量(升幂排列)
151 * @param[in] current 当前提升状态(g, h, s, t, modulus=m)
152 * @return 提升后的状态(g', h', s', t', modulus=m^2)
153 *
154 * @pre f == g*h (mod m)
155 * @pre s*g + t*h == 1 (mod m)
156 * @pre deg(s) < deg(h), deg(t) < deg(g)
157 *
158 * @see Zassenhaus, H. "On Hensel factorization, I."
159 * Journal of Number Theory, 1(3), 1969.
160 */
161LMCAS_API HenselLiftPair hl_two_factor_lift(
162 const std::vector<BigInt>& f,
163 const HenselLiftPair& current);
