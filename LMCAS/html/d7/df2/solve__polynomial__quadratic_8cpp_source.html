55 if (a_simplified->get_number_value_is_zero()) {
56 auto b_simplified = detail::make_expression_ptr(detail::normalize_node(detail::node(b), transform_budget));

63 throw std::invalid_argument(

65 }
66 if (is_purely_numeric(c_simplified)) {
67 return {};

69 throw std::invalid_argument(
70 "solve_biquadratic: equation is independent of the requested variable");
71 }
72
73 auto u_roots = solve_quadratic_internal(a, b, c, context);
74
75 std::vector<std::shared_ptr<SymbolicExpr>> results;
76 for (const auto& u : u_roots) {
77 auto pos_root = detail::make_expression_ptr(detail::normalize_node(detail::node(SymbolicExpr::sqrt(u)), transform_budget));
78 auto neg_root = detail::make_expression_ptr(detail::normalize_node(detail::node(negate(SymbolicExpr::sqrt(u))), transform_budget));
79 results.push_back(pos_root);
80 results.push_back(neg_root);
81 }
