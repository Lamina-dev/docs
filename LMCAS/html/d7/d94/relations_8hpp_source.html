127 int compare_same_type(const SymbolicNode& other) const override {
128 const auto& o = static_cast<const LogicalNode&>(other);
129 if (op_ != o.op_) return (int)op_ < (int)o.op_ ? -1 : 1;
130 /// 处理一元 Not 运算（right 为 nullptr）
131 bool l_null = !left_;
132 bool ol_null = !o.left_;
133 if (l_null != ol_null) return l_null ? -1 : 1;
134 if (!l_null) {
135 int cmp = left_->compare(*o.left_);
136 if (cmp != 0) return cmp;
137 }
138 bool r_null = !right_;
139 bool or_null = !o.right_;
140 if (r_null != or_null) return r_null ? -1 : 1;
141 if (!r_null) {
142 return right_->compare(*o.right_);
143 }
144 return 0;
145 }
