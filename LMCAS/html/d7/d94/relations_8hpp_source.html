127 if (op_ != o.op_) return (int)op_ < (int)o.op_ ? -1 : 1;
128 /// 处理一元 Not 运算（right 为 nullptr）
129 bool l_null = !left_;
130 bool ol_null = !o.left_;
131 if (l_null != ol_null) return l_null ? -1 : 1;
132 if (!l_null) {

134 if (cmp != 0) return cmp;
135 }
136 bool r_null = !right_;
137 bool or_null = !o.right_;
138 if (r_null != or_null) return r_null ? -1 : 1;
139 if (!r_null) {
140 return right_->compare(*o.right_);
141 }
142 return 0;
