48bool PolyBuilder::use_numeric_function(const FunctionNode& node) {
49 auto expression = LMCAS::detail::expression_from_node(
50 LMCAS::detail::make_node<FunctionNode>(node.type(), node.arguments()));
51 auto simplified = expression.simplify();
52 if (!simplified || !simplified->is_number()) return false;
53 auto number = std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(simplified));
54 if (!number) return false;
55 if (std::holds_alternative<Rational>(number->value())) {
56 result = Poly(ext_vars.size());
57 result.add_term(Monomial(ext_vars.size(), 0), std::get<Rational>(number->value()));
58 return true;
59 }
60 if (std::holds_alternative<BigInt>(number->value())) {
61 result = Poly(ext_vars.size());
62 result.add_term(Monomial(ext_vars.size(), 0), Rational(std::get<BigInt>(number->value())));
63 return true;
64 }
65 return false;
66}
