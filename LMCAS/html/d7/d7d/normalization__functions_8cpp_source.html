6bool NormalizationVisitor::normalize_logarithm(
7 FunctionNode::FuncType type,
8 const std::vector<std::shared_ptr<const SymbolicNode>>& arguments) {
9 if (type == FunctionNode::FuncType::Ln && arguments.size() == 1) {
10 return normalize_natural_logarithm(arguments.front());
11 }
12 if (type != FunctionNode::FuncType::Log || arguments.size() != 2) {
13 return false;
14 }
15 if (const auto power = std::dynamic_pointer_cast<const PowerNode>(arguments.front())) {
16 normalization_check_children(rewrite_budget(), 1, arguments[1]);
17 auto logarithm = detail::make_node<FunctionNode>(FunctionNode::FuncType::Ln,
18 std::vector<std::shared_ptr<const SymbolicNode>>{arguments[1]});
19 if (power->base()->compare(*arguments[1]) == 0 &&
20 normalization_proved(detail::query_positive_value(arguments[1], facts_, context_)) &&
21 normalization_proved(detail::query_real_value(power->exponent(), facts_, context_)) &&
22 normalization_proved(detail::query_nonzero_value(logarithm, facts_, Domain::Real, context_))) {
23 set_result(power->exponent());
24 return true;
25 }
26 }
27 normalization_check_children(rewrite_budget(), 1, arguments[0]);
28 auto numerator = detail::make_node<FunctionNode>(FunctionNode::FuncType::Ln,
29 std::vector<std::shared_ptr<const SymbolicNode>>{arguments[0]});
30 normalization_check_children(rewrite_budget(), 1, arguments[1]);
31 auto denominator = detail::make_node<FunctionNode>(FunctionNode::FuncType::Ln,
32 std::vector<std::shared_ptr<const SymbolicNode>>{arguments[1]});
33 normalization_check_children(rewrite_budget(), 2, denominator);
34 auto reciprocal = detail::make_node<PowerNode>(denominator, detail::make_node<NumberNode>(BigInt(-1)));
35 auto quotient = make_normalized_multiply_node({numerator, reciprocal}, rewrite_budget());
36 NormalizationVisitor visitor(context_, facts_, domain_, rewrite_budget());
37 quotient->accept(visitor);
38 set_result(visitor.get_result());
39 return true;
40}

42bool NormalizationVisitor::normalize_natural_logarithm(
43 const std::shared_ptr<const SymbolicNode>& argument) {
44 if (const auto power = std::dynamic_pointer_cast<const PowerNode>(argument)) {
45 if (!normalization_proved(detail::query_positive_value(power->base(), facts_, context_)) ||
46 !normalization_proved(detail::query_real_value(power->exponent(), facts_, context_))) {
47 return false;
48 }
49 normalization_check_children(rewrite_budget(), 1, power->base());
50 auto logarithm = detail::make_node<FunctionNode>(FunctionNode::FuncType::Ln,
51 std::vector<std::shared_ptr<const SymbolicNode>>{power->base()});
52 auto product = make_normalized_multiply_node({power->exponent(), logarithm}, rewrite_budget());
53 NormalizationVisitor visitor(context_, facts_, domain_, rewrite_budget());
54 product->accept(visitor);
55 set_result(visitor.get_result());
56 return true;
57 }
58 const auto function = std::dynamic_pointer_cast<const FunctionNode>(argument);
59 if (!function || function->type() != FunctionNode::FuncType::Exp) {
60 return false;
61 }
62 if (function->arguments().size() != 1 ||
63 !normalization_proved(detail::query_real_value(function->arguments().front(), facts_, context_))) {
64 return false;
65 }
66 set_result(function->arguments().front());
67 return true;
68}

72 std::shared_ptr<const SymbolicNode> positive;

75 }
76 switch (type) {

78 case FunctionNode::FuncType::Tan:
79 case FunctionNode::FuncType::ArcTan:

82 auto function = detail::make_node<FunctionNode>(type,

84 set_result(make_normalized_multiply_node(

110void NormalizationVisitor::visit(const FunctionNode& node) {
111 std::shared_ptr<const SymbolicNode> argument;
112 if (try_normalize_squared_norm(node, argument)) {
113 return;
114 }
115 std::vector<std::shared_ptr<const SymbolicNode>> arguments;
116 std::size_t nodes = 0;
117 if (argument) {
118 normalization_append(rewrite_budget(), nodes, arguments, argument);
119 } else {
120 for (const auto& operand : node.arguments()) {
121 operand->accept(*this);
122 normalization_append(rewrite_budget(), nodes, arguments, result);
123 }
124 }
125 if (arguments.size() == 1) {
126 if (const auto number = std::dynamic_pointer_cast<const NumberNode>(arguments.front())) {
127 if (auto value = normalization_numeric_function(node.type(), number, rewrite_budget())) {
128 set_result(std::move(value));
129 return;
130 }
131 }
132 }
133 if (normalize_logarithm(node.type(), arguments)) {
134 return;
135 }
136 if (arguments.size() == 1 && normalize_unary_function(node.type(), arguments.front())) {
137 return;
138 }
139 if (rewrite_budget()) { rewrite_budget()->append_size(1, nodes); }
140 auto function = detail::make_node<FunctionNode>(node.type(), arguments);
141 if (auto simplified = try_assumption_simplify(function)) {
142 simplified->accept(*this);
143 return;
144 }
145 set_result(function);
146}

160std::shared_ptr<const SymbolicNode> NormalizationVisitor::simplify_assumed_square_root(
161 const std::shared_ptr<const SymbolicNode>& argument) const {
162 const auto power = std::dynamic_pointer_cast<const PowerNode>(argument);
163 if (!power) {
164 return nullptr;
165 }
166 const auto exponent = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
167 BigInt value;
168 if (!try_get_integer_value(exponent, value)) {
169 return nullptr;
170 }
171 if (value != BigInt(2)) {
172 return nullptr;
173 }
174 auto nonnegative = detail::query_nonnegative_value(power->base(), facts_, context_);
175 if (normalization_proved(nonnegative)) {
176 return power->base();
177 }
178 auto real = detail::query_real_value(power->base(), facts_, context_);
179 if (!real) {
180 throw real.error();
181 }
182 if (real.value() != Tribool::True) {
183 return nullptr;
184 }
185 normalization_check_children(rewrite_budget(), 1, power->base());
186 return detail::make_node<FunctionNode>(FunctionNode::FuncType::Abs,
187 std::vector<std::shared_ptr<const SymbolicNode>>{power->base()});
188}

197 !normalization_proved(facts_.is_negative(argument, context_))) {
198 return nullptr;

204std::shared_ptr<const SymbolicNode> NormalizationVisitor::try_assumption_simplify(
205 const std::shared_ptr<const SymbolicNode>& node) {
206 const auto function = std::dynamic_pointer_cast<const FunctionNode>(node);
207 if (!function) {
208 return nullptr;
209 }
210 if (function->arguments().size() != 1) {
211 return nullptr;
212 }
213 const auto& argument = function->arguments().front();
214 switch (function->type()) {
215 case FunctionNode::FuncType::Sqrt:
216 return simplify_assumed_square_root(argument);
217 case FunctionNode::FuncType::Abs:
218 return simplify_assumed_absolute(argument);
219 default:
220 return nullptr;
221 }
