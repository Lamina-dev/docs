23InferenceEngine::DepthGuard::DepthGuard(
24 const InferenceEngine& engine, const SymbolicNode& node)
25 : engine_(engine), node_(&node) {
26 engine_.impl_->current_depth++;
27
28 if (engine_.impl_->current_depth > engine_.impl_->max_depth) {
29 abort_ = true;
30 return;
31 }
32
33 // Insert and detect cycles atomically. If allocation throws, restore the
34 // depth counter because a throwing constructor has no matching destructor.
35 try {
36 inserted_ = engine_.impl_->visited.insert(node_).second;
37 } catch (...) {
38 engine_.impl_->current_depth--;
39 if (engine_.impl_->current_depth == 0) {
40 engine_.impl_->visited.clear();
41 }
42 throw;
43 }
44 if (!inserted_) {
45 abort_ = true;
46 }
47}

49InferenceEngine::DepthGuard::~DepthGuard() {
50 if (inserted_) {
51 engine_.impl_->visited.erase(node_);
52 }
53
54 engine_.impl_->current_depth--;
55
56 if (engine_.impl_->current_depth == 0) {
57 engine_.impl_->visited.clear();
58 }
59}

62InferenceTriboolResult InferenceEngine::query_sign_of_checked(const SymbolicExpr& expr, Sign sign, ComputationContext& context) const {

255InferenceTriboolResult InferenceEngine::Impl::query_zero(
256 const InferenceEngine& engine,
257 const SymbolicExpr& expr, ComputationContext& context) const {
258 auto nn = engine.query_nonnegative_checked(expr, context);
259 if (!nn) {
260 return nn;
261 }
262 auto np = engine.query_nonpositive_checked(expr, context);
263 if (!np) {
264 return np;
265 }
266 if (nn.value() == Tribool::True && np.value() == Tribool::True) {
267 return InferenceTriboolResult::success(Tribool::True);
268 }
269 if (nn.value() == Tribool::False || np.value() == Tribool::False) {
270 return InferenceTriboolResult::success(Tribool::False);
271 }
272 auto nonzero = engine.query_nonzero_checked(expr, context);
273 if (!nonzero) return nonzero;
274 if (nonzero.value() == Tribool::True) return Tribool::False;
275 return InferenceTriboolResult::success(Tribool::Unknown);
276}
