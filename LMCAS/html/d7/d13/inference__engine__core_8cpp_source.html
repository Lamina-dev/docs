23InferenceEngine::DepthGuard::DepthGuard(const InferenceEngine& engine, const void* node)
24 : engine_(engine), node_(node) {
25 engine_.impl_->current_depth++;
26
27 if (engine_.impl_->current_depth > engine_.impl_->max_depth) {
28 abort_ = true;
29 return;
30 }
31
32 // Insert and detect cycles atomically. If allocation throws, restore the
33 // depth counter because a throwing constructor has no matching destructor.
34 if (node_) {
35 try {
36 inserted_ = engine_.impl_->visited.insert(node_).second;
37 } catch (...) {
38 engine_.impl_->current_depth--;
39 if (engine_.impl_->current_depth == 0) {
40 engine_.impl_->visited.clear();
41 }
42 throw;
43 }
44 if (!inserted_) {
45 abort_ = true;
46 }
47 }
48}
49
50InferenceEngine::DepthGuard::~DepthGuard() {
51 if (inserted_ && node_) {
52 engine_.impl_->visited.erase(node_);
53 }
54
55 engine_.impl_->current_depth--;
56
57 if (engine_.impl_->current_depth == 0) {
58 engine_.impl_->visited.clear();
59 }
60}

66InferenceTriboolResult InferenceEngine::query_sign_of_checked(const SymbolicExpr& expr, Sign sign) const {
67 switch (sign) {
68 case Sign::Positive: return query_positive_checked(expr);
69 case Sign::Negative: return query_negative_checked(expr);
70 case Sign::NonNegative: return query_nonnegative_checked(expr);
71 case Sign::NonPositive: return query_nonpositive_checked(expr);
72 case Sign::Zero:
73 // Zero means both NonNegative and NonPositive
74 {
75 auto nn = query_nonnegative_checked(expr);
76 if (!nn) return nn;
77 auto np = query_nonpositive_checked(expr);
78 if (!np) return np;
79 if (nn.value() == Tribool::True && np.value() == Tribool::True)
80 return InferenceTriboolResult::success(Tribool::True);
81 if (nn.value() == Tribool::False || np.value() == Tribool::False)
82 return InferenceTriboolResult::success(Tribool::False);
83 return InferenceTriboolResult::success(Tribool::Unknown);
84 }
85 case Sign::NonZero: return query_nonzero_checked(expr);
86 }
87 return InferenceTriboolResult::success(Tribool::Unknown);
88}

94InferenceTriboolResult InferenceEngine::query_domain_of_checked(const SymbolicExpr& expr, Domain domain) const {
95 switch (domain) {
96 case Domain::Integer: return query_integer_checked(expr);
97 case Domain::Real: return query_real_checked(expr);
98 case Domain::Rational: {
99 if (!lamina::detail::node(expr)) {
100 return InferenceTriboolResult::failure(
101 CasErrc::InvalidArgument,
102 "inference expression must not be null",
103 "query_domain_of_checked");
104 }
105 try {
106 // Rational: Integer subset Rational, so Integer implies Rational
107 auto integer = query_integer_checked(expr);
108 if (!integer) return InferenceTriboolResult::failure(integer.error());
109 if (integer.value() == Tribool::True) {
110 return InferenceTriboolResult::success(Tribool::True);
111 }
112 // Check if it's a Rational number literal
113 if (auto num = std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(expr))) {
114 if (std::holds_alternative<Rational>(num->value())) {
115 return InferenceTriboolResult::success(Tribool::True);
116 }
117 }
118 // Check variable domain
119 if (auto var = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(expr))) {
120 const auto& props = impl_->ctx.current_properties();
121 if (props.has_domain(var->name(), Domain::Rational)) {
122 return InferenceTriboolResult::success(Tribool::True);
123 }
124 }
125 return InferenceTriboolResult::success(Tribool::Unknown);
126 } catch (const std::bad_alloc&) {
127 return InferenceTriboolResult::failure(
128 CasErrc::ResourceLimit,
129 "inference query allocation failed",
130 "query_domain_of_checked");
131 } catch (const std::exception& ex) {
132 return InferenceTriboolResult::failure(
133 CasErrc::InternalInvariant,
134 ex.what(),
135 "query_domain_of_checked");
136 }
137 }
138 case Domain::Natural: {
139 return checked_inference_result<Tribool>(expr, "query_domain_of_checked",
140 [&]() {
141 // Natural: non-negative integers (0, 1, 2, ...)
142 // Check if it's a non-negative integer literal
143 if (auto num = std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(expr))) {
144 if (is_integer_number(*num)) {
145 if (std::holds_alternative<BigInt>(num->value())) {
146 const auto& b = std::get<BigInt>(num->value());
147 if (!b.IsNegative()) return Tribool::True;
148 } else if (std::holds_alternative<Rational>(num->value())) {
149 BigInt n = std::get<Rational>(num->value()).get_numerator();
150 if (!n.IsNegative()) return Tribool::True;
151 } else {
152 double v = std::get<lmmc_real_t>(num->value());
153 if (std::isfinite(v) && v >= 0.0 && v == std::floor(v))
154 return Tribool::True;
155 }
156 }
157 return Tribool::False;
158 }
159 // Check variable domain
160 if (auto var = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(expr))) {
161 const auto& props = impl_->ctx.current_properties();
162 if (props.has_domain(var->name(), Domain::Natural)) return Tribool::True;
163 }
164 return Tribool::Unknown;
165 });
166 }
167 default:
168 return InferenceTriboolResult::success(Tribool::Unknown);
169 }
170}

181InferenceTriboolResult InferenceEngine::query_positive_checked(const SymbolicExpr& expr) const {
182 const int infinity = infinity_sign(expr);
183 if (infinity != 0) {
184 return InferenceTriboolResult::success(
185 infinity > 0 ? Tribool::True : Tribool::False);
186 }
187 return checked_inference_result<Tribool>(expr, "query_positive_checked",
188 [&]() {
189 // Depth guard: detect cycles and enforce depth limit
190 DepthGuard guard(*this, lamina::detail::node(expr).get());
191 if (guard.should_abort()) return Tribool::Unknown;
192
193 // Handle NumberNode: determine sign directly from numeric value
194 if (auto num = std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(expr))) {
195 if (num->is_positive()) return Tribool::True;
196 if (num->is_zero()) return Tribool::False;
197 // Check if negative; if not negative and not zero, it must be positive.
198 if (std::holds_alternative<BigInt>(num->value())) {
199 return std::get<BigInt>(num->value()).IsNegative() ? Tribool::False : Tribool::True;
200 }
201 if (std::holds_alternative<Rational>(num->value())) {
202 return std::get<Rational>(num->value()) < Rational(0) ? Tribool::False : Tribool::True;
203 }
204 if (std::holds_alternative<lmmc_real_t>(num->value())) {
205 lmmc_real_t v = std::get<lmmc_real_t>(num->value());
206 if (!std::isfinite(v)) return Tribool::Unknown;
207 if (v > 0.0) return Tribool::True;
208 if (v < 0.0) return Tribool::False;
209 return Tribool::False; // zero
210 }
211 return Tribool::Unknown;
212 }
213
214 // Handle VariableNode: check PropertyStore in the AssumptionContext
215 if (auto var = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(expr))) {
216 const auto& props = impl_->ctx.current_properties();
217 if (props.has_sign(var->name(), Sign::Positive)) return Tribool::True;
218 // If NonPositive or Negative or Zero, then not positive
219 if (props.has_sign(var->name(), Sign::Negative) ||
220 props.has_sign(var->name(), Sign::Zero) ||
221 props.has_sign(var->name(), Sign::NonPositive)) return Tribool::False;
222 return Tribool::Unknown;
223 }
224
225 if (auto add = std::dynamic_pointer_cast<const AddNode>(lamina::detail::node(expr))) {
226 auto result = detail::propagate_result(infer_add_sign_checked(add.get(), Sign::Positive));
227 if (result != Tribool::Unknown) return result;
228 return detail::propagate_result(infer_sign_from_relations_checked(expr, Sign::Positive));
229 }
230 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(expr))) {
231 auto result = detail::propagate_result(infer_multiply_sign_checked(mul.get(), Sign::Positive));
232 if (result != Tribool::Unknown) return result;
233 return detail::propagate_result(infer_sign_from_relations_checked(expr, Sign::Positive));
234 }
235 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(lamina::detail::node(expr))) {
236 return detail::propagate_result(infer_power_property_checked(pow.get(), Sign::Positive));
237 }
238 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(expr))) {
239 return detail::propagate_result(infer_function_property_checked(func.get(), Sign::Positive));
240 }
241 // For other expression types (e.g., variables handled above), check relations
242 return detail::propagate_result(infer_sign_from_relations_checked(expr, Sign::Positive));
243 });
244}

555 if (auto var = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(expr))) {
556 const auto& props = impl_->ctx.current_properties();
557 if (props.has_sign(var->name(), Sign::NonZero)) return Tribool::True;
558 if (props.has_sign(var->name(), Sign::Zero)) return Tribool::False;
559 return Tribool::Unknown;
560 }
561
562 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(expr))) {
563 return detail::propagate_result(infer_multiply_sign_checked(mul.get(), Sign::NonZero));
564 }
565 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(lamina::detail::node(expr))) {
566 return detail::propagate_result(infer_power_property_checked(pow.get(), Sign::NonZero));
567 }
568 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(expr))) {
569 return detail::propagate_result(infer_function_property_checked(func.get(), Sign::NonZero));
570 }
571 // For addition, NonZero is hard to determine in general
572 // (positive + positive = nonzero, but that's covered by positive inference)
573 if (auto add = std::dynamic_pointer_cast<const AddNode>(lamina::detail::node(expr))) {
574 // If the sum is positive or negative, it's nonzero
575 auto pos = detail::propagate_result(infer_add_sign_checked(add.get(), Sign::Positive));
576 if (pos == Tribool::True) return Tribool::True;
577 auto neg = detail::propagate_result(infer_add_sign_checked(add.get(), Sign::Negative));
578 if (neg == Tribool::True) return Tribool::True;
579 return Tribool::Unknown;
580 }
581 return Tribool::Unknown;
582 });
583}

591InferenceTriboolResult InferenceEngine::infer_subtraction_sign_checked(const void* opaque_node, Sign target) const {
592 if (!opaque_node) {
593 return InferenceTriboolResult::failure(
594 CasErrc::InvalidArgument,
595 "subtraction inference node must not be null",
596 "infer_subtraction_sign_checked");
597 }
598
599 try {
600 const auto& node = *static_cast<const AddNode*>(opaque_node);
601 if (node.operands().size() != 2) return InferenceTriboolResult::success(Tribool::Unknown);
602
603 std::shared_ptr<const SymbolicNode> minuend_node;
604 std::shared_ptr<const SymbolicNode> subtrahend_node;
605
606 for (const auto& operand : node.operands()) {
607 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(operand);
608 if (mul && mul->operands().size() == 2) {
609 bool found_neg_one = false;
610 std::shared_ptr<const SymbolicNode> other_operand;
611 for (const auto& mul_op : mul->operands()) {
612 auto num = std::dynamic_pointer_cast<const NumberNode>(mul_op);
613 if (num) {
614 bool is_neg_one = false;
615 if (std::holds_alternative<BigInt>(num->value())) {
616 is_neg_one = (std::get<BigInt>(num->value()) == BigInt(-1));
617 } else if (std::holds_alternative<Rational>(num->value())) {
618 is_neg_one = (std::get<Rational>(num->value()) == Rational(-1));
619 } else if (std::holds_alternative<lmmc_real_t>(num->value())) {
620 lmmc_real_t v = std::get<lmmc_real_t>(num->value());
621 is_neg_one = (std::isfinite(v) && v == -1.0);
622 }
623 if (is_neg_one) {
624 found_neg_one = true;
625 continue;
626 }
627 }
628 other_operand = mul_op;
629 }
630 if (found_neg_one && other_operand) {
631 subtrahend_node = other_operand;
632 } else {
633 minuend_node = operand;
634 }
635 } else {
636 minuend_node = operand;
637 }
638 }
639
640 if (!minuend_node || !subtrahend_node) {
641 return InferenceTriboolResult::success(Tribool::Unknown);
642 }
643
644 auto minuend_expr = lamina::detail::expression_from_node(minuend_node);
645 auto subtrahend_expr = lamina::detail::expression_from_node(subtrahend_node);
646
647 Tribool min_pos = detail::propagate_result(query_sign_of_checked(minuend_expr, Sign::Positive));
648 Tribool min_neg = detail::propagate_result(query_sign_of_checked(minuend_expr, Sign::Negative));
649 Tribool sub_pos = detail::propagate_result(query_sign_of_checked(subtrahend_expr, Sign::Positive));
650 Tribool sub_neg = detail::propagate_result(query_sign_of_checked(subtrahend_expr, Sign::Negative));
651
652 if (min_pos == Tribool::True && sub_neg == Tribool::True) {
653 switch (target) {
654 case Sign::Positive: return InferenceTriboolResult::success(Tribool::True);
655 case Sign::NonNegative: return InferenceTriboolResult::success(Tribool::True);
656 case Sign::NonZero: return InferenceTriboolResult::success(Tribool::True);
657 case Sign::Negative: return InferenceTriboolResult::success(Tribool::False);
658 case Sign::NonPositive: return InferenceTriboolResult::success(Tribool::False);
659 case Sign::Zero: return InferenceTriboolResult::success(Tribool::False);
660 }
661 }
662
663 if (min_neg == Tribool::True && sub_pos == Tribool::True) {
664 switch (target) {
665 case Sign::Negative: return InferenceTriboolResult::success(Tribool::True);
666 case Sign::NonPositive: return InferenceTriboolResult::success(Tribool::True);
667 case Sign::NonZero: return InferenceTriboolResult::success(Tribool::True);
668 case Sign::Positive: return InferenceTriboolResult::success(Tribool::False);
669 case Sign::NonNegative: return InferenceTriboolResult::success(Tribool::False);
670 case Sign::Zero: return InferenceTriboolResult::success(Tribool::False);
671 }
672 }
673
674 Tribool min_nn = detail::propagate_result(query_sign_of_checked(minuend_expr, Sign::NonNegative));
675 Tribool sub_np = detail::propagate_result(query_sign_of_checked(subtrahend_expr, Sign::NonPositive));
676 if (min_nn == Tribool::True && sub_np == Tribool::True) {
677 switch (target) {
678 case Sign::NonNegative: return InferenceTriboolResult::success(Tribool::True);
679 case Sign::Negative: return InferenceTriboolResult::success(Tribool::False);
680 default: break;
681 }
682 }
683
684 Tribool min_np = detail::propagate_result(query_sign_of_checked(minuend_expr, Sign::NonPositive));
685 Tribool sub_nn = detail::propagate_result(query_sign_of_checked(subtrahend_expr, Sign::NonNegative));
686 if (min_np == Tribool::True && sub_nn == Tribool::True) {
687 switch (target) {
688 case Sign::NonPositive: return InferenceTriboolResult::success(Tribool::True);
689 case Sign::Positive: return InferenceTriboolResult::success(Tribool::False);
690 default: break;
691 }
692 }
693
694 return InferenceTriboolResult::success(Tribool::Unknown);
695 } catch (const detail::ResultPropagation& ex) {
696 return InferenceTriboolResult::failure(ex.error());
697 } catch (const std::bad_alloc&) {
698 return InferenceTriboolResult::failure(
699 CasErrc::ResourceLimit,
700 "subtraction sign inference allocation failed",
701 "infer_subtraction_sign_checked");
702 } catch (const std::exception& ex) {
703 return InferenceTriboolResult::failure(
704 CasErrc::InternalInvariant,
705 ex.what(),
706 "infer_subtraction_sign_checked");
707 }
708}

717 return checked_inference_result<Tribool>(expr, "query_algebraic_checked",
718 [&]() {
719 DepthGuard guard(*this, lamina::detail::node(expr).get());
720 if (guard.should_abort()) return Tribool::Unknown;
721
722 // NumberNode: BigInt and Rational are algebraic; finite doubles are Unknown
723 if (auto num = std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(expr))) {
724 if (std::holds_alternative<BigInt>(num->value())) return Tribool::True;
725 if (std::holds_alternative<Rational>(num->value())) return Tribool::True;
726 return Tribool::Unknown;
727 }
728
729 // VariableNode: check if domain is Algebraic or more specific
730 if (auto var = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(expr))) {
731 const auto& props = impl_->ctx.current_properties();
732 // Domain hierarchy: Algebraic ⊃ Rational ⊃ Integer ⊃ Natural ⊃ PositiveInt
733 if (props.has_domain(var->name(), Domain::Algebraic)) return Tribool::True;
734 // If transcendental, definitely not algebraic
735 if (props.is_transcendental(var->name())) return Tribool::False;
736 return Tribool::Unknown;
737 }
738
739 // Composite expressions: return Unknown for now
740 return Tribool::Unknown;
741 });
