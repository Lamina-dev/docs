103std::shared_ptr<SymbolicExpr> divergence(const VectorField& F,
104 const std::vector<std::string>& vars)
105{
106 if (F.size() != vars.size()) {
107 throw std::invalid_argument(
108 "divergence: F and vars must have the same dimension");
109 }
110 if (F.empty()) {
111 return SymbolicExpr::number(0);
112 }
113
114 std::shared_ptr<SymbolicExpr> sum = nullptr;
115 for (size_t i = 0; i < F.size(); ++i) {
116 if (!F[i]) {
117 continue;
118 }
119 auto partial = F[i]->differentiate(vars[i]);
120 if (!partial) {
121 continue;
122 }
123 partial = partial->simplify();
124 if (!sum) {
125 sum = partial;
126 } else {
127 sum = SymbolicExpr::add(sum, partial);
128 sum = sum->simplify();
129 }
130 }
131
132 if (!sum) {
133 return SymbolicExpr::number(0);
134 }
135 return sum->simplify();

182VectorField curl(const VectorField& F,
183 const std::vector<std::string>& vars)
184{
185 if (F.size() != vars.size()) {
186 throw std::invalid_argument(
187 "curl: F and vars must have the same dimension");
188 }
189
190 /// 二维标量旋度: ∂F₂/∂x₁ - ∂F₁/∂x₂
191 if (F.size() == 2 && vars.size() == 2) {
192 auto dF2_dx1 = F[1]->differentiate(vars[0]);
193 auto dF1_dx2 = F[0]->differentiate(vars[1]);
194
195 auto neg_dF1_dx2 = SymbolicExpr::multiply(
196 SymbolicExpr::number(-1), dF1_dx2);
197 auto scalar_curl = SymbolicExpr::add(dF2_dx1, neg_dF1_dx2);
198 scalar_curl = scalar_curl->simplify();
199
200 return VectorField{scalar_curl};
201 }
202
203 /// 三维旋度
204 if (F.size() != 3 || vars.size() != 3) {
205 throw std::invalid_argument(
206 "curl: requires 2D or 3D vector field");
207 }
208
209 /// curl_x = ∂F₃/∂x₂ - ∂F₂/∂x₃
210 auto dF3_dx2 = F[2]->differentiate(vars[1]);
211 auto dF2_dx3 = F[1]->differentiate(vars[2]);
212 auto curl_x = SymbolicExpr::add(
213 dF3_dx2,
214 SymbolicExpr::multiply(SymbolicExpr::number(-1), dF2_dx3));
215 curl_x = curl_x->simplify();
216
217 /// curl_y = ∂F₁/∂x₃ - ∂F₃/∂x₁
218 auto dF1_dx3 = F[0]->differentiate(vars[2]);
219 auto dF3_dx1 = F[2]->differentiate(vars[0]);
220 auto curl_y = SymbolicExpr::add(
221 dF1_dx3,
222 SymbolicExpr::multiply(SymbolicExpr::number(-1), dF3_dx1));
223 curl_y = curl_y->simplify();
224
225 /// curl_z = ∂F₂/∂x₁ - ∂F₁/∂x₂
226 auto dF2_dx1 = F[1]->differentiate(vars[0]);
227 auto dF1_dx2 = F[0]->differentiate(vars[1]);
228 auto curl_z = SymbolicExpr::add(
229 dF2_dx1,
230 SymbolicExpr::multiply(SymbolicExpr::number(-1), dF1_dx2));
231 curl_z = curl_z->simplify();
232
233 return VectorField{curl_x, curl_y, curl_z};
234}
