59Result<std::pair<ExprPtr, ExprPtr>> prepare_operands(
60 const SymbolicExpr& lhs, const SymbolicExpr& rhs,
61 ComputationContext& context, detail::RewriteBudget& budget) {
62 using OperandsResult = Result<std::pair<ExprPtr, ExprPtr>>;
63 auto left = std::make_shared<SymbolicExpr>(lhs);
64 auto right = std::make_shared<SymbolicExpr>(rhs);
65 if (auto quantity = std::dynamic_pointer_cast<const QuantityNode>(detail::node(lhs))) {
66 normalization_check_arithmetic<MultiplyNode>(&budget, 1, quantity->value());
67 auto stripped = strip_unit(left, UnitStripMode::BaseValue, context);
68 if (!stripped) { return OperandsResult::failure(stripped.error()); }
69 left = stripped.value();
70 }
71 if (auto quantity = std::dynamic_pointer_cast<const QuantityNode>(detail::node(rhs))) {
72 normalization_check_arithmetic<MultiplyNode>(&budget, 1, quantity->value());
73 auto stripped = strip_unit(right, UnitStripMode::BaseValue, context);
74 if (!stripped) { return OperandsResult::failure(stripped.error()); }
75 right = stripped.value();
76 }
77 return OperandsResult::success({std::move(left), std::move(right)});
78}
79Result<bool> apply_profile(const SymbolicExpr& lhs, const SymbolicExpr& rhs,
80 ComputationContext& context,
81 const EqvOptions& options, detail::RewriteBudget& budget) {
82 if (options.profile != EqvProfile::TrigBasic &&
83 options.profile != EqvProfile::ExpLogBasic) {
84 return prove_operands(lhs, rhs, context, options, budget);
85 }
86 const bool trig = options.profile == EqvProfile::TrigBasic;
87 budget.consume();
88 ExprPtr rewritten_lhs;
89 ExprPtr rewritten_rhs;
90 if (trig) {
91 rewritten_lhs = equivalence_detail::rewrite_trig_basic_identity(detail::node(lhs), budget);
92 rewritten_rhs = equivalence_detail::rewrite_trig_basic_identity(detail::node(rhs), budget);
93 } else {
94 rewritten_lhs = equivalence_detail::rewrite_exp_log_basic_identity(
95 detail::node(lhs), context.assumptions().get(), budget);
96 rewritten_rhs = equivalence_detail::rewrite_exp_log_basic_identity(
97 detail::node(rhs), context.assumptions().get(), budget);
98 }
99 rewritten_lhs = equivalence_detail::normalize_equivalence(rewritten_lhs, budget);
100 rewritten_rhs = equivalence_detail::normalize_equivalence(rewritten_rhs, budget);
101 EqvOptions core_options = options;
102 core_options.profile = EqvProfile::Core;
103 return equivalent_with_budget(*rewritten_lhs, *rewritten_rhs, context, core_options, &budget);
104}

106Result<bool> prove_operands(const SymbolicExpr& lhs, const SymbolicExpr& rhs,
107 ComputationContext& context,
108 const EqvOptions& options, detail::RewriteBudget& budget) {
109 budget.consume();
110 auto operands = prepare_operands(lhs, rhs, context, budget);
111 if (!operands) { return Result<bool>::failure(operands.error()); }
112 auto left = equivalence_detail::canonicalize_complex_product(
113 *operands.value().first, budget);
114 auto right = equivalence_detail::canonicalize_complex_product(
115 *operands.value().second, budget);
116 if (structurally_equal(*left, *right)) {
117 return Result<bool>::success(true);
118 }
119 normalization_check_arithmetic<MultiplyNode>(&budget, 1, detail::node(right));
120 auto negative = SymbolicExpr::multiply(SymbolicExpr::number(-1), right);
121 normalization_check_arithmetic<AddNode>(&budget, 0, detail::node(left), detail::node(negative));
122 auto difference = SymbolicExpr::add(left, negative);
123 if (!difference) {
124 return Result<bool>::failure(CasErrc::InternalInvariant,
125 "equivalence difference construction failed",
126 kEquivalentOperation);
127 }
128 difference = equivalence_detail::normalize_equivalence(difference, budget);
129 if (difference->is_zero()) {
130 return Result<bool>::success(true);
131 }
132 auto polynomial = equivalence_detail::prove_rational_polynomial_equivalence(
133 difference, context, options, budget);
134 if (!polynomial) { return Result<bool>::failure(polynomial.error()); }
135 if (polynomial.value()) { return Result<bool>::success(*polynomial.value()); }
136 return Result<bool>::failure(CasErrc::Inconclusive,
137 "equivalence proof is inconclusive",
138 kEquivalentOperation);
139}

141Result<bool> equivalent_with_budget(const SymbolicExpr& lhs, const SymbolicExpr& rhs,
142 ComputationContext& context, const EqvOptions& options,
143 detail::RewriteBudget* budget) {
144 try {
145 auto valid = equivalence_detail::validate_eqv_options(options);
146 if (!valid) { return Result<bool>::failure(valid.error()); }
147 auto step = context.consume_steps(1, kEquivalentOperation);
148 if (!step) { return Result<bool>::failure(step.error()); }
149 auto lhs_dimension = dimension_of(lhs);
150 if (!lhs_dimension) { return Result<bool>::failure(lhs_dimension.error()); }
151 auto rhs_dimension = dimension_of(rhs);
152 if (!rhs_dimension) { return Result<bool>::failure(rhs_dimension.error()); }
153 if (lhs_dimension.value() != rhs_dimension.value()) {
154 return Result<bool>::success(false);
155 }
156 if (budget) {
157 return options.profile == EqvProfile::Core
158 ? prove_operands(lhs, rhs, context, options, *budget)
159 : apply_profile(lhs, rhs, context, options, *budget);
160 }
161 const auto input_nodes = detail::RewriteBudget::input_nodes(
162 detail::node(lhs), detail::node(rhs), context,
163 options.budget.max_rewrite_depth, kEquivalentOperation);
164 const auto maximum = std::numeric_limits<std::size_t>::max();
165 const auto factor = options.budget.max_node_growth_factor;
166 const auto max_nodes = input_nodes > maximum / factor
167 ? maximum : input_nodes * factor;
168 detail::RewriteBudget initial_budget(
169 context, options.budget.max_rewrite_steps,
170 options.budget.max_rewrite_depth, max_nodes, kEquivalentOperation);
171 return options.profile == EqvProfile::Core
172 ? prove_operands(lhs, rhs, context, options, initial_budget)
173 : apply_profile(lhs, rhs, context, options, initial_budget);
174 } catch (const CasError& error) {
175 return Result<bool>::failure(error);
176 } catch (const std::bad_alloc&) {
177 return Result<bool>::failure(CasErrc::ResourceLimit,
178 "equivalence check allocation failed",
179 kEquivalentOperation);
180 } catch (const std::exception& error) {
181 return Result<bool>::failure(CasErrc::Inconclusive, error.what(),
182 kEquivalentOperation);
183 }
184}
