46Result<std::pair<ExprPtr, ExprPtr>> prepare_operands(
47 const SymbolicExpr& lhs, const SymbolicExpr& rhs,
48 ComputationContext& context, detail::RewriteBudget& budget) {
49 using OperandsResult = Result<std::pair<ExprPtr, ExprPtr>>;
50 auto left = std::make_shared<SymbolicExpr>(lhs);
51 auto right = std::make_shared<SymbolicExpr>(rhs);
52 if (auto quantity = std::dynamic_pointer_cast<const QuantityNode>(detail::node(lhs))) {
53 normalization_check_arithmetic<MultiplyNode>(&budget, 1, quantity->value());
54 auto stripped = strip_unit(left, UnitStripMode::BaseValue, context);
55 if (!stripped) { return OperandsResult::failure(stripped.error()); }
56 left = stripped.value();
57 }
58 if (auto quantity = std::dynamic_pointer_cast<const QuantityNode>(detail::node(rhs))) {
59 normalization_check_arithmetic<MultiplyNode>(&budget, 1, quantity->value());
60 auto stripped = strip_unit(right, UnitStripMode::BaseValue, context);
61 if (!stripped) { return OperandsResult::failure(stripped.error()); }
62 right = stripped.value();
63 }
64 return OperandsResult::success({std::move(left), std::move(right)});
65}
66Result<bool> apply_profile(const SymbolicExpr& lhs, const SymbolicExpr& rhs,
67 ComputationContext& context,
68 const EqvOptions& options, detail::RewriteBudget& budget) {
69 if (options.profile != EqvProfile::TrigBasic &&
70 options.profile != EqvProfile::ExpLogBasic) {
71 return prove_operands(lhs, rhs, context, options, budget);
72 }
73 const bool trig = options.profile == EqvProfile::TrigBasic;
74 budget.consume();
75 ExprPtr rewritten_lhs;
76 ExprPtr rewritten_rhs;
77 if (trig) {
78 rewritten_lhs = equivalence_detail::rewrite_trig_basic_identity(detail::node(lhs), budget);
79 rewritten_rhs = equivalence_detail::rewrite_trig_basic_identity(detail::node(rhs), budget);
80 } else {
81 rewritten_lhs = equivalence_detail::rewrite_exp_log_basic_identity(
82 detail::node(lhs), context.assumptions().get(), budget);
83 rewritten_rhs = equivalence_detail::rewrite_exp_log_basic_identity(
84 detail::node(rhs), context.assumptions().get(), budget);
85 }
86 rewritten_lhs = equivalence_detail::normalize_equivalence(rewritten_lhs, budget);
87 rewritten_rhs = equivalence_detail::normalize_equivalence(rewritten_rhs, budget);
88 EqvOptions core_options = options;
89 core_options.profile = EqvProfile::Core;
90 return equivalent_with_budget(*rewritten_lhs, *rewritten_rhs, context, core_options, &budget);
91}

93Result<bool> prove_operands(const SymbolicExpr& lhs, const SymbolicExpr& rhs,
94 ComputationContext& context,
95 const EqvOptions& options, detail::RewriteBudget& budget) {
96 budget.consume();
97 auto operands = prepare_operands(lhs, rhs, context, budget);
98 if (!operands) { return Result<bool>::failure(operands.error()); }
99 auto left = equivalence_detail::canonicalize_complex_product(
100 *operands.value().first, budget);
101 auto right = equivalence_detail::canonicalize_complex_product(
102 *operands.value().second, budget);
103 if (structurally_equal(*left, *right)) {
104 return Result<bool>::success(true);
105 }
106 normalization_check_arithmetic<MultiplyNode>(&budget, 1, detail::node(right));
107 auto negative = SymbolicExpr::multiply(SymbolicExpr::number(-1), right);
108 normalization_check_arithmetic<AddNode>(&budget, 0, detail::node(left), detail::node(negative));
109 auto difference = SymbolicExpr::add(left, negative);
110 if (!difference) {
111 return Result<bool>::failure(CasErrc::InternalInvariant,
112 "equivalence difference construction failed",
113 kEquivalentOperation);
114 }
115 difference = equivalence_detail::normalize_equivalence(difference, budget);
116 const auto value = detail::exact_rational_value(detail::node(difference));
117 if (value && value->is_zero()) {
118 return Result<bool>::success(true);
119 }
120 auto polynomial = equivalence_detail::prove_rational_polynomial_equivalence(
121 difference, context, options, budget);
122 if (!polynomial) { return Result<bool>::failure(polynomial.error()); }
123 if (polynomial.value()) { return Result<bool>::success(*polynomial.value()); }
124 return Result<bool>::failure(CasErrc::Inconclusive,
125 "equivalence proof is inconclusive",
126 kEquivalentOperation);
127}

129Result<bool> prove_checked_operands(
130 const SymbolicExpr& lhs, const SymbolicExpr& rhs,
131 ComputationContext& context, const EqvOptions& options,
132 detail::RewriteBudget& budget) {
133 budget.consume();
134 budget.measure(detail::node(lhs));
135 budget.measure(detail::node(rhs));
136 if (structurally_equal(lhs, rhs)) { return true; }
137 auto lhs_dimension = dimension_of(lhs);
138 if (!lhs_dimension) { return Result<bool>::failure(lhs_dimension.error()); }
139 auto rhs_dimension = dimension_of(rhs);
140 if (!rhs_dimension) { return Result<bool>::failure(rhs_dimension.error()); }
141 if (lhs_dimension.value() != rhs_dimension.value()) {
142 return Result<bool>::success(false);
143 }
144 return options.profile == EqvProfile::Core
145 ? prove_operands(lhs, rhs, context, options, budget)
146 : apply_profile(lhs, rhs, context, options, budget);
147}

149Result<bool> equivalent_with_budget(const SymbolicExpr& lhs, const SymbolicExpr& rhs,
150 ComputationContext& context, const EqvOptions& options,
151 detail::RewriteBudget* budget) {
152 try {
153 auto valid = equivalence_detail::validate_eqv_options(options);
154 if (!valid) { return Result<bool>::failure(valid.error()); }
155 auto step = context.consume_steps(1, kEquivalentOperation);
156 if (!step) { return Result<bool>::failure(step.error()); }
157 if (budget) {
158 return prove_checked_operands(lhs, rhs, context, options, *budget);
159 }
160 const auto input_nodes = detail::RewriteBudget::input_nodes(
161 detail::node(lhs), detail::node(rhs), context,
162 options.budget.max_rewrite_depth, kEquivalentOperation);
163 const auto maximum = std::numeric_limits<std::size_t>::max();
164 const auto factor = options.budget.max_node_growth_factor;
165 const auto growth_nodes = input_nodes > maximum / factor
166 ? maximum : input_nodes * factor;
167 const auto max_nodes = std::min(context.limits().max_ast_nodes, growth_nodes);
168 detail::RewriteBudget initial_budget(
169 context, options.budget.max_rewrite_steps,
170 options.budget.max_rewrite_depth, max_nodes, kEquivalentOperation);
171 return prove_checked_operands(lhs, rhs, context, options, initial_budget);
172 } catch (const CasError& error) {
173 return Result<bool>::failure(error);
174 } catch (const std::length_error& error) {
175 return Result<bool>::failure(CasErrc::ResourceLimit, error.what(),
176 kEquivalentOperation);
177 } catch (const std::bad_alloc&) {
178 return Result<bool>::failure(CasErrc::ResourceLimit,
179 "equivalence check allocation failed",
180 kEquivalentOperation);
181 } catch (const std::exception& error) {
182 return Result<bool>::failure(CasErrc::InternalInvariant, error.what(),
183 kEquivalentOperation);
184 }
185}
