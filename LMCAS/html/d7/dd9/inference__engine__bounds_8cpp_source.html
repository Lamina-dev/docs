9 * @brief Helper: extract a numeric double value from an Endpoint.
10 * Returns std::nullopt if the endpoint is not a finite numeric value.
11 */
12std::optional<double> endpoint_to_double(const Endpoint& ep) {
13 if (ep.is_neg_infinity || ep.is_pos_infinity) return std::nullopt;
14 if (!ep.value || !LMCAS::detail::node(ep.value)) return std::nullopt;
15 ComputationContext context;
16 auto evaluated = evaluate_numeric(*ep.value, NumericBindings{}, context);
17 if (!evaluated || !evaluated.value().is_finite()) return std::nullopt;
18 double v = evaluated.value().value;
19 if (!std::isfinite(v)) return std::nullopt;
20 return v;
21}

42/**
43 * @brief Helper: check if an exponent NumberNode is exactly 2 (for squaring detection).
44 */
45static bool is_exponent_two(const std::shared_ptr<const SymbolicNode>& node) {
46 auto num = std::dynamic_pointer_cast<const NumberNode>(node);
47 if (!num) return false;
48 if (std::holds_alternative<BigInt>(num->value())) {
49 return std::get<BigInt>(num->value()) == BigInt(2);
50 }
51 if (std::holds_alternative<Rational>(num->value())) {
52 return std::get<Rational>(num->value()) == Rational(2);
53 }
54 if (std::holds_alternative<lmmc_real_t>(num->value())) {
55 lmmc_real_t v = std::get<lmmc_real_t>(num->value());
56 return std::isfinite(v) && v == 2.0;
57 }
58 return false;
59}

80std::optional<Interval> InferenceEngine::propagate_bounds(const SymbolicExpr& expr) const {
81 if (!LMCAS::detail::node(expr)) return std::nullopt;
82
83 // --- NumberNode: point interval [n, n] ---
84 if (auto num = std::dynamic_pointer_cast<const NumberNode>(LMCAS::detail::node(expr))) {
85 double val = 0.0;
86 if (std::holds_alternative<BigInt>(num->value())) {
87 val = std::get<BigInt>(num->value()).to_double();
88 } else if (std::holds_alternative<Rational>(num->value())) {
89 val = std::get<Rational>(num->value()).to_double();
90 } else {
91 val = std::get<lmmc_real_t>(num->value());
92 }
93 if (!std::isfinite(val)) return std::nullopt;
94 return make_interval(val, val);
95 }
96
97 // --- VariableNode: look up bounds from AssumptionContext ---
98 if (auto var = std::dynamic_pointer_cast<const VariableNode>(LMCAS::detail::node(expr))) {
99 auto bounds = impl_->ctx.get_bounds(var->name());
100 if (!bounds.has_value()) return std::nullopt;
101
102 // Verify the bounds have numeric endpoints
103 auto lo = endpoint_to_double(bounds->lower);
104 auto hi = endpoint_to_double(bounds->upper);
105 if (!lo.has_value() || !hi.has_value()) return std::nullopt;
106
107 return bounds.value();
108 }
109
110 // --- AddNode: [a,b] + [c,d] → [a+c, b+d] for each operand ---
111 if (auto add = std::dynamic_pointer_cast<const AddNode>(LMCAS::detail::node(expr))) {
112 if (add->operands().empty()) return std::nullopt;
113
114 // Start with the bounds of the first operand
115 auto first_expr = LMCAS::detail::expression_from_node(add->operands()[0]);
116 auto result = propagate_bounds(first_expr);
117 if (!result.has_value()) return std::nullopt;
118
119 // Accumulate bounds for remaining operands
120 for (size_t i = 1; i < add->operands().size(); ++i) {
121 auto op_expr = LMCAS::detail::expression_from_node(add->operands()[i]);
122 auto op_bounds = propagate_bounds(op_expr);
123 if (!op_bounds.has_value()) return std::nullopt;
124
125 auto r_lo = endpoint_to_double(result->lower);
126 auto r_hi = endpoint_to_double(result->upper);
127 auto o_lo = endpoint_to_double(op_bounds->lower);
128 auto o_hi = endpoint_to_double(op_bounds->upper);
129
130 if (!r_lo || !r_hi || !o_lo || !o_hi) return std::nullopt;
131
132 // Addition: [a,b] + [c,d] = [a+c, b+d]
133 result = make_interval(*r_lo + *o_lo, *r_hi + *o_hi);
134 }
135
136 return result;
137 }
138
139 // --- MultiplyNode: handle multiplication and division ---
140 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(LMCAS::detail::node(expr))) {
141 if (mul->operands().empty()) return std::nullopt;
142
143 // Separate operands into numerator terms and denominator terms (PowerNode with exp -1)
144 std::vector<std::shared_ptr<const SymbolicNode>> numerator_ops;
145 std::vector<std::shared_ptr<const SymbolicNode>> denominator_ops;
146
147 for (const auto& operand : mul->operands()) {
148 auto pow_node = std::dynamic_pointer_cast<const PowerNode>(operand);
149 if (pow_node && is_exponent_neg_one(pow_node->exponent())) {
150 // This is a division: base^(-1) means we divide by base
151 denominator_ops.push_back(pow_node->base());
152 } else {
153 numerator_ops.push_back(operand);
154 }
155 }
156
157 // Compute numerator product bounds
158 std::optional<Interval> num_result;
159 if (numerator_ops.empty()) {
160 // All operands are denominators; numerator is implicitly 1
161 num_result = make_interval(1.0, 1.0);
162 } else {
163 auto first_expr = LMCAS::detail::expression_from_node(numerator_ops[0]);
164 num_result = propagate_bounds(first_expr);
165 if (!num_result.has_value()) return std::nullopt;
166
167 for (size_t i = 1; i < numerator_ops.size(); ++i) {
168 auto op_expr = LMCAS::detail::expression_from_node(numerator_ops[i]);
169 auto op_bounds = propagate_bounds(op_expr);
170 if (!op_bounds.has_value()) return std::nullopt;
171
172 auto a = endpoint_to_double(num_result->lower);
173 auto b = endpoint_to_double(num_result->upper);
174 auto c = endpoint_to_double(op_bounds->lower);
175 auto d = endpoint_to_double(op_bounds->upper);
176
177 if (!a || !b || !c || !d) return std::nullopt;
178
179 // Multiplication: [a,b] * [c,d] = [min(ac,ad,bc,bd), max(ac,ad,bc,bd)]
180 double ac = (*a) * (*c);
181 double ad = (*a) * (*d);
182 double bc = (*b) * (*c);
183 double bd = (*b) * (*d);
184
185 double lo = std::min({ac, ad, bc, bd});
186 double hi = std::max({ac, ad, bc, bd});
187 num_result = make_interval(lo, hi);
188 }
189 }
190
191 // If no denominator terms, return the numerator result
192 if (denominator_ops.empty()) {
193 return num_result;
194 }
195
196 // Compute denominator product bounds, then divide
197 // First compute the combined denominator interval
198 auto den_first_expr = LMCAS::detail::expression_from_node(denominator_ops[0]);
199 auto den_result = propagate_bounds(den_first_expr);
200 if (!den_result.has_value()) return std::nullopt;
201
202 for (size_t i = 1; i < denominator_ops.size(); ++i) {
203 auto op_expr = LMCAS::detail::expression_from_node(denominator_ops[i]);
204 auto op_bounds = propagate_bounds(op_expr);
205 if (!op_bounds.has_value()) return std::nullopt;
206
207 auto a = endpoint_to_double(den_result->lower);
208 auto b = endpoint_to_double(den_result->upper);
209 auto c = endpoint_to_double(op_bounds->lower);
210 auto d = endpoint_to_double(op_bounds->upper);
211
212 if (!a || !b || !c || !d) return std::nullopt;
213
214 double ac = (*a) * (*c);
215 double ad = (*a) * (*d);
216 double bc = (*b) * (*c);
217 double bd = (*b) * (*d);
218
219 double lo = std::min({ac, ad, bc, bd});
220 double hi = std::max({ac, ad, bc, bd});
221 den_result = make_interval(lo, hi);
222 }
223
224 // Now divide numerator by denominator
225 auto num_lo = endpoint_to_double(num_result->lower);
226 auto num_hi = endpoint_to_double(num_result->upper);
227 auto den_lo = endpoint_to_double(den_result->lower);
228 auto den_hi = endpoint_to_double(den_result->upper);
229
230 if (!num_lo || !num_hi || !den_lo || !den_hi) return std::nullopt;
231
232 // Division where divisor interval contains zero → unbounded (no finite interval)
233 if (*den_lo <= 0.0 && *den_hi >= 0.0) {
234 return std::nullopt;
235 }
236
237 // Division by positive interval [c,d] (c>0, a>=0): [a,b]/[c,d] → [a/d, b/c]
238 // General division: [a,b]/[c,d] where interval doesn't contain zero
239 // Use the general formula: compute all combinations and take min/max
240 double a = *num_lo, b = *num_hi, c = *den_lo, d = *den_hi;
241 double ac_d = a / c;
242 double ad_d = a / d;
243 double bc_d = b / c;
244 double bd_d = b / d;
245
246 double lo = std::min({ac_d, ad_d, bc_d, bd_d});
247 double hi = std::max({ac_d, ad_d, bc_d, bd_d});
248 return make_interval(lo, hi);
249 }
250
251 // --- PowerNode: handle squaring ---
252 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(LMCAS::detail::node(expr))) {
253 // Squaring: x^2 where x >= 0 → [a², b²]
254 if (is_exponent_two(pow->exponent())) {
255 auto base_expr = LMCAS::detail::expression_from_node(pow->base());
256 auto base_bounds = propagate_bounds(base_expr);
257 if (!base_bounds.has_value()) return std::nullopt;
258
259 auto a = endpoint_to_double(base_bounds->lower);
260 auto b = endpoint_to_double(base_bounds->upper);
261 if (!a || !b) return std::nullopt;
262
263 // If a >= 0 (non-negative base): [a², b²]
264 if (*a >= 0.0) {
265 return make_interval((*a) * (*a), (*b) * (*b));
266 }
267
268 // If b <= 0 (non-positive base): [b², a²]
269 if (*b <= 0.0) {
270 return make_interval((*b) * (*b), (*a) * (*a));
271 }
272
273 // If interval spans zero: [0, max(a², b²)]
274 double a_sq = (*a) * (*a);
275 double b_sq = (*b) * (*b);
276 return make_interval(0.0, std::max(a_sq, b_sq));
277 }
278
279 // For other exponents, we don't propagate bounds in this implementation
280 return std::nullopt;
281 }
282
283 // --- FunctionNode: sin/cos → [-1, 1] ---
284 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(LMCAS::detail::node(expr))) {
285 if (func->arguments().empty()) return std::nullopt;
286
287 auto arg_expr = LMCAS::detail::expression_from_node(func->arguments()[0]);
288 switch (func->type()) {
289 case FunctionNode::FuncType::Sin:
290 case FunctionNode::FuncType::Cos: {
291 // sin/cos of bounded input → [-1, 1]
292 auto arg_bounds = propagate_bounds(arg_expr);
293 if (!arg_bounds.has_value()) return std::nullopt;
294 // If the argument has finite bounds, sin/cos is bounded by [-1, 1]
295 return make_interval(-1.0, 1.0);
296 }
297 case FunctionNode::FuncType::ArcTan: {
298 auto argument_bounds = propagate_bounds(arg_expr);
299 if (argument_bounds) {
300 Interval result;
301 if (argument_bounds->lower.is_neg_infinity) {
302 result.lower = Endpoint::open(
303 SymbolicExpr::number(-LMMC_CONST_PI / 2.0));
304 } else if (auto lower =
305 endpoint_to_double(
306 argument_bounds->lower)) {
307 auto value = SymbolicExpr::number(std::atan(*lower));
308 result.lower = argument_bounds->lower.is_open
309 ? Endpoint::open(value)
310 : Endpoint::closed(value);
311 } else {
312 return std::nullopt;
313 }
314
315 if (argument_bounds->upper.is_pos_infinity) {
316 result.upper = Endpoint::open(
317 SymbolicExpr::number(LMMC_CONST_PI / 2.0));
318 } else if (auto upper =
319 endpoint_to_double(
320 argument_bounds->upper)) {
321 auto value = SymbolicExpr::number(std::atan(*upper));
322 result.upper = argument_bounds->upper.is_open
323 ? Endpoint::open(value)
324 : Endpoint::closed(value);
325 } else {
326 return std::nullopt;
327 }
328 return result;
329 }
330
331 auto real = query_real_checked(arg_expr);
332 if (real && real.value() == Tribool::True) {
333 return Interval{
334 Endpoint::open(SymbolicExpr::number(
335 -LMMC_CONST_PI / 2.0)),
336 Endpoint::open(SymbolicExpr::number(
337 LMMC_CONST_PI / 2.0))};
338 }
339 return std::nullopt;
340 }
341 default:
342 return std::nullopt;
343 }
344 }
345
346 return std::nullopt;
347}
