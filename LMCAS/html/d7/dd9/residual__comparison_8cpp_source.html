14ResidualCheckResult check_equivalent(
15 const ExprPtr& left,
16 const ExprPtr& right,
17 ComputationContext& context,
18 const LMCAS::EqvOptions& options) {
19 if (!left || !right || !detail::node(left) || !detail::node(right)) {
20 return ResidualCheckResult::failure(
21 CasErrc::InvalidArgument,
22 "equivalence check requires two expressions",
23 "residual.check_equivalent");
24 }
25 auto access = context.consume_steps(1, "residual.check_equivalent");
26 if (!access) return ResidualCheckResult::failure(access.error());
27 if (detail::node(left)->equals(*detail::node(right)))
28 return ResidualCheckResult::success(ProvedZeroResidual{
29 ExactNormalizationProof{left}});
30 try {
31 auto residual = SymbolicExpr::add(
32 left,

43 return ResidualCheckResult::failure(

68Result<Tribool> compare_defined_values(
69 const ExprPtr& left, const ExprPtr& right, const FactsQuery& facts,
70 ComputationContext& context, Domain domain) {
71 if (detail::node(left)->equals(*detail::node(right))) {
72 return Tribool::True;
73 }
74 auto residual = SymbolicExpr::add(
75 left,
76 SymbolicExpr::multiply(SymbolicExpr::number(-1), right))->simplify();
77 if (!residual) {
78 return Result<Tribool>::failure(
79 CasErrc::InternalInvariant,
80 "value comparison produced no residual",
81 "residual.compare_values");
82 }
83 auto nonzero = detail::query_nonzero_value(
84 detail::node(residual), facts, domain, context);
85 if (!nonzero) {
86 return Result<Tribool>::failure(nonzero.error());
87 }
88 if (nonzero.value() == Tribool::True) {
89 return Tribool::False;
90 }
91 if (nonzero.value() == Tribool::False) {
92 return Tribool::True;
93 }
94 auto identity = check_equivalent(left, right, context);
95 if (!identity) {
96 return Result<Tribool>::failure(identity.error());
97 }
98 return std::holds_alternative<ProvedZeroResidual>(identity.value())
99 ? Tribool::True : Tribool::Unknown;
100}

116 }

118 std::optional<detail::AssumptionFacts> assumed;
119 if (context.assumptions()) {

135 } catch (const CasError& error) {
136 return Result<Tribool>::failure(error);

138 return Result<Tribool>::failure(
139 CasErrc::ResourceLimit,
