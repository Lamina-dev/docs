14static CancellationNode positive_denominator_exponent(const NumberNode& exponent) {
15 const auto& value = exponent.value();
16 if (std::holds_alternative<BigInt>(value)) {
17 const auto& integer = std::get<BigInt>(value);
18 if (!integer.is_negative()) { return nullptr; }
19 return detail::make_node<NumberNode>(BigInt(0) - integer);
20 }
21 if (std::holds_alternative<lmmc_real_t>(value)) {
22 const auto real = std::get<lmmc_real_t>(value);
23 if (!(real < 0)) { return nullptr; }
24 return detail::make_node<NumberNode>(-real);
25 }
26 const auto& rational = std::get<Rational>(value);
27 if (!rational.get_numerator().is_negative()) { return nullptr; }
28 return detail::make_node<NumberNode>(Rational(
29 BigInt(0) - rational.get_numerator(), rational.get_denominator()));
30}

32static void classify_fraction_factor(const CancellationNode& factor,
33 std::vector<CancellationNode>& numerator, std::vector<CancellationNode>& denominator) {
34 auto power = std::dynamic_pointer_cast<const PowerNode>(factor);
35 if (power) {
36 auto exponent = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
37 auto positive = exponent ? positive_denominator_exponent(*exponent) : nullptr;
38 if (positive) {
39 auto number = std::dynamic_pointer_cast<const NumberNode>(positive);
40 if (number && number->is_one()) {
41 denominator.push_back(power->base());
42 } else {
43 denominator.push_back(detail::make_node<PowerNode>(power->base(), positive));
44 }
45 return;
46 }
47 }
48 numerator.push_back(factor);
49}

73static bool collect_fraction_terms(const AddNode& addition,
74 std::vector<CancellationFraction>& terms) {
75 bool has_denominator = false;
76 for (const auto& term : addition.operands()) {
77 std::vector<CancellationNode> numerator, denominator;
78 separate_num_den(term, numerator, denominator);
79 auto num_expr = build_product(numerator)->simplify();
80 auto den_expr = build_product(denominator)->simplify();
81 if (!den_expr->is_one()) { has_denominator = true; }
82 terms.push_back({num_expr, den_expr});
83 }
84 return has_denominator;
85}

95static CancellationExpr distinct_denominator_product(
96 const std::vector<CancellationFraction>& terms) {
97 auto total_den = SymbolicExpr::number(1);
98 std::vector<CancellationExpr> unique_dens;
99 for (const auto& term : terms) {
100 if (term.denominator->is_one()) { continue; }
101 bool found = false;
102 for (const auto& denominator : unique_dens) {
103 if (denominator->to_string() == term.denominator->to_string()) {
104 found = true;
105 break;
106 }
107 }
108 if (!found) { unique_dens.push_back(term.denominator); }
109 }
110 for (const auto& denominator : unique_dens) {
111 total_den = SymbolicExpr::multiply(total_den, denominator)->simplify();
112 }
113 return total_den;
114}

116static CancellationFraction combine_fraction_terms(
117 const std::vector<CancellationFraction>& terms) {
118 const bool same = same_denominators(terms);
119 auto denominator = same ? terms[0].denominator : distinct_denominator_product(terms);
120 std::vector<CancellationNode> operands;
121 for (const auto& term : terms) {
122 auto numerator = term.numerator;
123 if (!same) {
124 const auto& total_node = detail::node(denominator);
125 const auto& term_node = detail::node(term.denominator);
126 auto factor = total_node && term_node &&
127 total_node->compare(*term_node) == 0
128 ? SymbolicExpr::number(1)
129 : SymbolicExpr::divide(
130 denominator, term.denominator)->simplify();
131 numerator = SymbolicExpr::multiply(
132 term.numerator, factor)->simplify();
133 }
134 if (numerator && detail::node(numerator)) {
135 operands.push_back(detail::node(numerator));
136 }
137 }
138 if (operands.empty()) { return {SymbolicExpr::number(0), nullptr}; }
139 auto numerator = operands.size() == 1 ? detail::make_expression_ptr(operands[0])
140 : detail::make_expression_ptr(detail::make_node<AddNode>(operands));
141 numerator = same ? numerator->simplify() : numerator->expand()->simplify();
142 return {numerator, denominator};
143}

146template <typename Coefficient>
147static void cancel_polynomial_gcd(const Polynomial<Coefficient>& numerator,
148 const Polynomial<Coefficient>& denominator, CancellationExpr& cur_num,
149 CancellationExpr& cur_den) {
150 if (denominator.is_zero()) return;
151 auto gcd = Polynomial<Coefficient>::gcd(numerator, denominator);
152 if (gcd.degree() >= 1) {
153 auto [q_num, r_num] = numerator.div_mod(gcd);
154 auto [q_den, r_den] = denominator.div_mod(gcd);
155 if (!r_num.is_zero() || !r_den.is_zero()) return;
156 cur_num = poly_to_symbolic(q_num)->simplify();
157 cur_den = poly_to_symbolic(q_den)->simplify();
158 }
159}

161static CancellationExpr cancel_variable(CancellationExpr& cur_num,
162 CancellationExpr& cur_den, const std::string& var, const CancellationExpr& simp) {
163 auto num_expanded = cur_num->expand()->simplify();
164 auto den_expanded = cur_den->expand()->simplify();
165 auto rational_num = symbolic_to_poly<Rational>(num_expanded, var);
166 auto rational_den = symbolic_to_poly<Rational>(den_expanded, var);
167 if (rational_num && rational_den) {
168 if (rational_den.value().is_zero()) { return simp; }
169 if (rational_num.value().is_zero()) { return SymbolicExpr::number(0); }
170 cancel_polynomial_gcd(rational_num.value(), rational_den.value(), cur_num, cur_den);
171 return nullptr;
172 }
173 auto symbolic_num = symbolic_to_poly<SymbolicPolyCoeff>(num_expanded, var);
174 auto symbolic_den = symbolic_to_poly<SymbolicPolyCoeff>(den_expanded, var);
175 if (!symbolic_num || !symbolic_den) { return nullptr; }
176 if (symbolic_den.value().is_zero()) { return simp; }
177 if (symbolic_num.value().is_zero()) { return SymbolicExpr::number(0); }
178 cancel_polynomial_gcd(symbolic_num.value(), symbolic_den.value(), cur_num, cur_den);
179 return nullptr;
180}

182static CancellationExpr finish_cancellation(const CancellationExpr& cur_num,
183 const CancellationExpr& cur_den) {
184 if (cur_den->is_one()) { return cur_num; }
185 if (detail::node(cur_den)) {
186 auto diff = SymbolicExpr::add(detail::make_expression_ptr(detail::node(cur_den)),
187 SymbolicExpr::number(-1))->simplify();
188 if (diff->is_zero()) {
189 return SymbolicExpr::multiply(SymbolicExpr::number(-1), cur_num)->simplify();
190 }
191 }
192 return SymbolicExpr::divide(cur_num, cur_den)->simplify();
193}

195static CancellationExpr reduce_fraction(CancellationFraction fraction,
196 const CancellationExpr& simp, bool from_sum) {
197 auto rational_variables = free_variables(detail::node(fraction.numerator));
198 const auto denominator_variables = free_variables(detail::node(fraction.denominator));
199 rational_variables.insert(denominator_variables.begin(), denominator_variables.end());
200 if (!from_sum && rational_variables.empty()) {
201 return SymbolicExpr::divide(fraction.numerator, fraction.denominator)->simplify();
202 }
203 for (const auto& var : rational_variables) {
204 try {
205 auto terminal = cancel_variable(
206 fraction.numerator, fraction.denominator, var, simp);
207 if (terminal) { return terminal; }
208 } catch (const std::invalid_argument&) {
209 continue;
210 } catch (const std::out_of_range&) {
211 continue;
212 } catch (const std::runtime_error&) {
213 continue;
214 }
215 }
216 return finish_cancellation(fraction.numerator, fraction.denominator);
217}

221std::shared_ptr<SymbolicExpr> SymbolicExpr::cancel() const {
222 auto simp = simplify();
223 if (!simp || !detail::node(simp)) { return simp; }
224 if (auto addition = std::dynamic_pointer_cast<const AddNode>(detail::node(simp))) {
225 std::vector<CancellationFraction> terms;
226 if (collect_fraction_terms(*addition, terms)) {
227 auto combined = combine_fraction_terms(terms);
228 if (!combined.denominator) { return combined.numerator; }
229 combined.denominator = combined.denominator->expand()->simplify();
230 return reduce_fraction(std::move(combined), simp, true);
231 }
232 }
233 std::vector<CancellationNode> num_factors;
234 std::vector<CancellationNode> den_factors;
235 separate_num_den(detail::node(simp), num_factors, den_factors);
236 if (den_factors.empty()) { return simp; }
237 auto numerator = build_product(num_factors)->simplify();
238 auto denominator = build_product(den_factors)->simplify();
239 if (denominator->is_one()) { return numerator; }
240 return reduce_fraction({numerator, denominator}, simp, false);
241}
