136Result<Rational> UnitSystem::conversion_factor(
137 const std::string& source, const std::string& target) const {
138 auto source_unit = resolve(source);
139 if (!source_unit) return Result<Rational>::failure(source_unit.error());
140 auto target_unit = resolve(target);
141 if (!target_unit) return Result<Rational>::failure(target_unit.error());
142 if (source_unit.value().dimension != target_unit.value().dimension) {
143 return Result<Rational>::failure(
144 CasErrc::DimensionMismatch,
145 "unit conversion requires equal dimensions", kUnitOperation);
146 }
147 return Result<Rational>::success(
148 source_unit.value().scale_to_base / target_unit.value().scale_to_base);
149}
