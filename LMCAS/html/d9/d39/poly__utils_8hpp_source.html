75/** @brief Symbolic expression coefficient for symbolic polynomial algorithms. */
76struct SymbolicPolyCoeff {
77 std::shared_ptr<SymbolicExpr> val;
78
79 SymbolicPolyCoeff() : val(SymbolicExpr::number(0)) {}
80 explicit SymbolicPolyCoeff(int value) : val(SymbolicExpr::number(value)) {}
81 SymbolicPolyCoeff(std::shared_ptr<SymbolicExpr> value) : val(std::move(value)) {}
82
83 bool operator==(const SymbolicPolyCoeff& other) const {
84 if (!val || !other.val) return false;
85 if (val == other.val) return true;
86 auto difference = SymbolicExpr::add(
87 val,
88 SymbolicExpr::multiply(other.val, SymbolicExpr::number(-1)));
89 return difference->simplify()->is_zero();
90 }
91
92 bool operator!=(const SymbolicPolyCoeff& other) const {
93 return !(*this == other);
94 }
95
96 SymbolicPolyCoeff operator+(const SymbolicPolyCoeff& other) const {
97 return SymbolicPolyCoeff(SymbolicExpr::add(val, other.val));
98 }
99
100 SymbolicPolyCoeff operator-(const SymbolicPolyCoeff& other) const {
101 return SymbolicPolyCoeff(SymbolicExpr::add(
102 val,
103 SymbolicExpr::multiply(other.val, SymbolicExpr::number(-1))));
104 }
105
106 SymbolicPolyCoeff operator*(const SymbolicPolyCoeff& other) const {
107 return SymbolicPolyCoeff(SymbolicExpr::multiply(val, other.val));
108 }
109
110 SymbolicPolyCoeff operator/(const SymbolicPolyCoeff& other) const {
111 return SymbolicPolyCoeff(SymbolicExpr::divide(val, other.val));
112 }
113
114 SymbolicPolyCoeff operator-() const {
115 return SymbolicPolyCoeff(
116 SymbolicExpr::multiply(val, SymbolicExpr::number(-1)));
117 }
118
119 std::string ToString() const {
120 return val ? val->to_string() : "0";
121 }
122
123 friend SymbolicPolyCoeff abs(const SymbolicPolyCoeff& value) {
124 return value;
125 }
126
127 bool operator<(const SymbolicPolyCoeff&) const {
128 return false;
129 }
130};
