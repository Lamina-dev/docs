16static int tf_power_degree(const PowerNode& power, const std::string& var) {
17 auto exp_num = std::dynamic_pointer_cast<const NumberNode>(power.exponent());
18 if (!exp_num) return -1;
19
20 BigInt exponent(-1);
21 if (std::holds_alternative<BigInt>(exp_num->value())) {
22 exponent = std::get<BigInt>(exp_num->value());
23 } else if (std::holds_alternative<Rational>(exp_num->value())) {
24 const auto& r = std::get<Rational>(exp_num->value());
25 if (r.is_integer()) exponent = r.to_bigint();
26 } else if (std::holds_alternative<lmmc_real_t>(exp_num->value())) {
27 lmmc_real_t d = std::get<lmmc_real_t>(exp_num->value());
28 if (std::isfinite(d) && d >= 0 && d == std::floor(d) && d < 1000.0) {
29 exponent = Rational::from_double(d).to_bigint();
30 }
31 }
32
33 if (exponent.is_negative() || exponent > BigInt(std::numeric_limits<int>::max())) {
34 return -1;
35 }
36 const int e_val = static_cast<int>(exponent.try_to_int64().value());
37
38 int base_deg = tf_degree_in(power.base(), var);
39 if (base_deg < 0 ||
40 (e_val != 0 && base_deg > std::numeric_limits<int>::max() / e_val)) return -1;
41 return base_deg * e_val;
42}

44int tf_degree_in(const std::shared_ptr<const SymbolicNode>& node, const std::string& var) {
45 if (!node) return 0;
46
47 if (!expression_depends_on_variable(node, var)) return 0;
48
49 if (auto v = std::dynamic_pointer_cast<const VariableNode>(node)) {
50 return (!v->is_constant() && v->name() == var) ? 1 : 0;
51 }
52
53 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
54 int max_deg = 0;
55 for (const auto& op : add->operands()) {
56 int d = tf_degree_in(op, var);
57 if (d < 0) return -1;
58 max_deg = std::max(max_deg, d);
59 }
60 return max_deg;
61 }
62
63 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
64 int total_deg = 0;
65 for (const auto& op : mul->operands()) {
66 int d = tf_degree_in(op, var);
67 if (d < 0) return -1;
68 if (d > std::numeric_limits<int>::max() - total_deg) return -1;
69 total_deg += d;
70 }
71 return total_deg;
72 }
73
74 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
75 return tf_power_degree(*power, var);
76 }
77 if (std::dynamic_pointer_cast<const FunctionNode>(node)) {
78 return -1;
79 }
80
81 return 0;
82}

92static bool tf_validate_polynomial(
93 const std::shared_ptr<SymbolicExpr>& poly_expr,
94 const std::vector<std::string>& all_variables) {
95
96 if (!poly_expr || !LMCAS::detail::node(poly_expr)) return false;
97
98 for (const auto& var : all_variables) {
99 if (expression_depends_on_variable(LMCAS::detail::node(poly_expr), var)) {

101 if (deg < 0) return false;
102 }
103 }

137static Result<TfPolyBuildResult> tf_convert_main_variable(
138 TfPolyBuildResult result, const std::shared_ptr<SymbolicExpr>& expanded,
139 const std::vector<std::string>& all_variables) {
140 const auto main_var = result.main_variable;
141 auto converted = symbolic_to_poly<Rational>(expanded, main_var);
142 if (converted) {
143 result.poly = std::move(converted.value());
144 result.success = true;
145 return result;
146 }
147 if (converted.error().code != CasErrc::UnsupportedExpression) {
148 return Result<TfPolyBuildResult>::failure(converted.error());
149 }
150 for (const auto& var : all_variables) {
151 if (var == main_var) continue;
152 auto trial = symbolic_to_poly<Rational>(expanded, var);
153 if (!trial) {
154 if (trial.error().code == CasErrc::UnsupportedExpression) continue;
155 return Result<TfPolyBuildResult>::failure(trial.error());
156 }
157 result.main_variable = var;
158 result.poly = std::move(trial.value());
159 result.param_variables.clear();
160 for (const auto& other : all_variables) {
161 if (other != var) result.param_variables.push_back(other);
162 }
163 result.success = true;
164 return result;
165 }
166 result.success = false;
167 return result;
168}

171 const std::shared_ptr<SymbolicExpr>& poly_expr,
172 const std::vector<std::string>& indeterminates,
173 const std::string& original_var) try {
174
175 TfPolyBuildResult result;
176 result.success = false;
177
178 if (!poly_expr || !LMCAS::detail::node(poly_expr) || original_var.empty()) {
179 return Result<TfPolyBuildResult>::failure(CasErrc::InvalidArgument,
180 "expression and variable must be nonempty", "tf_build_polynomial");
181 }
182
183 auto before = tf_polynomial_variables(poly_expr, indeterminates, original_var);
184 if (!before.empty() && !tf_validate_polynomial(poly_expr, before)) return result;
185 auto expanded = poly_expr->expand();

187 expanded = poly_expr;
188 }
189 auto all_variables = tf_polynomial_variables(expanded, indeterminates, original_var);
190 if (all_variables.empty()) {
191 auto c = extract_coeff_value<Rational>(expanded);
192 if (!c) {
193 if (c.error().code == CasErrc::UnsupportedExpression) return result;
194 return Result<TfPolyBuildResult>::failure(c.error());
195 }
196 result.poly = Polynomial<Rational>({std::move(c.value())}, "x");
197 result.main_variable = "x";
198 result.success = true;
199 return result;
200 }
201
202 if (!tf_validate_polynomial(expanded, all_variables)) {
203 return result;
204 }
205 int max_degree = -1;
206 tf_select_main_variable(result, expanded, all_variables, max_degree);
207 if (result.main_variable.empty() || max_degree < 0) return result;
208 return tf_convert_main_variable(std::move(result), expanded, all_variables);
209}
210catch (const std::bad_alloc&) {
211 return Result<TfPolyBuildResult>::failure(CasErrc::ResourceLimit,
212 "allocation failed while constructing polynomial", "tf_build_polynomial");
