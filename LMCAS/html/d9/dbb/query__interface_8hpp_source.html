50/**
51 * @brief Cache key combining an expression's structural hash with the property type.
52 *
53 * Uses the expression structural hash - two structurally identical
54 * expressions will produce the same hash regardless of pointer identity.
55 */
56struct CacheKey {
57 std::size_t expression_hash;
58 PropType property;
59
60 bool operator==(const CacheKey& other) const {
61 return expression_hash == other.expression_hash && property == other.property;
62 }
63};
64
65/**
66 * @brief Hash function for CacheKey, combining expression hash and property type.
67 */
68struct CacheKeyHash {
69 std::size_t operator()(const CacheKey& key) const {
70 std::size_t seed = key.expression_hash;
71 seed ^= static_cast<std::size_t>(key.property) + 0x9e3779b9U
72 + (seed << 6U) + (seed >> 2U);
73 return seed;
74 }
75};

77/**
78 * @brief Unified query API for property questions on arbitrary SymbolicExpr trees.
79 *
80 * Provides query_positive, query_negative, query_nonnegative, query_real,
81 * query_integer, and query_nonzero methods that return Tribool results.
82 *
83 * 结果先按(表达式哈希,属性类型)分桶,再以结构相等性确认表达式身份.
84 * AssumptionContext 发生 push,pop 或 assume 变更时,缓存世代触发整体失效.
85 *
86 * Dispatch logic:
87 * - Null root node -> Unknown
88 * - MatrixNode, RelationalNode, LogicalNode -> Unknown
89 * - NumberNode with NaN -> False for integer, Unknown for sign queries
90 * - FunctionNode::Infinity -> derive sign from infinity's sign, False for integer
91 * - All other nodes -> delegate to InferenceEngine
92 */
