41ExpressionResult normalize(const Node& value, ComputationContext& context) {
42 auto step = context.consume_steps(1, kOperation);
43 if (!step) {
44 return ExpressionResult::failure(step.error());
45 }
46 RewriteBudget budget(context, context.limits().max_recursion_depth,
47 context.limits().max_ast_nodes, kOperation);
48 budget.measure(value);
49 std::optional<AssumptionFacts> facts;
50 if (context.assumptions()) facts.emplace(*context.assumptions());
51 NormalizationVisitor visitor(context, facts ? static_cast<const FactsQuery&>(*facts)
52 : no_facts(), Domain::Real, &budget);
53 value->accept(visitor);
54 auto result = visitor.get_result();
55 if (!result) {
56 return ExpressionResult::failure(CasErrc::InternalInvariant,
57 "quadratic normalization returned null", kOperation);
58 }
59 budget.measure(result);
60 return make_expression_ptr(std::move(result));
61}

63Result<Coefficients> normalize_coefficients(
64 const Coefficients& input, ComputationContext& context) {
65 for (const auto& coefficient : input) {
66 auto step = context.consume_steps(1, kOperation);
67 if (!step) {
68 return Result<Coefficients>::failure(step.error());
69 }
70 if (!coefficient || !node(coefficient)) {
71 return Result<Coefficients>::failure(CasErrc::InvalidArgument,
72 "quadratic coefficients must not be null", kOperation);
73 }
74 }
75 Coefficients result;
76 for (std::size_t i = 0; i < input.size(); ++i) {
77 auto simplified = normalize(node(input[i]), context);
78 if (!simplified) {
79 return Result<Coefficients>::failure(simplified.error());
80 }
81 result[i] = std::move(simplified.value());
82 }
83 return result;

86Result<void> prove_coefficients(const Coefficients& coefficients,
87 const AssumptionContext& facts,
88 ComputationContext& context) {
89 auto step = context.consume_steps(4, kOperation);
90 if (!step) {
91 return step;
92 }
93 if (node(coefficients[0])->is_zero()) {
94 return Result<void>::failure(CasErrc::DomainError,
95 "quadratic leading coefficient is zero", kOperation);
96 }
97 AssumptionFacts adapter(facts);
98 for (const auto& coefficient : coefficients) {
99 auto real = query_real_value(node(coefficient), adapter, context);
100 if (!real) {
101 return Result<void>::failure(real.error());
102 }
103 if (real.value() != Tribool::True) {
104 return Result<void>::failure(CasErrc::UnsupportedExpression,
105 "quadratic coefficients must be provably real", kOperation);
106 }
107 }
108 auto nonzero = facts.is_nonzero_checked(*coefficients[0], context);
109 if (!nonzero) {
110 return Result<void>::failure(nonzero.error());
111 }
112 if (nonzero.value() == Tribool::False) {
113 return Result<void>::failure(CasErrc::DomainError,
114 "quadratic leading coefficient is zero", kOperation);
115 }
116 if (nonzero.value() != Tribool::True) {
117 return Result<void>::failure(CasErrc::Inconclusive,
118 "quadratic leading coefficient is not provably nonzero", kOperation);
119 }
120 return Result<void>::success();
121}

123Result<bool> negative_discriminant(const SymbolicExpr& delta,
124 const AssumptionContext& facts,
125 ComputationContext& context) {
126 auto step = context.consume_steps(2, kOperation);
127 if (!step) {
128 return Result<bool>::failure(step.error());
129 }
130 auto nonnegative = facts.is_nonnegative_checked(delta, context);
131 if (!nonnegative) {
132 return Result<bool>::failure(nonnegative.error());
133 }
134 if (nonnegative.value() == Tribool::True) {
135 return false;
136 }
137 auto negative = facts.is_negative_checked(delta, context);
138 if (!negative) {
139 return Result<bool>::failure(negative.error());
140 }
141 if (negative.value() == Tribool::True) {
142 return true;
143 }
144 return Result<bool>::failure(CasErrc::Inconclusive,
145 "quadratic discriminant sign is unknown", kOperation);
146}

161Result<CartesianQuadraticRoots> construct_roots(
162 const Coefficients& coefficients, const Node& delta, bool negative,
163 ComputationContext& context) {
164 const auto denominator = multiply(integer(2), node(coefficients[0]));
165 const auto minus_b = negate(node(coefficients[1]));
166 const auto radical = SymbolicFactory::create_power(
167 negative ? negate(delta) : delta, SymbolicFactory::create_number(Rational(1, 2)));
168 const auto real = divide(minus_b, denominator);
169 const auto imag = divide(radical, denominator);
170 auto first = normalized_root(
171 negative ? real : add(real, imag),
172 negative ? imag : integer(0), context);
173 if (!first) {
174 return Result<CartesianQuadraticRoots>::failure(first.error());
175 }
176 auto second = normalized_root(
177 negative ? real : add(real, negate(imag)),
178 negative ? negate(imag) : integer(0), context);
179 if (!second) {
180 return Result<CartesianQuadraticRoots>::failure(second.error());
181 }
182 return CartesianQuadraticRoots{std::move(first.value()), std::move(second.value())};
183}

199Result<void> prove_zero(const Node& residual, ComputationContext& context) {
200 auto simplified = normalize(residual, context);
201 if (!simplified) {
202 return Result<void>::failure(simplified.error());
203 }
204 auto proof = check_zero_residual(simplified.value(), context);
205 if (!proof) {
206 return Result<void>::failure(proof.error());
207 }
208 if (!std::holds_alternative<ProvedZeroResidual>(proof.value())) {
209 return Result<void>::failure(CasErrc::Inconclusive,
210 "quadratic Cartesian residual could not be proved zero", kOperation);
211 }
212 return Result<void>::success();

215Result<void> verify_root(const Coefficients& coefficients,
216 const QuadraticCartesianRoot& root,
217 ComputationContext& context) {
218 auto step = context.consume_steps(16, kOperation);
219 if (!step) {
220 return step;
221 }
222 const auto zero = integer(0);
223 const CartesianNodes a{node(coefficients[0]), zero};
224 const CartesianNodes b{node(coefficients[1]), zero};
225 const CartesianNodes c{node(coefficients[2]), zero};
226 const CartesianNodes z{node(root.real), node(root.imag)};
227 const auto residual = cartesian_add(
228 cartesian_multiply(cartesian_add(cartesian_multiply(a, z), b), z), c);
229 auto real = prove_zero(residual.real, context);
230 if (!real) {
231 return real;
232 }
233 return prove_zero(residual.imag, context);
234}

236Result<CartesianQuadraticRoots> solve_real_quadratic(
237 const Coefficients& coefficients, const AssumptionContext& facts,
238 ComputationContext& context) {
239 auto valid = prove_coefficients(coefficients, facts, context);
240 if (!valid) {
241 return Result<CartesianQuadraticRoots>::failure(valid.error());
242 }
243 const auto b = node(coefficients[1]);
244 auto delta = normalize(add(multiply(b, b), negate(multiply(integer(4),
245 multiply(node(coefficients[0]), node(coefficients[2]))))), context);
246 if (!delta) {
247 return Result<CartesianQuadraticRoots>::failure(delta.error());
248 }
249 auto negative = negative_discriminant(*delta.value(), facts, context);
250 if (!negative) {
251 return Result<CartesianQuadraticRoots>::failure(negative.error());
252 }
253 auto roots = construct_roots(coefficients, node(delta.value()), negative.value(), context);
254 if (!roots) {
255 return roots;
256 }
257 for (const auto& root : roots.value()) {
258 auto verified = verify_root(coefficients, root, context);
259 if (!verified) {
260 return Result<CartesianQuadraticRoots>::failure(verified.error());
261 }
262 }
263 return roots;
264}
