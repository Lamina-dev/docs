47int infinity_sign(const Node& node) {
48 if (const auto* f = dynamic_cast<const FunctionNode*>(node.get())) {
49 if (f->type() == FunctionNode::FuncType::Infinity) return 1;
50 }
51 if (const auto* m = dynamic_cast<const MultiplyNode*>(node.get())) {
52 int result = 1;
53 bool infinity = false;
54 for (const auto& factor : m->operands()) {
55 if (infinity_sign(factor)) {
56 if (infinity) return 0;
57 infinity = true;
58 result *= infinity_sign(factor);
59 } else {
60 auto value = exact_number(factor, true);
61 if (!value || sign(*value) == 0) return 0;
62 result *= sign(*value);
63 }
64 }
65 return infinity ? result : 0;
66 }
67 return 0;

88Result<int> compare_boundaries(const Boundary& a, const Boundary& b,
89 ComputationContext& context) {
90 if (a.infinity || b.infinity)
91 { return a.infinity < b.infinity ? -1 : a.infinity > b.infinity ? 1 : 0; }
92 auto left = a.finite;
93 auto right = b.finite;
94 right.polynomial.variable_name = left.polynomial.variable_name;
95 if (!(left.upper < right.lower) && !(right.upper < left.lower)) {
96 auto equal = detail::equal_exact_real_algebraic(left, right, context);
97 if (!equal) { return Result<int>::failure(equal.error()); }
98 if (equal.value()) { return 0; }
99 while (!(left.upper < right.lower) && !(right.upper < left.lower)) {
100 auto refined = detail::refine_exact_real_algebraic(left, context, operation);
101 if (!refined) { return Result<int>::failure(refined.error()); }
102 refined = detail::refine_exact_real_algebraic(right, context, operation);
103 if (!refined) { return Result<int>::failure(refined.error()); }
104 }

121Result<Poly> product_checked(const Poly& a, const Poly& b, ComputationContext& context) {
122 const auto size = a.coeffs.size() + b.coeffs.size();
123 auto budget = context.require_expansion_terms(size, operation);
124 if (!budget) return Result<Poly>::failure(budget.error());
125 if (size > static_cast<std::size_t>(std::numeric_limits<int>::max()))
126 return Result<Poly>::failure(CasErrc::ResourceLimit,
127 "polynomial degree exceeds representation", operation);
128 auto step = context.consume_steps(a.coeffs.size() * b.coeffs.size() + 1, operation);
129 if (!step) return Result<Poly>::failure(step.error());
130 auto value = a * b;
131 budget = polynomial_budget(value, context);
132 if (!budget) return Result<Poly>::failure(budget.error());
133 return value;

136static Result<Poly> gcd_checked(Poly a, Poly b, ComputationContext& context) {
137 while (!b.is_zero()) {
138 auto step = context.consume_steps(a.coeffs.size() * b.coeffs.size() + 1, operation);
139 if (!step) return Result<Poly>::failure(step.error());
140 auto remainder = a.div_mod(b).second;
141 auto checked = polynomial_budget(remainder, context);
142 if (!checked) return Result<Poly>::failure(checked.error());
143 a = std::move(b);
144 b = std::move(remainder);
145 }
146 return a.make_monic();

149static RationalResult rational_operands_checked(const std::vector<Node>& operands,
150 bool sum, const std::string& var, ComputationContext& context) {
151 RationalFunction result{Poly(Rational(sum ? 0 : 1), var), Poly(Rational(1), var)};
152 for (const auto& operand : operands) {
153 auto child = rational_checked(operand, var, context);
154 if (!child || !child.value()) { return child; }
155 auto n = product_checked(result.numerator,
156 sum ? child.value()->denominator : child.value()->numerator, context);
157 if (!n) { return RationalResult::failure(n.error()); }
158 if (sum) {
159 auto other = product_checked(child.value()->numerator, result.denominator, context);
160 if (!other) { return RationalResult::failure(other.error()); }
161 n.value() = n.value() + other.value();
162 }
163 auto d = product_checked(result.denominator, child.value()->denominator, context);
164 if (!d) { return RationalResult::failure(d.error()); }
165 result = {std::move(n.value()), std::move(d.value())};
166 }
167 return std::optional<RationalFunction>{std::move(result)};

170static RationalResult rational_power_checked(const PowerNode& power,
171 const std::string& var, ComputationContext& context) {
172 auto exponent = exact_number(power.exponent());
173 if (!exponent || !exponent->is_integer()) { return std::optional<RationalFunction>{}; }
174 auto base = rational_checked(power.base(), var, context);
175 if (!base || !base.value()) { return base; }
176 BigInt count = exponent->to_bigint();
177 if (count < BigInt(0)) {
178 std::swap(base.value()->numerator, base.value()->denominator);
179 count = -count;
180 }
181 RationalFunction result{Poly(Rational(1), var), Poly(Rational(1), var)};
182 while (count != BigInt(0)) {
183 if (count % BigInt(2) != BigInt(0)) {
184 auto n = product_checked(result.numerator, base.value()->numerator, context);
185 auto d = product_checked(result.denominator, base.value()->denominator, context);
186 if (!n) { return RationalResult::failure(n.error()); }
187 if (!d) { return RationalResult::failure(d.error()); }
188 result = {std::move(n.value()), std::move(d.value())};
189 }
190 count = count / BigInt(2);
191 if (count == BigInt(0)) { break; }
192 auto n = product_checked(base.value()->numerator, base.value()->numerator, context);
193 auto d = product_checked(base.value()->denominator, base.value()->denominator, context);
194 if (!n) { return RationalResult::failure(n.error()); }
195 if (!d) { return RationalResult::failure(d.error()); }
196 base.value() = RationalFunction{std::move(n.value()), std::move(d.value())};
197 }
198 return std::optional<RationalFunction>{std::move(result)};

221static Result<void> recover_rational_cut(Algebraic& root, const BigInt& denominator_bound,
222 ComputationContext& context) {
223 if (root.is_rational()) { return Result<void>::success(); }
224 auto bits = context.require_integer_bits(2 * denominator_bound.bit_length() + 2, operation);
225 if (!bits) { return bits; }
226 const Rational separation(BigInt(1), BigInt(2) * denominator_bound * denominator_bound);
227 while (root.upper - root.lower >= separation) {
228 auto refined = detail::refine_exact_real_algebraic(root, context, operation);
229 if (!refined) { return refined; }
230 if (root.is_rational()) { return Result<void>::success(); }
231 }
232 /**
233 * @brief 有理根定理给出分母 q 的上界；当前包围宽度保证根 p/q 位于中点连分数的渐近分数中。
234 * 通过连分数恢复有理分割点，免去整数因子枚举。
235 */
236 const Rational midpoint = (root.lower + root.upper) / Rational(2);
237 BigInt numerator = midpoint.get_numerator(), denominator = midpoint.get_denominator();
238 BigInt old_p(0), p(1), old_q(1), q(0);
239 while (denominator != BigInt(0)) {
240 auto step = context.consume_steps(1, operation);
241 if (!step) { return step; }
242 BigInt quotient = numerator / denominator;
243 BigInt remainder = numerator % denominator;
244 if (remainder < BigInt(0)) { quotient = quotient - BigInt(1); remainder = remainder + denominator; }
245 BigInt next_p = quotient * p + old_p;
246 BigInt next_q = quotient * q + old_q;
247 if (next_q > denominator_bound) { break; }
248 Rational candidate(next_p, next_q);
249 if (candidate >= root.lower && candidate <= root.upper &&
250 root.polynomial.eval(candidate) == Rational(0)) {
251 root.lower = root.upper = candidate;
252 break;
253 }
254 old_p = std::move(p); p = std::move(next_p);
255 old_q = std::move(q); q = std::move(next_q);
256 numerator = std::move(denominator); denominator = std::move(remainder);
257 }
258 return Result<void>::success();
259}

261Result<std::vector<Algebraic>> roots_checked(const Poly& polynomial, ComputationContext& context) {
262 if (polynomial.degree() <= 0) return std::vector<Algebraic>{};
263 auto repeated = gcd_checked(polynomial, polynomial.differentiate(), context);
264 if (!repeated) return Result<std::vector<Algebraic>>::failure(repeated.error());
265 auto square_free = polynomial.div_mod(repeated.value()).first.make_monic();
266 BigInt denominator_bound(1);
267 for (const auto& coefficient : square_free.coeffs) {
268 const auto& denominator = coefficient.get_denominator();
269 denominator_bound = denominator_bound / BigInt::gcd(denominator_bound, denominator) * denominator;
270 auto checked = context.require_integer_bits(denominator_bound.bit_length(), operation);
271 if (!checked) return Result<std::vector<Algebraic>>::failure(checked.error());
272 }
273 auto isolated = detail::isolate_real_roots_exact(square_free, context, operation);
274 if (!isolated) return Result<std::vector<Algebraic>>::failure(isolated.error());
275 std::vector<Algebraic> roots;
276 for (std::size_t i = 0; i < isolated.value().size(); ++i) {
277 Algebraic root{square_free, isolated.value()[i].first, isolated.value()[i].second, i, 1};
278 auto recovered = recover_rational_cut(root, denominator_bound, context);
279 if (!recovered) return Result<std::vector<Algebraic>>::failure(recovered.error());
280 roots.push_back(std::move(root));
281 }
282 return roots;
283}

285Result<std::optional<int>> polynomial_sign_checked(const Poly& polynomial,
286 const Boundary& a, const Boundary& b, ComputationContext& context) {
287 if (polynomial.is_zero()) return std::optional<int>{0};
288 if (polynomial.degree() == 0) return std::optional<int>{sign(polynomial.lead_coeff())};
289 auto roots = roots_checked(polynomial, context);
290 if (!roots) return Result<std::optional<int>>::failure(roots.error());
291 for (const auto& root : roots.value()) {
292 auto left = compare_boundaries(a, Boundary{0, root}, context);
293 auto right = compare_boundaries(Boundary{0, root}, b, context);
294 if (!left) return Result<std::optional<int>>::failure(left.error());
295 if (!right) return Result<std::optional<int>>::failure(right.error());
296 if (left.value() < 0 && right.value() < 0) return std::optional<int>{};
297 }
298 if (a.infinity < 0)
299 return std::optional<int>{sign(polynomial.lead_coeff()) *
300 (polynomial.degree() % 2 ? -1 : 1)};
301 if (b.infinity > 0) return std::optional<int>{sign(polynomial.lead_coeff())};
302 auto left = a.finite;
303 auto right = b.finite;
304 while (!(left.upper < right.lower)) {
305 auto l = detail::refine_exact_real_algebraic(left, context, operation);
306 if (!l) return Result<std::optional<int>>::failure(l.error());
307 auto r = detail::refine_exact_real_algebraic(right, context, operation);
308 if (!r) return Result<std::optional<int>>::failure(r.error());
309 }
310 return std::optional<int>{sign(polynomial.eval((left.upper + right.lower) / Rational(2)))};
311}

313static Result<int> polynomial_at_root_sign(Poly polynomial, Algebraic root,
314 ComputationContext& context) {
315 if (polynomial.is_zero()) { return 0; }
316 if (root.is_rational()) { return sign(polynomial.eval(root.lower)); }
317 auto common = gcd_checked(polynomial, root.polynomial, context);
318 if (!common) { return Result<int>::failure(common.error()); }
319 if (common.value().degree() > 0) {
320 auto zeros = detail::count_real_roots_exact(common.value(), root.lower, root.upper, context, operation);
321 if (!zeros) { return Result<int>::failure(zeros.error()); }
322 if (zeros.value()) { return 0; }
323 }
324 for (;;) {
325 auto zeros = detail::count_real_roots_exact(polynomial, root.lower, root.upper, context, operation);
326 if (!zeros) { return Result<int>::failure(zeros.error()); }
327 if (!zeros.value()) { return sign(polynomial.eval(root.lower)); }

329 if (!refined) { return Result<int>::failure(refined.error()); }

339static Result<void> rational_pole_divergence(const RationalFunction& rational,
340 const Algebraic& root, const Boundary& a, const Boundary& b,
341 int& divergence, ComputationContext& context) {
342 auto left = compare_boundaries(a, Boundary{0, root}, context);
343 auto right = compare_boundaries(Boundary{0, root}, b, context);
344 if (!left) { return Result<void>::failure(left.error()); }
345 if (!right) { return Result<void>::failure(right.error()); }
346 if (left.value() > 0 || right.value() > 0) { return Result<void>::success(); }
347 Poly derivative = rational.denominator;
348 std::size_t multiplicity = 0;
349 int denominator_sign = 0;
350 do {
351 derivative = derivative.differentiate();
352 ++multiplicity;
353 auto value = polynomial_at_root_sign(derivative, root, context);
354 if (!value) { return Result<void>::failure(value.error()); }
355 denominator_sign = value.value();
356 } while (!denominator_sign);
357 auto numerator_sign = polynomial_at_root_sign(rational.numerator, root, context);
358 if (!numerator_sign) { return Result<void>::failure(numerator_sign.error()); }
359 int coefficient_sign = numerator_sign.value() * denominator_sign;
360 if (!coefficient_sign) { return Result<void>::failure(CasErrc::InternalInvariant,
361 "reduced rational pole has a zero numerator", operation); }
362 if (left.value() < 0 && !include_divergence(divergence, coefficient_sign * (multiplicity % 2 ? -1 : 1)))
363 { return divergent<void>(); }
364 if (right.value() < 0 && !include_divergence(divergence, coefficient_sign)) { return divergent<void>(); }
365 return Result<void>::success();
366}

368Result<int> rational_divergence_checked(RationalFunction& rational,
369 const Boundary& a, const Boundary& b, ComputationContext& context) {
370 if (rational.denominator.is_zero()) { return divergent<int>(); }
371 auto gcd = gcd_checked(rational.numerator, rational.denominator, context);
372 if (!gcd) { return Result<int>::failure(gcd.error()); }
373 rational.numerator = rational.numerator.div_mod(gcd.value()).first;
374 rational.denominator = rational.denominator.div_mod(gcd.value()).first;
375 if (rational.numerator.is_zero()) { return 0; }
376 auto roots = roots_checked(rational.denominator, context);
377 if (!roots) { return Result<int>::failure(roots.error()); }
378 int divergence = 0;
379 for (const auto& root : roots.value()) {
380 auto pole = rational_pole_divergence(rational, root, a, b, divergence, context);
381 if (!pole) { return Result<int>::failure(pole.error()); }
382 }
383 const int degree = rational.numerator.degree() - rational.denominator.degree();
384 if (degree >= -1) {
385 int coefficient = sign(rational.numerator.lead_coeff()) * sign(rational.denominator.lead_coeff());
386 if (a.infinity < 0 && !include_divergence(divergence, coefficient * (degree % 2 ? -1 : 1))) { return divergent<int>(); }
387 if (b.infinity > 0 && !include_divergence(divergence, coefficient)) { return divergent<int>(); }
388 }
389 return divergence;
390}
