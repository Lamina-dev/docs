48Polynomial<CoeffType>::pseudo_div_mod(const Polynomial<CoeffType>& other) const {
49 if (other.is_zero()) throw std::runtime_error("Division by zero polynomial");

51 if (is_zero()) {
52 return {Polynomial(other.variable_name), Polynomial(other.variable_name)};
53 }
54
55 int deg_rem = degree();
56 int deg_div = other.degree();
57
58 if (deg_rem < deg_div) {
59 return {Polynomial(variable_name), *this};
60 }
61
62 Polynomial remainder = *this;
63 Polynomial quotient(variable_name);
64 CoeffType lc_div = other.lead_coeff();
65
66 int delta = deg_rem - deg_div;
67 quotient.coeffs.resize(delta + 1, CoeffType(0));

70 int current_diff = deg_rem - deg_div;
71 CoeffType lc_rem = remainder.lead_coeff();
72
73 for (auto& c : quotient.coeffs) c = c * lc_div;
74
75 quotient.coeffs[current_diff] = quotient.coeffs[current_diff] + lc_rem;
76
77 Polynomial term(variable_name);
78 term.coeffs.resize(current_diff + 1, CoeffType(0));
79 term.coeffs[current_diff] = lc_rem;
80
81 for(auto& c : remainder.coeffs) c = c * lc_div;
82
83 for (size_t i = 0; i < other.coeffs.size(); ++i) {
84 size_t target_idx = i + current_diff;

87 remainder.coeffs[target_idx] = remainder.coeffs[target_idx] - other.coeffs[i] * lc_rem;
