36bool power_degrees_fit(const Poly& base, int exponent) {

118void PolyBuilder::visit(const PowerNode& node) {
119 PolyBuilder builder(vars, ext_vars, transcendental_map, aux_to_node, strict_mode);
120 node.base()->accept(builder);
121 if (builder.failed) {
122 failed = true;
123 return;
124 }
125 Poly base = builder.get_result();
126 auto exponent = integer_exponent(node);
127 if (!exponent || *exponent > BigInt(std::numeric_limits<int>::max()) ||
128 *exponent < 0) {
129 if (strict_mode) {
130 failed = true;
131 return;
132 }
133 represent_as_aux_or_fail(
134 LMCAS::detail::make_node<PowerNode>(node.base(), node.exponent()));
135 return;
136 }
137 if (*exponent == 0) {
138 result = Poly(ext_vars.size());
139 result.add_term(Monomial(ext_vars.size(), 0), Rational(1));
140 return;
141 }
142 const int count = static_cast<int>(exponent->try_to_int64().value());
143 if (!power_degrees_fit(base, count)) {
144 represent_as_aux_or_fail(node.clone());
145 return;
146 }
147 Poly product(ext_vars.size());
148 product.add_term(Monomial(ext_vars.size(), 0), Rational(1));
149 for (int index = 0; index < count; ++index) {
150 extend_variables(product, base.num_vars);
151 product = mul_poly(product, base);
152 }
153 result = product;
