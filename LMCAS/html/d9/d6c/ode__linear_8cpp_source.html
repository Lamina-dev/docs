36/**
37 * @internal
38 * @brief 将 double 值转为"干净"的数值表达式.
39 *
40 * 若值接近整数或简单分数,使用精确表示.
41 */
42static std::shared_ptr<SymbolicExpr> clean_number(double val) {
43 /// 只在转换可表示时提升为精确小整数，避免浮点到整数的越界转换。
44 double rounded = std::round(val);
45 int eq = 0;
46 if (std::isfinite(rounded) &&
47 rounded >= static_cast<double>(std::numeric_limits<int>::min()) &&
48 rounded <= static_cast<double>(std::numeric_limits<int>::max())) {
49 lmmc_double_nearly_equal_tol(
50 val, rounded, 1e-10, 1e-10, &eq);
51 if (eq && (rounded != 0.0 || val == 0.0)) {
52 return SymbolicExpr::number(static_cast<int>(rounded));
53 }
54 }
55
56 /// 检查是否接近简单分数 p/q (q <= 12)。
57 for (int q = 2; q <= 12; ++q) {
58 double p = val * q;
59 double p_rounded = std::round(p);
60 if (!std::isfinite(p_rounded) ||
61 p_rounded < static_cast<double>(
62 std::numeric_limits<int>::min()) ||
63 p_rounded > static_cast<double>(
64 std::numeric_limits<int>::max())) {
65 continue;
66 }
67 lmmc_double_nearly_equal_tol(
68 p, p_rounded, 1e-10, 1e-10, &eq);
69 if (eq && (p_rounded != 0.0 || p == 0.0)) {
70 return SymbolicExpr::divide(
71 SymbolicExpr::number(static_cast<int>(p_rounded)),
72 SymbolicExpr::number(q));
73 }
74 }
75
76 return SymbolicExpr::number(val);
77}

83/**
84 * @internal
85 * @brief 根据特征根构造齐次通解.
86 *
87 * - 实根 r(重数 m):C_k * x^k * e^(rx),k = 0, ..., m-1
88 * - 复根 alpha+/-betai(重数 m):x^k * e^(alphax) * (C_a*cos(betax) + C_b*sin(betax))
89 */
90[[maybe_unused]] static std::shared_ptr<SymbolicExpr> build_homogeneous_solution(
91 const std::vector<CharRoot>& roots,
92 const std::string& x,
93 std::vector<std::string>& constants)
94{
95 auto x_var = SymbolicExpr::variable(x);
96 std::shared_ptr<SymbolicExpr> solution = nullptr;
97 int const_idx = 1;
98
99 for (const auto& root : roots) {
100 for (int k = 0; k < root.multiplicity; ++k) {
101 if (!root.is_complex) {
102 /// 实根: C_i * x^k * e^(r*x)
103 std::string c_name = "C" + std::to_string(const_idx++);
104 constants.push_back(c_name);
105 auto C = SymbolicExpr::variable(c_name);
106
107 std::shared_ptr<SymbolicExpr> term = C;
108
109 /// x^k 因子
110 if (k > 0) {
111 auto x_pow = SymbolicExpr::power(x_var, SymbolicExpr::number(k));
112 term = SymbolicExpr::multiply(term, x_pow);
113 }
114
115 /// e^(r*x) 因子。精确零才可省略；小的非零根仍改变解。
116 if (root.real_part != 0.0) {
117 auto r_expr = clean_number(root.real_part);
118 auto exp_arg = SymbolicExpr::multiply(r_expr, x_var);
119 auto exp_term = SymbolicExpr::exp(exp_arg);
120 term = SymbolicExpr::multiply(term, exp_term);
121 }
122
123 solution = solution ? SymbolicExpr::add(solution, term) : term;
124 } else {
125 /// 复根 alpha+/-betai: 产生两个基本解
126 /// C_a * x^k * e^(alphax) * cos(betax)
127 /// C_b * x^k * e^(alphax) * sin(betax)
128 std::string ca_name = "C" + std::to_string(const_idx++);
129 std::string cb_name = "C" + std::to_string(const_idx++);
130 constants.push_back(ca_name);
131 constants.push_back(cb_name);
132 auto Ca = SymbolicExpr::variable(ca_name);
133 auto Cb = SymbolicExpr::variable(cb_name);
134
135 auto beta_expr = clean_number(root.imag_part);
136 auto beta_x = SymbolicExpr::multiply(beta_expr, x_var);
137 auto cos_term = SymbolicExpr::cos(beta_x);
138 auto sin_term = SymbolicExpr::sin(beta_x);
139
140 /// e^(alphax) 因子。精确零才可省略。
141 std::shared_ptr<SymbolicExpr> exp_factor = nullptr;
142 if (root.real_part != 0.0) {
143 auto alpha_expr = clean_number(root.real_part);
144 auto exp_arg = SymbolicExpr::multiply(alpha_expr, x_var);
145 exp_factor = SymbolicExpr::exp(exp_arg);
146 }
147
148 /// x^k 因子
149 std::shared_ptr<SymbolicExpr> x_pow_factor = nullptr;
150 if (k > 0) {
151 x_pow_factor = SymbolicExpr::power(x_var, SymbolicExpr::number(k));
152 }
153
154 /// 构造 cos 项: Ca * x^k * e^(alphax) * cos(betax)
155 auto term_cos = Ca;
156 if (x_pow_factor) term_cos = SymbolicExpr::multiply(term_cos, x_pow_factor);
157 if (exp_factor) term_cos = SymbolicExpr::multiply(term_cos, exp_factor);
158 term_cos = SymbolicExpr::multiply(term_cos, cos_term);
159
160 /// 构造 sin 项: Cb * x^k * e^(alphax) * sin(betax)
161 auto term_sin = Cb;
162 if (x_pow_factor) term_sin = SymbolicExpr::multiply(term_sin, x_pow_factor);
163 if (exp_factor) term_sin = SymbolicExpr::multiply(term_sin, exp_factor);
164 term_sin = SymbolicExpr::multiply(term_sin, sin_term);
165
166 solution = solution ? SymbolicExpr::add(solution, term_cos) : term_cos;
167 solution = SymbolicExpr::add(solution, term_sin);
168 }
169 }
170 }
171
172 return solution;
173}

175static std::shared_ptr<SymbolicExpr> build_euler_solution(
176 const std::vector<CharRoot>& roots,
177 const std::string& x,
178 std::vector<std::string>& constants)
179{
180 auto x_var = SymbolicExpr::variable(x);
181 auto ln_x = SymbolicExpr::ln(x_var);
182 std::shared_ptr<SymbolicExpr> solution = nullptr;
183 int const_idx = 1;
184
185 for (const auto& root : roots) {
186 for (int k = 0; k < root.multiplicity; ++k) {
187 std::shared_ptr<SymbolicExpr> log_factor = nullptr;
188 if (k > 0) {
189 log_factor = SymbolicExpr::power(ln_x, SymbolicExpr::number(k));
190 }
191
192 if (!root.is_complex) {
193 std::string c_name = "C" + std::to_string(const_idx++);
194 constants.push_back(c_name);
195 auto term = SymbolicExpr::variable(c_name);
196
197 auto exponent = clean_number(root.real_part);
198 term = SymbolicExpr::multiply(
199 term,
200 SymbolicExpr::power(x_var, exponent));
201 if (log_factor) {
202 term = SymbolicExpr::multiply(term, log_factor);
203 }
204 solution = solution ? SymbolicExpr::add(solution, term) : term;
205 } else {
206 std::string ca_name = "C" + std::to_string(const_idx++);
207 std::string cb_name = "C" + std::to_string(const_idx++);
208 constants.push_back(ca_name);
209 constants.push_back(cb_name);
210
211 auto alpha_expr = clean_number(root.real_part);
212 auto beta_expr = clean_number(root.imag_part);
213 auto beta_ln_x = SymbolicExpr::multiply(beta_expr, ln_x);
214 auto trig = SymbolicExpr::add(
215 SymbolicExpr::multiply(SymbolicExpr::variable(ca_name),
216 SymbolicExpr::cos(beta_ln_x)),
217 SymbolicExpr::multiply(SymbolicExpr::variable(cb_name),
218 SymbolicExpr::sin(beta_ln_x)));
219 auto term = SymbolicExpr::multiply(
220 SymbolicExpr::power(x_var, alpha_expr),
221 trig);
222 if (log_factor) {
223 term = SymbolicExpr::multiply(term, log_factor);
224 }
225 solution = solution ? SymbolicExpr::add(solution, term) : term;
226 }
227 }
228 }
229
230 return solution;
231}

233ODESolutionResult solve_higher_order_ode_checked(
234 const std::vector<double>& coeffs,
235 const std::shared_ptr<SymbolicExpr>& forcing,
236 const std::string& x,
237 const std::string& y,
238 ComputationContext& context)
239{
240 const std::string operation = "solve_higher_order_ode";
241 auto variables = validate_ode_variables(x, y, context, operation);
242 if (!variables) return ODESolutionResult::failure(variables.error());
243
244 auto coeff_check = validate_numeric_ode_coefficients(coeffs, 2, 7, operation);
245 if (!coeff_check) return ODESolutionResult::failure(coeff_check.error());
246 for (double coefficient : coeffs) {
247 if (!std::isfinite(coefficient / coeffs.front())) {
248 return ODESolutionResult::failure(
249 CasErrc::NumericFailure,
250 "normalizing the characteristic polynomial exceeds the finite numeric domain",
251 operation);
252 }
253 }
254
255 auto budget = context.consume_steps(coeffs.size() * 20 + 20, operation);
256 if (!budget) return ODESolutionResult::failure(budget.error());
257
258 const bool nonzero_forcing = has_nonzero_forcing(forcing);
259 if (nonzero_forcing &&
260 (!forcing->is_number() || coeffs.back() == 0.0)) {
261 return ODESolutionResult::failure(
262 CasErrc::Inconclusive,
263 "non-homogeneous constant-coefficient ODE currently requires a constant forcing and nonzero y coefficient",
264 operation);
265 }
266
267 try {
268 auto solution_result = solve_higher_order_ode_impl(
269 coeffs, nullptr, x, y);
270 if (!solution_result)
271 return ODESolutionResult::failure(solution_result.error());
272 auto solution = std::move(solution_result.value());
273 if (nonzero_forcing && solution.general_solution) {
274 auto particular = SymbolicExpr::divide(
275 forcing, SymbolicExpr::number(coeffs.back()))->simplify();
276 solution.general_solution = SymbolicExpr::add(
277 solution.general_solution, particular)->simplify();
278 }
279 return wrap_ode_solution(
280 std::move(solution),
281 ODEType::HigherOrder_ConstCoeff,
282 operation);
283 } catch (const std::bad_alloc&) {
284 return ODESolutionResult::failure(
285 CasErrc::ResourceLimit,
286 "allocation failed while solving higher-order ODE",
287 operation);
288 } catch (const std::exception& ex) {
289 return ODESolutionResult::failure(
290 CasErrc::InternalInvariant,
291 ex.what(),
292 operation);
293 }
294}

306static Result<ODESolution> solve_higher_order_ode_impl(
307 const std::vector<double>& coeffs,
308 const std::shared_ptr<SymbolicExpr>& forcing,
309 const std::string& x,
310 const std::string&)
311{
312 const std::string operation = "solve_higher_order_ode";
313 ODESolution result;
314 result.method_used = ODEType::HigherOrder_ConstCoeff;
315
316 if (coeffs.size() < 2 || has_nonzero_forcing(forcing)) {
317 return Result<ODESolution>::failure(
318 CasErrc::Inconclusive,
319 "higher-order ODE is outside the implemented support domain",
320 operation);
321 }
322
323 auto roots = find_characteristic_roots(coeffs, operation);
324 if (!roots) return Result<ODESolution>::failure(roots.error());
325 result.general_solution =
326 build_homogeneous_solution(roots.value(), x, result.constants);
327 if (result.general_solution) {
328 result.general_solution = result.general_solution->simplify();
329 }
330 return Result<ODESolution>::success(std::move(result));
331}

399static Result<ODESolution> solve_euler_ode_impl(
400 const std::vector<double>& euler_coeffs,
401 const std::shared_ptr<SymbolicExpr>& forcing,
402 const std::string& x,
403 const std::string&)
404{
405 const std::string operation = "solve_euler_ode";
406 ODESolution result;
407 result.method_used = ODEType::Euler;
408
409 if ((euler_coeffs.size() != 3 && euler_coeffs.size() != 4) ||
410 has_nonzero_forcing(forcing)) {
411 return Result<ODESolution>::failure(
412 CasErrc::Inconclusive,
413 "Euler ODE is outside the implemented support domain",
414 operation);
415 }
416
417 std::vector<double> characteristic;
418 if (euler_coeffs.size() == 3) {
419 const double a = euler_coeffs[0];
420 const double b = euler_coeffs[1];
421 const double c = euler_coeffs[2];
422 characteristic = {a, b - a, c};
423 } else {
424 const double a = euler_coeffs[0];
425 const double b = euler_coeffs[1];
426 const double c = euler_coeffs[2];
427 const double d = euler_coeffs[3];
428 characteristic = {a, b - 3.0 * a, 2.0 * a - b + c, d};
429 }
430
431 auto roots = find_characteristic_roots(characteristic, operation);
432 if (!roots) return Result<ODESolution>::failure(roots.error());
433 result.general_solution =
434 build_euler_solution(roots.value(), x, result.constants);
435 if (result.general_solution) {
436 result.general_solution = result.general_solution->simplify();
437 }
438 return Result<ODESolution>::success(std::move(result));
439}
