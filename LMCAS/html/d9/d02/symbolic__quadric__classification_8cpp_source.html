22double largest_off_diagonal(const double matrix[3][3], size_t& p, size_t& q)
23{
24 p = 0;
25 q = 1;
26 double off_diagonal = std::abs(matrix[p][q]);
27 for (size_t row = 0; row < 3; ++row) {
28 for (size_t column = row + 1; column < 3; ++column) {
29 const double candidate = std::abs(matrix[row][column]);
30 if (candidate > off_diagonal) {
31 off_diagonal = candidate;
32 p = row;
33 q = column;
34 }
35 }
36 }
37 return off_diagonal;
38}

40void rotate_principal_axes(
41 double matrix[3][3], double vectors[3][3], size_t p, size_t q)
42{
43 const double angle = 0.5 * std::atan2(
44 2.0 * matrix[p][q], matrix[q][q] - matrix[p][p]);
45 const double cosine = std::cos(angle);
46 const double sine = std::sin(angle);
47 const double pp = matrix[p][p];
48 const double pq = matrix[p][q];
49 const double qq = matrix[q][q];
50
51 for (size_t axis = 0; axis < 3; ++axis) {
52 if (axis == p || axis == q) continue;
53 const double axis_p = matrix[axis][p];
54 const double axis_q = matrix[axis][q];
55 matrix[axis][p] = cosine * axis_p - sine * axis_q;
56 matrix[p][axis] = matrix[axis][p];
57 matrix[axis][q] = sine * axis_p + cosine * axis_q;
58 matrix[q][axis] = matrix[axis][q];
59 }
60 matrix[p][p] = cosine * cosine * pp -
61 2.0 * sine * cosine * pq +
62 sine * sine * qq;
63 matrix[q][q] = sine * sine * pp +
64 2.0 * sine * cosine * pq +
65 cosine * cosine * qq;
66 matrix[p][q] = 0.0;
67 matrix[q][p] = 0.0;
68
69 for (size_t axis = 0; axis < 3; ++axis) {
70 const double vector_p = vectors[axis][p];
71 const double vector_q = vectors[axis][q];
72 vectors[axis][p] = cosine * vector_p - sine * vector_q;
73 vectors[axis][q] = sine * vector_p + cosine * vector_q;
74 }
75}

77SymmetricEigen3 diagonalize_symmetric_3x3(const double input[3][3])
78{
79 double matrix[3][3];
80 SymmetricEigen3 result{{0.0, 0.0, 0.0},
81 {{1.0, 0.0, 0.0},
82 {0.0, 1.0, 0.0},
83 {0.0, 0.0, 1.0}},
84 0.0,
85 0.0,
86 false};
87 for (size_t row = 0; row < 3; ++row) {
88 for (size_t column = 0; column < 3; ++column) {
89 result.coefficient_scale = std::max(
90 result.coefficient_scale, std::abs(input[row][column]));
91 }
92 }
93 if (result.coefficient_scale == 0.0) return result;
94 for (size_t row = 0; row < 3; ++row) {
95 for (size_t column = 0; column < 3; ++column) {
96 matrix[row][column] =
97 input[row][column] / result.coefficient_scale;
98 }
99 }
100 result.zero_tolerance =
101 256.0 * std::numeric_limits<double>::epsilon();
102
103 for (size_t iteration = 0; iteration < 64; ++iteration) {
104 size_t p;
105 size_t q;
106 if (largest_off_diagonal(matrix, p, q) <= result.zero_tolerance) break;
107 rotate_principal_axes(matrix, result.vectors, p, q);
108 }
109 for (size_t axis = 0; axis < 3; ++axis) {
110 result.values[axis] = matrix[axis][axis];
111 if (std::abs(result.values[axis]) <= result.zero_tolerance) {
112 result.unresolved_rank =
113 result.unresolved_rank || result.values[axis] != 0.0;
114 result.values[axis] = 0.0;
115 }
116 }
117 return result;
118}

120Result<double> finite_numeric_coeff(
121 const std::shared_ptr<SymbolicExpr>& expression,
122 ComputationContext& context,
123 const std::string& operation,
124 const std::string& message)
125{
126 auto simplified = expression ? expression->simplify() : nullptr;
127 if (!simplified || !node(simplified)) {
128 return Result<double>::failure(
129 CasErrc::Inconclusive, message, operation);
130 }
131 auto evaluated = evaluate_numeric(
132 *simplified, NumericBindings{}, context);
133 if (!evaluated) {
134 if (evaluated.error().code == CasErrc::Cancelled ||
135 evaluated.error().code == CasErrc::ResourceLimit) {
136 return Result<double>::failure(evaluated.error());
137 }
138 return Result<double>::failure(
139 CasErrc::Inconclusive, message, operation);
140 }
141 if (!evaluated.value().is_finite() ||
142 !std::isfinite(evaluated.value().value)) {
143 return Result<double>::failure(
144 CasErrc::Inconclusive, message, operation);
145 }
146 return Result<double>::success(evaluated.value().value);
147}

154{

199bool extract_axis_coefficients(
200 const std::shared_ptr<SymbolicExpr>& expanded,
201 const std::vector<std::string>& vars,
202 ComputationContext& context,
203 const std::string& operation,
204 double quadratic_matrix[3][3],
205 double linear[3],
206 CasError& error)
207{
208 Result<double> diagonal[3] = {
209 coefficient_of_square(expanded, vars[0], context, operation),
210 coefficient_of_square(expanded, vars[1], context, operation),
211 coefficient_of_square(expanded, vars[2], context, operation)};
212 Result<double> linear_coefficients[3] = {
213 coefficient_of_linear(expanded, vars[0], vars, context, operation),
214 coefficient_of_linear(expanded, vars[1], vars, context, operation),
215 coefficient_of_linear(expanded, vars[2], vars, context, operation)};
216 for (size_t axis = 0; axis < 3; ++axis) {
217 if (!diagonal[axis]) {
218 error = std::move(diagonal[axis].error());
219 return false;
220 }
221 if (!linear_coefficients[axis]) {
222 error = std::move(linear_coefficients[axis].error());
223 return false;
224 }
225 }
226 for (size_t axis = 0; axis < 3; ++axis) {
227 quadratic_matrix[axis][axis] = diagonal[axis].value();
228 linear[axis] = linear_coefficients[axis].value();
229 }
230 return true;
231}

233bool extract_mixed_coefficients(
234 const std::shared_ptr<SymbolicExpr>& expanded,
235 const std::vector<std::string>& vars,
236 ComputationContext& context,
237 const std::string& operation,
238 double quadratic_matrix[3][3],
239 CasError& error)
240{
241 for (size_t i = 0; i < 3; ++i) {
242 for (size_t j = i + 1; j < 3; ++j) {
243 auto mixed = expanded->differentiate(vars[i]);
244 if (!mixed || !node(mixed)) {
245 error = {
246 CasErrc::Inconclusive,
247 "quadric mixed derivative is outside the supported domain",
248 operation};
249 return false;
250 }
251 mixed = mixed->differentiate(vars[j]);
252 auto mixed_value = finite_numeric_coeff(
253 mixed, context, operation,
254 "quadric mixed coefficient cannot be evaluated");
255 if (!mixed_value) {
256 error = std::move(mixed_value.error());
257 return false;
258 }
259 quadratic_matrix[i][j] = mixed_value.value() * 0.5;
260 quadratic_matrix[j][i] = quadratic_matrix[i][j];
261 }
262 }
263 return true;
264}

266bool project_linear_coefficients(
267 const SymmetricEigen3& principal,
268 const double coefficients[3],
269 double linear[3],
270 double linear_tolerance[3],
271 const std::string& operation,
272 CasError& error)
273{
274 double projection_scale[3] = {0.0, 0.0, 0.0};
275 for (size_t row = 0; row < 3; ++row) {
276 const double normalized =
277 coefficients[row] / principal.coefficient_scale;
278 if (!std::isfinite(normalized)) {
279 error = {
280 CasErrc::Inconclusive,
281 "quadric linear part is outside the supported scale",
282 operation};
283 return false;
284 }
285 for (size_t axis = 0; axis < 3; ++axis) {
286 const double term = principal.vectors[row][axis] * normalized;
287 projection_scale[axis] =
288 std::max(projection_scale[axis], std::abs(term));
289 linear[axis] = std::fma(
290 principal.vectors[row][axis], normalized, linear[axis]);
291 }
292 }
293 for (size_t axis = 0; axis < 3; ++axis) {
294 linear_tolerance[axis] =
295 256.0 * std::numeric_limits<double>::epsilon() *
296 projection_scale[axis];
297 }
298 return true;
299}

342Result<CenteredQuadric> center_quadric(
343 const SymmetricEigen3& principal,
344 const double linear[3],
345 double constant,
346 const std::string& operation)
347{
348 double centered_constant = constant / principal.coefficient_scale;
349 if (!std::isfinite(centered_constant)) {
350 return Result<CenteredQuadric>::failure(
351 CasErrc::Inconclusive,
352 "quadric constant is outside the supported scale",
353 operation);
354 }
355 double centered_scale = std::max(1.0, std::abs(centered_constant));
356 for (size_t axis = 0; axis < 3; ++axis) {
357 if (principal.values[axis] == 0.0) continue;
358 const double factor =
359 (linear[axis] / principal.values[axis]) * 0.25;
360 const double shift = factor * linear[axis];
361 if (!std::isfinite(factor) || !std::isfinite(shift)) {
362 return Result<CenteredQuadric>::failure(
363 CasErrc::Inconclusive,
364 "quadric translation is outside the supported domain",
365 operation);
366 }
367 centered_constant = std::fma(
368 -factor, linear[axis], centered_constant);
369 centered_scale = std::max(centered_scale, std::abs(shift));
370 }
371 const double zero_tolerance =
372 256.0 * std::numeric_limits<double>::epsilon() * centered_scale;
373 return Result<CenteredQuadric>::success({centered_constant, zero_tolerance});
374}

376VectorStringResult classify_rank_two_locus(
377 int positive,
378 int negative,
379 double centered_constant,
380 double zero_tolerance,
381 const std::string& operation)
382{
383 if (std::abs(centered_constant) <= zero_tolerance) {
384 return VectorStringResult::failure(
385 CasErrc::Inconclusive,
386 "rank-two quadric has a degenerate real locus",
387 operation);
388 }
389 const bool indefinite = positive == 1 && negative == 1;
390 const bool real_definite =
391 (positive == 2 && centered_constant < -zero_tolerance) ||
392 (negative == 2 && centered_constant > zero_tolerance);
393 if (indefinite || real_definite) {
394 return VectorStringResult::success("cylinder");
395 }
396 return VectorStringResult::failure(
397 CasErrc::Inconclusive,
398 "rank-two quadric has no real cylinder locus",
399 operation);

402VectorStringResult classify_definite_locus(
403 const SymmetricEigen3& principal,
404 int positive,
405 int negative,
406 double centered_constant,
407 double zero_tolerance,
408 const std::string& operation)
409{
410 const bool real_ellipsoid =
411 (positive == 3 && centered_constant < -zero_tolerance) ||
412 (negative == 3 && centered_constant > zero_tolerance);
413 if (!real_ellipsoid) {
414 return VectorStringResult::failure(
415 CasErrc::Inconclusive,
416 std::abs(centered_constant) <= zero_tolerance
417 ? "definite quadric degenerates to a point"
418 : "definite quadric has no real locus",
419 operation);
420 }
421 const double min_value = std::min(
422 principal.values[0],
423 std::min(principal.values[1], principal.values[2]));
424 const double max_value = std::max(
425 principal.values[0],
426 std::max(principal.values[1], principal.values[2]));
427 if (max_value - min_value <= principal.zero_tolerance) {
428 return VectorStringResult::success("sphere");
429 }
430 return VectorStringResult::success("ellipsoid");
431}

433VectorStringResult classify_centered_locus(
434 const SymmetricEigen3& principal,
435 int rank,
436 double centered_constant,
437 double zero_tolerance,
438 const std::string& operation)
439{
440 const int positive =
441 (principal.values[0] > 0.0) +
442 (principal.values[1] > 0.0) +
443 (principal.values[2] > 0.0);
444 const int negative =
445 (principal.values[0] < 0.0) +
446 (principal.values[1] < 0.0) +
447 (principal.values[2] < 0.0);
448 if (rank == 2) {
449 return classify_rank_two_locus(
450 positive, negative, centered_constant, zero_tolerance, operation);
451 }
452 if (rank == 3) {

454 return classify_definite_locus(

458 if (std::abs(centered_constant) <= zero_tolerance) {
459 return VectorStringResult::success("cone");
460 }
461 return VectorStringResult::success("hyperboloid");
462 }
463 return VectorStringResult::failure(
464 CasErrc::Inconclusive,
465 "quadric is outside the currently classified support domain",
466 operation);

490VectorStringResult classify_quadric_impl(
491 const SurfaceSymbolic& surf,
492 ComputationContext& context,
493 const std::string& operation)
494{
495 const auto& vars = surf.vars;
496 auto expanded = surf.F->expand();
497 if (!expanded || !node(expanded)) expanded = surf.F;
498
499 double quadratic_matrix[3][3] = {};
500 double linear_coefficients[3];
501 CasError error;
502 if (!extract_axis_coefficients(
503 expanded, vars, context, operation, quadratic_matrix,
504 linear_coefficients, error)) {
505 return VectorStringResult::failure(std::move(error));
506 }
507 if (!extract_mixed_coefficients(
508 expanded, vars, context, operation, quadratic_matrix, error)) {
509 return VectorStringResult::failure(std::move(error));
510 }
511
512 const SymmetricEigen3 principal =
513 diagonalize_symmetric_3x3(quadratic_matrix);
514 if (principal.coefficient_scale == 0.0) {
515 return VectorStringResult::failure(
516 CasErrc::Inconclusive,
517 "quadric has no nonzero quadratic part",
518 operation);
519 }
520 if (principal.unresolved_rank) {
521 return VectorStringResult::failure(
522 CasErrc::Inconclusive,
523 "quadric rank is unresolved within the eigensolver error bound",
524 operation);
525 }
526
527 double linear[3] = {0.0, 0.0, 0.0};
528 double linear_tolerance[3];
529 if (!project_linear_coefficients(
530 principal, linear_coefficients, linear, linear_tolerance,
531 operation, error)) {
532 return VectorStringResult::failure(std::move(error));
533 }
534 const int rank =
535 (principal.values[0] != 0.0) +
536 (principal.values[1] != 0.0) +
537 (principal.values[2] != 0.0);
538
539 if (rank == 1) {
540 if (has_null_axis_linear_term(principal, linear, linear_tolerance)) {
541 return VectorStringResult::success("cylinder");
542 }
543 return VectorStringResult::failure(
544 CasErrc::Inconclusive,
545 "rank-one quadric is degenerate or has no real locus",
546 operation);
547 }
548 if (rank == 2 &&
549 has_null_axis_linear_term(principal, linear, linear_tolerance)) {
550 return VectorStringResult::success("paraboloid");
551 }
552 return classify_translated_quadric(
553 principal, rank, linear, expanded, vars, context, operation);
554}
