21SymmetricEigen3 diagonalize_symmetric_3x3(const double input[3][3])
22{
23 double matrix[3][3];
24 SymmetricEigen3 result{{0.0, 0.0, 0.0},
25 {{1.0, 0.0, 0.0},
26 {0.0, 1.0, 0.0},
27 {0.0, 0.0, 1.0}},
28 0.0,
29 0.0,
30 false};
31 for (size_t row = 0; row < 3; ++row) {
32 for (size_t column = 0; column < 3; ++column) {
33 result.coefficient_scale = std::max(
34 result.coefficient_scale, std::abs(input[row][column]));
35 }
36 }
37 if (result.coefficient_scale == 0.0) return result;
38 for (size_t row = 0; row < 3; ++row) {
39 for (size_t column = 0; column < 3; ++column) {
40 matrix[row][column] =
41 input[row][column] / result.coefficient_scale;
42 }
43 }
44 result.zero_tolerance =
45 256.0 * std::numeric_limits<double>::epsilon();
46
47 for (size_t iteration = 0; iteration < 64; ++iteration) {
48 size_t p = 0;
49 size_t q = 1;
50 double off_diagonal = std::abs(matrix[p][q]);
51 for (size_t row = 0; row < 3; ++row) {
52 for (size_t column = row + 1; column < 3; ++column) {
53 const double candidate = std::abs(matrix[row][column]);
54 if (candidate > off_diagonal) {
55 off_diagonal = candidate;
56 p = row;
57 q = column;
58 }
59 }
60 }
61 if (off_diagonal <= result.zero_tolerance) break;
62
63 const double angle = 0.5 * std::atan2(
64 2.0 * matrix[p][q], matrix[q][q] - matrix[p][p]);
65 const double cosine = std::cos(angle);
66 const double sine = std::sin(angle);
67 const double pp = matrix[p][p];
68 const double pq = matrix[p][q];
69 const double qq = matrix[q][q];
70
71 for (size_t axis = 0; axis < 3; ++axis) {
72 if (axis == p || axis == q) continue;
73 const double axis_p = matrix[axis][p];
74 const double axis_q = matrix[axis][q];
75 matrix[axis][p] = cosine * axis_p - sine * axis_q;
76 matrix[p][axis] = matrix[axis][p];
77 matrix[axis][q] = sine * axis_p + cosine * axis_q;
78 matrix[q][axis] = matrix[axis][q];
79 }
80 matrix[p][p] = cosine * cosine * pp -
81 2.0 * sine * cosine * pq +
82 sine * sine * qq;
83 matrix[q][q] = sine * sine * pp +
84 2.0 * sine * cosine * pq +
85 cosine * cosine * qq;
86 matrix[p][q] = 0.0;
87 matrix[q][p] = 0.0;
88
89 for (size_t axis = 0; axis < 3; ++axis) {
90 const double vector_p = result.vectors[axis][p];
91 const double vector_q = result.vectors[axis][q];
92 result.vectors[axis][p] =
93 cosine * vector_p - sine * vector_q;
94 result.vectors[axis][q] =
95 sine * vector_p + cosine * vector_q;
96 }
97 }
98 for (size_t axis = 0; axis < 3; ++axis) {
99 result.values[axis] = matrix[axis][axis];
100 if (std::abs(result.values[axis]) <= result.zero_tolerance) {
101 result.unresolved_rank =
102 result.unresolved_rank || result.values[axis] != 0.0;
103 result.values[axis] = 0.0;
104 }
105 }
106 return result;
107}

109Result<double> finite_numeric_coeff(
110 const std::shared_ptr<SymbolicExpr>& expression,
111 ComputationContext& context,
112 const std::string& operation,
113 const std::string& message)
114{
115 auto simplified = expression ? expression->simplify() : nullptr;
116 if (!simplified || !node(simplified)) {
117 return Result<double>::failure(
118 CasErrc::Inconclusive, message, operation);
119 }
120 auto evaluated = evaluate_numeric(
121 *simplified, NumericBindings{}, context);
122 if (!evaluated) {
123 if (evaluated.error().code == CasErrc::Cancelled ||
124 evaluated.error().code == CasErrc::ResourceLimit) {
125 return Result<double>::failure(evaluated.error());
126 }
127 return Result<double>::failure(
128 CasErrc::Inconclusive, message, operation);
129 }
130 if (!evaluated.value().is_finite() ||
131 !std::isfinite(evaluated.value().value)) {
132 return Result<double>::failure(
133 CasErrc::Inconclusive, message, operation);
134 }
135 return Result<double>::success(evaluated.value().value);
136}

149 auto coefficient_of_square = [&](const std::string& variable)
150 -> Result<double> {
151 auto derivative = expanded->differentiate(variable);
152 if (!derivative || !node(derivative)) {
153 return Result<double>::failure(
154 CasErrc::Inconclusive,
155 "quadric second derivative is outside the supported domain",
156 operation);
157 }
158 derivative = derivative->differentiate(variable);
159 auto value = finite_numeric_coeff(
160 derivative, context, operation,
161 "quadric second derivative coefficient cannot be evaluated");
162 if (!value) return value;
163 return Result<double>::success(value.value() / 2.0);
164 };
165 auto coefficient_of_linear = [&](const std::string& variable)
166 -> Result<double> {
167 auto derivative = expanded->differentiate(variable);
168 if (!derivative || !node(derivative)) {
169 return Result<double>::failure(
170 CasErrc::Inconclusive,
171 "quadric first derivative is outside the supported domain",
172 operation);
173 }
174 for (const auto& coordinate : vars) {
175 derivative = derivative->substitute(
176 coordinate, SymbolicExpr::number(0));
177 if (!derivative || !node(derivative)) {
178 return Result<double>::failure(
179 CasErrc::Inconclusive,
180 "quadric linear coefficient cannot be substituted",
181 operation);
182 }
183 }
184 return finite_numeric_coeff(
185 derivative, context, operation,
186 "quadric linear coefficient cannot be evaluated");
187 };

189 Result<double> diagonal[3] = {
190 coefficient_of_square(vars[0]),
191 coefficient_of_square(vars[1]),
192 coefficient_of_square(vars[2])};
193 Result<double> linear_coefficients[3] = {
194 coefficient_of_linear(vars[0]),
195 coefficient_of_linear(vars[1]),
196 coefficient_of_linear(vars[2])};
197 for (size_t axis = 0; axis < 3; ++axis) {
198 if (!diagonal[axis]) {
199 return VectorStringResult::failure(diagonal[axis].error());
200 }
201 if (!linear_coefficients[axis]) {
202 return VectorStringResult::failure(
203 linear_coefficients[axis].error());
204 }
205 }

207 double quadratic_matrix[3][3] = {
208 {diagonal[0].value(), 0.0, 0.0},
209 {0.0, diagonal[1].value(), 0.0},
210 {0.0, 0.0, diagonal[2].value()}};
211 for (size_t i = 0; i < 3; ++i) {
212 for (size_t j = i + 1; j < 3; ++j) {
213 auto mixed = expanded->differentiate(vars[i]);
214 if (!mixed || !node(mixed)) {
215 return VectorStringResult::failure(
216 CasErrc::Inconclusive,
217 "quadric mixed derivative is outside the supported domain",
218 operation);
219 }
220 mixed = mixed->differentiate(vars[j]);
221 auto mixed_value = finite_numeric_coeff(
222 mixed, context, operation,
223 "quadric mixed coefficient cannot be evaluated");
224 if (!mixed_value) {
225 return VectorStringResult::failure(mixed_value.error());
226 }
227 quadratic_matrix[i][j] = mixed_value.value() * 0.5;
228 quadratic_matrix[j][i] = quadratic_matrix[i][j];
229 }
230 }

232 const SymmetricEigen3 principal =
233 diagonalize_symmetric_3x3(quadratic_matrix);
234 if (principal.coefficient_scale == 0.0) {
235 return VectorStringResult::failure(
236 CasErrc::Inconclusive,
237 "quadric has no nonzero quadratic part",
238 operation);
239 }
240 if (principal.unresolved_rank) {
241 return VectorStringResult::failure(
242 CasErrc::Inconclusive,
243 "quadric rank is unresolved within the eigensolver error bound",
244 operation);
245 }
246
247 double linear[3] = {0.0, 0.0, 0.0};
248 double projection_scale[3] = {0.0, 0.0, 0.0};
249 for (size_t row = 0; row < 3; ++row) {
250 const double normalized =
251 linear_coefficients[row].value() / principal.coefficient_scale;
252 if (!std::isfinite(normalized)) {
253 return VectorStringResult::failure(
254 CasErrc::Inconclusive,
255 "quadric linear part is outside the supported scale",
256 operation);
257 }
258 for (size_t axis = 0; axis < 3; ++axis) {
259 const double term = principal.vectors[row][axis] * normalized;
260 projection_scale[axis] =
261 std::max(projection_scale[axis], std::abs(term));
262 linear[axis] = std::fma(
263 principal.vectors[row][axis], normalized, linear[axis]);
264 }
265 }
266 double linear_tolerance[3];
267 for (size_t axis = 0; axis < 3; ++axis) {
268 linear_tolerance[axis] =
269 256.0 * std::numeric_limits<double>::epsilon() *
270 projection_scale[axis];
271 }
272 const int rank =
273 (principal.values[0] != 0.0) +
274 (principal.values[1] != 0.0) +
275 (principal.values[2] != 0.0);
276 const int positive =
277 (principal.values[0] > 0.0) +
278 (principal.values[1] > 0.0) +
279 (principal.values[2] > 0.0);
280 const int negative =
281 (principal.values[0] < 0.0) +
282 (principal.values[1] < 0.0) +
283 (principal.values[2] < 0.0);
284
285 if (rank == 1) {
286 for (size_t axis = 0; axis < 3; ++axis) {
287 if (principal.values[axis] == 0.0 &&
288 std::abs(linear[axis]) > linear_tolerance[axis]) {
289 return VectorStringResult::success("cylinder");
290 }
291 }
292 return VectorStringResult::failure(
293 CasErrc::Inconclusive,
294 "rank-one quadric is degenerate or has no real locus",
295 operation);
296 }
297
298 if (rank == 2) {
299 for (size_t axis = 0; axis < 3; ++axis) {
300 if (principal.values[axis] == 0.0 &&
301 std::abs(linear[axis]) > linear_tolerance[axis]) {
302 return VectorStringResult::success("paraboloid");
303 }
304 }
305 }

307 auto constant = expanded;
308 for (const auto& coordinate : vars) {
309 constant = constant->substitute(
310 coordinate, SymbolicExpr::number(0));
311 if (!constant || !node(constant)) {
312 return VectorStringResult::failure(
313 CasErrc::Inconclusive,
314 "quadric constant term cannot be substituted",
315 operation);
316 }
317 }
318 auto constant_value = finite_numeric_coeff(
319 constant, context, operation,
320 "quadric constant term cannot be evaluated");
321 if (!constant_value) {
322 return VectorStringResult::failure(constant_value.error());
323 }
324
325 double centered_constant =
326 constant_value.value() / principal.coefficient_scale;
327 if (!std::isfinite(centered_constant)) {
328 return VectorStringResult::failure(
329 CasErrc::Inconclusive,
330 "quadric constant is outside the supported scale",
331 operation);
332 }
333 double centered_scale = std::max(1.0, std::abs(centered_constant));
334 for (size_t axis = 0; axis < 3; ++axis) {
335 if (principal.values[axis] == 0.0) continue;
336 const double factor =
337 (linear[axis] / principal.values[axis]) * 0.25;
338 const double shift = factor * linear[axis];
339 if (!std::isfinite(factor) || !std::isfinite(shift)) {
340 return VectorStringResult::failure(
341 CasErrc::Inconclusive,
342 "quadric translation is outside the supported domain",
343 operation);
344 }
345 centered_constant = std::fma(
346 -factor, linear[axis], centered_constant);
347 centered_scale = std::max(centered_scale, std::abs(shift));
348 }
349 const double zero_tolerance =
350 256.0 * std::numeric_limits<double>::epsilon() * centered_scale;
351 const bool centered_zero =
352 std::abs(centered_constant) <= zero_tolerance;
353
354 if (rank == 2) {
355 if (centered_zero) {
356 return VectorStringResult::failure(
357 CasErrc::Inconclusive,
358 "rank-two quadric has a degenerate real locus",
359 operation);
360 }
361 const bool indefinite = positive == 1 && negative == 1;
362 const bool real_definite =
363 (positive == 2 && centered_constant < -zero_tolerance) ||
364 (negative == 2 && centered_constant > zero_tolerance);
365 if (indefinite || real_definite) {
366 return VectorStringResult::success("cylinder");
367 }
368 return VectorStringResult::failure(
369 CasErrc::Inconclusive,
370 "rank-two quadric has no real cylinder locus",
371 operation);
372 }
373
374 if (rank == 3) {
375 if (negative == 0 || positive == 0) {
376 const bool real_ellipsoid =
377 (positive == 3 && centered_constant < -zero_tolerance) ||
378 (negative == 3 && centered_constant > zero_tolerance);
379 if (!real_ellipsoid) {
380 return VectorStringResult::failure(
381 CasErrc::Inconclusive,
382 centered_zero
383 ? "definite quadric degenerates to a point"
384 : "definite quadric has no real locus",
385 operation);
386 }
387 const double min_value = std::min(
388 principal.values[0],
389 std::min(principal.values[1], principal.values[2]));
390 const double max_value = std::max(
391 principal.values[0],
392 std::max(principal.values[1], principal.values[2]));
393 if (max_value - min_value <= principal.zero_tolerance) {
394 return VectorStringResult::success("sphere");
395 }
396 return VectorStringResult::success("ellipsoid");
397 }
398 if (centered_zero) {
399 return VectorStringResult::success("cone");
400 }
401 return VectorStringResult::success("hyperboloid");
402 }
403
404 return VectorStringResult::failure(
405 CasErrc::Inconclusive,
406 "quadric is outside the currently classified support domain",
407 operation);
408}
