13static bool is_alternating_factor(
14 const std::shared_ptr<const SymbolicNode>& node, const std::string& n) {
15 auto power = std::dynamic_pointer_cast<const PowerNode>(node);
16 if (!power) {
17 return false;
18 }
19 auto base = std::dynamic_pointer_cast<const NumberNode>(power->base());
20 auto exponent = std::dynamic_pointer_cast<const VariableNode>(power->exponent());
21 if (!base || !exponent || exponent->is_constant() || exponent->name() != n) {
22 return false;
23 }
24 return base->is_negative_one();
25}

44static bool series_extract_alternating(
45 const std::shared_ptr<const SymbolicNode>& node,
46 const std::string& n, std::shared_ptr<SymbolicExpr>& remainder) {
47 if (is_alternating_factor(node, n)) {
48 remainder = SymbolicExpr::number(1);
49 return true;
50 }
51 auto product = std::dynamic_pointer_cast<const MultiplyNode>(node);
52 if (!product) {
53 return false;
54 }
55 for (size_t i = 0; i < product->operands().size(); ++i) {
56 if (is_alternating_factor(product->operands()[i], n)) {
57 remainder = product_without_factor(*product, i);
58 return true;
59 }
60 }
61 return false;
62}

80static ExpressionResult alternating_sequence_bound(
81 const std::shared_ptr<SymbolicExpr>& remainder, const std::string& n,
82 bool upper, ComputationContext& context) {
83 auto limited = limit_expression_checked(
84 remainder, n, SymbolicExpr::infinity(), LimitDirection::Both, context);
85 if (!limited) {
86 return limited;
87 }
88 auto limit = std::move(limited.value());
89 auto simplified = limit ? limit->simplify() : nullptr;
90 if (simplified && !series_is_infinity(simplified)) {
91 auto magnitude = series_abs(simplified);
92 return upper ? magnitude : series_negate(magnitude);
93 }
94 if (series_is_infinity(simplified)) {
95 return upper ? SymbolicExpr::infinity() : SymbolicExpr::infinity(-1);
96 }
97 auto magnitude = series_abs(limit);
98 return upper ? magnitude : series_negate(magnitude);
99}
