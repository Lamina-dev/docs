15static bool is_alternating_factor(
16 const std::shared_ptr<const SymbolicNode>& node, const std::string& n) {
17 auto power = std::dynamic_pointer_cast<const PowerNode>(node);

22 auto exponent = std::dynamic_pointer_cast<const VariableNode>(power->exponent());

46static bool series_extract_alternating(
47 const std::shared_ptr<const SymbolicNode>& node,
48 const std::string& n, std::shared_ptr<SymbolicExpr>& remainder) {
49 if (is_alternating_factor(node, n)) {
50 remainder = SymbolicExpr::number(1);
51 return true;
52 }
53 auto product = std::dynamic_pointer_cast<const MultiplyNode>(node);
54 if (!product) {
55 return false;
56 }
57 for (size_t i = 0; i < product->operands().size(); ++i) {
58 if (is_alternating_factor(product->operands()[i], n)) {
59 remainder = product_without_factor(*product, i);
60 return true;
61 }
62 }
63 return false;
64}

89 const std::shared_ptr<SymbolicExpr>& remainder, const std::string& n,
90 bool upper, ComputationContext& context) try {

92 (context).limits().max_ast_nodes, "transform");
93 auto limited = limit_expression_checked(
94 remainder, n, SymbolicExpr::infinity(), LimitDirection::Both, context);
95 if (!limited) {
96 return limited;
97 }
98 auto limit = std::move(limited.value());
99 auto simplified = limit ? detail::make_expression_ptr(detail::normalize_node(detail::node(limit), transform_budget)) : nullptr;
100 if (simplified && !series_is_infinity(simplified)) {
101 auto magnitude = series_abs(simplified, context);
102 return upper ? magnitude : series_negate(magnitude, context);
103 }
104 if (series_is_infinity(simplified)) {
105 return upper ? SymbolicExpr::infinity() : SymbolicExpr::infinity(-1);
106 }
107 auto magnitude = series_abs(limit, context);

109} catch (const CasError& error) {
