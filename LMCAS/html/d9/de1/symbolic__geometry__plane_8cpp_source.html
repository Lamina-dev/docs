23bool plane_point_differences_overflow(
24 const std::vector<double>& p1,
25 const std::vector<double>& p2,
26 const std::vector<double>& p3) {
27 for (size_t i = 0; i < 3; ++i) {
28 if (!std::isfinite(p2[i] - p1[i]) ||
29 !std::isfinite(p3[i] - p1[i])) {
30 return true;
31 }
32 }
33 return false;

51Result<void> append_symbolic_plane_edges(
52 const std::vector<std::shared_ptr<SymbolicExpr>>& p1,
53 const std::vector<std::shared_ptr<SymbolicExpr>>& p2,
54 const std::vector<std::shared_ptr<SymbolicExpr>>& p3,
55 std::vector<std::shared_ptr<SymbolicExpr>>& v1,
56 std::vector<std::shared_ptr<SymbolicExpr>>& v2,
57 const std::string& operation) {
58 for (size_t i = 0; i < 3; ++i) {
59 auto left = simplify_checked(
60 SymbolicExpr::add(
61 p2[i],
62 SymbolicExpr::multiply(SymbolicExpr::number(-1), p1[i])),
63 operation,
64 "plane edge vector is outside the supported domain");
65 if (!left) return Result<void>::failure(left.error());
66 auto right = simplify_checked(
67 SymbolicExpr::add(
68 p3[i],
69 SymbolicExpr::multiply(SymbolicExpr::number(-1), p1[i])),
70 operation,
71 "plane edge vector is outside the supported domain");
72 if (!right) return Result<void>::failure(right.error());
73 v1.push_back(std::move(left.value()));
74 v2.push_back(std::move(right.value()));
75 }
76 return Result<void>::success();
77}

79Result<void> build_plane_edges(
80 const std::vector<std::shared_ptr<SymbolicExpr>>& p1,
81 const std::vector<std::shared_ptr<SymbolicExpr>>& p2,
82 const std::vector<std::shared_ptr<SymbolicExpr>>& p3,
83 std::vector<std::shared_ptr<SymbolicExpr>>& v1,
84 std::vector<std::shared_ptr<SymbolicExpr>>& v2,
85 ComputationContext& context,
86 const std::string& operation) {
87 std::vector<double> p1_values;
88 std::vector<double> p2_values;
89 std::vector<double> p3_values;
90 auto p1_scale = numeric_vector_scale_checked(
91 p1, context, operation, &p1_values);
92 auto p2_scale = numeric_vector_scale_checked(
93 p2, context, operation, &p2_values);
94 auto p3_scale = numeric_vector_scale_checked(
95 p3, context, operation, &p3_values);
96 if (is_plane_budget_error(p1_scale)) {
97 return Result<void>::failure(p1_scale.error());
98 }
99 if (is_plane_budget_error(p2_scale)) {
100 return Result<void>::failure(p2_scale.error());
101 }
102 if (is_plane_budget_error(p3_scale)) {
103 return Result<void>::failure(p3_scale.error());
104 }
105
106 const bool numeric_points =
107 p1_scale.has_value() && p2_scale.has_value() && p3_scale.has_value();
108 bool scale_differences = false;
109 double point_scale = 0.0;
110 if (numeric_points) {
111 point_scale = std::max(
112 p1_scale.value(), std::max(p2_scale.value(), p3_scale.value()));
113 scale_differences = plane_point_differences_overflow(
114 p1_values, p2_values, p3_values);
115 }
116 v1.reserve(3);
117 v2.reserve(3);
118 if (scale_differences) {
119 append_scaled_plane_edges(
120 p1_values, p2_values, p3_values, point_scale, v1, v2);
121 return Result<void>::success();
122 }
123 return append_symbolic_plane_edges(p1, p2, p3, v1, v2, operation);
124}

126Result<bool> scale_plane_edges_for_cross(
127 std::vector<std::shared_ptr<SymbolicExpr>>& v1,
128 std::vector<std::shared_ptr<SymbolicExpr>>& v2,
129 ComputationContext& context,
130 const std::string& operation) {
131 std::vector<double> v1_values;
132 std::vector<double> v2_values;
133 auto v1_scale = numeric_vector_scale_checked(
134 v1, context, operation, &v1_values);
135 auto v2_scale = numeric_vector_scale_checked(
136 v2, context, operation, &v2_values);
137 const bool numeric_edges = v1_scale.has_value() && v2_scale.has_value();
138 if (!numeric_edges) {
139 if (is_plane_budget_error(v1_scale) || is_plane_budget_error(v2_scale)) {
140 return Result<bool>::failure(
141 !v1_scale ? v1_scale.error() : v2_scale.error());
142 }
143 } else if (v1_scale.value() != 0.0 && v2_scale.value() != 0.0) {
144 const double safe_high =
145 std::sqrt(std::numeric_limits<double>::max() / 12.0);
146 const double safe_low =
147 std::sqrt(std::numeric_limits<double>::min());
148 const bool scale_cross =
149 v1_scale.value() > safe_high / v2_scale.value() ||
150 v1_scale.value() < safe_low / v2_scale.value();
151 if (scale_cross) {
152 for (size_t i = 0; i < 3; ++i) {
153 v1[i] = SymbolicExpr::number(
154 v1_values[i] / v1_scale.value());
155 v2[i] = SymbolicExpr::number(
156 v2_values[i] / v2_scale.value());
157 }
158 }
159 }
160 return Result<bool>::success(numeric_edges);
161}

164 const std::vector<std::shared_ptr<SymbolicExpr>>& normal,
165 bool numeric_edges,
166 ComputationContext& context,
167 const std::string& operation) {
168 Result<void> nonzero = numeric_edges

170 normal, context, operation,

172 "plane normal nonzero condition cannot be verified")
173 : Result<void>::failure(
174 CasErrc::Inconclusive,

176 operation);
177 if (!numeric_edges) {

179 if (!norm_sq) return Result<void>::failure(norm_sq.error());
180 nonzero = checked_nonzero_numeric_or_exact(
181 norm_sq.value(), context, operation,
182 "three points are collinear, so no unique plane exists",
183 "plane normal nonzero condition cannot be verified");
184 }
185 return nonzero;

188PlaneSymbolicResult assemble_plane(
189 std::vector<std::shared_ptr<SymbolicExpr>>&& normal,
190 const std::vector<std::shared_ptr<SymbolicExpr>>& point,
191 ComputationContext& context,
192 const std::string& operation) {
193 auto d = vector_dot_checked(normal, point, context);
194 if (!d) return PlaneSymbolicResult::failure(d.error());
195 auto d_simplified = simplify_checked(
196 d.value(), operation,
197 "plane constant is outside the supported domain");
198 if (!d_simplified) return PlaneSymbolicResult::failure(d_simplified.error());
199
200 PlaneSymbolic plane;
201 plane.normal = std::move(normal);
202 for (auto& component : plane.normal) {
203 auto simplified = simplify_checked(
204 component, operation,
205 "plane normal component is outside the supported domain");
206 if (!simplified) return PlaneSymbolicResult::failure(simplified.error());
207 component = std::move(simplified.value());
208 }
209 plane.d = std::move(d_simplified.value());
210 return PlaneSymbolicResult::success(std::move(plane));
211}
