194 return Result<void>::failure(CasErrc::NumericFailure,
195 inconclusive_message, operation);
196 }
197 if (numeric.value().value == 0.0) {

199 domain_message, operation);

372VectorAngleResult vector_angle_checked(
373 const std::vector<std::shared_ptr<SymbolicExpr>>& a,
374 const std::vector<std::shared_ptr<SymbolicExpr>>& b,
375 ComputationContext& context
376) {
377 const std::string operation = "vector_angle";
378 auto input = validate_same_dimension_vectors(a, b, context, operation);
379 if (!input) return VectorAngleResult::failure(input.error());
380 double norm_a = 0, norm_b = 0, dot = 0;
381 for (size_t i = 0; i < a.size(); ++i) {
382 auto step = context.consume_steps(1, operation);
383 if (!step) return VectorAngleResult::failure(step.error());
384 auto na_result = symbolic_vector_finite_numeric(a[i], context, operation);
385 if (!na_result) return VectorAngleResult::failure(na_result.error());
386 auto nb_result = symbolic_vector_finite_numeric(b[i], context, operation);
387 if (!nb_result) return VectorAngleResult::failure(nb_result.error());
388 double na = na_result.value();
389 double nb = nb_result.value();
390 norm_a += na * na;
391 norm_b += nb * nb;
392 dot += na * nb;
393 }
394 if (norm_a == 0.0 || norm_b == 0.0) {
395 return VectorAngleResult::failure(
396 CasErrc::DomainError,
397 "angle is undefined for zero-length vectors",
398 operation);
399 }
400 double cosv = dot / (std::sqrt(norm_a) * std::sqrt(norm_b));
401 // Clamp into [-1, 1] to avoid std::acos domain errors caused by rounding.
402 if (cosv > 1.0) cosv = 1.0;
403 else if (cosv < -1.0) cosv = -1.0;
404 return VectorAngleResult::success(std::acos(cosv));
405}

852VectorStringResult classify_quadric_checked(
853 const SurfaceSymbolic& surf,
854 ComputationContext& context) {
855 const std::string operation = "classify_quadric";
856 try {
857 auto valid = validate_surface_point(
858 surf,
859 std::vector<std::shared_ptr<SymbolicExpr>>(
860 surf.vars.size(), SymbolicExpr::number(0)),
861 context,
862 operation);
863 if (!valid) return VectorStringResult::failure(valid.error());
864 if (surf.vars.size() != 3) {
865 return VectorStringResult::failure(
866 CasErrc::InvalidArgument,
867 "quadric classification requires exactly three variables",
868 operation);
869 }
870 auto step = context.consume_steps(12, operation);
871 if (!step) return VectorStringResult::failure(step.error());
872
873 const auto& vars = surf.vars;
874 auto F = surf.F->expand();
875 if (!F || !lamina::detail::node(F)) F = surf.F;
876
877 auto numeric_coeff = [&](const std::shared_ptr<SymbolicExpr>& expr,
878 const std::string& message)
879 -> Result<double> {
880 auto simplified = expr ? expr->simplify() : nullptr;
881 if (!simplified || !lamina::detail::node(simplified)) {
882 return Result<double>::failure(CasErrc::Inconclusive,
883 message, operation);
884 }
885 auto value = symbolic_vector_finite_numeric(
886 simplified, context, operation);
887 if (!value) {
888 if (value.error().code == CasErrc::Cancelled ||
889 value.error().code == CasErrc::ResourceLimit) {
890 return Result<double>::failure(value.error());
891 }
892 return Result<double>::failure(CasErrc::Inconclusive,
893 message, operation);
894 }
895 return Result<double>::success(value.value());
896 };
897
898 auto coeff_of_sq = [&](const std::string& v) -> Result<double> {
899 auto d2 = F->differentiate(v);
900 if (!d2 || !lamina::detail::node(d2)) {
901 return Result<double>::failure(
902 CasErrc::Inconclusive,
903 "quadric second derivative is outside the supported domain",
904 operation);
905 }
906 d2 = d2->differentiate(v);
907 auto value = numeric_coeff(
908 d2,
909 "quadric second derivative coefficient cannot be evaluated");
910 if (!value) return value;
911 return Result<double>::success(value.value() / 2.0);
912 };
913 auto coeff_of_lin = [&](const std::string& v) -> Result<double> {
914 auto d = F->differentiate(v);
915 if (!d || !lamina::detail::node(d)) {
916 return Result<double>::failure(
917 CasErrc::Inconclusive,
918 "quadric first derivative is outside the supported domain",
919 operation);
920 }
921 for (const auto& w : vars) {
922 d = d->substitute(w, SymbolicExpr::number(0));
923 if (!d || !lamina::detail::node(d)) {
924 return Result<double>::failure(
925 CasErrc::Inconclusive,
926 "quadric linear coefficient cannot be substituted",
927 operation);
928 }
929 }
930 return numeric_coeff(
931 d,
932 "quadric linear coefficient cannot be evaluated");
933 };
934
935 auto a_result = coeff_of_sq(vars[0]);
936 if (!a_result) return VectorStringResult::failure(a_result.error());
937 auto b_result = coeff_of_sq(vars[1]);
938 if (!b_result) return VectorStringResult::failure(b_result.error());
939 auto c_result = coeff_of_sq(vars[2]);
940 if (!c_result) return VectorStringResult::failure(c_result.error());
941 auto lz_result = coeff_of_lin(vars[2]);
942 if (!lz_result) return VectorStringResult::failure(lz_result.error());
943
944 double a = a_result.value();
945 double b = b_result.value();
946 double c = c_result.value();
947 double lz = lz_result.value();
948
949 int n_sq = (a != 0) + (b != 0) + (c != 0);
950 int n_pos = (a > 0) + (b > 0) + (c > 0);
951 int n_neg = (a < 0) + (b < 0) + (c < 0);
952
953 if (n_sq == 2 && lz != 0 && c == 0) {
954 return VectorStringResult::success("paraboloid");
955 }
956 if (n_sq == 2) return VectorStringResult::success("cylinder");
957 if (n_sq == 3) {
958 if (n_neg == 0) {
959 if (a == b && b == c) {
960 return VectorStringResult::success("sphere");
961 }
962 return VectorStringResult::success("ellipsoid");
963 }
964 if (n_pos == 0) return VectorStringResult::success("ellipsoid");
965
966 auto constant = F;
967 for (const auto& w : vars) {
968 constant = constant->substitute(w, SymbolicExpr::number(0));
969 if (!constant || !lamina::detail::node(constant)) {
970 return VectorStringResult::failure(
971 CasErrc::Inconclusive,
972 "quadric constant term cannot be substituted",
973 operation);
974 }
975 }
976 auto cst_value = numeric_coeff(
977 constant,
978 "quadric constant term cannot be evaluated");
979 if (!cst_value) return VectorStringResult::failure(cst_value.error());
980 if (std::abs(cst_value.value()) < 1e-9) {
981 return VectorStringResult::success("cone");
982 }
983 return VectorStringResult::success("hyperboloid");
984 }
985 return VectorStringResult::success("unknown");
986 } catch (const std::bad_alloc&) {
987 return VectorStringResult::failure(CasErrc::ResourceLimit,
988 "vector geometry allocation failed",
989 operation);
990 } catch (const std::exception& e) {
991 return VectorStringResult::failure(CasErrc::Inconclusive,
992 e.what(), operation);
993 }
994}
