89std::optional<Rational> exact_rational_square_root(
90 const Rational& value) {
91 if (value < Rational(0)) return std::nullopt;
92 const BigInt numerator = value.get_numerator();
93 const BigInt denominator = value.get_denominator();
94 const BigInt numerator_root = numerator.sqrt();
95 const BigInt denominator_root = denominator.sqrt();
96 if (numerator_root * numerator_root != numerator ||
97 denominator_root * denominator_root != denominator) {
98 return std::nullopt;
99 }
100 return Rational(numerator_root, denominator_root);
101}

103std::vector<Poly> factor_biquadratic(const Poly& polynomial) {
104 if (polynomial.degree() != 4 ||
105 polynomial.coeffs[1] != Rational(0) ||
106 polynomial.coeffs[3] != Rational(0)) {
107 return {};
108 }
109 const Rational& c = polynomial.coeffs[0];
110 const Rational& b = polynomial.coeffs[2];
111 const Rational& a = polynomial.coeffs[4];
112 const Rational discriminant = b * b - Rational(4) * a * c;
113 auto square_root = exact_rational_square_root(discriminant);
114 if (!square_root) return {};
115 const Rational first =
116 (-b - *square_root) / (Rational(2) * a);
117 const Rational second =
118 (-b + *square_root) / (Rational(2) * a);
119 if (first == second) return {};
120 return {
121 Poly({-first, Rational(0), Rational(1)}, polynomial.variable_name),
122 Poly({-second, Rational(0), Rational(1)}, polynomial.variable_name)};
123}

125Result<Poly> exact_poly_divide(
126 const Poly& numerator,
127 const Poly& denominator,
128 ComputationContext& context,
129 const std::string& operation) {
130 auto step = context.consume_steps(1, operation);
131 if (!step) return Result<Poly>::failure(step.error());
132 if (denominator.is_zero()) {
133 return Result<Poly>::failure(
134 CasErrc::InternalInvariant,
135 "fraction-free resultant encountered a zero divisor",
136 operation);
137 }
138 auto division = numerator.div_mod(denominator);
139 if (!division.second.is_zero()) {
140 return Result<Poly>::failure(
141 CasErrc::InternalInvariant,
142 "fraction-free resultant division was not exact",
143 operation);
144 }
145 return Result<Poly>::success(std::move(division.first));
146}

178ComplexBivariate complex_bivariate_parts(const Poly& polynomial) {
179 BiPolynomial power_real;
180 power_real.add(0, 0, Rational(1));
181 BiPolynomial power_imaginary;
182 ComplexBivariate result;
183
184 for (std::size_t degree = 0; degree < polynomial.coeffs.size(); ++degree) {
185 add_scaled(result.real, power_real, polynomial.coeffs[degree]);
186 add_scaled(result.imaginary, power_imaginary,
187 polynomial.coeffs[degree]);
188
189 BiPolynomial next_real = shift_x(power_real);
190 add_scaled(next_real, shift_y(power_imaginary), Rational(-1));
191 BiPolynomial next_imaginary = shift_x(power_imaginary);
192 add_scaled(next_imaginary, shift_y(power_real), Rational(1));
193 power_real = std::move(next_real);
194 power_imaginary = std::move(next_imaginary);
195 }
196 return result;
197}

199std::vector<Poly> coefficients_in_y(
200 const BiPolynomial& polynomial,
201 const std::string& remaining_variable) {
202 std::size_t maximum_y = 0;
203 for (const auto& x_coefficients : polynomial.coefficients) {
204 if (!x_coefficients.empty()) {
205 maximum_y = std::max(maximum_y, x_coefficients.size() - 1);
206 }
207 }
208 std::vector<Poly> result(maximum_y + 1, Poly(remaining_variable));
209 for (std::size_t x = 0; x < polynomial.coefficients.size(); ++x) {
210 for (std::size_t y = 0; y < polynomial.coefficients[x].size(); ++y) {
211 if (result[y].coeffs.size() <= x) {
212 result[y].coeffs.resize(x + 1, Rational(0));
213 }
214 result[y].coeffs[x] = polynomial.coefficients[x][y];
215 }
216 }
217 while (result.size() > 1 && result.back().is_zero()) result.pop_back();
218 return result;
219}

241Result<Poly> sylvester_resultant(
242 std::vector<Poly> left,
243 std::vector<Poly> right,
244 const std::string& variable,
245 ComputationContext& context,
246 const std::string& operation) {
247 while (left.size() > 1 && left.back().is_zero()) left.pop_back();
248 while (right.size() > 1 && right.back().is_zero()) right.pop_back();
249 if (left.empty() || right.empty() ||
250 (left.size() == 1 && left[0].is_zero()) ||
251 (right.size() == 1 && right[0].is_zero())) {
252 return Result<Poly>::success(Poly(variable));
253 }
254
255 const std::size_t left_degree = left.size() - 1;
256 const std::size_t right_degree = right.size() - 1;
257 const std::size_t size = left_degree + right_degree;
258 if (size == 0) {
259 return Result<Poly>::success(constant_poly(Rational(1), variable));
260 }
261
262 auto step = context.consume_steps(size * size + 1, operation);
263 if (!step) return Result<Poly>::failure(step.error());
264 PolyMatrix matrix(size, std::vector<Poly>(size, Poly(variable)));
265 for (std::size_t row = 0; row < right_degree; ++row) {
266 for (std::size_t column = 0; column <= left_degree; ++column) {
267 matrix[row][row + column] = left[left_degree - column];
268 }
269 }
270 for (std::size_t row = 0; row < left_degree; ++row) {
271 for (std::size_t column = 0; column <= right_degree; ++column) {
272 matrix[right_degree + row][row + column] =
273 right[right_degree - column];
274 }
275 }
276
277 Poly previous = constant_poly(Rational(1), variable);
278 int determinant_sign = 1;
279 for (std::size_t pivot = 0; pivot + 1 < size; ++pivot) {
280 step = context.consume_steps(1, operation);
281 if (!step) return Result<Poly>::failure(step.error());
282 std::size_t replacement = pivot;
283 while (replacement < size && matrix[replacement][pivot].is_zero()) {
284 ++replacement;
285 }
286 if (replacement == size) return Result<Poly>::success(Poly(variable));
287 if (replacement != pivot) {
288 std::swap(matrix[pivot], matrix[replacement]);
289 determinant_sign = -determinant_sign;
290 }
291 const Poly pivot_value = matrix[pivot][pivot];
292 for (std::size_t row = pivot + 1; row < size; ++row) {
293 for (std::size_t column = pivot + 1; column < size; ++column) {
294 Poly numerator = matrix[row][column] * pivot_value -
295 matrix[row][pivot] * matrix[pivot][column];
296 if (pivot != 0) {
297 auto quotient = exact_poly_divide(
298 numerator, previous, context, operation);
299 if (!quotient) return quotient;
300 matrix[row][column] = std::move(quotient.value());
301 } else {
302 matrix[row][column] = std::move(numerator);
303 }
304 }
305 matrix[row][pivot] = Poly(variable);
306 }
307 previous = pivot_value;
308 }
309
310 Poly determinant = matrix[size - 1][size - 1];
311 if (determinant_sign < 0) {
312 determinant = scale_poly(std::move(determinant), Rational(-1));
313 }
314 determinant.trim();
315 return Result<Poly>::success(std::move(determinant));
316}

318std::pair<Poly, Poly> edge_image(
319 const Poly& polynomial,
320 const Rational& real_start,
321 const Rational& imaginary_start,
322 const Rational& real_delta,
323 const Rational& imaginary_delta) {
324 const Poly real_line(
325 std::vector<Rational>{real_start, real_delta}, "t");
326 const Poly imaginary_line(
327 std::vector<Rational>{imaginary_start, imaginary_delta}, "t");
328 Poly real("t");
329 Poly imaginary("t");
330 for (std::size_t position = polynomial.coeffs.size(); position-- > 0;) {
331 Poly next_real = real * real_line - imaginary * imaginary_line;
332 Poly next_imaginary = real * imaginary_line + imaginary * real_line;
333 if (next_real.coeffs.empty()) next_real.coeffs.resize(1, Rational(0));
334 next_real.coeffs[0] = next_real.coeffs[0] + polynomial.coeffs[position];
335 next_real.trim();
336 real = std::move(next_real);
337 imaginary = std::move(next_imaginary);
338 }
339 return {std::move(real), std::move(imaginary)};
340}

343 const Poly& polynomial,
344 const RationalRectangle& rectangle) {
345 const Rational width = rectangle.real_upper - rectangle.real_lower;
346 const Rational height =
347 rectangle.imaginary_upper - rectangle.imaginary_lower;
348 std::vector<std::pair<Poly, Poly>> edges;
349 edges.reserve(4);
350 edges.push_back(edge_image(

353 edges.push_back(edge_image(
354 polynomial, rectangle.real_upper, rectangle.imaginary_lower,

360 polynomial, rectangle.real_lower, rectangle.imaginary_upper,
361 Rational(0), -height));

365Result<bool> edge_is_clear(
366 const Poly& real,
367 const Poly& imaginary,
368 ComputationContext& context,
369 const std::string& operation) {
370 if (real.eval(Rational(0)) == Rational(0) &&
371 imaginary.eval(Rational(0)) == Rational(0)) {
372 return Result<bool>::success(false);
373 }
374 if (real.eval(Rational(1)) == Rational(0) &&
375 imaginary.eval(Rational(1)) == Rational(0)) {
376 return Result<bool>::success(false);
377 }
378 Poly common = Poly::gcd(real, imaginary);
379 if (common.degree() <= 0) return Result<bool>::success(true);
380 auto roots = count_real_roots_exact(
381 common, Rational(0), Rational(1), context, operation);
382 if (!roots) return Result<bool>::failure(roots.error());
383 return Result<bool>::success(roots.value() == 0);
384}

386Result<void> refine_interval(
387 const Poly& polynomial,
388 RationalInterval& interval,
389 ComputationContext& context,
390 const std::string& operation) {
391 auto step = context.consume_steps(1, operation);
392 if (!step) return step;
393 if (interval.first == interval.second) return Result<void>::success();
394 const Rational midpoint =
395 (interval.first + interval.second) / Rational(2);
396 const Rational middle_value = polynomial.eval(midpoint);
397 if (middle_value == Rational(0)) {
398 interval = {midpoint, midpoint};
399 return Result<void>::success();
400 }
401 const int lower_sign = rational_sign(polynomial.eval(interval.first));
402 if (lower_sign == 0) {
403 interval.second = interval.first;
404 } else if (lower_sign != rational_sign(middle_value)) {
405 interval.second = midpoint;
406 } else {
407 interval.first = midpoint;
408 }
409 return Result<void>::success();
410}

412Result<int> cauchy_index_on_edge(
413 const Poly& denominator,
414 const Poly& numerator,
415 ComputationContext& context,
416 const std::string& operation) {
417 int index = 0;
418 const auto factors = square_free_factorization(denominator);
419 for (const auto& [factor, multiplicity] : factors) {
420 if ((multiplicity & 1) == 0) continue;
421 auto roots = isolate_real_roots_exact(factor, context, operation);
422 if (!roots) return Result<int>::failure(roots.error());
423 for (auto interval : roots.value()) {
424 while (true) {
425 auto step = context.consume_steps(1, operation);
426 if (!step) return Result<int>::failure(step.error());
427 if (interval.first == interval.second) break;
428 if (interval.second <= Rational(0) ||
429 interval.first >= Rational(1)) {
430 break;
431 }
432 if (interval.first > Rational(0) &&
433 interval.second < Rational(1)) {
434 auto denominator_roots = count_real_roots_exact(
435 denominator, interval.first, interval.second,
436 context, operation);
437 if (!denominator_roots) {
438 return Result<int>::failure(denominator_roots.error());
439 }
440 auto numerator_roots = count_real_roots_exact(
441 numerator, interval.first, interval.second,
442 context, operation);
443 if (!numerator_roots) {
444 return Result<int>::failure(numerator_roots.error());
445 }
446 if (denominator_roots.value() == 1 &&
447 numerator_roots.value() == 0 &&
448 denominator.eval(interval.first) != Rational(0)) {
449 break;
450 }
451 }
452 auto refined = refine_interval(
453 factor, interval, context, operation);
454 if (!refined) return Result<int>::failure(refined.error());
455 }
456
457 if (interval.second <= Rational(0) ||
458 interval.first >= Rational(1)) {
459 continue;
460 }
461
462 Rational numerator_sample;
463 Rational left_sample;
464 if (interval.first == interval.second) {
465 const Rational root = interval.first;
466 if (!(root > Rational(0) && root < Rational(1))) continue;
467 numerator_sample = root;
468 left_sample = root / Rational(2);
469 while (true) {
470 auto step = context.consume_steps(1, operation);
471 if (!step) return Result<int>::failure(step.error());
472 auto count = count_real_roots_exact(
473 denominator, left_sample, root, context, operation);
474 if (!count) return Result<int>::failure(count.error());
475 if (count.value() == 1 &&
476 denominator.eval(left_sample) != Rational(0)) {
477 break;
478 }
479 left_sample = (left_sample + root) / Rational(2);
480 }
481 } else {
482 numerator_sample =
483 (interval.first + interval.second) / Rational(2);
484 left_sample = interval.first;
485 }
486
487 const int denominator_left_sign =
488 rational_sign(denominator.eval(left_sample));
489 const int numerator_sign =
490 rational_sign(numerator.eval(numerator_sample));
491 if (denominator_left_sign == 0 || numerator_sign == 0) {
492 return Result<int>::failure(
493 CasErrc::InternalInvariant,
494 "Cauchy-index pole sign could not be certified",
495 operation);
496 }
497 index += denominator_left_sign * numerator_sign;
498 }
499 }
500 return Result<int>::success(index);
501}

503Result<RectangleCount> count_rectangle_roots(
504 const Poly& polynomial,
505 const RationalRectangle& rectangle,
506 ComputationContext& context,
507 const std::string& operation) {
508 auto step = context.consume_steps(1, operation);
509 if (!step) return Result<RectangleCount>::failure(step.error());
510 const auto edges = rectangle_edges(polynomial, rectangle);
511 for (const auto& [real, imaginary] : edges) {
512 auto clear = edge_is_clear(real, imaginary, context, operation);
513 if (!clear) return Result<RectangleCount>::failure(clear.error());
514 if (!clear.value()) {
515 return Result<RectangleCount>::success(RectangleCount{false, 0});
516 }
517 }
518
519 long long shear_magnitude = 0;
520 int shear_direction = 1;
521 while (true) {
522 step = context.consume_steps(1, operation);
523 if (!step) return Result<RectangleCount>::failure(step.error());
524 const long long shear_value = shear_magnitude * shear_direction;
525 const Rational shear(shear_value);
526 bool valid = true;
527 for (const auto& [real, imaginary] : edges) {
528 const Poly denominator = real + scale_poly(imaginary, shear);
529 if (denominator.is_zero() ||
530 denominator.eval(Rational(0)) == Rational(0) ||
531 denominator.eval(Rational(1)) == Rational(0)) {
532 valid = false;
533 break;
534 }
535 }
536 if (valid) {
537 int total_index = 0;
538 for (const auto& [real, imaginary] : edges) {
539 const Poly denominator = real + scale_poly(imaginary, shear);
540 auto edge_index = cauchy_index_on_edge(
541 denominator, imaginary, context, operation);
542 if (!edge_index) {
543 return Result<RectangleCount>::failure(edge_index.error());
544 }
545 total_index += edge_index.value();
546 }
547 if ((total_index & 1) != 0) {
548 return Result<RectangleCount>::failure(
549 CasErrc::InternalInvariant,
550 "argument-principle Cauchy index was odd",
551 operation);
552 }
553 const int count = total_index / 2;
554 if (count < 0 || count > polynomial.degree()) {
555 return Result<RectangleCount>::failure(
556 CasErrc::InternalInvariant,
557 "argument-principle root count was outside polynomial degree",
558 operation);
559 }
560 return Result<RectangleCount>::success(
561 RectangleCount{true, static_cast<std::size_t>(count)});
562 }
563 if (shear_direction > 0) {
564 shear_direction = -1;
565 if (shear_magnitude == 0) shear_magnitude = 1;
566 } else {
567 shear_direction = 1;
568 ++shear_magnitude;
569 }
570 }
571}

573Result<void> separate_projection_intervals(
574 const Poly& projection,
575 std::vector<RationalInterval>& intervals,
576 ComputationContext& context,
577 const std::string& operation) {
578 if (intervals.size() < 2) return Result<void>::success();
579 bool separated = false;
580 std::size_t round = 0;
581 while (!separated) {
582 separated = true;
583 for (std::size_t index = 1; index < intervals.size(); ++index) {
584 if (intervals[index - 1].second < intervals[index].first) {
585 continue;
586 }
587 separated = false;
588 for (const std::size_t target : {index - 1, index}) {
589 if (intervals[target].first == intervals[target].second) {
590 continue;
591 }
592 if (round < 8) {
593 auto refined = refine_interval(
594 projection, intervals[target], context, operation);
595 if (!refined) return refined;
596 } else {
597 const double width =
598 (intervals[target].second -
599 intervals[target].first).to_double();
600 ExactRealAlgebraic value{
601 projection,
602 intervals[target].first,
603 intervals[target].second,
604 target,
605 1};
606 auto refined =
607 refine_exact_real_algebraic_to_tolerance(
608 value, width / 16.0, 1e-15,
609 context, operation);
610 if (!refined) return refined;
611 intervals[target] = {value.lower, value.upper};
612 }
613 }
614 }
615 ++round;
616 }
617 return Result<void>::success();
618}

620std::vector<RationalInterval> projection_bands(
621 const std::vector<RationalInterval>& intervals,
622 const Rational& bound) {
623 std::vector<RationalInterval> bands;
624 bands.reserve(intervals.size());
625 for (std::size_t index = 0; index < intervals.size(); ++index) {
626 const Rational lower = index == 0
627 ? -bound
628 : (intervals[index - 1].second + intervals[index].first) /
629 Rational(2);
630 const Rational upper = index + 1 == intervals.size()
631 ? bound
632 : (intervals[index].second + intervals[index + 1].first) /
633 Rational(2);
634 bands.emplace_back(lower, upper);
635 }
636 return bands;
637}

640Rational strict_root_bound(const Poly& polynomial) {
641 Rational maximum(0);
642 const Rational leading = polynomial.lead_coeff().abs();
643 for (int degree = 0; degree < polynomial.degree(); ++degree) {
644 const Rational ratio = polynomial.coeffs[degree].abs() / leading;
645 if (maximum < ratio) maximum = ratio;
646 }
647 return Rational(2) + maximum;

650Result<int> compare_isolations(
651 const RootIsolation& left,
652 const RootIsolation& right,
653 ComputationContext& context) {
654 const auto* left_real = std::get_if<RealIsolation>(&left);
655 const auto* right_real = std::get_if<RealIsolation>(&right);
656 if (left_real && !right_real) return Result<int>::success(-1);
657 if (!left_real && right_real) return Result<int>::success(1);
658 if (left_real && right_real) {
659 return compare_exact_real_algebraic(
660 left_real->value, right_real->value, context);
661 }
662 const auto& left_complex = std::get<ComplexIsolation>(left);
663 const auto& right_complex = std::get<ComplexIsolation>(right);
664 auto real_comparison = compare_exact_real_algebraic(
665 left_complex.real_projection,
666 right_complex.real_projection,
667 context);
668 if (!real_comparison || real_comparison.value() != 0) {
669 return real_comparison;
670 }
671 return compare_exact_real_algebraic(
672 left_complex.imaginary_projection,
673 right_complex.imaginary_projection,
674 context);
675}

691Result<std::vector<RootIsolation>> isolate_exact_roots(
692 const Polynomial<Rational>& input,
693 ComputationContext& context,
694 const std::string& operation) {
695 const std::size_t starting_steps = context.steps_used();
696 try {
697 auto step = context.consume_steps(1, operation);
698 if (!step) return Result<std::vector<RootIsolation>>::failure(step.error());
699 if (input.degree() <= 0) {
700 return Result<std::vector<RootIsolation>>::failure(
701 CasErrc::InvalidArgument,
702 "exact root isolation requires a nonconstant polynomial",
703 operation);
704 }
705 Poly polynomial = input.square_free_part().make_monic();
706 polynomial.variable_name = "_root";
707 for (const auto& entry : root_isolation_cache) {
708 if (!same_polynomial(entry.polynomial, polynomial)) continue;
709 const std::size_t remaining_cost =
710 entry.step_cost > 0 ? entry.step_cost - 1 : 0;
711 auto cached_step = context.consume_steps(
712 remaining_cost, operation + ".cache");
713 if (!cached_step) {
714 return Result<std::vector<RootIsolation>>::failure(
715 cached_step.error());
716 }
717 return Result<std::vector<RootIsolation>>::success(entry.roots);
718 }
719 auto remember = [&](std::vector<RootIsolation> roots) {
720 if (root_isolation_cache.size() >= 16) {
721 root_isolation_cache.erase(root_isolation_cache.begin());
722 }
723 root_isolation_cache.push_back(RootIsolationCacheEntry{
724 polynomial, roots, context.steps_used() - starting_steps});
725 return Result<std::vector<RootIsolation>>::success(
726 std::move(roots));
727 };
728
729 auto factors = factor_biquadratic(polynomial);
730 if (factors.empty() && polynomial.degree() >= 6) {
731 auto factor_result =
732 lamina::factor_univariate_bridge_checked(polynomial, context);
733 if (!factor_result) {
734 return Result<std::vector<RootIsolation>>::failure(
735 factor_result.error());
736 }
737 factors = std::move(factor_result.value().value);
738 }
739 if (factors.empty()) factors = {polynomial};
740 std::size_t factored_degree = 0;
741 bool strict_factorization = factors.size() > 1;
742 for (auto& factor : factors) {
743 factor = factor.square_free_part().make_monic();
744 factor.variable_name = "_root";
745 if (factor.degree() <= 0 ||
746 factor.degree() >= polynomial.degree()) {
747 strict_factorization = false;
748 break;
749 }
750 factored_degree += static_cast<std::size_t>(factor.degree());
751 }
752 if (strict_factorization &&
753 factored_degree ==
754 static_cast<std::size_t>(polynomial.degree())) {
755 std::vector<RootIsolation> combined;
756 combined.reserve(factored_degree);
757 for (const auto& factor : factors) {
758 auto factor_roots = isolate_exact_roots(
759 factor, context, operation + ".factor");
760 if (!factor_roots) {
761 return Result<std::vector<RootIsolation>>::failure(
762 factor_roots.error());
763 }
764 combined.insert(
765 combined.end(),
766 std::make_move_iterator(factor_roots.value().begin()),
767 std::make_move_iterator(factor_roots.value().end()));
768 }
769 for (std::size_t index = 1; index < combined.size(); ++index) {
770 RootIsolation current = std::move(combined[index]);
771 std::size_t position = index;
772 while (position > 0) {
773 auto comparison = compare_isolations(
774 combined[position - 1], current, context);
775 if (!comparison) {
776 return Result<std::vector<RootIsolation>>::failure(
777 comparison.error());
778 }
779 if (comparison.value() <= 0) break;
780 combined[position] = std::move(combined[position - 1]);
781 --position;
782 }
783 combined[position] = std::move(current);
784 }
785 return remember(std::move(combined));
786 }
787
788 auto real_roots = isolate_real_roots_exact(
789 polynomial, context, operation + ".real");
790 if (!real_roots) {
791 return Result<std::vector<RootIsolation>>::failure(
792 real_roots.error());
793 }
794 const std::size_t real_count = real_roots.value().size();
795 const std::size_t degree = static_cast<std::size_t>(polynomial.degree());
796 if (real_count > degree) {
797 return Result<std::vector<RootIsolation>>::failure(
798 CasErrc::InternalInvariant,
799 "real-root count exceeded polynomial degree",
800 operation);
801 }
802
803 std::vector<RootIsolation> result;
804 result.reserve(degree);
805 for (std::size_t index = 0; index < real_count; ++index) {
806 result.push_back(RealIsolation{ExactRealAlgebraic{
807 polynomial, real_roots.value()[index].first,
808 real_roots.value()[index].second, index, 1}});
809 }
810 if (real_count == degree) {
811 return remember(std::move(result));
812 }
813
814 const Poly certificate_polynomial = clear_denominators(polynomial);
815 const ComplexBivariate parts =
816 complex_bivariate_parts(certificate_polynomial);
817 auto real_resultant = sylvester_resultant(
818 coefficients_in_y(parts.real, "_re"),
819 coefficients_in_y(parts.imaginary, "_re"),
820 "_re", context, operation + ".resultant.real");
821 if (!real_resultant) {
822 return Result<std::vector<RootIsolation>>::failure(
823 real_resultant.error());
824 }
825 auto imaginary_resultant = sylvester_resultant(
826 coefficients_in_x(parts.real, "_im"),
827 coefficients_in_x(parts.imaginary, "_im"),
828 "_im", context, operation + ".resultant.imaginary");
829 if (!imaginary_resultant) {
830 return Result<std::vector<RootIsolation>>::failure(
831 imaginary_resultant.error());
832 }
833 if (real_resultant.value().degree() <= 0 ||
834 imaginary_resultant.value().degree() <= 0) {
835 return Result<std::vector<RootIsolation>>::failure(
836 CasErrc::InternalInvariant,
837 "complex coordinate resultant was identically zero",
838 operation);
839 }
840 Poly real_projection =
841 real_resultant.value().square_free_part().make_monic();
842 Poly imaginary_projection =
843 imaginary_resultant.value().square_free_part().make_monic();
844 auto real_projection_roots = isolate_real_roots_exact(
845 real_projection, context, operation + ".projection.real");
846 if (!real_projection_roots) {
847 return Result<std::vector<RootIsolation>>::failure(
848 real_projection_roots.error());
849 }
850 auto imaginary_projection_roots = isolate_real_roots_exact(
851 imaginary_projection, context, operation + ".projection.imaginary");
852 if (!imaginary_projection_roots) {
853 return Result<std::vector<RootIsolation>>::failure(
854 imaginary_projection_roots.error());
855 }
856 auto real_separated = separate_projection_intervals(
857 real_projection, real_projection_roots.value(), context,
858 operation + ".projection.real");
859 if (!real_separated) {
860 return Result<std::vector<RootIsolation>>::failure(
861 real_separated.error());
862 }
863 auto imaginary_separated = separate_projection_intervals(
864 imaginary_projection, imaginary_projection_roots.value(),
865 context, operation + ".projection.imaginary");
866 if (!imaginary_separated) {
867 return Result<std::vector<RootIsolation>>::failure(
868 imaginary_separated.error());
869 }
870
871 bool have_positive = false;
872 bool have_negative = false;
873 Rational positive_lower;
874 Rational negative_upper;
875 for (auto& interval : imaginary_projection_roots.value()) {
876 while (interval.first <= Rational(0) &&
877 interval.second >= Rational(0) &&
878 !(interval.first == Rational(0) &&
879 interval.second == Rational(0))) {
880 if (imaginary_projection.eval(Rational(0)) == Rational(0)) {
881 interval = {Rational(0), Rational(0)};
882 break;
883 }
884 auto refined = refine_interval(
885 imaginary_projection, interval, context, operation);
886 if (!refined) {
887 return Result<std::vector<RootIsolation>>::failure(
888 refined.error());
889 }
890 }
891 if (interval.first > Rational(0) &&
892 (!have_positive || interval.first < positive_lower)) {
893 positive_lower = interval.first;
894 have_positive = true;
895 }
896 if (interval.second < Rational(0) &&
897 (!have_negative || negative_upper < interval.second)) {
898 negative_upper = interval.second;
899 have_negative = true;
900 }
901 }
902
903 const std::size_t complex_count = degree - real_count;
904 if (!have_positive || !have_negative || (complex_count & 1U) != 0) {
905 return Result<std::vector<RootIsolation>>::failure(
906 CasErrc::InternalInvariant,
907 "non-real roots lacked conjugate imaginary projections",
908 operation);
909 }
910 const Rational bound = strict_root_bound(certificate_polynomial);
911 const auto real_bands = projection_bands(
912 real_projection_roots.value(), bound);
913 const auto imaginary_bands = projection_bands(
914 imaginary_projection_roots.value(), bound);
915 std::vector<ComplexIsolation> complex_roots;
916 complex_roots.reserve(complex_count);
917 for (std::size_t real_index = 0;
918 real_index < real_bands.size(); ++real_index) {
919 for (std::size_t imaginary_index = 0;
920 imaginary_index < imaginary_bands.size(); ++imaginary_index) {
921 const auto& imaginary_interval =
922 imaginary_projection_roots.value()[imaginary_index];
923 if (imaginary_projection.eval(Rational(0)) == Rational(0) &&
924 imaginary_interval.first <= Rational(0) &&
925 imaginary_interval.second >= Rational(0)) {
926 continue;
927 }
928 const RationalRectangle rectangle{
929 real_bands[real_index].first,
930 real_bands[real_index].second,
931 imaginary_bands[imaginary_index].first,
932 imaginary_bands[imaginary_index].second};
933 auto count = count_rectangle_roots(
934 certificate_polynomial, rectangle, context,
935 operation + ".rectangles.projection_grid");
936 if (!count) {
937 return Result<std::vector<RootIsolation>>::failure(
938 count.error());
939 }
940 if (!count.value().boundary_clear ||
941 count.value().count > 1) {
942 return Result<std::vector<RootIsolation>>::failure(
943 CasErrc::InternalInvariant,
944 "projection-band rectangle was not a clear one-root cell",
945 operation);
946 }
947 if (count.value().count == 0) continue;
948 complex_roots.push_back(ComplexIsolation{
949 rectangle.real_lower,
950 rectangle.real_upper,
951 rectangle.imaginary_lower,
952 rectangle.imaginary_upper,
953 ExactRealAlgebraic{
954 real_projection,
955 real_projection_roots.value()[real_index].first,
956 real_projection_roots.value()[real_index].second,
957 real_index, 1},
958 ExactRealAlgebraic{
959 imaginary_projection,
960 imaginary_interval.first,
961 imaginary_interval.second,
962 imaginary_index, 1}});
963 }
964 }
965 if (complex_roots.size() != complex_count) {
966 return Result<std::vector<RootIsolation>>::failure(
967 CasErrc::InternalInvariant,
968 "projection-band rectangles did not cover every non-real root",
969 operation);
970 }
971 std::sort(complex_roots.begin(), complex_roots.end(),
972 [](const ComplexIsolation& left,
973 const ComplexIsolation& right) {
974 if (left.real_projection.root_index !=
975 right.real_projection.root_index) {
976 return left.real_projection.root_index <
977 right.real_projection.root_index;
978 }
979 return left.imaginary_projection.root_index <
980 right.imaginary_projection.root_index;
981 });
982 for (auto& isolation : complex_roots) {
983 result.push_back(std::move(isolation));
984 }
985 if (result.size() != degree) {
986 return Result<std::vector<RootIsolation>>::failure(
987 CasErrc::InternalInvariant,
988 "certified root isolation did not cover the polynomial degree",
989 operation);
990 }
991 return remember(std::move(result));
992 } catch (const std::bad_alloc&) {
993 return Result<std::vector<RootIsolation>>::failure(
994 CasErrc::ResourceLimit,
995 "complex root isolation allocation failed",
996 operation);
997 } catch (const std::exception& error) {
998 return Result<std::vector<RootIsolation>>::failure(
999 CasErrc::InternalInvariant, error.what(), operation);
1000 }
1001}
