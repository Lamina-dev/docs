26Result<SymbolicExpr> equal_endpoint_integral(const SymbolicExpr& expr,
27 const std::string& var_name, const SymbolicExpr& lower, const SymbolicExpr& upper,
28 const DefiniteInterval& interval, ComputationContext& context) {
29 if (interval.a.value().infinity) { return unresolved_integral(expr, var_name, lower, upper); }
30 auto assumptions = real_assumptions(var_name, context);
31 if (!assumptions) { return Result<SymbolicExpr>::failure(assumptions.error()); }
32 detail::AssumptionFacts facts(assumptions.value());
33 auto value = expr.substitute(var_name, detail::make_expression_ptr(interval.lo));
34 auto defined = detail::query_definedness(detail::node(value), facts, Domain::Real, context);
35 if (!defined) { return Result<SymbolicExpr>::failure(defined.error()); }
36 if (defined.value() == Tribool::False) { return divergent<SymbolicExpr>(); }
37 return defined.value() == Tribool::True ? Result<SymbolicExpr>(*SymbolicExpr::number(0)) : unresolved_integral(expr, var_name, lower, upper);
38}

83Result<SymbolicExpr> accumulate_segments(Integrator& integrator, const SymbolicExpr& expr,
84 const SymbolicExpr& value_expression, const std::string& var_name,
85 const std::vector<SymbolicExpr>& partition, const std::optional<SymbolicExpr>& primitive,
86 bool reversed, ComputationContext& context) {
87 SymbolicExpr sum = *SymbolicExpr::number(0);
88 int infinity = 0;
89 for (std::size_t i = 1; i < partition.size(); ++i) {
90 std::optional<SymbolicExpr> local_primitive;
91 if (!primitive) {
92 auto integrated = integrate_segment_primitive(integrator, value_expression, var_name,
93 partition[i - 1], partition[i], context);
94 if (!integrated) { return integrated; }
95 local_primitive = std::move(integrated.value());
96 }
97 auto segment = definite_segment_checked(expr, primitive ? *primitive : *local_primitive, var_name,
98 partition[i - 1], partition[i], context);
99 if (!segment) { return segment; }
100 int contribution = infinity_sign(detail::node(segment.value()));
101 if (contribution) {
102 if (infinity && infinity != contribution) { return divergent<SymbolicExpr>(); }
103 infinity = contribution;
104 } else { sum = *SymbolicExpr::add(detail::make_expression_ptr(sum), detail::make_expression_ptr(segment.value())); }
105 }
106 if (infinity) { return *SymbolicExpr::infinity(infinity * (reversed ? -1 : 1)); }
107 if (reversed) { sum = *SymbolicExpr::multiply(SymbolicExpr::number(-1), detail::make_expression_ptr(sum)); }
108 return *sum.simplify();
109}

111Result<SymbolicExpr> integrate_ordered_interval(Integrator& integrator, const SymbolicExpr& expr,
112 const std::string& var_name, const SymbolicExpr& lower, const SymbolicExpr& upper,
113 const DefiniteInterval& interval, ComputationContext& context) {
114 const auto unresolved = [&]() { return unresolved_integral(expr, var_name, lower, upper); };
115 auto partition = definite_partition_checked(expr, var_name, interval.lo, interval.hi, context);
116 if (!partition) { return is_undecided(partition.error()) ? unresolved() : Result<SymbolicExpr>::failure(partition.error()); }
117 auto continuous = certify_partition(expr, var_name, partition.value(), context);
118 if (!continuous) { return Result<SymbolicExpr>::failure(continuous.error()); }
119 if (continuous.value() == Tribool::False) { return divergent<SymbolicExpr>(); }
120 if (continuous.value() != Tribool::True) { return unresolved(); }
121 auto reduced = reduced_integrand_checked(expr, var_name, interval, context);
122 if (!reduced) { return Result<SymbolicExpr>::failure(reduced.error()); }
123 if (reduced.value().divergence) {
124 return *SymbolicExpr::infinity(reduced.value().divergence * (interval.reversed ? -1 : 1));
125 }
126 const auto& value_expression = reduced.value().expression;
127 std::optional<SymbolicExpr> primitive;
128 if (!needs_segment_expression(detail::node(value_expression))) {
129 auto integrated = integrator.integrate_checked(value_expression, var_name, context);
130 if (!integrated) { return is_undecided(integrated.error()) ? unresolved() : integrated; }
131 if (contains_unevaluated_integral(detail::node(integrated.value()))) { return unresolved(); }
132 primitive = std::move(integrated.value());
133 }
134 auto sum = accumulate_segments(integrator, expr, value_expression, var_name,
135 partition.value(), primitive, interval.reversed, context);
136 if (!sum && is_undecided(sum.error())) { return unresolved(); }
137 return sum;
138}

140Result<SymbolicExpr> integrate_interval_checked(Integrator& integrator, const SymbolicExpr& expr,
141 const std::string& var_name, const SymbolicExpr& lower, const SymbolicExpr& upper,
142 ComputationContext& context) {
143 DefiniteInterval interval(lower, upper, context);
144 if (!interval.a && !is_undecided(interval.a.error())) { return Result<SymbolicExpr>::failure(interval.a.error()); }
145 if (!interval.b && !is_undecided(interval.b.error())) { return Result<SymbolicExpr>::failure(interval.b.error()); }
146 if (interval.a && interval.b) {
147 auto order = compare_boundaries(interval.a.value(), interval.b.value(), context);
148 if (!order) { return Result<SymbolicExpr>::failure(order.error()); }
149 if (!order.value()) {
150 return equal_endpoint_integral(expr, var_name, lower, upper, interval, context);
151 }
152 if (order.value() > 0) {
153 std::swap(interval.lo, interval.hi);
154 std::swap(interval.a.value(), interval.b.value());
155 interval.reversed = true;
156 }
157 }
158 return integrate_ordered_interval(integrator, expr, var_name, lower, upper, interval, context);
159}

162Result<SymbolicExpr> Integrator::integrate_def_checked(const SymbolicExpr& expr,
163 const std::string& var_name, const SymbolicExpr& lower, const SymbolicExpr& upper,
164 ComputationContext& context) {
165 auto access = context.consume_steps(1, operation);
166 if (!access) { return Result<SymbolicExpr>::failure(access.error()); }
167 if (var_name.empty() || depends_on_integration_variable(lower, var_name) ||
168 depends_on_integration_variable(upper, var_name))
169 { return Result<SymbolicExpr>::failure(CasErrc::InvalidArgument,
170 "integration endpoints must be independent of the integration variable", operation); }
171 try {
172 return integrate_interval_checked(*this, expr, var_name, lower, upper, context);
173 } catch (const std::bad_alloc&) {
174 return Result<SymbolicExpr>::failure(CasErrc::ResourceLimit,
175 "definite integration allocation failed", operation);
176 } catch (const std::exception& error) {
177 return Result<SymbolicExpr>::failure(CasErrc::InternalInvariant, error.what(), operation);
178 }
179}
