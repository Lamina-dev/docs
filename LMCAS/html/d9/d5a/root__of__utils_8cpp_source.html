57static std::shared_ptr<SymbolicExpr> symbolic_poly_to_expr(
58 const Polynomial<SymbolicPolyCoeff>& poly) {
59 if (poly.is_zero()) return SymbolicExpr::number(0);
60
61 std::vector<std::shared_ptr<SymbolicExpr>> terms;
62 for (int i = poly.degree(); i >= 0; --i) {
63 const auto& coeff = poly.coeffs[i];
64 if (coeff == SymbolicPolyCoeff(0)) continue;
65
66 auto coeff_expr = coeff.val;
67 if (coeff_expr) {
68 if (auto simplified = coeff_expr->simplify()) {
69 coeff_expr = simplified;
70 }
71 if (lamina::detail::node(coeff_expr) && coeff_expr->is_zero()) {
72 continue;
73 }
74 }
75 if (i == 0) {
76 terms.push_back(coeff_expr);
77 } else {
78 auto var_expr = SymbolicExpr::variable(poly.variable_name);
79 std::shared_ptr<SymbolicExpr> var_part;
80 if (i == 1) {
81 var_part = var_expr;
82 } else {
83 var_part = SymbolicExpr::power(var_expr, SymbolicExpr::number(i));
84 }
85
86 if (coeff_expr->is_one()) {
87 terms.push_back(var_part);
88 } else {
89 terms.push_back(SymbolicExpr::multiply(coeff_expr, var_part));
90 }
91 }
92 }
93
94 if (terms.empty()) return SymbolicExpr::number(0);
95 if (terms.size() == 1) return terms[0];
96
97 auto result = terms[0];
98 for (size_t i = 1; i < terms.size(); ++i) {
99 result = SymbolicExpr::add(result, terms[i]);
100 }
101 return result;
102}

112Result<RootOfRequest> parse_rootof_request(
113 const std::shared_ptr<SymbolicExpr>& rootof_expr,
114 ComputationContext& context,
115 const std::string& operation) {
116 if (!rootof_expr || !lamina::detail::node(rootof_expr)) {
117 return Result<RootOfRequest>::failure(
118 CasErrc::InvalidArgument, "RootOf expression cannot be null", operation);
119 }
120 auto root = std::dynamic_pointer_cast<const RootOfNode>(
121 lamina::detail::node(rootof_expr));
122 if (!root) {
123 return Result<RootOfRequest>::failure(
124 CasErrc::InvalidArgument,
125 "expression is not a valid RootOf", operation);
126 }
127 auto access = context.consume_steps(1, operation);
128 if (!access) return Result<RootOfRequest>::failure(access.error());
129 return Result<RootOfRequest>::success(RootOfRequest{
130 root->exact_id().polynomial, root->variable(), root->index()});
131}

136std::vector<std::shared_ptr<SymbolicExpr>> make_rootof_solutions(
137 const Polynomial<SymbolicPolyCoeff>& poly,
138 const std::string& var) {
139 if (poly.degree() <= 0) return {};
140 auto polynomial_expression = symbolic_poly_to_expr(poly);
141 ComputationContext context;
142 auto recognized = recognize_rational_polynomial(
143 *polynomial_expression, var, context);
144 if (!recognized || !recognized.value()) return {};
145 auto canonical = recognized.value()->square_free_part().make_monic();
146 canonical.variable_name = "_root";
147 std::vector<std::shared_ptr<SymbolicExpr>> solutions;
148 solutions.reserve(static_cast<std::size_t>(canonical.degree()));
149 for (int index = 0; index < canonical.degree(); ++index) {
150 solutions.push_back(lamina::detail::make_expression_ptr(
151 lamina::detail::make_node<RootOfNode>(
152 detail::ExactRootId{
153 canonical, static_cast<std::size_t>(index)},
154 var)));
155 }
156 return solutions;
157}

204static std::vector<std::shared_ptr<SymbolicExpr>> solve_closed_form_from_poly(
205 const Polynomial<SymbolicPolyCoeff>& poly,
206 const std::string& var)
207{
208 int deg = poly.degree();
209 if (deg <= 0) return {};
210
211 auto get_coeff = [&](int d) -> std::shared_ptr<SymbolicExpr> {
212 if (d < 0 || d > deg) return SymbolicExpr::number(0);
213 return poly.coeffs[d].val ? poly.coeffs[d].val : SymbolicExpr::number(0);
214 };
215
216 if (deg == 1) {
217
218 auto a = get_coeff(1);
219 auto b = get_coeff(0);
220 auto neg_b = SymbolicExpr::multiply(b, SymbolicExpr::number(-1));
221 return { SymbolicExpr::divide(neg_b, a)->simplify() };
222 } else if (deg == 2) {
223 auto a = get_coeff(2);
224 auto b = get_coeff(1);
225 auto c = get_coeff(0);
226
227 auto b2 = SymbolicExpr::power(b, SymbolicExpr::number(2));
228 auto four_ac = SymbolicExpr::multiply(SymbolicExpr::number(4), SymbolicExpr::multiply(a, c));
229 auto delta = SymbolicExpr::add(b2, SymbolicExpr::multiply(four_ac, SymbolicExpr::number(-1)));
230 auto sqrt_delta = SymbolicExpr::sqrt(delta);
231 auto neg_b = SymbolicExpr::multiply(b, SymbolicExpr::number(-1));
232 auto two_a = SymbolicExpr::multiply(SymbolicExpr::number(2), a);
233
234 auto x1 = SymbolicExpr::divide(SymbolicExpr::add(neg_b, sqrt_delta), two_a)->simplify();
235 auto x2 = SymbolicExpr::divide(SymbolicExpr::add(neg_b, SymbolicExpr::multiply(sqrt_delta, SymbolicExpr::number(-1))), two_a)->simplify();
236 return { x1, x2 };
237 } else if (deg == 3) {
238 auto a = get_coeff(3);
239 auto b = get_coeff(2);
240 auto c = get_coeff(1);
241 auto d = get_coeff(0);
242 return solve_cubic(a, b, c, d, var);
243 } else if (deg == 4) {
244 auto a = get_coeff(4);
245 auto b = get_coeff(3);
246 auto c = get_coeff(2);
247 auto d = get_coeff(1);
248 auto e = get_coeff(0);
249 return solve_quartic(a, b, c, d, e, var);
250 }
251
252 return {};
253}

258std::shared_ptr<SymbolicExpr> rootof_simplify(
259 const std::shared_ptr<SymbolicExpr>& rootof_expr)
260{
261 if (!rootof_expr || !lamina::detail::node(rootof_expr)) return rootof_expr;
262
263 auto root = std::dynamic_pointer_cast<const RootOfNode>(
264 lamina::detail::node(rootof_expr));
265 if (!root) return rootof_expr;
266
267 const std::string& var = root->variable();
268 const int k = static_cast<int>(root->index());
269 std::vector<SymbolicPolyCoeff> coefficients;
270 coefficients.reserve(root->exact_id().polynomial.coeffs.size());
271 for (const auto& coefficient : root->exact_id().polynomial.coeffs) {
272 coefficients.emplace_back(SymbolicExpr::number(coefficient));
273 }
274 Polynomial<SymbolicPolyCoeff> sym_poly(coefficients, var);
275
276 int degree = sym_poly.degree();
277 if (degree <= 0) return rootof_expr;
278
279 if (k < 0 || k >= degree) {
280 return rootof_expr;
281 }
282
283 if (degree == 1) {
284 auto roots = solve_closed_form_from_poly(sym_poly, var);
285 return roots.size() == 1 ? roots.front() : rootof_expr;
286 }
287 if (degree == 2) {
288 auto roots = solve_closed_form_from_poly(sym_poly, var);
289 if (roots.size() != 2) return rootof_expr;
290 return roots[static_cast<std::size_t>(1 - k)];
291 }
292
293 return rootof_expr;
294}
