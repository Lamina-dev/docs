14static std::shared_ptr<SymbolicExpr> symbolic_poly_to_expr(
15 const Polynomial<SymbolicPolyCoeff>& poly) {
16 if (poly.is_zero()) return SymbolicExpr::number(0);
17
18 std::vector<std::shared_ptr<SymbolicExpr>> terms;
19 for (int i = poly.degree(); i >= 0; --i) {
20 const auto& coeff = poly.coeffs[i];
21 if (coeff == SymbolicPolyCoeff(0)) continue;
22
23 auto coeff_expr = coeff.val;
24 if (i == 0) {
25 terms.push_back(coeff_expr);
26 } else {
27 auto var_expr = SymbolicExpr::variable(poly.variable_name);
28 std::shared_ptr<SymbolicExpr> var_part;
29 if (i == 1) {
30 var_part = var_expr;
31 } else {
32 var_part = SymbolicExpr::power(var_expr, SymbolicExpr::number(i));
33 }
34
35 if (coeff_expr->is_one()) {
36 terms.push_back(var_part);
37 } else {
38 terms.push_back(SymbolicExpr::multiply(coeff_expr, var_part));
39 }
40 }
41 }
42
43 if (terms.empty()) return SymbolicExpr::number(0);
44 if (terms.size() == 1) return terms[0];
45
46 auto result = terms[0];
47 for (size_t i = 1; i < terms.size(); ++i) {
48 result = SymbolicExpr::add(result, terms[i]);
49 }
50 return result;
51}

80RootOfEvaluationResult rootof_evaluate_checked(
81 const std::shared_ptr<SymbolicExpr>& rootof_expr,
82 ComputationContext& context)
83{
84 constexpr const char* operation = "rootof_evaluate";
85 if (!rootof_expr || !lamina::detail::node(rootof_expr)) {
86 return RootOfEvaluationResult::failure(
87 CasErrc::InvalidArgument, "RootOf expression cannot be null", operation);
88 }
89
90 auto func_node = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(rootof_expr));
91 if (!func_node || func_node->type() != FunctionNode::FuncType::RootOf) {
92 return RootOfEvaluationResult::failure(
93 CasErrc::InvalidArgument, "expression is not a RootOf", operation);
94 }
95
96 if (func_node->arguments().size() != 3) {
97 return RootOfEvaluationResult::failure(
98 CasErrc::InternalInvariant, "RootOf must contain three arguments", operation);
99 }
100
101 auto poly_node = func_node->arguments()[0];
102 auto var_node = func_node->arguments()[1];
103 auto index_node = func_node->arguments()[2];
104
105 auto var_sym = std::dynamic_pointer_cast<const VariableNode>(var_node);
106 if (!var_sym || var_sym->name().empty()) {
107 return RootOfEvaluationResult::failure(
108 CasErrc::InvalidArgument, "RootOf variable must be a named symbol", operation);
109 }
110 std::string var = var_sym->name();
111
112 auto idx_num = std::dynamic_pointer_cast<const NumberNode>(index_node);
113 if (!idx_num || std::holds_alternative<lmmc_real_t>(idx_num->value())) {
114 return RootOfEvaluationResult::failure(
115 CasErrc::InvalidArgument, "RootOf index must be an exact integer", operation);
116 }
117
118 BigInt index;
119 if (std::holds_alternative<BigInt>(idx_num->value())) {
120 index = std::get<BigInt>(idx_num->value());
121 } else {
122 const Rational& rational = std::get<Rational>(idx_num->value());
123 if (!rational.is_integer()) {
124 return RootOfEvaluationResult::failure(
125 CasErrc::InvalidArgument, "RootOf index must be an integer", operation);
126 }
127 index = rational.to_BigInt();
128 }
129 auto index_value = index.try_to_int64();
130 if (!index_value || *index_value < 0) {
131 return RootOfEvaluationResult::failure(
132 CasErrc::InvalidArgument, "RootOf index is negative or too large", operation);
133 }
134
135 auto poly_expr = lamina::detail::make_expression_ptr(poly_node);
136 auto recognized = recognize_rational_polynomial(*poly_expr, var, context);
137 if (!recognized) return RootOfEvaluationResult::failure(recognized.error());
138 if (!recognized.value()) {
139 return RootOfEvaluationResult::failure(
140 CasErrc::Inconclusive,
141 "RootOf numeric evaluation supports exact rational polynomials only",
142 operation);
143 }
144 const Polynomial<Rational>& rat_poly = *recognized.value();
145 const int degree = rat_poly.degree();
146 if (degree <= 0) {
147 return RootOfEvaluationResult::failure(
148 CasErrc::InvalidArgument, "RootOf polynomial must have positive degree", operation);
149 }
150 if (*index_value >= degree) {
151 return RootOfEvaluationResult::failure(
152 CasErrc::InvalidArgument, "RootOf index exceeds polynomial degree", operation);
153 }
154
155 auto isolation_step = context.consume_steps(
156 static_cast<std::size_t>(degree), operation);
157 if (!isolation_step) return RootOfEvaluationResult::failure(isolation_step.error());
158 auto intervals = isolate_real_roots(rat_poly);
159 if (static_cast<int>(intervals.size()) != degree) {
160 return RootOfEvaluationResult::failure(
161 CasErrc::Inconclusive,
162 "RootOf double evaluation currently requires every root to be real",
163 operation);
164 }
165
166 const auto& interval = intervals[static_cast<std::size_t>(*index_value)];
167 const lmmc_real_t lo = interval.first.to_double();
168 const lmmc_real_t hi = interval.second.to_double();
169 const lmmc_real_t x0 = (lo + hi) / 2.0;
170 auto df_expr = poly_expr->differentiate(var);
171
172 SolveOptions opts;
173 opts.tolerance = 1e-12;
174 opts.max_newton_iterations = 100;
175 auto refined = newton_raphson_checked(
176 poly_expr, df_expr, var, x0, lo, hi, context, opts);
177 if (!refined) return RootOfEvaluationResult::failure(refined.error());
178 if (!refined.value()) {
179 return RootOfEvaluationResult::failure(
180 CasErrc::Inconclusive, "isolated RootOf could not be numerically verified", operation);
181 }
182 return RootOfEvaluationResult::success(refined.value()->value);
183}

207static std::vector<std::shared_ptr<SymbolicExpr>> solve_closed_form_from_poly(
208 const Polynomial<SymbolicPolyCoeff>& poly,
209 const std::string& var)
210{
211 int deg = poly.degree();
212 if (deg <= 0) return {};
213
214 auto get_coeff = [&](int d) -> std::shared_ptr<SymbolicExpr> {
215 if (d < 0 || d > deg) return SymbolicExpr::number(0);
216 return poly.coeffs[d].val ? poly.coeffs[d].val : SymbolicExpr::number(0);
217 };
218
219 if (deg == 1) {
220
221 auto a = get_coeff(1);
222 auto b = get_coeff(0);
223 auto neg_b = SymbolicExpr::multiply(b, SymbolicExpr::number(-1));
224 return { SymbolicExpr::divide(neg_b, a)->simplify() };
225 } else if (deg == 2) {
226 auto a = get_coeff(2);
227 auto b = get_coeff(1);
228 auto c = get_coeff(0);
229
230 auto b2 = SymbolicExpr::power(b, SymbolicExpr::number(2));
231 auto four_ac = SymbolicExpr::multiply(SymbolicExpr::number(4), SymbolicExpr::multiply(a, c));
232 auto delta = SymbolicExpr::add(b2, SymbolicExpr::multiply(four_ac, SymbolicExpr::number(-1)));
233 auto sqrt_delta = SymbolicExpr::sqrt(delta);
234 auto neg_b = SymbolicExpr::multiply(b, SymbolicExpr::number(-1));
235 auto two_a = SymbolicExpr::multiply(SymbolicExpr::number(2), a);
236
237 auto x1 = SymbolicExpr::divide(SymbolicExpr::add(neg_b, sqrt_delta), two_a)->simplify();
238 auto x2 = SymbolicExpr::divide(SymbolicExpr::add(neg_b, SymbolicExpr::multiply(sqrt_delta, SymbolicExpr::number(-1))), two_a)->simplify();
239 return { x1, x2 };
240 } else if (deg == 3) {
241 auto a = get_coeff(3);
242 auto b = get_coeff(2);
243 auto c = get_coeff(1);
244 auto d = get_coeff(0);
245 return solve_cubic(a, b, c, d, var);
246 } else if (deg == 4) {
247 auto a = get_coeff(4);
248 auto b = get_coeff(3);
249 auto c = get_coeff(2);
250 auto d = get_coeff(1);
251 auto e = get_coeff(0);
252 return solve_quartic(a, b, c, d, e, var);
253 }
254
255 return {};
256}

258static std::vector<std::shared_ptr<SymbolicExpr>> sort_roots_by_convention(
259 const std::vector<std::shared_ptr<SymbolicExpr>>& roots)
260{
261 if (roots.empty()) return roots;
262
263 std::vector<RootWithIndex> evaluated;
264 evaluated.reserve(roots.size());
265
266 bool all_evaluated = true;
267 for (const auto& root : roots) {
268 RootWithIndex ri;
269 ri.expr = root;
270 auto [val, success] = try_eval_complex(root);
271 ri.eval_success = success;
272 if (success) {
273 ri.real_part = val.real();
274 ri.imag_part = val.imag();
275
276 ri.is_real = (std::abs(ri.imag_part) < 1e-10);
277 } else {
278 all_evaluated = false;
279 ri.real_part = 0.0;
280 ri.imag_part = 0.0;
281 ri.is_real = true;
282 }
283 evaluated.push_back(ri);
284 }
285
286 if (!all_evaluated) {
287 return roots;
288 }
289
290 std::vector<RootWithIndex> real_roots;
291 std::vector<RootWithIndex> complex_roots;
292
293 for (auto& ri : evaluated) {
294 if (ri.is_real) {
295 real_roots.push_back(ri);
296 } else {
297 complex_roots.push_back(ri);
298 }
299 }
300
301 std::sort(real_roots.begin(), real_roots.end(),
302 [](const RootWithIndex& a, const RootWithIndex& b) {
303 return a.real_part < b.real_part;
304 });
305
306 std::sort(complex_roots.begin(), complex_roots.end(),
307 [](const RootWithIndex& a, const RootWithIndex& b) {
308 if (std::abs(a.real_part - b.real_part) > 1e-10) {
309 return a.real_part < b.real_part;
310 }
311
312 return a.imag_part > b.imag_part;
313 });
314
315 std::vector<std::shared_ptr<SymbolicExpr>> sorted;
316 sorted.reserve(roots.size());
317 for (const auto& ri : real_roots) {
318 sorted.push_back(ri.expr);
319 }
320 for (const auto& ri : complex_roots) {
321 sorted.push_back(ri.expr);
322 }
323
324 return sorted;
325}

327static bool is_structural_polynomial(const std::shared_ptr<const SymbolicNode>& node,
328 const std::string& variable) {
329 if (!node) return false;
330 if (!expression_depends_on_variable(node, variable)) return true;
331
332 if (auto symbol = std::dynamic_pointer_cast<const VariableNode>(node)) {
333 return symbol->name() == variable;
334 }
335 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
336 return std::all_of(add->operands().begin(), add->operands().end(),
337 [&](const auto& operand) {
338 return is_structural_polynomial(operand, variable);
339 });
340 }
341 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
342 return std::all_of(multiply->operands().begin(), multiply->operands().end(),
343 [&](const auto& operand) {
344 return is_structural_polynomial(operand, variable);
345 });
346 }
347 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
348 auto exponent_node = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
349 if (!exponent_node || std::holds_alternative<lmmc_real_t>(exponent_node->value())) {
350 return false;
351 }
352 BigInt exponent;
353 if (std::holds_alternative<BigInt>(exponent_node->value())) {
354 exponent = std::get<BigInt>(exponent_node->value());
355 } else {
356 const Rational& rational = std::get<Rational>(exponent_node->value());
357 if (!rational.is_integer()) return false;
358 exponent = rational.to_BigInt();
359 }
360 auto value = exponent.try_to_int64();
361 return value && *value > 0 && *value < 1000 &&
362 is_structural_polynomial(power->base(), variable);
363 }
364 return false;
365}

388 const std::shared_ptr<SymbolicExpr>& rootof_expr)
389{
390 if (!rootof_expr || !lamina::detail::node(rootof_expr)) return rootof_expr;
391
392 auto func_node = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(rootof_expr));
393 if (!func_node || func_node->type() != FunctionNode::FuncType::RootOf) {
394 return rootof_expr;
395 }
396
397 if (func_node->arguments().size() != 3) {
398 return rootof_expr;
399 }
400
401 auto poly_node = func_node->arguments()[0];
402 auto var_node = func_node->arguments()[1];
403 auto index_node = func_node->arguments()[2];
404

406 if (!var_sym) return rootof_expr;
407 std::string var = var_sym->name();
408
409 auto index = exact_root_index(index_node);
410 if (!index) return rootof_expr;
411 const int k = *index;
412
413 if (!is_structural_polynomial(poly_node, var)) return rootof_expr;
414
415 auto poly_expr = lamina::detail::make_expression_ptr(poly_node);
416 auto sym_poly = symbolic_to_poly<SymbolicPolyCoeff>(poly_expr, var);
417
418 int degree = sym_poly.degree();
419 if (degree <= 0) return rootof_expr;
420
421 if (k < 0 || k >= degree) {
422 return rootof_expr;
423 }
424
425 if (degree <= 4) {

427 if (roots.empty()) {
428 return rootof_expr;
429 }
430

432
433 if (k >= 0 && k < static_cast<int>(sorted_roots.size())) {
434 return sorted_roots[k];

438 }
