26// Binomial coefficient C(n,k) for small n; safe for n up to ~62.
27long long binomial_ll(int n, int k) {
28 if (k < 0 || k > n) return 0;
29 if (k == 0 || k == n) return 1;
30 if (k > n - k) k = n - k;
31 long long c = 1;
32 for (int i = 0; i < k; ++i) {
33 c = c * static_cast<long long>(n - i) / static_cast<long long>(i + 1);
34 }
35 return c;
36}

38// Match a single-argument FunctionNode whose argument is the integration
39// variable itself. Returns 0 (sin), 1 (cos), 2 (tan), 3 (sec), or -1 on
40// no match for this strategy.
41int trig_match_of_var(const std::shared_ptr<const SymbolicNode>& node, const std::string& var) {
42 auto fn = std::dynamic_pointer_cast<const FunctionNode>(node);
43 if (!fn) return -1;
44 if (fn->arguments().size() != 1) return -1;
45 auto v = std::dynamic_pointer_cast<const VariableNode>(fn->arguments()[0]);
46 if (!v || v->name() != var) return -1;
47 using FT = FunctionNode::FuncType;
48 switch (fn->type()) {
49 case FT::Sin: return 0;
50 case FT::Cos: return 1;
51 case FT::Tan: return 2;
52 case FT::Sec: return 3;
53 default: return -1;
54 }

68// Match a single factor of the form trig(var) or trig(var)^k where trig is
69// sin/cos/tan/sec and k is a non-negative integer. Returns true on success
70// with the kind (0..3) and the integer power.
71bool trig_extract_factor(const std::shared_ptr<const SymbolicNode>& node,
72 const std::string& var,
73 int& kind_out, int& power_out) {
74 int kind = trig_match_of_var(node, var);
75 if (kind >= 0) {
76 kind_out = kind;
77 power_out = 1;
78 return true;
79 }
80 auto pn = std::dynamic_pointer_cast<const PowerNode>(node);
81 if (!pn) return false;
82 int base_kind = trig_match_of_var(pn->base(), var);
83 if (base_kind < 0) return false;
84 int p = 0;
85 if (!trig_extract_nonneg_int(pn->exponent(), p)) return false;
86 kind_out = base_kind;
87 power_out = p;
88 return true;
89}

93bool TrigCombinationStrategy::extract_sin_cos_powers(
94 const SymbolicExpr& expr, const std::string& var, int& m_out, int& n_out) {
95
96 int m = 0, n = 0;
97 std::vector<std::shared_ptr<const SymbolicNode>> factors;
98 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(expr))) {
99 factors = mul->operands();
100 } else {
101 factors.push_back(lamina::detail::node(expr));
102 }
103
104 for (const auto& f : factors) {
105 int kind = -1, p = 0;
106 if (!trig_extract_factor(f, var, kind, p)) return false;
107 // Only sin/cos contribute to (m,n); tan/sec disqualify this form.
108 if (kind == 0) m += p;
109 else if (kind == 1) n += p;
110 else return false;
111 }
112
113 if (m == 0 && n == 0) return false; // not a non-trivial sin/cos product
114 m_out = m;
115 n_out = n;
116 return true;
117}

137std::shared_ptr<SymbolicExpr> TrigCombinationStrategy::try_integrate(
138 const SymbolicExpr& expr, const std::string& var, Integrator& ctx,
139 ComputationContext&, int depth) {
140
141 // sin^m(x) * cos^n(x)
142 int m = 0, n = 0;
143 if (extract_sin_cos_powers(expr, var, m, n)) {
144 if (m < 0 || n < 0) return nullptr;
145 if (m + n > 8) return nullptr;
146 return integrate_sin_m_cos_n(m, n, 1, var, ctx, depth);
147 }
148
149 // tan^n(x)
150 int tn = 0;
151 if (extract_tan_power(expr, var, tn)) {
152 if (tn < 2 || tn > 8) return nullptr;
153 return integrate_tan_power(tn, var, ctx, depth);
154 }
155
156 // sec^n(x) -- only even powers per design.
157 int sn = 0;
158 if (extract_sec_power(expr, var, sn)) {
159 if (sn < 2 || sn > 8) return nullptr;
160 if (sn % 2 != 0) return nullptr;
161 return integrate_sec_power(sn, var, ctx, depth);
162 }
163
164 return nullptr;
165}

168 int m, int n, long long scale,
169 const std::string& var, Integrator& ctx, int depth) {

183std::shared_ptr<SymbolicExpr> TrigCombinationStrategy::integrate_odd_case(
184 int m, int n, long long scale,
185 const std::string& var, Integrator&, int) {
186
187 if (scale == 0) return nullptr;
188
189 // Pick which factor to peel:
190 // m odd -> peel one sin, set u = cos(scale*x), du = -scale*sin(scale*x)dx
191 // so sin(scale*x)dx = -du/scale.
192 // remaining: sin^(m-1) cos^n = (1-u^2)^k * u^n, k=(m-1)/2.
193 // n odd -> peel one cos, set u = sin(scale*x), du = +scale*cos(scale*x)dx
194 // so cos(scale*x)dx = du/scale.
195 // remaining: sin^m cos^(n-1) = u^m * (1-u^2)^k, k=(n-1)/2.
196 bool peel_sin = (m % 2 == 1);
197 int k = 0;
198 int other_pow = 0;
199 long long sign_factor = 1; // includes the sign from du sign.
200 bool u_is_cos = false;
201
202 if (peel_sin) {
203 k = (m - 1) / 2;
204 other_pow = n;
205 sign_factor = -1;
206 u_is_cos = true;
207 } else {
208 k = (n - 1) / 2;
209 other_pow = m;
210 sign_factor = 1;
211 u_is_cos = false;
212 }
213
214 auto u_expr = u_is_cos ? make_cos_scaled(scale, var)
215 : make_sin_scaled(scale, var);
216
217 // Result = sum_{i=0..k} C(k,i)*(-1)^i * u^(2i+other_pow+1) / (scale*(2i+other_pow+1)) * sign_factor
218 std::vector<std::shared_ptr<const SymbolicNode>> add_terms;
219 for (int i = 0; i <= k; ++i) {
220 long long bin = binomial_ll(k, i);
221 long long alt_sign = ((i % 2) == 0) ? 1 : -1;
222 long long num = sign_factor * alt_sign * bin;
223 int u_pow = 2 * i + other_pow + 1; // always >= 1
224 long long den = scale * static_cast<long long>(u_pow);
225 if (den == 0) return nullptr;
226
227 std::shared_ptr<SymbolicExpr> u_to_pow;
228 if (u_pow == 1) {
229 u_to_pow = u_expr;
230 } else {
231 u_to_pow = SymbolicExpr::power(u_expr, SymbolicExpr::number(u_pow));
232 }
233 // Normalize sign so the rational denominator is positive.
234 if (den < 0) { num = -num; den = -den; }
235 auto coeff = SymbolicExpr::number(Rational(BigInt(num), BigInt(den)));
236 auto term = SymbolicExpr::multiply(coeff, u_to_pow);
237 add_terms.push_back(lamina::detail::node(term));
238 }
239
240 if (add_terms.empty()) return SymbolicExpr::number(0);
241 if (add_terms.size() == 1) {
242 return lamina::detail::make_expression_ptr(add_terms[0]);
243 }
244 return lamina::detail::make_expression_ptr(lamina::detail::make_node<AddNode>(add_terms));
245}

247std::shared_ptr<SymbolicExpr> TrigCombinationStrategy::integrate_even_case(
248 int m, int n, long long scale,
249 const std::string& var, Integrator& ctx, int depth) {
250
251 if ((m % 2 != 0) || (n % 2 != 0)) return nullptr;
252 int p = m / 2;
253 int q = n / 2;
254 int K = p + q;
255
256 // sin^(2p)(c*x)*cos^(2q)(c*x)
257 // = (1/2^(p+q)) * sum_{i,j} C(p,i)(-1)^i C(q,j) cos^(i+j)(2c*x)
258 // Compute coefficients a[k] = sum_{i+j=k} C(p,i)*C(q,j)*(-1)^i for k=0..K.
259 std::vector<long long> a(K + 1, 0);
260 for (int i = 0; i <= p; ++i) {
261 long long bp = binomial_ll(p, i);
262 long long si = ((i % 2) == 0) ? 1 : -1;
263 for (int j = 0; j <= q; ++j) {
264 long long bq = binomial_ll(q, j);
265 a[i + j] += si * bp * bq;
266 }
267 }
268
269 long long pow2 = 1;
270 for (int t = 0; t < K; ++t) pow2 *= 2;
271 if (pow2 == 0) pow2 = 1;
272
273 std::vector<std::shared_ptr<const SymbolicNode>> add_terms;
274 for (int k = 0; k <= K; ++k) {
275 if (a[k] == 0) continue;
276 // Recurse: integrate cos^k(2c*x) at the new scale 2c.
277 auto inner = integrate_sin_m_cos_n(0, k, scale * 2, var, ctx, depth + 1);
278 if (!inner) return nullptr;
279 long long num = a[k];
280 long long den = pow2;
281 if (den < 0) { num = -num; den = -den; }
282 auto coeff = SymbolicExpr::number(Rational(BigInt(num), BigInt(den)));
283 auto term = SymbolicExpr::multiply(coeff, inner);
284 add_terms.push_back(lamina::detail::node(term));
285 }
286
287 if (add_terms.empty()) return SymbolicExpr::number(0);
288 if (add_terms.size() == 1) {
289 return lamina::detail::make_expression_ptr(add_terms[0]);
290 }
291 return lamina::detail::make_expression_ptr(lamina::detail::make_node<AddNode>(add_terms));
292}

294std::shared_ptr<SymbolicExpr> TrigCombinationStrategy::integrate_tan_power(
295 int n, const std::string& var, Integrator& ctx, int depth) {
296
297 if (n < 0) return nullptr;
298 auto v = SymbolicExpr::variable(var);
299
300 if (n == 0) {
301 // int 1 dx = x
302 return v;
303 }
304 if (n == 1) {
305 // int tan(x) dx = -ln(cos(x))
306 auto cos_x = SymbolicExpr::cos(v);
307 auto ln_cos = SymbolicExpr::ln(cos_x);
308 return SymbolicExpr::multiply(SymbolicExpr::number(-1), ln_cos);
309 }
310
311 // n >= 2: tan^n = tan^(n-2)*(sec^2 - 1)
312 // int tan^n = tan^(n-1)/(n-1) - int tan^(n-2)
313 auto tan_x = SymbolicExpr::tan(v);
314 std::shared_ptr<SymbolicExpr> tan_pow_term;
315 if (n - 1 == 1) {
316 tan_pow_term = tan_x;
317 } else {
318 tan_pow_term = SymbolicExpr::power(tan_x, SymbolicExpr::number(n - 1));
319 }
320 auto first = SymbolicExpr::multiply(trig_rational(1, n - 1), tan_pow_term);
321
322 auto rest = integrate_tan_power(n - 2, var, ctx, depth + 1);
323 if (!rest) return nullptr;
324 auto neg_rest = SymbolicExpr::multiply(SymbolicExpr::number(-1), rest);
325 return SymbolicExpr::add(first, neg_rest);
326}

328std::shared_ptr<SymbolicExpr> TrigCombinationStrategy::integrate_sec_power(
329 int n, const std::string& var, Integrator& ctx, int depth) {
330
331 if (n < 2 || (n % 2) != 0) return nullptr;
332
333 using FT = FunctionNode::FuncType;
334 auto v = SymbolicExpr::variable(var);
335 auto tan_x = SymbolicExpr::tan(v);
336 auto make_sec_x = [&]() -> std::shared_ptr<SymbolicExpr> {
337 return lamina::detail::make_expression_ptr(
338 lamina::detail::make_node<FunctionNode>(FT::Sec,
339 std::vector<std::shared_ptr<const SymbolicNode>>{lamina::detail::node(v)}));
340 };
341
342 if (n == 2) {
343 return tan_x;
344 }
345
346 auto sec_x = make_sec_x();
347 std::shared_ptr<SymbolicExpr> sec_pow;
348 if (n - 2 == 1) {
349 sec_pow = sec_x;
350 } else {
351 sec_pow = SymbolicExpr::power(sec_x, SymbolicExpr::number(n - 2));
352 }
353
354 // First term: sec^(n-2)(x) * tan(x) / (n-1)
355 auto inner = SymbolicExpr::multiply(sec_pow, tan_x);
356 auto first = SymbolicExpr::multiply(trig_rational(1, n - 1), inner);
357
358 // Second term: ((n-2)/(n-1)) * int sec^(n-2)
359 auto rest = integrate_sec_power(n - 2, var, ctx, depth + 1);
360 if (!rest) return nullptr;
361 auto coeff = SymbolicExpr::number(Rational(BigInt(static_cast<long long>(n - 2)),
362 BigInt(static_cast<long long>(n - 1))));
363 auto second = SymbolicExpr::multiply(coeff, rest);
364
365 return SymbolicExpr::add(first, second);
366}
