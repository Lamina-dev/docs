11void eliminate_lu_column(RationalRows& lower, RationalRows& upper,
12 std::size_t column) {
13 const std::size_t n = upper.size();
14 for (std::size_t row = column + 1; row < n; ++row) {
15 const Rational multiplier =
16 upper[row][column] / upper[column][column];
17 lower[row][column] = multiplier;
18 for (std::size_t col = column; col < n; ++col) {
19 upper[row][col] =
20 upper[row][col] -
21 multiplier * upper[column][col];
22 }
23 }

26Result<void> factor_lu(RationalRows& lower, RationalRows& upper,
27 RationalRows& permutation, const std::string& operation) {
28 const std::size_t n = upper.size();
29 for (std::size_t i = 0; i < n; ++i) {
30 lower[i][i] = Rational(1);
31 permutation[i][i] = Rational(1);
32 }
33 for (std::size_t column = 0; column < n; ++column) {
34 std::size_t pivot = column;
35 while (pivot < n && upper[pivot][column] == Rational(0)) ++pivot;
36 if (pivot == n) {
37 return Result<void>::failure(
38 CasErrc::DomainError,
39 "PLU decomposition requires a nonsingular matrix",
40 operation);
41 }
42 if (pivot != column) {
43 std::swap(upper[pivot], upper[column]);
44 std::swap(permutation[pivot], permutation[column]);
45 for (std::size_t prior = 0; prior < column; ++prior) {
46 std::swap(lower[pivot][prior], lower[column][prior]);
47 }
48 }
49 eliminate_lu_column(lower, upper, column);
50 }
51 return Result<void>::success();
52}
