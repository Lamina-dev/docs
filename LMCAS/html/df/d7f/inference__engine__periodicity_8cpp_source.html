27 Result<PeriodEvidence> infer(const Node& node) {
28 auto entered = context_.enter_recursion(periodic_operation);
29 if (!entered) { return Result<PeriodEvidence>::failure(entered.error()); }
30 struct Leave {
31 ComputationContext& context;
32 ~Leave() { context.leave_recursion(); }
33 } leave{context_};
34 if (depth_ >= max_depth_) return PeriodEvidence{};
35 ++depth_;
36 struct LeaveProof {
37 int& depth;
38 ~LeaveProof() { --depth; }
39 } leave_proof{depth_};
40
41 if (const auto* symbol = dynamic_cast<const VariableNode*>(node.get())) {
42 if (!symbol->is_constant() &&
43 assumptions_.has_period_declarations(symbol->name())) {
44 return declared_period(node, *symbol);
45 }
46 }
47 auto constant_real = real_defined(node);
48 if (!constant_real) { return Result<PeriodEvidence>::failure(constant_real.error()); }
49 if (constant_real.value() == Tribool::True && independent(node))
50 { return PeriodEvidence{Tribool::True, std::nullopt, true}; }
51
52 return function_period(node);
53 }

56 Result<PeriodEvidence> function_period(const Node& node) {
57 const auto* function = dynamic_cast<const FunctionNode*>(node.get());
58 if (!function) { return PeriodEvidence{}; }
59 if (function->type() == FunctionNode::FuncType::Infinity)
60 { return PeriodEvidence{Tribool::False, std::nullopt, false}; }
61 if (function->arguments().size() != 1) { return PeriodEvidence{}; }
62 const Node& argument = function->arguments().front();
63 if (function->type() == FunctionNode::FuncType::ArcTan) {
64 return arctangent_period(argument);
65 }
66 const auto type = function->type();
67 if (type != FunctionNode::FuncType::Sin && type != FunctionNode::FuncType::Cos &&
68 type != FunctionNode::FuncType::Tan) { return PeriodEvidence{}; }
69 return affine_period(node, argument, type);
70 }

72 Result<PeriodEvidence> declared_period(const Node& node, const VariableNode& symbol) {
73 auto period = assumptions_.get_period(symbol.name(), variable_);
74 if (!period) { return PeriodEvidence{}; }
75 auto defined = detail::query_definedness(
76 detail::node(*period), facts_, Domain::Real, context_);
77 if (!defined) { return Result<PeriodEvidence>::failure(defined.error()); }
78 auto positive = detail::query_positive_value(
79 detail::node(*period), facts_, context_);
80 if (!positive) { return Result<PeriodEvidence>::failure(positive.error()); }
81 if (defined.value() != Tribool::True || positive.value() != Tribool::True ||
82 !independent(detail::node(*period)))
83 { return PeriodEvidence{}; }
84 auto real = detail::query_real_value(node, facts_, context_);
85 if (!real) { return Result<PeriodEvidence>::failure(real.error()); }
86 return PeriodEvidence{
87 Tribool::True, *period, real.value() == Tribool::True};
88 }

90 Result<PeriodEvidence> arctangent_period(const Node& argument) {
91 const auto* symbol = dynamic_cast<const VariableNode*>(argument.get());
92 if (symbol && !symbol->is_constant() && symbol->name() == variable_ &&
93 !assumptions_.has_period_declarations(symbol->name())) {
94 return PeriodEvidence{Tribool::False, std::nullopt, true};
95 }
96 auto inner = infer(argument);
97 if (!inner) { return inner; }
98 if (inner.value().periodic == Tribool::True && inner.value().real_valued)
99 { return inner; }
100 return PeriodEvidence{};
101 }

103 Result<PeriodEvidence> affine_period(const Node& node, const Node& argument, FunctionNode::FuncType type) {
104 auto defined = detail::query_definedness(argument, facts_, Domain::Real, context_);
105 if (!defined) { return Result<PeriodEvidence>::failure(defined.error()); }
106 if (defined.value() != Tribool::True) { return PeriodEvidence{}; }
107 auto affine = detail::recognize_affine(detail::expression_from_node(argument), variable_, context_);
108 if (!affine) { return Result<PeriodEvidence>::failure(affine.error()); }
109 if (!affine.value()) { return PeriodEvidence{}; }
110 for (auto* coefficient : {&affine.value()->slope, &affine.value()->offset}) {
111 auto normalized = detail::simplify_expression(*coefficient, context_);
112 if (!normalized) { return Result<PeriodEvidence>::failure(normalized.error()); }
113 *coefficient = std::move(normalized.value());
114 }
115 const auto& slope = affine.value()->slope;
116 const auto& offset = affine.value()->offset;
117 for (const auto& coefficient : {slope, offset}) {
118 auto real = real_defined(detail::node(*coefficient));
119 if (!real) { return Result<PeriodEvidence>::failure(real.error()); }
120 if (real.value() != Tribool::True || !independent(detail::node(*coefficient)))
121 { return PeriodEvidence{}; }
122 }
123 auto nonzero = detail::query_nonzero_value(detail::node(*slope), facts_, Domain::Real, context_);
124 if (!nonzero) { return Result<PeriodEvidence>::failure(nonzero.error()); }
125 return period_from_slope(node, type, slope, nonzero.value());
126 }

128 Result<PeriodEvidence> period_from_slope(const Node& node, FunctionNode::FuncType type, const std::shared_ptr<SymbolicExpr>& slope, Tribool nonzero) {
129 if (nonzero == Tribool::False) {
130 auto constant_domain = detail::query_definedness(node, facts_, Domain::Real, context_);
131 if (!constant_domain) { return Result<PeriodEvidence>::failure(constant_domain.error()); }
132 if (constant_domain.value() == Tribool::True)
133 { return PeriodEvidence{Tribool::True, std::nullopt, true}; }
134 return PeriodEvidence{};
135 }
136 if (nonzero != Tribool::True) { return PeriodEvidence{}; }
137 auto positive = detail::query_positive_value(detail::node(*slope), facts_, context_);
138 if (!positive) { return Result<PeriodEvidence>::failure(positive.error()); }
139 std::shared_ptr<SymbolicExpr> magnitude;
140 if (positive.value() == Tribool::True) {
141 magnitude = slope;
142 } else if (positive.value() == Tribool::False) {
143 magnitude = SymbolicExpr::multiply(SymbolicExpr::number(-1), slope);
144 } else {
145 magnitude = detail::make_expression_ptr(detail::make_node<FunctionNode>(
146 FunctionNode::FuncType::Abs, std::vector<Node>{detail::node(*slope)}));
147 }
148 auto exact_pi = detail::constant_expression("pi");
149 if (!exact_pi) { return Result<PeriodEvidence>::failure(exact_pi.error()); }
150 auto numerator = type == FunctionNode::FuncType::Tan ? exact_pi.value() :
151 SymbolicExpr::multiply(SymbolicExpr::number(2), exact_pi.value());
152 auto period = detail::simplify_expression(SymbolicExpr::divide(numerator, magnitude), context_);
153 if (!period) { return Result<PeriodEvidence>::failure(period.error()); }
154 return PeriodEvidence{Tribool::True, *period.value(), true};
155 }

157 bool independent(const Node& node) const {
158 detail::RewriteBudget budget(context_, context_.limits().max_recursion_depth,
159 std::numeric_limits<std::size_t>::max(), periodic_operation);
160 const auto names = free_variables(node, &budget);
161 for (const auto& name : names) {
162 auto step = context_.consume_steps(1, periodic_operation);
163 if (!step) throw step.error();
164 if (name == variable_ || assumptions_.has_period_declarations(name)) return false;
165 }
166 return true;
167 }

172 return detail::query_real_value(node, facts_, context_);

183Result<PeriodEvidence> periodicity_checked(
184 const SymbolicExpr& expression, const std::string& variable,
185 const AssumptionContext& assumptions, const InferenceEngine& engine,
186 ComputationContext& context) {
187 if (variable.empty()) {
188 return Result<PeriodEvidence>::failure(CasErrc::InvalidArgument,
189 "independent variable must not be empty", periodic_operation);
190 }
191 return checked_inference_result<PeriodEvidence>(expression, periodic_operation, context,
192 [&]() -> Result<PeriodEvidence> {
193 PeriodInference inference(assumptions, engine, variable, context);
194 return inference.infer(detail::node(expression));
