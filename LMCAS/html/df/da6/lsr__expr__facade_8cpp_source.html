909ExprMatchResult expr_match(const ExprPtr& pattern,
910 const ExprPtr& target,
911 const std::vector<std::string>& wildcards,
912 ComputationContext& context) {
913 auto step = context.consume_steps(1, kExprMatchOperation);
914 if (!step) return ExprMatchResult::failure(step.error());
915 if (!pattern || !lamina::detail::node(pattern)) {
916 return expr_match_failure(CasErrc::InvalidArgument,
917 "match pattern cannot be null",
918 kExprMatchOperation);
919 }
920 if (!target || !lamina::detail::node(target)) {
921 return expr_match_failure(CasErrc::InvalidArgument,
922 "match target cannot be null",
923 kExprMatchOperation);
924 }
925
926 std::unordered_set<std::string> wildcard_set;
927 wildcard_set.reserve(wildcards.size());
928 for (const auto& wildcard : wildcards) {
929 if (wildcard.empty()) {
930 return expr_match_failure(CasErrc::InvalidArgument,
931 "wildcard names cannot be empty",
932 kExprMatchOperation);
933 }
934 if (!wildcard_set.insert(wildcard).second) {
935 return expr_match_failure(CasErrc::InvalidArgument,
936 "wildcard names must be unique",
937 kExprMatchOperation);
938 }
939 }
940
941 try {
942 MatchMap raw_bindings;
943 const bool matched = Matcher::match(
944 *pattern, *target, wildcard_set, raw_bindings);
945 if (!matched) {
946 return ExprMatchResult::success(ExprMatch{false, {}});
947 }
948
949 std::vector<std::string> names;
950 names.reserve(raw_bindings.size());
951 for (const auto& binding : raw_bindings) {
952 names.push_back(binding.first);
953 }
954 std::sort(names.begin(), names.end());
955
956 ExprMatch result;
957 result.matched = true;
958 result.bindings.reserve(names.size());
959 for (const auto& name : names) {
960 const auto& value = raw_bindings.at(name);
961 auto node = lamina::detail::node(value);
962 if (!node) {
963 return expr_match_failure(CasErrc::InternalInvariant,
964 "matcher produced a null binding",
965 kExprMatchOperation);
966 }
967 result.bindings.push_back(
968 ExprMatchBinding{name,
969 lamina::detail::make_expression_ptr(node)});
970 }
971 return ExprMatchResult::success(std::move(result));
972 } catch (const std::bad_alloc&) {
973 return expr_match_failure(CasErrc::ResourceLimit,
974 "expression matching allocation failed",
975 kExprMatchOperation);
976 } catch (const std::exception& error) {
977 return expr_match_failure(CasErrc::InternalInvariant, error.what(),
978 kExprMatchOperation);
979 }
980}
