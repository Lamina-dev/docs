17std::shared_ptr<SymbolicExpr> substitution_primitive(
18 const SymbolicExpr& candidate_term, const SymbolicExpr& u) {
19 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(LMCAS::detail::node(candidate_term))) {
20 auto n = LMCAS::detail::expression_from_node(pow->exponent());
21 auto np1 = SymbolicExpr::add(detail::make_expression_ptr(n), SymbolicExpr::number(1))->simplify();
22 if (np1->is_zero()) {
23 return SymbolicExpr::ln(detail::make_expression_ptr(u));
24 } else {
25 return SymbolicExpr::divide(
26 SymbolicExpr::power(detail::make_expression_ptr(u), np1), np1);
27 }
28 } else if (auto func = std::dynamic_pointer_cast<const FunctionNode>(LMCAS::detail::node(candidate_term))) {
29 if (func->type() == FunctionNode::FuncType::Cos) {
30 return SymbolicExpr::sin(detail::make_expression_ptr(u));
31 } else if (func->type() == FunctionNode::FuncType::Sin) {
32 return SymbolicExpr::multiply(SymbolicExpr::number(-1), SymbolicExpr::cos(detail::make_expression_ptr(u)));
33 } else if (func->type() == FunctionNode::FuncType::Exp) {
34 return SymbolicExpr::exp(detail::make_expression_ptr(u));
35 }
36 }
37 return nullptr;
38}

40Result<std::shared_ptr<SymbolicExpr>> substitute_candidate(
41 const SymbolicExpr& expr, const SymbolicExpr& candidate_term,
42 const std::string& var, ComputationContext& computation) {
43 auto u = substitution_argument(candidate_term);
44 if (!u || !depends_on_integration_variable(*u, var)) { return nullptr; }
45 auto d_ptr = u->differentiate(var);
46 if (!d_ptr) { return nullptr; }
47 auto du = d_ptr->simplify();
48 if (du->is_zero()) { return nullptr; }
49
50 auto f_u = candidate_term;
51 auto term_times_du = SymbolicExpr::multiply(detail::make_expression_ptr(f_u), detail::make_expression_ptr(*du));
52 auto ratio = SymbolicExpr::divide(detail::make_expression_ptr(expr), term_times_du)->simplify();
53 bool ratio_independent = !depends_on_integration_variable(*ratio, var);
54 if (!ratio_independent && computation.assumptions()) {
55 auto nonzero =
56 computation.assumptions()->is_nonzero(*du);
57 if (!nonzero) {
58 return Result<std::shared_ptr<SymbolicExpr>>::failure(
59 nonzero.error());
60 }
61 if (nonzero.value() == Tribool::True) {
62 ratio_independent = !depends_on_integration_variable(*ratio, var);
63 }
64 }
65 if (!ratio_independent) { return nullptr; }
66 auto primitive = substitution_primitive(candidate_term, *u);
67 if (!primitive) { return nullptr; }
68 return SymbolicExpr::multiply(ratio, primitive);
69}

72Result<std::shared_ptr<SymbolicExpr>> SubstitutionStrategy::try_integrate_raw(
73 const SymbolicExpr& expr, const std::string& var, Integrator&,
74 ComputationContext& computation, int) {
75 std::vector<std::shared_ptr<const SymbolicNode>> ops;
76 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(LMCAS::detail::node(expr))) {
77 ops = mul->operands();
78 } else {
79 ops.push_back(LMCAS::detail::node(expr));
80 }
81 for (const auto& operand : ops) {
82 auto candidate = detail::expression_from_node(operand);
83 auto result = substitute_candidate(expr, candidate, var, computation);
84 if (!result || result.value()) { return result; }
85 }
86 return nullptr;

108LinearWrapper classify_linear_wrapper(const SymbolicExpr& expr, const std::string& var) {
109 LinearWrapper wrapper;
110 wrapper.function = std::dynamic_pointer_cast<const FunctionNode>(detail::node(expr));
111 if (wrapper.function) {
112 if (!linear_substitutable_function(*wrapper.function)) { return wrapper; }
113 wrapper.argument = detail::expression_from_node(wrapper.function->arguments()[0]);
114 wrapper.kind = LinearWrapperKind::Function;
115 return wrapper;
116 }
117 wrapper.power = std::dynamic_pointer_cast<const PowerNode>(detail::node(expr));
118 if (!wrapper.power) { return wrapper; }
119 if (expression_depends_on_variable(wrapper.power->exponent(), var)) { return wrapper; }
120 wrapper.function = std::dynamic_pointer_cast<const FunctionNode>(wrapper.power->base());
121 if (wrapper.function) {
122 if (!linear_substitutable_function(*wrapper.function)) { return wrapper; }
123 wrapper.argument = detail::expression_from_node(wrapper.function->arguments()[0]);
124 wrapper.kind = LinearWrapperKind::PowerOfFunction;
125 } else {
126 wrapper.argument = detail::expression_from_node(wrapper.power->base());
127 wrapper.kind = LinearWrapperKind::PowerOfLinear;
128 }
129 return wrapper;
130}

148Result<bool> LinearSubstitutionStrategy::extract_linear_arg(
149 const SymbolicExpr& arg,
150 const std::string& var,
151 std::shared_ptr<SymbolicExpr>& a_out,
152 std::shared_ptr<SymbolicExpr>& b_out) {
153 if (!expression_depends_on_variable(LMCAS::detail::node(arg), var)) { return false; }
154
155 auto converted = symbolic_to_poly<SymbolicPolyCoeff>(detail::make_expression_ptr(arg), var);
156 if (!converted) {
157 if (converted.error().code == CasErrc::UnsupportedExpression) { return false; }
158 return Result<bool>::failure(converted.error());
159 }
160 auto poly = std::move(converted.value());
161 if (poly.degree() != 1) { return false; }
162 if (poly.coeffs.size() < 2) { return false; }
163
164 auto a_expr = poly.coeffs[1].val;
165 auto b_expr = poly.coeffs[0].val;
166 if (!a_expr || !b_expr) { return false; }
167 if (expression_depends_on_variable(LMCAS::detail::node(a_expr), var)) { return false; }
168 if (expression_depends_on_variable(LMCAS::detail::node(b_expr), var)) { return false; }
169
170 auto a_simp = a_expr->simplify();
171 auto b_simp = b_expr->simplify();
172 if (!a_simp) a_simp = a_expr;
173 if (!b_simp) b_simp = b_expr;
174 if (a_simp->is_zero()) { return false; }
175
176 a_out = a_simp;
177 b_out = b_simp;
178 return true;
179}

181Result<std::shared_ptr<SymbolicExpr>> LinearSubstitutionStrategy::try_integrate_raw(
182 const SymbolicExpr& expr, const std::string& var, Integrator& ctx,
183 ComputationContext& computation, int depth) {
184 auto wrapper = classify_linear_wrapper(expr, var);
185 auto& arg_expr = wrapper.argument;
186 std::shared_ptr<SymbolicExpr> a_coeff, b_coeff;
187 if (!arg_expr) { return nullptr; }
188 auto extracted = extract_linear_arg(*arg_expr, var, a_coeff, b_coeff);
189 if (!extracted) {
190 return Result<std::shared_ptr<SymbolicExpr>>::failure(extracted.error());
191 }
192 if (!extracted.value()) { return nullptr; }
193
194 /**
195 * @brief arg == var（a = 1，b = 0）由前序 TableLookup 或 PowerRule 处理。
196 * 此处返回 nullptr，将控制权交给下一策略。
197 */
198 bool a_is_one = a_coeff && a_coeff->is_one();
199 bool b_is_zero = b_coeff && b_coeff->is_zero();
200 if (a_is_one && b_is_zero) { return nullptr; }
201
202 const std::string dummy_name = "__lin_sub_u__";
203 auto dummy_var = SymbolicExpr::variable(dummy_name);
204 auto test_expr = linear_dummy_expression(wrapper, dummy_var);
205 if (!test_expr) { return nullptr; }
206
207 TableLookupStrategy table_only;
208 auto table_attempt = table_only.try_integrate(
209 *test_expr, dummy_name, ctx, computation, depth);
210 if (!table_attempt) {
211 return Result<std::shared_ptr<SymbolicExpr>>::failure(
212 table_attempt.error());
213 }
214 auto* table_candidate =
215 std::get_if<IntegrationCandidate>(&table_attempt.value());
216 if (!table_candidate || !table_candidate->expression) { return nullptr; }
217 auto F_substituted = table_candidate->expression->substitute(
218 dummy_name, detail::make_expression_ptr(*arg_expr));
219 if (!F_substituted) { return nullptr; }
220
221 auto inv_a = SymbolicExpr::power(a_coeff, SymbolicExpr::number(-1));
222 auto result = SymbolicExpr::multiply(inv_a, F_substituted);
223
224 auto simplified = result->simplify();
225 if (simplified && !contains_unevaluated_integral(LMCAS::detail::node(simplified))) {
226 return simplified;
227 }
228 return result;
229}
