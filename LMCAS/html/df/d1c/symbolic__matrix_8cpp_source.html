126ExpressionResult matrix_inverse_checked(const std::shared_ptr<SymbolicExpr>& A,
127 ComputationContext& context) {
128 const std::string operation = "matrix_inverse";
129 auto mat = require_square_matrix(A, context, operation);
130 if (!mat) return ExpressionResult::failure(mat.error());
131 auto determinant_budget =
132 context.consume_steps(mat.value()->rows() * mat.value()->cols() + 1, operation);
133 if (!determinant_budget) return ExpressionResult::failure(determinant_budget.error());
134
135 auto det = SymbolicExpr::determinant(A);
136 if (!det || !lamina::detail::node(det)) {
137 return ExpressionResult::failure(
138 CasErrc::InternalInvariant,
139 "matrix determinant could not be constructed",
140 operation);
141 }
142 auto det_simplified = det->simplify();
143 if (!det_simplified || !lamina::detail::node(det_simplified)) {
144 return ExpressionResult::failure(
145 CasErrc::InternalInvariant,
146 "matrix determinant simplification produced an empty expression",
147 operation);
148 }
149 if (lamina::detail::node(det_simplified)->is_zero()) {
150 return ExpressionResult::failure(
151 CasErrc::DomainError,
152 "matrix is singular and has no inverse",
153 operation);
154 }
155 if (!matrix_is_provably_nonzero_number(det_simplified)) {
156 return ExpressionResult::failure(
157 CasErrc::Inconclusive,
158 "matrix determinant is not provably non-zero in the current support domain",
159 operation);
160 }
161
162 auto inverse_budget =
163 context.consume_steps(mat.value()->rows() * mat.value()->cols() * 4 + 1, operation);
164 if (!inverse_budget) return ExpressionResult::failure(inverse_budget.error());
165 return require_matrix_result(SymbolicExpr::inverse(A), operation);
166}

256MatrixEigenvalueResult matrix_eigenvalues_checked(const std::shared_ptr<SymbolicExpr>& A,
257 ComputationContext& context) {
258 const std::string operation = "matrix_eigenvalues";
259 auto mat_result = require_square_matrix(A, context, operation);
260 if (!mat_result) return MatrixEigenvalueResult::failure(mat_result.error());
261 auto mat = mat_result.value();
262
263 bool triangular = true;
264 for (size_t r = 0; r < mat->rows() && triangular; ++r) {
265 for (size_t c = 0; c < r; ++c) {
266 auto below = lamina::detail::make_expression_ptr(mat->get(r, c))->simplify();
267 if (!lamina::detail::node(below)->is_zero()) {
268 triangular = false;
269 break;
270 }
271 }
272 }
273 if (triangular) {
274 std::vector<std::shared_ptr<SymbolicExpr>> values;
275 values.reserve(mat->rows());
276 for (size_t i = 0; i < mat->rows(); ++i) {
277 values.push_back(lamina::detail::make_expression_ptr(mat->get(i, i))->simplify());
278 }
279 return MatrixEigenvalueResult::success(std::move(values));
280 }
281
282 auto values_matrix = SymbolicExpr::eigenvalues(A);
283 auto values_node = values_matrix ? std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(values_matrix)) : nullptr;
284 if (!values_node || values_node->rows() == 0) {
285 return MatrixEigenvalueResult::failure(
286 CasErrc::UnsupportedExpression,
287 "eigenvalue computation did not produce a matrix result",
288 operation);
289 }
290
291 std::vector<std::shared_ptr<SymbolicExpr>> values;
292 for (size_t c = 0; c < values_node->cols(); ++c) {
293 values.push_back(lamina::detail::make_expression_ptr(values_node->get(0, c))->simplify());
294 }

309MatrixEigenvectorResult matrix_eigenvectors_checked(const std::shared_ptr<SymbolicExpr>& A,
310 ComputationContext& context) {
311 const std::string operation = "matrix_eigenvectors";
312 auto mat_result = require_square_matrix(A, context, operation);
313 if (!mat_result) return MatrixEigenvectorResult::failure(mat_result.error());
314 auto mat = mat_result.value();
315
316 if (mat->rows() == 2) {
317 auto a = lamina::detail::make_expression_ptr(mat->get(0, 0))->simplify();
318 auto b = lamina::detail::make_expression_ptr(mat->get(0, 1))->simplify();
319 auto lower = lamina::detail::make_expression_ptr(mat->get(1, 0))->simplify();
320 auto d = lamina::detail::make_expression_ptr(mat->get(1, 1))->simplify();
321 auto delta = SymbolicExpr::add(d, SymbolicExpr::multiply(SymbolicExpr::number(-1), a))->simplify();
322 if (lamina::detail::node(lower)->is_zero() && !lamina::detail::node(delta)->is_zero()) {
323 return MatrixEigenvectorResult::success({
324 {SymbolicExpr::number(1), SymbolicExpr::number(0)},
325 {b, delta}
326 });
327 }
328 }
329
330 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> vectors;
331 auto pairs = SymbolicExpr::eigenvectors(A);
332 for (const auto& [lambda, vector_matrices] : pairs) {
333 (void)lambda;
334 for (const auto& vector_matrix : vector_matrices) {
335 auto vec_node = vector_matrix ? std::dynamic_pointer_cast<const MatrixNode>(lamina::detail::node(vector_matrix)) : nullptr;
336 if (!vec_node || vec_node->cols() != 1) continue;
337 std::vector<std::shared_ptr<SymbolicExpr>> vector;
338 for (size_t r = 0; r < vec_node->rows(); ++r) {
339 vector.push_back(lamina::detail::make_expression_ptr(vec_node->get(r, 0))->simplify());
340 }
341 vectors.push_back(std::move(vector));
342 }
343 }
344 if (vectors.empty()) {
345 return MatrixEigenvectorResult::failure(
346 CasErrc::UnsupportedExpression,
347 "eigenvector computation did not produce supported vectors",
348 operation);
349 }
350 return MatrixEigenvectorResult::success(std::move(vectors));
