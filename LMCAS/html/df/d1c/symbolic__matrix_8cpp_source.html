9ExpressionResult matrix_multiply_checked(const std::shared_ptr<SymbolicExpr>& A,
10 const std::shared_ptr<SymbolicExpr>& B,
11 ComputationContext& context) {
12 const std::string operation = "matrix_multiply";
13 auto left = require_matrix(A, context, operation);
14 if (!left) return ExpressionResult::failure(left.error());
15 auto right = require_matrix(B, context, operation);
16 if (!right) return ExpressionResult::failure(right.error());
17 if (left.value()->cols() != right.value()->rows()) {
18 return ExpressionResult::failure(
19 CasErrc::InvalidArgument, "matrix dimensions are not compatible", operation);
20 }
21 try {
22 detail::RewriteBudget budget(context, context.limits().max_recursion_depth,
23 context.limits().max_ast_nodes, "matrix_multiply");
24 const auto count = normalization_check_count(
25 &budget, left.value()->rows(), right.value()->cols());
26 auto output_terms = context.require_expansion_terms(count, operation);
27 if (!output_terms) { return ExpressionResult::failure(output_terms.error()); }
28 auto inner_terms = context.require_expansion_terms(left.value()->cols(), operation);
29 if (!inner_terms) { return ExpressionResult::failure(inner_terms.error()); }
30 normalization_check_arithmetic<MultiplyNode>(&budget, 0, left.value(), right.value());
31 auto product = detail::make_node<MultiplyNode>(
32 std::vector<std::shared_ptr<const SymbolicNode>>{left.value(), right.value()});
33 NormalizationVisitor visitor(context, detail::no_facts(), Domain::Real, &budget);
34 product->accept(visitor);
35 auto result = visitor.get_result();
36 budget.measure(result);
37 if (!std::dynamic_pointer_cast<const MatrixNode>(result)) {
38 return ExpressionResult::failure(CasErrc::InternalInvariant,
39 "matrix multiplication did not produce an explicit matrix", operation);
40 }
41 return detail::make_expression_ptr(std::move(result));
42 } catch (const CasError& error) {
43 return ExpressionResult::failure(error);
44 } catch (const std::bad_alloc&) {
45 return ExpressionResult::failure(CasErrc::ResourceLimit,
46 "allocation failed while multiplying matrices", operation);
47 } catch (const std::exception& ex) {
48 return ExpressionResult::failure(CasErrc::InternalInvariant, ex.what(), operation);
49 }
50}
