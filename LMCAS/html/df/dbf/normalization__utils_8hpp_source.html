50template <class Parent, class... Nodes>
51std::size_t normalization_check_arithmetic(detail::RewriteBudget* budget,
52 std::size_t extra_nodes, const std::shared_ptr<const Nodes>&... children) {
53 if (!budget) {
54 return 0;
55 }
56 std::size_t nodes = extra_nodes, flattened = 0;
57 const auto append = [&](const auto& child) {
58 nodes = budget->append_size(nodes, budget->measure(child));
59 if (dynamic_cast<const Parent*>(child.get())) {
60 ++flattened;
61 }
62 };
63 (append(children), ...);
64 return budget->append_size(1, nodes - flattened);
65}
