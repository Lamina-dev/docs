18 * @brief 计算 AST 节点的多项式次数
19 * @param node 输入节点
20 * @return 节点对应的多项式次数
21 */
22inline int get_node_degree_helper(const std::shared_ptr<const SymbolicNode>& node) {
23 if (!node) return 0;
24 if (std::dynamic_pointer_cast<const VariableNode>(node)) return 1;
25 if (auto p = std::dynamic_pointer_cast<const PowerNode>(node)) {
26 if (auto e = std::dynamic_pointer_cast<const NumberNode>(p->exponent())) {
27 if (std::holds_alternative<BigInt>(e->value())) return (int)std::get<BigInt>(e->value()).to_int();
28 if (std::holds_alternative<lmmc_real_t>(e->value())) return (int)std::get<lmmc_real_t>(e->value());
29 }
30 return 1;
31 }
32 if (auto m = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
33 int d = 0;
34 for (auto& op : m->operands()) d += get_node_degree_helper(op);
35 return d;
36 }
37 return 0;
38}

59/** @brief 节点比较器，按多项式次数降序排列，用于同类项合并 */
60struct NodeCompare { bool operator()(const std::shared_ptr<const SymbolicNode>& lhs, const std::shared_ptr<const SymbolicNode>& rhs) const {
61 if (!lhs && !rhs) return false;
62 if (!lhs) return true;
63 if (!rhs) return false;
64
65 int d1 = get_node_degree_helper(lhs);
66 int d2 = get_node_degree_helper(rhs);
67 if (d1 != d2) return d1 > d2;
68
69 bool isNum1 = std::dynamic_pointer_cast<const NumberNode>(lhs) != nullptr;
70 bool isNum2 = std::dynamic_pointer_cast<const NumberNode>(rhs) != nullptr;
71 if (isNum1 != isNum2) return isNum1;
72
73 if (lhs->type_priority() != rhs->type_priority()) {
74 return lhs->type_priority() < rhs->type_priority();
75 }
76
77 return lhs->compare(*rhs) < 0;
78 }
79};

141 * @param arg 待检查的节点
142 * @param out_positive 输出参数，存放取正后的节点
143 * @return 若节点为负则返回 true

145inline bool check_negative_arg(const std::shared_ptr<const SymbolicNode>& arg, std::shared_ptr<const SymbolicNode>& out_positive) {
146 if (auto num = std::dynamic_pointer_cast<const NumberNode>(arg)) {
147 if (std::holds_alternative<lmmc_real_t>(num->value()) && std::get<lmmc_real_t>(num->value()) < 0) {
148 out_positive = LMCAS::detail::make_node<NumberNode>(std::abs(std::get<lmmc_real_t>(num->value())));
149 return true;
150 }
151 if (std::holds_alternative<BigInt>(num->value()) &&
152 std::get<BigInt>(num->value()).IsNegative()) {
153 out_positive = LMCAS::detail::make_node<NumberNode>(std::get<BigInt>(num->value()) * BigInt(-1));
154 return true;
155 }
156 if (std::holds_alternative<Rational>(num->value()) &&
157 std::get<Rational>(num->value()).get_numerator().IsNegative()) {
158 out_positive = LMCAS::detail::make_node<NumberNode>(std::get<Rational>(num->value()) * Rational(-1));
159 return true;
160 }
161 return false;
162 }
163 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(arg)) {
164 if (!mul->operands().empty()) {
165 if (auto num = std::dynamic_pointer_cast<const NumberNode>(mul->operands()[0])) {
166 bool is_neg = false;
167 std::shared_ptr<const NumberNode> pos_num = nullptr;
168 if (std::holds_alternative<lmmc_real_t>(num->value()) && std::get<lmmc_real_t>(num->value()) < 0) {
169 is_neg = true;
170 pos_num = LMCAS::detail::make_node<NumberNode>(std::abs(std::get<lmmc_real_t>(num->value())));
171 } else if (std::holds_alternative<BigInt>(num->value()) &&
172 std::get<BigInt>(num->value()).IsNegative()) {
173 is_neg = true;
174 pos_num = LMCAS::detail::make_node<NumberNode>(std::get<BigInt>(num->value()) * BigInt(-1));
175 } else if (std::holds_alternative<Rational>(num->value()) &&
176 std::get<Rational>(num->value()).get_numerator().IsNegative()) {
177 is_neg = true;
178 pos_num = LMCAS::detail::make_node<NumberNode>(std::get<Rational>(num->value()) * Rational(-1));
179 }
180
181 if (is_neg) {
182 std::vector<std::shared_ptr<const SymbolicNode>> new_ops = mul->operands();
183
184 bool is_minus_one = false;
185 if (std::holds_alternative<lmmc_real_t>(num->value())) {
186 int eq_1;
187 lmmc_double_nearly_equal_tol(std::get<lmmc_real_t>(num->value()), -1.0, 1e-9, 1e-9, &eq_1);
188 is_minus_one = (eq_1 != 0);
189 }
190 else if (std::holds_alternative<BigInt>(num->value())) is_minus_one = (std::get<BigInt>(num->value()) == BigInt(-1));
191 else if (std::holds_alternative<Rational>(num->value())) is_minus_one = (std::get<Rational>(num->value()) == Rational(-1));
192
193 if (is_minus_one) {
194 new_ops.erase(new_ops.begin());
195 if (new_ops.empty()) out_positive = LMCAS::detail::make_node<NumberNode>(BigInt(1));
196 else if (new_ops.size() == 1) out_positive = new_ops[0];
197 else out_positive = make_normalized_multiply_node(new_ops);
198 } else {
199
200 new_ops[0] = pos_num;
201 out_positive = make_normalized_multiply_node(new_ops);
202 }
203 return true;
204 }
205 }
206 }
207 }
208 return false;
209}

211/**
212 * @brief 提取节点中 π 的有理系数（如 2π/3 中的 2/3）
213 * @param node 待检查的节点
214 * @param k 输出参数，存放 π 的有理系数
215 * @return 若节点为 k*π 形式则返回 true
216 */
217inline bool get_pi_coeff(const std::shared_ptr<const SymbolicNode>& node, Rational& k) {
218 if (auto v = std::dynamic_pointer_cast<const VariableNode>(node)) {
219 if (v->name() == "pi") {
220 k = Rational(1);
221 return true;
222 }
223 }
224 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
225 bool has_pi = false;
226 k = Rational(1);
227
228 for (const auto& op : mul->operands()) {
229 if (auto v = std::dynamic_pointer_cast<const VariableNode>(op)) {
230 if (v->name() == "pi") {
231 if (has_pi) return false;
232 has_pi = true;
233 } else return false;
234 } else if (auto n = std::dynamic_pointer_cast<const NumberNode>(op)) {
235 if (std::holds_alternative<Rational>(n->value())) k = k * std::get<Rational>(n->value());
236 else if (std::holds_alternative<BigInt>(n->value())) k = k * Rational(std::get<BigInt>(n->value()));
237 else return false;
238 } else return false;
239 }
240 return has_pi;
241 }
242 return false;
243}
