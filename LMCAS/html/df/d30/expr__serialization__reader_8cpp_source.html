31 std::size_t read_length() {
32 std::size_t length = 0;
33 while (position_ < source_.size() && source_[position_] >= '0' &&
34 source_[position_] <= '9') {
35 const auto digit = static_cast<std::size_t>(source_[position_++] - '0');
36 if (length > (std::numeric_limits<std::size_t>::max() - digit) / 10)
37 invalid("serialized expression length overflow");
38 length = length * 10 + digit;
39 }
40 return length;
41 }
42 std::string_view field() {
43 check(context_.consume_steps(1, kDecode));
44 const auto begin = position_;
45 const auto length = read_length();
46 if (begin == position_) {
47 invalid("invalid serialized expression length");
48 }
49 if (position_ == source_.size()) {
50 invalid("invalid serialized expression length");
51 }
52 if (source_[position_++] != ':') {
53 invalid("invalid serialized expression length");
54 }
55 if (position_ - begin > 2 && source_[begin] == '0') {
56 invalid("invalid serialized expression length");
57 }
58 if (length >= source_.size() - position_) {
59 invalid("truncated serialized expression field");
60 }
61 if (source_[position_ + length] != ',') {
62 invalid("truncated serialized expression field");
63 }
64 auto result = std::string_view(source_).substr(position_, length);
65 position_ += length + 1;
66 return result;
67 }
68 std::size_t unsigned_field() {
69 const auto text = field();
70 std::size_t result = 0;
71 auto parsed = std::from_chars(text.data(), text.data() + text.size(), result);
72 if (text.empty() || (text.size() > 1 && text[0] == '0') ||
73 parsed.ec != std::errc{} || parsed.ptr != text.data() + text.size())
74 invalid("invalid serialized expression integer");
75 return result;

102 BigInt integer() {
103 auto text = field();
104 if (text.empty()) {
105 invalid("invalid serialized expression integer");
106 }
107 std::size_t first = text.front() == '-' ? 1 : 0;
108 if (first == text.size()) {
109 invalid("invalid serialized expression integer");
110 }
111 if (text.size() - first > 1 && text[first] == '0') {
112 invalid("invalid serialized expression integer");
113 }
114 check_integer_digits(text, first);
115 if (first != 0 && text.substr(first) == "0") {
116 invalid("invalid serialized expression integer");
117 }
118 // Decimal length gives an inexpensive lower bound before allocating BigInt.
119 if (text.size() - first > context_.limits().max_integer_bits / 3 + 1) {
120 throw CasError{CasErrc::ResourceLimit, "integer bit budget exceeded", kDecode};
121 }
122 BigInt result{std::string(text)};
123 check(context_.require_integer_bits(result.bit_length(), kDecode));
124 return result;
125 }

291LMCAS_API ExpressionResult parse_serialized_expr(const std::string& source,
292 ComputationContext& context) {
293 auto input = context.require_input_bytes(source.size(), kDecode);
294 if (!input) return ExpressionResult::failure(input.error());
295 if (source.compare(0, kHeader.size(), kHeader.data(), kHeader.size()) != 0)
296 return ExpressionResult::failure(CasErrc::ParseError,
297 "unknown serialized expression version", kDecode);
298 try {
299 return detail::make_expression_ptr(Reader(source, context).finish());
300 } catch (const CasError& error) {
301 return ExpressionResult::failure(error);
302 } catch (const std::bad_alloc&) {
303 return ExpressionResult::failure(CasErrc::ResourceLimit,
304 "expression parse allocation failed", kDecode);
305 } catch (const std::exception& error) {
306 return ExpressionResult::failure(CasErrc::ParseError, error.what(), kDecode);
307 }
308}
