33 std::size_t read_length() {
34 std::size_t length = 0;
35 while (position_ < source_.size() && source_[position_] >= '0' &&
36 source_[position_] <= '9') {
37 const auto digit = static_cast<std::size_t>(source_[position_++] - '0');
38 if (length > (std::numeric_limits<std::size_t>::max() - digit) / 10)
39 invalid("serialized expression length overflow");
40 length = length * 10 + digit;
41 }
42 return length;
43 }
44 std::string_view field() {
45 check(context_.consume_steps(1, kDecode));
46 const auto begin = position_;
47 const auto length = read_length();
48 if (begin == position_) {
49 invalid("invalid serialized expression length");
50 }
51 if (position_ == source_.size()) {
52 invalid("invalid serialized expression length");
53 }
54 if (source_[position_++] != ':') {
55 invalid("invalid serialized expression length");
56 }
57 if (position_ - begin > 2 && source_[begin] == '0') {
58 invalid("invalid serialized expression length");
59 }
60 if (length >= source_.size() - position_) {
61 invalid("truncated serialized expression field");
62 }
63 if (source_[position_ + length] != ',') {
64 invalid("truncated serialized expression field");
65 }
66 auto result = std::string_view(source_).substr(position_, length);
67 position_ += length + 1;
68 return result;
69 }
70 std::size_t unsigned_field() {
71 const auto text = field();
72 std::size_t result = 0;
73 auto parsed = std::from_chars(text.data(), text.data() + text.size(), result);
74 if (text.empty() || (text.size() > 1 && text[0] == '0') ||
75 parsed.ec != std::errc{} || parsed.ptr != text.data() + text.size())
76 invalid("invalid serialized expression integer");
77 return result;
78 }
79 std::size_t count() {

83 if (n > context_.limits().max_ast_nodes)

85 return n;

100 void check_integer_digits(std::string_view text, std::size_t first) const {
101 for (std::size_t i = first; i < text.size(); ++i)
102 if (text[i] < '0' || text[i] > '9') invalid("invalid serialized expression integer");

104 BigInt integer() {
105 auto text = field();
106 if (text.empty()) {
107 invalid("invalid serialized expression integer");
108 }
109 std::size_t first = text.front() == '-' ? 1 : 0;
110 if (first == text.size()) {
111 invalid("invalid serialized expression integer");
112 }
113 if (text.size() - first > 1 && text[first] == '0') {
114 invalid("invalid serialized expression integer");
115 }
116 check_integer_digits(text, first);
117 if (first != 0 && text.substr(first) == "0") {
118 invalid("invalid serialized expression integer");
119 }
120 // Decimal length gives an inexpensive lower bound before allocating BigInt.
121 if (text.size() - first > context_.limits().max_integer_bits / 3 + 1) {
122 throw CasError{CasErrc::ResourceLimit, "integer bit budget exceeded", kDecode};
123 }
124 BigInt result{std::string(text)};
125 check(context_.require_integer_bits(result.bit_length(), kDecode));
126 return result;
127 }
128 Rational rational() {

132 return Rational(numerator, denominator);

141 int hex_digit(char c) const {
142 if (c >= '0' && c <= '9') return c - '0';
143 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
144 invalid("invalid binary64 bits");
145 }
146 NodePtr binary64() {
147 auto text = field();
148 if (text.size() != 16) invalid("invalid binary64 bits");
149 std::uint64_t bits = 0;
150 for (char c : text) {
151 const auto digit = hex_digit(c);
152 bits = bits * 16 + static_cast<unsigned>(digit);
153 }
154 double value;
155 std::memcpy(&value, &bits, sizeof(value));
156 if (!std::isfinite(value)) invalid("nonfinite binary64 value");
157 return detail::make_node<NumberNode>(value);
158 }
159 NodePtr matrix() {
160 auto rows = unsigned_field(), cols = unsigned_field();
161 if (!rows || !cols || rows > std::numeric_limits<std::size_t>::max() / cols ||
162 rows * cols > source_.size() - position_)
163 invalid("invalid matrix dimensions");
164 if (rows * cols > context_.limits().max_ast_nodes)
165 throw CasError{CasErrc::ResourceLimit, "AST node budget exhausted", kDecode};
166 MatrixNode::DenseStorage elements;
167 elements.reserve(rows * cols);
168 for (std::size_t i = 0; i < rows * cols; ++i) elements.push_back(node());
169 return detail::make_node<MatrixNode>(rows, cols, std::move(elements));
170 }

182 NodePtr quantity() {
183 auto dimensions = count();
184 DimensionSignature::Exponents exponents;

190 invalid("invalid dimension exponent");
191 }

193 return detail::make_node<QuantityNode>(node(), std::move(dimension), Rational(1), "");
194 }
195 NodePtr root_of() {
196 auto name = std::string(field());
197 auto index = unsigned_field(), size = count();
198 std::vector<Rational> coefficients;
199 coefficients.reserve(size);
200 for (std::size_t i = 0; i < size; ++i) coefficients.push_back(rational());
201 if (coefficients.size() < 2 || coefficients.back().is_zero() ||
202 index >= coefficients.size() - 1) invalid("invalid root identity");
203 detail::ExactRootId id{Polynomial<Rational>(coefficients), index};
204 return detail::make_node<RootOfNode>(std::move(id), std::move(name));
205 }

309LMCAS_API ExpressionResult parse_serialized_expr(const std::string& source,
310 ComputationContext& context) {
311 auto input = context.require_input_bytes(source.size(), kDecode);
312 if (!input) return ExpressionResult::failure(input.error());
313 if (source.compare(0, kHeader.size(), kHeader.data(), kHeader.size()) != 0)
314 return ExpressionResult::failure(CasErrc::ParseError,
315 "unknown serialized expression version", kDecode);
316 try {
317 return detail::make_expression_ptr(Reader(source, context).finish());
318 } catch (const CasError& error) {
319 return ExpressionResult::failure(error);
320 } catch (const std::bad_alloc&) {
321 return ExpressionResult::failure(CasErrc::ResourceLimit,
322 "expression parse allocation failed", kDecode);
323 } catch (const std::exception& error) {
324 return ExpressionResult::failure(CasErrc::ParseError, error.what(), kDecode);
325 }
