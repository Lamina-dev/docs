28Result<ExprSet> ExprSet::make(std::vector<ExprPtr> elements) {
29 ComputationContext context;
30 for (const auto& element : elements) {
31 if (!element) {
32 return expr_set_failure(CasErrc::InvalidArgument,
33 "set<Expr> elements cannot be null",
34 kExprSetOperation);
35 }
36 }
37 auto expression = make_finite_set(std::move(elements), context);
38 if (!expression) { return Result<ExprSet>::failure(expression.error()); }
39 return Result<ExprSet>::success(
40 ExprSet(std::make_shared<const SymbolicExpr>(*expression.value())));
41}
