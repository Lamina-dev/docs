26bool exact_rational_sqrt(const Rational& value, Rational& root) {
27 if (value < Rational(0)) return false;
28 const BigInt numerator_root = value.get_numerator().sqrt();
29 const BigInt denominator_root = value.get_denominator().sqrt();
30 if (numerator_root * numerator_root != value.get_numerator() ||
31 denominator_root * denominator_root != value.get_denominator()) {
32 return false;
33 }
34 root = Rational(numerator_root, denominator_root);
35 return true;
36}

38ExprPtr sqrt_rational_expression(const Rational& value) {
39 Rational root;
40 if (exact_rational_sqrt(value, root)) {
41 return rational_expression(root);
42 }
43 /// 从 num*den 提取最大完全平方因子,使 sqrt(28) 规范化为 2*sqrt(7).
44 BigInt numerator = value.get_numerator();
45 BigInt denominator = value.get_denominator();
46 BigInt outside_numerator(1);
47 BigInt outside_denominator(1);
48 for (BigInt factor(2); factor * factor <= numerator; factor = factor + BigInt(1)) {
49 while (numerator % (factor * factor) == BigInt(0)) {
50 numerator = numerator / (factor * factor);
51 outside_numerator = outside_numerator * factor;
52 }
53 }
54 for (BigInt factor(2); factor * factor <= denominator; factor = factor + BigInt(1)) {
55 while (denominator % (factor * factor) == BigInt(0)) {
56 denominator = denominator / (factor * factor);
57 outside_denominator = outside_denominator * factor;
58 }
59 }
60 const Rational inside(numerator, denominator);
61 auto radical = SymbolicExpr::sqrt(rational_expression(inside))->simplify();
62 if (!radical) {
63 return SymbolicExpr::sqrt(rational_expression(value));
64 }
65 auto scaled = SymbolicExpr::multiply(
66 rational_expression(Rational(outside_numerator, outside_denominator)),
67 radical)->simplify();
68 return scaled ? scaled : radical;
69}

83ExprResult normalize_equation_to_zero(const ExprPtr& expression,
84 const bool require_equality,
85 ComputationContext& context) {
86 if (!expression) {
87 return ExprResult::failure(CasErrc::InvalidArgument,
88 "equation cannot be null",
89 kSolveExprSetOperation);
90 }
91 const auto& node = lamina::detail::node(*expression);
92 if (const auto relation =
93 std::dynamic_pointer_cast<const RelationalNode>(node)) {
94 if (relation->op() == RelationalNode::Op::EQ) {
95 auto left = lamina::detail::make_expression_ptr(relation->left());
96 auto right = lamina::detail::make_expression_ptr(relation->right());
97 if (!left || !right) {
98 return ExprResult::failure(
99 CasErrc::InternalInvariant,
100 "equation relation has a null operand",
101 kSolveExprSetOperation);
102 }
103 auto difference = SymbolicExpr::add(
104 std::move(left),
105 SymbolicExpr::multiply(SymbolicExpr::number(-1),
106 std::move(right)));
107 if (!difference) {
108 return ExprResult::failure(
109 CasErrc::UnsupportedExpression,
110 "equation normalization produced no expression",
111 kSolveExprSetOperation);
112 }
113 auto simplified = difference->simplify();
114 if (!simplified) {
115 return ExprResult::failure(
116 CasErrc::InternalInvariant,
117 "equation normalization simplification failed",
118 kSolveExprSetOperation);
119 }
120 (void)context;
121 return ExprResult::success(std::move(simplified));
122 }
123 if (require_equality) {
124 return ExprResult::failure(
125 CasErrc::InvalidArgument,
126 "solve requires an equality relation",
127 kSolveExprSetOperation);
128 }
129 } else if (require_equality) {
130 return ExprResult::failure(CasErrc::InvalidArgument,
131 "solve requires an equality relation",
132 kSolveExprSetOperation);
133 }
134 return ExprResult::success(expression);
135}

137Result<std::optional<ExprSet>> try_lsr_closed_form_rational_poly_roots(
138 const ExprPtr& equation,
139 const std::string& variable,
140 ComputationContext& context) {
141 if (!equation) {
142 return Result<std::optional<ExprSet>>::failure(
143 CasErrc::InvalidArgument, "equation cannot be null",
144 kSolveExprSetOperation);
145 }
146 if (variable.empty()) {
147 return Result<std::optional<ExprSet>>::failure(
148 CasErrc::InvalidArgument, "solve variable cannot be empty",
149 kSolveExprSetOperation);
150 }
151
152 auto recognized = recognize_rational_polynomial(*equation, variable, context);
153 if (!recognized) {
154 return Result<std::optional<ExprSet>>::failure(recognized.error());
155 }
156 if (!recognized.value()) {
157 return Result<std::optional<ExprSet>>::success(std::nullopt);
158 }
159
160 const Polynomial<Rational>& polynomial = *recognized.value();
161 const int degree = polynomial.degree();
162 if (degree < 1 || degree > 2) {
163 return Result<std::optional<ExprSet>>::success(std::nullopt);
164 }
165
166 std::vector<ExprPtr> roots;
167 if (degree == 1) {
168 const Rational b = polynomial_coeff_or_zero(polynomial, 1);
169 if (b.is_zero()) {
170 return Result<std::optional<ExprSet>>::success(std::nullopt);
171 }
172 const Rational c = polynomial_coeff_or_zero(polynomial, 0);
173 roots.push_back(rational_expression((-c) / b)->simplify());
174 } else {
175 const Rational a = polynomial_coeff_or_zero(polynomial, 2);
176 const Rational b = polynomial_coeff_or_zero(polynomial, 1);
177 const Rational c = polynomial_coeff_or_zero(polynomial, 0);
178 if (a.is_zero()) {
179 return Result<std::optional<ExprSet>>::success(std::nullopt);
180 }
181
182 const Rational two_a = Rational(2) * a;
183 const Rational discriminant = b * b - Rational(4) * a * c;
184 const Rational real_component = (-b) / two_a;
185
186 if (discriminant < Rational(0)) {
187 const Rational positive_discriminant = -discriminant;
188 const Rational positive_denominator = two_a.abs();
189 auto imag_magnitude = SymbolicExpr::divide(
190 sqrt_rational_expression(positive_discriminant),
191 rational_expression(positive_denominator))->simplify();
192 auto positive = verified_lsr_complex(
193 rational_expression(real_component), imag_magnitude);
194 if (!positive) {
195 return Result<std::optional<ExprSet>>::failure(positive.error());
196 }
197 auto negative_imag = SymbolicExpr::multiply(
198 SymbolicExpr::number(-1), imag_magnitude)->simplify();
199 auto negative = verified_lsr_complex(
200 rational_expression(real_component), negative_imag);
201 if (!negative) {
202 return Result<std::optional<ExprSet>>::failure(negative.error());
203 }
204 roots.push_back(negative.value());
205 roots.push_back(positive.value());
206 } else {
207 auto sqrt_discriminant = sqrt_rational_expression(discriminant);
208 auto numerator_left = SymbolicExpr::add(
209 rational_expression(-b),
210 SymbolicExpr::multiply(SymbolicExpr::number(-1),
211 sqrt_discriminant))->simplify();
212 roots.push_back(SymbolicExpr::divide(
213 numerator_left, rational_expression(two_a))->simplify());
214 if (!discriminant.is_zero()) {
215 auto numerator_right = SymbolicExpr::add(
216 rational_expression(-b), sqrt_discriminant)->simplify();
217 roots.push_back(SymbolicExpr::divide(
218 numerator_right, rational_expression(two_a))->simplify());
219 }
220 }
221 }
222
223 auto set = ExprSet::make(std::move(roots));
224 if (!set) {
225 return Result<std::optional<ExprSet>>::failure(set.error());
226 }
227 return Result<std::optional<ExprSet>>::success(std::move(set.value()));
228}

255ExprSetResult solve_normalized_expr_set(const ExprPtr& normalized,
256 const std::string& variable,
257 ComputationContext& context,
258 const SolveOptions& options) {
259 auto closed_form = try_lsr_closed_form_rational_poly_roots(
260 normalized, variable, context);
261 if (!closed_form) {
262 return ExprSetResult::failure(closed_form.error());
263 }
264 if (closed_form.value()) {
265 return ExprSetResult::success(std::move(*closed_form.value()));
266 }
267
268 auto solved = solve_set(normalized, variable, context, options);
269 if (!solved) {
270 return ExprSetResult::failure(solved.error());
271 }
272
273 const auto& solution_set = solved.value();
274 if (std::holds_alternative<EmptySolutions>(solution_set)) {
275 return expr_set({});
276 }
277 const auto* finite = std::get_if<FiniteSolutions>(&solution_set);
278 if (!finite) {
279 return expr_set_failure(
280 CasErrc::Inconclusive,
281 "solution set is not a finite enumerable set<Expr>",
282 kSolveExprSetOperation);
283 }
284
285 std::vector<ExprPtr> elements;
286 elements.reserve(finite->values.size());
287 for (const auto& solution : finite->values) {
288 if (!solution.conditions.empty()) {
289 return expr_set_failure(CasErrc::Inconclusive,
290 "conditional finite solutions cannot be lowered to set<Expr>",
291 kSolveExprSetOperation);
292 }
293 elements.push_back(solution.value);
294 }
295 return expr_set(std::move(elements));
296}
