30Result<Rational> exact_double_rational(double value,
31 ComputationContext& context,
32 const std::string& operation) {
33 if (!std::isfinite(value)) {
34 return Result<Rational>::failure(
35 CasErrc::NumericFailure,
36 "finite interval endpoint evaluated to NaN or infinity",
37 operation);
38 }
39 if (value == 0.0) return Result<Rational>::success(Rational(0));
40
41 int exponent = 0;
42 const double fraction = std::frexp(value, &exponent);
43 const auto mantissa = static_cast<std::int64_t>(std::ldexp(fraction, 53));
44 const int binary_exponent = exponent - 53;
45 const std::size_t required_bits = binary_exponent < 0
46 ? static_cast<std::size_t>(-binary_exponent) + 1
47 : static_cast<std::size_t>(binary_exponent) + 54;
48 if (required_bits > context.limits().max_integer_bits) {
49 return Result<Rational>::failure(
50 CasErrc::ResourceLimit,
51 "IEEE endpoint conversion exceeds the integer bit budget",
52 operation);
53 }
54
55 BigInt numerator(static_cast<long long>(mantissa));
56 BigInt denominator(1);
57 if (binary_exponent >= 0) {
58 numerator <<= static_cast<mp_size_t>(binary_exponent);
59 } else {
60 denominator <<= static_cast<mp_size_t>(-binary_exponent);
61 }
62 return Result<Rational>::success(Rational(numerator, denominator));
63}

78std::optional<ComparableEndpoint> parse_quadratic_surd(
79 const std::shared_ptr<const SymbolicNode>& node) {
80 if (!node) return std::nullopt;
81 if (auto number = exact_number_value(node)) {
82 return ComparableEndpoint{0, *number, Rational(0), Rational(0)};
83 }
84
85 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(node)) {
86 if (function->type() != FunctionNode::FuncType::Sqrt ||
87 function->arguments().size() != 1) {
88 return std::nullopt;
89 }
90 auto radicand = exact_number_value(function->arguments()[0]);
91 if (!radicand || *radicand <= Rational(0)) return std::nullopt;
92 return ComparableEndpoint{0, Rational(0), Rational(1), *radicand};
93 }

95 auto combine_add = [](const ComparableEndpoint& left,
96 const ComparableEndpoint& right)
97 -> std::optional<ComparableEndpoint> {
98 if (left.infinity != 0 || right.infinity != 0) return std::nullopt;
99 if (left.radical_coefficient != Rational(0) &&
100 right.radical_coefficient != Rational(0) &&
101 left.radicand != right.radicand) {
102 return std::nullopt;
103 }
104 const Rational radicand = left.radical_coefficient != Rational(0)
105 ? left.radicand : right.radicand;
106 return ComparableEndpoint{
107 0,
108 left.rational + right.rational,
109 left.radical_coefficient + right.radical_coefficient,
110 radicand};
111 };
112
113 auto combine_multiply = [](const ComparableEndpoint& left,
114 const ComparableEndpoint& right)
115 -> std::optional<ComparableEndpoint> {
116 if (left.infinity != 0 || right.infinity != 0) return std::nullopt;
117 const bool left_radical = left.radical_coefficient != Rational(0);
118 const bool right_radical = right.radical_coefficient != Rational(0);
119 if (left_radical && right_radical && left.radicand != right.radicand) {
120 return std::nullopt;
121 }
122 const Rational radicand = left_radical ? left.radicand : right.radicand;
123 const Rational rational = left.rational * right.rational +
124 left.radical_coefficient * right.radical_coefficient * radicand;
125 const Rational coefficient =
126 left.rational * right.radical_coefficient +
127 left.radical_coefficient * right.rational;
128 return ComparableEndpoint{0, rational, coefficient, radicand};
129 };
130
131 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
132 ComparableEndpoint result{0, Rational(0), Rational(0), Rational(0)};
133 for (const auto& operand : add->operands()) {
134 auto parsed = parse_quadratic_surd(operand);
135 if (!parsed) return std::nullopt;
136 auto combined = combine_add(result, *parsed);
137 if (!combined) return std::nullopt;
138 result = std::move(*combined);
139 }
140 return result;
141 }
142
143 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
144 ComparableEndpoint result{0, Rational(1), Rational(0), Rational(0)};
145 for (const auto& operand : multiply->operands()) {
146 auto parsed = parse_quadratic_surd(operand);
147 if (!parsed) return std::nullopt;
148 auto combined = combine_multiply(result, *parsed);
149 if (!combined) return std::nullopt;
150 result = std::move(*combined);
151 }
152 return result;
153 }
154
155 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
156 auto exponent = exact_number_value(power->exponent());
157 if (!exponent) return std::nullopt;
158 if (*exponent == Rational(1, 2)) {
159 auto radicand = exact_number_value(power->base());
160 if (!radicand || *radicand <= Rational(0)) return std::nullopt;
161 return ComparableEndpoint{0, Rational(0), Rational(1), *radicand};
162 }
163 auto base = parse_quadratic_surd(power->base());
164 if (!base || *exponent != Rational(-1)) return std::nullopt;
165 const Rational norm = base->rational * base->rational -
166 base->radical_coefficient * base->radical_coefficient * base->radicand;
167 if (norm == Rational(0)) return std::nullopt;
168 return ComparableEndpoint{
169 0,
170 base->rational / norm,
171 (Rational(0) - base->radical_coefficient) / norm,
172 base->radicand};
173 }
174
175 return std::nullopt;
176}

183int quadratic_surd_sign(const Rational& rational,
184 const Rational& coefficient,
185 const Rational& radicand) {
186 const int rational_part_sign = rational_sign(rational);
187 const int radical_part_sign = rational_sign(coefficient);
188 if (radical_part_sign == 0) return rational_part_sign;
189 if (rational_part_sign == 0 || rational_part_sign == radical_part_sign) {
190 return radical_part_sign;
191 }
192
193 const Rational rational_square = rational * rational;
194 const Rational radical_square = coefficient * coefficient * radicand;
195 if (rational_square == radical_square) return 0;
196 if (rational_part_sign > 0) {
197 return rational_square > radical_square ? 1 : -1;
198 }
199 return radical_square > rational_square ? 1 : -1;
200}

202Result<int> compare_comparable(const ComparableEndpoint& left,
203 const ComparableEndpoint& right,
204 const std::string& operation) {
205 if (left.infinity < right.infinity) return Result<int>::success(-1);
206 if (left.infinity > right.infinity) return Result<int>::success(1);
207 if (left.infinity != 0) return Result<int>::success(0);
208
209 const bool left_radical = left.radical_coefficient != Rational(0);
210 const bool right_radical = right.radical_coefficient != Rational(0);
211 if (!left_radical && !right_radical) {
212 if (left.rational < right.rational) return Result<int>::success(-1);
213 if (left.rational > right.rational) return Result<int>::success(1);
214 return Result<int>::success(0);
215 }
216 if (left_radical && right_radical && left.radicand != right.radicand) {
217 return Result<int>::failure(
218 CasErrc::Inconclusive,
219 "exact comparison across distinct quadratic extensions is not proven",
220 operation);
221 }
222 const Rational radicand = left_radical ? left.radicand : right.radicand;
223 return Result<int>::success(quadratic_surd_sign(
224 left.rational - right.rational,
225 left.radical_coefficient - right.radical_coefficient,
226 radicand));
227}

229Result<ComparableEndpoint> comparable_endpoint(
230 const Endpoint& endpoint,
231 ComputationContext& context,
232 const std::string& operation) {
233 auto step = context.consume_steps(1, operation);
234 if (!step) return Result<ComparableEndpoint>::failure(step.error());
235
236 if (endpoint.is_neg_infinity && endpoint.is_pos_infinity) {
237 return Result<ComparableEndpoint>::failure(
238 CasErrc::InvalidArgument,
239 "an endpoint cannot be both negative and positive infinity",
240 operation);
241 }
242 if (endpoint.is_neg_infinity || endpoint.is_pos_infinity) {
243 if (endpoint.value) {
244 return Result<ComparableEndpoint>::failure(
245 CasErrc::InvalidArgument,
246 "infinite endpoints cannot also contain a finite value",
247 operation);
248 }
249 if (!endpoint.is_open) {
250 return Result<ComparableEndpoint>::failure(
251 CasErrc::InvalidArgument,
252 "infinite interval endpoints must be open",
253 operation);
254 }
255 return Result<ComparableEndpoint>::success(
256 ComparableEndpoint{endpoint.is_neg_infinity ? -1 : 1,
257 Rational(0), Rational(0), Rational(0)});
258 }
259 if (!endpoint.value || !lamina::detail::node(endpoint.value)) {
260 return Result<ComparableEndpoint>::failure(
261 CasErrc::InvalidArgument,
262 "finite interval endpoint must contain an expression",
263 operation);
264 }
265
266 auto simplified = endpoint.value->simplify();
267 if (!simplified || !lamina::detail::node(simplified)) {
268 return Result<ComparableEndpoint>::failure(
269 CasErrc::InternalInvariant,
270 "interval endpoint simplification produced a null expression",
271 operation);
272 }
273
274 if (auto number = std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(simplified))) {
275 if (std::holds_alternative<BigInt>(number->value())) {
276 return Result<ComparableEndpoint>::success(ComparableEndpoint{
277 0, Rational(std::get<BigInt>(number->value())),
278 Rational(0), Rational(0)});
279 }
280 if (std::holds_alternative<Rational>(number->value())) {
281 return Result<ComparableEndpoint>::success(ComparableEndpoint{
282 0, std::get<Rational>(number->value()), Rational(0), Rational(0)});
283 }
284 auto rational = exact_double_rational(
285 std::get<lmmc_real_t>(number->value()), context, operation);
286 if (!rational) return Result<ComparableEndpoint>::failure(rational.error());
287 return Result<ComparableEndpoint>::success(
288 ComparableEndpoint{0, std::move(rational.value()),
289 Rational(0), Rational(0)});
290 }
291
292 if (auto surd = parse_quadratic_surd(lamina::detail::node(simplified))) {
293 return Result<ComparableEndpoint>::success(std::move(*surd));
294 }
295
296 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(simplified))) {
297 return Result<ComparableEndpoint>::failure(
298 CasErrc::UnboundSymbol,
299 "interval endpoint contains unbound symbol '" + variable->name() + "'",
300 operation);
301 }
302 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(simplified))) {
303 if (function->arguments().size() == 1) {
304 auto argument = exact_number_value(function->arguments()[0]);
305 if (argument &&
306 ((function->type() == FunctionNode::FuncType::Ln &&
307 *argument <= Rational(0)) ||
308 (function->type() == FunctionNode::FuncType::Sqrt &&
309 *argument < Rational(0)))) {
310 return Result<ComparableEndpoint>::failure(
311 CasErrc::DomainError,
312 "interval endpoint is outside the real function domain",
313 operation);
314 }
315 }
316 }
317 return Result<ComparableEndpoint>::failure(
318 CasErrc::Inconclusive,
319 "exact ordering of a symbolic interval endpoint is not proven",
320 operation);
321}

323Result<std::vector<CheckedInterval>> checked_interval_views(
324 const std::vector<Interval>& intervals,
325 ComputationContext& context,
326 const std::string& operation) {
327 auto normalized = normalize_intervals_checked(intervals, context);
328 if (!normalized) {
329 return Result<std::vector<CheckedInterval>>::failure(normalized.error());
330 }
331
332 std::vector<CheckedInterval> checked;
333 auto normalized_intervals = std::move(normalized.value());
334 checked.reserve(normalized_intervals.size());
335 for (auto& interval : normalized_intervals) {
336 auto lower = comparable_endpoint(interval.lower, context, operation);
337 if (!lower) return Result<std::vector<CheckedInterval>>::failure(lower.error());
338 auto upper = comparable_endpoint(interval.upper, context, operation);
339 if (!upper) return Result<std::vector<CheckedInterval>>::failure(upper.error());
340 checked.push_back(CheckedInterval{
341 std::move(interval), std::move(lower.value()), std::move(upper.value())});
342 }
343 return Result<std::vector<CheckedInterval>>::success(std::move(checked));
344}

476Result<IntervalUnion> IntervalUnion::intersect_checked(
477 const IntervalUnion& other,
478 ComputationContext& context) const {
479 constexpr const char* operation = "interval_union_intersect";
480 auto left = checked_interval_views(intervals_, context, operation);
481 if (!left) return Result<IntervalUnion>::failure(left.error());
482 auto right = checked_interval_views(other.intervals_, context, operation);
483 if (!right) return Result<IntervalUnion>::failure(right.error());
484
485 const auto& a_intervals = left.value();
486 const auto& b_intervals = right.value();
487 std::vector<Interval> result;
488 std::size_t i = 0;
489 std::size_t j = 0;
490 while (i < a_intervals.size() && j < b_intervals.size()) {
491 const auto& a = a_intervals[i];
492 const auto& b = b_intervals[j];
493 auto lower_order = compare_comparable(a.lower, b.lower, operation);
494 if (!lower_order) return Result<IntervalUnion>::failure(lower_order.error());
495 auto upper_order = compare_comparable(a.upper, b.upper, operation);
496 if (!upper_order) return Result<IntervalUnion>::failure(upper_order.error());
497 const bool use_a_lower = lower_order.value() >= 0;
498 const bool use_a_upper = upper_order.value() <= 0;
499
500 Interval candidate{
501 use_a_lower ? a.interval.lower : b.interval.lower,
502 use_a_upper ? a.interval.upper : b.interval.upper
503 };
504 auto empty = interval_is_empty_checked(candidate, context);
505 if (!empty) return Result<IntervalUnion>::failure(empty.error());
506 if (!empty.value()) {
507 result.push_back(std::move(candidate));
508 }
509
510 if (upper_order.value() < 0) {
511 ++i;
512 } else {
513 ++j;
514 }
515 }
516
517 auto normalized = normalize_intervals_checked(std::move(result), context);
518 if (!normalized) return Result<IntervalUnion>::failure(normalized.error());
519 return Result<IntervalUnion>::success(
520 IntervalUnion::from_checked_normalized(std::move(normalized.value())));
521}

552Result<IntervalUnion> IntervalUnion::complement_checked(
553 ComputationContext& context) const {
554 constexpr const char* operation = "interval_union_complement";
555 auto checked = checked_interval_views(intervals_, context, operation);
556 if (!checked) return Result<IntervalUnion>::failure(checked.error());
557 const auto& intervals = checked.value();
558
559 if (intervals.empty()) {
560 return Result<IntervalUnion>::success(
561 IntervalUnion::from_checked_normalized({Interval::entire_line()}));
562 }
563 if (intervals.size() == 1 &&
564 intervals[0].interval.lower.is_neg_infinity &&
565 intervals[0].interval.upper.is_pos_infinity) {
566 return Result<IntervalUnion>::success(IntervalUnion::from_checked_normalized({}));
567 }
568
569 std::vector<Interval> result;
570 const Interval& first = intervals.front().interval;
571 if (!first.lower.is_neg_infinity) {
572 result.push_back(Interval{
573 Endpoint::neg_inf(),
574 complement_upper_from_lower(first.lower)
575 });
576 }
577
578 for (std::size_t i = 0; i + 1 < intervals.size(); ++i) {
579 const Interval& current = intervals[i].interval;
580 const Interval& next = intervals[i + 1].interval;
581 result.push_back(Interval{
582 complement_lower_from_upper(current.upper),
583 complement_upper_from_lower(next.lower)
584 });
585 }
586
587 const Interval& last = intervals.back().interval;
588 if (!last.upper.is_pos_infinity) {
589 result.push_back(Interval{
590 complement_lower_from_upper(last.upper),
591 Endpoint::pos_inf()
592 });
593 }
594
595 auto normalized = normalize_intervals_checked(std::move(result), context);
596 if (!normalized) return Result<IntervalUnion>::failure(normalized.error());
597 return Result<IntervalUnion>::success(
598 IntervalUnion::from_checked_normalized(std::move(normalized.value())));
599}

606Result<bool> interval_contains_checked(
607 const Interval& interval,
608 double value,
609 ComputationContext& context) {
610 constexpr const char* operation = "interval_contains";
611 auto point_step = context.consume_steps(1, operation);
612 if (!point_step) return Result<bool>::failure(point_step.error());
613 auto point = exact_double_rational(value, context, operation);
614 if (!point) return Result<bool>::failure(point.error());
615 const ComparableEndpoint point_key{
616 0, std::move(point.value()), Rational(0), Rational(0)};
617
618 auto lower = comparable_endpoint(interval.lower, context, operation);
619 if (!lower) return Result<bool>::failure(lower.error());
620 auto upper = comparable_endpoint(interval.upper, context, operation);
621 if (!upper) return Result<bool>::failure(upper.error());
622
623 auto lower_cmp = compare_comparable(point_key, lower.value(), operation);
624 if (!lower_cmp) return Result<bool>::failure(lower_cmp.error());
625 if (lower_cmp.value() < 0 ||
626 (lower_cmp.value() == 0 && interval.lower.is_open)) {
627 return Result<bool>::success(false);
628 }
629 auto upper_cmp = compare_comparable(point_key, upper.value(), operation);
630 if (!upper_cmp) return Result<bool>::failure(upper_cmp.error());
631 if (upper_cmp.value() > 0 ||
632 (upper_cmp.value() == 0 && interval.upper.is_open)) {
633 return Result<bool>::success(false);
634 }
635 return Result<bool>::success(true);
636}

667Result<std::vector<Interval>> normalize_intervals_checked(

669 ComputationContext& context) {
670 struct CheckedInterval {
671 Interval interval;
672 ComparableEndpoint lower;
673 ComparableEndpoint upper;
674 };
675
676 std::vector<CheckedInterval> checked;
677 checked.reserve(intervals.size());
678 for (auto& interval : intervals) {
679 auto lower = comparable_endpoint(
680 interval.lower, context, kCheckedIntervalOperation);
681 if (!lower) return Result<std::vector<Interval>>::failure(lower.error());
682 auto upper = comparable_endpoint(
683 interval.upper, context, kCheckedIntervalOperation);
684 if (!upper) return Result<std::vector<Interval>>::failure(upper.error());
685 auto comparison = compare_comparable(
686 lower.value(), upper.value(), kCheckedIntervalOperation);
687 if (!comparison) {
688 return Result<std::vector<Interval>>::failure(comparison.error());
689 }
690 const bool empty = comparison.value() > 0 ||
691 (comparison.value() == 0 &&
692 (interval.lower.is_open || interval.upper.is_open));
693 if (!empty) {
694 checked.push_back(CheckedInterval{
695 std::move(interval), std::move(lower.value()), std::move(upper.value())});
696 }
697 }
698
699 for (std::size_t i = 1; i < checked.size(); ++i) {
700 std::size_t position = i;
701 while (position > 0) {
702 auto comparison = compare_comparable(
703 checked[position - 1].lower,
704 checked[position].lower,
705 kCheckedIntervalOperation);
706 if (!comparison) {
707 return Result<std::vector<Interval>>::failure(comparison.error());
708 }
709 const bool out_of_order = comparison.value() > 0 ||
710 (comparison.value() == 0 &&
711 checked[position - 1].interval.lower.is_open &&
712 !checked[position].interval.lower.is_open);
713 if (!out_of_order) break;
714 std::swap(checked[position - 1], checked[position]);
715 --position;
716 }
717 }
718
719 std::vector<CheckedInterval> merged;
720 merged.reserve(checked.size());
721 for (auto& next : checked) {
722 if (merged.empty()) {
723 merged.push_back(std::move(next));
724 continue;
725 }
726
727 CheckedInterval& current = merged.back();
728 auto boundary = compare_comparable(
729 current.upper, next.lower, kCheckedIntervalOperation);
730 if (!boundary) {
731 return Result<std::vector<Interval>>::failure(boundary.error());
732 }
733 const bool overlaps = boundary.value() > 0 ||
734 (boundary.value() == 0 &&
735 (!current.interval.upper.is_open || !next.interval.lower.is_open));
736 if (!overlaps) {
737 merged.push_back(std::move(next));
738 continue;
739 }
740
741 auto upper_comparison = compare_comparable(
742 current.upper, next.upper, kCheckedIntervalOperation);
743 if (!upper_comparison) {
744 return Result<std::vector<Interval>>::failure(upper_comparison.error());
745 }
746 if (upper_comparison.value() < 0 ||
747 (upper_comparison.value() == 0 && current.interval.upper.is_open &&
748 !next.interval.upper.is_open)) {
749 current.interval.upper = std::move(next.interval.upper);
750 current.upper = std::move(next.upper);
751 }
752 }
753
754 std::vector<Interval> result;
755 result.reserve(merged.size());
756 for (auto& interval : merged) {
757 result.push_back(std::move(interval.interval));
758 }
759 return Result<std::vector<Interval>>::success(std::move(result));
760}

847static std::shared_ptr<SymbolicExpr> parse_numeric_value(const std::string& str, size_t& pos) {
848 size_t start = pos;
849 bool has_digit = false;
850 bool has_dot = false;
851
852 if (pos < str.size() && (str[pos] == '-' || str[pos] == '+')) {
853 ++pos;
854 }
855
856 while (pos < str.size() && std::isdigit(static_cast<unsigned char>(str[pos]))) {
857 has_digit = true;
858 ++pos;
859 }
860
861 if (pos < str.size() && str[pos] == '.') {
862 has_dot = true;
863 ++pos;
864 while (pos < str.size() && std::isdigit(static_cast<unsigned char>(str[pos]))) {
865 has_digit = true;
866 ++pos;
867 }
868 }
869
870 if (!has_digit) {
871 pos = start;
872 return nullptr;
873 }
874
875 std::string num_str = str.substr(start, pos - start);
876 if (has_dot) {
877 double val = std::stod(num_str);
878 return SymbolicExpr::number(val);
879 } else {
880
881 long long val = std::stoll(num_str);
882 return SymbolicExpr::number(val);
883 }
884}

886static bool parse_endpoint_value(const std::string& str, size_t& pos, Endpoint& ep) {
887 pos = skip_ws(str, pos);
888
889 if (starts_with_at(str, pos, "-\xe2\x88\x9e")) {
890 ep.is_neg_infinity = true;
891 ep.is_pos_infinity = false;
892 ep.is_open = true;
893 ep.value = nullptr;
894 pos += 4;
895 return true;
896 }
897
898 if (starts_with_at(str, pos, "+\xe2\x88\x9e")) {
899 ep.is_pos_infinity = true;
900 ep.is_neg_infinity = false;
901 ep.is_open = true;
902 ep.value = nullptr;
903 pos += 4;
904 return true;
905 }
906
907 auto val = parse_numeric_value(str, pos);
908 if (!val) return false;
909
910 ep.value = val;
911 ep.is_neg_infinity = false;
912 ep.is_pos_infinity = false;

914 return true;

917static bool parse_single_interval(const std::string& str, size_t& pos, Interval& iv) {
918 pos = skip_ws(str, pos);
919 if (pos >= str.size()) return false;
920
921 char lower_bracket = str[pos];
922 if (lower_bracket != '(' && lower_bracket != '[') return false;
923 bool lower_open = (lower_bracket == '(');
924 ++pos;
925
926 Endpoint lower;
927 if (!parse_endpoint_value(str, pos, lower)) return false;
928
929 if (!lower.is_neg_infinity && !lower.is_pos_infinity) {
930 lower.is_open = lower_open;
931 }
932
933 pos = skip_ws(str, pos);
934 if (pos >= str.size() || str[pos] != ',') return false;
935 ++pos;
936
937 Endpoint upper;
938 if (!parse_endpoint_value(str, pos, upper)) return false;
939
940 pos = skip_ws(str, pos);
941 if (pos >= str.size()) return false;
942 char upper_bracket = str[pos];
943 if (upper_bracket != ')' && upper_bracket != ']') return false;
944 bool upper_open = (upper_bracket == ')');
945 ++pos;
946
947 if (!upper.is_neg_infinity && !upper.is_pos_infinity) {
948 upper.is_open = upper_open;
949 }
950
951 iv.lower = lower;
952 iv.upper = upper;
953 return true;
954}

956std::optional<IntervalUnion> IntervalUnion::parse(const std::string& str) {
957 if (str.empty()) return std::nullopt;
958
959 if (str == "\xe2\x88\x85") {
960 return IntervalUnion::empty();
961 }
962
963 std::vector<Interval> intervals;
964 size_t pos = 0;
965
966 Interval iv;
967 if (!parse_single_interval(str, pos, iv)) return std::nullopt;
968 intervals.push_back(iv);
969
970 while (pos < str.size()) {
971 pos = skip_ws(str, pos);
972 if (pos >= str.size()) break;
973
974 if (!starts_with_at(str, pos, "\xe2\x88\xaa")) {
975 return std::nullopt;
976 }
977 pos += 3;
978
979 pos = skip_ws(str, pos);
980
981 Interval next_iv;
982 if (!parse_single_interval(str, pos, next_iv)) return std::nullopt;
983 intervals.push_back(next_iv);
984 }
985
986 return IntervalUnion(std::move(intervals));
987}

1017 } else if (upper_cond) {

1019 } else {

1022 }
1023 };
