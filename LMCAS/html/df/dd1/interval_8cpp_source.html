43Result<Rational> exact_double_rational(double value,
44 ComputationContext& context,
45 const std::string& operation) {
46 if (!std::isfinite(value)) {
47 return Result<Rational>::failure(
48 CasErrc::NumericFailure,
49 "finite interval endpoint evaluated to NaN or infinity",
50 operation);
51 }
52 if (value == 0.0) return Result<Rational>::success(Rational(0));
53
54 int exponent = 0;
55 const double fraction = std::frexp(value, &exponent);
56 const auto mantissa = static_cast<std::int64_t>(std::ldexp(fraction, 53));
57 const int binary_exponent = exponent - 53;
58 const std::size_t required_bits = binary_exponent < 0
59 ? static_cast<std::size_t>(-binary_exponent) + 1
60 : static_cast<std::size_t>(binary_exponent) + 54;
61 if (required_bits > context.limits().max_integer_bits) {
62 return Result<Rational>::failure(
63 CasErrc::ResourceLimit,
64 "IEEE endpoint conversion exceeds the integer bit budget",
65 operation);
66 }
67
68 BigInt numerator(static_cast<long long>(mantissa));
69 BigInt denominator(1);
70 if (binary_exponent >= 0) {
71 numerator <<= static_cast<mp_size_t>(binary_exponent);
72 } else {
73 denominator <<= static_cast<mp_size_t>(-binary_exponent);
74 }
75 return Result<Rational>::success(Rational(numerator, denominator));
76}

91std::optional<ComparableEndpoint> parse_quadratic_surd(
92 const std::shared_ptr<const SymbolicNode>& node) {
93 if (!node) return std::nullopt;
94 if (auto number = exact_number_value(node)) {
95 return ComparableEndpoint{0, *number, Rational(0), Rational(0)};
96 }
97
98 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(node)) {
99 if (function->type() != FunctionNode::FuncType::Sqrt ||
100 function->arguments().size() != 1) {
101 return std::nullopt;
102 }
103 auto radicand = exact_number_value(function->arguments()[0]);
104 if (!radicand || *radicand <= Rational(0)) return std::nullopt;
105 return ComparableEndpoint{0, Rational(0), Rational(1), *radicand};
106 }

108 auto combine_add = [](const ComparableEndpoint& left,
109 const ComparableEndpoint& right)
110 -> std::optional<ComparableEndpoint> {
111 if (left.infinity != 0 || right.infinity != 0) return std::nullopt;
112 if (left.radical_coefficient != Rational(0) &&
113 right.radical_coefficient != Rational(0) &&
114 left.radicand != right.radicand) {
115 return std::nullopt;
116 }
117 const Rational radicand = left.radical_coefficient != Rational(0)
118 ? left.radicand : right.radicand;
119 return ComparableEndpoint{
120 0,
121 left.rational + right.rational,
122 left.radical_coefficient + right.radical_coefficient,
123 radicand};
124 };
125
126 auto combine_multiply = [](const ComparableEndpoint& left,
127 const ComparableEndpoint& right)
128 -> std::optional<ComparableEndpoint> {
129 if (left.infinity != 0 || right.infinity != 0) return std::nullopt;
130 const bool left_radical = left.radical_coefficient != Rational(0);
131 const bool right_radical = right.radical_coefficient != Rational(0);
132 if (left_radical && right_radical && left.radicand != right.radicand) {
133 return std::nullopt;
134 }
135 const Rational radicand = left_radical ? left.radicand : right.radicand;
136 const Rational rational = left.rational * right.rational +
137 left.radical_coefficient * right.radical_coefficient * radicand;
138 const Rational coefficient =
139 left.rational * right.radical_coefficient +
140 left.radical_coefficient * right.rational;
141 return ComparableEndpoint{0, rational, coefficient, radicand};
142 };
143
144 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
145 ComparableEndpoint result{0, Rational(0), Rational(0), Rational(0)};
146 for (const auto& operand : add->operands()) {
147 auto parsed = parse_quadratic_surd(operand);
148 if (!parsed) return std::nullopt;
149 auto combined = combine_add(result, *parsed);
150 if (!combined) return std::nullopt;
151 result = std::move(*combined);
152 }
153 return result;
154 }
155
156 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
157 ComparableEndpoint result{0, Rational(1), Rational(0), Rational(0)};
158 for (const auto& operand : multiply->operands()) {
159 auto parsed = parse_quadratic_surd(operand);
160 if (!parsed) return std::nullopt;
161 auto combined = combine_multiply(result, *parsed);
162 if (!combined) return std::nullopt;
163 result = std::move(*combined);
164 }
165 return result;
166 }
167
168 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
169 auto exponent = exact_number_value(power->exponent());
170 if (!exponent) return std::nullopt;
171 if (*exponent == Rational(1, 2)) {
172 auto radicand = exact_number_value(power->base());
173 if (!radicand || *radicand <= Rational(0)) return std::nullopt;
174 return ComparableEndpoint{0, Rational(0), Rational(1), *radicand};
175 }
176 auto base = parse_quadratic_surd(power->base());
177 if (!base || *exponent != Rational(-1)) return std::nullopt;
178 const Rational norm = base->rational * base->rational -
179 base->radical_coefficient * base->radical_coefficient * base->radicand;
180 if (norm == Rational(0)) return std::nullopt;
181 return ComparableEndpoint{
182 0,
183 base->rational / norm,
184 (Rational(0) - base->radical_coefficient) / norm,
185 base->radicand};
186 }
187
188 return std::nullopt;
189}

197Result<int> compare_comparable(const ComparableEndpoint& left,
198 const ComparableEndpoint& right,
199 ComputationContext& context,
200 const std::string& operation) {
201 (void)operation;
202 if (left.infinity < right.infinity) return Result<int>::success(-1);
203 if (left.infinity > right.infinity) return Result<int>::success(1);
204 if (left.infinity != 0) return Result<int>::success(0);
205
206 const bool left_radical =
207 left.radical_coefficient != Rational(0) || left.algebraic.has_value();
208 const bool right_radical =
209 right.radical_coefficient != Rational(0) || right.algebraic.has_value();
210 if (!left_radical && !right_radical) {
211 if (left.rational < right.rational) return Result<int>::success(-1);
212 if (left.rational > right.rational) return Result<int>::success(1);
213 return Result<int>::success(0);
214 }
215 auto compare_algebraic_to_rational =
216 [](const lamina::detail::ExactRealAlgebraic& algebraic,
217 const Rational& rational) -> int {
218 if (algebraic.upper < rational) return -1;
219 if (rational < algebraic.lower) return 1;
220 const Rational value = algebraic.polynomial.eval(rational);
221 if (value == Rational(0)) return 0;
222 const Rational lower_value =
223 algebraic.polynomial.eval(algebraic.lower);
224 const bool root_before_rational =
225 rational_sign(lower_value) != rational_sign(value);
226 return root_before_rational ? -1 : 1;
227 };
228 if (left.algebraic && right.radical_coefficient == Rational(0)) {
229 return Result<int>::success(
230 compare_algebraic_to_rational(*left.algebraic, right.rational));
231 }
232 if (right.algebraic && left.radical_coefficient == Rational(0)) {
233 return Result<int>::success(
234 -compare_algebraic_to_rational(*right.algebraic, left.rational));
235 }
236 auto to_algebraic = [&](const ComparableEndpoint& endpoint)
237 -> lamina::detail::ExactRealAlgebraicResult {
238 if (endpoint.algebraic) {
239 return lamina::detail::ExactRealAlgebraicResult::success(
240 *endpoint.algebraic);
241 }
242 if (endpoint.radical_coefficient == Rational(0)) {
243 return lamina::detail::make_exact_real_algebraic(
244 Polynomial<Rational>({
245 Rational(0) - endpoint.rational,
246 Rational(1)
247 }),
248 0, 1, context);
249 }
250 const Rational a = endpoint.rational;
251 const Rational b = endpoint.radical_coefficient;
252 const Rational d = endpoint.radicand;
253 auto value = lamina::detail::make_exact_real_algebraic(
254 Polynomial<Rational>({
255 a * a - b * b * d,
256 Rational(-2) * a,
257 Rational(1)
258 }),
259 b > Rational(0) ? 1 : 0, 1, context);
260 return value;
261 };
262 auto left_algebraic = to_algebraic(left);
263 if (!left_algebraic) return Result<int>::failure(left_algebraic.error());
264 auto right_algebraic = to_algebraic(right);
265 if (!right_algebraic) return Result<int>::failure(right_algebraic.error());
266 return lamina::detail::compare_exact_real_algebraic(
267 left_algebraic.value(), right_algebraic.value(), context);
268}

270Result<ComparableEndpoint> comparable_endpoint(
271 const Endpoint& endpoint,
272 ComputationContext& context,
273 const std::string& operation) {
274 auto step = context.consume_steps(1, operation);
275 if (!step) return Result<ComparableEndpoint>::failure(step.error());
276
277 if (endpoint.is_neg_infinity && endpoint.is_pos_infinity) {
278 return Result<ComparableEndpoint>::failure(
279 CasErrc::InvalidArgument,
280 "an endpoint cannot be both negative and positive infinity",
281 operation);
282 }
283 if (endpoint.is_neg_infinity || endpoint.is_pos_infinity) {
284 if (endpoint.value) {
285 return Result<ComparableEndpoint>::failure(
286 CasErrc::InvalidArgument,
287 "infinite endpoints cannot also contain a finite value",
288 operation);
289 }
290 if (!endpoint.is_open) {
291 return Result<ComparableEndpoint>::failure(
292 CasErrc::InvalidArgument,
293 "infinite interval endpoints must be open",
294 operation);
295 }
296 return Result<ComparableEndpoint>::success(
297 ComparableEndpoint{endpoint.is_neg_infinity ? -1 : 1,
298 Rational(0), Rational(0), Rational(0)});
299 }
300 if (!endpoint.value || !lamina::detail::node(endpoint.value)) {
301 return Result<ComparableEndpoint>::failure(
302 CasErrc::InvalidArgument,
303 "finite interval endpoint must contain an expression",
304 operation);
305 }
306
307 auto simplified = endpoint.value->simplify();
308 if (!simplified || !lamina::detail::node(simplified)) {
309 return Result<ComparableEndpoint>::failure(
310 CasErrc::InternalInvariant,
311 "interval endpoint simplification produced a null expression",
312 operation);
313 }
314
315 if (auto number = std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(simplified))) {
316 if (std::holds_alternative<BigInt>(number->value())) {
317 return Result<ComparableEndpoint>::success(ComparableEndpoint{
318 0, Rational(std::get<BigInt>(number->value())),
319 Rational(0), Rational(0)});
320 }
321 if (std::holds_alternative<Rational>(number->value())) {
322 return Result<ComparableEndpoint>::success(ComparableEndpoint{
323 0, std::get<Rational>(number->value()), Rational(0), Rational(0)});
324 }
325 auto rational = exact_double_rational(
326 std::get<lmmc_real_t>(number->value()), context, operation);
327 if (!rational) return Result<ComparableEndpoint>::failure(rational.error());
328 return Result<ComparableEndpoint>::success(
329 ComparableEndpoint{0, std::move(rational.value()),
330 Rational(0), Rational(0)});
331 }
332
333 if (auto surd = parse_quadratic_surd(lamina::detail::node(simplified))) {
334 if (surd->radical_coefficient != Rational(0)) {
335 const Rational a = surd->rational;
336 const Rational b = surd->radical_coefficient;
337 const Rational d = surd->radicand;
338 auto algebraic = lamina::detail::make_exact_real_algebraic(
339 Polynomial<Rational>({
340 a * a - b * b * d,
341 Rational(-2) * a,
342 Rational(1)
343 }),
344 b > Rational(0) ? 1 : 0, 1, context);
345 if (!algebraic) {
346 return Result<ComparableEndpoint>::failure(algebraic.error());
347 }
348 surd->algebraic = std::move(algebraic.value());
349 }
350 return Result<ComparableEndpoint>::success(std::move(*surd));
351 }
352
353 if (auto root = std::dynamic_pointer_cast<const RootOfNode>(
354 lamina::detail::node(simplified))) {
355 auto isolated = lamina::detail::isolate_exact_root(
356 root->exact_id(), context, operation);
357 if (!isolated) {
358 return Result<ComparableEndpoint>::failure(isolated.error());
359 }
360 auto* real = std::get_if<lamina::detail::RealIsolation>(
361 &isolated.value());
362 if (!real) {
363 return Result<ComparableEndpoint>::failure(
364 CasErrc::DomainError,
365 "interval endpoint RootOf is not real",
366 operation);
367 }
368 ComparableEndpoint endpoint_value;
369 endpoint_value.algebraic = std::move(real->value);
370 return Result<ComparableEndpoint>::success(std::move(endpoint_value));
371 }
372
373 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(simplified))) {
374 return Result<ComparableEndpoint>::failure(
375 CasErrc::UnboundSymbol,
376 "interval endpoint contains unbound symbol '" + variable->name() + "'",
377 operation);
378 }
379 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(lamina::detail::node(simplified))) {
380 if (function->arguments().size() == 1) {
381 auto argument = exact_number_value(function->arguments()[0]);
382 if (argument &&
383 ((function->type() == FunctionNode::FuncType::Ln &&
384 *argument <= Rational(0)) ||
385 (function->type() == FunctionNode::FuncType::Sqrt &&
386 *argument < Rational(0)))) {
387 return Result<ComparableEndpoint>::failure(
388 CasErrc::DomainError,
389 "interval endpoint is outside the real function domain",
390 operation);
391 }
392 }
393 }
394 return Result<ComparableEndpoint>::failure(
395 CasErrc::Inconclusive,
396 "exact ordering of a symbolic interval endpoint is not proven",
397 operation);
398}

400Result<std::vector<CheckedInterval>> checked_interval_views(
401 const std::vector<Interval>& intervals,
402 ComputationContext& context,
403 const std::string& operation) {
404 auto normalized = normalize_intervals_checked(intervals, context);
405 if (!normalized) {
406 return Result<std::vector<CheckedInterval>>::failure(normalized.error());
407 }
408
409 std::vector<CheckedInterval> checked;
410 auto normalized_intervals = std::move(normalized.value());
411 checked.reserve(normalized_intervals.size());
412 for (auto& interval : normalized_intervals) {
413 auto lower = comparable_endpoint(interval.lower, context, operation);
414 if (!lower) return Result<std::vector<CheckedInterval>>::failure(lower.error());
415 auto upper = comparable_endpoint(interval.upper, context, operation);
416 if (!upper) return Result<std::vector<CheckedInterval>>::failure(upper.error());
417 checked.push_back(CheckedInterval{
418 std::move(interval), std::move(lower.value()), std::move(upper.value())});
419 }
420 return Result<std::vector<CheckedInterval>>::success(std::move(checked));
421}

553Result<IntervalUnion> IntervalUnion::intersect_checked(
554 const IntervalUnion& other,
555 ComputationContext& context) const {
556 constexpr const char* operation = "interval_union_intersect";
557 auto left = checked_interval_views(intervals_, context, operation);
558 if (!left) return Result<IntervalUnion>::failure(left.error());
559 auto right = checked_interval_views(other.intervals_, context, operation);
560 if (!right) return Result<IntervalUnion>::failure(right.error());
561
562 const auto& a_intervals = left.value();
563 const auto& b_intervals = right.value();
564 std::vector<Interval> result;
565 std::size_t i = 0;
566 std::size_t j = 0;
567 while (i < a_intervals.size() && j < b_intervals.size()) {
568 const auto& a = a_intervals[i];
569 const auto& b = b_intervals[j];
570 auto lower_order = compare_comparable(a.lower, b.lower, context, operation);
571 if (!lower_order) return Result<IntervalUnion>::failure(lower_order.error());
572 auto upper_order = compare_comparable(a.upper, b.upper, context, operation);
573 if (!upper_order) return Result<IntervalUnion>::failure(upper_order.error());
574 const bool use_a_lower = lower_order.value() >= 0;
575 const bool use_a_upper = upper_order.value() <= 0;
576
577 Interval candidate{
578 use_a_lower ? a.interval.lower : b.interval.lower,
579 use_a_upper ? a.interval.upper : b.interval.upper
580 };
581 auto empty = interval_is_empty_checked(candidate, context);
582 if (!empty) return Result<IntervalUnion>::failure(empty.error());
583 if (!empty.value()) {
584 result.push_back(std::move(candidate));
585 }
586
587 if (upper_order.value() < 0) {
588 ++i;
589 } else {
590 ++j;
591 }
592 }
593
594 auto normalized = normalize_intervals_checked(std::move(result), context);
595 if (!normalized) return Result<IntervalUnion>::failure(normalized.error());
596 return Result<IntervalUnion>::success(
597 IntervalUnion::from_checked_normalized(std::move(normalized.value())));
598}

629Result<IntervalUnion> IntervalUnion::complement_checked(
630 ComputationContext& context) const {
631 constexpr const char* operation = "interval_union_complement";
632 auto checked = checked_interval_views(intervals_, context, operation);
633 if (!checked) return Result<IntervalUnion>::failure(checked.error());
634 const auto& intervals = checked.value();
635
636 if (intervals.empty()) {
637 return Result<IntervalUnion>::success(
638 IntervalUnion::from_checked_normalized({Interval::entire_line()}));
639 }
640 if (intervals.size() == 1 &&
641 intervals[0].interval.lower.is_neg_infinity &&
642 intervals[0].interval.upper.is_pos_infinity) {
643 return Result<IntervalUnion>::success(IntervalUnion::from_checked_normalized({}));
644 }
645
646 std::vector<Interval> result;
647 const Interval& first = intervals.front().interval;
648 if (!first.lower.is_neg_infinity) {
649 result.push_back(Interval{
650 Endpoint::neg_inf(),
651 complement_upper_from_lower(first.lower)
652 });
653 }
654
655 for (std::size_t i = 0; i + 1 < intervals.size(); ++i) {
656 const Interval& current = intervals[i].interval;
657 const Interval& next = intervals[i + 1].interval;
658 result.push_back(Interval{
659 complement_lower_from_upper(current.upper),
660 complement_upper_from_lower(next.lower)
661 });
662 }
663
664 const Interval& last = intervals.back().interval;
665 if (!last.upper.is_pos_infinity) {
666 result.push_back(Interval{

743Result<std::vector<Interval>> normalize_intervals_checked(
744 std::vector<Interval> intervals,
745 ComputationContext& context) {
746 struct CheckedInterval {
747 Interval interval;
748 ComparableEndpoint lower;
749 ComparableEndpoint upper;
750 };
751
752 std::vector<CheckedInterval> checked;
753 checked.reserve(intervals.size());
754 for (auto& interval : intervals) {
755 auto lower = comparable_endpoint(
756 interval.lower, context, kCheckedIntervalOperation);
757 if (!lower) return Result<std::vector<Interval>>::failure(lower.error());
758 auto upper = comparable_endpoint(
759 interval.upper, context, kCheckedIntervalOperation);
760 if (!upper) return Result<std::vector<Interval>>::failure(upper.error());
761 auto comparison = compare_comparable(lower.value(), upper.value(), context, kCheckedIntervalOperation);
762 if (!comparison) {
763 return Result<std::vector<Interval>>::failure(comparison.error());
764 }
765 const bool empty = comparison.value() > 0 ||
766 (comparison.value() == 0 &&
767 (interval.lower.is_open || interval.upper.is_open));
768 if (!empty) {
769 checked.push_back(CheckedInterval{
770 std::move(interval), std::move(lower.value()), std::move(upper.value())});
771 }
772 }
773
774 for (std::size_t i = 1; i < checked.size(); ++i) {
775 std::size_t position = i;
776 while (position > 0) {
777 auto comparison = compare_comparable(checked[position - 1].lower, checked[position].lower, context, kCheckedIntervalOperation);
778 if (!comparison) {
779 return Result<std::vector<Interval>>::failure(comparison.error());
780 }
781 const bool out_of_order = comparison.value() > 0 ||
782 (comparison.value() == 0 &&
783 checked[position - 1].interval.lower.is_open &&
784 !checked[position].interval.lower.is_open);
785 if (!out_of_order) break;
786 std::swap(checked[position - 1], checked[position]);
787 --position;
788 }
789 }
790
791 std::vector<CheckedInterval> merged;
792 merged.reserve(checked.size());
793 for (auto& next : checked) {
794 if (merged.empty()) {
795 merged.push_back(std::move(next));
796 continue;
797 }
798
799 CheckedInterval& current = merged.back();
800 auto boundary = compare_comparable(current.upper, next.lower, context, kCheckedIntervalOperation);
801 if (!boundary) {
802 return Result<std::vector<Interval>>::failure(boundary.error());
803 }
804 const bool overlaps = boundary.value() > 0 ||
805 (boundary.value() == 0 &&
806 (!current.interval.upper.is_open || !next.interval.lower.is_open));
807 if (!overlaps) {
808 merged.push_back(std::move(next));
809 continue;
810 }
811
812 auto upper_comparison = compare_comparable(current.upper, next.upper, context, kCheckedIntervalOperation);
813 if (!upper_comparison) {
814 return Result<std::vector<Interval>>::failure(upper_comparison.error());
815 }
816 if (upper_comparison.value() < 0 ||
817 (upper_comparison.value() == 0 && current.interval.upper.is_open &&
818 !next.interval.upper.is_open)) {
819 current.interval.upper = std::move(next.interval.upper);
820 current.upper = std::move(next.upper);
821 }
822 }
823
824 std::vector<Interval> result;
825 result.reserve(merged.size());
826 for (auto& interval : merged) {
827 result.push_back(std::move(interval.interval));
828 }
829 return Result<std::vector<Interval>>::success(std::move(result));
830}

917static std::shared_ptr<SymbolicExpr> parse_numeric_value(const std::string& str, size_t& pos) {
918 size_t start = pos;
919 bool has_digit = false;
920 bool has_dot = false;
921
922 if (pos < str.size() && (str[pos] == '-' || str[pos] == '+')) {
923 ++pos;
924 }
925
926 while (pos < str.size() && std::isdigit(static_cast<unsigned char>(str[pos]))) {
927 has_digit = true;
928 ++pos;
929 }
930
931 if (pos < str.size() && str[pos] == '.') {
932 has_dot = true;
933 ++pos;
934 while (pos < str.size() && std::isdigit(static_cast<unsigned char>(str[pos]))) {
935 has_digit = true;
936 ++pos;
937 }
938 }
939
940 if (!has_digit) {
941 pos = start;
942 return nullptr;
943 }
944
945 std::string num_str = str.substr(start, pos - start);
946 if (has_dot) {
947 double val = std::stod(num_str);
948 return SymbolicExpr::number(val);
949 } else {
950
951 long long val = std::stoll(num_str);
952 return SymbolicExpr::number(val);
953 }
954}

956static bool parse_endpoint_value(const std::string& str, size_t& pos, Endpoint& ep) {
957 pos = skip_ws(str, pos);
958
959 if (starts_with_at(str, pos, "-\xe2\x88\x9e")) {
960 ep.is_neg_infinity = true;
961 ep.is_pos_infinity = false;
962 ep.is_open = true;
963 ep.value = nullptr;
964 pos += 4;
965 return true;
966 }

969 ep.is_pos_infinity = true;
970 ep.is_neg_infinity = false;
971 ep.is_open = true;
972 ep.value = nullptr;
973 pos += 4;

981 ep.is_neg_infinity = false;

987static bool parse_single_interval(const std::string& str, size_t& pos, Interval& iv) {
988 pos = skip_ws(str, pos);
989 if (pos >= str.size()) return false;
990
991 char lower_bracket = str[pos];
992 if (lower_bracket != '(' && lower_bracket != '[') return false;
993 bool lower_open = (lower_bracket == '(');
994 ++pos;
995
996 Endpoint lower;
997 if (!parse_endpoint_value(str, pos, lower)) return false;
998
999 if (!lower.is_neg_infinity && !lower.is_pos_infinity) {
1000 lower.is_open = lower_open;
1001 }
1002
1003 pos = skip_ws(str, pos);
1004 if (pos >= str.size() || str[pos] != ',') return false;
1005 ++pos;
1006
1007 Endpoint upper;
1008 if (!parse_endpoint_value(str, pos, upper)) return false;
1009
1010 pos = skip_ws(str, pos);
1011 if (pos >= str.size()) return false;
1012 char upper_bracket = str[pos];
1013 if (upper_bracket != ')' && upper_bracket != ']') return false;
1014 bool upper_open = (upper_bracket == ')');
1015 ++pos;
1016
1017 if (!upper.is_neg_infinity && !upper.is_pos_infinity) {
1018 upper.is_open = upper_open;
1019 }
1020
1021 iv.lower = lower;
1022 iv.upper = upper;
1023 return true;
1024}

1026std::optional<IntervalUnion> IntervalUnion::parse(const std::string& str) {
1027 if (str.empty()) return std::nullopt;
1028
1029 if (str == "\xe2\x88\x85") {
1030 return IntervalUnion::empty();
1031 }
1032
1033 std::vector<Interval> intervals;
1034 size_t pos = 0;
1035
1036 Interval iv;

1038 intervals.push_back(iv);
1039
1040 while (pos < str.size()) {
1041 pos = skip_ws(str, pos);
1042 if (pos >= str.size()) break;
1043
1044 if (!starts_with_at(str, pos, "\xe2\x88\xaa")) {
1045 return std::nullopt;

1047 pos += 3;
1048
1049 pos = skip_ws(str, pos);

1053 intervals.push_back(next_iv);
1054 }

1085 } else if (lower_cond) {

1087 } else if (upper_cond) {
1088 return upper_cond;
1089 } else {
1090
1091 return nullptr;
1092 }
1093 };
1094

1096
