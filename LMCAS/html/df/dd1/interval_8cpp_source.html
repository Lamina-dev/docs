122Result<bool> interval_contains_checked(
123 const Interval& interval,
124 double value,
125 ComputationContext& context) {
126 constexpr const char* operation = "interval_contains";
127 auto point_step = context.consume_steps(1, operation);
128 if (!point_step) {
129 return Result<bool>::failure(point_step.error());
130 }
131 auto point = exact_double_rational(value, context, operation);
132 if (!point) {
133 return Result<bool>::failure(point.error());
134 }
135 const ComparableEndpoint point_key{
136 0, std::move(point.value()), Rational(0), Rational(0)};
137
138 auto lower = comparable_endpoint(interval.lower, context, operation);
139 if (!lower) {
140 return Result<bool>::failure(lower.error());
141 }
142 auto upper = comparable_endpoint(interval.upper, context, operation);
143 if (!upper) {
144 return Result<bool>::failure(upper.error());
145 }
146
147 auto lower_cmp = compare_comparable(point_key, lower.value(), context);
148 if (!lower_cmp) {
149 return Result<bool>::failure(lower_cmp.error());
150 }
151 if (lower_cmp.value() < 0 ||
152 (lower_cmp.value() == 0 && interval.lower.is_open)) {
153 return Result<bool>::success(false);
154 }
155 auto upper_cmp = compare_comparable(point_key, upper.value(), context);
156 if (!upper_cmp) {
157 return Result<bool>::failure(upper_cmp.error());
158 }
159 if (upper_cmp.value() > 0 ||
160 (upper_cmp.value() == 0 && interval.upper.is_open)) {
161 return Result<bool>::success(false);
162 }
163 return Result<bool>::success(true);
164}

173Result<bool> interval_is_empty_checked(
174 const Interval& interval,
175 ComputationContext& context) {
176 constexpr const char* operation = "interval_is_empty";
177 auto lower = comparable_endpoint(interval.lower, context, operation);
178 if (!lower) {
179 return Result<bool>::failure(lower.error());
180 }
181 auto upper = comparable_endpoint(interval.upper, context, operation);
182 if (!upper) {
183 return Result<bool>::failure(upper.error());
184 }
185 auto comparison = compare_comparable(lower.value(), upper.value(), context);
186 if (!comparison) {
187 return Result<bool>::failure(comparison.error());
188 }
189 return Result<bool>::success(
190 comparison.value() > 0 ||
191 (comparison.value() == 0 &&
