202inline bool MultiplyNode::contains_matrix(const SymbolicNode& node) {
203 if (dynamic_cast<const MatrixNode*>(&node)) return true;
204 if (const auto* power = dynamic_cast<const PowerNode*>(&node)) {
205 return contains_matrix(*power->base());
206 }
207 const std::vector<std::shared_ptr<const SymbolicNode>>* children = nullptr;
208 if (const auto* sum = dynamic_cast<const AddNode*>(&node)) {
209 children = &sum->operands();
210 } else if (const auto* product = dynamic_cast<const MultiplyNode*>(&node)) {
211 children = &product->operands();
212 } else if (const auto* function = dynamic_cast<const FunctionNode*>(&node)) {
213 children = &function->arguments();
214 } else if (const auto* function = dynamic_cast<const UninterpretedFunctionNode*>(&node)) {
215 children = &function->arguments();
216 }
217 return children && std::any_of(
218 children->begin(), children->end(),
219 [](const auto& child) { return contains_matrix(*child); });

222inline std::variant<MatrixNode::DenseStorage, MatrixNode::SparseStorage> MatrixNode::create_storage_from_grid(
223 const std::vector<std::vector<std::shared_ptr<const SymbolicNode>>>& grid, size_t total_elements, size_t ncols) {
224
225 size_t non_zeros = 0;
226 for (const auto& row : grid) {
227 for (const auto& item : row) {
228 if (item && !item->is_zero()) non_zeros++;
229 }
230 }
231
232 bool use_sparse = total_elements > 0 && ((double)non_zeros / total_elements < 0.2);
233
234 if (use_sparse) {
235 SparseStorage s;
236 size_t r = 0;
237 for (const auto& row : grid) {
238 size_t c = 0;
239 for (const auto& item : row) {
240 if (item && !item->is_zero()) {
241 s[r * ncols + c] = item;
242 }
243 c++;
244 }
245 r++;
246 }
247 return s;
248 } else {
249 DenseStorage d;
250 d.reserve(total_elements);
251 for (const auto& row : grid) {
252 d.insert(d.end(), row.begin(), row.end());
253 }
254 return d;
255 }
