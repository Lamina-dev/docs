181AssumptionVoidResult AssumptionContext::assume_conditional_checked(
182 const SymbolicExpr& condition,
183 const SymbolicExpr& conclusion) {
184 constexpr const char* operation = "assume_conditional";
185 if (!std::dynamic_pointer_cast<const RelationalNode>(lamina::detail::node(condition))) {
186 return AssumptionVoidResult::failure(
187 CasErrc::InvalidArgument, "condition expression must be relational", operation);
188 }
189 if (!std::dynamic_pointer_cast<const RelationalNode>(lamina::detail::node(conclusion))) {
190 return AssumptionVoidResult::failure(
191 CasErrc::InvalidArgument, "conclusion expression must be relational", operation);
192 }
193 try {
194 Tribool cond_result = evaluate_condition(condition);
195 if (cond_result == Tribool::True &&
196 evaluate_condition(conclusion) == Tribool::False) {
197 return AssumptionVoidResult::failure(
198 CasErrc::InvalidArgument,
199 "condition is satisfied but conclusion contradicts the current assumption state",
200 operation);
201 }
202 scope_stack_.back().conditionals.push_back({condition, conclusion});
203 ++cache_generation_;
204 } catch (const std::bad_alloc&) {
205 return AssumptionVoidResult::failure(
206 CasErrc::ResourceLimit, "assumption allocation failed", operation);
207 } catch (const std::invalid_argument& ex) {
208 return AssumptionVoidResult::failure(CasErrc::InvalidArgument, ex.what(), operation);
209 } catch (const std::exception& ex) {
210 return AssumptionVoidResult::failure(CasErrc::InternalInvariant, ex.what(), operation);
211 }
212 return AssumptionVoidResult::success();
213}

225Tribool AssumptionContext::evaluate_condition(const SymbolicExpr& condition) const {
226 if (!lamina::detail::node(condition)) {
227 return Tribool::Unknown;
228 }
229 auto rel_node = std::dynamic_pointer_cast<const RelationalNode>(lamina::detail::node(condition));
230 if (!rel_node) {
231 return Tribool::Unknown;
232 }
233
234 auto lhs = lamina::detail::expression_from_node(rel_node->left());
235 auto rhs = lamina::detail::expression_from_node(rel_node->right());
236 RelationalNode::Op op = rel_node->op();
237 for (auto it = scope_stack_.rbegin(); it != scope_stack_.rend(); ++it) {
238 if (it->relations.has_relation(lhs, rhs, op)) {
239 return Tribool::True;
240 }
241 }
242 // For patterns like "variable > 0", check if the variable has the corresponding sign.
243 auto lhs_var = std::dynamic_pointer_cast<const VariableNode>(rel_node->left());
244 auto rhs_num = std::dynamic_pointer_cast<const NumberNode>(rel_node->right());
245
246 if (lhs_var && rhs_num && rhs_num->is_zero()) {
247 const std::string& name = lhs_var->name();
248 switch (op) {
249 case RelationalNode::Op::GT:
250 if (has_sign(name, Sign::Positive)) return Tribool::True;
251 if (has_sign(name, Sign::Negative) || has_sign(name, Sign::NonPositive) || has_sign(name, Sign::Zero))
252 return Tribool::False;
253 break;
254 case RelationalNode::Op::GEQ:
255 if (has_sign(name, Sign::NonNegative) || has_sign(name, Sign::Positive) || has_sign(name, Sign::Zero))
256 return Tribool::True;
257 if (has_sign(name, Sign::Negative))
258 return Tribool::False;
259 break;
260 case RelationalNode::Op::LT:
261 if (has_sign(name, Sign::Negative)) return Tribool::True;
262 if (has_sign(name, Sign::Positive) || has_sign(name, Sign::NonNegative) || has_sign(name, Sign::Zero))
263 return Tribool::False;
264 break;
265 case RelationalNode::Op::LEQ:
266 if (has_sign(name, Sign::NonPositive) || has_sign(name, Sign::Negative) || has_sign(name, Sign::Zero))
267 return Tribool::True;
268 if (has_sign(name, Sign::Positive))
269 return Tribool::False;
270 break;
271 case RelationalNode::Op::NEQ:
272 if (has_sign(name, Sign::NonZero) || has_sign(name, Sign::Positive) || has_sign(name, Sign::Negative))
273 return Tribool::True;
274 if (has_sign(name, Sign::Zero))
275 return Tribool::False;
276 break;
277 case RelationalNode::Op::EQ:
278 if (has_sign(name, Sign::Zero)) return Tribool::True;
279 if (has_sign(name, Sign::Positive) || has_sign(name, Sign::Negative) || has_sign(name, Sign::NonZero))
280 return Tribool::False;
281 break;
282 }
283 }
284 auto lhs_num = std::dynamic_pointer_cast<const NumberNode>(rel_node->left());
285 auto rhs_var = std::dynamic_pointer_cast<const VariableNode>(rel_node->right());
286
287 if (lhs_num && lhs_num->is_zero() && rhs_var) {
288 const std::string& name = rhs_var->name();
289 switch (op) {
290 case RelationalNode::Op::LT: // 0 < var → var > 0
291 if (has_sign(name, Sign::Positive)) return Tribool::True;
292 if (has_sign(name, Sign::Negative) || has_sign(name, Sign::NonPositive) || has_sign(name, Sign::Zero))
293 return Tribool::False;
294 break;
295 case RelationalNode::Op::LEQ: // 0 <= var → var >= 0
296 if (has_sign(name, Sign::NonNegative) || has_sign(name, Sign::Positive) || has_sign(name, Sign::Zero))
297 return Tribool::True;
298 if (has_sign(name, Sign::Negative))
299 return Tribool::False;
300 break;
301 case RelationalNode::Op::GT: // 0 > var → var < 0
302 if (has_sign(name, Sign::Negative)) return Tribool::True;
303 if (has_sign(name, Sign::Positive) || has_sign(name, Sign::NonNegative) || has_sign(name, Sign::Zero))
304 return Tribool::False;
305 break;
306 case RelationalNode::Op::GEQ: // 0 >= var → var <= 0
307 if (has_sign(name, Sign::NonPositive) || has_sign(name, Sign::Negative) || has_sign(name, Sign::Zero))
308 return Tribool::True;
309 if (has_sign(name, Sign::Positive))
310 return Tribool::False;
311 break;
312 case RelationalNode::Op::NEQ: // 0 != var → var != 0
313 if (has_sign(name, Sign::NonZero) || has_sign(name, Sign::Positive) || has_sign(name, Sign::Negative))
314 return Tribool::True;
315 if (has_sign(name, Sign::Zero))
316 return Tribool::False;
317 break;
318 case RelationalNode::Op::EQ: // 0 == var → var == 0
319 if (has_sign(name, Sign::Zero)) return Tribool::True;
320 if (has_sign(name, Sign::Positive) || has_sign(name, Sign::Negative) || has_sign(name, Sign::NonZero))
321 return Tribool::False;
322 break;
323 }
324 }
325 return Tribool::Unknown;
326}

385AssumptionVoidResult AssumptionContext::assume_checked(const SymbolicExpr& relation) {
386 constexpr const char* operation = "assume";
387 if (!lamina::detail::node(relation)) {
388 return AssumptionVoidResult::failure(
389 CasErrc::InvalidArgument, "relation expression must not be null", operation);
390 }
391 if (!std::dynamic_pointer_cast<const RelationalNode>(lamina::detail::node(relation))) {
392 return AssumptionVoidResult::failure(
393 CasErrc::InvalidArgument, "relation expression root must be relational", operation);
394 }
395 try {
396 auto rel_node = std::dynamic_pointer_cast<const RelationalNode>(lamina::detail::node(relation));
397 auto lhs = lamina::detail::expression_from_node(rel_node->left());
398 auto rhs = lamina::detail::expression_from_node(rel_node->right());
399 auto result = scope_stack_.back().relations.add_relation_checked(
400 lhs, rhs, rel_node->op(), scope_stack_.back().properties);
401 if (!result.has_value()) {
402 return AssumptionVoidResult::failure(result.error());
403 }
404 ++cache_generation_;
405 } catch (const std::bad_alloc&) {
406 return AssumptionVoidResult::failure(
407 CasErrc::ResourceLimit, "assumption allocation failed", operation);
408 } catch (const std::invalid_argument& ex) {
409 return AssumptionVoidResult::failure(CasErrc::InvalidArgument, ex.what(), operation);
410 } catch (const std::exception& ex) {
411 return AssumptionVoidResult::failure(CasErrc::InternalInvariant, ex.what(), operation);
412 }
413 return AssumptionVoidResult::success();
414}

781/// Serialize an interval endpoint to a numeric string.
782std::string endpoint_to_string(const Endpoint& ep) {

789/// Serialize an interval to string format: [lo, hi] or (lo, hi) etc.
790std::string interval_to_string(const Interval& iv) {
791 std::string result;
792 result += iv.lower.is_open ? "(" : "[";
793 result += endpoint_to_string(iv.lower);
794 result += ", ";
795 result += endpoint_to_string(iv.upper);
796 result += iv.upper.is_open ? ")" : "]";
797 return result;
798}

800/// Parse an interval from string format like [0.000000, 1.000000]
801Interval parse_interval(const std::string& s, int line_num) {
802 if (s.size() < 5) {
803 throw std::invalid_argument(
804 "Line " + std::to_string(line_num) + ": malformed interval '" + s + "'");
805 }
806
807 Interval iv;
808 bool lower_open = (s[0] == '(');
809 bool upper_open = (s.back() == ')');
810 std::string inner = s.substr(1, s.size() - 2);
811 auto comma_pos = inner.find(", ");
812 if (comma_pos == std::string::npos) {
813 throw std::invalid_argument(
814 "Line " + std::to_string(line_num) + ": malformed interval '" + s + "'");
815 }
816
817 std::string lo_str = inner.substr(0, comma_pos);
818 std::string hi_str = inner.substr(comma_pos + 2);
819 if (lo_str == "-inf") {
820 iv.lower = Endpoint::neg_inf();
821 } else {
822 double lo_val = std::stod(lo_str);
823 auto lo_expr = lamina::detail::make_expression_ptr(
824 lamina::detail::make_node<NumberNode>(static_cast<lmmc_real_t>(lo_val)));
825 iv.lower = lower_open ? Endpoint::open(lo_expr) : Endpoint::closed(lo_expr);
826 }
827 if (hi_str == "+inf") {
828 iv.upper = Endpoint::pos_inf();
829 } else {
830 double hi_val = std::stod(hi_str);
831 auto hi_expr = lamina::detail::make_expression_ptr(
832 lamina::detail::make_node<NumberNode>(static_cast<lmmc_real_t>(hi_val)));
833 iv.upper = upper_open ? Endpoint::open(hi_expr) : Endpoint::closed(hi_expr);
834 }
835
836 return iv;
837}

842
843std::string AssumptionContext::serialize() const {
844 std::ostringstream out;
845
846 for (int scope_idx = 0; scope_idx < static_cast<int>(scope_stack_.size()); ++scope_idx) {
847 out << "SCOPE " << scope_idx << "\n";
848 const auto& scope = scope_stack_[scope_idx];
849 const auto& props = scope.properties;
850 auto symbols = props.get_all_symbols();
851 for (const auto& sym : symbols) {
852 Domain dom = props.get_domain(sym);
853 if (dom != Domain::Complex) {
854 out << "DOMAIN " << sym << " " << domain_to_string(dom) << "\n";
855 }
856 auto signs = props.get_signs(sym);
857 for (Sign s : signs) {
858 out << "SIGN " << sym << " " << sign_to_string(s) << "\n";
859 }
860 Parity par = props.get_parity(sym);
861 if (par != Parity::Unknown) {
862 out << "PARITY " << sym << " " << parity_to_string(par) << "\n";
863 }
864 Boundedness bnd = props.get_boundedness(sym);
865 if (bnd != Boundedness::Unknown) {
866 out << "BOUNDED " << sym << " " << boundedness_to_string(bnd) << "\n";
867 }
868 if (props.is_transcendental(sym)) {
869 out << "TRANSCENDENTAL " << sym << "\n";
870 }
871 Finiteness fin = props.get_finiteness(sym);
872 if (fin != Finiteness::Unknown) {
873 out << "FINITENESS " << sym << " " << finiteness_to_string(fin) << "\n";
874 }
875 Definiteness def = props.get_definiteness(sym);
876 if (def != Definiteness::Unknown) {
877 out << "DEFINITENESS " << sym << " " << definiteness_to_string(def) << "\n";
878 }
879 auto period = props.get_period(sym);
880 if (period.has_value() && *period) {
881 out << "PERIODIC " << sym << " " << (*period)->to_string() << "\n";
882 }
883 }
884 for (const auto& sym : symbols) {
885 auto cont_decls = props.get_continuity_decls(sym);
886 for (const auto& decl : cont_decls) {
887 if (decl.is_differentiable) {
888 out << "DIFFERENTIABLE " << sym << " "
889 << interval_to_string(decl.interval) << "\n";
890 } else {
891 out << "CONTINUOUS " << sym << " "
892 << interval_to_string(decl.interval) << "\n";
893 }
894 }
895
896 auto mono_decls = props.get_monotonicity_decls(sym);
897 for (const auto& decl : mono_decls) {
898 out << "MONOTONICITY " << sym << " " << decl.variable << " "
899 << interval_to_string(decl.interval) << " "
900 << monotonicity_to_string(decl.type) << "\n";
901 }
902 }
903 const auto& relations = scope.relations.get_relations();
904 for (const auto& rel : relations) {
905 std::string lhs_str = rel.lhs.to_string();
906 std::string rhs_str = rel.rhs.to_string();
907 out << "RELATION " << lhs_str << " " << relop_to_string(rel.op)
908 << " " << rhs_str << "\n";
909 }
910 for (const auto& cond : scope.conditionals) {
911 std::string cond_str = cond.condition.to_string();
912 std::string concl_str = cond.conclusion.to_string();
913 out << "CONDITIONAL (" << cond_str << ") => (" << concl_str << ")\n";
914 }
915 }
916
917 out << "END\n";
918 return out.str();
919}

922
923AssumptionContext AssumptionContext::deserialize_impl(const std::string& data) {
924 AssumptionContext ctx;
925 std::istringstream input(data);
926 std::string line;
927 int line_num = 0;
928 int current_scope = -1;
929 bool ended = false;
930
931 while (std::getline(input, line)) {
932 ++line_num;
933 if (line.empty()) continue;
934 if (!line.empty() && line.back() == '\r') {
935 line.pop_back();
936 }
937 if (line.empty()) continue;
938 if (line == "END") {
939 ended = true;
940 break;
941 }
942 std::istringstream ls(line);
943 std::string keyword;
944 ls >> keyword;
945
946 if (keyword == "SCOPE") {
947 int idx;
948 if (!(ls >> idx)) {
949 throw std::invalid_argument(
950 "Line " + std::to_string(line_num) + ": SCOPE missing index");
951 }
952 if (idx == 0) {
953 current_scope = 0;
954 } else {
955 ctx.push();
956 current_scope = idx;
957 }
958 } else if (keyword == "DOMAIN") {
959 if (current_scope < 0) {
960 throw std::invalid_argument(
961 "Line " + std::to_string(line_num) + ": DOMAIN before SCOPE");
962 }
963 std::string sym, dom_str;
964 if (!(ls >> sym >> dom_str)) {
965 throw std::invalid_argument(
966 "Line " + std::to_string(line_num) + ": DOMAIN requires symbol and domain");
967 }
968 Domain dom = parse_domain(dom_str, line_num);
969 require_deserialization_update(
970 ctx.current_properties().declare_domain_checked(sym, dom), line_num, keyword);
971 } else if (keyword == "SIGN") {
972 if (current_scope < 0) {
973 throw std::invalid_argument(
974 "Line " + std::to_string(line_num) + ": SIGN before SCOPE");
975 }
976 std::string sym, sign_str;
977 if (!(ls >> sym >> sign_str)) {
978 throw std::invalid_argument(
979 "Line " + std::to_string(line_num) + ": SIGN requires symbol and sign");
980 }
981 Sign s = parse_sign(sign_str, line_num);
982 require_deserialization_update(
983 ctx.current_properties().declare_sign_checked(sym, s), line_num, keyword);
984 } else if (keyword == "PARITY") {
985 if (current_scope < 0) {
986 throw std::invalid_argument(
987 "Line " + std::to_string(line_num) + ": PARITY before SCOPE");
988 }
989 std::string sym, par_str;
990 if (!(ls >> sym >> par_str)) {
991 throw std::invalid_argument(
992 "Line " + std::to_string(line_num) + ": PARITY requires symbol and parity");
993 }
994 Parity p = parse_parity(par_str, line_num);
995 require_deserialization_update(
996 ctx.current_properties().declare_parity_checked(sym, p), line_num, keyword);
997 } else if (keyword == "BOUNDED") {
998 if (current_scope < 0) {
999 throw std::invalid_argument(
1000 "Line " + std::to_string(line_num) + ": BOUNDED before SCOPE");
1001 }
1002 std::string sym, bnd_str;
1003 if (!(ls >> sym >> bnd_str)) {
1004 throw std::invalid_argument(
1005 "Line " + std::to_string(line_num) + ": BOUNDED requires symbol and boundedness");
1006 }
1007 Boundedness b = parse_boundedness(bnd_str, line_num);
1008 require_deserialization_update(
1009 ctx.current_properties().declare_bounded_checked(sym, b), line_num, keyword);
1010 } else if (keyword == "TRANSCENDENTAL") {
1011 if (current_scope < 0) {
1012 throw std::invalid_argument(
1013 "Line " + std::to_string(line_num) + ": TRANSCENDENTAL before SCOPE");
1014 }
1015 std::string sym;
1016 if (!(ls >> sym)) {
1017 throw std::invalid_argument(
1018 "Line " + std::to_string(line_num) + ": TRANSCENDENTAL requires symbol");
1019 }
1020 require_deserialization_update(
1021 ctx.current_properties().declare_transcendental_checked(sym), line_num, keyword);
1022 } else if (keyword == "FINITENESS") {
1023 if (current_scope < 0) {
1024 throw std::invalid_argument(
1025 "Line " + std::to_string(line_num) + ": FINITENESS before SCOPE");
1026 }
1027 std::string sym, fin_str;
1028 if (!(ls >> sym >> fin_str)) {
1029 throw std::invalid_argument(
1030 "Line " + std::to_string(line_num) + ": FINITENESS requires symbol and value");
1031 }
1032 Finiteness f = parse_finiteness(fin_str, line_num);
1033 require_deserialization_update(
1034 ctx.current_properties().declare_finiteness_checked(sym, f), line_num, keyword);
1035 } else if (keyword == "DEFINITENESS") {
1036 if (current_scope < 0) {
1037 throw std::invalid_argument(
1038 "Line " + std::to_string(line_num) + ": DEFINITENESS before SCOPE");
1039 }
1040 std::string sym, def_str;
1041 if (!(ls >> sym >> def_str)) {
1042 throw std::invalid_argument(
1043 "Line " + std::to_string(line_num) + ": DEFINITENESS requires symbol and value");
1044 }
1045 Definiteness d = parse_definiteness(def_str, line_num);
1046 require_deserialization_update(
1047 ctx.current_properties().declare_definiteness_checked(sym, d), line_num, keyword);
1048 } else if (keyword == "PERIODIC") {
1049 if (current_scope < 0) {
1050 throw std::invalid_argument(
1051 "Line " + std::to_string(line_num) + ": PERIODIC before SCOPE");
1052 }
1053 std::string sym;
1054 if (!(ls >> sym)) {
1055 throw std::invalid_argument(
1056 "Line " + std::to_string(line_num) + ": PERIODIC requires symbol");
1057 }
1058 std::string period_str;
1059 std::getline(ls, period_str);
1060 size_t start = period_str.find_first_not_of(' ');
1061 if (start == std::string::npos) {
1062 throw std::invalid_argument(
1063 "Line " + std::to_string(line_num) + ": PERIODIC requires period value");
1064 }
1065 period_str = period_str.substr(start);
1066 std::shared_ptr<SymbolicExpr> period_expr;
1067 try {
1068 double period_val = std::stod(period_str);
1069 period_expr = lamina::detail::make_expression_ptr(
1070 lamina::detail::make_node<NumberNode>(static_cast<lmmc_real_t>(period_val)));
1071 } catch (...) {
1072 period_expr = lamina::detail::make_expression_ptr(
1073 lamina::detail::make_node<VariableNode>(period_str));
1074 }
1075 require_deserialization_update(
1076 ctx.current_properties().declare_periodic_checked(sym, period_expr),
1077 line_num, keyword);
1078 } else if (keyword == "RELATION") {
1079 if (current_scope < 0) {
1080 throw std::invalid_argument(
1081 "Line " + std::to_string(line_num) + ": RELATION before SCOPE");
1082 }
1083 // The rest of the line after "RELATION " is: lhs_str OP rhs_str
1084 std::string rest;
1085 std::getline(ls, rest);
1086 size_t rstart = rest.find_first_not_of(' ');
1087 if (rstart == std::string::npos) {
1088 throw std::invalid_argument(
1089 "Line " + std::to_string(line_num) + ": RELATION requires lhs, op, and rhs");
1090 }
1091 rest = rest.substr(rstart);
1092 std::string op_str;
1093 size_t op_pos = std::string::npos;
1094 for (const auto& op_candidate : {"GEQ", "LEQ", "NEQ"}) {
1095 std::string search = std::string(" ") + op_candidate + " ";
1096 size_t pos = rest.find(search);
1097 if (pos != std::string::npos) {
1098 op_str = op_candidate;
1099 op_pos = pos;
1100 break;
1101 }
1102 }
1103 if (op_pos == std::string::npos) {
1104 for (const auto& op_candidate : {"GT", "LT", "EQ"}) {
1105 std::string search = std::string(" ") + op_candidate + " ";
1106 size_t pos = rest.find(search);
1107 if (pos != std::string::npos) {
1108 op_str = op_candidate;
1109 op_pos = pos;
1110 break;
1111 }
1112 }
1113 }
1114
1115 if (op_pos == std::string::npos) {
1116 throw std::invalid_argument(
1117 "Line " + std::to_string(line_num) + ": RELATION missing operator");
1118 }
1119
1120 std::string lhs_str = rest.substr(0, op_pos);
1121 std::string rhs_str = rest.substr(op_pos + 1 + op_str.size() + 1);
1122 RelationalNode::Op op = parse_relop(op_str, line_num);
1123 auto parse_simple_expr = [&](const std::string& s) -> SymbolicExpr {
1124 try {
1125 double val = std::stod(s);
1126 return lamina::detail::expression_from_node(lamina::detail::make_node<NumberNode>(
1127 static_cast<lmmc_real_t>(val)));
1128 } catch (...) {}
1129 return lamina::detail::expression_from_node(lamina::detail::make_node<VariableNode>(s));
1130 };
1131
1132 SymbolicExpr lhs = parse_simple_expr(lhs_str);
1133 SymbolicExpr rhs = parse_simple_expr(rhs_str);
1134 auto relation_result = ctx.current_relations().add_relation_checked(
1135 lhs, rhs, op, ctx.current_properties());
1136 if (!relation_result.has_value()) {
1137 throw std::invalid_argument(
1138 "Line " + std::to_string(line_num) + ": " +
1139 relation_result.error().message);
1140 }
1141 } else if (keyword == "CONDITIONAL") {
1142 if (current_scope < 0) {
1143 throw std::invalid_argument(
1144 "Line " + std::to_string(line_num) + ": CONDITIONAL before SCOPE");
1145 }
1146 std::string rest;
1147 std::getline(ls, rest);
1148 size_t rstart = rest.find_first_not_of(' ');
1149 if (rstart == std::string::npos) {
1150 throw std::invalid_argument(
1151 "Line " + std::to_string(line_num) +
1152 ": CONDITIONAL requires condition and conclusion");
1153 }
1154 rest = rest.substr(rstart);
1155 size_t arrow_pos = rest.find(") => (");
1156 if (arrow_pos == std::string::npos) {
1157 throw std::invalid_argument(
1158 "Line " + std::to_string(line_num) +
1159 ": CONDITIONAL malformed, expected '(cond) => (concl)'");
1160 }
1161
1162 // Extract condition string (between first '(' and the arrow)
1163 if (rest[0] != '(') {
1164 throw std::invalid_argument(
1165 "Line " + std::to_string(line_num) +
1166 ": CONDITIONAL condition must start with '('");
1167 }
1168 std::string cond_str = rest.substr(1, arrow_pos - 1);
1169
1170 // Extract conclusion string (between '(' after '=>' and final ')')
1171 std::string concl_part = rest.substr(arrow_pos + 6); // skip ") => ("
1172 if (concl_part.empty() || concl_part.back() != ')') {
1173 throw std::invalid_argument(
1174 "Line " + std::to_string(line_num) +
1175 ": CONDITIONAL conclusion must end with ')'");
1176 }
1177 std::string concl_str = concl_part.substr(0, concl_part.size() - 1);
1178 // We parse simple "var OP val" patterns for condition and conclusion.
1179 // Complex expression deserialization is deferred per task spec.
1180 auto parse_relational = [&](const std::string& s) -> SymbolicExpr {
1181 std::string op_token;
1182 size_t op_pos_local = std::string::npos;
1183 for (const auto& candidate : {"GEQ", "LEQ", "NEQ"}) {
1184 std::string search = std::string(" ") + candidate + " ";
1185 size_t pos = s.find(search);
1186 if (pos != std::string::npos) {
1187 op_token = candidate;
1188 op_pos_local = pos;
1189 break;
1190 }
1191 }
1192 if (op_pos_local == std::string::npos) {
1193 for (const auto& candidate : {"GT", "LT", "EQ"}) {
1194 std::string search = std::string(" ") + candidate + " ";
1195 size_t pos = s.find(search);
1196 if (pos != std::string::npos) {
1197 op_token = candidate;
1198 op_pos_local = pos;
1199 break;
1200 }
1201 }
1202 }
1203 if (op_pos_local == std::string::npos) {
1204 throw std::invalid_argument(
1205 "Line " + std::to_string(line_num) +
1206 ": CONDITIONAL cannot parse expression '" + s + "'");
1207 }
1208 std::string lhs_s = s.substr(0, op_pos_local);
1209 std::string rhs_s = s.substr(op_pos_local + 1 + op_token.size() + 1);
1210 RelationalNode::Op rel_op = parse_relop(op_token, line_num);
1211 auto parse_token = [](const std::string& tok) -> std::shared_ptr<const SymbolicNode> {
1212 try {
1213 double val = std::stod(tok);
1214 return lamina::detail::make_node<NumberNode>(static_cast<lmmc_real_t>(val));
1215 } catch (...) {}
1216 return lamina::detail::make_node<VariableNode>(tok);
1217 };
1218
1219 auto lhs_node = parse_token(lhs_s);
1220 auto rhs_node = parse_token(rhs_s);
1221 return lamina::detail::expression_from_node(lamina::detail::make_node<RelationalNode>(lhs_node, rhs_node, rel_op));
1222 };
1223
1224 SymbolicExpr cond_expr = parse_relational(cond_str);
1225 SymbolicExpr concl_expr = parse_relational(concl_str);
1226 require_deserialization_update(
1227 ctx.assume_conditional_checked(cond_expr, concl_expr), line_num, keyword);
1228 } else if (keyword == "CONTINUOUS") {
1229 if (current_scope < 0) {
1230 throw std::invalid_argument(
1231 "Line " + std::to_string(line_num) + ": CONTINUOUS before SCOPE");
1232 }
1233 std::string sym;
1234 if (!(ls >> sym)) {
1235 throw std::invalid_argument(
