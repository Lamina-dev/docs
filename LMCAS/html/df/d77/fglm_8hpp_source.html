24/** @brief FGLM 算法中使用的多项式表示,以 (单项式, 系数) 对列表存储 */
25struct FGLMPoly {
26 std::vector<std::pair<Monomial, Rational>> terms; ///< 项列表
27 size_t num_vars; ///< 变量个数
28
29 /** @brief 默认构造 */
30 FGLMPoly() : num_vars(0) {}
31
32 /**
33 * @brief 构造指定变量数的零多项式
34 * @param n 变量个数
35 */
36 explicit FGLMPoly(size_t n) : num_vars(n) {}
37
38 /** @brief 判断是否为零多项式 */
39 bool is_zero() const { return terms.empty(); }
40
41 /** @brief 获取首项单项式 */
42 Monomial LM() const {
43 if (terms.empty()) return Monomial();
44 return terms.front().first;
45 }
46
47 /** @brief 获取首项系数 */
48 Rational LC() const {
49 if (terms.empty()) return Rational(0);
50 return terms.front().second;
51 }
52
53 /**
54 * @brief 添加一项
55 * @param m 单项式
56 * @param c 系数(零系数不添加)
57 */
58 void add_term(const Monomial& m, const Rational& c) {
59 if (!c.is_zero()) {
60 terms.emplace_back(m, c);
61 }
62 }
63
64 /**
65 * @brief 按指定单项式序排序各项
66 * @param order 单项式序
67 */
68 void sort_terms(const MonomialOrder& order) {
69 std::sort(terms.begin(), terms.end(),
70 [&order](const std::pair<Monomial, Rational>& a,
71 const std::pair<Monomial, Rational>& b) {
72 return order(a.first, b.first);
73 });
74 }
75
76 /** @brief 合并同类项并去除零项 */
77 void normalize() {
78
79 std::vector<std::pair<Monomial, Rational>> cleaned;
80 for (auto& [m, c] : terms) {
81 if (c.is_zero()) continue;
82 if (!cleaned.empty() && cleaned.back().first == m) {
83 cleaned.back().second = cleaned.back().second + c;
84 if (cleaned.back().second.is_zero()) {
85 cleaned.pop_back();
86 }
87 } else {
88 cleaned.emplace_back(m, c);
89 }
90 }
91 terms = std::move(cleaned);
92 }

94 /**
95 * @brief 对给定基进行多项式约化
96 * @param basis Gröbner 基
97 * @param order 单项式序
98 * @return 约化后的余式
99 */
100 FGLMPoly reduce(const std::vector<FGLMPoly>& basis,
101 const MonomialOrder& order) const {
102 FGLMPoly r(num_vars);
103 FGLMPoly f = *this;
104 f.sort_terms(order);
105 f.normalize();
106
107 int max_steps = 10000;
108 while (!f.is_zero() && max_steps-- > 0) {
109 bool reduced = false;
110 for (const auto& g : basis) {
111 if (g.is_zero()) continue;
112 Monomial lm_f = f.LM();
113 Monomial lm_g = g.LM();
114 if (divides_monomial(lm_g, lm_f)) {
115
116 Monomial quot_mon(num_vars, 0);
117 for (size_t i = 0; i < num_vars; ++i) {
118 int fi = (i < lm_f.size()) ? lm_f[i] : 0;
119 int gi = (i < lm_g.size()) ? lm_g[i] : 0;
120 quot_mon[i] = fi - gi;
121 }
122 Rational quot_coeff = f.LC() / g.LC();
123
124 FGLMPoly subtracted(num_vars);
125 for (const auto& [gm, gc] : g.terms) {
126 Monomial product_mon(num_vars, 0);
127 for (size_t i = 0; i < num_vars; ++i) {
128 int qi = (i < quot_mon.size()) ? quot_mon[i] : 0;
129 int gmi = (i < gm.size()) ? gm[i] : 0;
130 product_mon[i] = qi + gmi;
131 }
132 subtracted.add_term(product_mon, quot_coeff * gc);
133 }
134
135 for (const auto& [sm, sc] : subtracted.terms) {
136 f.add_term(sm, -sc);
137 }
138 f.sort_terms(order);
139 f.normalize();
140 reduced = true;
141 break;
142 }
143 }
144 if (!reduced) {
145
146 r.add_term(f.LM(), f.LC());
147 f.terms.erase(f.terms.begin());
148 f.normalize();
149 }
150 }
151 r.sort_terms(order);
152 r.normalize();
153 return r;
154 }

191/**
192 * @brief 判断理想是否为零维(每个变量都有纯幂次首项)
193 * @param basis Gröbner 基
194 * @param num_vars 变量个数
195 * @return 零维返回 true
196 */
197inline bool is_zero_dimensional(const std::vector<FGLMPoly>& basis, size_t num_vars) {
198 if (basis.empty()) return false;
199
200 for (size_t var = 0; var < num_vars; ++var) {
201 bool found = false;
202 for (const auto& g : basis) {
203 if (g.is_zero()) continue;
204 Monomial lm = g.LM();
205
206 bool is_pure = true;
207 for (size_t i = 0; i < num_vars; ++i) {
208 int exp_i = (i < lm.size()) ? lm[i] : 0;
209 if (i == var) {
210 if (exp_i == 0) { is_pure = false; break; }
211 } else {
212 if (exp_i != 0) { is_pure = false; break; }
213 }
214 }
215 if (is_pure) { found = true; break; }
216 }
217 if (!found) return false;
218 }
219 return true;
220}

222/**
223 * @brief 计算商空间维数(标准单项式个数)
224 * @param basis Gröbner 基
225 * @param num_vars 变量个数
226 * @param max_degree 搜索的最大次数上限
227 * @return 商空间维数;-1 表示当前搜索界内维数保持未知
228 */
229inline int quotient_dimension(const std::vector<FGLMPoly>& basis,
230 size_t num_vars,
231 int max_degree = 50) {
232 if (basis.empty()) return -1;
233
234 std::vector<Monomial> lead_mons;
235 for (const auto& g : basis) {
236 if (!g.is_zero()) {
237 lead_mons.push_back(g.LM());
238 }
239 }
240
241 int count = 0;
242 const int limit = 1000;
243
244 for (int d = 0; d <= max_degree && count <= limit; ++d) {
245
246 std::vector<Monomial> degree_mons;
247 Monomial current(num_vars, 0);
248
249 std::function<void(size_t, int)> generate =
250 [&](size_t var_idx, int remaining_deg) {
251 if (var_idx == num_vars - 1) {
252 current[var_idx] = remaining_deg;
253 degree_mons.push_back(current);
254 current[var_idx] = 0;
255 return;
256 }
257 for (int e = 0; e <= remaining_deg; ++e) {
258 current[var_idx] = e;
259 generate(var_idx + 1, remaining_deg - e);
260 }
261 current[var_idx] = 0;
262 };
263
264 generate(0, d);
265
266 for (const auto& m : degree_mons) {
267
268 bool divisible = false;
269 for (const auto& lm : lead_mons) {
270 if (divides_monomial(lm, m)) {
271 divisible = true;
272 break;
273 }
274 }
275 if (!divisible) {
276 ++count;
277 if (count > limit) return count;
278 }
279 }
280
281 if (degree_mons.size() > 0) {
282 bool any_standard = false;
283 for (const auto& m : degree_mons) {
284 bool divisible = false;
285 for (const auto& lm : lead_mons) {
286 if (divides_monomial(lm, m)) { divisible = true; break; }
287 }
288 if (!divisible) { any_standard = true; break; }
289 }
290 if (!any_standard && d > 0) break;
291 }
292 }
293 return count;
294}

307/** @brief 按单项式序递增枚举单项式的迭代器 */
308class MonomialEnumerator {
309public:
310 /**
311 * @brief 构造单项式枚举器
312 * @param num_vars 变量个数
313 * @param order 单项式序
314 * @param max_deg 最大次数上限
315 */
316 MonomialEnumerator(size_t num_vars, const MonomialOrder& order, int max_deg = 100)
317 : num_vars_(num_vars), order_(order), max_degree_(max_deg) {
318
319 Monomial one(num_vars, 0);
320 push(one);
321 }
322
323 /**
324 * @brief 获取下一个单项式
325 * @param out 输出单项式
326 * @return 成功返回 true,枚举结束返回 false
327 */
328 bool next(Monomial& out) {
329 if (heap_.empty()) return false;
330 out = heap_.front();
331 std::pop_heap(heap_.begin(), heap_.end(), cmp_);
332 heap_.pop_back();
333
334 int out_deg = total_degree(out);
335 if (out_deg < max_degree_) {
336 for (size_t i = 0; i < num_vars_; ++i) {
337 Monomial succ = out;
338 succ[i] += 1;
339 push(succ);
340 }
341 }
342 return true;
343 }
344
345private:
346 size_t num_vars_;
347 MonomialOrder order_;
348 int max_degree_;
349 std::vector<Monomial> heap_;
350 std::set<Monomial> visited_;
351
352 struct HeapCmp {
353 const MonomialOrder* order;
354 HeapCmp() : order(nullptr) {}
355 HeapCmp(const MonomialOrder* o) : order(o) {}
356 bool operator()(const Monomial& a, const Monomial& b) const {
357
358 return order->operator()(a, b);
359 }
360 };
361 HeapCmp cmp_{&order_};
362
363 void push(const Monomial& m) {
364 if (visited_.count(m)) return;
365 visited_.insert(m);
366 heap_.push_back(m);
367 std::push_heap(heap_.begin(), heap_.end(), cmp_);
368 }
369};

371/** @brief 增量式高斯消元器,用于检测向量线性相关性 */
372class GaussianEliminator {
373public:
374 GaussianEliminator() {}
375
376 /**
377 * @brief 尝试将向量加入行空间
378 * @param v 待加入的向量
379 * @param combination 输出线性组合系数(当线性相关时有效)
380 * @return true 表示线性无关(已加入),false 表示线性相关
381 */
382 bool add_vector(const std::vector<Rational>& v,
383 std::vector<Rational>& combination) {
384 size_t n = v.size();
385
386 for (auto& row : rows_) {
387 row.resize(n, Rational(0));
388 }
389
390 std::vector<Rational> working = v;
391
392 combination.assign(basis_count_, Rational(0));
393
394 for (size_t i = 0; i < pivots_.size(); ++i) {
395 size_t col = pivots_[i];
396 if (col >= working.size()) continue;
397 if (working[col].is_zero()) continue;
398
399 Rational factor = working[col] / rows_[i][col];
400 for (size_t j = 0; j < n; ++j) {
401 if (j < working.size() && j < rows_[i].size()) {
402 working[j] = working[j] - factor * rows_[i][j];
403 }
404 }
405
406 for (size_t j = 0; j < combination.size(); ++j) {
407 combination[j] = combination[j] - factor * combinations_[i][j];
408 }
409 }
410
411 size_t pivot_col = n;
412 for (size_t j = 0; j < n; ++j) {
413 if (!working[j].is_zero()) {
414 pivot_col = j;
415 break;
416 }
417 }
418
419 if (pivot_col == n) {
420
421 for (auto& c : combination) {
422 c = -c;
423 }
424 return false;
425 }
426
427 Rational pivot_val = working[pivot_col];
428 for (size_t j = 0; j < n; ++j) {
429 working[j] = working[j] / pivot_val;
430 }
431
432 std::vector<Rational> comb_row(basis_count_ + 1, Rational(0));
433 for (size_t j = 0; j < combination.size(); ++j) {
434 comb_row[j] = combination[j] / pivot_val;
435 }
436
437 comb_row[basis_count_] = Rational(1) / pivot_val;
438
439 rows_.push_back(working);
440 pivots_.push_back(pivot_col);
441 combinations_.push_back(comb_row);
442 ++basis_count_;
443
444 return true;
445 }
446
447 /** @brief 获取当前秩 */
448 size_t rank() const { return basis_count_; }
449
450private:
451 std::vector<std::vector<Rational>> rows_;
452 std::vector<size_t> pivots_;
453 std::vector<std::vector<Rational>> combinations_;
454 size_t basis_count_ = 0;
455};

458 * @brief 将多项式转换为坐标向量(相对于单项式基)
459 * @param p 多项式
460 * @param all_monomials 单项式基(可能被扩展)

464inline std::vector<Rational> poly_to_vector(
465 const FGLMPoly& p,
466 std::vector<Monomial>& all_monomials,
467 std::map<Monomial, size_t>& monomial_index) {

478 for (const auto& [m, c] : p.terms) {
479 v[monomial_index[m]] = c;
480 }
481 return v;
482}

486/**
487 * @brief FGLM 算法:将零维理想的 Gröbner 基从源序转换到目标序
488 * @param source_basis 源序下的 Gröbner 基
489 * @param source_order 源单项式序
490 * @param target_order 目标单项式序
491 * @param num_vars 变量个数
492 * @return 目标序下的 Gröbner 基
493 * @throw std::runtime_error 理想非零维或维数过大时抛出
494 * @see J.-C. Faugère, P. Gianni, D. Lazard, and T. Mora,
495 * "Efficient Computation of Zero-dimensional Gröbner Bases by Change of Ordering,"
496 * Journal of Symbolic Computation 16(4), 1993.
497 */
498inline std::vector<FGLMPoly> fglm_convert(
499 const std::vector<FGLMPoly>& source_basis,
500 const MonomialOrder& source_order,
501 const MonomialOrder& target_order,
502 size_t num_vars) {
503
504 if (source_basis.empty()) {
505 return {};
506 }
507
508 if (!is_zero_dimensional(source_basis, num_vars)) {
509 throw std::runtime_error(
510 "FGLM: ideal is not zero-dimensional. "
511 "FGLM only works for zero-dimensional ideals.");
512 }
513
514 int dim = quotient_dimension(source_basis, num_vars);
515 if (dim <= 0 || dim > 1000) {
516 throw std::runtime_error(
517 "FGLM: quotient dimension is too large or could not be determined. "
518 "dim = " + std::to_string(dim));
519 }
520
521 std::vector<FGLMPoly> target_basis;
522
523 std::vector<Monomial> basis_monomials;
524
525 std::vector<Monomial> standard_monomials;
526 std::map<Monomial, size_t> monomial_index;
527 {
528 std::vector<Monomial> source_lms;
529 for (const auto& g : source_basis) {
530 if (!g.is_zero()) source_lms.push_back(g.LM());
531 }
532
533 for (int d = 0; d <= dim + 2 && (int)standard_monomials.size() < dim; ++d) {
534 Monomial current(num_vars, 0);
535 std::function<void(size_t, int)> gen = [&](size_t var_idx, int remaining) {
536 if (var_idx == num_vars - 1) {
537 current[var_idx] = remaining;
538
539 bool is_std = true;
540 for (const auto& lm : source_lms) {
541 if (divides_monomial(lm, current)) { is_std = false; break; }
542 }
543 if (is_std) standard_monomials.push_back(current);
544 current[var_idx] = 0;
545 return;
546 }
547 for (int e = 0; e <= remaining; ++e) {
548 current[var_idx] = e;
549 gen(var_idx + 1, remaining - e);
550 }
551 current[var_idx] = 0;
552 };
553 gen(0, d);
554 }
555
556 for (size_t i = 0; i < standard_monomials.size(); ++i) {
557 monomial_index[standard_monomials[i]] = i;
558 }
559 }
560
561 auto nf_to_vector = [&](const FGLMPoly& nf) -> std::vector<Rational> {
562 std::vector<Rational> v(standard_monomials.size(), Rational(0));
563 for (const auto& [m, c] : nf.terms) {
564 auto it = monomial_index.find(m);
565 if (it != monomial_index.end()) {
566 v[it->second] = c;
567 }
568 }
569 return v;
570 };
571
572 detail::GaussianEliminator gauss;
573
574 int max_mono_degree = dim + 5;
575 detail::MonomialEnumerator enumerator(num_vars, target_order, max_mono_degree);
576
577 std::vector<Monomial> target_lms;
578
579 const int max_iterations = dim * 10 + 50;
580 int iterations = 0;
581
582 Monomial t;
583 while (enumerator.next(t) && iterations < max_iterations) {
584 ++iterations;
585
586 bool skip = false;
587 for (const auto& lm : target_lms) {
588 if (divides_monomial(lm, t)) {
589 skip = true;
590 break;
591 }
592 }
593 if (skip) continue;
594
595 FGLMPoly t_poly = FGLMPoly::from_monomial(t, num_vars);
596 FGLMPoly nf = normal_form(t_poly, source_basis, source_order);
597
598 std::vector<Rational> v = nf_to_vector(nf);
599
600 std::vector<Rational> combination;
601 bool independent = gauss.add_vector(v, combination);
602
603 if (independent) {
604
605 basis_monomials.push_back(t);
606
607 if (static_cast<int>(basis_monomials.size()) >= dim) {
608
609 }
610 } else {
611
612 FGLMPoly new_elem(num_vars);
613 new_elem.add_term(t, Rational(1));
614
615 for (size_t i = 0; i < combination.size() && i < basis_monomials.size(); ++i) {
616 if (!combination[i].is_zero()) {
617
618 new_elem.add_term(basis_monomials[i], -combination[i]);
619 }
620 }
621
622 new_elem.sort_terms(target_order);
623 new_elem.normalize();
624
625 if (!new_elem.is_zero()) {
626 Rational lc = new_elem.LC();
627 if (!(lc == Rational(1))) {
628 for (auto& [m, c] : new_elem.terms) {
629 c = c / lc;
630 }
631 }
632 }
633
634 if (!new_elem.is_zero()) {
635 target_lms.push_back(new_elem.LM());
636 target_basis.push_back(std::move(new_elem));
637 }
638 }
639
640 if (static_cast<int>(basis_monomials.size()) >= dim) {
641
642 bool complete = true;
643 for (size_t var = 0; var < num_vars; ++var) {
644 bool has_gen = false;
645 for (const auto& lm : target_lms) {
646
647 bool is_pure = true;
648 for (size_t i = 0; i < num_vars; ++i) {
649 int exp_i = (i < lm.size()) ? lm[i] : 0;
650 if (i == var) {
651 if (exp_i == 0) { is_pure = false; break; }
652 } else {
653 if (exp_i != 0) { is_pure = false; break; }
654 }
655 }
656 if (is_pure) { has_gen = true; break; }
657 }
658 if (!has_gen) { complete = false; break; }
659 }
660 if (complete) break;
661 }
662 }
663
664 return target_basis;
665}
