18/** @brief FGLM 算法中使用的多项式表示,以 (单项式, 系数) 对列表存储 */
19struct LMCAS_API FGLMPoly {
20 std::vector<std::pair<Monomial, Rational>> terms; ///< 项列表
21 size_t num_vars; ///< 变量个数
22
23 /** @brief 默认构造 */
24 FGLMPoly() : num_vars(0) {}
25
26 /**
27 * @brief 构造指定变量数的零多项式
28 * @param n 变量个数
29 */
30 explicit FGLMPoly(size_t n) : num_vars(n) {}
31
32 /** @brief 判断是否为零多项式 */
33 bool is_zero() const { return terms.empty(); }
34
35 /** @brief 获取首项单项式 */
36 Monomial lead_monomial() const {
37 if (terms.empty()) return Monomial();
38 return terms.front().first;
39 }
40
41 /** @brief 获取首项系数 */
42 Rational lead_coeff() const {
43 if (terms.empty()) return Rational(0);
44 return terms.front().second;
45 }
46
47 /**
48 * @brief 添加一项
49 * @param m 单项式
50 * @param c 系数(零系数不添加)
51 */
52 void add_term(const Monomial& m, const Rational& c) {
53 if (!c.is_zero()) {
54 terms.emplace_back(m, c);
55 }
56 }
