19std::shared_ptr<const SymbolicNode> collect_log_power_factors(
20 const std::shared_ptr<const SymbolicNode>& exponent,
21 std::vector<std::shared_ptr<const SymbolicNode>>& coefficients) {
22 std::shared_ptr<const SymbolicNode> logarithm;
23 auto collect = [&](const std::shared_ptr<const SymbolicNode>& op) {
24 auto f = std::dynamic_pointer_cast<const FunctionNode>(op);
25 if (!logarithm && f && f->type() == FunctionNode::FuncType::Ln && f->arguments().size() == 1)
26 { logarithm = op; }
27 else { coefficients.push_back(op); }
28 };
29 if (auto product = std::dynamic_pointer_cast<const MultiplyNode>(exponent))
30 { for (const auto& op : product->operands()) { collect(op); } }
31 else { collect(exponent); }
32 return logarithm;
33}
34}

38std::shared_ptr<const SymbolicNode> LimitVisitor::linear_exponent_coefficient(
39 const std::shared_ptr<const SymbolicNode>& node) const {
40 if (is_limit_variable(node)) {
41 return LMCAS::detail::make_node<NumberNode>(BigInt(1));
42 }
43 auto product = std::dynamic_pointer_cast<const MultiplyNode>(node);
44 if (!product) {
45 return nullptr;
46 }
47 bool has_variable = false;
48 std::vector<std::shared_ptr<const SymbolicNode>> factors;
49 for (const auto& operand : product->operands()) {
50 if (!has_variable && is_limit_variable(operand)) has_variable = true;
51 else factors.push_back(operand);
52 }
53 if (!has_variable) {
54 return nullptr;
55 }
56 return make_product_or_one(factors);
57}

59bool LimitVisitor::exponential_power_limit(const PowerNode& node) {
60 auto sum = std::dynamic_pointer_cast<const AddNode>(node.base());
61 if (!sum) {
62 return false;
63 }
64 auto coefficient = linear_exponent_coefficient(node.exponent());
65 if (!coefficient) {
66 return false;
67 }
68 auto term = one_plus_term(*sum);
69 if (!term) {
70 return false;
71 }
72 auto numerator = variable_fraction_numerator(term);
73 if (!numerator) {
74 return false;
75 }
76 if (!is_zero_limit(eval_limit(term))) return false;
77 auto inner = LMCAS::detail::make_node<MultiplyNode>(
78 std::vector<std::shared_ptr<const SymbolicNode>>{numerator, coefficient});
79 auto value = eval_limit(inner);
80 if (!value) return false;
81 auto exponential = LMCAS::detail::make_node<FunctionNode>(
82 FunctionNode::FuncType::Exp,
83 std::vector<std::shared_ptr<const SymbolicNode>>{value});
84 NormalizationVisitor norm(*context_, detail::no_facts(), Domain::Real, rewrite_budget());
85 exponential->accept(norm);
86 result = norm.get_result();
87 return true;
88}

91 const std::shared_ptr<const SymbolicNode>& base,
92 const std::shared_ptr<const SymbolicNode>& exponent) {
93 auto logarithm = LMCAS::detail::make_node<FunctionNode>(
94 FunctionNode::FuncType::Ln, std::vector<std::shared_ptr<const SymbolicNode>>{base});
95 auto product = LMCAS::detail::make_node<MultiplyNode>(
96 std::vector<std::shared_ptr<const SymbolicNode>>{exponent, logarithm});
97 auto value = eval_limit(product);
98 if (!value) {
99 return nullptr;
100 }
101 if (!is_inf(value)) {
102 auto exponential = LMCAS::detail::make_node<FunctionNode>(
103 FunctionNode::FuncType::Exp, std::vector<std::shared_ptr<const SymbolicNode>>{value});
104 NormalizationVisitor norm(*context_, detail::no_facts(), Domain::Real, rewrite_budget());
105 exponential->accept(norm);
106 return norm.get_result();
107 }
108 if (!is_neg_inf(value)) {
109 return signed_infinity(1);
110 }
111 return LMCAS::detail::make_node<NumberNode>(BigInt(0));

136void LimitVisitor::visit(const PowerNode& node) {
137 if (is_limit_at_neg_infinity()) {
138 auto value = handle_neg_infinity_limit(
139 LMCAS::detail::make_node<PowerNode>(node.base(), node.exponent()));
140 if (value) {
141 result = value;
142 return;
143 }
144 }
145 if (is_limit_at_infinity() && exponential_power_limit(node)) {
146 return;
147 }
148 if (node.base()->is_one() && defined_near_point(node.exponent())) {
149 result = detail::make_node<NumberNode>(BigInt(1));
150 return;
151 }
152 if (node.exponent()->is_zero()) {
153 auto sign = determine_sign_near_point(node.base(), direction);
154 if (sign && *sign != 0) {
155 result = detail::make_node<NumberNode>(BigInt(1));
156 return;
157 }
158 }
159 node.base()->accept(*this);
160 auto base = result;
161 node.exponent()->accept(*this);
162 auto exponent = result;
163 if (!base || !exponent) { result = nullptr; return; }
164 if (classify_power_form(base, exponent) != IndeterminateForm::None) {
165 auto value = resolve_exponential_form(node.base(), node.exponent());
166 if (value) {
167 result = value;
168 return;
169 }
170 }
171 auto singular = singular_power_limit(node, base, exponent);
172 if (singular) {
173 result = singular;
174 return;
175 }
176 NormalizationVisitor norm(*context_, detail::no_facts(), Domain::Real, rewrite_budget());
177 LMCAS::detail::make_node<PowerNode>(base, exponent)->accept(norm);
178 result = norm.get_result();
179}

182 switch (type) {
183 case FunctionNode::FuncType::Exp:
184 result = negative ? std::static_pointer_cast<const SymbolicNode>(
185 LMCAS::detail::make_node<NumberNode>(BigInt(0))) : signed_infinity(1);
186 return true;

188 result = negative ? nullptr : signed_infinity(1);

193 std::vector<std::shared_ptr<const SymbolicNode>>{
194 result, LMCAS::detail::make_node<VariableNode>("pi", true)});
195 return true;
196 default:
197 return false;

201bool LimitVisitor::composed_function_limit(const FunctionNode& node) {
202 if (node.arguments().size() != 1 ||
203 node.type() == FunctionNode::FuncType::Infinity) {
204 return false;
205 }
206 auto inner = eval_limit(node.arguments()[0]);
207 if (!inner) {
208 return false;
209 }
210 if (!is_inf(inner)) {
211 auto candidate = LMCAS::detail::make_node<FunctionNode>(node.type(),
212 std::vector<std::shared_ptr<const SymbolicNode>>{inner});
213 std::optional<detail::AssumptionFacts> assumed;
214 if (assumption_ctx_) { assumed.emplace(*assumption_ctx_); }
215 const auto& facts = assumed ? static_cast<const FactsQuery&>(*assumed) : detail::no_facts();
216 auto defined = detail::query_definedness(candidate, facts, Domain::Real, *context_);
217 if (!defined) { throw defined.error(); }
218 if (defined && defined.value() == Tribool::True) {
219 NormalizationVisitor norm(*context_, detail::no_facts(), Domain::Real, rewrite_budget());
220 candidate->accept(norm);
221 result = norm.get_result();
222 return true;
223 }
224 if (node.type() == FunctionNode::FuncType::Ln && is_zero_limit(inner)) {
225 auto sign = determine_sign_near_point(node.arguments()[0], direction);
226 if (sign && *sign > 0) {
227 result = signed_infinity(-1);
228 return true;
229 }
230 }
231 }
232 if (is_inf(inner)) {
233 return infinite_function_limit(node.type(), is_neg_inf(inner));
234 }
235 return false;
236}

238std::shared_ptr<const SymbolicNode> LimitVisitor::positive_tail_log_power(const FunctionNode& node) const {
239 if (!is_limit_at_infinity() || node.type() != FunctionNode::FuncType::Exp ||
240 node.arguments().size() != 1) { return nullptr; }
241 std::vector<std::shared_ptr<const SymbolicNode>> coefficients;
242 auto logarithm = collect_log_power_factors(node.arguments()[0], coefficients);
243 if (!logarithm) { return nullptr; }
244 auto f = std::dynamic_pointer_cast<const FunctionNode>(logarithm);
245 std::optional<detail::AssumptionFacts> assumed;
246 if (assumption_ctx_) { assumed.emplace(*assumption_ctx_); }
247 const auto& facts = assumed ? static_cast<const FactsQuery&>(*assumed) : detail::no_facts();
248 auto exponent = make_product_or_one(coefficients);
249 auto real_exponent = detail::query_real_value(exponent, facts, *context_);
250 if (!real_exponent) { throw real_exponent.error(); }
251 if (real_exponent.value() != Tribool::True || !defined_near_point(f->arguments()[0])) { return nullptr; }
252 auto positive = detail::query_positive_value(f->arguments()[0], facts, *context_);
253 if (!positive) { throw positive.error(); }
254 if (positive.value() != Tribool::True) {
255 LimitVisitor witness(var, point, direction, assumption_ctx_, context_);
256 auto sign = witness.determine_sign_near_point(f->arguments()[0], direction);
257 if (!sign || *sign <= 0) { return nullptr; }
258 }
259 return detail::make_node<PowerNode>(f->arguments()[0], exponent);
260}

262void LimitVisitor::visit(const FunctionNode& node) {
263 if (is_limit_at_neg_infinity() && node.type() != FunctionNode::FuncType::Infinity) {
264 auto value = handle_neg_infinity_limit(
265 LMCAS::detail::make_node<FunctionNode>(node.type(), node.arguments()));
266 if (value) {
267 result = value;
268 return;
269 }
270 }
271 if (auto rewritten = positive_tail_log_power(node)) {
272 result = eval_limit(rewritten);
273 return;
274 }
275 if (node.type() == FunctionNode::FuncType::Sgn && node.arguments().size() == 1) {
276 auto value = evaluate_sgn_limit(node.arguments()[0]);
277 result = value ? *value : nullptr;
278 return;
279 }
280 if (node.type() == FunctionNode::FuncType::Abs && node.arguments().size() == 1) {
281 auto value = evaluate_abs_limit(node.arguments()[0]);
282 if (value) {
283 result = *value;
284 return;
285 }
286 }
287 if (composed_function_limit(node)) {
288 return;
289 }
290 std::vector<std::shared_ptr<const SymbolicNode>> arguments;
291 for (const auto& argument : node.arguments()) {
292 argument->accept(*this);
293 if (!result) return;
294 arguments.push_back(result);
295 }
296 NormalizationVisitor norm(*context_, detail::no_facts(), Domain::Real, rewrite_budget());
297 LMCAS::detail::make_node<FunctionNode>(node.type(), arguments)->accept(norm);
298 result = norm.get_result();
299}
