31static VectorCalculusExprResult vector_calculus_try_definite(
32 const std::shared_ptr<SymbolicExpr>& integrand,
33 const std::string& var,
34 const std::shared_ptr<SymbolicExpr>& a,
35 const std::shared_ptr<SymbolicExpr>& b,
36 ComputationContext& context,
37 detail::RewriteBudget& transform_budget)
38{
39 Integrator integrator;
40 auto integrated = integrator.integrate_def_checked(
41 *integrand, var, *a, *b, context);
42 if (!integrated) {
43 return VectorCalculusExprResult::failure(integrated.error());
44 }
45 SymbolicExpr result = std::move(integrated.value());
46
47 if (contains_unevaluated_integral(LMCAS::detail::node(result))) {
48 return std::shared_ptr<SymbolicExpr>{};
49 }
50 auto res = LMCAS::detail::make_expression_ptr(result);
51 auto simplified = detail::make_expression_ptr(detail::normalize_node(detail::node(res), transform_budget));
52 if (simplified &&
53 contains_unevaluated_integral(LMCAS::detail::node(simplified))) {
54 return std::shared_ptr<SymbolicExpr>{};
55 }
56 return simplified ? simplified : res;
57}

59static VectorCalculusExprResult vector_calculus_numerical_definite(
60 const std::shared_ptr<SymbolicExpr>& integrand,
61 const std::string& var,
62 const std::shared_ptr<SymbolicExpr>& a,
63 const std::shared_ptr<SymbolicExpr>& b,
64 ComputationContext& context,
65 detail::RewriteBudget& transform_budget)
66{
67 double a_val = 0.0;
68 double b_val = 0.0;
69 if (!vector_calculus_checked_finite_numeric(a, a_val, context) ||
70 !vector_calculus_checked_finite_numeric(b, b_val, context)) {
71 return std::shared_ptr<SymbolicExpr>{};
72 }
73
74 int n = 1000;
75 double h = (b_val - a_val) / n;
76 if (!std::isfinite(h)) return std::shared_ptr<SymbolicExpr>{};
77 double sum = 0.0;
78
79 for (int i = 0; i <= n; ++i) {
80 auto step_result =
81 context.consume_steps(1, "vector_calculus.numeric_integral");
82 if (!step_result) {
83 return VectorCalculusExprResult::failure(step_result.error());
84 }
85 double xi = a_val + i * h;
86 if (!std::isfinite(xi)) return std::shared_ptr<SymbolicExpr>{};
87 auto xi_expr = SymbolicExpr::number(xi);
88 auto fi = detail::make_expression_ptr(detail::substitute_node(detail::node(integrand), var, detail::node(xi_expr), transform_budget));
89 double fi_val = 0.0;
90 if (!vector_calculus_checked_finite_numeric(fi, fi_val, context)) {
91 return std::shared_ptr<SymbolicExpr>{};
92 }
93
94 double weight = 1.0;
95 if (i == 0 || i == n) {
96 weight = 1.0;
97 } else if (i % 2 == 1) {
98 weight = 4.0;
99 } else {
100 weight = 2.0;
101 }
102 sum += weight * fi_val;
103 if (!std::isfinite(sum)) return std::shared_ptr<SymbolicExpr>{};
104 }
105 sum *= h / 3.0;
106 if (!std::isfinite(sum)) return std::shared_ptr<SymbolicExpr>{};
107
108 return SymbolicExpr::number(sum);
109}

156 const std::string& var,
157 const std::string& operation,
158 detail::RewriteBudget& transform_budget)
159{
160 if (!expr || !LMCAS::detail::node(expr)) {

164 try {
165 auto derivative = detail::make_expression_ptr(detail::differentiate_node(detail::node(expr), var, transform_budget));
166 return vector_calculus_simplify_strict(derivative, operation,

169 if (error.code != CasErrc::UnsupportedExpression) {
170 return VectorCalculusExprResult::failure(error);
171 }
172 return vector_calculus_inconclusive(
173 operation, "expression derivative is outside the supported domain");
174 }

177static VectorCalculusExprResult vector_calculus_substitute_coords_strict(
178 const std::shared_ptr<SymbolicExpr>& expr,
179 const std::vector<std::string>& coord_vars,
180 const VectorField& values,
181 const std::string& operation,
182 detail::RewriteBudget& transform_budget)
183{
184 if (!expr || !LMCAS::detail::node(expr) || coord_vars.size() != values.size()) {
185 return vector_calculus_inconclusive(
186 operation, "coordinate substitution is outside the supported domain");
187 }
188 auto substituted = expr;
189 for (size_t i = 0; i < coord_vars.size(); ++i) {
190 if (!values[i] || !LMCAS::detail::node(values[i])) {
191 return vector_calculus_inconclusive(
192 operation, "coordinate substitution is outside the supported domain");
193 }
194 substituted = detail::make_expression_ptr(detail::substitute_node(detail::node(substituted), coord_vars[i], detail::node(values[i]), transform_budget));
195 if (!substituted || !LMCAS::detail::node(substituted)) {
196 return vector_calculus_inconclusive(
197 operation, "coordinate substitution is outside the supported domain");
198 }
199 }
200 return vector_calculus_simplify_strict(substituted, operation,
201 "coordinate substitution is outside the supported domain", transform_budget);
202}

242static Result<VectorField> vector_calculus_cross_product_partials_strict(
243 const VectorField& parametrization,
244 const std::string& u,
245 const std::string& v,
246 const std::string& operation,
247 detail::RewriteBudget& transform_budget)
248{
249 VectorField r_u;
250 VectorField r_v;
251 r_u.reserve(3);
252 r_v.reserve(3);
253 for (size_t i = 0; i < 3; ++i) {
254 auto du = vector_calculus_differentiate_strict(parametrization[i], u, operation, transform_budget);
255 if (!du) return Result<VectorField>::failure(du.error());
256 auto dv = vector_calculus_differentiate_strict(parametrization[i], v, operation, transform_budget);
257 if (!dv) return Result<VectorField>::failure(dv.error());
258 r_u.push_back(std::move(du.value()));
259 r_v.push_back(std::move(dv.value()));
260 }
261
262 VectorField cross;
263 cross.reserve(3);
264 for (std::size_t index = 0; index < 3; ++index) {
265 auto component = vector_calculus_cross_component(
266 r_u, r_v, index);
267 auto checked = vector_calculus_simplify_strict(component, operation,
268 "surface normal construction is outside the supported domain", transform_budget);
269 if (!checked) {
270 return Result<VectorField>::failure(checked.error());
271 }
272 cross.push_back(std::move(checked.value()));
273 }
274 return Result<VectorField>::success(std::move(cross));
275}

278vector_calculus_detail::curve_integral_scalar_strict(

306 if (!speed_sq_checked) {

314 if (!speed_checked) {
315 return speed_checked;

320 if (!integrand_checked) {
321 return integrand_checked;

486static VectorCalculusExprResult vector_calculus_constant_integrand(
487 const std::shared_ptr<SymbolicExpr>& integrand,
488 const std::string& variable,
489 ComputationContext& context,
490 detail::RewriteBudget& transform_budget) {

513 const std::string& t,
514 const std::shared_ptr<SymbolicExpr>& a,

544VectorCalculusExprResult vector_calculus_detail::divergence_theorem_strict(
545 const VectorField& F,
546 const std::vector<std::string>& vars,
547 const std::pair<std::shared_ptr<SymbolicExpr>, std::shared_ptr<SymbolicExpr>>& x_bounds,
548 const std::pair<std::shared_ptr<SymbolicExpr>, std::shared_ptr<SymbolicExpr>>& y_bounds,
549 const std::pair<std::shared_ptr<SymbolicExpr>, std::shared_ptr<SymbolicExpr>>& z_bounds,
550 ComputationContext& context,
551 const std::string& operation,
552 detail::RewriteBudget& transform_budget)
553{
554 std::shared_ptr<SymbolicExpr> div_F;
555 for (size_t i = 0; i < 3; ++i) {
556 auto partial = vector_calculus_differentiate_strict(F[i], vars[i], operation, transform_budget);
557 if (!partial) return partial;
558 div_F = div_F ? SymbolicExpr::add(div_F, partial.value()) : partial.value();
559 auto div_checked = vector_calculus_simplify_strict(div_F, operation, "divergence theorem integrand is outside the supported domain", transform_budget);
560 if (!div_checked) return div_checked;
561 div_F = std::move(div_checked.value());
562 }
563
564 std::vector<IntegrationStep> steps;
565 steps.push_back({vars[2], z_bounds.first, z_bounds.second});
566 steps.push_back({vars[1], y_bounds.first, y_bounds.second});
567 steps.push_back({vars[0], x_bounds.first, x_bounds.second});
568 return vector_calculus_multiple_integral_strict(div_F, steps, context, operation, transform_budget);
569}

571static Result<VectorField> vector_calculus_curl_strict(
572 const VectorField& F,
573 const std::vector<std::string>& vars,
574 const std::string& operation,
575 detail::RewriteBudget& transform_budget)
576{
577 auto dFz_dy = vector_calculus_differentiate_strict(F[2], vars[1], operation, transform_budget);
578 if (!dFz_dy) {
579 return Result<VectorField>::failure(dFz_dy.error());
580 }
581 auto dFy_dz = vector_calculus_differentiate_strict(F[1], vars[2], operation, transform_budget);
582 if (!dFy_dz) {
583 return Result<VectorField>::failure(dFy_dz.error());
584 }
585 auto dFx_dz = vector_calculus_differentiate_strict(F[0], vars[2], operation, transform_budget);
586 if (!dFx_dz) {
587 return Result<VectorField>::failure(dFx_dz.error());
588 }
589 auto dFz_dx = vector_calculus_differentiate_strict(F[2], vars[0], operation, transform_budget);
590 if (!dFz_dx) {
591 return Result<VectorField>::failure(dFz_dx.error());
592 }
593 auto dFy_dx = vector_calculus_differentiate_strict(F[1], vars[0], operation, transform_budget);
594 if (!dFy_dx) {
595 return Result<VectorField>::failure(dFy_dx.error());
596 }
597 auto dFx_dy = vector_calculus_differentiate_strict(F[0], vars[1], operation, transform_budget);
598 if (!dFx_dy) {
599 return Result<VectorField>::failure(dFx_dy.error());
600 }
601
602 auto cx = SymbolicExpr::add(
603 dFz_dy.value(),
604 SymbolicExpr::multiply(SymbolicExpr::number(-1), dFy_dz.value()));
605 auto cx_checked = vector_calculus_simplify_strict(cx, operation, "curl construction is outside the supported domain", transform_budget);
606 if (!cx_checked) {
607 return Result<VectorField>::failure(cx_checked.error());
608 }
609
610 auto cy = SymbolicExpr::add(
611 dFx_dz.value(),
612 SymbolicExpr::multiply(SymbolicExpr::number(-1), dFz_dx.value()));
613 auto cy_checked = vector_calculus_simplify_strict(cy, operation, "curl construction is outside the supported domain", transform_budget);
614 if (!cy_checked) {
615 return Result<VectorField>::failure(cy_checked.error());
616 }
617
618 auto cz = SymbolicExpr::add(
619 dFy_dx.value(),
620 SymbolicExpr::multiply(SymbolicExpr::number(-1), dFx_dy.value()));
621 auto cz_checked = vector_calculus_simplify_strict(cz, operation, "curl construction is outside the supported domain", transform_budget);
622 if (!cz_checked) {
623 return Result<VectorField>::failure(cz_checked.error());
624 }
625
626 return Result<VectorField>::success(
627 VectorField{std::move(cx_checked.value()),
628 std::move(cy_checked.value()),
629 std::move(cz_checked.value())});
630}

632VectorCalculusExprResult vector_calculus_detail::stokes_theorem_strict(
633 const VectorField& F,
634 const std::vector<std::string>& vars,
635 const VectorField& parametrization,
636 const std::string& u,
637 const std::string& v,
638 const std::pair<std::shared_ptr<SymbolicExpr>, std::shared_ptr<SymbolicExpr>>& u_bounds,
639 const std::pair<std::shared_ptr<SymbolicExpr>, std::shared_ptr<SymbolicExpr>>& v_bounds,

641 const std::string& operation,
642 detail::RewriteBudget& transform_budget)
643{
644 auto curl_F = vector_calculus_curl_strict(F, vars, operation, transform_budget);
645 if (!curl_F) return VectorCalculusExprResult::failure(curl_F.error());
646

652 operation, transform_budget);

654
655 std::vector<IntegrationStep> steps;
656 steps.push_back({v, v_bounds.first, v_bounds.second});
657 steps.push_back({u, u_bounds.first, u_bounds.second});
658 return vector_calculus_multiple_integral_strict(dot_product.value(), steps, context, operation, transform_budget);
659}
