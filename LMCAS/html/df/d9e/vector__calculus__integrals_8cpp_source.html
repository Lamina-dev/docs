27 * @internal
28 * @brief 尝试符号定积分，若结果仍含未求值积分节点则返回 nullptr。
29 */
30static std::shared_ptr<SymbolicExpr> vector_calculus_try_definite(
31 const std::shared_ptr<SymbolicExpr>& integrand,
32 const std::string& var,
33 const std::shared_ptr<SymbolicExpr>& a,
34 const std::shared_ptr<SymbolicExpr>& b)
35{
36 Integrator integrator;
37 SymbolicExpr result = integrator.integrate_def(*integrand, var, *a, *b);
38
39 if (vector_calculus_contains_unevaluated_integral(lamina::detail::node(result))) {
40 return nullptr;
41 }
42 auto res = lamina::detail::make_expression_ptr(result);
43 auto simplified = res->simplify();
44 if (simplified && vector_calculus_contains_unevaluated_integral(lamina::detail::node(simplified))) {
45 return nullptr;
46 }
47 return simplified ? simplified : res;
48}

50/**
51 * @internal
52 * @brief 数值定积分回退（复合 Simpson 法）。
53 */
54static std::shared_ptr<SymbolicExpr> vector_calculus_numerical_definite(
55 const std::shared_ptr<SymbolicExpr>& integrand,
56 const std::string& var,
57 const std::shared_ptr<SymbolicExpr>& a,
58 const std::shared_ptr<SymbolicExpr>& b)
59{
60 ComputationContext context;
61 double a_val = 0.0;
62 double b_val = 0.0;
63 if (!vector_calculus_checked_finite_numeric(a, a_val, &context) ||
64 !vector_calculus_checked_finite_numeric(b, b_val, &context)) {
65 return nullptr;
66 }
67
68 int n = 1000;
69 double h = (b_val - a_val) / n;
70 if (!std::isfinite(h)) return nullptr;
71 double sum = 0.0;
72
73 for (int i = 0; i <= n; ++i) {
74 double xi = a_val + i * h;
75 if (!std::isfinite(xi)) return nullptr;
76 auto xi_expr = SymbolicExpr::number(xi);
77 auto fi = integrand->substitute(var, xi_expr);
78 double fi_val = 0.0;
79 if (!vector_calculus_checked_finite_numeric(fi, fi_val, &context)) {
80 return nullptr;
81 }
82
83 double weight = 1.0;
84 if (i == 0 || i == n) {
85 weight = 1.0;
86 } else if (i % 2 == 1) {
87 weight = 4.0;
88 } else {
89 weight = 2.0;
90 }
91 sum += weight * fi_val;
92 if (!std::isfinite(sum)) return nullptr;
93 }
94 sum *= h / 3.0;
95 if (!std::isfinite(sum)) return nullptr;
96
97 return SymbolicExpr::number(sum);
98}

221static Result<VectorField> vector_calculus_cross_product_partials_strict(
222 const VectorField& parametrization,
223 const std::string& u,
224 const std::string& v,
225 const std::string& operation)
226{
227 VectorField r_u;
228 VectorField r_v;
229 r_u.reserve(3);
230 r_v.reserve(3);
231 for (size_t i = 0; i < 3; ++i) {
232 auto du = vector_calculus_differentiate_strict(
233 parametrization[i], u, operation);
234 if (!du) return Result<VectorField>::failure(du.error());
235 auto dv = vector_calculus_differentiate_strict(
236 parametrization[i], v, operation);
237 if (!dv) return Result<VectorField>::failure(dv.error());
238 r_u.push_back(std::move(du.value()));
239 r_v.push_back(std::move(dv.value()));
240 }
241
242 auto cross_x = SymbolicExpr::add(
243 SymbolicExpr::multiply(r_u[1], r_v[2]),
244 SymbolicExpr::multiply(SymbolicExpr::number(-1),
245 SymbolicExpr::multiply(r_u[2], r_v[1])));
246 auto cross_x_checked = vector_calculus_simplify_strict(
247 cross_x, operation, "surface normal construction is outside the supported domain");
248 if (!cross_x_checked) return Result<VectorField>::failure(cross_x_checked.error());
249
250 auto cross_y = SymbolicExpr::add(
251 SymbolicExpr::multiply(r_u[2], r_v[0]),
252 SymbolicExpr::multiply(SymbolicExpr::number(-1),
253 SymbolicExpr::multiply(r_u[0], r_v[2])));
254 auto cross_y_checked = vector_calculus_simplify_strict(
255 cross_y, operation, "surface normal construction is outside the supported domain");
256 if (!cross_y_checked) return Result<VectorField>::failure(cross_y_checked.error());
257
258 auto cross_z = SymbolicExpr::add(
259 SymbolicExpr::multiply(r_u[0], r_v[1]),
260 SymbolicExpr::multiply(SymbolicExpr::number(-1),
261 SymbolicExpr::multiply(r_u[1], r_v[0])));
262 auto cross_z_checked = vector_calculus_simplify_strict(
263 cross_z, operation, "surface normal construction is outside the supported domain");
264 if (!cross_z_checked) return Result<VectorField>::failure(cross_z_checked.error());
265
266 return Result<VectorField>::success(
267 VectorField{std::move(cross_x_checked.value()),
268 std::move(cross_y_checked.value()),
269 std::move(cross_z_checked.value())});
270}

583 parametrization, u, v, operation);
584 if (!cross) return VectorCalculusExprResult::failure(cross.error());

588 for (size_t i = 0; i < 3; ++i) {
589 auto curl_i_composed = vector_calculus_substitute_coords_strict(
590 curl_F.value()[i], coord_vars, parametrization, operation);
591 if (!curl_i_composed) return curl_i_composed;

593 auto term = SymbolicExpr::multiply(curl_i_composed.value(), cross.value()[i]);
594 auto term_checked = vector_calculus_simplify_strict(
595 term, operation, "Stokes theorem integrand is outside the supported domain");
596 if (!term_checked) return term_checked;
597
598 dot_product = dot_product ? SymbolicExpr::add(dot_product, term_checked.value())

600 auto dot_checked = vector_calculus_simplify_strict(
601 dot_product, operation, "Stokes theorem integrand is outside the supported domain");
602 if (!dot_checked) return dot_checked;
603 dot_product = std::move(dot_checked.value());
604 }
605
606 std::vector<IntegrationStep> steps;
607 steps.push_back({v, v_bounds.first, v_bounds.second});
608 steps.push_back({u, u_bounds.first, u_bounds.second});
609 return vector_calculus_multiple_integral_strict(dot_product, steps, context, operation);
610}

661 throw std::invalid_argument("curve_integral_scalar: bounds must not be null");
662 }

665 auto coord_vars = vector_calculus_coord_vars(dim);
666

668 auto f_composed = f;
669 for (size_t i = 0; i < dim; ++i) {
670 f_composed = f_composed->substitute(coord_vars[i], parametrization[i]);
671 }
672 f_composed = f_composed->simplify();
673
674 /// 计算 |r'(t)| = √(∑(r_i'(t))²)
675 std::shared_ptr<SymbolicExpr> speed_sq = nullptr;
676 for (size_t i = 0; i < dim; ++i) {
677 auto deriv = parametrization[i]->differentiate(t);

681 sq = sq->simplify();

683 speed_sq = sq;
684 } else {
685 speed_sq = SymbolicExpr::add(speed_sq, sq);
686 speed_sq = speed_sq->simplify();
687 }
688 }
689
690 if (!speed_sq) {
691 return SymbolicExpr::number(0);
692 }
693
694 auto speed = SymbolicExpr::sqrt(speed_sq);
695 speed = speed->simplify();
696
697 /// 被积函数: f(r(t)) · |r'(t)|
698 auto integrand = SymbolicExpr::multiply(f_composed, speed);
699 integrand = integrand->simplify();
700
701 /// 计算定积分 ∫ₐᵇ f(r(t))·|r'(t)| dt
702 return vector_calculus_integrate_with_fallback(integrand, t, a, b);

804/**
805 * @internal
806 * @brief 计算参数化曲面的法向量 r_u × r_v。
807 *
808 * 参数化曲面 r(u,v) = [x(u,v), y(u,v), z(u,v)]，
809 * 返回叉积 r_u × r_v 的三个分量。
810 */
811static VectorField vector_calculus_cross_product_partials(
812 const VectorField& parametrization,
813 const std::string& u, const std::string& v)
814{
815 /// r_u = [∂x/∂u, ∂y/∂u, ∂z/∂u]
816 VectorField r_u;
817 r_u.reserve(3);
818 for (size_t i = 0; i < 3; ++i) {
819 auto deriv = parametrization[i]->differentiate(u);
820 r_u.push_back(deriv ? deriv->simplify() : SymbolicExpr::number(0));
821 }
822
823 /// r_v = [∂x/∂v, ∂y/∂v, ∂z/∂v]
824 VectorField r_v;
825 r_v.reserve(3);
826 for (size_t i = 0; i < 3; ++i) {
827 auto deriv = parametrization[i]->differentiate(v);
828 r_v.push_back(deriv ? deriv->simplify() : SymbolicExpr::number(0));
829 }
830
831 /// 叉积: r_u × r_v = (r_u[1]*r_v[2] - r_u[2]*r_v[1],
832 /// r_u[2]*r_v[0] - r_u[0]*r_v[2],
833 /// r_u[0]*r_v[1] - r_u[1]*r_v[0])
834 auto cross_x = SymbolicExpr::add(
835 SymbolicExpr::multiply(r_u[1], r_v[2]),
836 SymbolicExpr::multiply(SymbolicExpr::number(-1),
837 SymbolicExpr::multiply(r_u[2], r_v[1])));
838 cross_x = cross_x->simplify();
839
840 auto cross_y = SymbolicExpr::add(
841 SymbolicExpr::multiply(r_u[2], r_v[0]),
842 SymbolicExpr::multiply(SymbolicExpr::number(-1),
843 SymbolicExpr::multiply(r_u[0], r_v[2])));
844 cross_y = cross_y->simplify();
845
846 auto cross_z = SymbolicExpr::add(
847 SymbolicExpr::multiply(r_u[0], r_v[1]),
848 SymbolicExpr::multiply(SymbolicExpr::number(-1),
849 SymbolicExpr::multiply(r_u[1], r_v[0])));
850 cross_z = cross_z->simplify();
851
852 return VectorField{cross_x, cross_y, cross_z};
853}
