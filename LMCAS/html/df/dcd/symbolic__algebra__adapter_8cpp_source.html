9ResultantMatrix sylvester_matrix(const Polynomial<Rational>& left,
10 const Polynomial<Rational>& right) {
11 const int left_degree = left.degree();
12 const int right_degree = right.degree();
13 const int size = left_degree + right_degree;
14 ResultantMatrix matrix(size, std::vector<Rational>(size, Rational(0)));
15 for (int row = 0; row < right_degree; ++row) {
16 for (int column = 0; column <= left_degree; ++column) {
17 matrix[row][row + column] = left.coeffs[left_degree - column];
18 }
19 }
20 for (int row = 0; row < left_degree; ++row) {
21 for (int column = 0; column <= right_degree; ++column) {
22 matrix[right_degree + row][row + column] =
23 right.coeffs[right_degree - column];
24 }

29void eliminate_resultant_column(ResultantMatrix& matrix, int pivot,

42Rational resultant_determinant(ResultantMatrix& matrix) {
43 Rational previous(1);
44 int sign = 1;
45 const int size = static_cast<int>(matrix.size());
46 for (int pivot = 0; pivot < size; ++pivot) {
47 if (matrix[pivot][pivot] == Rational(0)) {
48 int replacement = pivot + 1;
49 while (replacement < size &&
50 matrix[replacement][pivot] == Rational(0)) {
51 ++replacement;
52 }
53 if (replacement == size) { return Rational(0); }
54 std::swap(matrix[pivot], matrix[replacement]);
55 sign = -sign;
56 }
57 eliminate_resultant_column(matrix, pivot, previous);
58 previous = matrix[pivot][pivot];
59 }
60 auto determinant = matrix.back().back();
61 if (sign < 0) { determinant = Rational(0) - determinant; }
62 return determinant;
63}
