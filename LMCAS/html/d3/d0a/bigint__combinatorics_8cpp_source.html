8Result<unsigned int> kernel_argument(const BigInt& value, const char* operation) {
9 if (value.is_negative()) {
10 return Result<unsigned int>::failure(
11 CasErrc::DomainError, "combinatorial arguments must be nonnegative", operation);
12 }
13 const auto converted = value.try_to_uint64();
14 if (!converted || *converted > std::numeric_limits<unsigned int>::max()) {
15 return Result<unsigned int>::failure(
16 CasErrc::ResourceLimit, "argument exceeds the LMMP unsigned kernel range", operation);
17 }
18 return static_cast<unsigned int>(*converted);

21Result<void> reserve_product(const BigInt& largest_factor, unsigned int factors,
22 ComputationContext& context, const char* operation) {
23 auto steps = context.consume_steps(factors, operation);
24 if (!steps) {
25 return steps;
26 }
27 const auto factor_bits = largest_factor.bit_length();
28 if (factors != 0 && factor_bits > std::numeric_limits<std::size_t>::max() / factors) {
29 return Result<void>::failure(CasErrc::ResourceLimit,
30 "combinatorial bit bound overflows", operation);
31 }
32 return context.require_integer_bits(factor_bits * factors, operation);
33}

35Result<BigInt> finish_combinatorial(BigInt value, ComputationContext& context,

63Result<BigInt> BigInt::factorial_checked(const BigInt& n,
64 ComputationContext& context) try {
65 constexpr auto operation = "BigInt::factorial_checked";
66 auto access = context.consume_steps(1, operation);
67 if (!access) {
68 return Result<BigInt>::failure(access.error());
69 }
70 auto argument = kernel_argument(n, operation);
71 if (!argument) {
72 return Result<BigInt>::failure(argument.error());
73 }
74 if (argument.value() <= 1) {
75 return finish_combinatorial(BigInt(1), context, operation);
76 }
77 auto budget = reserve_product(n, argument.value(), context, operation);
78 if (!budget) {
79 return Result<BigInt>::failure(budget.error());
80 }
81 detail::ensure_lmmc_lifecycle();
82 mp_bitcnt_t binary_shift = 0;
83 const auto limbs = lmmp_factorial_size_(argument.value(), &binary_shift);
84 BigInt result;
85 result.realloc_to(limbs);
86 result.size_ = lmmp_factorial_(result.data_, binary_shift, limbs, argument.value());
87 result.sign_ = POSITIVE;
88 result.normalize();
89 return finish_combinatorial(std::move(result), context, operation);
90} catch (const std::bad_alloc&) {
91 return Result<BigInt>::failure(CasErrc::ResourceLimit,
92 "factorial allocation failed", "BigInt::factorial_checked");
93} catch (const std::length_error& error) {
94 return Result<BigInt>::failure(CasErrc::ResourceLimit, error.what(), "BigInt::factorial_checked");
