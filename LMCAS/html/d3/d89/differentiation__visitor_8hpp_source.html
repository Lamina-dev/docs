67 void visit(const MultiplyNode& node) override {
68 if (node.operands().empty()) {
69 result = SymbolicFactory::create_number(BigInt(0));
70 return;
71 }
72
73 std::vector<std::shared_ptr<const SymbolicNode>> sum_terms;
74
75 for (size_t i = 0; i < node.operands().size(); ++i) {
76
77 node.operands()[i]->accept(*this);
78 auto d_term = result;
79
80 if (d_term->is_zero()) continue;
81
82 std::vector<std::shared_ptr<const SymbolicNode>> prod_terms;
83
84 for (size_t j = 0; j < node.operands().size(); ++j) {
85 if (i == j) {
86 prod_terms.push_back(d_term);
87 } else {
88 prod_terms.push_back(node.operands()[j]);
89 }
90 }
91 sum_terms.push_back(SymbolicFactory::create_multiply(prod_terms));
92 }
93
94 if (sum_terms.empty()) {
95 result = SymbolicFactory::create_number(BigInt(0));
96 } else {
97 result = SymbolicFactory::create_add(sum_terms);
98 }
99 }

101 void visit(const PowerNode& node) override {
102
103 node.base()->accept(*this);
104 auto du = result;
105 node.exponent()->accept(*this);
106 auto dv = result;
107
108 auto u = node.base();
109 auto v = node.exponent();
110
111 if (dv->is_zero()) {
112
113 auto n = v;
114
115 auto n_minus_1 = SymbolicFactory::create_add({n, SymbolicFactory::create_number(BigInt(-1))});
116
117 auto u_pow = lamina::detail::make_node<PowerNode>(u, n_minus_1);
118
119 result = SymbolicFactory::create_multiply({n, u_pow, du});
120 } else {
121
122 auto ln_u = lamina::detail::make_node<FunctionNode>(FunctionNode::FuncType::Ln, std::vector<std::shared_ptr<const SymbolicNode>>{u});
123
124 auto t1 = SymbolicFactory::create_multiply({dv, ln_u});
125
126 auto u_inv = lamina::detail::make_node<PowerNode>(u, SymbolicFactory::create_number(BigInt(-1)));
127 auto t2 = SymbolicFactory::create_multiply({v, du, u_inv});
128
129 auto sum = SymbolicFactory::create_add({t1, t2});
130
131 auto u_pow_v = lamina::detail::make_node<PowerNode>(u, v);
132 result = SymbolicFactory::create_multiply({u_pow_v, sum});
133 }
134 }

136 void visit(const FunctionNode& node) override {
137 if (node.arguments().size() != 1) {
138 result = SymbolicFactory::create_number(BigInt(0));
139 return;
140 }
141
142 auto& arg = node.arguments()[0];
143 arg->accept(*this);
144 auto d_arg = result;
145
146 if (d_arg->is_zero()) {
147 result = SymbolicFactory::create_number(BigInt(0));
148 return;
149 }
150
151 std::shared_ptr<const SymbolicNode> d_outer;
152
153 switch (node.type()) {
154 case FunctionNode::FuncType::Sin:
155 d_outer = lamina::detail::make_node<FunctionNode>(FunctionNode::FuncType::Cos, node.arguments());
156 break;
157 case FunctionNode::FuncType::Cos:
158 d_outer = SymbolicFactory::create_multiply({
159 SymbolicFactory::create_number(BigInt(-1)),
160 lamina::detail::make_node<FunctionNode>(FunctionNode::FuncType::Sin, node.arguments())
161 });
162 break;
163 case FunctionNode::FuncType::Tan:
164 {
165 auto sec = lamina::detail::make_node<FunctionNode>(FunctionNode::FuncType::Sec, node.arguments());
166 d_outer = lamina::detail::make_node<PowerNode>(sec, SymbolicFactory::create_number(BigInt(2)));
167 }
168 break;
169 case FunctionNode::FuncType::Cot:
170 {
171 // d/dx cot(x) = -csc(x)^2
172 auto csc = lamina::detail::make_node<FunctionNode>(FunctionNode::FuncType::Csc, node.arguments());
173 auto csc_sq = lamina::detail::make_node<PowerNode>(csc, SymbolicFactory::create_number(BigInt(2)));
174 d_outer = SymbolicFactory::create_multiply({
175 SymbolicFactory::create_number(BigInt(-1)),
176 csc_sq
177 });
178 }
179 break;
180 case FunctionNode::FuncType::Sec:
181 {
182 // d/dx sec(x) = sec(x) * tan(x)
183 auto sec = lamina::detail::make_node<FunctionNode>(FunctionNode::FuncType::Sec, node.arguments());
184 auto tan = lamina::detail::make_node<FunctionNode>(FunctionNode::FuncType::Tan, node.arguments());
185 d_outer = SymbolicFactory::create_multiply({sec, tan});
186 }
187 break;
188 case FunctionNode::FuncType::Csc:
189 {
190 // d/dx csc(x) = -csc(x) * cot(x)
191 auto csc = lamina::detail::make_node<FunctionNode>(FunctionNode::FuncType::Csc, node.arguments());
192 auto cot = lamina::detail::make_node<FunctionNode>(FunctionNode::FuncType::Cot, node.arguments());
193 d_outer = SymbolicFactory::create_multiply({
194 SymbolicFactory::create_number(BigInt(-1)),
195 csc, cot
196 });
197 }
198 break;
199 case FunctionNode::FuncType::ArcSin:
200 {
201 // d/dx arcsin(x) = 1/sqrt(1 - x^2) = (1 - x^2)^(-1/2)
202 auto arg_sq = lamina::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(BigInt(2)));
203 auto neg_arg_sq = SymbolicFactory::create_multiply({
204 SymbolicFactory::create_number(BigInt(-1)), arg_sq
205 });
206 auto one_minus_sq = SymbolicFactory::create_add({
207 SymbolicFactory::create_number(BigInt(1)), neg_arg_sq
208 });
209 d_outer = lamina::detail::make_node<PowerNode>(
210 one_minus_sq, SymbolicFactory::create_number(Rational(-1, 2)));
211 }
212 break;
213 case FunctionNode::FuncType::ArcCos:
214 {
215 // d/dx arccos(x) = -1/sqrt(1 - x^2) = -(1 - x^2)^(-1/2)
216 auto arg_sq = lamina::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(BigInt(2)));
217 auto neg_arg_sq = SymbolicFactory::create_multiply({
218 SymbolicFactory::create_number(BigInt(-1)), arg_sq
219 });
220 auto one_minus_sq = SymbolicFactory::create_add({
221 SymbolicFactory::create_number(BigInt(1)), neg_arg_sq
222 });
223 auto inv_sqrt = lamina::detail::make_node<PowerNode>(
224 one_minus_sq, SymbolicFactory::create_number(Rational(-1, 2)));
225 d_outer = SymbolicFactory::create_multiply({
226 SymbolicFactory::create_number(BigInt(-1)), inv_sqrt
227 });
228 }
229 break;
230 case FunctionNode::FuncType::ArcTan:
231 {
232 // d/dx arctan(x) = 1/(1 + x^2) = (1 + x^2)^(-1)
233 auto arg_sq = lamina::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(BigInt(2)));
234 auto one_plus_sq = SymbolicFactory::create_add({
235 SymbolicFactory::create_number(BigInt(1)), arg_sq
236 });
237 d_outer = lamina::detail::make_node<PowerNode>(
238 one_plus_sq, SymbolicFactory::create_number(BigInt(-1)));
239 }
240 break;
241 case FunctionNode::FuncType::Sinh:
242 d_outer = lamina::detail::make_node<FunctionNode>(
243 FunctionNode::FuncType::Cosh, node.arguments());
244 break;
245 case FunctionNode::FuncType::Cosh:
246 d_outer = lamina::detail::make_node<FunctionNode>(
247 FunctionNode::FuncType::Sinh, node.arguments());
248 break;
249 case FunctionNode::FuncType::Tanh:
250 {
251 // d/dx tanh(x) = sech(x)^2 = 1/cosh(x)^2 = cosh(x)^(-2)
252 auto cosh = lamina::detail::make_node<FunctionNode>(
253 FunctionNode::FuncType::Cosh, node.arguments());
254 d_outer = lamina::detail::make_node<PowerNode>(
255 cosh, SymbolicFactory::create_number(BigInt(-2)));
256 }
257 break;
258 case FunctionNode::FuncType::Exp:
259 d_outer = lamina::detail::make_node<FunctionNode>(FunctionNode::FuncType::Exp, node.arguments());
260 break;
261 case FunctionNode::FuncType::Ln:
262 d_outer = lamina::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(BigInt(-1)));
263 break;
264 case FunctionNode::FuncType::Sqrt:
265 d_outer = SymbolicFactory::create_multiply({
266 SymbolicFactory::create_number(Rational(1, 2)),
267 lamina::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(Rational(-1, 2)))
268 });
269 break;
270 case FunctionNode::FuncType::Abs:
271 {
272 /// d/dx |u| = u / |u| = sgn(u) (undefined at u=0)
273 auto abs_arg = lamina::detail::make_node<FunctionNode>(
274 FunctionNode::FuncType::Abs, node.arguments());
275 auto abs_inv = lamina::detail::make_node<PowerNode>(abs_arg, SymbolicFactory::create_number(BigInt(-1)));
276 d_outer = SymbolicFactory::create_multiply({arg, abs_inv});
277 }
278 break;
279 case FunctionNode::FuncType::LambertW:
280 {
281 auto W = lamina::detail::make_node<FunctionNode>(FunctionNode::FuncType::LambertW, node.arguments());
282 auto one = SymbolicFactory::create_number(BigInt(1));
283 auto one_plus_W = SymbolicFactory::create_add({one, W});
284 auto denom = SymbolicFactory::create_multiply({arg, one_plus_W});
285 auto denom_inv = lamina::detail::make_node<PowerNode>(denom, SymbolicFactory::create_number(BigInt(-1)));
286 d_outer = SymbolicFactory::create_multiply({W, denom_inv});
287 }
288 break;
289 case FunctionNode::FuncType::RootOf:
290 throw std::runtime_error("RootOf differentiation is not mathematically supported yet.");
291 case FunctionNode::FuncType::Erf:
292 {
293 // d/dx erf(x) = (2/sqrt(pi)) * exp(-x^2)
294 auto pi = lamina::detail::make_node<VariableNode>("pi");
295 auto sqrt_pi = lamina::detail::make_node<FunctionNode>(
296 FunctionNode::FuncType::Sqrt,
297 std::vector<std::shared_ptr<const SymbolicNode>>{pi});
298 auto sqrt_pi_inv = lamina::detail::make_node<PowerNode>(sqrt_pi, SymbolicFactory::create_number(BigInt(-1)));
299 auto two = SymbolicFactory::create_number(BigInt(2));
300 auto neg_one = SymbolicFactory::create_number(BigInt(-1));
301 auto arg_sq = lamina::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(BigInt(2)));
302 auto neg_arg_sq = SymbolicFactory::create_multiply({neg_one, arg_sq});
303 auto exp_term = lamina::detail::make_node<FunctionNode>(
304 FunctionNode::FuncType::Exp,
305 std::vector<std::shared_ptr<const SymbolicNode>>{neg_arg_sq});
306 d_outer = SymbolicFactory::create_multiply({two, sqrt_pi_inv, exp_term});
307 }
308 break;
309 case FunctionNode::FuncType::Ei:
310 {
311 // d/dx Ei(x) = exp(x) / x
312 auto exp_arg = lamina::detail::make_node<FunctionNode>(
313 FunctionNode::FuncType::Exp, node.arguments());
314 auto arg_inv = lamina::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(BigInt(-1)));
315 d_outer = SymbolicFactory::create_multiply({exp_arg, arg_inv});
316 }
317 break;
318 case FunctionNode::FuncType::Si:
319 {
320 // d/dx Si(x) = sin(x) / x
321 auto sin_arg = lamina::detail::make_node<FunctionNode>(
322 FunctionNode::FuncType::Sin, node.arguments());
323 auto arg_inv = lamina::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(BigInt(-1)));
324 d_outer = SymbolicFactory::create_multiply({sin_arg, arg_inv});
325 }
326 break;
327 case FunctionNode::FuncType::Ci:
328 {
329 // d/dx Ci(x) = cos(x) / x
330 auto cos_arg = lamina::detail::make_node<FunctionNode>(
331 FunctionNode::FuncType::Cos, node.arguments());
332 auto arg_inv = lamina::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(BigInt(-1)));
333 d_outer = SymbolicFactory::create_multiply({cos_arg, arg_inv});
334 }
335 break;
336 case FunctionNode::FuncType::Li:
337 {
338 // d/dx Li(x) = 1 / ln(x)
339 auto ln_arg = lamina::detail::make_node<FunctionNode>(
340 FunctionNode::FuncType::Ln, node.arguments());
341 d_outer = lamina::detail::make_node<PowerNode>(ln_arg, SymbolicFactory::create_number(BigInt(-1)));
342 }
343 break;
344 default:
345 d_outer = SymbolicFactory::create_number(BigInt(0));
346 }
347
348 result = SymbolicFactory::create_multiply(std::vector<std::shared_ptr<const SymbolicNode>>{d_outer, d_arg});
349 }

351 void visit(const MatrixNode& node) override {
352
353 if (std::holds_alternative<MatrixNode::DenseStorage>(node.storage())) {
354 const auto& dense = std::get<MatrixNode::DenseStorage>(node.storage());
355 MatrixNode::DenseStorage new_dense;
356 for(const auto& item : dense) {
357 if (item) {
358 item->accept(*this);
359 new_dense.push_back(result);
360 } else {
361 new_dense.push_back(nullptr);
362 }
363 }
364 result = lamina::detail::make_node<MatrixNode>(node.rows(), node.cols(), new_dense);
365 } else {
366 const auto& sparse = std::get<MatrixNode::SparseStorage>(node.storage());
367 MatrixNode::SparseStorage new_sparse;
368 for(const auto& [idx, item] : sparse) {
369 item->accept(*this);
370 if (!result->is_zero()) {
371 new_sparse[idx] = result;
372 }
373 }
374 result = lamina::detail::make_node<MatrixNode>(node.rows(), node.cols(), new_sparse);
375 }
376 }
