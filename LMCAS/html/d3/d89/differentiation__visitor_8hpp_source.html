60 void visit(const AddNode& node) override {
61 std::vector<std::shared_ptr<const SymbolicNode>> diff_ops;
62 for (const auto& op : node.operands()) {
63 op->accept(*this);
64 diff_ops.push_back(result);
65 }

69 void visit(const MultiplyNode& node) override {
70 if (node.operands().empty()) {
71 result = SymbolicFactory::create_number(BigInt(0));
72 return;
73 }
74
75 std::vector<std::shared_ptr<const SymbolicNode>> sum_terms;
76
77 for (size_t i = 0; i < node.operands().size(); ++i) {
78
79 node.operands()[i]->accept(*this);
80 auto d_term = result;
81
82 if (d_term->is_zero()) continue;
83
84 std::vector<std::shared_ptr<const SymbolicNode>> prod_terms;
85
86 for (size_t j = 0; j < node.operands().size(); ++j) {
87 if (i == j) {
88 prod_terms.push_back(d_term);
89 } else {
90 prod_terms.push_back(node.operands()[j]);
91 }
92 }
93 sum_terms.push_back(SymbolicFactory::create_multiply(prod_terms));
94 }
95
96 if (sum_terms.empty()) {
97 result = SymbolicFactory::create_number(BigInt(0));
98 } else {
99 result = SymbolicFactory::create_add(sum_terms);
100 }
101 }

103 void visit(const PowerNode& node) override {
104
105 node.base()->accept(*this);
106 auto du = result;
107 node.exponent()->accept(*this);
108 auto dv = result;
109
110 auto u = node.base();
111 auto v = node.exponent();
112
113 if (dv->is_zero()) {
114
115 auto n = v;
116
117 auto n_minus_1 = SymbolicFactory::create_add({n, SymbolicFactory::create_number(BigInt(-1))});
118
119 auto u_pow = LMCAS::detail::make_node<PowerNode>(u, n_minus_1);
120
121 result = SymbolicFactory::create_multiply({n, u_pow, du});
122 } else {
123
124 auto ln_u = LMCAS::detail::make_node<FunctionNode>(FunctionNode::FuncType::Ln, std::vector<std::shared_ptr<const SymbolicNode>>{u});
125
126 auto t1 = SymbolicFactory::create_multiply({dv, ln_u});
127
128 auto u_inv = LMCAS::detail::make_node<PowerNode>(u, SymbolicFactory::create_number(BigInt(-1)));
129 auto t2 = SymbolicFactory::create_multiply({v, du, u_inv});
130
131 auto sum = SymbolicFactory::create_add({t1, t2});
132
133 auto u_pow_v = LMCAS::detail::make_node<PowerNode>(u, v);
134 result = SymbolicFactory::create_multiply({u_pow_v, sum});
135 }
136 }

138 void visit(const FunctionNode& node) override {
139 if (node.arguments().size() != 1) {
140 result = SymbolicFactory::create_number(BigInt(0));
141 return;
142 }
143
144 auto& arg = node.arguments()[0];
145 arg->accept(*this);
146 auto d_arg = result;
147
148 if (d_arg->is_zero()) {
149 result = SymbolicFactory::create_number(BigInt(0));
150 return;
151 }
152
153 std::shared_ptr<const SymbolicNode> d_outer;
154
155 switch (node.type()) {
156 case FunctionNode::FuncType::Sin:
157 d_outer = LMCAS::detail::make_node<FunctionNode>(FunctionNode::FuncType::Cos, node.arguments());
158 break;
159 case FunctionNode::FuncType::Cos:
160 d_outer = SymbolicFactory::create_multiply({
161 SymbolicFactory::create_number(BigInt(-1)),
162 LMCAS::detail::make_node<FunctionNode>(FunctionNode::FuncType::Sin, node.arguments())
163 });
164 break;
165 case FunctionNode::FuncType::Tan:
166 {
167 auto sec = LMCAS::detail::make_node<FunctionNode>(FunctionNode::FuncType::Sec, node.arguments());
168 d_outer = LMCAS::detail::make_node<PowerNode>(sec, SymbolicFactory::create_number(BigInt(2)));
169 }
170 break;
171 case FunctionNode::FuncType::Cot:
172 {
173 // d/dx cot(x) = -csc(x)^2
174 auto csc = LMCAS::detail::make_node<FunctionNode>(FunctionNode::FuncType::Csc, node.arguments());
175 auto csc_sq = LMCAS::detail::make_node<PowerNode>(csc, SymbolicFactory::create_number(BigInt(2)));
176 d_outer = SymbolicFactory::create_multiply({
177 SymbolicFactory::create_number(BigInt(-1)),
178 csc_sq
179 });
180 }
181 break;
182 case FunctionNode::FuncType::Sec:
183 {
184 // d/dx sec(x) = sec(x) * tan(x)
185 auto sec = LMCAS::detail::make_node<FunctionNode>(FunctionNode::FuncType::Sec, node.arguments());
186 auto tan = LMCAS::detail::make_node<FunctionNode>(FunctionNode::FuncType::Tan, node.arguments());
187 d_outer = SymbolicFactory::create_multiply({sec, tan});
188 }
189 break;
190 case FunctionNode::FuncType::Csc:
191 {
192 // d/dx csc(x) = -csc(x) * cot(x)
193 auto csc = LMCAS::detail::make_node<FunctionNode>(FunctionNode::FuncType::Csc, node.arguments());
194 auto cot = LMCAS::detail::make_node<FunctionNode>(FunctionNode::FuncType::Cot, node.arguments());
195 d_outer = SymbolicFactory::create_multiply({
196 SymbolicFactory::create_number(BigInt(-1)),
197 csc, cot
198 });
199 }
200 break;
201 case FunctionNode::FuncType::ArcSin:
202 {
203 // d/dx arcsin(x) = 1/sqrt(1 - x^2) = (1 - x^2)^(-1/2)
204 auto arg_sq = LMCAS::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(BigInt(2)));
205 auto neg_arg_sq = SymbolicFactory::create_multiply({
206 SymbolicFactory::create_number(BigInt(-1)), arg_sq
207 });
208 auto one_minus_sq = SymbolicFactory::create_add({
209 SymbolicFactory::create_number(BigInt(1)), neg_arg_sq
210 });
211 d_outer = LMCAS::detail::make_node<PowerNode>(
212 one_minus_sq, SymbolicFactory::create_number(Rational(-1, 2)));
213 }
214 break;
215 case FunctionNode::FuncType::ArcCos:
216 {
217 // d/dx arccos(x) = -1/sqrt(1 - x^2) = -(1 - x^2)^(-1/2)
218 auto arg_sq = LMCAS::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(BigInt(2)));
219 auto neg_arg_sq = SymbolicFactory::create_multiply({
220 SymbolicFactory::create_number(BigInt(-1)), arg_sq
221 });
222 auto one_minus_sq = SymbolicFactory::create_add({
223 SymbolicFactory::create_number(BigInt(1)), neg_arg_sq
224 });
225 auto inv_sqrt = LMCAS::detail::make_node<PowerNode>(
226 one_minus_sq, SymbolicFactory::create_number(Rational(-1, 2)));
227 d_outer = SymbolicFactory::create_multiply({
228 SymbolicFactory::create_number(BigInt(-1)), inv_sqrt
229 });
230 }
231 break;
232 case FunctionNode::FuncType::ArcTan:
233 {
234 // d/dx arctan(x) = 1/(1 + x^2) = (1 + x^2)^(-1)
235 auto arg_sq = LMCAS::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(BigInt(2)));
236 auto one_plus_sq = SymbolicFactory::create_add({
237 SymbolicFactory::create_number(BigInt(1)), arg_sq
238 });
239 d_outer = LMCAS::detail::make_node<PowerNode>(
240 one_plus_sq, SymbolicFactory::create_number(BigInt(-1)));
241 }
242 break;
243 case FunctionNode::FuncType::Sinh:
244 d_outer = LMCAS::detail::make_node<FunctionNode>(
245 FunctionNode::FuncType::Cosh, node.arguments());
246 break;
247 case FunctionNode::FuncType::Cosh:
248 d_outer = LMCAS::detail::make_node<FunctionNode>(
249 FunctionNode::FuncType::Sinh, node.arguments());
250 break;
251 case FunctionNode::FuncType::Tanh:
252 {
253 // d/dx tanh(x) = sech(x)^2 = 1/cosh(x)^2 = cosh(x)^(-2)
254 auto cosh = LMCAS::detail::make_node<FunctionNode>(
255 FunctionNode::FuncType::Cosh, node.arguments());
256 d_outer = LMCAS::detail::make_node<PowerNode>(
257 cosh, SymbolicFactory::create_number(BigInt(-2)));
258 }
259 break;
260 case FunctionNode::FuncType::Exp:
261 d_outer = LMCAS::detail::make_node<FunctionNode>(FunctionNode::FuncType::Exp, node.arguments());
262 break;
263 case FunctionNode::FuncType::Ln:
264 d_outer = LMCAS::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(BigInt(-1)));
265 break;
266 case FunctionNode::FuncType::Sqrt:
267 d_outer = SymbolicFactory::create_multiply({
268 SymbolicFactory::create_number(Rational(1, 2)),
269 LMCAS::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(Rational(-1, 2)))
270 });
271 break;
272 case FunctionNode::FuncType::Abs:
273 {
274 /// d/dx |u| = u / |u| = sgn(u) (undefined at u=0)
275 auto abs_arg = LMCAS::detail::make_node<FunctionNode>(
276 FunctionNode::FuncType::Abs, node.arguments());
277 auto abs_inv = LMCAS::detail::make_node<PowerNode>(abs_arg, SymbolicFactory::create_number(BigInt(-1)));
278 d_outer = SymbolicFactory::create_multiply({arg, abs_inv});
279 }
280 break;
281 case FunctionNode::FuncType::LambertW:
282 {
283 auto W = LMCAS::detail::make_node<FunctionNode>(FunctionNode::FuncType::LambertW, node.arguments());
284 auto one = SymbolicFactory::create_number(BigInt(1));
285 auto one_plus_W = SymbolicFactory::create_add({one, W});
286 auto denom = SymbolicFactory::create_multiply({arg, one_plus_W});
287 auto denom_inv = LMCAS::detail::make_node<PowerNode>(denom, SymbolicFactory::create_number(BigInt(-1)));
288 d_outer = SymbolicFactory::create_multiply({W, denom_inv});
289 }
290 break;
291 case FunctionNode::FuncType::Erf:
292 {
293 // d/dx erf(x) = (2/sqrt(pi)) * exp(-x^2)
294 auto pi = LMCAS::detail::make_node<VariableNode>("pi");
295 auto sqrt_pi = LMCAS::detail::make_node<FunctionNode>(
296 FunctionNode::FuncType::Sqrt,
297 std::vector<std::shared_ptr<const SymbolicNode>>{pi});
298 auto sqrt_pi_inv = LMCAS::detail::make_node<PowerNode>(sqrt_pi, SymbolicFactory::create_number(BigInt(-1)));
299 auto two = SymbolicFactory::create_number(BigInt(2));
300 auto neg_one = SymbolicFactory::create_number(BigInt(-1));
301 auto arg_sq = LMCAS::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(BigInt(2)));
302 auto neg_arg_sq = SymbolicFactory::create_multiply({neg_one, arg_sq});
303 auto exp_term = LMCAS::detail::make_node<FunctionNode>(
304 FunctionNode::FuncType::Exp,
305 std::vector<std::shared_ptr<const SymbolicNode>>{neg_arg_sq});
306 d_outer = SymbolicFactory::create_multiply({two, sqrt_pi_inv, exp_term});
307 }
308 break;
309 case FunctionNode::FuncType::Ei:
310 {
311 // d/dx Ei(x) = exp(x) / x
312 auto exp_arg = LMCAS::detail::make_node<FunctionNode>(
313 FunctionNode::FuncType::Exp, node.arguments());
314 auto arg_inv = LMCAS::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(BigInt(-1)));
315 d_outer = SymbolicFactory::create_multiply({exp_arg, arg_inv});
316 }
317 break;
318 case FunctionNode::FuncType::Si:
319 {
320 // d/dx Si(x) = sin(x) / x
321 auto sin_arg = LMCAS::detail::make_node<FunctionNode>(
322 FunctionNode::FuncType::Sin, node.arguments());
323 auto arg_inv = LMCAS::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(BigInt(-1)));
324 d_outer = SymbolicFactory::create_multiply({sin_arg, arg_inv});
325 }
326 break;
327 case FunctionNode::FuncType::Ci:
328 {
329 // d/dx Ci(x) = cos(x) / x
330 auto cos_arg = LMCAS::detail::make_node<FunctionNode>(
331 FunctionNode::FuncType::Cos, node.arguments());
332 auto arg_inv = LMCAS::detail::make_node<PowerNode>(arg, SymbolicFactory::create_number(BigInt(-1)));
333 d_outer = SymbolicFactory::create_multiply({cos_arg, arg_inv});
334 }
335 break;
336 case FunctionNode::FuncType::Li:
337 {
338 // d/dx Li(x) = 1 / ln(x)
339 auto ln_arg = LMCAS::detail::make_node<FunctionNode>(
340 FunctionNode::FuncType::Ln, node.arguments());
341 d_outer = LMCAS::detail::make_node<PowerNode>(ln_arg, SymbolicFactory::create_number(BigInt(-1)));
342 }
343 break;
344 default:
345 d_outer = SymbolicFactory::create_number(BigInt(0));
346 }
347
348 result = SymbolicFactory::create_multiply(std::vector<std::shared_ptr<const SymbolicNode>>{d_outer, d_arg});
349 }

351 void visit(const MatrixNode& node) override {
352
353 if (std::holds_alternative<MatrixNode::DenseStorage>(node.storage())) {
354 const auto& dense = std::get<MatrixNode::DenseStorage>(node.storage());
355 MatrixNode::DenseStorage new_dense;
356 for(const auto& item : dense) {
357 if (item) {
358 item->accept(*this);
359 new_dense.push_back(result);
360 } else {
361 new_dense.push_back(nullptr);
362 }
363 }
364 result = LMCAS::detail::make_node<MatrixNode>(node.rows(), node.cols(), new_dense);
365 } else {
366 const auto& sparse = std::get<MatrixNode::SparseStorage>(node.storage());
367 MatrixNode::SparseStorage new_sparse;
368 for(const auto& [idx, item] : sparse) {
369 item->accept(*this);
370 if (!result->is_zero()) {
371 new_sparse[idx] = result;
372 }
373 }
374 result = LMCAS::detail::make_node<MatrixNode>(node.rows(), node.cols(), new_sparse);
375 }
376 }

443 void visit(const IntegralNode& node) override {
444 if (!node.is_definite()) {
445 if (node.variable() == var) {
446 result = node.body()->clone();
447 return;
448 }
449 node.body()->accept(*this);
450 result = LMCAS::detail::make_node<IntegralNode>(
451 result, node.variable());
452 return;
453 }
454
455 std::vector<std::shared_ptr<const SymbolicNode>> terms;
456 if (node.variable() != var) {
457 node.body()->accept(*this);
458 if (!result->is_zero()) {
459 terms.push_back(LMCAS::detail::make_node<IntegralNode>(
460 result, node.variable(), node.lower(), node.upper()));
461 }
462 }
463
464 auto body_expression = LMCAS::detail::make_expression_ptr(
465 LMCAS::detail::expression_from_node(node.body()));
466 auto upper_expression = LMCAS::detail::make_expression_ptr(
467 LMCAS::detail::expression_from_node(node.upper()));
468 auto lower_expression = LMCAS::detail::make_expression_ptr(
469 LMCAS::detail::expression_from_node(node.lower()));
470
471 node.upper()->accept(*this);
472 auto upper_derivative = result;
473 if (!upper_derivative->is_zero()) {
474 auto at_upper = body_expression->substitute(
475 node.variable(), upper_expression);
476 terms.push_back(SymbolicFactory::create_multiply(
477 {LMCAS::detail::node(at_upper), upper_derivative}));
478 }
479
480 node.lower()->accept(*this);
481 auto lower_derivative = result;
482 if (!lower_derivative->is_zero()) {
483 auto at_lower = body_expression->substitute(
484 node.variable(), lower_expression);
485 terms.push_back(SymbolicFactory::create_multiply({
486 SymbolicFactory::create_number(BigInt(-1)),
487 LMCAS::detail::node(at_lower),
488 lower_derivative}));
489 }
490 result = SymbolicFactory::create_add(std::move(terms));
491 }
