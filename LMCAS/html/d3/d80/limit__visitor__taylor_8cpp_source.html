19std::string substitute_infinity_series(
20 const std::shared_ptr<const SymbolicNode>& num,
21 const std::shared_ptr<const SymbolicNode>& den, const std::string& var,
22 std::shared_ptr<SymbolicExpr>& num_expr, std::shared_ptr<SymbolicExpr>& den_expr) {
23 /// x -> infinity: 代换 x = 1/t,在 t -> 0 处展开
24 std::string t_var = "__lim_t__";
25 auto names = detail::all_variable_names(num);
26 auto denominator_names = detail::all_variable_names(den);
27 names.insert(denominator_names.begin(), denominator_names.end());
28 while (names.count(t_var)) { t_var += "_"; }
29 auto t_expr = SymbolicExpr::variable(t_var);
30 auto one_over_t = SymbolicExpr::power(t_expr, SymbolicExpr::number(-1));
31
32 num_expr = num_expr->substitute(var, one_over_t);
33 den_expr = den_expr->substitute(var, one_over_t);
34
35 if (num_expr) { num_expr = num_expr->simplify(); }
36 if (den_expr) { den_expr = den_expr->simplify(); }
37
38 return t_var;
39}

42std::shared_ptr<const SymbolicNode> LimitVisitor::divergent_leading_ratio(
43 const std::shared_ptr<SymbolicExpr>& ratio, int difference) {
44 auto sign = ratio ? get_node_sign(LMCAS::detail::node(ratio)) : std::nullopt;
45 if (!sign || !*sign) { return nullptr; }
46 if (difference % 2 && direction.empty() && !is_inf(point)) { return nullptr; }
47 if (difference % 2 && direction == "-" && !is_inf(point)) { *sign = -*sign; }
48 return signed_infinity(*sign);
49}

53/**
54 * @brief 对 L'Hôpital 法则产生的导数比进行代数化简后求极限.
55 *
56 * 当 dN/dD 含有公因子(如 x^-2)时,直接分别求极限会导致
57 * 无限 0/0 循环.此方法先通过 simplify() 约去公因子,
58 * 再对化简后的表达式求极限.
59 *
60 * @param[in] ratio_node 导数比 dN * dD^(-1) 的 AST 节点
61 * @return 极限结果,化简无效时返回 nullptr
62 */
63std::shared_ptr<const SymbolicNode> LimitVisitor::simplify_and_eval_ratio(
64 const std::shared_ptr<const SymbolicNode>& ratio_node) {
65
66 if (!ratio_node) {
67 return nullptr;
68 }
69
70 auto ratio_expr = LMCAS::detail::make_expression_ptr(ratio_node->clone());
71 auto simplified = ratio_expr->simplify();
72 if (!simplified || !LMCAS::detail::node(simplified)) {
73 return nullptr;
74 }
75
76 if (detail::node(ratio_expr)->compare(*detail::node(simplified)) == 0) return nullptr;
77
78 /// Reject simplified forms that are AddNodes (sums). When simplify() expands
79 /// a fraction like (1-cos(x))/(sin(x)+x*cos(x)) into a sum of terms with
80 /// negative powers, evaluating the limit of that sum can trigger infinity-infinity detection,
81 /// which calls resolve_inf_minus_inf -> apply_lhopital -> simplify_and_eval_ratio
82 /// again, creating an infinite loop.
83 if (std::dynamic_pointer_cast<const AddNode>(LMCAS::detail::node(simplified))) {
84 return nullptr;
85 }
86
87 return derivative_limit(LMCAS::detail::node(simplified), true);
88}

90/**
91 * @brief Taylor 展开回退策略实现.
92 *
93 * 当极限点为有限值时,直接在该点展开 Taylor 级数.
94 * 当极限点为无穷时,先做 x = 1/t 代换,再在 t = 0 处展开.
95 * 从 order=4 开始逐步增加到 max_order,直到找到非零首项.
96 *
97 * @param[in] num 分子 AST 节点
98 * @param[in] den 分母 AST 节点
99 * @param[in] max_order 最大展开阶数
100 * @return 极限结果节点;nullptr 表示当前规则保持结果未知
101 */
102std::shared_ptr<const SymbolicNode> LimitVisitor::taylor_fallback(
103 const std::shared_ptr<const SymbolicNode>& num,
104 const std::shared_ptr<const SymbolicNode>& den,
105 int max_order) {
106
107 if (!num || !den) {
108 return nullptr;
109 }
110 if (!is_inf(point) && (!analytic_at_point(num) || !analytic_at_point(den))) { return nullptr; }
111
112 auto num_expr = LMCAS::detail::make_expression_ptr(num->clone());
113 auto den_expr = LMCAS::detail::make_expression_ptr(den->clone());
114 auto point_expr = LMCAS::detail::make_expression_ptr(point->clone());
115
116 bool at_infinity = is_inf(point);
117
118 std::string expand_var = var;
119 std::shared_ptr<SymbolicExpr> expand_point;
120
121 if (at_infinity) {
122 expand_var = substitute_infinity_series(num, den, var, num_expr, den_expr);
123 expand_point = SymbolicExpr::number(0);
124 } else {
125 expand_point = point_expr;
126 }
127
128 if (!num_expr || !den_expr || !expand_point) {
129 return nullptr;
130 }
131 if (at_infinity) {
132 LimitVisitor analytic(expand_var, detail::node(expand_point), "+", assumption_ctx_, context_);
133 if (!analytic.analytic_at_point(detail::node(num_expr)) ||
134 !analytic.analytic_at_point(detail::node(den_expr))) { return nullptr; }
135 }
136
137 /// 从 order=4 开始,逐步增加到 max_order
138 for (int order = 4; order <= max_order; order += 2) {
139 auto value = taylor_series_limit(num_expr, den_expr, expand_var, expand_point, order);
140 if (value) {
141 return value;
142 }
143 }
144
145 return nullptr;
146}

148std::shared_ptr<const SymbolicNode> LimitVisitor::taylor_series_limit(
149 const std::shared_ptr<SymbolicExpr>& numerator,
150 const std::shared_ptr<SymbolicExpr>& denominator,
151 const std::string& expand_var,
152 const std::shared_ptr<SymbolicExpr>& expand_point, int order) {
153 auto numerator_series = numerator->series(expand_var, expand_point, order);
154 auto denominator_series = denominator->series(expand_var, expand_point, order);
155 if (!numerator_series || !denominator_series) {
156 return nullptr;
157 }
158 numerator_series = numerator_series->simplify();
159 denominator_series = denominator_series->simplify();
160 auto numerator_leading = find_leading_term(numerator_series, expand_var, expand_point, order);
161 auto denominator_leading = find_leading_term(denominator_series, expand_var, expand_point, order);
162 return leading_terms_limit(numerator_leading, denominator_leading);
163}

165std::shared_ptr<const SymbolicNode> LimitVisitor::leading_terms_limit(
166 const std::pair<std::shared_ptr<const SymbolicNode>, int>& numerator,
167 const std::pair<std::shared_ptr<const SymbolicNode>, int>& denominator) {
168 if (!numerator.first || !denominator.first) {
169 return nullptr;
170 }
171 if (denominator.first->is_zero()) {
172 return nullptr;
173 }
174 if (numerator.first->is_zero() || numerator.second > denominator.second) {
175 return LMCAS::detail::make_node<NumberNode>(BigInt(0));
176 }
177 auto ratio = SymbolicExpr::multiply(
178 LMCAS::detail::make_expression_ptr(numerator.first),
179 SymbolicExpr::power(LMCAS::detail::make_expression_ptr(denominator.first),
180 SymbolicExpr::number(-1)));
181 ratio = ratio->simplify();
182 if (numerator.second < denominator.second) {
183 return divergent_leading_ratio(ratio, denominator.second - numerator.second);
184 }
185 return ratio ? LMCAS::detail::node(ratio) : nullptr;
186}

188/**
189 * @internal
190 * @brief 从 Taylor 级数中提取关于 (x - a) 的首个非零项.
191 *
192 * 通过逐阶求导并在展开点求值来确定首个非零系数及其阶数.
193 *
194 * @param[in] series_expr 级数表达式
195 * @param[in] expand_var 展开变量名
196 * @param[in] expand_point 展开中心点
197 * @param[in] max_order 最大检查阶数
198 * @return (首项系数节点, 首项阶数) 对
199 */
200std::pair<std::shared_ptr<const SymbolicNode>, int> LimitVisitor::find_leading_term(
201 const std::shared_ptr<SymbolicExpr>& series_expr,
202 const std::string& expand_var,
203 const std::shared_ptr<SymbolicExpr>& expand_point,
204 int max_order) {
205
206 if (!series_expr) {
207 return {nullptr, 0};
208 }
209
210 auto current = series_expr;
211
212 for (int n = 0; n <= max_order; ++n) {
213 /// 在展开点求值得到第 n 阶系数(乘以 n!)
214 auto val = current->substitute(expand_var, expand_point);
215 if (!val) {
216 return {nullptr, 0};
217 }
218 val = val->simplify();
219
220 auto sign = val ? get_node_sign(LMCAS::detail::node(val)) : std::nullopt;
221 if (!sign) return {nullptr, 0};
222 if (*sign) return {LMCAS::detail::node(val), n};
223
224 /// 对当前表达式求导以获取下一阶系数
225 current = current->differentiate(expand_var);
226 if (!current) {
227 return {nullptr, 0};
228 }
229 current = current->simplify();
230 }
231
232 return {nullptr, max_order + 1};
233}

235/**
236 * @internal
237 * @brief 判断数值节点的符号.
238 * @param[in] node AST 节点
239 * @return 正数返回 1,负数返回 -1,零或未知符号返回 0
240 */
241int LimitVisitor::get_sign(const std::shared_ptr<const SymbolicNode>& node) {
242 if (!node) {
243 return 0;
244 }
245 if (auto num = std::dynamic_pointer_cast<const NumberNode>(node)) {
246 if (std::holds_alternative<double>(num->value())) {
247 double v = std::get<double>(num->value());
248 if (v > 0) {
249 return 1;
250 }
251 if (v < 0) {
252 return -1;
253 }
254 return 0;
255 }
256 if (std::holds_alternative<BigInt>(num->value())) {
257 BigInt v = std::get<BigInt>(num->value());
258 if (v > BigInt(0)) {
259 return 1;
260 }
261 if (v < BigInt(0)) {
262 return -1;
263 }
264 return 0;
265 }
266 if (std::holds_alternative<Rational>(num->value())) {
267 Rational v = std::get<Rational>(num->value());
268 if (v > Rational(0)) {
269 return 1;
270 }
271 if (v < Rational(0)) {
272 return -1;
273 }
274 return 0;
275 }
276 }
277 /// 对于乘法节点,符号为各因子符号之积
278 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
279 int sign = 1;
280 for (auto& op : mul->operands()) {
281 int s = get_sign(op);
282 if (s == 0) {
283 return 0;
284 }
285 sign *= s;
286 }
287 return sign;
288 }
289 return 0;
290}
