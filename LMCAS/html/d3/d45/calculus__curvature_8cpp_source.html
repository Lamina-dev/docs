20ExpressionResult curvature_checked(
21 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
22 ComputationContext& context)
23{
24 const std::string operation = "curvature";
25 auto input = calculus_utils_validate_expr(f, var, context, operation);
26 if (!input) {
27 return ExpressionResult::failure(input.error());
28 }
29
30 auto first = checked_derivative(f, var, context, operation.c_str());
31 if (!first) {
32 return ExpressionResult::failure(first.error());
33 }
34 const auto& f_prime = first.value();
35 auto second = checked_derivative(f_prime, var, context, operation.c_str());
36 if (!second) {
37 return ExpressionResult::failure(second.error());
38 }
39 const auto& f_double_prime = second.value();
40
41 auto abs_f_pp = calculus_utils_make_abs(f_double_prime);
42 if (!abs_f_pp || !LMCAS::detail::node(abs_f_pp)) {
43 return ExpressionResult::failure(CasErrc::InternalInvariant,
44 "absolute value node construction failed",
45 operation);
46 }
47
48 auto f_prime_sq = SymbolicExpr::power(f_prime, SymbolicExpr::number(2));
49 auto one_plus_fp_sq = SymbolicExpr::add(SymbolicExpr::number(1), f_prime_sq);
50
51 auto three_half = SymbolicExpr::number(Rational(3, 2));
52 auto denom = SymbolicExpr::power(one_plus_fp_sq, three_half);
53 auto result = SymbolicExpr::divide(abs_f_pp, denom);
54 auto simplified = result->simplify();
55 return ExpressionResult::success(simplified ? simplified : result);
56}

66ExpressionResult curvature_parametric_checked(
67 const std::shared_ptr<SymbolicExpr>& x_t,
68 const std::shared_ptr<SymbolicExpr>& y_t, const std::string& t,
69 ComputationContext& context)
70{
71 const std::string operation = "curvature_parametric";
72 auto input = calculus_utils_validate_two_exprs(x_t, y_t, t, context, operation);
73 if (!input) {
74 return ExpressionResult::failure(input.error());
75 }
76
77 auto x_first = checked_derivative(x_t, t, context, operation.c_str());
78 if (!x_first) {
79 return ExpressionResult::failure(x_first.error());
80 }
81 const auto& x_prime = x_first.value();
82 auto x_second = checked_derivative(x_prime, t, context, operation.c_str());
83 if (!x_second) {
84 return ExpressionResult::failure(x_second.error());
85 }
86 const auto& x_double_prime = x_second.value();
87
88 auto y_first = checked_derivative(y_t, t, context, operation.c_str());
89 if (!y_first) {
90 return ExpressionResult::failure(y_first.error());
91 }
92 const auto& y_prime = y_first.value();
93 auto y_second = checked_derivative(y_prime, t, context, operation.c_str());
94 if (!y_second) {
95 return ExpressionResult::failure(y_second.error());
96 }
97 const auto& y_double_prime = y_second.value();
98
99 if (x_prime->is_zero() && y_prime->is_zero()) {
100 return ExpressionResult::failure(CasErrc::DomainError,
101 "parametric curvature is undefined for zero velocity",
102 operation);
103 }
104
105 auto cross_term = SymbolicExpr::add(
106 SymbolicExpr::multiply(x_prime, y_double_prime),
107 SymbolicExpr::multiply(
108 SymbolicExpr::number(-1),
109 SymbolicExpr::multiply(y_prime, x_double_prime)));
110
111 auto abs_cross = calculus_utils_make_abs(cross_term);
112 if (!abs_cross || !LMCAS::detail::node(abs_cross)) {
113 return ExpressionResult::failure(CasErrc::InternalInvariant,
114 "absolute value node construction failed",
115 operation);
116 }
117
118 auto x_prime_sq = SymbolicExpr::power(x_prime, SymbolicExpr::number(2));
119 auto y_prime_sq = SymbolicExpr::power(y_prime, SymbolicExpr::number(2));
120 auto sum_sq = SymbolicExpr::add(x_prime_sq, y_prime_sq);
121
122 auto three_half = SymbolicExpr::number(Rational(3, 2));
123 auto denom = SymbolicExpr::power(sum_sq, three_half);
124 auto result = SymbolicExpr::divide(abs_cross, denom);
125 auto simplified = result->simplify();
126 return ExpressionResult::success(simplified ? simplified : result);
127}

139static Result<bool> inflection_curve_defined(
140 const std::shared_ptr<SymbolicExpr>& f,
141 const std::shared_ptr<SymbolicExpr>& f_prime,
142 const std::shared_ptr<SymbolicExpr>& f_double_prime,
143 const std::shared_ptr<SymbolicExpr>& point, const std::string& var,
144 const FactsQuery& facts, const std::string& operation, ComputationContext& context) {
145 auto value = f->substitute(var, point);
146 auto first = f_prime->substitute(var, point);
147 auto second = f_double_prime->substitute(var, point);
148 for (const auto& expression : {value, first, second}) {
149 auto defined = detail::query_definedness(detail::node(expression), facts, Domain::Real, context);
150 if (!defined) { return Result<bool>::failure(defined.error()); }
151 if (defined.value() == Tribool::False) { return false; }
152 auto real_value = detail::query_real_value(detail::node(expression), facts, context);
153 if (!real_value) { return Result<bool>::failure(real_value.error()); }
154 if (real_value.value() == Tribool::False) { return false; }
155 if (defined.value() != Tribool::True || real_value.value() != Tribool::True) {
156 return Result<bool>::failure(CasErrc::Inconclusive,
157 "real curve domain at an inflection candidate is unproved",
158 operation);
159 }
160 }
161 return true;
162}

164static SymbolicExprVectorResult classify_inflection_candidates(
165 const FiniteSolutions& finite, const std::shared_ptr<SymbolicExpr>& f,
166 const std::shared_ptr<SymbolicExpr>& f_prime,
167 const std::shared_ptr<SymbolicExpr>& f_double_prime, const std::string& var,
168 const FactsQuery& facts, ComputationContext& context, const std::string& operation) {
169 std::vector<std::shared_ptr<SymbolicExpr>> points;
170 points.reserve(finite.values.size());
171 for (const auto& solution : finite.values) {
172 auto step = context.consume_steps(1, operation);
173 if (!step) { return SymbolicExprVectorResult::failure(step.error()); }
174 if (!solution.value || !LMCAS::detail::node(solution.value)) {
175 return SymbolicExprVectorResult::failure(CasErrc::InternalInvariant,
176 "checked solver returned a null inflection candidate", operation);
177 }
178 auto real = detail::query_real_value(detail::node(solution.value), facts, context);
179 if (!real) { return SymbolicExprVectorResult::failure(real.error()); }
180 if (real.value() == Tribool::False) { continue; }
181 if (real.value() != Tribool::True || !solution.conditions.empty()) {
182 return SymbolicExprVectorResult::failure(CasErrc::Inconclusive,
183 "inflection candidate is not proved unconditionally real", operation);
184 }
185 auto defined = inflection_curve_defined(
186 f, f_prime, f_double_prime, solution.value, var, facts, operation, context);
187 if (!defined) { return SymbolicExprVectorResult::failure(defined.error()); }
188 if (!defined.value()) { continue; }
189 points.push_back(solution.value);
190 }
191 return SymbolicExprVectorResult::success(std::move(points));
192}

194SymbolicExprVectorResult inflection_points_checked(
195 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
196 ComputationContext& context)
197{
198 const std::string operation = "inflection_points";
199 auto input = calculus_utils_validate_expr(f, var, context, operation);
200 if (!input) {
201 return SymbolicExprVectorResult::failure(input.error());
202 }
203
204 auto first = checked_derivative(f, var, context, operation.c_str());
205 if (!first) {
206 return SymbolicExprVectorResult::failure(first.error());
207 }
208 const auto& f_prime = first.value();
209
210 auto second = checked_derivative(f_prime, var, context, operation.c_str());
211 if (!second) {
212 return SymbolicExprVectorResult::failure(second.error());
213 }
214 const auto& f_double_prime = second.value();
215
216 auto equation = f_double_prime->simplify();
217 if (!equation || !LMCAS::detail::node(equation)) { equation = f_double_prime; }
218
219 auto solved = solve_equation(equation, var, context, SolveOptions{});
220 if (!solved) {
221 return SymbolicExprVectorResult::failure(solved.error());
222 }
223
224 const auto& solutions = solved.value();
225 if (std::holds_alternative<EmptySolutions>(solutions)) {
226 return SymbolicExprVectorResult::success({});
227 }
228 std::optional<detail::AssumptionFacts> assumed_facts;
229 if (context.assumptions()) { assumed_facts.emplace(*context.assumptions()); }
230 const FactsQuery& facts = assumed_facts
231 ? static_cast<const FactsQuery&>(*assumed_facts) : detail::no_facts();
232 if (const auto* finite = std::get_if<FiniteSolutions>(&solutions)) {
233 return classify_inflection_candidates(
234 *finite, f, f_prime, f_double_prime, var, facts, context, operation);
235 }
236
237
238 return SymbolicExprVectorResult::failure(
239 CasErrc::Inconclusive,
240 "inflection equation is outside the finite exact support domain",
241 operation);
242}
