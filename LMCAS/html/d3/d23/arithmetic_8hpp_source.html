43class AddNode : public SymbolicNode {
44private:

48
49 explicit AddNode(std::vector<std::shared_ptr<const SymbolicNode>> ops)
50 : operands_([&ops]() {
51 if (ops.empty()) {
52 throw std::invalid_argument("AddNode requires at least one operand");
53 }
54 std::vector<std::shared_ptr<const SymbolicNode>> flattened;
55 for (const auto& op : ops) {
56 if (!op) {
57 throw std::invalid_argument("AddNode operand cannot be null");
58 }
59 if (auto add = std::dynamic_pointer_cast<const AddNode>(op)) {
60 flattened.insert(flattened.end(), add->operands_.begin(), add->operands_.end());
61 } else {
62 flattened.push_back(op);
63 }
64 }
65 std::sort(flattened.begin(), flattened.end(), [](const auto& a, const auto& b) {
66 bool a_num = a->is_number();
67 bool b_num = b->is_number();
68 if (a_num != b_num) return !a_num;
69 return a->compare(*b) < 0;
70 });
71 return flattened;
