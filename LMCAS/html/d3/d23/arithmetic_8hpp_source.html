50/**
51 * @brief 加法节点，表示多个操作数的求和。
52 *
53 * 构造时自动扁平化嵌套的 AddNode，并按规范顺序排序操作数。
54 */
55class AddNode : public SymbolicNode {
56private:
57 LMCAS_AST_NODE_FACTORY_FRIEND;
58
59 static std::vector<std::shared_ptr<const SymbolicNode>> normalize_operands(
60 std::vector<std::shared_ptr<const SymbolicNode>> ops) {
61 if (ops.empty()) {
62 throw std::invalid_argument("AddNode requires at least one operand");
63 }
64 std::vector<std::shared_ptr<const SymbolicNode>> flattened;
65 for (const auto& op : ops) {
66 if (!op) {
67 throw std::invalid_argument("AddNode operand cannot be null");
68 }
69 if (auto add = std::dynamic_pointer_cast<const AddNode>(op)) {
70 flattened.insert(flattened.end(), add->operands_.begin(), add->operands_.end());
71 } else {
72 flattened.push_back(op);
73 }
74 }
75 std::sort(flattened.begin(), flattened.end(), [](const auto& a, const auto& b) {
76 bool a_num = a->is_number();
77 bool b_num = b->is_number();
78 if (a_num != b_num) return !a_num;
79 return a->compare(*b) < 0;
80 });
81 return flattened;
82 }
83
84 const std::vector<std::shared_ptr<const SymbolicNode>> operands_;
85
86 explicit AddNode(std::vector<std::shared_ptr<const SymbolicNode>> ops)
87 : operands_(normalize_operands(std::move(ops))) {}
88
89public:
90 const std::vector<std::shared_ptr<const SymbolicNode>>& operands() const noexcept {
91 return operands_;
92 }
93
94 int type_priority() const override { return 5; }
95
96protected:
97 std::size_t compute_hash() const override {
98 std::size_t seed = 0;
99 hash_combine(seed, type_priority());
100 for (const auto& op : operands_) {
101 hash_combine(seed, op->hash());
102 }
103 return seed;
104 }
105
106 int compare_same_type(const SymbolicNode& other) const override {
107 const auto& o = static_cast<const AddNode&>(other);
108 if (operands_.size() != o.operands_.size()) {
109 return operands_.size() < o.operands_.size() ? -1 : 1;
110 }
111 for (size_t i = 0; i < operands_.size(); ++i) {
112 int cmp = operands_[i]->compare(*o.operands_[i]);
113 if (cmp != 0) return cmp;
114 }
115 return 0;
116 }
117
118public:
119 void accept(LMCAS::detail::SymbolicVisitor& visitor) const override { LMCAS::detail::SymbolicVisitor::DepthGuard guard(visitor); visitor.visit(*this); }
120 std::shared_ptr<const SymbolicNode> clone() const override {
121 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
122 new_ops.reserve(operands_.size());
123 for (const auto& op : operands_) new_ops.push_back(op->clone());
124 return LMCAS::detail::make_node<AddNode>(std::move(new_ops));
125 }
126};

226 : base_(std::move(b)), exponent_(std::move(e)) {

239
240protected:
241 std::size_t compute_hash() const override {
242 std::size_t seed = 0;
243 hash_combine(seed, type_priority());

246 return seed;
247 }

249 int compare_same_type(const SymbolicNode& other) const override {
250 const auto& o = static_cast<const PowerNode&>(other);

254
255 return base_->compare(*o.base_);

261 return LMCAS::detail::make_node<PowerNode>(base_->clone(), exponent_->clone());
