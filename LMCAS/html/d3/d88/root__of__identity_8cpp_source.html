49static std::shared_ptr<SymbolicExpr> symbolic_root_term(
50 const std::shared_ptr<SymbolicExpr>& coefficient,
51 const std::string& variable, int degree) {
52 if (degree == 0) return coefficient;
53 auto symbol = SymbolicExpr::variable(variable);
54 auto power = degree == 1
55 ? symbol : SymbolicExpr::power(symbol, SymbolicExpr::number(degree));
56 return coefficient->is_one()
57 ? power : SymbolicExpr::multiply(coefficient, power);
58}

60static std::shared_ptr<SymbolicExpr> symbolic_poly_to_expr(
61 const Polynomial<SymbolicPolyCoeff>& poly) {
62 if (poly.is_zero()) return SymbolicExpr::number(0);
63
64 std::vector<std::shared_ptr<SymbolicExpr>> terms;
65 for (int i = poly.degree(); i >= 0; --i) {
66 const auto& coeff = poly.coeffs[i];
67 if (coeff == SymbolicPolyCoeff(0)) continue;
68
69 auto coeff_expr = coeff.val;
70 if (coeff_expr) {
71 if (auto simplified = coeff_expr->simplify()) {
72 coeff_expr = simplified;
73 }
74 if (LMCAS::detail::node(coeff_expr) && coeff_expr->is_zero()) {
75 continue;
76 }
77 }
78 terms.push_back(symbolic_root_term(coeff_expr, poly.variable_name, i));
79 }
80
81 if (terms.empty()) return SymbolicExpr::number(0);
82 if (terms.size() == 1) return terms[0];
83
84 auto result = terms[0];
85 for (size_t i = 1; i < terms.size(); ++i) {
86 result = SymbolicExpr::add(result, terms[i]);
87 }
88 return result;
89}

92std::vector<std::shared_ptr<SymbolicExpr>> make_rootof_solutions(
93 const Polynomial<SymbolicPolyCoeff>& poly,
94 const std::string& var) {
95 if (poly.degree() <= 0) return {};
96 auto polynomial_expression = symbolic_poly_to_expr(poly);
97 ComputationContext context;
98 auto recognized = recognize_rational_polynomial(
99 *polynomial_expression, var, context);
100 if (!recognized || !recognized.value()) return {};
101 auto canonical = recognized.value()->square_free_part().make_monic();
102 canonical.variable_name = "_root";
103 std::vector<std::shared_ptr<SymbolicExpr>> solutions;
104 solutions.reserve(static_cast<std::size_t>(canonical.degree()));
105 for (int index = 0; index < canonical.degree(); ++index) {
106 solutions.push_back(LMCAS::detail::make_expression_ptr(
107 LMCAS::detail::make_node<RootOfNode>(
108 detail::ExactRootId{
109 canonical, static_cast<std::size_t>(index)},
110 var)));
111 }
112 return solutions;
113}

118Result<const RootOfNode*> parse_rootof_request(
119 const std::shared_ptr<SymbolicExpr>& rootof_expr,
120 ComputationContext& context,
121 const std::string& operation) {
122 if (!rootof_expr || !LMCAS::detail::node(rootof_expr)) {
123 return Result<const RootOfNode*>::failure(
124 CasErrc::InvalidArgument, "RootOf expression cannot be null", operation);
125 }
126 auto root = std::dynamic_pointer_cast<const RootOfNode>(
127 LMCAS::detail::node(rootof_expr));
128 if (!root) {
129 return Result<const RootOfNode*>::failure(
130 CasErrc::InvalidArgument,
131 "expression is not a valid RootOf", operation);
132 }
133 auto access = context.consume_steps(1, operation);
134 if (!access) return Result<const RootOfNode*>::failure(access.error());
135 return Result<const RootOfNode*>::success(root.get());
136}
