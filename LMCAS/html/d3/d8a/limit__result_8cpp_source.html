44Result<int> finite_sign(const NodePtr& node, const FactsQuery& facts, ComputationContext& context) {
45 auto nz = detail::query_nonzero_value(node, facts, Domain::Real, context);
46 if (!nz) { return Result<int>::failure(nz.error()); }
47 if (nz.value() == Tribool::False) { return 0; }
48 auto positive = detail::query_positive_value(node, facts, context);
49 if (!positive) { return Result<int>::failure(positive.error()); }
50 if (positive.value() == Tribool::True) { return 1; }
51 auto nonnegative = detail::query_nonnegative_value(node, facts, context);
52 if (!nonnegative) { return Result<int>::failure(nonnegative.error()); }
53 if (nonnegative.value() == Tribool::False) { return -1; }
54 auto negative = facts.is_negative(node, context);
55 if (!negative) { return Result<int>::failure(negative.error()); }
56 if (negative.value() == Tribool::True) { return -1; }
57 return Result<int>::failure(CasErrc::Inconclusive, "finite sign is unknown", "limit");
58}

62LimitResult classify_infinite_sum(const AddNode& sum, const std::string& variable,
63 ComputationContext& context, const FactsQuery& facts, Domain domain) {
64 int sign = 0;
65 std::vector<NodePtr> finite_terms;
66 for (const auto& term : sum.operands()) {
67 auto value = classify_limit_result(term, variable, context, facts, domain);
68 if (!value) { return value; }
69 int next = infinity_sign(value.value().value);
70 if (sign && next && sign != next) { return unknown(); }
71 if (next) { sign = next; }
72 else { finite_terms.push_back(detail::node(std::get<FiniteLimit>(value.value().value).value)); }
73 }
74 if (sign) { return sign > 0 ? outcome(PositiveInfinityLimit{}) : outcome(NegativeInfinityLimit{}); }
75 auto finite = detail::make_expression_ptr(detail::make_node<AddNode>(std::move(finite_terms)))->simplify();
76 return finite ? classify_limit_result(detail::node(finite), variable, context, facts, domain) : unknown();
77}

79LimitResult classify_infinite_product(const MultiplyNode& product, const std::string& variable,
80 ComputationContext& context, const FactsQuery& facts, Domain domain) {
81 int sign = 1;
82 bool infinite = false;
83 std::vector<NodePtr> finite_factors;
84 for (const auto& factor : product.operands()) {
85 auto value = classify_limit_result(factor, variable, context, facts, domain);
86 if (!value) { return value; }
87 int next = infinity_sign(value.value().value);
88 if (next) { infinite = true; sign *= next; }
89 else { finite_factors.push_back(detail::node(std::get<FiniteLimit>(value.value().value).value)); }
90 }
91 if (infinite) {
92 for (const auto& factor : finite_factors) {
93 auto s = finite_sign(factor, facts, context);
94 if (!s) { return LimitResult::failure(s.error()); }
95 if (!s.value()) { return unknown(); }
96 sign *= s.value();
97 }
98 return sign > 0 ? outcome(PositiveInfinityLimit{}) : outcome(NegativeInfinityLimit{});
99 }
100 auto finite = detail::make_expression_ptr(detail::make_node<MultiplyNode>(std::move(finite_factors)))->simplify();
101 return finite ? classify_limit_result(detail::node(finite), variable, context, facts, domain) : unknown();
102}

105 ComputationContext& context, const FactsQuery& facts, Domain domain) {
106 auto base = classify_limit_result(power.base(), variable, context, facts, domain);

109 BigInt integer;

120LimitResult classify_infinite_function(const FunctionNode& function, const std::string& variable,
121 ComputationContext& context, const FactsQuery& facts, Domain domain) {
122 if (function.arguments().size() != 1) { return unknown(); }
123 auto argument = classify_limit_result(function.arguments()[0], variable, context, facts, domain);
124 if (!argument) { return argument; }
125 int sign = infinity_sign(argument.value().value);
126 if (!sign) {
127 auto applied = detail::make_node<FunctionNode>(function.type(), std::vector<NodePtr>{
128 detail::node(std::get<FiniteLimit>(argument.value().value).value)});
129 return classify_limit_result(applied, variable, context, facts, domain);
130 }
131 switch (function.type()) {
132 case FunctionNode::FuncType::Exp: {
133 return sign > 0 ? outcome(PositiveInfinityLimit{}) : outcome(FiniteLimit{SymbolicExpr::number(0)});
134 }
135 case FunctionNode::FuncType::Abs: { return outcome(PositiveInfinityLimit{}); }
136 case FunctionNode::FuncType::Ln: case FunctionNode::FuncType::Sqrt: {
137 return sign > 0 ? outcome(PositiveInfinityLimit{}) : unknown();
138 }
139 case FunctionNode::FuncType::ArcTan: {
140 return outcome(FiniteLimit{SymbolicExpr::multiply(SymbolicExpr::number(Rational(sign,2)),
141 LMCAS::detail::make_expression_ptr(
142 LMCAS::detail::make_node<VariableNode>("pi", true)))});
143 }
144 default: { return unknown(); }
145 }

148LimitResult classify_limit_result(const NodePtr& node, const std::string& variable,
149 ComputationContext& context, const FactsQuery& facts, Domain domain) {
150 auto step = context.consume_steps(1, "limit.classify");
151 if (!step) { return LimitResult::failure(step.error()); }
152 if (!node || detail::contains_node_type<LimitNode>(node) || expression_depends_on_variable(node, variable))
153 { return unknown(); }
154 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(node)) {
155 if (function->type() == FunctionNode::FuncType::Infinity) { return outcome(PositiveInfinityLimit{}); }
156 }
157 if (has_infinity(node)) {
158 if (auto sum = std::dynamic_pointer_cast<const AddNode>(node)) {
159 return classify_infinite_sum(*sum, variable, context, facts, domain);
160 }
161 if (auto product = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
162 return classify_infinite_product(*product, variable, context, facts, domain);
163 }
164 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
165 return classify_infinite_power(*power, variable, context, facts, domain);
166 }
167 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(node)) {
168 return classify_infinite_function(*function, variable, context, facts, domain);
169 }
170 return unknown();
171 }
172 auto defined = detail::query_definedness(node, facts, domain, context);
173 if (!defined) { return LimitResult::failure(defined.error()); }
174 if (defined.value() != Tribool::True) { return unknown(); }
175 return outcome(FiniteLimit{detail::make_expression_ptr(node)});
176}

206bool continuous_function_branch(FunctionNode::FuncType type, const NodePtr& argument,
207 const ComplexContinuityPoint* point) {
208 switch (type) {
209 case FunctionNode::FuncType::Sin: { return true; }
210 case FunctionNode::FuncType::Cos: { return true; }
211 case FunctionNode::FuncType::Tan: { return true; }
212 case FunctionNode::FuncType::Exp: { return true; }
213 case FunctionNode::FuncType::Abs: { return true; }
214 case FunctionNode::FuncType::Ln: { return !point || point->positive(argument); }
215 case FunctionNode::FuncType::Sqrt: { return !point || point->positive(argument); }
216 case FunctionNode::FuncType::ArcTan: { return point == nullptr; }
217 case FunctionNode::FuncType::ArcSin: { return point == nullptr; }
218 case FunctionNode::FuncType::ArcCos: { return point == nullptr; }
219 default: { return false; }
220 }
221}
222bool elementary_continuous_structure(const NodePtr& node, const ComplexContinuityPoint* point) {
223 if (std::dynamic_pointer_cast<const NumberNode>(node) ||
224 std::dynamic_pointer_cast<const VariableNode>(node)) { return true; }
225 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
226 return power_continuous_structure(*power, point);
227 }
228 if (auto complex = std::dynamic_pointer_cast<const ComplexNode>(node)) {
229 return point && elementary_continuous_structure(complex->real(), point) &&
230 elementary_continuous_structure(complex->imag(), point);
231 }
232 const std::vector<NodePtr>* operands = nullptr;
233 if (auto sum = std::dynamic_pointer_cast<const AddNode>(node)) {
234 operands = &sum->operands();
235 } else if (auto product = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
236 operands = &product->operands();
237 }
238 if (operands) {
239 for (const auto& operand : *operands) {
240 if (!elementary_continuous_structure(operand, point)) { return false; }
241 }
242 return true;
243 }
244 auto function = std::dynamic_pointer_cast<const FunctionNode>(node);
245 return function && function->arguments().size() == 1 &&
246 continuous_function_branch(function->type(), function->arguments()[0], point) &&
247 elementary_continuous_structure(function->arguments()[0], point);
248}

250class SideBranchRewriter final : public detail::SymbolicRewriter {
251public:
252 SideBranchRewriter(LimitVisitor& visitor, const FactsQuery& facts, ComputationContext& context)
253 : visitor_(visitor), facts_(facts), context_(context) {}
254 void visit(const PiecewiseNode& node) override {
255 auto selected = visitor_.select_piecewise_near_point(node);
256 set_result(selected ? rewrite(selected) : current());
257 }
258 void visit(const PowerNode& node) override {
259 bool changed = false;
260 auto base = rewrite_child(node.base(), changed);
261 auto exponent = rewrite_child(node.exponent(), changed);
262 const auto square = std::dynamic_pointer_cast<const PowerNode>(base);
263 if (square) {
264 auto outer = extract_coeff_value<Rational>(detail::make_expression_ptr(exponent));
265 auto inner = extract_coeff_value<Rational>(detail::make_expression_ptr(square->exponent()));
266 if (outer && inner && outer.value() == Rational(1, 2) &&
267 inner.value() == Rational(2) && rewrite_absolute(square->base())) return;
268 }
269 set_result(changed ? detail::make_node<PowerNode>(base, exponent) : current());
270 }
271 void visit(const FunctionNode& node) override {
272 bool changed = false;
273 auto args = rewrite_children(node.arguments(), changed);
274 if (args.size() == 1) {
275 if (node.type() == FunctionNode::FuncType::Abs && rewrite_absolute(args[0])) { return; }
276 if (node.type() == FunctionNode::FuncType::Sqrt && rewrite_square_absolute(args[0])) { return; }
277 }
278 set_result(changed ? detail::make_node<FunctionNode>(node.type(), std::move(args)) : current());
279 }
280private:
281 bool rewrite_square_absolute(const NodePtr& argument) {
282 auto square = std::dynamic_pointer_cast<const PowerNode>(argument);
283 if (!square) { return false; }
284 auto exponent = extract_coeff_value<Rational>(detail::make_expression_ptr(square->exponent()));
285 return exponent && exponent.value() == Rational(2) && rewrite_absolute(square->base());
286 }
287 bool rewrite_absolute(const NodePtr& argument) {
288 auto real = detail::query_real_value(argument, facts_, context_);
289 if (!real) throw real.error();
290 auto sign = real.value() == Tribool::True ? visitor_.sign_near_point(argument) : std::nullopt;
291 if (!sign) return false;
292 set_result(*sign < 0 ? detail::node(SymbolicExpr::multiply(
293 SymbolicExpr::number(-1), detail::make_expression_ptr(argument))) : argument);
294 return true;
295 }
296 LimitVisitor& visitor_;
297 const FactsQuery& facts_;
298 ComputationContext& context_;
299};

301std::optional<LimitResult> scaled_one_sided_limit(const MultiplyNode& product,
302 const std::string& variable, const LimitExprPtr& point, LimitDirection direction,
303 ComputationContext& context, const FactsQuery& facts) {
304 std::shared_ptr<const SymbolicNode> varying;
305 bool invertible_scale = true;
306 for (const auto& factor : product.operands()) {
307 if (expression_depends_on_variable(factor, variable)) {
308 if (varying) { invertible_scale = false; break; }
309 varying = factor;
310 } else {
311 auto nonzero = detail::query_nonzero_value(factor, facts, Domain::Real, context);
312 if (!nonzero) { return LimitResult::failure(nonzero.error()); }
313 if (nonzero.value() != Tribool::True) { invertible_scale = false; break; }
314 }
315 }
316 if (invertible_scale && varying) {
317 auto inner = limit_checked(detail::make_expression_ptr(varying), variable, point, direction, context);
318 if (!inner && inner.error().code != CasErrc::Inconclusive) { return inner; }
319 if (inner && std::holds_alternative<LimitDoesNotExist>(inner.value().value)) { return inner; }
320 }
321 return std::nullopt;
322}

324std::optional<LimitResult> oscillatory_one_sided_limit(const NodePtr& node,
325 const std::string& variable, const LimitExprPtr& point, LimitDirection direction,
326 ComputationContext& context, const FactsQuery& facts, LimitVisitor& visitor) {
327 auto f = std::dynamic_pointer_cast<const FunctionNode>(node);
328 if (!f || f->arguments().size() != 1) { return std::nullopt; }
329 if (f->type() != FunctionNode::FuncType::Sin && f->type() != FunctionNode::FuncType::Cos) {
330 return std::nullopt;
331 }
332 auto argument = detail::make_expression_ptr(f->arguments()[0]);
333 auto inner = limit_checked(argument, variable, point, direction, context);
334 if (!inner && inner.error().code != CasErrc::Inconclusive) { return inner; }
335 if (!inner || !infinity_sign(inner.value().value) || !elementary_continuous_structure(f->arguments()[0])) {
336 return std::nullopt;
337 }
338 auto domain = detail::domain_constraints(f->arguments()[0], facts, Domain::Real, context);
339 if (!domain) { return LimitResult::failure(domain.error()); }
340 if (!domain.value()) { return std::nullopt; }
341 for (const auto& condition : *domain.value()) {
342 if (!visitor.condition_holds_near_point(detail::node(condition))) { return std::nullopt; }
343 }
344 return outcome(LimitDoesNotExist{});
345}

347LimitResult evaluate_one_sided_node(const NodePtr& node, const std::string& variable,
348 const LimitExprPtr& point, LimitDirection direction, ComputationContext& context,
349 const AssumptionContext& local, const FactsQuery& facts, LimitVisitor& visitor) {
350 if (!expression_depends_on_variable(node, variable))
351 { return classify_limit_result(node, variable, context, facts); }
352 auto original_domain = detail::domain_constraints(node, facts, Domain::Real, context);
353 if (!original_domain) { return LimitResult::failure(original_domain.error()); }
354 if (!original_domain.value()) { return unknown(); }
355 for (const auto& condition : *original_domain.value())
356 { if (!visitor.condition_holds_near_point(detail::node(condition))) { return unknown(); } }
357 if (auto product = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
358 auto scaled = scaled_one_sided_limit(*product, variable, point, direction, context, facts);
359 if (scaled) { return std::move(*scaled); }
360 }
361 auto oscillatory = oscillatory_one_sided_limit(node, variable, point, direction, context, facts, visitor);
362 if (oscillatory) { return std::move(*oscillatory); }
363 auto prepared = local.simplify(detail::expression_from_node(node));
364 if (!prepared) { return unknown(); }
365 detail::node(prepared)->accept(visitor);
366 return classify_limit_result(visitor.get_result(), variable, context, facts);
367}

370LimitResult limit_one_sided_checked(const LimitExprPtr& expression, const std::string& variable,
371 const LimitExprPtr& point, LimitDirection direction, ComputationContext& context) {
372 AssumptionContext local = context.assumptions() ? *context.assumptions() : AssumptionContext{};
373 auto real_variable = local.assume_domain_checked(variable, Domain::Real);
374 if (!real_variable) { return assumption_limit_failure(real_variable.error()); }
375 if (has_infinity(detail::node(point))) {
376 const int sign = std::dynamic_pointer_cast<const FunctionNode>(detail::node(point)) ? 1 : -1;
377 auto tail = local.assume_sign_checked(variable, sign < 0 ? Sign::Negative : Sign::Positive);
378 if (!tail) { return assumption_limit_failure(tail.error()); }
379 }
380 detail::AssumptionFacts facts(local);
381 const char* token = direction == LimitDirection::FromBelow ? "-" : "+";
382 LimitVisitor visitor(variable, detail::node(point), token, &local, &context);
383 if (!has_infinity(detail::node(point))) {
384 auto sign = visitor.sign_near_point(detail::make_node<VariableNode>(variable));
385 if (sign && *sign != 0) {
386 auto side = local.assume_sign_checked(variable, *sign < 0 ? Sign::Negative : Sign::Positive);
387 if (!side) { return assumption_limit_failure(side.error()); }
388 }
389 }
390 SideBranchRewriter branches(visitor, facts, context);
391 auto node = branches.rewrite(detail::node(expression));
392 return evaluate_one_sided_node(node, variable, point, direction, context, local, facts, visitor);
393}

395LimitResult combine_one_sided_limits(const LimitOutcome& left, const LimitOutcome& right,
396 ComputationContext& context) {
397 if (std::holds_alternative<LimitDoesNotExist>(left) || std::holds_alternative<LimitDoesNotExist>(right))
398 return outcome(LimitDoesNotExist{});
399 int ls = infinity_sign(left), rs = infinity_sign(right);
400 if (ls || rs) {
401 if (ls == rs) return outcome(left);
402 return outcome(LimitDoesNotExist{});
403 }
404 const auto& lv = std::get<FiniteLimit>(left).value;
405 const auto& rv = std::get<FiniteLimit>(right).value;
406 auto equal = detail::compare_pointwise_values(lv, rv, context, Domain::Real);
407 if (!equal) return LimitResult::failure(equal.error());
408 if (equal.value() == Tribool::True) return outcome(left);
409 if (equal.value() == Tribool::False) return outcome(LimitDoesNotExist{});
410 return unknown();
411}

436Result<bool> complex_polynomial_nonzero(const RelationalNode& relation, const std::string& variable,
437 const FactsQuery& facts, ComputationContext& context, std::vector<NodePtr>& nonzero_bases) {
438 auto difference = SymbolicExpr::add(detail::make_expression_ptr(relation.left()),
439 SymbolicExpr::multiply(SymbolicExpr::number(-1), detail::make_expression_ptr(relation.right())));
440 auto polynomial = symbolic_to_poly<SymbolicPolyCoeff>(difference, variable);
441 if (!polynomial) {
442 if (polynomial.error().code == CasErrc::UnsupportedExpression) { return false; }
443 return Result<bool>::failure(polynomial.error());
444 }
445 bool nonzero = false;
446 for (const auto& coefficient : polynomial.value().coeffs) {
447 auto step = context.consume_steps(1, "limit.complex_domain");
448 if (!step) { return Result<bool>::failure(step.error()); }
449 auto defined = detail::query_definedness(detail::node(coefficient.val), facts, Domain::Complex, context);
450 if (!defined) { return Result<bool>::failure(defined.error()); }
451 if (defined.value() != Tribool::True) { return false; }
452 auto value = detail::query_nonzero_value(detail::node(coefficient.val), facts, Domain::Complex, context);
453 if (!value) { return Result<bool>::failure(value.error()); }
454 nonzero |= value.value() == Tribool::True;
455 }
456 if (!nonzero) { return false; }
457 auto normalized = difference->simplify();
458 if (!normalized) { return false; }
459 nonzero_bases.push_back(detail::node(normalized));
460 return true;
461}

463Result<bool> complex_punctured_condition(const NodePtr& condition, const std::string& variable,
464 const FactsQuery& facts, ComputationContext& context, std::vector<NodePtr>& nonzero_bases) {
465 auto step = context.consume_steps(1, "limit.complex_condition");
466 if (!step) { return Result<bool>::failure(step.error()); }
467 auto logical = std::dynamic_pointer_cast<const LogicalNode>(condition);
468 if (logical && logical->op() == LogicalNode::Op::Or) {
469 auto left = complex_punctured_condition(logical->left(), variable, facts, context, nonzero_bases);
470 if (!left || left.value()) { return left; }
471 return complex_punctured_condition(logical->right(), variable, facts, context, nonzero_bases);
472 }
473 auto relation = std::dynamic_pointer_cast<const RelationalNode>(condition);
474 if (!relation || relation->op() != RelationalNode::Op::NEQ) { return false; }
475 return complex_polynomial_nonzero(*relation, variable, facts, context, nonzero_bases);
476}

490Result<void> validate_limit_input(const LimitExprPtr& expression, const std::string& variable,
491 const LimitExprPtr& point, LimitDirection direction, Domain domain) {
492 if (!expression || !point || !detail::node(expression) || !detail::node(point) || variable.empty()) {
493 return Result<void>::failure(CasErrc::InvalidArgument,
494 "limit requires an expression, named variable, and point", "limit");
495 }
496 if (expression_depends_on_variable(detail::node(point), variable)) {
497 return Result<void>::failure(CasErrc::InvalidArgument,
498 "limit point depends on the approaching variable", "limit");
499 }
500 if (domain != Domain::Real && domain != Domain::Complex) {
501 return Result<void>::failure(CasErrc::InvalidArgument, "limit domain must be Real or Complex", "limit");
502 }
503 if (domain == Domain::Complex && direction != LimitDirection::Both) {
504 return Result<void>::failure(CasErrc::InvalidArgument, "complex limits have no ordered sides", "limit");
505 }
506 return Result<void>::success();
507}

509LimitResult limit_complex_at_point(const LimitExprPtr& expression, const std::string& variable,
510 const LimitExprPtr& point, ComputationContext& context, const FactsQuery& facts) {
511 if (has_infinity(detail::node(expression))) { return unknown(); }
512 if (!expression_depends_on_variable(detail::node(expression), variable)) {
513 return classify_limit_result(detail::node(expression), variable, context, facts, Domain::Complex);
514 }
515 ComplexContinuityPoint continuity_point{variable, detail::node(point), facts, context};
516 if (!elementary_continuous_structure(detail::node(expression), &continuity_point)) { return unknown(); }
517 std::vector<NodePtr> nonzero_bases;
518 auto domain_proof = complex_punctured_domain(detail::node(expression), variable, facts, context, nonzero_bases);
519 if (!domain_proof) { return LimitResult::failure(domain_proof.error()); }
520 if (!domain_proof.value()) { return unknown(); }
521 ComplexPuncturedFacts punctured(facts, nonzero_bases);
522 NormalizationVisitor normalizer(context, punctured, Domain::Complex);
523 detail::node(expression)->accept(normalizer);
524 auto prepared = normalizer.get_result();
525 if (!prepared) { return unknown(); }
526 auto substituted = substitute_free(prepared, variable, detail::node(point));
527 if (!substituted) { return unknown(); }
528 auto defined = detail::query_definedness(substituted, facts, Domain::Complex, context);
529 if (!defined) { return LimitResult::failure(defined.error()); }
530 if (defined.value() != Tribool::True) { return unknown(); }
531 substituted->accept(normalizer);
532 auto value = normalizer.get_result();
533 return value ? classify_limit_result(value, variable, context, facts, Domain::Complex) : unknown();
534}
535
536LimitResult limit_real_at_point(const LimitExprPtr& expression, const std::string& variable,
537 const LimitExprPtr& point, LimitDirection direction, int endpoint_sign, ComputationContext& context) {
538 if (direction != LimitDirection::Both || endpoint_sign) {
539 return limit_one_sided_checked(expression, variable, point, direction, context);
540 }
541 auto left = limit_one_sided_checked(expression, variable, point, LimitDirection::FromBelow, context);
542 if (!left && left.error().code != CasErrc::Inconclusive) { return left; }
543 auto right = limit_one_sided_checked(expression, variable, point, LimitDirection::FromAbove, context);
544 if (!right && right.error().code != CasErrc::Inconclusive) { return right; }
545 if ((left && std::holds_alternative<LimitDoesNotExist>(left.value().value)) ||
546 (right && std::holds_alternative<LimitDoesNotExist>(right.value().value))) {
547 return outcome(LimitDoesNotExist{});
548 }
549 if (!left || !right) { return unknown(); }
550 return combine_one_sided_limits(left.value().value, right.value().value, context);
551}

553LimitResult limit_at_checked_point(const LimitExprPtr& expression, const std::string& variable,
554 const LimitExprPtr& point, LimitDirection direction, ComputationContext& context,
555 const FactsQuery& facts, Domain domain) {
556 auto endpoint = classify_limit_result(detail::node(point), variable, context, facts, domain);
557 if (!endpoint) { return endpoint; }
558 const int endpoint_sign = infinity_sign(endpoint.value().value);
559 if (domain == Domain::Complex && endpoint_sign) { return unknown(); }
560 auto canonical_point = endpoint_sign ? SymbolicExpr::infinity(endpoint_sign)
561 : std::get<FiniteLimit>(endpoint.value().value).value->simplify();
562 if (!canonical_point) { return unknown(); }
563 if (domain == Domain::Complex) {
564 return limit_complex_at_point(expression, variable, canonical_point, context, facts);
565 }
566 return limit_real_at_point(expression, variable, canonical_point, direction, endpoint_sign, context);
567}
568}

570LimitResult limit_checked(const LimitExprPtr& expression, const std::string& variable,
571 const LimitExprPtr& point, LimitDirection direction, ComputationContext& context, Domain domain) {
572 auto valid = validate_limit_input(expression, variable, point, direction, domain);
573 if (!valid) { return LimitResult::failure(valid.error()); }
574 auto step = context.consume_steps(1, "limit");
575 if (!step) { return LimitResult::failure(step.error()); }
576 try {
577 std::optional<detail::AssumptionFacts> assumed;
578 if (context.assumptions()) { assumed.emplace(*context.assumptions()); }
579 const auto& facts = assumed ? static_cast<const FactsQuery&>(*assumed) : detail::no_facts();
580 return limit_at_checked_point(expression, variable, point, direction, context, facts, domain);
581 } catch (const CasError& error) {
582 return LimitResult::failure(error);
583 } catch (const std::bad_alloc&) {
584 return LimitResult::failure(CasErrc::ResourceLimit, "limit allocation failed", "limit");
585 } catch (const std::exception& error) {
586 return LimitResult::failure(CasErrc::InternalInvariant, error.what(), "limit");
587 }
588}
589LimitResult limit_checked(const LimitExprPtr& expression, const std::string& variable,
590 const LimitExprPtr& point, LimitDirection direction, Domain domain) {
591 ComputationContext context;
592 return limit_checked(expression, variable, point, direction, context, domain);
593}
594LimitExpressionResult limit_expression_checked(const LimitExprPtr& expression, const std::string& variable,
595 const LimitExprPtr& point, LimitDirection direction, ComputationContext& context, Domain domain) {
596 auto result = limit_checked(expression, variable, point, direction, context, domain);
597 if (!result) return LimitExpressionResult::failure(result.error());
598 const auto& value = result.value().value;
599 if (auto finite = std::get_if<FiniteLimit>(&value)) return LimitExpressionResult::success(finite->value);
600 if (int sign = infinity_sign(value)) return LimitExpressionResult::success(SymbolicExpr::infinity(sign));
601 return LimitExpressionResult::failure(CasErrc::Inconclusive, "limit does not exist", "limit");
602}
603LimitExpressionResult limit_expression_checked(const LimitExprPtr& expression, const std::string& variable,
604 const LimitExprPtr& point, LimitDirection direction, Domain domain) {
