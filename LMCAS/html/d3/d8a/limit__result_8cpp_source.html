18bool is_negative_infinity(const std::shared_ptr<const SymbolicNode>& node) {
19 auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node);
20 if (!multiply) return false;
21 bool negative = false;
22 bool infinity = false;
23 for (const auto& operand : multiply->operands()) {
24 infinity = infinity || is_infinity(operand);
25 auto number = std::dynamic_pointer_cast<const NumberNode>(operand);
26 if (number && std::holds_alternative<BigInt>(number->value()) &&
27 std::get<BigInt>(number->value()) == BigInt(-1)) {
28 negative = true;
29 }
30 }
31 return negative && infinity;
32}

41LimitResult limit_checked(
42 const LimitExprPtr& expression,
43 const std::string& variable,
44 const LimitExprPtr& point,
45 LimitDirection direction,
46 ComputationContext& context) {
47 constexpr const char* operation = "limit";
48 if (!expression || !point || !lamina::detail::node(expression) ||
49 !lamina::detail::node(point) || variable.empty()) {
50 return LimitResult::failure(
51 CasErrc::InvalidArgument,
52 "limit requires an expression, named variable, and point",
53 operation);
54 }
55 auto step = context.consume_steps(1, operation);
56 if (!step) return LimitResult::failure(step.error());
57
58 if (is_infinity(lamina::detail::node(point)) ||
59 is_negative_infinity(lamina::detail::node(point))) {
60 auto function = std::dynamic_pointer_cast<const FunctionNode>(
61 lamina::detail::node(expression));
62 if (function && function->arguments().size() == 1 &&
63 (function->type() == FunctionNode::FuncType::Sin ||
64 function->type() == FunctionNode::FuncType::Cos) &&
65 depends_on(expression, variable)) {
66 return LimitResult::success(Verified<LimitOutcome>{
67 LimitDoesNotExist{}, ByConstructionProof{}});
68 }
69 }
70
71 try {
72 const char* direction_token = "";
73 if (direction == LimitDirection::FromAbove) direction_token = "+";
74 if (direction == LimitDirection::FromBelow) direction_token = "-";
75 LimitVisitor visitor(
76 variable, lamina::detail::node(point), direction_token,
77 context.assumptions().get());
78 lamina::detail::node(expression)->accept(visitor);
79 auto result_node = visitor.get_result();
80 if (!result_node ||
81 lamina::detail::contains_node_type<LimitNode>(result_node) ||
82 expression_depends_on_variable(result_node, variable)) {
83 return LimitResult::failure(
84 CasErrc::Inconclusive,
85 "limit could not be proved in the supported domain",
86 operation);
87 }
88 auto value = lamina::detail::make_expression_ptr(result_node);
89 if (is_infinity(result_node)) {
90 return LimitResult::success(Verified<LimitOutcome>{
91 PositiveInfinityLimit{}, ByConstructionProof{}});
92 }
93 if (is_negative_infinity(result_node)) {
94 return LimitResult::success(Verified<LimitOutcome>{
95 NegativeInfinityLimit{}, ByConstructionProof{}});
96 }
97 return LimitResult::success(Verified<LimitOutcome>{
98 FiniteLimit{std::move(value)}, ByConstructionProof{}});
99 } catch (const std::bad_alloc&) {
100 return LimitResult::failure(
101 CasErrc::ResourceLimit, "limit allocation failed", operation);
102 } catch (const std::exception& error) {
103 return LimitResult::failure(
104 CasErrc::InternalInvariant, error.what(), operation);
105 }
106}

123 auto result = limit_checked(
124 expression, variable, point, direction, context);
125 if (!result) return LimitExpressionResult::failure(result.error());
126 const auto& outcome = result.value().value;
127 if (const auto* finite = std::get_if<FiniteLimit>(&outcome)) {
128 return LimitExpressionResult::success(finite->value);
129 }
130 if (std::holds_alternative<PositiveInfinityLimit>(outcome)) {
131 return LimitExpressionResult::success(SymbolicExpr::infinity(1));
132 }
133 if (std::holds_alternative<NegativeInfinityLimit>(outcome)) {
134 return LimitExpressionResult::success(SymbolicExpr::infinity(-1));
135 }
