85Result<SymbolicExpr> apply_assumption_simplifications(
86 const SymbolicExpr& expr, const std::string& var,
87 const AssumptionContext* ctx) {

90 SymbolicExpr var_expr = *SymbolicExpr::variable(var);
91 auto positive = ctx->is_positive(var_expr);
92 if (!positive) {
93 return Result<SymbolicExpr>::failure(positive.error());
94 }
95
96 if (positive.value() == Tribool::True) {
97 auto new_root = simplify_abs_positive(
