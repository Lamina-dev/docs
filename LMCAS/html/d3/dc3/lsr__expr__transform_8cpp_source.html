67 kSubstituteOperation);

164ExprMatchResult expr_match(const ExprPtr& pattern,
165 const ExprPtr& target,
166 const std::vector<std::string>& wildcards,
167 ComputationContext& context) {
168 auto step = context.consume_steps(1, kExprMatchOperation);
169 if (!step) return ExprMatchResult::failure(step.error());
170 if (!pattern || !lamina::detail::node(pattern)) {
171 return expr_match_failure(CasErrc::InvalidArgument,
172 "match pattern cannot be null",
173 kExprMatchOperation);
174 }
175 if (!target || !lamina::detail::node(target)) {
176 return expr_match_failure(CasErrc::InvalidArgument,
177 "match target cannot be null",
178 kExprMatchOperation);
179 }
180
181 std::unordered_set<std::string> wildcard_set;
182 wildcard_set.reserve(wildcards.size());
183 for (const auto& wildcard : wildcards) {
184 if (wildcard.empty()) {
185 return expr_match_failure(CasErrc::InvalidArgument,
186 "wildcard names cannot be empty",
187 kExprMatchOperation);
188 }
189 if (!wildcard_set.insert(wildcard).second) {
190 return expr_match_failure(CasErrc::InvalidArgument,
191 "wildcard names must be unique",
192 kExprMatchOperation);
193 }
194 }
195
196 try {
197 MatchMap raw_bindings;
198 const bool matched = Matcher::match(
199 *pattern, *target, wildcard_set, raw_bindings);
200 if (!matched) {
201 return ExprMatchResult::success(ExprMatch{false, {}});
202 }
203
204 std::vector<std::string> names;
205 names.reserve(raw_bindings.size());
206 for (const auto& binding : raw_bindings) {
207 names.push_back(binding.first);
208 }
209 std::sort(names.begin(), names.end());
210
211 ExprMatch result;
212 result.matched = true;
213 result.bindings.reserve(names.size());
214 for (const auto& name : names) {
215 const auto& value = raw_bindings.at(name);
216 auto node = lamina::detail::node(value);
217 if (!node) {
218 return expr_match_failure(CasErrc::InternalInvariant,
219 "matcher produced a null binding",
220 kExprMatchOperation);
221 }
222 result.bindings.push_back(
223 ExprMatchBinding{name,
224 lamina::detail::make_expression_ptr(node)});
225 }
226 return ExprMatchResult::success(std::move(result));
227 } catch (const std::bad_alloc&) {
228 return expr_match_failure(CasErrc::ResourceLimit,
229 "expression matching allocation failed",
230 kExprMatchOperation);
231 } catch (const std::exception& error) {
232 return expr_match_failure(CasErrc::InternalInvariant, error.what(),
233 kExprMatchOperation);
234 }
235}
