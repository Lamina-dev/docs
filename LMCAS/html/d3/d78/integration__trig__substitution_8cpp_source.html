7/// 检测形如 (c0 + c2*x²)^p 的二次根式幂，其中 p = ±1/2。
8/// 返回 a²=|c0/c2| 信息与符号模式。pattern:
9/// 1 => a²-x² (c2<0, c0>0, 归一化后 a²-x²)
10/// 2 => a²+x² (c2>0, c0>0)
11/// 3 => x²-a² (c2>0, c0<0)
12/// 同时输出 a_sq=|c0|/|c2| 以及 c2 的绝对值（要求 c2=±1 以保持简单）。
13struct QuadRadical {
14 int pattern = 0; // 1,2,3
15 double exponent = 0; // +0.5 or -0.5
16 double a_sq = 0; // a²
17};

19bool trigsub_match_radical(const std::shared_ptr<const SymbolicNode>& node, const std::string& var,
20 QuadRadical& out) {
21 auto pw = std::dynamic_pointer_cast<const PowerNode>(node);
22 if (!pw) return false;
23 auto en = std::dynamic_pointer_cast<const NumberNode>(pw->exponent());
24 if (!en) return false;
25 double e;
26 if (std::holds_alternative<lmmc_real_t>(en->value())) e = std::get<lmmc_real_t>(en->value());
27 else if (std::holds_alternative<Rational>(en->value())) e = std::get<Rational>(en->value()).to_double();
28 else if (std::holds_alternative<BigInt>(en->value())) e = std::get<BigInt>(en->value()).to_double();
29 else return false;
30 if (std::abs(e - 0.5) > 1e-9 && std::abs(e + 0.5) > 1e-9) return false;
31
32 /// base 必须是 c0 + c2*x²（关于 var 的二次、无一次项）
33 auto base = lamina::detail::expression_from_node(pw->base());
34 auto b = base.expand();
35 if (!b) b = lamina::detail::make_expression_ptr(pw->base());
36 /// 提取关于 var 的系数：c0（常数）、c1（一次）、c2（二次）
37 /// 用求导法：c2 = (1/2) d²/dx² ; c1 = d/dx |_{x=0} ; c0 = base|_{x=0}
38 auto d1 = b->differentiate(var);
39 auto d2 = d1->differentiate(var);
40 auto zero = SymbolicExpr::number(0);
41 auto c0e = b->substitute(var, zero)->simplify();
42 auto c1e = d1->substitute(var, zero)->simplify();
43 auto c2e = SymbolicExpr::multiply(SymbolicExpr::number(Rational(1,2)), d2)->simplify();
44 /// 必须 c1=0，且 c2 为非零常数，c0 常数，且 d2 不依赖 var（纯二次）
45 if (!lamina::detail::node(c1e) || !lamina::detail::node(c1e)->is_zero()) return false;
46 if (expression_depends_on_variable(lamina::detail::node(c2e), var)) return false;
47 if (expression_depends_on_variable(lamina::detail::node(c0e), var)) return false;
48 if (!lamina::detail::node(c2e)->is_number() || !lamina::detail::node(c0e)->is_number()) return false;
49 auto c0_checked = try_checked_numeric_constant(*c0e);
50 auto c2_checked = try_checked_numeric_constant(*c2e);
51 if (!c0_checked || !c2_checked) return false;
52 double c0 = *c0_checked;
53 double c2 = *c2_checked;
54 if (std::abs(c2) < 1e-12) return false;
55 /// 仅支持 c2 = ±1（标准型 a²±x² / x²-a²）
56 if (std::abs(std::abs(c2) - 1.0) > 1e-9) return false;
57
58 out.exponent = e;
59 if (c2 < 0 && c0 > 0) { out.pattern = 1; out.a_sq = c0; } // a² - x²
60 else if (c2 > 0 && c0 > 0) { out.pattern = 2; out.a_sq = c0; } // a² + x²
61 else if (c2 > 0 && c0 < 0) { out.pattern = 3; out.a_sq = -c0; } // x² - a²
62 else return false;
63 return true;
64}

68std::shared_ptr<SymbolicExpr> TrigSubstitutionStrategy::try_integrate(
69 const SymbolicExpr& expr, const std::string& var, Integrator&,
70 ComputationContext&, int) {
71 if (!lamina::detail::node(expr)) return nullptr;
72
73 auto x = SymbolicExpr::variable(var);
74
75 /// 仅处理被积函数恰为单个二次根式幂的常见闭式情形：
76 /// ∫ (a²-x²)^(-1/2) dx = arcsin(x/a)
77 /// ∫ (a²+x²)^(-1/2) dx = arcsinh(x/a) = ln(x + √(x²+a²))
78 /// ∫ (x²-a²)^(-1/2) dx = arccosh(x/a) = ln(x + √(x²-a²))
79 /// ∫ (a²-x²)^( 1/2) dx = (x/2)√(a²-x²) + (a²/2)arcsin(x/a)
80 QuadRadical qr;
81 if (!trigsub_match_radical(lamina::detail::node(expr), var, qr)) return nullptr;
82
83 double a_sq = qr.a_sq;
84 auto a_sq_expr = SymbolicExpr::number(a_sq);
85 auto a_expr = SymbolicExpr::sqrt(a_sq_expr); // a = √(a²)
86
87 /// 构造 √(模式) 表达式
88 auto x_sq = SymbolicExpr::power(x, SymbolicExpr::number(2));
89 auto make_radicand = [&]() -> std::shared_ptr<SymbolicExpr> {
90 if (qr.pattern == 1) // a²-x²
91 return SymbolicExpr::add(a_sq_expr, SymbolicExpr::multiply(SymbolicExpr::number(-1), x_sq));
92 if (qr.pattern == 2) // a²+x²
93 return SymbolicExpr::add(a_sq_expr, x_sq);
94 return SymbolicExpr::add(x_sq, SymbolicExpr::multiply(SymbolicExpr::number(-1), a_sq_expr)); // x²-a²
95 };
96 auto x_over_a = SymbolicExpr::divide(x, a_expr);
97
98 auto arcsin = [&](const std::shared_ptr<SymbolicExpr>& u) {
99 return lamina::detail::make_expression_ptr(lamina::detail::make_node<FunctionNode>(
100 FunctionNode::FuncType::ArcSin, std::vector<std::shared_ptr<const SymbolicNode>>{lamina::detail::node(u)}));
101 };
102 auto ln = [&](const std::shared_ptr<SymbolicExpr>& u) { return SymbolicExpr::ln(u); };
103
104 if (qr.exponent < 0) {
105 /// (...)^(-1/2)
106 if (qr.pattern == 1) {
107 /// arcsin(x/a)
108 return arcsin(x_over_a)->simplify();
109 }
110 if (qr.pattern == 2) {
111 /// ln(x + √(x²+a²))
112 auto rad = SymbolicExpr::sqrt(SymbolicExpr::add(x_sq, a_sq_expr));
113 return ln(SymbolicExpr::add(x, rad))->simplify();
114 }
115 /// pattern 3: ln(x + √(x²-a²))
116 auto rad = SymbolicExpr::sqrt(SymbolicExpr::add(x_sq,
117 SymbolicExpr::multiply(SymbolicExpr::number(-1), a_sq_expr)));
118 return ln(SymbolicExpr::add(x, rad))->simplify();
119 } else {
120 /// (...)^(+1/2)
121 auto rad = SymbolicExpr::sqrt(make_radicand());
122 auto half = SymbolicExpr::number(Rational(1, 2));
123 if (qr.pattern == 1) {
124 /// (x/2)√(a²-x²) + (a²/2)arcsin(x/a)
125 auto term1 = SymbolicExpr::multiply(SymbolicExpr::multiply(half, x), rad);
126 auto term2 = SymbolicExpr::multiply(SymbolicExpr::multiply(half, a_sq_expr), arcsin(x_over_a));
127 return SymbolicExpr::add(term1, term2)->simplify();
128 }
129 if (qr.pattern == 2) {
130 /// (x/2)√(a²+x²) + (a²/2)ln(x+√(x²+a²))
131 auto lrad = SymbolicExpr::sqrt(SymbolicExpr::add(x_sq, a_sq_expr));
132 auto term1 = SymbolicExpr::multiply(SymbolicExpr::multiply(half, x), rad);
133 auto term2 = SymbolicExpr::multiply(SymbolicExpr::multiply(half, a_sq_expr),
134 ln(SymbolicExpr::add(x, lrad)));
135 return SymbolicExpr::add(term1, term2)->simplify();
136 }
137 /// pattern 3: (x/2)√(x²-a²) - (a²/2)ln(x+√(x²-a²))
138 auto lrad = SymbolicExpr::sqrt(SymbolicExpr::add(x_sq,
139 SymbolicExpr::multiply(SymbolicExpr::number(-1), a_sq_expr)));
140 auto term1 = SymbolicExpr::multiply(SymbolicExpr::multiply(half, x), rad);
141 auto term2 = SymbolicExpr::multiply(
142 SymbolicExpr::multiply(SymbolicExpr::multiply(SymbolicExpr::number(-1), half), a_sq_expr),
143 ln(SymbolicExpr::add(x, lrad)));
144 return SymbolicExpr::add(term1, term2)->simplify();
145 }
146}
