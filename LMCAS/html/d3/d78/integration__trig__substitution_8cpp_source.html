7/// 检测形如 (c0 + c2*x²)^p 的二次根式幂，其中 p = ±1/2。
8/// 返回 a²=|c0/c2| 信息与符号模式。pattern:
9/// 1 => a²-x² (c2<0, c0>0, 归一化后 a²-x²)
10/// 2 => a²+x² (c2>0, c0>0)
11/// 3 => x²-a² (c2>0, c0<0)
12/// 同时输出 a_sq=|c0|/|c2| 以及 c2 的绝对值（要求 c2=±1 以保持简单）。
13struct QuadRadical {
14 int pattern = 0; // 1,2,3
15 double exponent = 0; // +0.5 or -0.5
16 double a_sq = 0; // a²
17};

19bool trigsub_half_exponent(const NumberNode& en, double& e) {
20 if (std::holds_alternative<lmmc_real_t>(en.value())) e = std::get<lmmc_real_t>(en.value());
21 else if (std::holds_alternative<Rational>(en.value())) e = std::get<Rational>(en.value()).to_double();
22 else if (std::holds_alternative<BigInt>(en.value())) e = std::get<BigInt>(en.value()).to_double();
23 else return false;
24 if (std::abs(e - 0.5) > 1e-9 && std::abs(e + 0.5) > 1e-9) { return false; }
25 return true;
26}

28bool trigsub_quadratic_coefficients(const PowerNode& pw, const std::string& var,
29 ComputationContext& context, double& c0, double& c2) {
30 /// base 必须是 c0 + c2*x²（关于 var 的二次、无一次项）
31 auto base = LMCAS::detail::expression_from_node(pw.base());
32 auto b = base.expand();
33 if (!b) b = LMCAS::detail::make_expression_ptr(pw.base());
34 /// 提取关于 var 的系数：c0（常数）、c1（一次）、c2（二次）
35 /// 用求导法：c2 = (1/2) d²/dx² ; c1 = d/dx |_{x=0} ; c0 = base|_{x=0}
36 auto d1 = b->differentiate(var);
37 auto d2 = d1->differentiate(var);
38 auto zero = SymbolicExpr::number(0);
39 auto c0e = b->substitute(var, zero)->simplify();
40 auto c1e = d1->substitute(var, zero)->simplify();
41 auto c2e = SymbolicExpr::multiply(SymbolicExpr::number(Rational(1,2)), d2)->simplify();
42 /// 必须 c1=0，且 c2 为非零常数，c0 常数，且 d2 不依赖 var（纯二次）
43 if (!LMCAS::detail::node(c1e) || !LMCAS::detail::node(c1e)->is_zero()) { return false; }
44 if (expression_depends_on_variable(LMCAS::detail::node(c2e), var)) { return false; }
45 if (expression_depends_on_variable(LMCAS::detail::node(c0e), var)) { return false; }
46 if (!LMCAS::detail::node(c2e)->is_number() || !LMCAS::detail::node(c0e)->is_number()) { return false; }
47 auto c0_checked = try_checked_numeric_constant(*c0e, context);
48 auto c2_checked = try_checked_numeric_constant(*c2e, context);
49 if (!c0_checked || !c2_checked) { return false; }
50 c0 = *c0_checked;
51 c2 = *c2_checked;
52 if (std::abs(c2) < 1e-12) { return false; }
53 /// 仅支持 c2 = ±1（标准型 a²±x² / x²-a²）
54 if (std::abs(std::abs(c2) - 1.0) > 1e-9) { return false; }
55 return true;
56}

58bool trigsub_match_radical(const std::shared_ptr<const SymbolicNode>& node,
59 const std::string& var,
60 ComputationContext& context,
61 QuadRadical& out) {
62 auto pw = std::dynamic_pointer_cast<const PowerNode>(node);
63 if (!pw) { return false; }
64 auto en = std::dynamic_pointer_cast<const NumberNode>(pw->exponent());
65 if (!en) { return false; }
66 double e;
67 if (!trigsub_half_exponent(*en, e)) { return false; }
68 double c0, c2;
69 if (!trigsub_quadratic_coefficients(*pw, var, context, c0, c2)) { return false; }
70 out.exponent = e;
71 if (c2 < 0 && c0 > 0) { out.pattern = 1; out.a_sq = c0; } // a² - x²
72 else if (c2 > 0 && c0 > 0) { out.pattern = 2; out.a_sq = c0; } // a² + x²
73 else if (c2 > 0 && c0 < 0) { out.pattern = 3; out.a_sq = -c0; } // x² - a²
74 else return false;
75 return true;
76}

80Result<std::shared_ptr<SymbolicExpr>> TrigSubstitutionStrategy::try_integrate_raw(
81 const SymbolicExpr& expr, const std::string& var, Integrator&,
82 ComputationContext& context, int) {
83 if (!LMCAS::detail::node(expr)) { return nullptr; }
84
85 auto x = SymbolicExpr::variable(var);
86
87 /// 仅处理被积函数恰为单个二次根式幂的常见闭式情形：
88 /// ∫ (a²-x²)^(-1/2) dx = arcsin(x/a)
89 /// ∫ (a²+x²)^(-1/2) dx = arcsinh(x/a) = ln(x + √(x²+a²))
90 /// ∫ (x²-a²)^(-1/2) dx = arccosh(x/a) = ln(x + √(x²-a²))
91 /// ∫ (a²-x²)^( 1/2) dx = (x/2)√(a²-x²) + (a²/2)arcsin(x/a)
92 QuadRadical qr;
93 if (!trigsub_match_radical(
94 LMCAS::detail::node(expr), var, context, qr)) {
95 return nullptr;
96 }
97
98 double a_sq = qr.a_sq;
99 auto a_sq_expr = SymbolicExpr::number(a_sq);
100 auto a_expr = SymbolicExpr::sqrt(a_sq_expr); // a = √(a²)
101
102 /// 构造 √(模式) 表达式
103 auto x_sq = SymbolicExpr::power(x, SymbolicExpr::number(2));
104 auto make_radicand = [&]() -> std::shared_ptr<SymbolicExpr> {
105 if (qr.pattern == 1) // a²-x²
106 return SymbolicExpr::add(a_sq_expr, SymbolicExpr::multiply(SymbolicExpr::number(-1), x_sq));
107 if (qr.pattern == 2) // a²+x²
108 return SymbolicExpr::add(a_sq_expr, x_sq);
109 return SymbolicExpr::add(x_sq, SymbolicExpr::multiply(SymbolicExpr::number(-1), a_sq_expr)); // x²-a²
110 };
111 auto x_over_a = SymbolicExpr::divide(x, a_expr);
112
113 auto arcsin = [&](const std::shared_ptr<SymbolicExpr>& u) {
114 return LMCAS::detail::make_expression_ptr(LMCAS::detail::make_node<FunctionNode>(
115 FunctionNode::FuncType::ArcSin, std::vector<std::shared_ptr<const SymbolicNode>>{LMCAS::detail::node(u)}));
116 };
117 auto ln = [&](const std::shared_ptr<SymbolicExpr>& u) { return SymbolicExpr::ln(u); };
118
119 if (qr.exponent < 0) {
120 /// (...)^(-1/2)
121 if (qr.pattern == 1) {
122 /// arcsin(x/a)
123 return arcsin(x_over_a)->simplify();
124 }
125 if (qr.pattern == 2) {
126 /// ln(x + √(x²+a²))
127 auto rad = SymbolicExpr::sqrt(SymbolicExpr::add(x_sq, a_sq_expr));
128 return ln(SymbolicExpr::add(x, rad))->simplify();
129 }
130 /// pattern 3: ln(x + √(x²-a²))
131 auto rad = SymbolicExpr::sqrt(SymbolicExpr::add(x_sq,
132 SymbolicExpr::multiply(SymbolicExpr::number(-1), a_sq_expr)));
133 return ln(SymbolicExpr::add(x, rad))->simplify();
134 } else {
135 /// (...)^(+1/2)
136 auto rad = SymbolicExpr::sqrt(make_radicand());
137 auto half = SymbolicExpr::number(Rational(1, 2));
138 if (qr.pattern == 1) {
139 /// (x/2)√(a²-x²) + (a²/2)arcsin(x/a)
140 auto term1 = SymbolicExpr::multiply(SymbolicExpr::multiply(half, x), rad);
141 auto term2 = SymbolicExpr::multiply(SymbolicExpr::multiply(half, a_sq_expr), arcsin(x_over_a));
142 return SymbolicExpr::add(term1, term2)->simplify();
143 }
144 if (qr.pattern == 2) {
145 /// (x/2)√(a²+x²) + (a²/2)ln(x+√(x²+a²))
146 auto lrad = SymbolicExpr::sqrt(SymbolicExpr::add(x_sq, a_sq_expr));
147 auto term1 = SymbolicExpr::multiply(SymbolicExpr::multiply(half, x), rad);
148 auto term2 = SymbolicExpr::multiply(SymbolicExpr::multiply(half, a_sq_expr),
149 ln(SymbolicExpr::add(x, lrad)));
150 return SymbolicExpr::add(term1, term2)->simplify();
151 }
152 /// pattern 3: (x/2)√(x²-a²) - (a²/2)ln(x+√(x²-a²))
153 auto lrad = SymbolicExpr::sqrt(SymbolicExpr::add(x_sq,
154 SymbolicExpr::multiply(SymbolicExpr::number(-1), a_sq_expr)));
155 auto term1 = SymbolicExpr::multiply(SymbolicExpr::multiply(half, x), rad);
156 auto term2 = SymbolicExpr::multiply(
157 SymbolicExpr::multiply(SymbolicExpr::multiply(SymbolicExpr::number(-1), half), a_sq_expr),
158 ln(SymbolicExpr::add(x, lrad)));
159 return SymbolicExpr::add(term1, term2)->simplify();
160 }
