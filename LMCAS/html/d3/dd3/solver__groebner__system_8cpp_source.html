21std::shared_ptr<SymbolicExpr> clear_denominators(
22 const SymbolicExpr& equation, Expressions& constraints) {
23 if (!detail::node(equation)) return nullptr;
24 std::vector<std::shared_ptr<const SymbolicNode>> factors;
25 Expressions local_constraints;
26 if (!collect_denominator_factors(detail::node(equation), factors, local_constraints)) {
27 return nullptr;
28 }
29 auto cleared = to_ptr(equation);
30 if (factors.empty()) {
31 cleared = cleared->simplify();
32 } else if (auto sum = std::dynamic_pointer_cast<const AddNode>(detail::node(cleared))) {
33 std::vector<std::shared_ptr<const SymbolicNode>> terms;
34 terms.reserve(sum->operands().size());
35 for (const auto& term : sum->operands()) {
36 auto product = multiply_no_expand(term, factors);
37 terms.push_back(detail::node(product));
38 }
39 cleared = detail::make_expression_ptr(detail::make_node<AddNode>(terms));
40 } else {
41 cleared = multiply_no_expand(detail::node(cleared), factors);
42 }
43 if (!cleared || !detail::node(cleared) || !is_polynomial_node(detail::node(cleared))) {
44 return nullptr;
45 }
46 for (const auto& constraint : local_constraints) constraints.push_back(constraint);
47 return cleared;
48}

60PolynomialSystemResult filter_denominators(PolynomialSolutions candidates,
61 const Expressions& constraints, ComputationContext& context) {
62 std::optional<detail::AssumptionFacts> adapter;
63 if (context.assumptions()) adapter.emplace(*context.assumptions());
64 const FactsQuery& facts = adapter ? static_cast<const FactsQuery&>(*adapter) : detail::no_facts();
65 PolynomialSolutions filtered;
66 filtered.reserve(candidates.size());
67 for (auto& solution : candidates) {
68 bool valid = true;
69 for (const auto& denominator : constraints) {
70 auto bound = substitute_solution(denominator, solution);
71 if (!bound) return PolynomialSystemResult::failure(CasErrc::InternalInvariant,
72 "denominator substitution failed", "solve_polynomial_system");
73 auto nonzero = detail::query_nonzero_value(detail::node(bound), facts, Domain::Complex, context);
74 if (!nonzero) return PolynomialSystemResult::failure(nonzero.error());
75 if (nonzero.value() == Tribool::False) { valid = false; break; }
76 if (nonzero.value() != Tribool::True)
77 return PolynomialSystemResult::failure(CasErrc::Inconclusive,
78 "candidate denominator is not proved nonzero", "solve_polynomial_system");
79 }
80 if (valid) filtered.push_back(std::move(solution));
81 }
82 return filtered;
83}

96Result<void> validate_basis_input(const std::vector<SymbolicExpr>& equations,
97 const std::vector<std::string>& variables,
98 groebner_detail::PolyContext& conversion,
99 ComputationContext& context) {
100 constexpr const char* operation = "solve_polynomial_system";
101 for (const auto& equation : equations) {
102 auto budget = context.consume_steps(1, operation);
103 if (!budget) return Result<void>::failure(budget.error());
104 groebner_detail::PolyBuilder builder(variables, conversion, true);
105 detail::node(equation)->accept(builder);
106 if (builder.failed) return Result<void>::failure(
107 CasErrc::Inconclusive, "equation is not an exact polynomial in the system variables", operation);
108 }
109 return Result<void>::success();

129Result<Expressions> simplified_basis(const std::vector<SymbolicExpr>& equations,
130 const std::vector<std::string>& variables,
131 ComputationContext& context) {
132 constexpr const char* operation = "solve_polynomial_system";
133 groebner_detail::PolyContext conversion(variables);
134 auto validated = validate_basis_input(equations, variables, conversion, context);
135 if (!validated) return Result<Expressions>::failure(validated.error());
136 auto groebner = Solver::reduced_groebner_basis(equations, variables);
137 Expressions basis;
138 std::vector<FGLMPoly> leading;
139 basis.reserve(groebner.size());
140 leading.reserve(groebner.size());
141 for (const auto& polynomial : groebner) {
142 std::shared_ptr<SymbolicExpr> simplified;
143 auto converted = convert_basis_polynomial(polynomial, variables, conversion, context, simplified);
144 if (!converted) return Result<Expressions>::failure(converted.error());
145 if (simplified->is_number() && !simplified->is_zero()) return Expressions{simplified};
146 if (converted.value().is_zero()) continue;
147 FGLMPoly term(variables.size());
148 term.add_term(converted.value().lead_monomial(), converted.value().lead_coeff());
149 leading.push_back(std::move(term));
150 basis.push_back(std::move(simplified));
151 }
152 if (!is_zero_dimensional(leading, variables.size())) {
153 return Result<Expressions>::failure(CasErrc::Inconclusive,
154 "polynomial system has no certified finite solution set", operation);
155 }
156 return basis;

187Result<std::optional<Expressions>> PolynomialSystemSearch::substitute_basis(
188 int variable, const PolynomialSolution& partial) const {
189 Expressions reduced;
190 reduced.reserve(basis_.size());
191 for (const auto& polynomial : basis_) {
192 auto substituted = substitute_solution(polynomial, partial);
193 if (!substituted) { return Result<std::optional<Expressions>>::failure(
194 CasErrc::InternalInvariant, "basis substitution failed", "solve_polynomial_system"); }
195 if (substituted->is_zero()) { continue; }
196 if (!depends_on_remaining(*substituted, variable)) {
197 auto proof = check_zero_residual(substituted, context_);
198 if (!proof) { return Result<std::optional<Expressions>>::failure(proof.error()); }
199 if (std::holds_alternative<ProvedZeroResidual>(proof.value())) { continue; }
200 std::optional<detail::AssumptionFacts> adapter;
201 if (context_.assumptions()) { adapter.emplace(*context_.assumptions()); }
202 const FactsQuery& facts = adapter ? static_cast<const FactsQuery&>(*adapter) : detail::no_facts();
203 auto nonzero = detail::query_nonzero_value(detail::node(substituted), facts, Domain::Complex, context_);
204 if (!nonzero) { return Result<std::optional<Expressions>>::failure(nonzero.error()); }
205 if (nonzero.value() == Tribool::True) { return std::optional<Expressions>{}; }
206 return Result<std::optional<Expressions>>::failure(CasErrc::Inconclusive,
207 "remaining system constraint is not proved", "solve_polynomial_system");
208 }
209 reduced.push_back(substituted);
210 }
211 return std::optional<Expressions>{std::move(reduced)};
212}

214Result<std::shared_ptr<SymbolicExpr>> PolynomialSystemSearch::select_equation(
215 const Expressions& reduced, int variable, bool& appears) const {
216 std::shared_ptr<SymbolicExpr> target;
217 int best_degree = std::numeric_limits<int>::max();
218 const auto& name = variables_[variable];
219 for (const auto& polynomial : reduced) {
220 if (!contains(*polynomial, name)) continue;
221 appears = true;
222 if (depends_on_remaining(*polynomial, variable - 1)) continue;
223 auto univariate = symbolic_to_poly<SymbolicPolyCoeff>(polynomial, name);
224 if (!univariate) {
225 if (univariate.error().code != CasErrc::UnsupportedExpression) {
226 return Result<std::shared_ptr<SymbolicExpr>>::failure(univariate.error());
227 }
228 continue;
229 }
230 int degree = univariate.value().degree();
231 if (degree >= 1 && degree < best_degree) {
232 best_degree = degree;
233 target = polynomial;
234 }
235 }
236 return target;
237}

239PolynomialSystemResult PolynomialSystemSearch::solve(
240 int variable, const PolynomialSolution& partial) const {
241 auto reduced = substitute_basis(variable, partial);
242 if (!reduced) return PolynomialSystemResult::failure(reduced.error());
243 if (!reduced.value()) return PolynomialSolutions{};
244 if (variable < 0) return PolynomialSolutions{partial};
245 const auto& name = variables_[variable];
246 bool appears = false;
247 auto target = select_equation(*reduced.value(), variable, appears);
248 if (!target) return PolynomialSystemResult::failure(target.error());
249 if (!appears) {
250 return PolynomialSystemResult::failure(CasErrc::Inconclusive,
251 "remaining variable is not constrained", "solve_polynomial_system");
252 }
253 if (!target.value()) {
254 return PolynomialSystemResult::failure(CasErrc::Inconclusive,
255 "No convertible polynomial target for remaining variable", "solve_polynomial_system");
256 }
257 auto solved = solve_finite_checked(target.value(), name, context_, SolveOptions{});
258 if (!solved) return PolynomialSystemResult::failure(solved.error());
259 PolynomialSolutions results;
260 for (const auto& root : solved.value()) {
261 auto next = partial;
262 next.insert_or_assign(name, *root);
263 auto solved_branch = solve(variable - 1, next);
264 if (!solved_branch) return solved_branch;
265 auto& values = solved_branch.value();
266 results.insert(results.end(), std::make_move_iterator(values.begin()),
267 std::make_move_iterator(values.end()));
268 }
269 return results;
270}

272PolynomialSystemResult solve_polynomial_system_impl(
273 const std::vector<SymbolicExpr>& equations,
274 const std::vector<std::string>& variables, ComputationContext& context) {
275 std::vector<SymbolicExpr> cleared_equations;
276 Expressions constraints;
277 cleared_equations.reserve(equations.size());
278 for (const auto& equation : equations) {
279 auto cleared = clear_denominators(equation, constraints);
280 if (!cleared) return PolynomialSystemResult::failure(CasErrc::Inconclusive,
281 "equation cannot be converted to a polynomial without losing its domain", "solve_polynomial_system");
282 cleared_equations.push_back(*cleared);
283 }
284 if (cleared_equations.size() == 1 && variables.size() == 1) {
285 auto solved = solve_single_equation(cleared_equations[0], variables[0], context);
286 if (!solved) return solved;
287 return filter_denominators(std::move(solved.value()), constraints, context);
288 }
289 auto basis = simplified_basis(cleared_equations, variables, context);
290 if (!basis) return PolynomialSystemResult::failure(basis.error());
291 if (basis.value().size() == 1 && basis.value().front()->is_number() &&
292 !basis.value().front()->is_zero()) return PolynomialSolutions{};
293 PolynomialSystemSearch search(basis.value(), variables, context);
294 auto solved = search.solve(static_cast<int>(variables.size()) - 1, PolynomialSolution{});
295 if (!solved) return solved;
296 return filter_denominators(std::move(solved.value()), constraints, context);
297}

301 const std::vector<SymbolicExpr>& equations,
302 const std::vector<std::string>& variables, ComputationContext& context) {
303 constexpr const char* operation = "solve_polynomial_system";
304 if (equations.empty() || variables.empty()) {

306 CasErrc::InvalidArgument,
307 "polynomial system requires equations and variables", operation);
308 }

315 CasErrc::ResourceLimit,

317 } catch (const std::exception& error) {
318 return PolynomialSystemResult::failure(CasErrc::InternalInvariant, error.what(), operation);
319 }
