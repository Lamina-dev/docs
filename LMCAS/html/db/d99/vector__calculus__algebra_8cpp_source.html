20VectorCalculusExprResult dot_product(const VectorField& a, const VectorField& b)
21{
22 if (a.size() != b.size()) {
23 return VectorCalculusExprResult::failure(
24 CasErrc::DimensionMismatch,
25 "vectors must have the same dimension",
26 "dot_product");
27 }
28 std::shared_ptr<SymbolicExpr> sum = SymbolicExpr::number(0);
29 for (size_t i = 0; i < a.size(); ++i) {
30 if (!a[i] || !b[i]) continue;
31 sum = SymbolicExpr::add(sum, SymbolicExpr::multiply(a[i], b[i]));
32 }
33 return VectorCalculusExprResult::success(sum->simplify());
34}

37{
38 if (a.size() != 3 || b.size() != 3) {

44 auto sub = [](const std::shared_ptr<SymbolicExpr>& p,
45 const std::shared_ptr<SymbolicExpr>& q) {

55VectorCalculusFieldResult vector_project(const VectorField& a, const VectorField& b)
56{
57 if (a.size() != b.size()) {
58 return VectorCalculusFieldResult::failure(
59 CasErrc::DimensionMismatch,
60 "vectors must have the same dimension",
61 "vector_project");
62 }
63 auto bb_result = dot_product(b, b);
64 if (!bb_result) return VectorCalculusFieldResult::failure(bb_result.error());
65 auto bb = bb_result.value();
66 if (LMCAS::detail::node(bb) && LMCAS::detail::node(bb)->is_zero()) {
67 return VectorCalculusFieldResult::success(
68 VectorField(a.size(), SymbolicExpr::number(0)));
69 }
70 auto ab_result = dot_product(a, b);
71 if (!ab_result) return VectorCalculusFieldResult::failure(ab_result.error());
72 auto ab = ab_result.value();
73 auto coeff = SymbolicExpr::divide(ab, bb);
74 VectorField result;
75 result.reserve(b.size());
76 for (const auto& comp : b) {
77 result.push_back(SymbolicExpr::multiply(coeff, comp)->simplify());
78 }
79 return VectorCalculusFieldResult::success(std::move(result));
80}

97{
98 auto aa_result = dot_product(a, a);

103 auto bb = bb_result.value();
104 if ((LMCAS::detail::node(aa) && LMCAS::detail::node(aa)->is_zero()) ||
105 (LMCAS::detail::node(bb) && LMCAS::detail::node(bb)->is_zero())) {
106 return VectorCalculusExprResult::success(nullptr);

108 auto ab_result = dot_product(a, b);
109 if (!ab_result) return ab_result;

112 auto arccos_node = LMCAS::detail::make_node<FunctionNode>(

115 return VectorCalculusExprResult::success(
