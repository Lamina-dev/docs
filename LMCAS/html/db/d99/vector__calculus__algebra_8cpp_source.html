20std::shared_ptr<SymbolicExpr> dot_product(const VectorField& a, const VectorField& b)
21{
22 if (a.size() != b.size()) {
23 throw std::invalid_argument("dot_product: vectors must have the same dimension");
24 }
25 std::shared_ptr<SymbolicExpr> sum = SymbolicExpr::number(0);
26 for (size_t i = 0; i < a.size(); ++i) {
27 if (!a[i] || !b[i]) continue;
28 sum = SymbolicExpr::add(sum, SymbolicExpr::multiply(a[i], b[i]));
29 }
30 return sum->simplify();
31}

49VectorField vector_project(const VectorField& a, const VectorField& b)
50{
51 if (a.size() != b.size()) {
52 throw std::invalid_argument("vector_project: vectors must have the same dimension");
53 }
54 auto bb = dot_product(b, b);
55 if (lamina::detail::node(bb) && lamina::detail::node(bb)->is_zero()) {
56 return VectorField(a.size(), SymbolicExpr::number(0));
57 }
58 auto ab = dot_product(a, b);
59 auto coeff = SymbolicExpr::divide(ab, bb);
60 VectorField result;
61 result.reserve(b.size());
62 for (const auto& comp : b) {
63 result.push_back(SymbolicExpr::multiply(coeff, comp)->simplify());
64 }
65 return result;
66}
