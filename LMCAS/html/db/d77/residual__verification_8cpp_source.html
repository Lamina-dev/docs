19ResidualCheckResult classify_residual(
20 const ExprPtr& expression,
21 ComputationContext& context,
22 const std::string& operation,
23 bool final_stage) {
24 auto proof = detail::classify_exact_zero(
25 expression, context, operation);
26 if (!proof) return ResidualCheckResult::failure(proof.error());
27 if (proof.value() == detail::ZeroProof::Zero) {
28 return ResidualCheckResult::success(ProvedZeroResidual{
29 ExactNormalizationProof{expression}});
30 }
31 const auto variables = free_variables(detail::node(expression));
32 if (variables.size() == 1) {
33 auto polynomial = recognize_rational_polynomial(
34 *expression, *variables.begin(), context);
35 if (!polynomial) {
36 return ResidualCheckResult::failure(polynomial.error());
37 }
38 if (polynomial.value() && !polynomial.value()->is_zero()) {
39 return ResidualCheckResult::success(
40 ProvedNonzeroResidual{expression});
41 }
42 }
43 if (proof.value() == detail::ZeroProof::NonZero) {
44 return ResidualCheckResult::success(
45 ProvedNonzeroResidual{expression});
46 }
47 if (final_stage) {
48 return ResidualCheckResult::success(UnprovedResidual{expression});
49 }
50 return ResidualCheckResult::success(UnprovedResidual{expression});
51}

59ResidualCheckResult check_zero_residual(
60 const ExprPtr& residual,
61 ComputationContext& context,
62 const lsr::EqvOptions& options) {
63 constexpr const char* operation = "residual.check_zero";
64 if (!residual || !detail::node(residual)) {
65 return ResidualCheckResult::failure(
66 CasErrc::InvalidArgument,
67 "residual expression cannot be null", operation);
68 }
69 auto access = context.consume_steps(1, operation);
70 if (!access) return ResidualCheckResult::failure(access.error());
71
72 try {
73 auto current = normalized(residual);
74 auto first = classify_residual(current, context, operation, false);
75 if (!first || is_decided(first.value())) return first;
76
77 current = normalized(current->expand());
78 auto expanded = classify_residual(current, context, operation, false);
79 if (!expanded || is_decided(expanded.value())) return expanded;
80
81 current = normalized(current->cancel());
82 auto cancelled = classify_residual(current, context, operation, false);
83 if (!cancelled || is_decided(cancelled.value())) return cancelled;
84
85 current = normalized(current->expand());
86 current = normalized(current->cancel());
87 auto combined = classify_residual(current, context, operation, false);
88 if (!combined || is_decided(combined.value())) return combined;
89
90 const auto zero = SymbolicExpr::number(0);
91 for (const auto profile : {
92 lsr::EqvProfile::Core,
93 lsr::EqvProfile::TrigBasic,
94 lsr::EqvProfile::ExpLogBasic}) {
95 auto profile_options = options;
96 profile_options.profile = profile;
97 auto equivalent = lsr::equivalent_core(
98 *current, *zero, context, profile_options);
99 if (!equivalent) {
100 if (equivalent.error().code == CasErrc::UnsupportedExpression ||
101 equivalent.error().code == CasErrc::Inconclusive) {
102 continue;
103 }
104 return ResidualCheckResult::failure(equivalent.error());
105 }
106 if (equivalent.value()) {
107 return ResidualCheckResult::success(ProvedZeroResidual{
108 ExactResidualProof{current}});
109 }
110 }
111 return ResidualCheckResult::success(UnprovedResidual{current});
112 } catch (const std::bad_alloc&) {
113 return ResidualCheckResult::failure(
114 CasErrc::ResourceLimit,
115 "residual verification allocation failed", operation);
116 } catch (const std::exception& error) {
117 return ResidualCheckResult::failure(
118 CasErrc::InternalInvariant, error.what(), operation);
119 }
120}

122ResidualCheckResult check_equivalent(
123 const ExprPtr& left,
124 const ExprPtr& right,
125 ComputationContext& context,
126 const lsr::EqvOptions& options) {
127 if (!left || !right || !detail::node(left) || !detail::node(right)) {
128 return ResidualCheckResult::failure(
129 CasErrc::InvalidArgument,
130 "equivalence check requires two expressions",
131 "residual.check_equivalent");
132 }
133 try {
134 auto residual = SymbolicExpr::add(
135 left,
136 SymbolicExpr::multiply(SymbolicExpr::number(-1), right));
137 return check_zero_residual(residual, context, options);
138 } catch (const std::bad_alloc&) {
139 return ResidualCheckResult::failure(
140 CasErrc::ResourceLimit,
141 "equivalence residual allocation failed",
142 "residual.check_equivalent");
143 } catch (const std::exception& error) {
144 return ResidualCheckResult::failure(
145 CasErrc::InternalInvariant, error.what(),
146 "residual.check_equivalent");
147 }
