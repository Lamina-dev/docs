85 return false;

139 bool register_atom(const Node& node, bool canonicalize_radicals) {
140 if (!dynamic_cast<const VariableNode*>(node.get()) &&
141 !dynamic_cast<const FunctionNode*>(node.get()) &&
142 !dynamic_cast<const PowerNode*>(node.get()) &&
143 !dynamic_cast<const RootOfNode*>(node.get())) { return false; }
144 for (const auto& atom : atoms_) { if (atom->equals(*node)) { return true; } }
145 for (const auto& radical : canonical_radicals_)

149 std::optional<bool> register_canonical_radical(
150 const Node& node, const Rational& square) {
151 const auto& numerator = square.get_numerator();
152 const auto& denominator = square.get_denominator();
153 if (!checked(context_.require_integer_bits(
154 numerator.bit_length() + denominator.bit_length() + 2,
155 "residual.formal"))) {
156 return false;
157 }
158 auto root = Irrational::sqrt_checked(
159 numerator * denominator, 1.0, context_);
160 if (!root) {
161 failure_ = root.error();
162 return false;
163 }
164 if (!checked(context_.reserve_nodes(5, "residual.formal"))) {
165 return false;
166 }
167 auto canonical = detail::node(denominator == BigInt(1)
168 ? root.value().to_symbolic()
169 : (root.value() * Rational(BigInt(1), denominator)).to_symbolic());
170 if (canonical->equals(*node)) {
171 return std::nullopt;
172 }
173 if (!collect(canonical, false) ||
174 !checked(context_.require_expansion_terms(
175 atoms_.size() + canonical_radicals_.size() + 1,
176 "residual.formal"))) {
177 return false;
178 }
179 canonical_radicals_.emplace_back(node, std::move(canonical));
180 return true;

216 std::optional<MultiPoly> multiply(const MultiPoly& a, const MultiPoly& b) {
217 const auto count = static_cast<std::size_t>(a.num_terms()) *
218 static_cast<std::size_t>(b.num_terms());
219 if (!checked(context_.require_expansion_terms(count, "residual.formal")) ||
220 !checked(context_.consume_steps(count + 1, "residual.formal"))) return std::nullopt;
221 std::size_t a_bits = 0, b_bits = 0;
222 std::int64_t a_degree = 0, b_degree = 0;
223 for (const auto& term : a.terms()) {
224 a_bits = std::max(a_bits, term.second.get_numerator().bit_length() +
225 term.second.get_denominator().bit_length());
226 std::int64_t degree = 0;
227 for (int exponent : term.first) degree += exponent;
228 a_degree = std::max(a_degree, degree);
229 }
230 for (const auto& term : b.terms()) {
231 b_bits = std::max(b_bits, term.second.get_numerator().bit_length() +
232 term.second.get_denominator().bit_length());
233 std::int64_t degree = 0;
234 for (int exponent : term.first) degree += exponent;
235 b_degree = std::max(b_degree, degree);
236 }
237 if (a_degree + b_degree > std::numeric_limits<int>::max()) {
238 checked(Result<void>::failure(CasErrc::ResourceLimit,
239 "formal polynomial degree exceeds its representation", "residual.formal"));
240 return std::nullopt;
241 }
242 if (!checked(context_.require_integer_bits(
243 (a_bits + b_bits + 1) * (count + 1), "residual.formal"))) return std::nullopt;
244 return a * b;
245 }
246 std::optional<Fraction> combine(Fraction a, Fraction b, bool sum) {
247 auto denominator = multiply(a.denominator, b.denominator);
248 if (!denominator) return std::nullopt;
249 if (!sum) {
250 auto numerator = multiply(a.numerator, b.numerator);
251 if (!numerator) return std::nullopt;
252 return Fraction{std::move(*numerator), std::move(*denominator)};
253 }
254 auto left = multiply(a.numerator, b.denominator);
255 auto right = multiply(b.numerator, a.denominator);
256 if (!left || !right) return std::nullopt;
257 if (!checked(context_.require_expansion_terms(
258 left->num_terms() + right->num_terms(), "residual.formal"))) return std::nullopt;
259 std::size_t left_bits = 0, right_bits = 0;
260 for (const auto& term : left->terms())
261 left_bits = std::max(left_bits, term.second.get_numerator().bit_length() +
262 term.second.get_denominator().bit_length());
263 for (const auto& term : right->terms())
264 right_bits = std::max(right_bits, term.second.get_numerator().bit_length() +
265 term.second.get_denominator().bit_length());
266 if (!checked(context_.require_integer_bits(
267 left_bits + right_bits + 1, "residual.formal"))) return std::nullopt;
268 return Fraction{*left + *right, std::move(*denominator)};
269 }
270 std::optional<Fraction> convert_integer_power(const PowerNode& power, std::int64_t n) {
271 auto base = convert(power.base());
272 if (!base) { return std::nullopt; }
273 if (n < 0) { std::swap(base->numerator, base->denominator); n = -n; }
274 std::optional<Fraction> result = Fraction{constant(Rational(1)), constant(Rational(1))};
275 while (n) {
276 if (n & 1) {
277 result = combine(std::move(*result), *base, false);
278 if (!result) { return std::nullopt; }
279 }
280 n >>= 1;
281 if (n) {
282 base = combine(*base, *base, false);
283 if (!base) { return std::nullopt; }
284 }
285 }
286 return result;
287 }

296 std::optional<Fraction> convert_operands(const std::vector<Node>& operands, bool sum) {
297 std::optional<Fraction> result = Fraction{
298 constant(Rational(sum ? 0 : 1)), constant(Rational(1))};
299 for (const auto& child : operands) {
300 auto operand = convert(child);

307 std::optional<Fraction> convert(const Node& node) {
308 if (!checked(context_.enter_recursion("residual.formal"))) { return std::nullopt; }
309 ResidualRecursionScope scope(context_);
310 if (dynamic_cast<const NumberNode*>(node.get())) {
311 return convert_number(node);
312 }
313 for (const auto& radical : canonical_radicals_) {
314 if (radical.first->equals(*node)) {
315 return convert(radical.second);
316 }
317 }
318 for (std::size_t i = 0; i < atoms_.size(); ++i) {
319 if (!atoms_[i]->equals(*node)) { continue; }
320 Monomial powers(atoms_.size(), 0);
321 powers[i] = 1;
322 return Fraction{
323 MultiPoly({{std::move(powers), Rational(1)}}, variables_),
324 constant(Rational(1))};
325 }
326 auto sum = dynamic_cast<const AddNode*>(node.get());
327 auto product = dynamic_cast<const MultiplyNode*>(node.get());
328 if (sum || product) {
329 return convert_operands(
330 sum ? sum->operands() : product->operands(), sum != nullptr);
331 }
332 if (auto power = dynamic_cast<const PowerNode*>(node.get())) {
333 if (auto exponent = integer_power(*power)) {
334 return convert_integer_power(*power, *exponent);
335 }
336 }
337 return std::nullopt;
338 }
339 std::optional<MultiPoly> reduce_radicals(const MultiPoly& polynomial) {
340 MultiPoly reduced(Rational(0), variables_);
341 for (auto term : polynomial.terms()) {
342 std::vector<std::pair<Node, int>> symbolic_squares;
343 for (std::size_t i = 0; i < radical_squares_.size(); ++i) {
344 if (term.first[i] < 2) { continue; }
345 const auto exponent = term.first[i] / 2;
346 if (radical_squares_[i]) {
347 const auto& square = *radical_squares_[i];
348 auto bits = term.second.get_numerator().bit_length() +
349 term.second.get_denominator().bit_length() +
350 static_cast<std::size_t>(exponent) *
351 (square.get_numerator().bit_length() +
352 square.get_denominator().bit_length());
353 if (!checked(context_.require_integer_bits(
354 bits, "residual.formal"))) {
355 return std::nullopt;
356 }
357 term.second =
358 term.second * square.power(BigInt(exponent));
359 } else if (radical_radicands_[i]) {
360 symbolic_squares.emplace_back(
361 radical_radicands_[i], exponent);
362 } else {
363 continue;
364 }
365 if (!checked(context_.consume_steps(
366 static_cast<std::size_t>(exponent) + 1,
367 "residual.formal"))) {
368 return std::nullopt;
369 }
370 term.first[i] %= 2;
371 }
372 MultiPoly expanded({term}, variables_);
373 for (const auto& [radicand, exponent] : symbolic_squares) {
374 auto square = convert(radicand);
375 if (!square ||
376 square->denominator != constant(Rational(1))) {
377 return std::nullopt;
378 }
379 for (int power = 0; power < exponent; ++power) {
380 auto product = multiply(expanded, square->numerator);
381 if (!product) { return std::nullopt; }
382 expanded = std::move(*product);
383 }
384 }
385 if (!checked(context_.require_expansion_terms(
386 static_cast<std::size_t>(reduced.num_terms()) +
387 static_cast<std::size_t>(expanded.num_terms()),
388 "residual.formal"))) {
389 return std::nullopt;
390 }
391 reduced = reduced + expanded;
392 }
393 return reduced;
394 }

397 Result<bool> prove(const Node& expression) {
398 if (!collect(expression))
399 { return failure_ ? Result<bool>::failure(*failure_) : Result<bool>::success(false); }
400 auto fraction = convert(expression);
401 if (failure_) { return Result<bool>::failure(*failure_); }
402 if (!fraction) { return false; }
403 const bool has_radicals = std::any_of(
404 radical_squares_.begin(), radical_squares_.end(),
405 [](const auto& value) { return value.has_value(); }) ||
406 std::any_of(
407 radical_radicands_.begin(), radical_radicands_.end(),
408 [](const auto& value) { return value != nullptr; });
409 if (!has_radicals) {
410 return fraction->numerator.is_zero() &&
411 !fraction->denominator.is_zero();
412 }
413 auto numerator = reduce_radicals(fraction->numerator);
414 auto denominator = reduce_radicals(fraction->denominator);
415 if (failure_) { return Result<bool>::failure(*failure_); }
416 return numerator && denominator && numerator->is_zero() && !denominator->is_zero();
417 }

425RootReplacementResult replace_root_operands(const detail::SymbolicNodePtr& node,
426 const std::vector<detail::SymbolicNodePtr>& operands, bool sum,
427 const detail::ExactRootId*& identity, ComputationContext& context, const std::string& operation) {
428 bool changed = false;
429 std::vector<detail::SymbolicNodePtr> replaced;
430 for (auto operand = operands.begin(); operand != operands.end(); ++operand) {
431 auto child = replace_exact_root(*operand, identity, context, operation);
432 if (!child || !child.value()) { return child; }
433 if (!changed && *child.value() != *operand) {
434 auto reserved = context.reserve_nodes(1, operation);
435 if (!reserved) { return RootReplacementResult::failure(reserved.error()); }
436 replaced.reserve(operands.size());
437 replaced.insert(replaced.end(), operands.begin(), operand);
438 changed = true;
439 }
440 if (changed) { replaced.push_back(std::move(*child.value())); }
441 }
442 if (!changed) { return RootReplacementResult::success(node); }
443 return RootReplacementResult::success(sum
444 ? SymbolicFactory::create_add(std::move(replaced))
445 : SymbolicFactory::create_multiply(std::move(replaced)));
446}

464RootReplacementResult replace_exact_root(
465 const detail::SymbolicNodePtr& node,
466 const detail::ExactRootId*& identity,
467 ComputationContext& context,
468 const std::string& operation) {
469 auto entered = context.enter_recursion(operation);
470 if (!entered) { return RootReplacementResult::failure(entered.error()); }
471 ResidualRecursionScope scope(context);
472 if (const auto* root = dynamic_cast<const RootOfNode*>(node.get())) {
473 if (identity && identity->compare(root->exact_id()) != 0) {
474 return RootReplacementResult::success(std::nullopt);
475 }
476 identity = &root->exact_id();
477 auto reserved = context.reserve_nodes(1, operation);
478 if (!reserved) { return RootReplacementResult::failure(reserved.error()); }
479 return RootReplacementResult::success(
480 SymbolicFactory::create_variable("_residual_root"));
481 }
482 if (dynamic_cast<const NumberNode*>(node.get())) {
483 return RootReplacementResult::success(node);
484 }
485 const auto* sum = dynamic_cast<const AddNode*>(node.get());
486 const auto* product = dynamic_cast<const MultiplyNode*>(node.get());
487 if (sum || product) {
488 const auto& operands = sum ? sum->operands() : product->operands();
489 return replace_root_operands(node, operands, sum != nullptr, identity, context, operation);
490 }
491 if (const auto* power = dynamic_cast<const PowerNode*>(node.get())) {
492 return replace_root_power(node, *power, identity, context, operation);
493 }
494 return RootReplacementResult::success(std::nullopt);
495}

505Result<bool> has_zero_root_remainder(
506 Polynomial<Rational> remainder,
507 const Polynomial<Rational>& defining,
508 ComputationContext& context,
509 const std::string& operation) {
510 auto terms = context.require_expansion_terms(
511 std::max(remainder.coeffs.size(), defining.coeffs.size()), operation);
512 if (!terms) return Result<bool>::failure(terms.error());
513 for (const auto& coefficient : remainder.coeffs) {
514 auto checked = check_residual_coefficient(coefficient, context, operation);
515 if (!checked) return Result<bool>::failure(checked.error());
516 }
517 for (const auto& coefficient : defining.coeffs) {
518 auto checked = check_residual_coefficient(coefficient, context, operation);
519 if (!checked) return Result<bool>::failure(checked.error());
520 }
521 while (remainder.degree() >= defining.degree()) {
522 const auto offset = remainder.coeffs.size() - defining.coeffs.size();
523 const auto& factor = remainder.coeffs.back();
524 for (std::size_t index = 0; index + 1 < defining.coeffs.size(); ++index) {
525 auto step = context.consume_steps(1, operation);
526 if (!step) return Result<bool>::failure(step.error());
527 auto product = factor * defining.coeffs[index];
528 auto checked = check_residual_coefficient(product, context, operation);
529 if (!checked) return Result<bool>::failure(checked.error());
530 auto& coefficient = remainder.coeffs[offset + index];
531 coefficient = coefficient - product;
532 checked = check_residual_coefficient(coefficient, context, operation);
533 if (!checked) return Result<bool>::failure(checked.error());
534 }
535 remainder.coeffs.pop_back();
536 remainder.trim();
537 }
538 return Result<bool>::success(remainder.is_zero());
539}

541ResidualCheckResult prove_exact_root_residual(
542 const ExprPtr& expression, ComputationContext& context,
543 const std::string& operation) {
544 const detail::ExactRootId* identity = nullptr;
545 auto replaced = replace_exact_root(
546 detail::node(expression), identity, context, operation);
547 if (!replaced) return ResidualCheckResult::failure(replaced.error());
548 if (!replaced.value() || !identity) {
549 return ResidualCheckResult::success(UnprovedResidual{expression});
550 }
551 auto polynomial = recognize_rational_polynomial(
552 *detail::make_expression_ptr(*replaced.value()),
553 "_residual_root", context);
554 if (!polynomial) return ResidualCheckResult::failure(polynomial.error());
555 if (polynomial.value()) {
556 auto zero = has_zero_root_remainder(
557 std::move(*polynomial.value()), identity->polynomial, context, operation);
558 if (!zero) return ResidualCheckResult::failure(zero.error());
559 if (zero.value()) {
560 return ResidualCheckResult::success(ProvedZeroResidual{
561 ExactResidualProof{expression}});
562 }
563 }
564 return ResidualCheckResult::success(UnprovedResidual{expression});
565}

573ResidualCheckResult classify_residual(
574 const ExprPtr& expression,
575 ComputationContext& context,
576 const std::string& operation) {
577 auto root_proof = prove_exact_root_residual(expression, context, operation);
578 if (!root_proof ||
579 std::holds_alternative<ProvedZeroResidual>(root_proof.value())) {
580 return root_proof;
581 }
582 auto proof = detail::classify_exact_zero(
583 expression, context, operation);
584 if (!proof) { return ResidualCheckResult::failure(proof.error()); }
585 if (proof.value() == detail::ZeroProof::Zero) {
586 return ResidualCheckResult::success(ProvedZeroResidual{
587 ExactNormalizationProof{expression}});
588 }
589 const auto variables = free_variables(detail::node(expression));
590 if (variables.size() == 1) {
591 auto polynomial = recognize_rational_polynomial(
592 *expression, *variables.begin(), context);
593 if (!polynomial) {
594 return ResidualCheckResult::failure(polynomial.error());
595 }
596 if (polynomial.value() && !polynomial.value()->is_zero()) {
597 return ResidualCheckResult::success(
598 ProvedNonIdentityResidual{expression});
599 }
600 }
601 if (proof.value() == detail::ZeroProof::NonZero) {
602 return ResidualCheckResult::success(
603 ProvedNonIdentityResidual{expression});
604 }
605 return ResidualCheckResult::success(UnprovedResidual{expression});
606}

639void collect_residual_profiles(
640 const detail::SymbolicNodePtr& node, ResidualProfiles& profiles) {
641 if (!node || (profiles.trig && profiles.exp_log)) {
642 return;
643 }
644 if (const auto function =
645 std::dynamic_pointer_cast<const FunctionNode>(node)) {
646 collect_function_profile(*function, profiles);
647 return;
648 }
649 if (const auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
650 collect_residual_profile_children(add->operands(), profiles);
651 return;
652 }
653 if (const auto multiply =
654 std::dynamic_pointer_cast<const MultiplyNode>(node)) {
655 collect_residual_profile_children(multiply->operands(), profiles);
656 return;
657 }
658 if (const auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
659 collect_residual_profiles(power->base(), profiles);
660 collect_residual_profiles(power->exponent(), profiles);
661 return;
662 }
663 if (const auto complex =
664 std::dynamic_pointer_cast<const ComplexNode>(node)) {
665 collect_residual_profiles(complex->real(), profiles);
666 collect_residual_profiles(complex->imag(), profiles);
667 }

670ResidualCheckResult prove_with_profiles(
671 const ExprPtr& current, ComputationContext& context,
672 const EqvOptions& options) {
673 ResidualProfiles required;
674 collect_residual_profiles(detail::node(current), required);
675 const auto zero = SymbolicExpr::number(0);
676 for (const auto profile : {
677 EqvProfile::Core, EqvProfile::TrigBasic, EqvProfile::ExpLogBasic}) {
678 if ((profile == EqvProfile::TrigBasic && !required.trig) ||
679 (profile == EqvProfile::ExpLogBasic && !required.exp_log)) {
680 continue;
681 }
682 auto profile_options = options;
683 profile_options.profile = profile;
684 auto equivalent = equivalent_core(*current, *zero, context, profile_options);
685 if (!equivalent) {
686 if (equivalent.error().code == CasErrc::UnsupportedExpression ||
687 equivalent.error().code == CasErrc::Inconclusive) {
688 continue;
689 }
690 return ResidualCheckResult::failure(equivalent.error());
691 }
692 if (equivalent.value()) {
693 return ResidualCheckResult::success(ProvedZeroResidual{
694 ExactResidualProof{current}});
695 }
696 }
697 return ResidualCheckResult::success(UnprovedResidual{current});
698}
699
700ResidualCheckResult prove_normalized_residual(const ExprPtr& residual,
701 ComputationContext& context, const EqvOptions& options, const std::string& operation) {
702 auto current = normalized(residual);
703 auto first = classify_residual(current, context, operation);
704 if (!first || is_decided(first.value())) { return first; }
705 current = normalized(current->expand());
706 auto expanded = classify_residual(current, context, operation);
707 if (!expanded || is_decided(expanded.value())) { return expanded; }
708 /**
709 * @brief 通分证明受计算预算约束，替代符号系数的 Euclid 及因子搜索。
710 * 支持绑定整数名同时出现在参数分母中的解族。
711 */
712 auto formal = FormalResidual(context).prove(detail::node(current));
713 if (!formal) { return ResidualCheckResult::failure(formal.error()); }
714 if (formal.value()) {
715 return ResidualCheckResult::success(ProvedZeroResidual{ExactResidualProof{current}});
716 }
717 return prove_with_profiles(current, context, options);
718}

722ResidualCheckResult check_zero_residual(
723 const ExprPtr& residual,
724 ComputationContext& context,
725 const LMCAS::EqvOptions& options) {
726 constexpr const char* operation = "residual.check_zero";
727 if (!residual || !detail::node(residual)) {
728 return ResidualCheckResult::failure(
729 CasErrc::InvalidArgument,
730 "residual expression cannot be null", operation);
731 }
732 auto access = context.consume_steps(1, operation);
733 if (!access) { return ResidualCheckResult::failure(access.error()); }
734
735 try {
736 auto root_proof = prove_exact_root_residual(residual, context, operation);
737 if (!root_proof ||
738 std::holds_alternative<ProvedZeroResidual>(root_proof.value())) {
739 return root_proof;
740 }
741 auto formal = FormalResidual(context).prove(detail::node(residual));
742 if (!formal) { return ResidualCheckResult::failure(formal.error()); }
743 if (formal.value())
744 { return ResidualCheckResult::success(ProvedZeroResidual{ExactResidualProof{residual}}); }
745 return prove_normalized_residual(residual, context, options, operation);
746 } catch (const CasError& error) {
747 return ResidualCheckResult::failure(error);
748 } catch (const std::bad_alloc&) {
749 return ResidualCheckResult::failure(
750 CasErrc::ResourceLimit,
751 "residual verification allocation failed", operation);
752 } catch (const std::exception& error) {
753 return ResidualCheckResult::failure(
754 CasErrc::InternalInvariant, error.what(), operation);
755 }
756}
