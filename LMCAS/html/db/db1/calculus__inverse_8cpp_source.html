12{

92static bool calculus_utils_contains_nonreal_sqrt(const std::shared_ptr<const SymbolicNode>& node)
93{

112 calculus_utils_contains_nonreal_sqrt(power->exponent());
113 }
114 if (auto complex = std::dynamic_pointer_cast<const ComplexNode>(node)) {
115 return calculus_utils_contains_nonreal_sqrt(complex->real()) ||
116 calculus_utils_contains_nonreal_sqrt(complex->imag());

127static CalculusRealCandidateStatus calculus_utils_extract_real_candidate(
128 const std::shared_ptr<SymbolicExpr>& candidate,
129 std::shared_ptr<SymbolicExpr>& real_candidate)
130{
131 real_candidate.reset();
132 auto simplified_candidate = rootof_simplify(candidate);
133 if (!simplified_candidate || !LMCAS::detail::node(simplified_candidate)) {
134 return CalculusRealCandidateStatus::Inconclusive;
135 }
136 if (LMCAS::detail::contains_node_type<RootOfNode>(LMCAS::detail::node(simplified_candidate))) {
137 return CalculusRealCandidateStatus::Inconclusive;
138 }
139 if (calculus_utils_contains_nonreal_sqrt(LMCAS::detail::node(simplified_candidate))) {
140 return CalculusRealCandidateStatus::NonReal;
141 }
142
143 if (auto complex = std::dynamic_pointer_cast<const ComplexNode>(
144 LMCAS::detail::node(simplified_candidate))) {
145 auto imag = LMCAS::detail::make_expression_ptr(complex->imag())->simplify();
146 if (!imag || !LMCAS::detail::node(imag)) {
147 return CalculusRealCandidateStatus::Inconclusive;
148 }
149 if (imag->is_zero()) {
150 auto real = LMCAS::detail::make_expression_ptr(complex->real())->simplify();
151 if (!real || !LMCAS::detail::node(real)) {
152 return CalculusRealCandidateStatus::Inconclusive;
153 }
154 real_candidate = real;
155 return CalculusRealCandidateStatus::Real;
156 }
157 return CalculusRealCandidateStatus::NonReal;
158 }
159
160 if (calculus_utils_contains_complex(LMCAS::detail::node(simplified_candidate))) {
161 return CalculusRealCandidateStatus::Inconclusive;
162 }
163
164 real_candidate = simplified_candidate;
165 return CalculusRealCandidateStatus::Real;
166}
167static bool contains_inverse_candidate(
168 const std::vector<std::shared_ptr<SymbolicExpr>>& candidates,
169 const std::shared_ptr<SymbolicExpr>& candidate)
170{
171 for (const auto& existing : candidates) {
172 if (existing && LMCAS::detail::node(existing) &&
173 LMCAS::detail::node(existing)->equals(*LMCAS::detail::node(candidate))) {
174 return true;
175 }
176 }
177 return false;

180static SymbolicExprVectorResult verified_real_inverse_candidates(
181 const std::vector<std::shared_ptr<SymbolicExpr>>& candidates,
182 const std::shared_ptr<SymbolicExpr>& equation, const std::string& var,
183 const std::string& operation)
184{
185 std::vector<std::shared_ptr<SymbolicExpr>> real_candidates;
186 real_candidates.reserve(candidates.size());
187 for (const auto& candidate : candidates) {
188 std::shared_ptr<SymbolicExpr> real_candidate;
189 auto status = calculus_utils_extract_real_candidate(candidate, real_candidate);
190 if (status == CalculusRealCandidateStatus::Inconclusive) {
191 return SymbolicExprVectorResult::failure(
192 CasErrc::Inconclusive,
193 "inverse derivative candidate reality could not be proven", operation);
194 }
195 if (status == CalculusRealCandidateStatus::NonReal) {
196 continue;
197 }
198 auto residual = equation->substitute(var, real_candidate);
199 auto simplified = residual ? residual->simplify() : nullptr;
200 if (!simplified || !LMCAS::detail::node(simplified)) {
201 return SymbolicExprVectorResult::failure(
202 CasErrc::InternalInvariant,
203 "inverse derivative candidate verification produced a null residual", operation);
204 }
205 if (!simplified->is_zero()) {
206 continue;
207 }
208 if (!contains_inverse_candidate(real_candidates, real_candidate)) {
209 real_candidates.push_back(real_candidate);
210 }
211 }
212 return SymbolicExprVectorResult::success(std::move(real_candidates));
213}

215static ExpressionResult inverse_branch_derivative(
216 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
217 const std::vector<std::shared_ptr<SymbolicExpr>>& candidates,
218 const std::string& operation)
219{
220 if (candidates.empty()) {
221 return ExpressionResult::failure(
222 CasErrc::DomainError,
223 "inverse point is outside the proven range",
224 operation);
225 }
226 if (candidates.size() != 1) {
227 return ExpressionResult::failure(
228 CasErrc::Inconclusive,
229 "inverse derivative requires a unique inverse branch",
230 operation);
231 }
232
233 auto f_prime = f->differentiate(var);
234 if (!f_prime || !LMCAS::detail::node(f_prime)) {
235 return ExpressionResult::failure(
236 CasErrc::Inconclusive,
237 "inverse derivative could not construct the derivative",
238 operation);
239 }
240
241 auto f_prime_at_x0 = f_prime->substitute(var, candidates[0]);
242 auto simplified_derivative = f_prime_at_x0 ? f_prime_at_x0->simplify() : nullptr;
243 if (!simplified_derivative || !LMCAS::detail::node(simplified_derivative)) {
244 return ExpressionResult::failure(
245 CasErrc::InternalInvariant,
246 "inverse derivative substitution produced a null expression",
247 operation);
248 }
249
250 if (simplified_derivative->is_zero()) {
251 return ExpressionResult::failure(
252 CasErrc::DomainError,
253 "inverse derivative is undefined where f' is zero",
254 operation);
255 }
256
257 auto result = SymbolicExpr::divide(SymbolicExpr::number(1), simplified_derivative);
258 auto simplified = result ? result->simplify() : nullptr;
259 if (!simplified || !LMCAS::detail::node(simplified)) {
260 return ExpressionResult::failure(
261 CasErrc::InternalInvariant,
262 "inverse derivative result construction failed",
263 operation);
264 }
265 return ExpressionResult::success(simplified);
266}

268static SymbolicExprVectorResult explicit_inverse_candidates(
269 const FiniteSolutions& finite, const std::string& operation)
270{
271 std::vector<std::shared_ptr<SymbolicExpr>> values;
272 values.reserve(finite.values.size());
273 for (const auto& solution : finite.values) {
274 if (!solution.value || !LMCAS::detail::node(solution.value)) {
275 return SymbolicExprVectorResult::failure(
276 CasErrc::InternalInvariant,
277 "checked solver returned a null inverse candidate",
278 operation);
279 }
280 auto candidate = rootof_simplify(solution.value);
281 if (!candidate || !LMCAS::detail::node(candidate)) {
282 return SymbolicExprVectorResult::failure(
283 CasErrc::InternalInvariant,
284 "checked solver returned a malformed inverse candidate",
285 operation);
286 }
287 values.push_back(candidate);
288 }
289 return SymbolicExprVectorResult::success(std::move(values));
290}

292ExpressionResult inverse_derivative_checked(
293 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
294 const std::shared_ptr<SymbolicExpr>& point,
295 ComputationContext& context)
296{
297 const std::string operation = "inverse_derivative";
298 auto input = calculus_utils_validate_expr_target(f, var, point, context, operation);
299 if (!input) {
300 return ExpressionResult::failure(input.error());
301 }
302 auto step = context.consume_steps(4, operation);
303 if (!step) {
304 return ExpressionResult::failure(step.error());
305 }
306
307 try {
308 auto eq = SymbolicExpr::add(
309 f, SymbolicExpr::multiply(SymbolicExpr::number(-1), point));
310
311 std::vector<std::shared_ptr<SymbolicExpr>> candidates;
312 auto inverse = inverse_function_checked(f, var, point, context);
313 if (inverse) {
314 candidates = inverse.value();
315 } else {
316 return ExpressionResult::failure(inverse.error());
317 }
318 auto verified = verified_real_inverse_candidates(candidates, eq, var, operation);
319 if (!verified) {
320 return ExpressionResult::failure(verified.error());
321 }
322 return inverse_branch_derivative(f, var, verified.value(), operation);
323 } catch (const std::bad_alloc&) {
324 return ExpressionResult::failure(CasErrc::ResourceLimit,
325 "inverse derivative allocation failed",
326 operation);
327 } catch (const std::exception& e) {
328 return ExpressionResult::failure(CasErrc::InternalInvariant,
329 e.what(), operation);
330 }
331}

341SymbolicExprVectorResult inverse_function_checked(
342 const std::shared_ptr<SymbolicExpr>& f, const std::string& var,
343 const std::shared_ptr<SymbolicExpr>& y, ComputationContext& context)
344{
345 const std::string operation = "inverse_function";
346 auto input = calculus_utils_validate_expr_target(f, var, y, context, operation);
347 if (!input) {
348 return SymbolicExprVectorResult::failure(input.error());
349 }
350 auto step = context.consume_steps(4, operation);
351 if (!step) {
352 return SymbolicExprVectorResult::failure(step.error());
353 }
354
355 try {
356 auto eq = SymbolicExpr::eq(f, y);
357
358 auto solved = solve_equation(eq, var, context, SolveOptions{});
359 if (!solved) {
360 return SymbolicExprVectorResult::failure(solved.error());
361 }
362
363 const auto& solutions = solved.value();
364 if (std::holds_alternative<EmptySolutions>(solutions)) {
365 return SymbolicExprVectorResult::success({});
366 }
367 if (const auto* finite = std::get_if<FiniteSolutions>(&solutions)) {
368 return explicit_inverse_candidates(*finite, operation);
369 }
370
371 return SymbolicExprVectorResult::failure(
372 CasErrc::Inconclusive,
373 "inverse equation is outside the finite exact support domain",
374 operation);
375 } catch (const std::bad_alloc&) {
376 return SymbolicExprVectorResult::failure(CasErrc::ResourceLimit,
377 "inverse function allocation failed",
378 operation);
379 } catch (const std::exception& e) {
380 return SymbolicExprVectorResult::failure(CasErrc::InternalInvariant,
381 e.what(), operation);
382 }
383}
