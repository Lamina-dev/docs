12std::shared_ptr<SymbolicExpr> rd_integrate_linear_term(
13 const Polynomial<Rational>& numerator, const Polynomial<Rational>& factor,
14 int power, const std::string& var) {
15 Rational r = Rational(0) - factor.coeffs[0];
16 Rational A = (numerator.coeffs.size() > 0) ? numerator.coeffs[0] : Rational(0);
17 if (A == Rational(0)) { return SymbolicExpr::number(0); }
18
19 if (power == 1) {
20 auto inner = rd_var_minus(var, r);
21 return SymbolicExpr::multiply(SymbolicExpr::number(A), SymbolicExpr::ln(inner));
22 }
23 Rational coeff = Rational(0) - A / Rational(BigInt(power - 1));
24 auto base = rd_var_minus(var, r);
25 auto exp = SymbolicExpr::number(BigInt(power - 1));
26 auto pw = SymbolicExpr::power(base, exp);
27 auto inv_pw = SymbolicExpr::power(pw, SymbolicExpr::number(BigInt(-1)));
28 return SymbolicExpr::multiply(SymbolicExpr::number(coeff), inv_pw);
29}

110 auto negative_root = SymbolicExpr::multiply(
111 SymbolicExpr::number(BigInt(-1)), square_root);

118 SymbolicExpr::number(residual), inverse_root);

124std::shared_ptr<SymbolicExpr> rd_integrate_quadratic_term(
125 const Polynomial<Rational>& numerator, const Polynomial<Rational>& factor,
126 const std::string& var) {
127 const Rational q = factor.coeffs.empty()
128 ? Rational(0) : factor.coeffs[0];
129 const Rational p = factor.coeffs.size() > 1
130 ? factor.coeffs[1] : Rational(0);
131 const Rational discriminant = p * p - Rational(4) * q;
132 const Rational B = numerator.coeffs.size() > 1
133 ? numerator.coeffs[1] : Rational(0);
134 const Rational C = numerator.coeffs.empty()
135 ? Rational(0) : numerator.coeffs[0];
136 const Rational residual = C - (B * p) / Rational(2);
137
138 if (discriminant == Rational(0)) {
139 return rd_integrate_repeated_quadratic(
140 numerator, factor, p, B, residual, var);
141 }
142
143 auto result = rd_quadratic_logarithm(factor, B);
144 if (residual == Rational(0)) {
145 return result;
146 }
147 auto shifted = rd_shifted_double_variable(p, var);
148 if (discriminant.get_numerator().is_negative()) {
149 return rd_integrate_negative_discriminant(
150 result, shifted, residual, discriminant);
151 }
152 return rd_integrate_positive_discriminant(
153 result, shifted, residual, discriminant);
154}

158detail::IntegralParts detail::integrate_rational_term(
159 const Polynomial<Rational>& numerator,
160 const Polynomial<Rational>& factor,
161 int power, const std::string& var) {
162 const SymbolicExpr zero = *SymbolicExpr::number(0);
163 if (numerator.is_zero()) { return {zero, zero}; }
164 if (factor.degree() == 1) {
165 return {*rd_integrate_linear_term(numerator, factor, power, var), zero};
166 }
167 if (factor.degree() == 2 && power == 1) {
168 return {*rd_integrate_quadratic_term(numerator, factor, var), zero};
169 }
170 return {zero, *rd_fraction_term(numerator, factor, power)};
171}
