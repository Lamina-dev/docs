6std::shared_ptr<SymbolicExpr> rd_integrate_linear_term(
7 const Polynomial<Rational>& numerator, const Polynomial<Rational>& factor,
8 int power, const std::string& var) {
9 Rational r = Rational(0) - factor.coeffs[0];
10 Rational A = (numerator.coeffs.size() > 0) ? numerator.coeffs[0] : Rational(0);
11 if (A == Rational(0)) { return SymbolicExpr::number(0); }
12
13 if (power == 1) {
14 auto inner = rd_var_minus(var, r);
15 return SymbolicExpr::multiply(SymbolicExpr::number(A), SymbolicExpr::ln(inner));
16 }
17 Rational coeff = Rational(0) - A / Rational(BigInt(power - 1));
18 auto base = rd_var_minus(var, r);
19 auto exp = SymbolicExpr::number(BigInt(power - 1));
20 auto pw = SymbolicExpr::power(base, exp);
21 auto inv_pw = SymbolicExpr::power(pw, SymbolicExpr::number(BigInt(-1)));
22 return SymbolicExpr::multiply(SymbolicExpr::number(coeff), inv_pw);
23}

60std::shared_ptr<SymbolicExpr> rd_quadratic_logarithm(
61 const Polynomial<Rational>& factor, const Rational& B) {
62 const Rational half = B / Rational(2);
63 auto result = SymbolicExpr::number(0);
64 if (half == Rational(0)) {
65 return result;
66 }
67 auto denominator = poly_to_symbolic(factor);
68 auto logarithm = SymbolicExpr::multiply(
69 SymbolicExpr::number(half), SymbolicExpr::ln(denominator));
70 return SymbolicExpr::add(result, logarithm);

82std::shared_ptr<SymbolicExpr> rd_integrate_negative_discriminant(
83 const std::shared_ptr<SymbolicExpr>& result,
84 const std::shared_ptr<SymbolicExpr>& shifted,
85 const Rational& residual, const Rational& discriminant) {
86 auto square_root = SymbolicExpr::sqrt(
87 SymbolicExpr::number(Rational(0) - discriminant));
88 auto inverse_root = SymbolicExpr::power(
89 square_root, SymbolicExpr::number(BigInt(-1)));
90 auto arctangent = make_arctan(
91 SymbolicExpr::multiply(shifted, inverse_root));
92 auto coefficient = SymbolicExpr::multiply(
93 SymbolicExpr::multiply(
94 SymbolicExpr::number(residual),
95 SymbolicExpr::number(BigInt(2))),
96 inverse_root);
97 return SymbolicExpr::add(
98 result, SymbolicExpr::multiply(coefficient, arctangent));

101std::shared_ptr<SymbolicExpr> rd_integrate_positive_discriminant(
102 const std::shared_ptr<SymbolicExpr>& result,
103 const std::shared_ptr<SymbolicExpr>& shifted,
104 const Rational& residual, const Rational& discriminant) {
105 auto square_root =
106 SymbolicExpr::sqrt(SymbolicExpr::number(discriminant));
107 auto negative_root = SymbolicExpr::multiply(
108 SymbolicExpr::number(BigInt(-1)), square_root);
109 auto ratio = SymbolicExpr::divide(
110 SymbolicExpr::add(shifted, negative_root),
111 SymbolicExpr::add(shifted, square_root));
112 auto inverse_root = SymbolicExpr::power(
113 square_root, SymbolicExpr::number(BigInt(-1)));
114 auto coefficient = SymbolicExpr::multiply(
115 SymbolicExpr::number(residual), inverse_root);
116 return SymbolicExpr::add(
117 result,
118 SymbolicExpr::multiply(coefficient, SymbolicExpr::ln(ratio)));

121std::shared_ptr<SymbolicExpr> rd_integrate_quadratic_term(
122 const Polynomial<Rational>& numerator, const Polynomial<Rational>& factor,
123 int power, const std::string& var) {
124 if (power >= 2) {
125 return rd_unevaluated_quadratic_term(
126 numerator, factor, power, var);
127 }
128
129 const Rational q = factor.coeffs.empty()
130 ? Rational(0) : factor.coeffs[0];
131 const Rational p = factor.coeffs.size() > 1
132 ? factor.coeffs[1] : Rational(0);
133 const Rational discriminant = p * p - Rational(4) * q;
134 const Rational B = numerator.coeffs.size() > 1
135 ? numerator.coeffs[1] : Rational(0);
136 const Rational C = numerator.coeffs.empty()
137 ? Rational(0) : numerator.coeffs[0];
138 const Rational residual = C - (B * p) / Rational(2);
139
140 if (discriminant == Rational(0)) {
141 return rd_integrate_repeated_quadratic(
142 numerator, factor, p, B, residual, var);
143 }
144
145 auto result = rd_quadratic_logarithm(factor, B);
146 if (residual == Rational(0)) {
147 return result;
148 }
149 auto shifted = rd_shifted_double_variable(p, var);
150 if (discriminant.get_numerator().is_negative()) {
151 return rd_integrate_negative_discriminant(
152 result, shifted, residual, discriminant);
153 }
154 return rd_integrate_positive_discriminant(
155 result, shifted, residual, discriminant);
156}
