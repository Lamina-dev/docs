26std::shared_ptr<SymbolicExpr> SymbolicExpr::series(
27 const std::string& variable,
28 const std::shared_ptr<SymbolicExpr>& point,
29 int order,
30 const LMCAS::AssumptionContext*) const {
31 if (!impl_->root || !point) return nullptr;
32 if (order < 0) return nullptr;
33
34 auto variable_expr = SymbolicExpr::variable(variable);
35 auto delta = SymbolicExpr::add(
36 variable_expr,
37 SymbolicExpr::multiply(SymbolicExpr::number(-1), point));
38 std::vector<std::shared_ptr<const SymbolicNode>> terms;
39 auto derivative = LMCAS::detail::make_expression_ptr(impl_->root->clone());
40
41 for (int n = 0; n <= order; ++n) {
42 if (n > 0) {
43 derivative = derivative->differentiate(variable);
44 if (!derivative) return nullptr;
45 derivative = derivative->simplify();
46 }
47
48 auto coefficient = derivative->substitute(variable, point);
49 if (!coefficient) return nullptr;
50 coefficient = coefficient->simplify();
51
52 auto term = coefficient;
53 if (n > 0) {
54 term = SymbolicExpr::multiply(
55 term,
56 SymbolicExpr::power(delta, SymbolicExpr::number(n)));
57 }
58 if (n > 1) {
59 auto factorial = BigInt::factorial(static_cast<unsigned int>(n));
60 term = SymbolicExpr::multiply(
61 term,
62 SymbolicExpr::number(Rational(BigInt(1), factorial)));
63 }
64 terms.push_back(LMCAS::detail::node(term));
65 }
66
67 if (terms.empty()) return SymbolicExpr::number(0);
68 return LMCAS::detail::make_expression_ptr(
69 LMCAS::detail::make_node<AddNode>(terms))->simplify();
70}
