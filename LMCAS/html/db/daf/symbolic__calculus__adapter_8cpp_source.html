42std::shared_ptr<SymbolicExpr> SymbolicExpr::series(
43 const std::string& variable,
44 const std::shared_ptr<SymbolicExpr>& point,
45 int order,
46 const lamina::AssumptionContext*) const {
47 if (!impl_->root || !point) return nullptr;
48 if (order < 0) return SymbolicExpr::number(0);
49
50 auto variable_expr = SymbolicExpr::variable(variable);
51 auto delta = SymbolicExpr::add(
52 variable_expr,
53 SymbolicExpr::multiply(SymbolicExpr::number(-1), point));
54 std::vector<std::shared_ptr<const SymbolicNode>> terms;
55 auto derivative = lamina::detail::make_expression_ptr(impl_->root->clone());
56
57 for (int n = 0; n <= order; ++n) {
58 if (n > 0) {
59 derivative = derivative->differentiate(variable);
60 if (!derivative) break;
61 derivative = derivative->simplify();
62 }
63
64 auto coefficient = derivative->substitute(variable, point);
65 if (!coefficient) break;
66 coefficient = coefficient->simplify();
67
68 auto term = coefficient;
69 if (n > 0) {
70 term = SymbolicExpr::multiply(
71 term,
72 SymbolicExpr::power(delta, SymbolicExpr::number(n)));
73 }
74 if (n > 1) {
75 auto factorial = BigInt::factorial(static_cast<unsigned int>(n));
76 term = SymbolicExpr::multiply(
77 term,
78 SymbolicExpr::number(Rational(BigInt(1), factorial)));
79 }
80 terms.push_back(lamina::detail::node(term));
81 }
82
83 if (terms.empty()) return SymbolicExpr::number(0);
84 return lamina::detail::make_expression_ptr(
85 lamina::detail::make_node<AddNode>(terms))->simplify();
86}
