11ExpressionResult matrix_characteristic_polynomial_checked(
12 const std::shared_ptr<SymbolicExpr>& A, const std::string& variable,
13 ComputationContext& context) {
14 constexpr const char* operation = "matrix_characteristic_polynomial";
15 auto matrix_result = require_square_matrix(A, context, operation);
16 if (!matrix_result) return ExpressionResult::failure(matrix_result.error());
17 if (variable.empty()) {
18 return ExpressionResult::failure(
19 CasErrc::InvalidArgument,
20 "characteristic polynomial variable cannot be empty", operation);
21 }
22 try {
23 const auto& matrix = *matrix_result.value();
24 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> entries(
25 matrix.rows(),
26 std::vector<std::shared_ptr<SymbolicExpr>>(matrix.cols()));
27 auto lambda = SymbolicExpr::variable(variable);
28 for (std::size_t row = 0; row < matrix.rows(); ++row) {
29 for (std::size_t column = 0; column < matrix.cols(); ++column) {
30 auto value = detail::make_expression_ptr(
31 matrix.get(row, column));
32 entries[row][column] = row == column
33 ? SymbolicExpr::add(
34 value,
35 SymbolicExpr::multiply(
36 lambda, SymbolicExpr::number(-1)))
37 : std::move(value);
38 }
39 }
40 auto determinant = matrix_determinant_checked(
41 SymbolicExpr::matrix(entries), context);
42 if (!determinant) return ExpressionResult::failure(determinant.error());
43 return ExpressionResult::success(
44 std::move(determinant.value()));
45 } catch (const std::bad_alloc&) {
46 return ExpressionResult::failure(
47 CasErrc::ResourceLimit,
48 "allocation failed while constructing characteristic polynomial",
49 operation);
50 } catch (const std::exception& ex) {
51 return ExpressionResult::failure(
52 CasErrc::InternalInvariant, ex.what(), operation);
53 }

62MatrixEigenvalueResult matrix_eigenvalues_checked(
63 const std::shared_ptr<SymbolicExpr>& A, ComputationContext& context) {
64 constexpr const char* operation = "matrix_eigenvalues";
65 auto matrix_result = require_square_matrix(A, context, operation);
66 if (!matrix_result) {
67 return MatrixEigenvalueResult::failure(matrix_result.error());
68 }
69 const auto& matrix = *matrix_result.value();
70 bool triangular = true;
71 for (std::size_t row = 0; row < matrix.rows() && triangular; ++row) {
72 for (std::size_t column = 0; column < row; ++column) {
73 auto value =
74 detail::make_expression_ptr(matrix.get(row, column))->simplify();
75 if (!detail::node(value)->is_zero()) {
76 triangular = false;
77 break;
78 }
79 }
80 }
81 if (triangular) {
82 std::vector<std::shared_ptr<SymbolicExpr>> values;
83 values.reserve(matrix.rows());
84 for (std::size_t index = 0; index < matrix.rows(); ++index) {
85 values.push_back(
86 detail::make_expression_ptr(
87 matrix.get(index, index))->simplify());
88 }
89 return MatrixEigenvalueResult::success(std::move(values));
90 }
91
92 auto polynomial = matrix_characteristic_polynomial_checked(
93 A, "lambda", context);
94 if (!polynomial) {
95 return MatrixEigenvalueResult::failure(polynomial.error());
96 }
97 auto roots = solve_finite_checked(
98 polynomial.value(), "lambda", context, SolveOptions{});
99 if (!roots) return MatrixEigenvalueResult::failure(roots.error());
100 std::vector<std::shared_ptr<SymbolicExpr>> values;
101 std::set<std::string> seen;
102 for (auto& root : roots.value()) {
103 const auto key = root->to_string();
104 if (seen.insert(key).second) values.push_back(std::move(root));
105 }
106 if (values.empty()) {
107 return MatrixEigenvalueResult::failure(
108 CasErrc::Inconclusive,
109 "eigenvalue solver produced no finite roots", operation);
110 }
111 return MatrixEigenvalueResult::success(std::move(values));
112}

123Result<bool> eigenvalue_already_collected(const ExprPtr& lambda,
124 const Eigenspaces& spaces, ComputationContext& context,
125 const detail::AssumptionFacts& facts) {
126 constexpr const char* operation = "matrix_eigenspaces";
127 for (const auto& space : spaces) {
128 if (detail::node(lambda)->equals(*detail::node(space.first))) { return true; }
129 auto difference = SymbolicExpr::add(lambda,
130 SymbolicExpr::multiply(SymbolicExpr::number(-1), space.first));
131 auto proof = detail::classify_exact_zero(difference, context, operation);
132 if (!proof) { return Result<bool>::failure(proof.error()); }
133 if (proof.value() == detail::ZeroProof::Zero) { return true; }
134 auto nonzero = detail::query_nonzero_value(detail::node(difference->simplify()), facts, Domain::Complex, context);
135 if (!nonzero) { return Result<bool>::failure(nonzero.error()); }
136 if (nonzero.value() != Tribool::True) {
137 return Result<bool>::failure(CasErrc::Inconclusive,
138 "eigenvalues cannot be proved pointwise distinct", operation);
139 }
140 }
141 return false;
142}

144MatrixEigenvectorResult eigenvalue_nullspace(const MatrixNode& matrix,
145 const ExprPtr& lambda, ComputationContext& context) {
146 std::vector<std::vector<std::shared_ptr<SymbolicExpr>>> shifted(
147 matrix.rows(),

150 for (std::size_t column = 0; column < matrix.cols(); ++column) {
151 auto value = detail::make_expression_ptr(matrix.get(row, column));

153 ? SymbolicExpr::add(value, SymbolicExpr::multiply(

157 }

162 "eigenvalue has no proved nonzero nullspace basis", "matrix_eigenspaces");
163 }

169detail::matrix_eigenspaces_checked(const ExprPtr& A, ComputationContext& context) {
170 using SpacesResult = Result<Eigenspaces>;
171 constexpr const char* operation = "matrix_eigenspaces";
172 auto matrix_result = require_square_matrix(A, context, operation);
173 if (!matrix_result) { return SpacesResult::failure(matrix_result.error()); }
174 auto eigenvalues = matrix_eigenvalues_checked(A, context);
175 if (!eigenvalues) { return SpacesResult::failure(eigenvalues.error()); }
176 const auto& matrix = *matrix_result.value();
177 Eigenspaces spaces;
178 AssumptionContext assumptions;
179 detail::AssumptionFacts facts(context.assumptions() ? *context.assumptions() : assumptions);
180 try {
181 for (const auto& eigenvalue : eigenvalues.value()) {
182 auto lambda = eigenvalue->simplify();
183 auto duplicate = eigenvalue_already_collected(lambda, spaces, context, facts);
184 if (!duplicate) { return SpacesResult::failure(duplicate.error()); }
185 if (duplicate.value()) { continue; }
186 auto nullspace = eigenvalue_nullspace(matrix, lambda, context);
187 if (!nullspace) { return SpacesResult::failure(nullspace.error()); }
188 spaces.emplace_back(std::move(lambda), std::move(nullspace.value()));
189 }
190 return SpacesResult::success(std::move(spaces));
191 } catch (const std::bad_alloc&) {
192 return SpacesResult::failure(CasErrc::ResourceLimit,
193 "allocation failed while computing eigenspaces", operation);
194 } catch (const std::exception& ex) {
195 return SpacesResult::failure(CasErrc::InternalInvariant, ex.what(), operation);
196 }
197}
