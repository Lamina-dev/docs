126void NormalizationVisitor::visit(const PowerNode& node) {
127 std::shared_ptr<const SymbolicNode> base;
128 if (try_normalize_squared_norm(node, base)) {
129 return;
130 }
131 if (!base) {
132 node.base()->accept(*this);
133 base = result;
134 }
135 node.exponent()->accept(*this);
136 auto exponent = result;
137 if (normalize_power_identity(base, exponent)) {
138 return;
139 }
140 const auto base_number = std::dynamic_pointer_cast<const NumberNode>(base);
141 const auto exponent_number = std::dynamic_pointer_cast<const NumberNode>(exponent);
142 if (base_number && exponent_number &&
143 normalization_can_discard(base, facts_, domain_, context_) && normalization_can_discard(exponent, facts_, domain_, context_)) {
144 if (auto folded = normalization_numeric_power(base_number, exponent_number, rewrite_budget())) {
145 set_result(std::move(folded));
146 return;
147 }
148 }
149 if (const auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(base)) {
150 if (normalize_product_power(*multiply, exponent)) {

152 }
153 }
154 if (const auto power = std::dynamic_pointer_cast<const PowerNode>(base)) {
155 if (normalize_nested_power(*power, exponent)) {
156 return;
157 }
158 }
159 normalization_check_children(rewrite_budget(), 1, base, exponent);
160 set_result(detail::make_node<PowerNode>(base, exponent));
161}
