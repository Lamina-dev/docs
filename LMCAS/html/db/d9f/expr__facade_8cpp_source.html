23ExprResult make_unary_math_expr(const ExprPtr& expression,
24 ComputationContext& context,
25 const char* function_name,
26 ExprPtr (*factory)(ExprPtr)) {
27 auto step = context.consume_steps(1, kMathOperation);
28 if (!step) return ExprResult::failure(step.error());
29 if (!expression || !LMCAS::detail::node(expression)) {
30 return expression_failure(CasErrc::InvalidArgument,
31 std::string(function_name) +
32 " argument cannot be null",
33 kMathOperation);
34 }
35 try {
36 auto result = factory(expression);
37 if (!result || !LMCAS::detail::node(result)) {
38 return expression_failure(CasErrc::InternalInvariant,
39 std::string(function_name) +
40 " expression construction failed",
41 kMathOperation);
42 }
43 return ExprResult::success(std::move(result));
44 } catch (const std::bad_alloc&) {
45 return expression_failure(CasErrc::ResourceLimit,
46 std::string(function_name) +
47 " expression allocation failed",
48 kMathOperation);
49 } catch (const std::exception& error) {
50 return expression_failure(CasErrc::InvalidArgument, error.what(),
51 kMathOperation);
52 }
53}

138ExprResult function(const std::string& name, std::vector<ExprPtr> arguments) {
139 constexpr const char* operation = "LMCAS.function";
140 if (name.empty()) {
141 return expression_failure(CasErrc::InvalidArgument,
142 "function name cannot be empty", operation);
143 }
144 std::vector<std::shared_ptr<const SymbolicNode>> nodes;
145 nodes.reserve(arguments.size());
146 for (const auto& argument : arguments) {
147 if (!argument || !LMCAS::detail::node(argument)) {
148 return expression_failure(CasErrc::InvalidArgument,
149 "function arguments cannot be null", operation);
150 }
151 nodes.push_back(LMCAS::detail::node(argument));
152 }
153 try {
154 return ExprResult::success(LMCAS::detail::make_expression_ptr(
155 LMCAS::detail::make_node<UninterpretedFunctionNode>(
156 name, std::move(nodes))));
157 } catch (const std::bad_alloc&) {
158 return expression_failure(CasErrc::ResourceLimit,
159 "function allocation failed", operation);
160 } catch (const std::exception& error) {
161 return expression_failure(CasErrc::InvalidArgument,
162 error.what(), operation);
163 }
164}
