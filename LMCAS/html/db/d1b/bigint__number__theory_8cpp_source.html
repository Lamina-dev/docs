7BigInt binary_modular_power(const BigInt& base, const BigInt& exp, const BigInt& mod) {
8 BigInt result = 1;
9 BigInt residue = base % mod;
10 if (residue.is_negative()) {
11 residue += mod;
12 }
13 BigInt exponent = exp;
14 while (!exponent.is_zero()) {
15 if (exponent.is_odd()) {
16 result = (result * residue) % mod;
17 }
18 residue = (residue * residue) % mod;
19 exponent >>= 1;
20 }
21 return result;
22}

116 for (auto p : small_primes) {
117 if ((*this % p).is_zero()) return false;
118 }

127 s++;
128 }

130 static const uint64_t mr_bases[] = {2, 3, 5, 7, 11, 13, 17};
131 for (auto base : mr_bases) {
132 BigInt x = BigInt::pow_mod(BigInt(base), d, n);
133 if (x == 1 || x == n_minus_one) continue;
134
135 bool composite = true;
136 for (int r = 1; r < s; ++r) {
137 x = BigInt::pow_mod(x, two, n);
138 if (x == n_minus_one) {
139 composite = false;

141 }
