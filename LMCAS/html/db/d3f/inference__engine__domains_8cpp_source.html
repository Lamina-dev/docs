9InferenceTriboolResult InferenceEngine::infer_add_domain_checked(
10 const AddNode& node, Domain target, ComputationContext& context) const {
11 try {
12 if (node.operands().empty()) {
13 return InferenceTriboolResult::success(Tribool::Unknown);
14 }
15 for (const auto& operand : node.operands()) {
16 auto op_expr = LMCAS::detail::expression_from_node(operand);
17 Tribool op_result = Tribool::Unknown;
18 switch (target) {
19 case Domain::Integer: {
20 auto queried = query_integer_checked(op_expr, context);
21 if (!queried) {
22 return queried;
23 }
24 op_result = queried.value();
25 break;
26 }
27 case Domain::Real: {
28 auto queried = query_real_checked(op_expr, context);
29 if (!queried) {
30 return queried;
31 }
32 op_result = queried.value();
33 break;
34 }
35 default:
36 return InferenceTriboolResult::success(Tribool::Unknown);
37 }
38
39 if (op_result != Tribool::True) {
40 if (target == Domain::Real) {
41 auto integer = query_integer_checked(op_expr, context);
42 if (!integer) {
43 return integer;
44 }
45 if (integer.value() == Tribool::True) {
46 continue;
47 }
48 }
49 return InferenceTriboolResult::success(Tribool::Unknown);
50 }
51 }
52
53 return InferenceTriboolResult::success(Tribool::True);
54 } catch (const CasError& error) {
55 return InferenceTriboolResult::failure(error);
56 } catch (const std::bad_alloc&) {
57 return InferenceTriboolResult::failure(
58 CasErrc::ResourceLimit,
59 "addition domain inference allocation failed",
60 "infer_add_domain_checked");
61 } catch (const std::exception& ex) {
62 return InferenceTriboolResult::failure(
63 CasErrc::InternalInvariant,
64 ex.what(),
65 "infer_add_domain_checked");
66 }

96 const InferenceEngine& engine, const MultiplyNode& node, Domain target, ComputationContext& context) const {
97 if (target != Domain::Integer && target != Domain::Real) { return Tribool::Unknown; }
98 for (const auto& operand : node.operands()) {
99 const auto expression = detail::expression_from_node(operand);
100 auto result = target == Domain::Integer ? engine.query_integer_checked(expression, context)

112 if (target == Domain::Real) {

114 const std::shared_ptr<const SymbolicNode> borrowed(

116 auto real = detail::query_real_value(borrowed, facts, context);
117 if (!real) { return real; }

143InferenceTriboolResult InferenceEngine::Impl::integral_power_domain(
144 const InferenceEngine& engine, const SymbolicExpr& base,
145 const NumberNode& exponent, Domain target, ComputationContext& context) {
146 if (target == Domain::Integer) {
147 auto base_int_result = engine.query_integer_checked(base, context);
148 if (!base_int_result) {
149 return base_int_result;
150 }
151 const Tribool base_int = base_int_result.value();
152 if (base_int == Tribool::True && is_positive_integer_number(exponent)) {
153 return InferenceTriboolResult::success(Tribool::True);
154 }
155 if (base_int == Tribool::True && is_zero_number(exponent)) {
156 return InferenceTriboolResult::success(Tribool::True);
157 }
158 }
159 return Tribool::Unknown;
