122protected:
123 std::size_t compute_hash() const override {
124 std::size_t seed = static_cast<std::size_t>(type_priority());
125 hash_combine(seed, std::hash<std::string>{}(name_));
126 for (const auto& argument : arguments_) hash_combine(seed, argument->hash());
127 return seed;

130 int compare_same_type(const SymbolicNode& other) const override {
131 const auto& function = static_cast<const UninterpretedFunctionNode&>(other);
132 int comparison = name_.compare(function.name_);
133 if (comparison != 0) return comparison;
134 const auto count = std::min(arguments_.size(), function.arguments_.size());
135 for (std::size_t index = 0; index < count; ++index) {
136 comparison = arguments_[index]->compare(*function.arguments_[index]);
137 if (comparison != 0) return comparison;
138 }
139 if (arguments_.size() == function.arguments_.size()) return 0;
140 return arguments_.size() < function.arguments_.size() ? -1 : 1;
141 }

166public:
167 using DenseStorage = std::vector<std::shared_ptr<const SymbolicNode>>; ///< 稠密存储（按行优先展开）

170 /**
171 * @brief 验证并返回网格的列数（要求各行列数一致）。
172 * @param grid 二维节点网格
173 * @return 列数
174 * @throw std::invalid_argument 各行列数不一致时抛出
175 */
176 static size_t validate_grid_columns(const std::vector<std::vector<std::shared_ptr<const SymbolicNode>>>& grid) {
177 if (grid.empty()) {
178 throw std::invalid_argument("MatrixNode: matrix must have at least one row");
179 }
180 size_t ncols = grid[0].size();
181 if (ncols == 0) {
182 throw std::invalid_argument("MatrixNode: matrix must have at least one column");
183 }
184 for (const auto& item : grid[0]) {
185 if (!item) {
186 throw std::invalid_argument("MatrixNode: matrix elements cannot be null");
187 }
188 }
189 for (size_t i = 1; i < grid.size(); ++i) {
190 if (grid[i].size() != ncols) {
191 throw std::invalid_argument("MatrixNode: all rows must have the same number of columns");
192 }
193 for (const auto& item : grid[i]) {
194 if (!item) {
195 throw std::invalid_argument("MatrixNode: matrix elements cannot be null");
196 }
197 }
198 }
199 return ncols;
200 }

202 static DenseStorage validate_dense_storage(size_t r, size_t c, DenseStorage dense) {
203 if (r == 0 || c == 0) {
204 throw std::invalid_argument("MatrixNode: matrix dimensions must be non-zero");
205 }
206 if (r > std::numeric_limits<size_t>::max() / c) {
207 throw std::length_error("MatrixNode: matrix dimensions overflow");
208 }
209 if (dense.size() != r * c) {
210 throw std::invalid_argument("MatrixNode: dense storage size does not match dimensions");
211 }
212 for (const auto& item : dense) {
213 if (!item) {
214 throw std::invalid_argument("MatrixNode: dense storage elements cannot be null");
215 }
216 }
217 return dense;
218 }

220 static SparseStorage validate_sparse_storage(size_t r, size_t c, SparseStorage sparse) {
221 if (r == 0 || c == 0) {
222 throw std::invalid_argument("MatrixNode: matrix dimensions must be non-zero");
223 }
224 if (r > std::numeric_limits<size_t>::max() / c) {
225 throw std::length_error("MatrixNode: matrix dimensions overflow");
226 }
227 const size_t total = r * c;
228 for (const auto& [idx, item] : sparse) {
229 if (idx >= total) {
230 throw std::invalid_argument("MatrixNode: sparse storage index is out of bounds");
231 }
232 if (!item) {
233 throw std::invalid_argument("MatrixNode: sparse storage elements cannot be null");
234 }
235 }
236 return sparse;
237 }

280 hash_combine(seed, rows_);

304 int compare_same_type(const SymbolicNode& other) const override {
305 const auto& o = static_cast<const MatrixNode&>(other);
306 if (rows_ != o.rows_) return rows_ < o.rows_ ? -1 : 1;
307 if (cols_ != o.cols_) return cols_ < o.cols_ ? -1 : 1;
308
309 for (size_t r = 0; r < rows_; ++r) {
310 for (size_t c = 0; c < cols_; ++c) {
311 auto v1 = this->get(r, c);
312 auto v2 = o.get(r, c);
313
314 bool z1 = !v1 || v1->is_zero();
315 bool z2 = !v2 || v2->is_zero();
316
317 if (z1 && z2) continue;
318 if (z1) return -1;
319 if (z2) return 1;
320
321 int cmp = v1->compare(*v2);
322 if (cmp != 0) return cmp;
323 }
324 }
325 return 0;

350 /**
351 * @brief 获取指定位置的矩阵元素。
352 * @param r 行索引（从 0 开始）
353 * @param c 列索引（从 0 开始）
354 * @return 对应节点，越界返回 nullptr，稀疏缺失返回零节点
355 */
356 std::shared_ptr<const SymbolicNode> get(size_t r, size_t c) const {
357 if (r >= rows_ || c >= cols_) return nullptr;
358
359 size_t idx = r * cols_ + c;
360 if (std::holds_alternative<DenseStorage>(storage_)) {
361 return std::get<DenseStorage>(storage_)[idx];
362 } else {
363 const auto& sparse = std::get<SparseStorage>(storage_);
364 auto it = sparse.find(idx);
365 if (it != sparse.end()) return it->second;
366 static const std::shared_ptr<const SymbolicNode> zero =
367 lamina::detail::make_node<NumberNode>(BigInt(0));
368 return zero;
369 }
370 }

379inline std::variant<MatrixNode::DenseStorage, MatrixNode::SparseStorage> MatrixNode::create_storage_from_grid(
380 const std::vector<std::vector<std::shared_ptr<const SymbolicNode>>>& grid, size_t total_elements, size_t ncols) {
381
382 size_t non_zeros = 0;
383 for (const auto& row : grid) {
384 for (const auto& item : row) {
385 if (item && !item->is_zero()) non_zeros++;
386 }
387 }
388
389 bool use_sparse = total_elements > 0 && ((double)non_zeros / total_elements < 0.2);
390
391 if (use_sparse) {
392 SparseStorage s;
393 size_t r = 0;
394 for (const auto& row : grid) {
395 size_t c = 0;
396 for (const auto& item : row) {
397 if (item && !item->is_zero()) {
398 s[r * ncols + c] = item;
399 }
400 c++;
401 }
402 r++;
403 }
404 return s;
405 } else {
406 DenseStorage d;
407 d.reserve(total_elements);
408 for (const auto& row : grid) {
409 d.insert(d.end(), row.begin(), row.end());
410 }
411 return d;
412 }
