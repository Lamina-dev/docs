98/** A symbolic call whose function name has no built-in semantics. */
99class UninterpretedFunctionNode : public SymbolicNode {
100private:
101 LMCAS_AST_NODE_FACTORY_FRIEND;
102 const std::string name_;
103 const std::vector<std::shared_ptr<const SymbolicNode>> arguments_;
104
105 UninterpretedFunctionNode(
106 std::string name,
107 std::vector<std::shared_ptr<const SymbolicNode>> arguments)
108 : name_(std::move(name)), arguments_(std::move(arguments)) {
109 if (name_.empty()) {
110 throw std::invalid_argument("uninterpreted function name cannot be empty");
111 }
112 for (const auto& argument : arguments_) {
113 if (!argument) {
114 throw std::invalid_argument("uninterpreted function arguments cannot be null");
115 }
116 }
117 }
118
119public:
120 const std::string& name() const noexcept { return name_; }
121 const auto& arguments() const noexcept { return arguments_; }
122 int type_priority() const override { return 111; }
123
124protected:
125 std::size_t compute_hash() const override {
126 std::size_t seed = static_cast<std::size_t>(type_priority());
127 hash_combine(seed, std::hash<std::string>{}(name_));
128 for (const auto& argument : arguments_) hash_combine(seed, argument->hash());
129 return seed;
130 }
131
132 int compare_same_type(const SymbolicNode& other) const override {
133 const auto& function = static_cast<const UninterpretedFunctionNode&>(other);
134 int comparison = name_.compare(function.name_);
135 if (comparison != 0) return comparison;
136 const auto count = std::min(arguments_.size(), function.arguments_.size());
137 for (std::size_t index = 0; index < count; ++index) {
138 comparison = arguments_[index]->compare(*function.arguments_[index]);
139 if (comparison != 0) return comparison;
140 }
141 if (arguments_.size() == function.arguments_.size()) return 0;
142 return arguments_.size() < function.arguments_.size() ? -1 : 1;
143 }
144
145public:
146 void accept(LMCAS::detail::SymbolicVisitor& visitor) const override {
147 LMCAS::detail::SymbolicVisitor::DepthGuard guard(visitor);
148 visitor.visit(*this);
149 }
150
151 std::shared_ptr<const SymbolicNode> clone() const override {
152 std::vector<std::shared_ptr<const SymbolicNode>> arguments;
153 arguments.reserve(arguments_.size());
154 for (const auto& argument : arguments_) arguments.push_back(argument->clone());
155 return LMCAS::detail::make_node<UninterpretedFunctionNode>(
156 name_, std::move(arguments));
157 }
158};

168 * @brief 矩阵节点，支持稠密和稀疏两种存储方式。

170 * 当非零元素占比低于 20% 时自动使用稀疏存储，否则使用稠密存储。
171 */
172class MatrixNode : public SymbolicNode {
173public:
174 using DenseStorage = std::vector<std::shared_ptr<const SymbolicNode>>; ///< 稠密存储（按行优先展开）
175 using SparseStorage = std::map<size_t, std::shared_ptr<const SymbolicNode>>; ///< 稀疏存储（索引 -> 节点）
176
177 /**
178 * @brief 验证并返回网格的列数（要求各行列数一致）。
179 * @param grid 二维节点网格
180 * @return 列数
181 * @throw std::invalid_argument 各行列数不一致时抛出
182 */
183 static size_t validate_grid_columns(const std::vector<std::vector<std::shared_ptr<const SymbolicNode>>>& grid) {
184 if (grid.empty()) {
185 throw std::invalid_argument("MatrixNode: matrix must have at least one row");
186 }
187 size_t ncols = grid[0].size();
188 if (ncols == 0) {
189 throw std::invalid_argument("MatrixNode: matrix must have at least one column");
190 }
191 for (const auto& item : grid[0]) {
192 if (!item) {
193 throw std::invalid_argument("MatrixNode: matrix elements cannot be null");
194 }
195 }
196 for (size_t i = 1; i < grid.size(); ++i) {
197 if (grid[i].size() != ncols) {
198 throw std::invalid_argument("MatrixNode: all rows must have the same number of columns");
199 }
200 for (const auto& item : grid[i]) {
201 if (!item) {
202 throw std::invalid_argument("MatrixNode: matrix elements cannot be null");
203 }
204 }
205 }
206 return ncols;
207 }
208
209 static DenseStorage validate_dense_storage(size_t r, size_t c, DenseStorage dense) {
210 if (r == 0 || c == 0) {
211 throw std::invalid_argument("MatrixNode: matrix dimensions must be non-zero");
212 }
213 if (r > std::numeric_limits<size_t>::max() / c) {
214 throw std::length_error("MatrixNode: matrix dimensions overflow");
215 }
216 if (dense.size() != r * c) {
217 throw std::invalid_argument("MatrixNode: dense storage size does not match dimensions");
218 }
219 for (const auto& item : dense) {
220 if (!item) {
221 throw std::invalid_argument("MatrixNode: dense storage elements cannot be null");
222 }
223 }
224 return dense;
225 }
226
227 static SparseStorage validate_sparse_storage(size_t r, size_t c, SparseStorage sparse) {
228 if (r == 0 || c == 0) {
229 throw std::invalid_argument("MatrixNode: matrix dimensions must be non-zero");
230 }
231 if (r > std::numeric_limits<size_t>::max() / c) {
232 throw std::length_error("MatrixNode: matrix dimensions overflow");
233 }
234 const size_t total = r * c;
235 for (const auto& [idx, item] : sparse) {
236 if (idx >= total) {
237 throw std::invalid_argument("MatrixNode: sparse storage index is out of bounds");
238 }
239 if (!item) {
240 throw std::invalid_argument("MatrixNode: sparse storage elements cannot be null");
241 }
242 }
243 return sparse;
244 }

246 /**
247 * @brief 根据稀疏度选择存储方式，从二维网格创建存储。
248 * @param grid 二维节点网格
249 * @param total_elements 总元素数
250 * @param ncols 列数
251 * @return 稠密或稀疏存储
252 */
253 static std::variant<DenseStorage, SparseStorage> create_storage_from_grid(
254 const std::vector<std::vector<std::shared_ptr<const SymbolicNode>>>& grid, size_t total_elements, size_t ncols);
255
256private:
257 LMCAS_AST_NODE_FACTORY_FRIEND;
258
259 const size_t rows_;
260 const size_t cols_;
261 const std::variant<DenseStorage, SparseStorage> storage_;
262
263 MatrixNode(const std::vector<std::vector<std::shared_ptr<const SymbolicNode>>>& grid)
264 : rows_(grid.size()),
265 cols_(validate_grid_columns(grid)),
266 storage_(create_storage_from_grid(grid, rows_ * cols_, cols_)) {}
267
268 MatrixNode(size_t r, size_t c, DenseStorage dense)
269 : rows_(r), cols_(c), storage_(validate_dense_storage(r, c, std::move(dense))) {}
270
271 MatrixNode(size_t r, size_t c, SparseStorage sparse)
272 : rows_(r), cols_(c), storage_(validate_sparse_storage(r, c, std::move(sparse))) {}
273
274public:
275 size_t rows() const noexcept { return rows_; }
276 size_t cols() const noexcept { return cols_; }
277 const std::variant<DenseStorage, SparseStorage>& storage() const noexcept {
278 return storage_;
279 }
280
281 int type_priority() const override { return 6; }
282
283protected:
284 std::size_t compute_hash() const override {
285 std::size_t seed = 0;
286 hash_combine(seed, type_priority());
287 hash_combine(seed, rows_);
288 hash_combine(seed, cols_);
289
290 if (std::holds_alternative<DenseStorage>(storage_)) {
291 const auto& dense = std::get<DenseStorage>(storage_);
292 for (size_t i = 0; i < dense.size(); ++i) {
293 if (dense[i] && !dense[i]->is_zero()) {
294 hash_combine(seed, i);
295 hash_combine(seed, dense[i]->hash());
296 }
297 }
298 } else {
299 const auto& sparse = std::get<SparseStorage>(storage_);
300 for (const auto& [idx, val] : sparse) {
301
302 if (val && !val->is_zero()) {
303 hash_combine(seed, idx);
304 hash_combine(seed, val->hash());
305 }
306 }
307 }
308 return seed;
309 }

311 int compare_same_type(const SymbolicNode& other) const override {
312 const auto& o = static_cast<const MatrixNode&>(other);
313 if (rows_ != o.rows_) return rows_ < o.rows_ ? -1 : 1;
314 if (cols_ != o.cols_) return cols_ < o.cols_ ? -1 : 1;
315
316 for (size_t r = 0; r < rows_; ++r) {
317 for (size_t c = 0; c < cols_; ++c) {
318 auto v1 = this->get(r, c);
319 auto v2 = o.get(r, c);
320
321 bool z1 = !v1 || v1->is_zero();
322 bool z2 = !v2 || v2->is_zero();
323
324 if (z1 && z2) continue;
325 if (z1) return -1;
326 if (z2) return 1;
327
328 int cmp = v1->compare(*v2);
329 if (cmp != 0) return cmp;
330 }
331 }
332 return 0;
333 }

336 void accept(LMCAS::detail::SymbolicVisitor& visitor) const override { LMCAS::detail::SymbolicVisitor::DepthGuard guard(visitor); visitor.visit(*this); }

338 std::shared_ptr<const SymbolicNode> clone() const override {
339 if (std::holds_alternative<DenseStorage>(storage_)) {
340 const auto& dense = std::get<DenseStorage>(storage_);
341 DenseStorage new_dense;
342 new_dense.reserve(dense.size());
343 for (const auto& e : dense) {
344 new_dense.push_back(e->clone());
345 }
346 return LMCAS::detail::make_node<MatrixNode>(rows_, cols_, std::move(new_dense));
347 } else {
348 const auto& sparse = std::get<SparseStorage>(storage_);
349 SparseStorage new_sparse;
350 for(const auto& [idx, node] : sparse) {
351 new_sparse[idx] = node->clone();
352 }
353 return LMCAS::detail::make_node<MatrixNode>(rows_, cols_, std::move(new_sparse));
354 }

357 /**
358 * @brief 获取指定位置的矩阵元素。
359 * @param r 行索引（从 0 开始）
360 * @param c 列索引（从 0 开始）
361 * @return 对应节点，越界返回 nullptr，稀疏缺失返回零节点
362 */
363 std::shared_ptr<const SymbolicNode> get(size_t r, size_t c) const {
364 if (r >= rows_ || c >= cols_) return nullptr;
365
366 size_t idx = r * cols_ + c;
367 if (std::holds_alternative<DenseStorage>(storage_)) {
368 return std::get<DenseStorage>(storage_)[idx];
369 } else {
370 const auto& sparse = std::get<SparseStorage>(storage_);
371 auto it = sparse.find(idx);
372 if (it != sparse.end()) return it->second;
373 static const std::shared_ptr<const SymbolicNode> zero =
374 LMCAS::detail::make_node<NumberNode>(BigInt(0));
375 return zero;
376 }
377 }

386inline bool MultiplyNode::contains_matrix(const SymbolicNode& node) {
387 if (dynamic_cast<const MatrixNode*>(&node)) return true;
388 if (const auto* power = dynamic_cast<const PowerNode*>(&node)) {
389 return contains_matrix(*power->base());
390 }
391 const std::vector<std::shared_ptr<const SymbolicNode>>* children = nullptr;
392 if (const auto* sum = dynamic_cast<const AddNode*>(&node)) {
393 children = &sum->operands();
394 } else if (const auto* product = dynamic_cast<const MultiplyNode*>(&node)) {
395 children = &product->operands();
396 } else if (const auto* function = dynamic_cast<const FunctionNode*>(&node)) {
397 children = &function->arguments();
398 } else if (const auto* function = dynamic_cast<const UninterpretedFunctionNode*>(&node)) {
399 children = &function->arguments();
400 }
401 return children && std::any_of(
402 children->begin(), children->end(),
403 [](const auto& child) { return contains_matrix(*child); });
404}

406inline std::variant<MatrixNode::DenseStorage, MatrixNode::SparseStorage> MatrixNode::create_storage_from_grid(
407 const std::vector<std::vector<std::shared_ptr<const SymbolicNode>>>& grid, size_t total_elements, size_t ncols) {
408
409 size_t non_zeros = 0;
410 for (const auto& row : grid) {
411 for (const auto& item : row) {
412 if (item && !item->is_zero()) non_zeros++;
413 }
414 }
415
416 bool use_sparse = total_elements > 0 && ((double)non_zeros / total_elements < 0.2);
417
418 if (use_sparse) {
419 SparseStorage s;
420 size_t r = 0;
421 for (const auto& row : grid) {
422 size_t c = 0;
423 for (const auto& item : row) {
424 if (item && !item->is_zero()) {
425 s[r * ncols + c] = item;
426 }
427 c++;
428 }
429 r++;
430 }
431 return s;
432 } else {
433 DenseStorage d;
434 d.reserve(total_elements);
435 for (const auto& row : grid) {
436 d.insert(d.end(), row.begin(), row.end());
437 }
438 return d;
439 }
