7InferenceTriboolResult InferenceEngine::infer_add_sign_checked(const void* opaque_node, Sign target) const {
8 if (!opaque_node) {
9 return InferenceTriboolResult::failure(
10 CasErrc::InvalidArgument,
11 "addition inference node must not be null",
12 "infer_add_sign_checked");
13 }
14
15 try {
16 const auto& node = *static_cast<const AddNode*>(opaque_node);
17 if (node.operands().empty()) return InferenceTriboolResult::success(Tribool::Unknown);
18
19 // First, try subtraction pattern detection for 2-operand AddNodes
20 if (node.operands().size() == 2) {
21 // Check if one operand is a negated term (MultiplyNode with -1 coefficient)
22 bool has_negated = false;
23 for (const auto& operand : node.operands()) {
24 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(operand);
25 if (mul && mul->operands().size() == 2) {
26 for (const auto& mul_op : mul->operands()) {
27 auto num = std::dynamic_pointer_cast<const NumberNode>(mul_op);
28 if (num) {
29 bool is_neg_one = false;
30 if (std::holds_alternative<BigInt>(num->value())) {
31 is_neg_one = (std::get<BigInt>(num->value()) == BigInt(-1));
32 } else if (std::holds_alternative<Rational>(num->value())) {
33 is_neg_one = (std::get<Rational>(num->value()) == Rational(-1));
34 } else if (std::holds_alternative<lmmc_real_t>(num->value())) {
35 lmmc_real_t v = std::get<lmmc_real_t>(num->value());
36 is_neg_one = (std::isfinite(v) && v == -1.0);
37 }
38 if (is_neg_one) {
39 has_negated = true;
40 break;
41 }
42 }
43 }
44 if (has_negated) break;
45 }
46 }
47 if (has_negated) {
48 Tribool sub_result = detail::propagate_result(infer_subtraction_sign_checked(&node, target));
49 if (sub_result != Tribool::Unknown) {
50 return InferenceTriboolResult::success(sub_result);
51 }
52 }
53 }
54
55 // For sign inference on addition:
56 // - If all operands have the target sign property -> sum has that property
57 // - If any operand has Unknown -> result is Unknown
58 // - If operands have mixed definite signs -> result is Unknown
59
60 bool all_have_property = true;
61
62 for (const auto& operand : node.operands()) {
63 auto op_expr = lamina::detail::expression_from_node(operand);
64 Tribool op_result = Tribool::Unknown;
65 switch (target) {
66 case Sign::Positive:
67 case Sign::Negative:
68 case Sign::NonNegative:
69 case Sign::NonPositive:
70 op_result = detail::propagate_result(query_sign_of_checked(op_expr, target));
71 break;
72 default:
73 return InferenceTriboolResult::success(Tribool::Unknown);
74 }
75
76 if (op_result == Tribool::Unknown) {
77 return InferenceTriboolResult::success(Tribool::Unknown);
78 }
79 if (op_result == Tribool::False) {
80 all_have_property = false;
81 }
82 }
83
84 if (all_have_property) {
85 return InferenceTriboolResult::success(Tribool::True);
86 }
87
88 return InferenceTriboolResult::success(Tribool::Unknown);
89 } catch (const detail::ResultPropagation& ex) {
90 return InferenceTriboolResult::failure(ex.error());
91 } catch (const std::bad_alloc&) {
92 return InferenceTriboolResult::failure(
93 CasErrc::ResourceLimit,
94 "addition sign inference allocation failed",
95 "infer_add_sign_checked");
96 } catch (const std::exception& ex) {
97 return InferenceTriboolResult::failure(
98 CasErrc::InternalInvariant,
99 ex.what(),
100 "infer_add_sign_checked");
101 }

107InferenceTriboolResult InferenceEngine::infer_sign_from_relations_checked(
108 const SymbolicExpr& expr, Sign target) const {
109 if (!lamina::detail::node(expr)) {
110 return InferenceTriboolResult::failure(
111 CasErrc::InvalidArgument,
112 "relation sign inference expression must not be null",
113 "infer_sign_from_relations_checked");
114 }
115
116 try {
117 const auto& rel_store = impl_->ctx.current_relations();
118 auto zero_expr = lamina::detail::expression_from_node(
119 lamina::detail::make_node<NumberNode>(BigInt(0)));
120
121 if (auto add = std::dynamic_pointer_cast<const AddNode>(lamina::detail::node(expr))) {
122 if (add->operands().empty()) return InferenceTriboolResult::success(Tribool::Unknown);
123
124 bool all_gt_zero = true;
125 bool all_geq_zero = true;
126
127 for (const auto& operand : add->operands()) {
128 auto op_expr = lamina::detail::expression_from_node(operand);
129 bool op_gt = rel_store.has_relation(op_expr, zero_expr, RelationalNode::Op::GT);
130 bool op_geq = rel_store.has_relation(op_expr, zero_expr, RelationalNode::Op::GEQ);
131
132 if (!op_gt) {
133 if (auto var = std::dynamic_pointer_cast<const VariableNode>(operand)) {
134 op_gt = impl_->ctx.current_properties().has_sign(var->name(), Sign::Positive);
135 } else if (auto num = std::dynamic_pointer_cast<const NumberNode>(operand)) {
136 op_gt = num->is_positive();
137 }
138 }
139 if (!op_geq && !op_gt) {
140 if (auto var = std::dynamic_pointer_cast<const VariableNode>(operand)) {
141 op_geq = impl_->ctx.current_properties().has_sign(var->name(), Sign::NonNegative);
142 } else if (auto num = std::dynamic_pointer_cast<const NumberNode>(operand)) {
143 op_geq = num->is_zero() || num->is_positive();
144 }
145 }
146
147 if (!op_gt) all_gt_zero = false;
148 if (!op_gt && !op_geq) all_geq_zero = false;
149 }
150
151 if (target == Sign::Positive && all_gt_zero) {
152 return InferenceTriboolResult::success(Tribool::True);
153 }
154 if (target == Sign::NonNegative && all_geq_zero) {
155 return InferenceTriboolResult::success(Tribool::True);
156 }
157 }
158
159 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(lamina::detail::node(expr))) {
160 if (mul->operands().empty()) return InferenceTriboolResult::success(Tribool::Unknown);
161
162 if (target == Sign::Positive) {
163 bool all_gt_zero = true;
164
165 for (const auto& operand : mul->operands()) {
166 auto op_expr = lamina::detail::expression_from_node(operand);
167 bool op_gt = rel_store.has_relation(op_expr, zero_expr, RelationalNode::Op::GT);
168
169 if (!op_gt) {
170 if (auto var = std::dynamic_pointer_cast<const VariableNode>(operand)) {
171 op_gt = impl_->ctx.current_properties().has_sign(var->name(), Sign::Positive);
172 } else if (auto num = std::dynamic_pointer_cast<const NumberNode>(operand)) {
173 op_gt = num->is_positive();
174 }
175 }
176
177 if (!op_gt) {
178 all_gt_zero = false;
179 break;
180 }
181 }
182
183 if (all_gt_zero) return InferenceTriboolResult::success(Tribool::True);
184 }
185 }
186
187 if (target == Sign::Positive) {
188 const auto& relations = rel_store.get_relations();
189 for (const auto& rel : relations) {
190 if (rel.op != RelationalNode::Op::GT) continue;
191
192 if (!lamina::detail::node(rel.lhs) || !lamina::detail::node(expr)) continue;
193 if (!lamina::detail::node(rel.lhs)->equals(*lamina::detail::node(expr))) continue;
194 if (!lamina::detail::node(rel.rhs)) continue;
195
196 if (auto rhs_num = std::dynamic_pointer_cast<const NumberNode>(lamina::detail::node(rel.rhs))) {
197 if (rhs_num->is_zero() || rhs_num->is_positive()) {
198 return InferenceTriboolResult::success(Tribool::True);
199 }
200 }
201
202 if (auto rhs_var = std::dynamic_pointer_cast<const VariableNode>(lamina::detail::node(rel.rhs))) {
203 if (impl_->ctx.current_properties().has_sign(rhs_var->name(), Sign::NonNegative)) {
204 return InferenceTriboolResult::success(Tribool::True);
205 }
206 }
207
208 if (rel_store.has_relation(rel.rhs, zero_expr, RelationalNode::Op::GEQ) ||
209 rel_store.has_relation(rel.rhs, zero_expr, RelationalNode::Op::GT)) {
210 return InferenceTriboolResult::success(Tribool::True);
211 }
212 }
213 }
214
215 return InferenceTriboolResult::success(Tribool::Unknown);
216 } catch (const std::bad_alloc&) {
217 return InferenceTriboolResult::failure(
218 CasErrc::ResourceLimit,
219 "relation sign inference allocation failed",
220 "infer_sign_from_relations_checked");
221 } catch (const std::exception& ex) {
222 return InferenceTriboolResult::failure(
223 CasErrc::InternalInvariant,
224 ex.what(),
225 "infer_sign_from_relations_checked");
226 }

261 Tribool int_result = detail::propagate_result(query_integer_checked(op_expr));
262 if (int_result == Tribool::True) continue;
263 }
264 return InferenceTriboolResult::success(Tribool::Unknown);
265 }
266 }
267
268 return InferenceTriboolResult::success(Tribool::True);
269 } catch (const detail::ResultPropagation& ex) {
270 return InferenceTriboolResult::failure(ex.error());
271 } catch (const std::bad_alloc&) {
272 return InferenceTriboolResult::failure(
273 CasErrc::ResourceLimit,
274 "addition domain inference allocation failed",
275 "infer_add_domain_checked");
276 } catch (const std::exception& ex) {

278 CasErrc::InternalInvariant,
279 ex.what(),
280 "infer_add_domain_checked");

286
287InferenceTriboolResult InferenceEngine::infer_division_sign_checked(const void* opaque_node, Sign target) const {
288 if (!opaque_node) {
289 return InferenceTriboolResult::failure(
290 CasErrc::InvalidArgument,
291 "division inference node must not be null",
292 "infer_division_sign_checked");
293 }
294
295 try {
296 const auto& node = *static_cast<const MultiplyNode*>(opaque_node);
297 // Division pattern: MultiplyNode with exactly 2 children where one is PowerNode(den, -1)
298 if (node.operands().size() != 2) return InferenceTriboolResult::success(Tribool::Unknown);
299
300 // Find the PowerNode with exponent -1 (denominator) and the other operand (numerator)
301 std::shared_ptr<const SymbolicNode> numerator_node;
302 std::shared_ptr<const SymbolicNode> denominator_node;
303
304 for (const auto& operand : node.operands()) {
305 auto pow_node = std::dynamic_pointer_cast<const PowerNode>(operand);
306 if (pow_node && is_exponent_neg_one(pow_node->exponent())) {
307 denominator_node = pow_node->base();
308 } else {
309 numerator_node = operand;
310 }
311 }
312
313 // Must have exactly one denominator and one numerator
314 if (!denominator_node || !numerator_node) return InferenceTriboolResult::success(Tribool::Unknown);
315
316 auto num_expr = lamina::detail::expression_from_node(numerator_node);
317 auto den_expr = lamina::detail::expression_from_node(denominator_node);
318 // Check if denominator is zero -> return Unknown for all sign queries
319 Tribool den_nn = detail::propagate_result(query_sign_of_checked(den_expr, Sign::NonNegative));
320 Tribool den_np = detail::propagate_result(query_sign_of_checked(den_expr, Sign::NonPositive));
321 if (den_nn == Tribool::True && den_np == Tribool::True) {
322 // Denominator is zero
323 return InferenceTriboolResult::success(Tribool::Unknown);
324 }
325
326 // Determine numerator sign
327 Tribool num_pos = detail::propagate_result(query_sign_of_checked(num_expr, Sign::Positive));
328 Tribool num_neg = detail::propagate_result(query_sign_of_checked(num_expr, Sign::Negative));
329
330 // Determine denominator sign
331 Tribool den_pos = detail::propagate_result(query_sign_of_checked(den_expr, Sign::Positive));
332 Tribool den_neg = detail::propagate_result(query_sign_of_checked(den_expr, Sign::Negative));
333
334 // If denominator sign is unknown, return Unknown
335 if (den_pos != Tribool::True && den_neg != Tribool::True) {
336 return InferenceTriboolResult::success(Tribool::Unknown);
337 }
338
339 // If numerator sign is unknown, return Unknown
340 if (num_pos != Tribool::True && num_neg != Tribool::True) {
341 // Check if numerator is zero
342 Tribool num_nn = detail::propagate_result(query_sign_of_checked(num_expr, Sign::NonNegative));
343 Tribool num_np_check = detail::propagate_result(query_sign_of_checked(num_expr, Sign::NonPositive));
344 if (num_nn == Tribool::True && num_np_check == Tribool::True) {
345 // Numerator is zero -> result is zero
346 switch (target) {
347 case Sign::Zero: return InferenceTriboolResult::success(Tribool::True);
348 case Sign::NonNegative: return InferenceTriboolResult::success(Tribool::True);
349 case Sign::NonPositive: return InferenceTriboolResult::success(Tribool::True);
350 case Sign::Positive: return InferenceTriboolResult::success(Tribool::False);
351 case Sign::Negative: return InferenceTriboolResult::success(Tribool::False);
352 case Sign::NonZero: return InferenceTriboolResult::success(Tribool::False);
353 }
354 }
355 return InferenceTriboolResult::success(Tribool::Unknown);
356 }
357
358 // Apply sign table for division
359 // positive / positive -> positive
360 // negative / negative -> positive
361 // positive / negative -> negative
362 // negative / positive -> negative
363 bool result_positive = (num_pos == Tribool::True && den_pos == Tribool::True) ||
364 (num_neg == Tribool::True && den_neg == Tribool::True);
365 bool result_negative = (num_pos == Tribool::True && den_neg == Tribool::True) ||
366 (num_neg == Tribool::True && den_pos == Tribool::True);
367
368 if (result_positive) {
369 switch (target) {
370 case Sign::Positive: return InferenceTriboolResult::success(Tribool::True);
371 case Sign::NonNegative: return InferenceTriboolResult::success(Tribool::True);
372 case Sign::NonZero: return InferenceTriboolResult::success(Tribool::True);
373 case Sign::Negative: return InferenceTriboolResult::success(Tribool::False);
374 case Sign::NonPositive: return InferenceTriboolResult::success(Tribool::False);
375 case Sign::Zero: return InferenceTriboolResult::success(Tribool::False);
376 }
377 }
378
379 if (result_negative) {
380 switch (target) {
381 case Sign::Negative: return InferenceTriboolResult::success(Tribool::True);
382 case Sign::NonPositive: return InferenceTriboolResult::success(Tribool::True);
383 case Sign::NonZero: return InferenceTriboolResult::success(Tribool::True);
384 case Sign::Positive: return InferenceTriboolResult::success(Tribool::False);
385 case Sign::NonNegative: return InferenceTriboolResult::success(Tribool::False);
386 case Sign::Zero: return InferenceTriboolResult::success(Tribool::False);
387 }
388 }
389
390 return InferenceTriboolResult::success(Tribool::Unknown);
391 } catch (const detail::ResultPropagation& ex) {
392 return InferenceTriboolResult::failure(ex.error());
393 } catch (const std::bad_alloc&) {
394 return InferenceTriboolResult::failure(
395 CasErrc::ResourceLimit,
396 "division sign inference allocation failed",
397 "infer_division_sign_checked");
398 } catch (const std::exception& ex) {
399 return InferenceTriboolResult::failure(
400 CasErrc::InternalInvariant,
401 ex.what(),
402 "infer_division_sign_checked");
403 }

408
409InferenceTriboolResult InferenceEngine::infer_multiply_sign_checked(const void* opaque_node, Sign target) const {
410 if (!opaque_node) {
411 return InferenceTriboolResult::failure(
412 CasErrc::InvalidArgument,
413 "multiplication inference node must not be null",
414 "infer_multiply_sign_checked");
415 }
416
417 try {
418 const auto& node = *static_cast<const MultiplyNode*>(opaque_node);
419 if (node.operands().empty()) return InferenceTriboolResult::success(Tribool::Unknown);
420
421 // Check for division pattern: exactly 2 operands with one being PowerNode(den, -1)
422 if (node.operands().size() == 2) {
423 bool has_power_neg_one = false;
424 for (const auto& operand : node.operands()) {
425 auto pow_node = std::dynamic_pointer_cast<const PowerNode>(operand);
426 if (pow_node && is_exponent_neg_one(pow_node->exponent())) {
427 has_power_neg_one = true;
428 break;
429 }
430 }
431 if (has_power_neg_one) {
432 Tribool div_result = detail::propagate_result(infer_division_sign_checked(&node, target));
433 if (div_result != Tribool::Unknown) {
434 return InferenceTriboolResult::success(div_result);
435 }
436 }
437 }
438
439 // Step 1: Check if any operand is Zero -> product is Zero
440 bool has_zero = false;
441 for (const auto& operand : node.operands()) {
442 auto op_expr = lamina::detail::expression_from_node(operand);
443 // Check if operand is zero (both nonnegative and nonpositive)
444 Tribool nn = detail::propagate_result(query_sign_of_checked(op_expr, Sign::NonNegative));
445 Tribool np = detail::propagate_result(query_sign_of_checked(op_expr, Sign::NonPositive));
446 if (nn == Tribool::True && np == Tribool::True) {
447 has_zero = true;
448 break;
449 }
450 // Also check the node directly
451 if (operand->is_zero()) {
452 has_zero = true;
453 break;
454 }
455 }
456
457 if (has_zero) {
458 // Product is Zero
459 switch (target) {
460 case Sign::Zero: return InferenceTriboolResult::success(Tribool::True);
461 case Sign::NonNegative: return InferenceTriboolResult::success(Tribool::True);
462 case Sign::NonPositive: return InferenceTriboolResult::success(Tribool::True);
463 case Sign::Positive: return InferenceTriboolResult::success(Tribool::False);
464 case Sign::Negative: return InferenceTriboolResult::success(Tribool::False);
465 case Sign::NonZero: return InferenceTriboolResult::success(Tribool::False);
466 }
467 }
468
469 // Step 2: Handle NonZero query -- all NonZero -> product is NonZero
470 if (target == Sign::NonZero) {
471 for (const auto& operand : node.operands()) {
472 auto op_expr = lamina::detail::expression_from_node(operand);
473 Tribool nz = detail::propagate_result(query_sign_of_checked(op_expr, Sign::NonZero));
474 if (nz != Tribool::True) {
475 // Also check if operand is positive or negative (both imply nonzero)
476 Tribool pos = detail::propagate_result(query_sign_of_checked(op_expr, Sign::Positive));
477 if (pos == Tribool::True) continue;
478 Tribool neg = detail::propagate_result(query_sign_of_checked(op_expr, Sign::Negative));
479 if (neg == Tribool::True) continue;
480 return InferenceTriboolResult::success(Tribool::Unknown);
481 }
482 }
483 return InferenceTriboolResult::success(Tribool::True);
484 }
485
486 // Step 3: Determine sign by counting negatives
487 // We need each operand to have a definite sign (Positive or Negative)
488 // or at least NonNegative/NonPositive for weaker results.
489 int negative_count = 0;
490 bool all_definite = true; // all are Positive or Negative
491 bool all_nonneg_or_nonpos = true; // all are at least NonNeg or NonPos
492 bool has_unknown = false;
493
494 for (const auto& operand : node.operands()) {
495 auto op_expr = lamina::detail::expression_from_node(operand);
496 Tribool pos = detail::propagate_result(query_sign_of_checked(op_expr, Sign::Positive));
497 Tribool neg = detail::propagate_result(query_sign_of_checked(op_expr, Sign::Negative));
498
499 if (pos == Tribool::True) {
500 // Positive operand -- doesn't change sign
501 continue;
502 } else if (neg == Tribool::True) {
503 // Negative operand -- flips sign
504 negative_count++;
505 continue;
506 }
507
508 // Not definitively Positive or Negative
509 all_definite = false;
510
511 // Check weaker properties
512 Tribool nn = detail::propagate_result(query_sign_of_checked(op_expr, Sign::NonNegative));
513 Tribool np = detail::propagate_result(query_sign_of_checked(op_expr, Sign::NonPositive));
514
515 if (nn == Tribool::True) {
516 // NonNegative but not Positive (could be zero)
517 continue;
518 } else if (np == Tribool::True) {
519 // NonPositive but not Negative (could be zero)
520 negative_count++;
521 continue;
522 }
523
524 // Unknown sign
525 all_nonneg_or_nonpos = false;
526 has_unknown = true;
527 }
528
529 if (has_unknown) {
530 return InferenceTriboolResult::success(Tribool::Unknown);
531 }
532
533 // Determine result based on parity of negative count
534 bool even_negatives = (negative_count % 2 == 0);
535
536 if (all_definite) {
537 // All operands are definitively Positive or Negative
538 switch (target) {
539 case Sign::Positive:
540 return InferenceTriboolResult::success(even_negatives ? Tribool::True : Tribool::False);
541 case Sign::Negative:
542 return InferenceTriboolResult::success(even_negatives ? Tribool::False : Tribool::True);
543 case Sign::NonNegative:
544 return InferenceTriboolResult::success(even_negatives ? Tribool::True : Tribool::False);
545 case Sign::NonPositive:
546 return InferenceTriboolResult::success(even_negatives ? Tribool::False : Tribool::True);
547 case Sign::NonZero:
548 return InferenceTriboolResult::success(Tribool::True); // product of nonzero values is nonzero
549 default:
550 return InferenceTriboolResult::success(Tribool::Unknown);
551 }
552 }
553
554 if (all_nonneg_or_nonpos) {
555 switch (target) {
556 case Sign::NonNegative:
557 return InferenceTriboolResult::success(even_negatives ? Tribool::True : Tribool::False);
558 case Sign::NonPositive:
559 return InferenceTriboolResult::success(even_negatives ? Tribool::False : Tribool::True);
560 case Sign::Positive:
561 return InferenceTriboolResult::success(Tribool::Unknown); // could be zero
562 case Sign::Negative:
563 return InferenceTriboolResult::success(Tribool::Unknown); // could be zero
564 default:
565 return InferenceTriboolResult::success(Tribool::Unknown);
566 }
567 }
568
569 return InferenceTriboolResult::success(Tribool::Unknown);
570 } catch (const detail::ResultPropagation& ex) {
571 return InferenceTriboolResult::failure(ex.error());
572 } catch (const std::bad_alloc&) {
573 return InferenceTriboolResult::failure(
574 CasErrc::ResourceLimit,
575 "multiplication sign inference allocation failed",
576 "infer_multiply_sign_checked");
577 } catch (const std::exception& ex) {
578 return InferenceTriboolResult::failure(
579 CasErrc::InternalInvariant,
580 ex.what(),
581 "infer_multiply_sign_checked");
582 }

587
588InferenceTriboolResult InferenceEngine::infer_multiply_domain_checked(const void* opaque_node, Domain target) const {
589 if (!opaque_node) {
590 return InferenceTriboolResult::failure(
591 CasErrc::InvalidArgument,
592 "multiplication domain inference node must not be null",
593 "infer_multiply_domain_checked");
594 }
595
596 try {
597 const auto& node = *static_cast<const MultiplyNode*>(opaque_node);
598 if (node.operands().empty()) return InferenceTriboolResult::success(Tribool::Unknown);
599
600 if (target == Domain::Integer) {
601 for (const auto& operand : node.operands()) {
602 auto op_expr = lamina::detail::expression_from_node(operand);
603 Tribool result = detail::propagate_result(query_integer_checked(op_expr));
604 if (result != Tribool::True) {
605 return InferenceTriboolResult::success(Tribool::Unknown);
606 }
607 }
608 return InferenceTriboolResult::success(Tribool::True);
609 }
610
611 if (target == Domain::Real) {
612 for (const auto& operand : node.operands()) {
613 auto op_expr = lamina::detail::expression_from_node(operand);
614 Tribool real_result = detail::propagate_result(query_real_checked(op_expr));
615 if (real_result == Tribool::True) continue;
616

643/**
644 * @brief Helper: check if a NumberNode holds an integer value.
645 * Works for BigInt (always integer), Rational (if denominator == 1),
646 * and double (if finite and equal to its floor).
647 */
648bool is_integer_number(const NumberNode& num) {
649 if (std::holds_alternative<BigInt>(num.value())) {
650 return true;
651 }
652 if (std::holds_alternative<Rational>(num.value())) {
653 return std::get<Rational>(num.value()).is_integer();
654 }
655 // double case
656 double v = std::get<lmmc_real_t>(num.value());
657 return std::isfinite(v) && v == std::floor(v);
658}
659
660/**
661 * @brief Helper: check if a NumberNode holds an even integer value.
662 */
663bool is_even_integer_number(const NumberNode& num) {
664 if (!is_integer_number(num)) return false;
665
666 if (std::holds_alternative<BigInt>(num.value())) {
667 return std::get<BigInt>(num.value()).is_even();
668 }
669 if (std::holds_alternative<Rational>(num.value())) {
670 // Rational with denominator 1 - check numerator
671 BigInt n = std::get<Rational>(num.value()).get_numerator();
672 return n.is_even();
673 }
674 // double case - guard against overflow for very large values
675 double v = std::get<lmmc_real_t>(num.value());
676 if (std::fabs(v) >= static_cast<double>(LLONG_MAX)) {
677 return std::fmod(v, 2.0) == 0.0;
678 }
679 long long iv = static_cast<long long>(v);
680 return (iv % 2) == 0;
681}
682
683/**
684 * @brief Helper: check if a NumberNode holds a positive integer value (> 0).
685 */
686bool is_positive_integer_number(const NumberNode& num) {
687 if (!is_integer_number(num)) return false;
688
689 if (std::holds_alternative<BigInt>(num.value())) {
690 const auto& b = std::get<BigInt>(num.value());
691 return !b.IsNegative() && !(b == BigInt(0));
692 }
693 if (std::holds_alternative<Rational>(num.value())) {
694 BigInt n = std::get<Rational>(num.value()).get_numerator();
695 return !n.IsNegative() && !(n == BigInt(0));
696 }
697 // double case
698 double v = std::get<lmmc_real_t>(num.value());
699 return v > 0.0;
700}

702/**
703 * @brief Helper: check if a NumberNode holds the value 0.
704 */
705bool is_zero_number(const NumberNode& num) {
706 return num.is_zero();

863InferenceTriboolResult InferenceEngine::infer_function_property_checked(const void* opaque_node, Sign target) const {
864 if (!opaque_node) {
865 return InferenceTriboolResult::failure(
866 CasErrc::InvalidArgument,
867 "function inference node must not be null",
868 "infer_function_property_checked");
869 }
870
871 try {
872 const auto& node = *static_cast<const FunctionNode*>(opaque_node);
873 // Need at least one argument for all built-in function rules
874 if (node.arguments().empty()) return InferenceTriboolResult::success(Tribool::Unknown);
875
876 auto arg_expr = lamina::detail::expression_from_node(node.arguments()[0]);
877 switch (node.type()) {
878 case FunctionNode::FuncType::Exp: {
879 // exp(Real) -> Positive
880 // exp is always positive for real arguments
881 Tribool arg_real = detail::propagate_result(query_domain_of_checked(arg_expr, Domain::Real));
882 if (arg_real != Tribool::True) return InferenceTriboolResult::success(Tribool::Unknown);
883
884 switch (target) {
885 case Sign::Positive: return InferenceTriboolResult::success(Tribool::True);
886 case Sign::NonNegative: return InferenceTriboolResult::success(Tribool::True);
887 case Sign::NonZero: return InferenceTriboolResult::success(Tribool::True);
888 case Sign::Negative: return InferenceTriboolResult::success(Tribool::False);
889 case Sign::NonPositive: return InferenceTriboolResult::success(Tribool::False);
890 case Sign::Zero: return InferenceTriboolResult::success(Tribool::False);
891 }
892 return InferenceTriboolResult::success(Tribool::Unknown);
893 }
894
895 case FunctionNode::FuncType::Sin: {
896 /// sin(Real) 属于 Real 且值域为 [-1,1];其符号由具体参数值决定.
897 Tribool arg_real = detail::propagate_result(query_domain_of_checked(arg_expr, Domain::Real));
898 if (arg_real != Tribool::True) return InferenceTriboolResult::success(Tribool::Unknown);
899
900 return InferenceTriboolResult::success(Tribool::Unknown);
901 }
902
903 case FunctionNode::FuncType::Cos: {
904 /// cos(Real) 属于 Real 且值域为 [-1,1];其符号由具体参数值决定.
905 Tribool arg_real = detail::propagate_result(query_domain_of_checked(arg_expr, Domain::Real));
906 if (arg_real != Tribool::True) return InferenceTriboolResult::success(Tribool::Unknown);
907
908 return InferenceTriboolResult::success(Tribool::Unknown);
909 }
910
911 case FunctionNode::FuncType::Abs: {
912 Tribool arg_real = detail::propagate_result(query_domain_of_checked(arg_expr, Domain::Real));
913 if (arg_real != Tribool::True) return InferenceTriboolResult::success(Tribool::Unknown);
914
915 // Check if argument is known Positive, Negative, or NonZero
916 Tribool arg_pos = detail::propagate_result(query_sign_of_checked(arg_expr, Sign::Positive));
917 Tribool arg_neg = detail::propagate_result(query_sign_of_checked(arg_expr, Sign::Negative));
918 Tribool arg_nz = detail::propagate_result(query_sign_of_checked(arg_expr, Sign::NonZero));
919
920 // If argument is Positive or Negative (or NonZero), abs is Positive
921 bool abs_is_positive = (arg_pos == Tribool::True) ||
922 (arg_neg == Tribool::True) ||
923 (arg_nz == Tribool::True);
924
925 switch (target) {
926 case Sign::NonNegative: return InferenceTriboolResult::success(Tribool::True);
927 case Sign::Negative: return InferenceTriboolResult::success(Tribool::False);
928 case Sign::Positive:
929 return InferenceTriboolResult::success(abs_is_positive ? Tribool::True : Tribool::Unknown);
930 case Sign::NonZero:
931 return InferenceTriboolResult::success(abs_is_positive ? Tribool::True : Tribool::Unknown);
932 case Sign::NonPositive:
933 return InferenceTriboolResult::success(abs_is_positive ? Tribool::False : Tribool::Unknown);
934 case Sign::Zero:
935 return InferenceTriboolResult::success(abs_is_positive ? Tribool::False : Tribool::Unknown);
936 }
937 return InferenceTriboolResult::success(Tribool::Unknown);
938 }
939
940 case FunctionNode::FuncType::Ln: {
941 // ln(Positive) -> Real
942 /// ln 在 x>1,0<x<1 与 x=1 时分别为正,负与零,通用符号保持 Unknown.
943 Tribool arg_positive = detail::propagate_result(query_sign_of_checked(arg_expr, Sign::Positive));
944 if (arg_positive != Tribool::True) return InferenceTriboolResult::success(Tribool::Unknown);
945
946 // ln(Positive) has no definite sign property
947 return InferenceTriboolResult::success(Tribool::Unknown);
948 }
949
950 case FunctionNode::FuncType::Sqrt: {
951 // sqrt(NonNegative) -> NonNegative, Real
952 Tribool arg_nonneg = detail::propagate_result(query_sign_of_checked(arg_expr, Sign::NonNegative));
953 if (arg_nonneg != Tribool::True) return InferenceTriboolResult::success(Tribool::Unknown);
954
955 switch (target) {
956 case Sign::NonNegative: return InferenceTriboolResult::success(Tribool::True);
957 case Sign::Negative: return InferenceTriboolResult::success(Tribool::False);
958 case Sign::NonPositive: return InferenceTriboolResult::success(Tribool::Unknown); // sqrt could be zero
959 case Sign::Positive: return InferenceTriboolResult::success(Tribool::Unknown); // sqrt(0) = 0
960 case Sign::Zero: return InferenceTriboolResult::success(Tribool::Unknown); // depends on argument
961 case Sign::NonZero: return InferenceTriboolResult::success(Tribool::Unknown); // depends on argument
962 }
963 return InferenceTriboolResult::success(Tribool::Unknown);
964 }
965
966 case FunctionNode::FuncType::Tan: {
967 // tan(Real) -> Real
968 // tan can be any real value; no sign determination possible.
969 Tribool arg_real = detail::propagate_result(query_domain_of_checked(arg_expr, Domain::Real));
970 if (arg_real != Tribool::True) return InferenceTriboolResult::success(Tribool::Unknown);
971
972 return InferenceTriboolResult::success(Tribool::Unknown);
973 }
974
975 case FunctionNode::FuncType::ArcTan:
976 // atan 在实数域严格递增且 atan(0)=0,六类符号性质均与参数一致.
977 return query_sign_of_checked(arg_expr, target);
978
979 case FunctionNode::FuncType::ArcSin:
980 case FunctionNode::FuncType::ArcCos:
981 return InferenceTriboolResult::success(Tribool::Unknown);
982
983 default:
984 // Unrecognized function -> Unknown for all properties
985 return InferenceTriboolResult::success(Tribool::Unknown);
986 }
987 } catch (const detail::ResultPropagation& ex) {
988 return InferenceTriboolResult::failure(ex.error());
989 } catch (const std::bad_alloc&) {
990 return InferenceTriboolResult::failure(
991 CasErrc::ResourceLimit,
992 "function sign inference allocation failed",
993 "infer_function_property_checked");
994 } catch (const std::exception& ex) {
995 return InferenceTriboolResult::failure(
996 CasErrc::InternalInvariant,
997 ex.what(),
998 "infer_function_property_checked");
999 }
