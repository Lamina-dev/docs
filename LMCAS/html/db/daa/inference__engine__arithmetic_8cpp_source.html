43 if (!mul || mul->operands().size() != 2) {

86InferenceTriboolResult InferenceEngine::Impl::sum_sign(
87 const InferenceEngine& engine,
88 const AddNode& node,
89 Sign target, ComputationContext& context) const {
90 bool all_have_property = true;
91
92 for (const auto& operand : node.operands()) {
93 auto step = context.consume_steps(1, "inference.arithmetic");
94 if (!step) throw step.error();
95 auto op_expr = LMCAS::detail::expression_from_node(operand);
96 Tribool op_result = Tribool::Unknown;
97 if (target != Sign::Positive && target != Sign::Negative &&
98 target != Sign::NonNegative && target != Sign::NonPositive) {
99 return InferenceTriboolResult::success(Tribool::Unknown);
100 }
101 auto queried = engine.query_sign_of_checked(op_expr, target, context);
102 if (!queried) {
103 return queried;
104 }
105 op_result = queried.value();
106
107 if (op_result == Tribool::Unknown) {
108 return InferenceTriboolResult::success(Tribool::Unknown);
109 }
110 if (op_result == Tribool::False) {
111 all_have_property = false;
112 }
113 }
114
115 if (all_have_property) {
116 return InferenceTriboolResult::success(Tribool::True);
117 }
118
119 return InferenceTriboolResult::success(Tribool::Unknown);
120}

122InferenceTriboolResult InferenceEngine::Impl::subtraction_sign(
123 const InferenceEngine& engine,
124 const SymbolicExpr& minuend_expr,
125 const SymbolicExpr& subtrahend_expr,
126 Sign target, ComputationContext& context) const {
127 auto min_pos_result =
128 engine.query_sign_of_checked(minuend_expr, Sign::Positive, context);
129 if (!min_pos_result) {
130 return min_pos_result;
131 }
132 const Tribool min_pos = min_pos_result.value();
133 auto min_neg_result =
134 engine.query_sign_of_checked(minuend_expr, Sign::Negative, context);
135 if (!min_neg_result) {
136 return min_neg_result;
137 }
138 const Tribool min_neg = min_neg_result.value();
139 auto sub_pos_result =
140 engine.query_sign_of_checked(subtrahend_expr, Sign::Positive, context);
141 if (!sub_pos_result) {
142 return sub_pos_result;
143 }
144 const Tribool sub_pos = sub_pos_result.value();
145 auto sub_neg_result =
146 engine.query_sign_of_checked(subtrahend_expr, Sign::Negative, context);
147 if (!sub_neg_result) {
148 return sub_neg_result;
149 }
150 const Tribool sub_neg = sub_neg_result.value();
151
152 if (min_pos == Tribool::True && sub_neg == Tribool::True) {
153 return known_sign(1, target);
154 }
155
156 if (min_neg == Tribool::True && sub_pos == Tribool::True) {
157 return known_sign(-1, target);
158 }
159
160 return weak_subtraction_sign(engine, minuend_expr, subtrahend_expr, target, context);
161}
