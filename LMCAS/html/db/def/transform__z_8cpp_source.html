8static bool zt_is_exp_seq(const std::shared_ptr<SymbolicExpr>& f, const std::string& n,
9 std::shared_ptr<SymbolicExpr>& base_out) {
10 if (!f || !detail::node(f)) {
11 return false;
12 }
13 auto power = std::dynamic_pointer_cast<const PowerNode>(detail::node(f));
14 if (!power) {
15 return false;
16 }
17 auto exponent = std::dynamic_pointer_cast<const VariableNode>(power->exponent());
18 if (!exponent || exponent->is_constant() || exponent->name() != n) {
19 return false;
20 }
21 auto base = detail::make_expression_ptr(power->base());
22 if (te_depends_on(base, n)) {
23 return false;
24 }
25 base_out = base;
26 return true;
27}

29static bool zt_is_weighted_exp(const std::shared_ptr<SymbolicExpr>& f,
30 const std::string& n,
31 std::shared_ptr<SymbolicExpr>& base) {
32 auto product = std::dynamic_pointer_cast<const MultiplyNode>(detail::node(f));
33 if (!product || product->operands().size() != 2) {
34 return false;
35 }
36 for (std::size_t i = 0; i < 2; ++i) {
37 auto variable = std::dynamic_pointer_cast<const VariableNode>(product->operands()[i]);
38 auto other = detail::make_expression_ptr(product->operands()[1 - i]);
39 if (variable && !variable->is_constant() && variable->name() == n &&
40 zt_is_exp_seq(other, n, base)) {
41 return true;
42 }
43 }
44 return false;
45}

49static ZPolynomialPattern zt_polynomial_pattern(
50 const std::shared_ptr<SymbolicExpr>& f, const std::string& n) {
51 auto variable = std::dynamic_pointer_cast<const VariableNode>(detail::node(f));
52 if (variable && !variable->is_constant() && variable->name() == n) {
53 return ZPolynomialPattern::Linear;
54 }
55 auto power = std::dynamic_pointer_cast<const PowerNode>(detail::node(f));
56 if (!power) {
57 return ZPolynomialPattern::None;
58 }
59 variable = std::dynamic_pointer_cast<const VariableNode>(power->base());
60 if (!variable || variable->is_constant() || variable->name() != n) {
61 return ZPolynomialPattern::None;
62 }
63 auto exponent = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
64 BigInt integer;
65 if (!exponent || !try_get_integer_value(exponent, integer)) {
66 return ZPolynomialPattern::None;
67 }
68 if (integer == BigInt(2)) return ZPolynomialPattern::Quadratic;
69 if (integer == BigInt(3)) return ZPolynomialPattern::Cubic;
70 return ZPolynomialPattern::None;
71}

108static std::vector<std::shared_ptr<SymbolicExpr>> zt_roc(
109 const std::shared_ptr<SymbolicExpr>& f, const std::string& n, const std::string& z) {
110 if (!f || !detail::node(f)) return {};
111 auto addition = std::dynamic_pointer_cast<const AddNode>(detail::node(f));
112 if (addition) {
113 std::vector<std::shared_ptr<SymbolicExpr>> combined;
114 for (const auto& operand : addition->operands()) {
115 auto conditions = zt_roc(detail::make_expression_ptr(operand), n, z);
116 combined.insert(combined.end(), conditions.begin(), conditions.end());
117 }
118 return combined;
119 }
120 if (!te_depends_on(f, n)) return zt_unit_roc(z);
121 auto [coefficient, body] = te_split_coeff(f, n);
122 if (!coefficient->is_one()) return zt_roc(body, n, z);
123 return zt_lookup_roc(f, n, z);
124}

126static std::shared_ptr<SymbolicExpr> zt_polynomial(
127 const std::shared_ptr<SymbolicExpr>& f, const std::string& n,
128 const std::shared_ptr<SymbolicExpr>& z, const std::string& z_var) {
129 auto pattern = zt_polynomial_pattern(f, n);
130 if (pattern == ZPolynomialPattern::None) return nullptr;
131 if (pattern == ZPolynomialPattern::Cubic) {
132 auto square = SymbolicExpr::power(SymbolicExpr::variable(n), SymbolicExpr::number(2));
133 auto quadratic = zt_polynomial(square, n, z, z_var);
134 return SymbolicExpr::multiply(
135 SymbolicExpr::multiply(SymbolicExpr::number(-1), z),
136 quadratic->differentiate(z_var))->simplify();
137 }
138 auto denominator = SymbolicExpr::power(SymbolicExpr::add(z, SymbolicExpr::number(-1)),
139 SymbolicExpr::number(pattern == ZPolynomialPattern::Linear ? 2 : 3));
140 auto numerator = pattern == ZPolynomialPattern::Linear ? z : SymbolicExpr::multiply(z,
141 SymbolicExpr::add(z, SymbolicExpr::number(1)));
142 return SymbolicExpr::divide(numerator, denominator)->simplify();
143}

145static std::shared_ptr<SymbolicExpr> zt_exponential(
146 const std::shared_ptr<SymbolicExpr>& f, const std::string& n,
147 const std::shared_ptr<SymbolicExpr>& z) {
148 std::shared_ptr<SymbolicExpr> base;
149 const bool plain = zt_is_exp_seq(f, n, base);
150 if (!plain && !zt_is_weighted_exp(f, n, base)) return nullptr;
151 auto difference = SymbolicExpr::add(z, SymbolicExpr::multiply(SymbolicExpr::number(-1), base));
152 if (plain) return SymbolicExpr::divide(z, difference)->simplify();
153 auto denominator = SymbolicExpr::power(difference, SymbolicExpr::number(2));
154 return SymbolicExpr::divide(SymbolicExpr::multiply(base, z), denominator)->simplify();
155}

157static std::shared_ptr<SymbolicExpr> zt_trigonometric(
158 const std::shared_ptr<SymbolicExpr>& f, const std::string& n,
159 const std::shared_ptr<SymbolicExpr>& z) {
160 bool is_sin = false;
161 std::shared_ptr<SymbolicExpr> frequency;
162 if (!te_is_trig(f, n, is_sin, frequency)) return nullptr;
163 auto cosine = SymbolicExpr::cos(frequency);
164 auto cross = SymbolicExpr::multiply(SymbolicExpr::multiply(SymbolicExpr::number(-2), z), cosine);
165 auto denominator = SymbolicExpr::add(SymbolicExpr::add(
166 SymbolicExpr::power(z, SymbolicExpr::number(2)), cross), SymbolicExpr::number(1));
167 auto factor = is_sin ? SymbolicExpr::sin(frequency) : SymbolicExpr::add(z,
168 SymbolicExpr::multiply(SymbolicExpr::number(-1), cosine));
169 return SymbolicExpr::divide(SymbolicExpr::multiply(z, factor), denominator)->simplify();
170}

195static std::shared_ptr<SymbolicExpr> z_transform_core(
196 const std::shared_ptr<SymbolicExpr>& f, const std::string& n,
197 const std::string& z, ComputationContext& context) {
198 auto step = context.consume_steps(1, "z_transform.recursive");
199 if (!step) {
200 return nullptr;
201 }
202 if (!f || !detail::node(f)) {
203 return nullptr;
204 }
205 auto zv = SymbolicExpr::variable(z);
206 auto addition = std::dynamic_pointer_cast<const AddNode>(detail::node(f));
207 if (addition) {
208 return zt_sum(f, *addition, n, z, context);
209 }
210 auto [coefficient, body] = te_split_coeff(f, n);
211 if (!coefficient->is_one()) {
212 auto transformed = z_transform_core(body, n, z, context);
213 if (transformed) {
214 return SymbolicExpr::multiply(coefficient, transformed)->simplify();
215 }
216 }
217 if (!te_depends_on(f, n)) {
218 auto denominator = SymbolicExpr::add(zv, SymbolicExpr::number(-1));
219 return SymbolicExpr::multiply(f, SymbolicExpr::divide(zv, denominator))->simplify();
220 }
221 auto polynomial = zt_polynomial(f, n, zv, z);
222 if (polynomial) {
223 return polynomial;
224 }
225 auto exponential = zt_exponential(f, n, zv);
226 if (exponential) {
227 return exponential;
228 }
229 auto trig = zt_trigonometric(f, n, zv);
230 return trig ? trig : zt_unevaluated(f, n, z);
231}

233TransformEngineResult z_transform_checked(
234 const std::shared_ptr<SymbolicExpr>& f_n,
235 const std::string& n,
236 const std::string& z,
237 ComputationContext& context) {
238 const std::string operation = "z_transform";
239 auto valid = te_validate_expr_vars(f_n, n, z, context, operation);
240 if (!valid) return TransformEngineResult::failure(valid.error());
241 auto step = context.consume_steps(8, operation);
242 if (!step) return TransformEngineResult::failure(step.error());
243 try {
244 auto expression = z_transform_core(f_n, n, z, context);
245 auto final_access = context.consume_steps(0, operation);
246 if (!final_access) {
247 return TransformEngineResult::failure(final_access.error());
248 }
249 auto result = te_wrap_transform_result(std::move(expression), operation);
250 if (result) {
251 result.value().value.roc = zt_roc(f_n, n, z);
252 }
253 return result;
254 } catch (const std::bad_alloc&) {
255 return TransformEngineResult::failure(CasErrc::ResourceLimit,
256 "transform allocation failed",
257 operation);
258 } catch (const std::exception& e) {
259 return TransformEngineResult::failure(CasErrc::InternalInvariant,
260 e.what(), operation);
261 }
262}
