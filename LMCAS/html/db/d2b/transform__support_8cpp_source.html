14 const std::string& input_var,

94te_split_coeff(const std::shared_ptr<SymbolicExpr>& e, const std::string& v) {
95 if (!e || !LMCAS::detail::node(e) || !te_depends_on(e, v)) return {e ? e : SymbolicExpr::number(1), SymbolicExpr::number(1)};
96 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(LMCAS::detail::node(e));
97 if (!mul) return {SymbolicExpr::number(1), e};
98 std::vector<std::shared_ptr<const SymbolicNode>> cp, vp;
99 for (auto& op : mul->operands()) {
100 if (expression_depends_on_variable(op, v)) vp.push_back(op); else cp.push_back(op);
101 }
102 if (cp.empty()) return {SymbolicExpr::number(1), e};
103 if (vp.empty()) return {e, SymbolicExpr::number(1)};
104 auto c = (cp.size()==1) ? LMCAS::detail::make_expression_ptr(cp[0])
105 : LMCAS::detail::make_expression_ptr(LMCAS::detail::make_node<MultiplyNode>(cp));
106 auto b = (vp.size()==1) ? LMCAS::detail::make_expression_ptr(vp[0])
107 : LMCAS::detail::make_expression_ptr(LMCAS::detail::make_node<MultiplyNode>(vp));
108 return {c, b};
109}

111std::shared_ptr<SymbolicExpr> te_linear_coefficient(
112 const std::shared_ptr<const SymbolicNode>& argument, const std::string& variable) {
113 auto symbol = std::dynamic_pointer_cast<const VariableNode>(argument);
114 if (symbol && !symbol->is_constant() && symbol->name() == variable) {
115 return detail::make_expression_ptr(SymbolicFactory::create_number(BigInt(1)));
116 }
117 auto product = std::dynamic_pointer_cast<const MultiplyNode>(argument);
118 if (!product || product->operands().size() != 2) {
119 return nullptr;
120 }
121 for (std::size_t i = 0; i < 2; ++i) {
122 symbol = std::dynamic_pointer_cast<const VariableNode>(product->operands()[i]);
123 const auto& other = product->operands()[1 - i];
124 if (!symbol || symbol->is_constant() || symbol->name() != variable) continue;
125 if (expression_depends_on_variable(other, variable)) continue;
126 return detail::make_expression_ptr(other);
127 }
128 return nullptr;
129}

145std::optional<double> te_atomic_gaussian_rate(
146 const std::shared_ptr<const SymbolicNode>& argument,
147 const std::string& variable) {
148 auto parts = te_split_coeff(
149 detail::make_expression_ptr(argument), variable);
150 const auto coefficient = te_number_value(detail::node(parts.first));
151 if (!coefficient || !std::isfinite(*coefficient) ||
152 !te_is_variable_square(detail::node(parts.second), variable) ||
153 !(*coefficient < 0.0)) {
154 return std::nullopt;
155 }
156 const double rate = -*coefficient;
157 return std::isfinite(rate) && rate > 0.0
158 ? std::optional<double>(rate)
159 : std::nullopt;
160}
