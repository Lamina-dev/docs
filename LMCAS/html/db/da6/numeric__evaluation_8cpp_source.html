37 auto entered = context_.enter_recursion(kOperation);
38 if (!entered) {
39 return Result<ApproxReal>::failure(entered.error());
40 }
41 RecursionScope scope(context_);
42 if (!node) {

44 }
45 if (const auto* value = dynamic_cast<const NumberNode*>(node.get())) {
46 return number(*value);

74Result<ApproxReal> NumericEvaluator::function(const FunctionNode& node) {
75 if (node.type() == FunctionNode::FuncType::Infinity && node.arguments().empty()) {
76 return Result<ApproxReal>::success(ApproxReal{
77 std::numeric_limits<double>::infinity(), 0.0, NumericStatus::PositiveInfinity});
78 }
79 if (node.type() == FunctionNode::FuncType::Atan2) {
80 return atan2(node);
81 }
82 if (node.type() == FunctionNode::FuncType::Max || node.type() == FunctionNode::FuncType::Min) {
83 return extremum(node);
84 }
85 if (node.type() == FunctionNode::FuncType::Log &&
86 node.arguments().size() == 2) {
87 auto value = evaluate(node.arguments()[0]);
88 if (!value) {
89 return value;
90 }
91 auto base = evaluate(node.arguments()[1]);
92 if (!base) {
93 return base;
94 }
95 const double numeric_value = value.value().value;
96 const double numeric_base = base.value().value;
97 if (!(numeric_value > 0.0)) {
98 return numeric_failure(
99 CasErrc::DomainError,
100 "logarithm requires a positive real argument");
101 }
102 if (!(numeric_base > 0.0) || numeric_base == 1.0) {
103 return numeric_failure(
104 CasErrc::DomainError,
105 "logarithm base must be positive and not equal to one");
106 }
107 return numeric_approximation(
108 std::log(numeric_value) / std::log(numeric_base));
109 }
110 if (node.arguments().size() != 1) {
111 return numeric_failure(CasErrc::UnsupportedExpression,
112 "function is not supported by real numeric evaluation");
113 }
114 auto argument = evaluate(node.arguments()[0]);
115 if (!argument) {
116 return argument;
117 }
118 return evaluate_unary_numeric(node.type(), argument.value().value);
119}
