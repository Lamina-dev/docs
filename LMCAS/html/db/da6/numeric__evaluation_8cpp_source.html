33Result<ApproxReal> make_approx(double value) {
34 if (std::isnan(value)) {
35 return failure(CasErrc::NumericFailure, "numeric evaluation produced NaN");
36 }
37
38 ApproxReal result;
39 result.value = value;
40 if (std::isinf(value)) {
41 result.status = value > 0 ? NumericStatus::PositiveInfinity
42 : NumericStatus::NegativeInfinity;
43 result.absolute_error = 0.0;
44 } else {
45 result.status = NumericStatus::Finite;
46 result.absolute_error = std::numeric_limits<double>::epsilon() *
47 std::max(1.0, std::abs(value)) * 4.0;
48 }
49 return Result<ApproxReal>::success(result);
50}

52Result<ApproxReal> evaluate_node(const std::shared_ptr<const SymbolicNode>& node,
53 const NumericBindings& bindings,
54 ComputationContext& context) {
55 auto entered = context.enter_recursion(kOperation);
56 if (!entered) return Result<ApproxReal>::failure(entered.error());
57 RecursionScope scope(context);
58
59 if (!node) {
60 return failure(CasErrc::InvalidArgument, "expression contains a null node");
61 }
62
63 if (auto number = std::dynamic_pointer_cast<const NumberNode>(node)) {
64 double value = 0.0;
65 try {
66 if (std::holds_alternative<lmmc_real_t>(number->value())) {
67 value = static_cast<double>(std::get<lmmc_real_t>(number->value()));
68 } else if (std::holds_alternative<BigInt>(number->value())) {
69 value = std::get<BigInt>(number->value()).to_double();
70 } else {
71 value = std::get<Rational>(number->value()).to_double();
72 }
73 } catch (const std::exception& error) {
74 return failure(CasErrc::NumericFailure, error.what());
75 }
76 if (!std::isfinite(value)) {
77 return failure(CasErrc::NumericFailure,
78 "exact number cannot be represented as a finite double");
79 }
80 return make_approx(value);
81 }
82
83 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(node)) {
84 if (variable->name() == "pi" || variable->name() == "π") {
85 return make_approx(static_cast<double>(LMMC_CONST_PI));
86 }
87 if (variable->name() == "e") {
88 return make_approx(std::exp(1.0));
89 }
90 if (variable->name() == "phi") {
91 return make_approx((1.0 + std::sqrt(5.0)) / 2.0);
92 }
93 auto it = bindings.find(variable->name());
94 if (it == bindings.end()) {
95 return failure(CasErrc::UnboundSymbol,
96 "no numeric binding for symbol '" + variable->name() + "'");
97 }
98 return make_approx(it->second);
99 }
100
101 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
102 double value = 0.0;
103 for (const auto& operand : add->operands()) {
104 auto evaluated = evaluate_node(operand, bindings, context);
105 if (!evaluated) return evaluated;
106 value += evaluated.value().value;
107 }
108 return make_approx(value);
109 }
110
111 if (auto multiply = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
112 double value = 1.0;
113 for (const auto& operand : multiply->operands()) {
114 auto evaluated = evaluate_node(operand, bindings, context);
115 if (!evaluated) return evaluated;
116 value *= evaluated.value().value;
117 }
118 return make_approx(value);
119 }
120
121 if (auto power = std::dynamic_pointer_cast<const PowerNode>(node)) {
122 auto base = evaluate_node(power->base(), bindings, context);
123 if (!base) return base;
124 auto exponent = evaluate_node(power->exponent(), bindings, context);
125 if (!exponent) return exponent;
126 if (base.value().value == 0.0 && exponent.value().value <= 0.0) {
127 return failure(CasErrc::DomainError,
128 "zero cannot be raised to a non-positive power");
129 }
130 return make_approx(std::pow(base.value().value, exponent.value().value));
131 }
132
133 if (std::dynamic_pointer_cast<const RootOfNode>(node)) {
134 auto root_expression = lamina::detail::make_expression_ptr(node);
135 auto root = rootof_evaluate_checked(root_expression, context);
136 if (!root) return Result<ApproxReal>::failure(root.error());
137 return make_approx(root.value());
138 }
139
140 if (auto function = std::dynamic_pointer_cast<const FunctionNode>(node)) {
141 if (function->type() == FunctionNode::FuncType::Infinity &&
142 function->arguments().empty()) {
143 return make_approx(std::numeric_limits<double>::infinity());
144 }
145
146 if (function->type() == FunctionNode::FuncType::Atan2) {
147 if (function->arguments().size() != 2) {
148 return failure(CasErrc::InvalidArgument, "atan2 requires two arguments");
149 }
150 auto y = evaluate_node(function->arguments()[0], bindings, context);
151 if (!y) return y;
152 auto x = evaluate_node(function->arguments()[1], bindings, context);
153 if (!x) return x;
154 return make_approx(std::atan2(y.value().value, x.value().value));
155 }
156
157 if (function->type() == FunctionNode::FuncType::Max ||
158 function->type() == FunctionNode::FuncType::Min) {
159 if (function->arguments().empty()) {
160 return failure(CasErrc::InvalidArgument, "min/max requires an argument");
161 }
162 auto first = evaluate_node(function->arguments().front(), bindings, context);
163 if (!first) return first;
164 double value = first.value().value;
165 for (std::size_t i = 1; i < function->arguments().size(); ++i) {
166 auto next = evaluate_node(function->arguments()[i], bindings, context);
167 if (!next) return next;
168 value = function->type() == FunctionNode::FuncType::Max
169 ? std::max(value, next.value().value)
170 : std::min(value, next.value().value);
171 }
172 return make_approx(value);
173 }
174
175
176 if (function->arguments().size() != 1) {
177 return failure(CasErrc::UnsupportedExpression,
178 "function is not supported by real numeric evaluation");
179 }
180 auto argument = evaluate_node(function->arguments()[0], bindings, context);
181 if (!argument) return argument;
182 const double x = argument.value().value;
183 double result = 0.0;
184
185 switch (function->type()) {
186 case FunctionNode::FuncType::Sin: result = std::sin(x); break;
187 case FunctionNode::FuncType::Cos: result = std::cos(x); break;
188 case FunctionNode::FuncType::Tan: result = std::tan(x); break;
189 case FunctionNode::FuncType::Cot:
190 if (std::sin(x) == 0.0) return failure(CasErrc::DomainError, "cot is undefined");
191 result = 1.0 / std::tan(x);
192 break;
193 case FunctionNode::FuncType::Sec:
194 if (std::cos(x) == 0.0) return failure(CasErrc::DomainError, "sec is undefined");
195 result = 1.0 / std::cos(x);
196 break;
197 case FunctionNode::FuncType::Csc:
198 if (std::sin(x) == 0.0) return failure(CasErrc::DomainError, "csc is undefined");
199 result = 1.0 / std::sin(x);
200 break;
201 case FunctionNode::FuncType::ArcSin:
202 if (x < -1.0 || x > 1.0) return failure(CasErrc::DomainError, "asin real domain is [-1, 1]");
203 result = std::asin(x);
204 break;
205 case FunctionNode::FuncType::ArcCos:
206 if (x < -1.0 || x > 1.0) return failure(CasErrc::DomainError, "acos real domain is [-1, 1]");
207 result = std::acos(x);
208 break;
209 case FunctionNode::FuncType::ArcTan: result = std::atan(x); break;
210 case FunctionNode::FuncType::Sinh: result = std::sinh(x); break;
211 case FunctionNode::FuncType::Cosh: result = std::cosh(x); break;
212 case FunctionNode::FuncType::Tanh: result = std::tanh(x); break;
213 case FunctionNode::FuncType::Ln:
214 case FunctionNode::FuncType::Log:
215 if (x <= 0.0) return failure(CasErrc::DomainError, "logarithm requires a positive real argument");
216 result = std::log(x);
217 break;
218 case FunctionNode::FuncType::Abs: result = std::abs(x); break;
219 case FunctionNode::FuncType::Sqrt:
220 if (x < 0.0) return failure(CasErrc::DomainError, "real square root requires a non-negative argument");
221 result = std::sqrt(x);
222 break;
223 case FunctionNode::FuncType::Exp: result = std::exp(x); break;
224 case FunctionNode::FuncType::Sgn: result = (x > 0.0) - (x < 0.0); break;
225 case FunctionNode::FuncType::Floor: result = std::floor(x); break;
226 case FunctionNode::FuncType::Ceil: result = std::ceil(x); break;
227 case FunctionNode::FuncType::Round: result = std::round(x); break;
228 case FunctionNode::FuncType::Erf: result = std::erf(x); break;
229 case FunctionNode::FuncType::LambertW: {
230 lmmc_real_t value = 0.0;
231 if (lmmc_lambertw(static_cast<lmmc_real_t>(x), &value) != LMMC_STATUS_OK) {
232 return failure(CasErrc::DomainError, "LambertW evaluation failed on the real branch");
233 }
234 result = static_cast<double>(value);
235 break;
236 }
237 case FunctionNode::FuncType::RealPart:
238 case FunctionNode::FuncType::Conjugate:
239 result = x;
240 break;
241 case FunctionNode::FuncType::ImagPart:
242 case FunctionNode::FuncType::ComplexArg:
243 result = 0.0;
244 break;
245 case FunctionNode::FuncType::ComplexAbs:
246 result = std::abs(x);
247 break;
248 default:
249 return failure(CasErrc::UnsupportedExpression,
250 "function is not supported by real numeric evaluation");
251 }
252 return make_approx(result);
253 }
254
255 return failure(CasErrc::UnsupportedExpression,
256 "expression node is not real-numerically evaluable");
257}
