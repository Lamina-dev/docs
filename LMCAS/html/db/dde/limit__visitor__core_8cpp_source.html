83bool LimitVisitor::is_inf(const std::shared_ptr<const SymbolicNode>& node) const {
84 if (!node) return false;
85 if (auto f = std::dynamic_pointer_cast<const FunctionNode>(node)) return f->type() == FunctionNode::FuncType::Infinity;
86 if (auto m = std::dynamic_pointer_cast<const MultiplyNode>(node)) for (auto& op : m->operands()) if (is_inf(op)) return true;
87 if (auto p = std::dynamic_pointer_cast<const PowerNode>(node)) {
88 if (is_inf(p->base())) return true;
89 /// 0^(negative) is infinity
90 if (p->base() && p->base()->is_zero()) {
91 if (auto e_num = std::dynamic_pointer_cast<const NumberNode>(p->exponent())) {
92 double ev = 0;
93 if (std::holds_alternative<double>(e_num->value())) ev = std::get<double>(e_num->value());
94 else if (std::holds_alternative<BigInt>(e_num->value())) ev = std::get<BigInt>(e_num->value()).to_double();
95 else if (std::holds_alternative<Rational>(e_num->value())) ev = std::get<Rational>(e_num->value()).to_double();
96 if (ev < 0) return true;
97 }
98 }
99 }
100 if (auto a = std::dynamic_pointer_cast<const AddNode>(node)) for (auto& op : a->operands()) if (is_inf(op)) return true;
101 return false;

145bool LimitVisitor::is_bounded(const std::shared_ptr<const SymbolicNode>& node) const {
146 if (!node) return false;
147 /// sin(expr) and cos(expr) are bounded in [-1, 1]
148 /// arctan(expr) is bounded in (-π/2, π/2)
149 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(node)) {
150 return func->type() == FunctionNode::FuncType::Sin
151 || func->type() == FunctionNode::FuncType::Cos
152 || func->type() == FunctionNode::FuncType::ArcTan;
153 }
154 /// Product of bounded functions is bounded
155 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
156 bool all_bounded_or_const = true;
157 bool has_bounded = false;
158 for (const auto& op : mul->operands()) {
159 if (is_bounded(op)) { has_bounded = true; }
160 else if (std::dynamic_pointer_cast<const NumberNode>(op)) { /* constants are fine */ }
161 else { all_bounded_or_const = false; break; }
162 }
163 return all_bounded_or_const && has_bounded;
164 }
165 /// bounded^(even positive integer) is bounded (e.g., sin(x)^2)
166 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
167 if (is_bounded(pow->base())) {
168 if (auto num = std::dynamic_pointer_cast<const NumberNode>(pow->exponent())) {
169 double e = 0;
170 if (std::holds_alternative<double>(num->value())) e = std::get<double>(num->value());
171 else if (std::holds_alternative<BigInt>(num->value())) e = std::get<BigInt>(num->value()).to_double();
172 else if (std::holds_alternative<Rational>(num->value())) e = std::get<Rational>(num->value()).to_double();
173 if (e > 0) return true;
174 }
175 }
176 }
177 return false;
178}

182 *
183 * 检测形如 constant + bounded 或 constant * bounded 的表达式，

198 if (is_bounded(op) || is_bounded_expression(op)) {
199 has_bounded = true;
200 } else if (std::dynamic_pointer_cast<const NumberNode>(op)) {
201 /// constants are bounded

253std::shared_ptr<const SymbolicNode> LimitVisitor::try_squeeze(const std::shared_ptr<const SymbolicNode>& expr) {
254 /// Case 1: MultiplyNode — product of bounded × zero-tending → 0
255 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(expr)) {
256 std::vector<std::shared_ptr<const SymbolicNode>> bounded_factors, other_factors;
257 for (const auto& op : mul->operands()) {
258 if (is_bounded(op)) {
259 bounded_factors.push_back(op);
260 } else if (is_bounded_expression(op)) {
261 bounded_factors.push_back(op);
262 } else {
263 other_factors.push_back(op);
264 }
265 }
266 if (bounded_factors.empty() || other_factors.empty()) return nullptr;
267
268 /// Check if the non-bounded factors tend to zero
269 auto remaining = other_factors.size() == 1
270 ? other_factors[0]
271 : make_product_or_one(other_factors);
272 if (tends_to_zero(remaining)) {
273 return lamina::detail::make_node<NumberNode>(BigInt(0));
274 }
275 return nullptr;
276 }
277
278 /// Case 2: AddNode — general squeeze theorem.
279 /// If f(x) = g(x) + h(x) where h(x) is a product of bounded × zero-tending,
280 /// then lim f = lim g (since h squeezes to 0).
281 /// This handles the case: f bounded between g and (g + bounded×zero) where
282 /// lim(lower) = lim(upper) = lim g = L.
283 if (auto add = std::dynamic_pointer_cast<const AddNode>(expr)) {
284 std::vector<std::shared_ptr<const SymbolicNode>> squeeze_to_zero_terms;
285 std::vector<std::shared_ptr<const SymbolicNode>> other_terms;
286
287 for (const auto& op : add->operands()) {
288 /// Check if this term is a product of bounded × zero-tending
289 if (auto term_mul = std::dynamic_pointer_cast<const MultiplyNode>(op)) {
290 auto term_squeeze = try_squeeze(op);
291 if (term_squeeze && term_squeeze->is_zero()) {
292 squeeze_to_zero_terms.push_back(op);
293 continue;
294 }
295 }
296 other_terms.push_back(op);
297 }
298
299 /// If we found squeeze-to-zero terms and have remaining terms with computable limits
300 if (!squeeze_to_zero_terms.empty() && !other_terms.empty()) {
301 auto remaining_expr = other_terms.size() == 1
302 ? other_terms[0]
303 : std::static_pointer_cast<const SymbolicNode>(lamina::detail::make_node<AddNode>(other_terms));
304 /// Compute the limit of the remaining terms
305 auto remaining_limit = eval_limit(remaining_expr);
306 if (remaining_limit && !is_inf(remaining_limit)) {
307 return remaining_limit;
308 }
309 }
310 }
311
312 return nullptr;
313}

348std::shared_ptr<const SymbolicNode> LimitVisitor::resolve_zero_times_inf(const std::vector<std::shared_ptr<const SymbolicNode>>& factors, const std::vector<std::shared_ptr<const SymbolicNode>>& factor_vals) {
349 std::vector<std::shared_ptr<const SymbolicNode>> zero_f, inf_f, other_f;
350 for (size_t i = 0; i < factors.size(); ++i) {
351 if (factor_vals[i] && factor_vals[i]->is_zero()) zero_f.push_back(factors[i]);
352 else if (is_inf(factor_vals[i])) inf_f.push_back(factors[i]);
353 else other_f.push_back(factors[i]);
354 }
355 if (zero_f.empty() || inf_f.empty()) return nullptr;
356 auto f = make_product_or_one(zero_f);
357 auto g = make_product_or_one(inf_f);
358
359 /// Choose between 0/0 form (f/(1/g)) and ∞/∞ form (g/(1/f)).
360 /// Heuristic: if the zero factor has exponential growth class (e.g., e^(-x)),
361 /// prefer ∞/∞ form because L'Hôpital on 0/0 won't converge (polynomial grows).
362 /// Otherwise, try 0/0 first as it's simpler for most cases.
363 bool prefer_inf_over_inf = false;
364 if (is_limit_at_infinity() || is_limit_at_neg_infinity()) {
365 GrowthClass f_growth = classify_growth(f);
366 GrowthClass g_growth = classify_growth(g);
367 /// If zero factor is exponential (e.g., e^(-x)→0) and inf factor is polynomial,
368 /// the ∞/∞ form (polynomial/exponential) converges in one step.
369 if (f_growth == GrowthClass::Exponential && g_growth == GrowthClass::Polynomial) {
370 prefer_inf_over_inf = true;
371 }
372 /// If zero factor is polynomial (e.g., 1/x→0) and inf factor is logarithmic,
373 /// the 0/0 form works well.
374 }
375
376 std::shared_ptr<const SymbolicNode> res = nullptr;
377
378 if (prefer_inf_over_inf) {
379 /// Try ∞/∞ form first: g / (1/f)
380 auto f_inv = lamina::detail::make_node<PowerNode>(f, lamina::detail::make_node<NumberNode>(BigInt(-1)));
381 res = apply_lhopital(g, f_inv);
382 if (!res) {
383 auto g_inv = lamina::detail::make_node<PowerNode>(g, lamina::detail::make_node<NumberNode>(BigInt(-1)));
384 res = apply_lhopital(f, g_inv);
385 }
386 } else {
387 /// Try 0/0 form first: f / (1/g)
388 auto g_inv = lamina::detail::make_node<PowerNode>(g, lamina::detail::make_node<NumberNode>(BigInt(-1)));
389 res = apply_lhopital(f, g_inv);
390 if (!res) {
391 auto f_inv = lamina::detail::make_node<PowerNode>(f, lamina::detail::make_node<NumberNode>(BigInt(-1)));
392 res = apply_lhopital(g, f_inv);
393 }
394 }
395
396 if (res && !other_f.empty()) {
397 std::vector<std::shared_ptr<const SymbolicNode>> ff; ff.push_back(res);
398 for (auto& of : other_f) { auto ov = eval_limit(of); if (ov) ff.push_back(ov); }
399 if (ff.size() == 1) return ff[0];
400 auto prod = lamina::detail::make_node<MultiplyNode>(ff);
401 NormalizationVisitor norm; prod->accept(norm); return norm.get_result();
402 }
403 return res;
404}

406std::shared_ptr<const SymbolicNode> LimitVisitor::resolve_inf_minus_inf(const AddNode& node, const std::vector<std::shared_ptr<const SymbolicNode>>&) {
407 std::vector<std::shared_ptr<const SymbolicNode>> nums, dens;
408 bool has_nontrivial_den = false;
409 for (auto& op : node.operands()) { auto nd = extract_num_den(op); nums.push_back(nd.first); dens.push_back(nd.second); if (!nd.second->is_one()) has_nontrivial_den = true; }
410
411 /// If no operand has a denominator, combining into a fraction won't help.
412 /// Fall back to Taylor expansion at the limit point.
413 if (!has_nontrivial_den) {
414 /// Try Taylor fallback: treat the whole expression as numerator / 1
415 auto whole_expr = node.operands().size() == 1 ? node.operands()[0] : std::static_pointer_cast<const SymbolicNode>(lamina::detail::make_node<AddNode>(node.operands()));
416 auto one_node = lamina::detail::make_node<NumberNode>(BigInt(1));
417 auto tf_result = taylor_fallback(whole_expr, one_node);
418 return tf_result;
419 }
420
421 auto common_den = make_product_or_one(dens);
422 std::vector<std::shared_ptr<const SymbolicNode>> new_num_terms;
423 for (size_t i = 0; i < nums.size(); ++i) {
424 std::vector<std::shared_ptr<const SymbolicNode>> tf; tf.push_back(nums[i]);
425 for (size_t j = 0; j < dens.size(); ++j) { if (j != i) tf.push_back(dens[j]); }
426 new_num_terms.push_back(make_product_or_one(tf));
427 }
428 auto combined_num = new_num_terms.size() == 1 ? new_num_terms[0] : std::static_pointer_cast<const SymbolicNode>(lamina::detail::make_node<AddNode>(new_num_terms));
429 NormalizationVisitor norm;
430 combined_num->accept(norm); combined_num = norm.get_result();
431 common_den->accept(norm); common_den = norm.get_result();
432 auto nv = eval_limit(combined_num); auto dv = eval_limit(common_den);
433 bool nz = nv && nv->is_zero(), dz = dv && dv->is_zero();
434 if ((nz && dz) || (is_inf(nv) && is_inf(dv))) return apply_lhopital(combined_num, common_den);
435 if (!dz && !is_inf(dv) && dv) {
436 auto ratio = lamina::detail::make_node<MultiplyNode>(std::vector<std::shared_ptr<const SymbolicNode>>{combined_num, lamina::detail::make_node<PowerNode>(common_den, lamina::detail::make_node<NumberNode>(BigInt(-1)))});
437 return eval_limit(ratio);
438 }
439 return nullptr;
440}

459std::shared_ptr<const SymbolicNode> LimitVisitor::apply_lhopital(const std::shared_ptr<const SymbolicNode>& num, const std::shared_ptr<const SymbolicNode>& den) {
460 EvaluationScope scope;
461 if (!scope) return nullptr;
462 if (lhopital_depth_ >= max_lhopital_depth) return taylor_fallback(num, den);
463 DifferentiationVisitor diff_vis(var);
464 num->accept(diff_vis); auto dN = diff_vis.get_result();
465 den->accept(diff_vis); auto dD = diff_vis.get_result();
466 if (!dN || !dD) return taylor_fallback(num, den);
467 NormalizationVisitor norm;
468 dN->accept(norm); dN = norm.get_result();
469 dD->accept(norm); dD = norm.get_result();
470 if (dD->is_zero()) return taylor_fallback(num, den);
471
472 /// Try to simplify the ratio dN/dD algebraically before evaluating limits.
473 /// This handles cases like x*ln(1+1/x) where L'Hôpital produces derivatives
474 /// with common factors (e.g., x^-2) that cancel, avoiding infinite 0/0 loops.
475 {
476 std::vector<std::shared_ptr<const SymbolicNode>> ratio_ops = {
477 dN, lamina::detail::make_node<PowerNode>(dD, lamina::detail::make_node<NumberNode>(BigInt(-1)))};
478 auto ratio_node = lamina::detail::make_node<MultiplyNode>(ratio_ops);
479 /// Wrap in SymbolicExpr for simplification (defined in limit_visitor.cpp)
480 auto simp_result = simplify_and_eval_ratio(ratio_node);
481 if (simp_result) return simp_result;
482 }
483
484 /// Construct the ratio dN/dD. Instead of creating a MultiplyNode that might
485 /// be misinterpreted as 0×∞, evaluate as a proper fraction: compute limits
486 /// of dN and dD separately and check for 0/0 or ∞/∞ forms.
487 LimitVisitor sub_n(var, point, direction, assumption_ctx_);
488 sub_n.lhopital_depth_ = this->lhopital_depth_ + 1;
489 dN->accept(sub_n);
490 auto val_n = sub_n.get_result();
491 if (val_n) { NormalizationVisitor n2; val_n->accept(n2); val_n = n2.get_result(); }
492
493 LimitVisitor sub_d(var, point, direction, assumption_ctx_);
494 sub_d.lhopital_depth_ = this->lhopital_depth_ + 1;
495 dD->accept(sub_d);
496 auto val_d = sub_d.get_result();
497 if (val_d) { NormalizationVisitor n2; val_d->accept(n2); val_d = n2.get_result(); }
498
499 if (!val_n || !val_d) return taylor_fallback(num, den);
500
501 bool n_zero = val_n->is_zero();
502 bool d_zero = val_d->is_zero();
503 bool n_inf = is_inf(val_n);
504 bool d_inf = is_inf(val_d);
505
506 /// If still indeterminate (0/0 or ∞/∞), recurse with increased depth
507 if ((n_zero && d_zero) || (n_inf && d_inf)) {
508 LimitVisitor sub(var, point, direction, assumption_ctx_);
509 sub.lhopital_depth_ = this->lhopital_depth_ + 1;
510 /// Build ratio as MultiplyNode for the sub-visitor
511 std::vector<std::shared_ptr<const SymbolicNode>> ratio_ops = {dN, lamina::detail::make_node<PowerNode>(dD, lamina::detail::make_node<NumberNode>(BigInt(-1)))};
512 auto ratio = lamina::detail::make_node<MultiplyNode>(ratio_ops);
513 ratio->accept(sub);
514 return sub.get_result();
515 }
516
517 /// Denominator is zero but numerator isn't → ±∞
518 if (d_zero && !n_zero) {
519 int sign_n = 1;
520 if (auto nn = std::dynamic_pointer_cast<const NumberNode>(val_n)) {
521 auto s = get_node_sign(nn);
522 if (s) sign_n = *s;
523 }
524 std::vector<std::shared_ptr<const SymbolicNode>> inf_args;
525 auto inf_node = lamina::detail::make_node<FunctionNode>(FunctionNode::FuncType::Infinity, inf_args);
526 if (sign_n < 0) {
527 std::vector<std::shared_ptr<const SymbolicNode>> m = {lamina::detail::make_node<NumberNode>(BigInt(-1)), inf_node};
528 return lamina::detail::make_node<MultiplyNode>(m);
529 }
530 return inf_node;
531 }
532
533 /// Normal case: compute the ratio of limits
534 if (!d_zero && !d_inf) {
535 auto ratio_result = lamina::detail::make_node<MultiplyNode>(std::vector<std::shared_ptr<const SymbolicNode>>{
536 val_n, lamina::detail::make_node<PowerNode>(val_d, lamina::detail::make_node<NumberNode>(BigInt(-1)))});
537 ratio_result->accept(norm);
538 return norm.get_result();
539 }
540
541 /// Numerator is finite, denominator is ∞ → 0
542 if (!n_inf && !n_zero && d_inf) {
543 return lamina::detail::make_node<NumberNode>(BigInt(0));
544 }
545
546 /// Fallback: evaluate the ratio expression directly
547 std::vector<std::shared_ptr<const SymbolicNode>> ratio_ops = {dN, lamina::detail::make_node<PowerNode>(dD, lamina::detail::make_node<NumberNode>(BigInt(-1)))};
548 auto ratio = lamina::detail::make_node<MultiplyNode>(ratio_ops);
549 LimitVisitor sub(var, point, direction, assumption_ctx_);
550 sub.lhopital_depth_ = this->lhopital_depth_ + 1;
551 ratio->accept(sub);
552 return sub.get_result();
553}

555int LimitVisitor::determine_sign_near_point(const std::shared_ptr<const SymbolicNode>& expr, const std::string& dir) {
556 if (dir.empty()) return 0;
557 DifferentiationVisitor diff_vis(var);
558 std::shared_ptr<const SymbolicNode> curr = expr;
559 for (int i = 1; i <= 3; ++i) {
560 curr->accept(diff_vis); auto deriv = diff_vis.get_result(); if (!deriv) break;
561 LimitVisitor sv2(var, point, dir, assumption_ctx_); deriv->accept(sv2); auto val = sv2.get_result(); if (!val) break;
562 NormalizationVisitor norm; val->accept(norm); val = norm.get_result();
563 if (!val->is_zero()) {
564 int s = 1;
565 if (auto num = std::dynamic_pointer_cast<const NumberNode>(val)) {
566 if (std::holds_alternative<double>(num->value())) s = std::get<double>(num->value()) > 0 ? 1 : -1;
567 else if (std::holds_alternative<BigInt>(num->value())) s = std::get<BigInt>(num->value()) > BigInt(0) ? 1 : -1;
568 else if (std::holds_alternative<Rational>(num->value())) s = std::get<Rational>(num->value()) > Rational(0) ? 1 : -1;
569 }
570 if (i % 2 != 0 && dir == "-") s *= -1;
571 return s;
572 }
573 curr = deriv;
574 }
575 return 0;
576}

578void LimitVisitor::visit(const AddNode& node) {
579 /// Handle negative infinity: substitute x = -t and evaluate lim(t→+∞)
580 if (is_limit_at_neg_infinity()) {
581 auto neg_inf_result = handle_neg_infinity_limit(lamina::detail::make_node<AddNode>(node.operands()));
582 if (neg_inf_result) { result = neg_inf_result; return; }
583 }
584 /// General squeeze theorem for AddNode: detect terms that are products of
585 /// bounded × zero-tending (squeeze to 0) and compute limit of remaining terms.
586 auto squeeze_result = try_squeeze(lamina::detail::make_node<AddNode>(std::vector<std::shared_ptr<const SymbolicNode>>(node.operands().begin(), node.operands().end())));
587 if (squeeze_result) { result = squeeze_result; return; }

589 for (auto& op : node.operands()) { op->accept(*this); new_ops.push_back(result); }
590 auto form = classify_add_form(new_ops);
591 if (form == IndeterminateForm::InfMinusInf) { auto resolved = resolve_inf_minus_inf(node, new_ops); if (resolved) { result = resolved; return; } }
592 NormalizationVisitor norm; lamina::detail::make_node<AddNode>(new_ops)->accept(norm); result = norm.get_result();
593}

595void LimitVisitor::visit(const MultiplyNode& node) {
596 /// Handle negative infinity: substitute x = -t and evaluate lim(t→+∞)
597 if (is_limit_at_neg_infinity()) {
598 auto neg_inf_result = handle_neg_infinity_limit(lamina::detail::make_node<MultiplyNode>(std::vector<std::shared_ptr<const SymbolicNode>>(node.operands().begin(), node.operands().end())));
599 if (neg_inf_result) { result = neg_inf_result; return; }
600 }
601 {
602 auto point_num = std::dynamic_pointer_cast<const NumberNode>(point);
603 bool at_zero_from_right = point_num && point_num->is_zero() && direction == "+";
604 bool has_positive_var_power = false;
605 bool has_log_var = false;
606
607 auto is_positive_power_of_limit_var = [&](const std::shared_ptr<const SymbolicNode>& op) {
608 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(op)) {
609 return variable->name() == var;
610 }
611 if (auto power = std::dynamic_pointer_cast<const PowerNode>(op)) {
612 auto base_var = std::dynamic_pointer_cast<const VariableNode>(power->base());
613 auto exponent = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
614 return base_var && base_var->name() == var && exponent && exponent->is_positive();
615 }
616 return false;
617 };
618
619 auto is_log_of_limit_var = [&](const std::shared_ptr<const SymbolicNode>& op) {
620 auto function = std::dynamic_pointer_cast<const FunctionNode>(op);
621 if (!function || function->arguments().size() != 1) return false;
622 if (function->type() != FunctionNode::FuncType::Ln &&
623 function->type() != FunctionNode::FuncType::Log) return false;
624 auto argument_var = std::dynamic_pointer_cast<const VariableNode>(function->arguments()[0]);
625 return argument_var && argument_var->name() == var;
626 };
627
628 if (at_zero_from_right) {
629 for (const auto& op : node.operands()) {
630 has_positive_var_power = has_positive_var_power || is_positive_power_of_limit_var(op);
631 has_log_var = has_log_var || is_log_of_limit_var(op);
632 }
633 if (has_positive_var_power && has_log_var) {
634 result = lamina::detail::make_node<NumberNode>(BigInt(0));
635 return;
636 }
637 }
638 }
639 if (is_limit_at_infinity()) {
640 for (const auto& op : node.operands()) {
641 auto function = std::dynamic_pointer_cast<const FunctionNode>(op);
642 if (!function || function->arguments().size() != 1 ||
643 (function->type() != FunctionNode::FuncType::Ln &&
644 function->type() != FunctionNode::FuncType::Log)) {
645 continue;
646 }
647 auto add = std::dynamic_pointer_cast<const AddNode>(function->arguments()[0]);
648 if (!add) continue;
649
650 std::shared_ptr<const SymbolicNode> small_num;
651 bool has_one = false;
652 for (const auto& add_op : add->operands()) {
653 if (add_op->is_one()) {
654 has_one = true;
655 continue;
656 }
657 small_num = add_op;
658 }
659 if (!has_one || !small_num) continue;
660
661 std::shared_ptr<const SymbolicNode> numerator = lamina::detail::make_node<NumberNode>(BigInt(1));
662 bool denominator_is_var = false;
663 if (auto power = std::dynamic_pointer_cast<const PowerNode>(small_num)) {
664 auto base_var = std::dynamic_pointer_cast<const VariableNode>(power->base());
665 auto exponent = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
666 if (base_var && base_var->name() == var && exponent) {
667 double exponent_value = 0.0;
668 if (std::holds_alternative<double>(exponent->value())) exponent_value = std::get<double>(exponent->value());
669 else if (std::holds_alternative<BigInt>(exponent->value())) exponent_value = std::get<BigInt>(exponent->value()).to_double();
670 else if (std::holds_alternative<Rational>(exponent->value())) exponent_value = std::get<Rational>(exponent->value()).to_double();
671 denominator_is_var = exponent_value == -1.0;
672 }
673 } else if (auto small_mul = std::dynamic_pointer_cast<const MultiplyNode>(small_num)) {
674 std::vector<std::shared_ptr<const SymbolicNode>> numerator_parts;
675 for (const auto& small_factor : small_mul->operands()) {
676 auto power = std::dynamic_pointer_cast<const PowerNode>(small_factor);
677 auto base_var = power ? std::dynamic_pointer_cast<const VariableNode>(power->base()) : nullptr;
678 auto exponent = power ? std::dynamic_pointer_cast<const NumberNode>(power->exponent()) : nullptr;
679 if (base_var && base_var->name() == var && exponent) {
680 double exponent_value = 0.0;
681 if (std::holds_alternative<double>(exponent->value())) exponent_value = std::get<double>(exponent->value());
682 else if (std::holds_alternative<BigInt>(exponent->value())) exponent_value = std::get<BigInt>(exponent->value()).to_double();
683 else if (std::holds_alternative<Rational>(exponent->value())) exponent_value = std::get<Rational>(exponent->value()).to_double();
684 if (exponent_value == -1.0) {
685 denominator_is_var = true;
686 continue;
687 }
688 }
689 numerator_parts.push_back(small_factor);
690 }
691 if (!numerator_parts.empty()) {
692 numerator = numerator_parts.size() == 1
693 ? numerator_parts[0]
694 : make_product_or_one(numerator_parts);
695 }
696 }
697 if (!denominator_is_var) continue;
698
699 std::vector<std::shared_ptr<const SymbolicNode>> remaining_limits = {numerator};
700 bool has_var_factor = false;
701 for (const auto& remaining : node.operands()) {
702 if (remaining.get() == op.get()) continue;
703 if (auto variable = std::dynamic_pointer_cast<const VariableNode>(remaining)) {
704 if (variable->name() == var) {
705 has_var_factor = true;
706 continue;
707 }
708 }
709 auto remaining_limit = eval_limit(remaining);
710 if (!remaining_limit) {
711 remaining_limits.clear();
712 break;
713 }
714 remaining_limits.push_back(remaining_limit);
715 }
716 if (!has_var_factor || remaining_limits.empty()) continue;
717 auto reduced = remaining_limits.size() == 1
718 ? remaining_limits[0]
719 : make_product_or_one(remaining_limits);
720 NormalizationVisitor norm;
721 reduced->accept(norm);
722 result = norm.get_result();
723 return;
724 }
725 for (const auto& op : node.operands()) {
726 auto function = std::dynamic_pointer_cast<const FunctionNode>(op);
727 if (!function || function->arguments().size() != 1 ||
728 function->type() != FunctionNode::FuncType::Exp) {
729 continue;
730 }
731
732 auto exponent_limit = eval_limit(function->arguments()[0]);
733 if (!is_neg_inf(exponent_limit)) continue;
734
735 bool other_growth_is_slower = true;
736 for (const auto& other : node.operands()) {
737 if (other.get() == op.get()) continue;
738 GrowthClass growth = classify_growth(other);
739 if (growth == GrowthClass::Exponential ||
740 growth == GrowthClass::Unknown) {
741 other_growth_is_slower = false;
742 break;
743 }
744 }
745 if (other_growth_is_slower) {
746 result = lamina::detail::make_node<NumberNode>(BigInt(0));
747 return;
748 }
749 }
750 }
751 {
752 for (const auto& op : node.operands()) {
753 auto function = std::dynamic_pointer_cast<const FunctionNode>(op);
754 if (!function || function->arguments().size() != 1) continue;
755 if (function->type() != FunctionNode::FuncType::Sin &&
756 function->type() != FunctionNode::FuncType::Tan) {
757 continue;
758 }
759
760 const auto& argument = function->arguments()[0];
761 std::shared_ptr<const SymbolicNode> reciprocal_argument;
762 for (const auto& candidate : node.operands()) {
763 if (auto power = std::dynamic_pointer_cast<const PowerNode>(candidate)) {
764 auto exponent = std::dynamic_pointer_cast<const NumberNode>(power->exponent());
765 if (exponent && power->base() && power->base()->compare(*argument) == 0) {
766 double exponent_value = 0.0;
767 if (std::holds_alternative<double>(exponent->value())) {
768 exponent_value = std::get<double>(exponent->value());
769 } else if (std::holds_alternative<BigInt>(exponent->value())) {
770 exponent_value = std::get<BigInt>(exponent->value()).to_double();
771 } else if (std::holds_alternative<Rational>(exponent->value())) {
772 exponent_value = std::get<Rational>(exponent->value()).to_double();
773 }
774 if (exponent_value == -1.0) {
775 reciprocal_argument = candidate;
776 break;
777 }
778 }
779 }
780 }
781 if (!reciprocal_argument) continue;
782
783 auto argument_limit = eval_limit(argument);
784 if (argument_limit && argument_limit->is_zero()) {
785 std::vector<std::shared_ptr<const SymbolicNode>> remaining_limits;
786 for (const auto& remaining : node.operands()) {
787 if (remaining.get() == op.get() ||
788 remaining.get() == reciprocal_argument.get()) {
789 continue;
790 }
791 auto remaining_limit = eval_limit(remaining);
792 if (!remaining_limit) {
793 remaining_limits.clear();
794 break;
795 }
796 remaining_limits.push_back(remaining_limit);
797 }
798 remaining_limits.push_back(lamina::detail::make_node<NumberNode>(BigInt(1)));
799 auto reduced = remaining_limits.size() == 1
800 ? remaining_limits[0]
801 : make_product_or_one(remaining_limits);
802 NormalizationVisitor norm;
803 reduced->accept(norm);
804 result = norm.get_result();
805 return;
806 }
807 }
808 }
809 auto squeeze_result = try_squeeze(lamina::detail::make_node<MultiplyNode>(std::vector<std::shared_ptr<const SymbolicNode>>(node.operands().begin(), node.operands().end())));
810 if (squeeze_result) { result = squeeze_result; return; }
811 std::vector<std::shared_ptr<const SymbolicNode>> subs_ops;
812 for (auto& op : node.operands()) { op->accept(*this); subs_ops.push_back(result); }
813 NormalizationVisitor norm;
814 auto subs_res = lamina::detail::make_node<MultiplyNode>(subs_ops); subs_res->accept(norm); auto final_subs = norm.get_result();
815 std::vector<std::shared_ptr<const SymbolicNode>> num_nodes, den_nodes;
816 for (auto& op : node.operands()) {
817 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(op)) {
818 if (auto num = std::dynamic_pointer_cast<const NumberNode>(pow->exponent())) {
819 double e = 0;
820 if (std::holds_alternative<double>(num->value())) e = std::get<double>(num->value());
821 else if (std::holds_alternative<BigInt>(num->value())) e = std::get<BigInt>(num->value()).to_double();
822 else if (std::holds_alternative<Rational>(num->value())) e = std::get<Rational>(num->value()).to_double();
823 if (e < 0) { den_nodes.push_back(lamina::detail::make_node<PowerNode>(pow->base(), lamina::detail::make_node<NumberNode>(-e))); continue; }
824 }
825 }
826 num_nodes.push_back(op);
827 }
828 if (den_nodes.empty()) {
829 auto prod_form = classify_product_form(subs_ops);
830 if (prod_form == IndeterminateForm::ZeroTimesInf) { auto resolved = resolve_zero_times_inf(node.operands(), subs_ops); if (resolved) { result = resolved; return; } }
831 result = final_subs; return;
832 }
833 auto N = make_product_or_one(num_nodes);
834 auto D = make_product_or_one(den_nodes);
835 auto val_n = eval_limit(N); auto val_d = eval_limit(D);
836 bool n_zero = val_n && val_n->is_zero(), d_zero = val_d && val_d->is_zero();
837 bool n_inf = is_inf(val_n), d_inf = is_inf(val_d);
838 /// Limits at infinity: try rational function degree comparison and growth-rate comparison
839 if (is_limit_at_infinity() || is_limit_at_neg_infinity()) {
840 if (n_inf && d_inf) {
841 auto rat_result = limit_rational_at_infinity(N, D);
842 if (rat_result) { result = rat_result; return; }
843 auto growth_result = limit_by_growth_comparison(N, D);
844 if (growth_result) { result = growth_result; return; }
845 }
846 }
847 if ((n_zero && d_zero) || (n_inf && d_inf)) { auto lhop = apply_lhopital(N, D); if (lhop) { result = lhop; return; } }
848 if (!n_zero && d_zero) {
849 int sign_n = 1;
850 if (auto num = std::dynamic_pointer_cast<const NumberNode>(val_n)) {
851 if (std::holds_alternative<double>(num->value())) sign_n = std::get<double>(num->value()) > 0 ? 1 : -1;
852 else if (std::holds_alternative<BigInt>(num->value())) sign_n = std::get<BigInt>(num->value()) > BigInt(0) ? 1 : -1;
853 else if (std::holds_alternative<Rational>(num->value())) sign_n = std::get<Rational>(num->value()) > Rational(0) ? 1 : -1;
854 }
855 int sign_d = (direction == "-" ? -1 : 1);
856 int final_sign = sign_n * sign_d;
857 std::vector<std::shared_ptr<const SymbolicNode>> inf_args;
858 auto inf_node = lamina::detail::make_node<FunctionNode>(FunctionNode::FuncType::Infinity, inf_args);
859 if (final_sign < 0) { std::vector<std::shared_ptr<const SymbolicNode>> m_args = {lamina::detail::make_node<NumberNode>(BigInt(-1)), inf_node}; result = lamina::detail::make_node<MultiplyNode>(m_args); }
860 else { result = inf_node; }
861 return;
862 }
863 result = final_subs;
864}
