48void LimitVisitor::visit(const MatrixNode& node) {
49 if (const auto* dense =
50 std::get_if<MatrixNode::DenseStorage>(&node.storage())) {
51 MatrixNode::DenseStorage entries;
52 entries.reserve(dense->size());
53 for (const auto& entry : *dense) {
54 entry->accept(*this);
55 entries.push_back(result);
56 }
57 result = LMCAS::detail::make_node<MatrixNode>(
58 node.rows(), node.cols(), std::move(entries));
59 return;
60 }
61 MatrixNode::SparseStorage entries;
62 for (const auto& [index, entry] :
63 std::get<MatrixNode::SparseStorage>(node.storage())) {
64 entry->accept(*this);
65 entries.emplace(index, result);
66 }
67 result = LMCAS::detail::make_node<MatrixNode>(
68 node.rows(), node.cols(), std::move(entries));
69}
