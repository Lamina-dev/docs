8/**
9 * @brief 匹配 sin^2(f) 或 cos^2(f)，提取函数类型和参数。
10 * @note 对应 PowerNode(FunctionNode(Sin/Cos, [f]), NumberNode(2))。
11 *
12 * @param[in] node 待检测的 AST 节点
13 * @param[out] func_type 输出函数类型(Sin 或 Cos)
14 * @param[out] argument 输出函数参数节点
15 * @return 匹配成功返回 true
16 * @internal
17 */
18static bool tf_is_trig_squared(
19 const std::shared_ptr<const SymbolicNode>& node,
20 FunctionNode::FuncType& func_type,
21 std::shared_ptr<const SymbolicNode>& argument) {
22
23 auto pow = std::dynamic_pointer_cast<const PowerNode>(node);
24 if (!pow) {
25 return false;
26 }
27
28 auto exp_num = std::dynamic_pointer_cast<const NumberNode>(pow->exponent());
29 if (!exp_num) {
30 return false;
31 }
32
33 bool is_two = false;
34 if (std::holds_alternative<BigInt>(exp_num->value())) {
35 is_two = (std::get<BigInt>(exp_num->value()) == BigInt(2));
36 } else if (std::holds_alternative<Rational>(exp_num->value())) {
37 is_two = (std::get<Rational>(exp_num->value()) == Rational(2));
38 } else if (std::holds_alternative<lmmc_real_t>(exp_num->value())) {
39 lmmc_real_t v = std::get<lmmc_real_t>(exp_num->value());
40 int eq;
41 lmmc_double_nearly_equal(v, 2.0, &eq);
42 is_two = (eq != 0);
43 }
44 if (!is_two) {
45 return false;
46 }
47
48 auto func = std::dynamic_pointer_cast<const FunctionNode>(pow->base());
49 if (!func || func->arguments().size() != 1) {
50 return false;
51 }
52
53 if (func->type() != FunctionNode::FuncType::Sin &&
54 func->type() != FunctionNode::FuncType::Cos) {
55 return false;
56 }
57
58 func_type = func->type();
59 argument = func->arguments()[0];
60 return true;
61}

63/**
64 * @brief 从 a*sin^2(f) 或 a*cos^2(f) 提取系数和三角平方项。
65 * @note 单独的三角平方项系数为 1。
66 *
67 * @param[in] node 待分析的加法操作数节点
68 * @param[out] coeff 输出系数节点(nullptr 表示系数为 1)
69 * @param[out] func_type 输出函数类型(Sin 或 Cos)
70 * @param[out] argument 输出函数参数节点
71 * @return 匹配成功返回 true
72 * @internal
73 */
74static bool tf_extract_coeff_trig_squared(
75 const std::shared_ptr<const SymbolicNode>& node,
76 std::shared_ptr<const SymbolicNode>& coeff,
77 FunctionNode::FuncType& func_type,
78 std::shared_ptr<const SymbolicNode>& argument) {
79
80 if (tf_is_trig_squared(node, func_type, argument)) {
81 coeff = nullptr; /**< 表示系数为 1。 */
82 return true;
83 }
84
85 auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node);
86 if (!mul || mul->operands().size() < 2) {
87 return false;
88 }
89
90 for (size_t i = 0; i < mul->operands().size(); ++i) {
91 if (tf_is_trig_squared(mul->operands()[i], func_type, argument)) {
92 std::vector<std::shared_ptr<const SymbolicNode>> coeff_ops;
93 for (size_t j = 0; j < mul->operands().size(); ++j) {
94 if (j != i) {
95 coeff_ops.push_back(mul->operands()[j]);
96 }
97 }
98
99 if (coeff_ops.size() == 1) {
100 coeff = coeff_ops[0];
101 } else {
102 coeff = LMCAS::detail::make_node<MultiplyNode>(std::move(coeff_ops));
103 }
104 return true;
105 }
106 }
107
108 return false;
109}

120static bool complementary_square(
121 const std::shared_ptr<const SymbolicNode>& node, const TrigSquare& square) {
122 TrigSquare other;
123 if (!tf_extract_coeff_trig_squared(
124 node, other.coefficient, other.type, other.argument)) {
125 return false;
126 }
127 const auto target = square.type == FunctionNode::FuncType::Sin
128 ? FunctionNode::FuncType::Cos : FunctionNode::FuncType::Sin;
129 if (other.type != target || !square.argument || !other.argument ||
130 !square.argument->equals(*other.argument)) {
131 return false;
132 }
133 if (!square.coefficient || !other.coefficient) {
134 return !square.coefficient && !other.coefficient;
135 }
136 return square.coefficient->equals(*other.coefficient);
137}

150static std::shared_ptr<const SymbolicNode> simplify_pythagorean_sum(
151 const AddNode& sum, const std::string& var) {
152 std::vector<std::shared_ptr<const SymbolicNode>> operands;
153 operands.reserve(sum.operands().size());
154 for (const auto& operand : sum.operands()) {
155 operands.push_back(tf_simplify_pythagorean_node(operand, var));
156 }
157 std::vector<bool> consumed(operands.size(), false);
158 std::vector<std::shared_ptr<const SymbolicNode>> result;
159 for (std::size_t i = 0; i < operands.size(); ++i) {
160 if (consumed[i]) {
161 continue;
162 }
163 TrigSquare square;
164 if (!tf_extract_coeff_trig_squared(
165 operands[i], square.coefficient, square.type, square.argument)) {
166 result.push_back(operands[i]);
167 continue;
168 }
169 const auto partner = find_complementary_square(operands, consumed, i + 1, square);
170 if (partner == operands.size()) {
171 result.push_back(operands[i]);
172 continue;
173 }
174 consumed[i] = true;
175 consumed[partner] = true;
176 if (square.coefficient) {
177 result.push_back(square.coefficient);
178 } else {
179 result.push_back(LMCAS::detail::make_node<NumberNode>(BigInt(1)));
180 }
181 }
182 if (result.empty()) {
183 return LMCAS::detail::make_node<NumberNode>(BigInt(0));
184 }
185 if (result.size() == 1) {
186 return result[0];
187 }
188 return LMCAS::detail::make_node<AddNode>(std::move(result));
189}

191/**
192 * @brief 递归化简 a*sin^2(f) + a*cos^2(f) 为 a，系数为 1 时结果为 1。
193 * @note 在加法节点中配对，要求参数 f 与系数 a 分别结构相等。
194 *
195 * @param[in] node 当前 AST 节点
196 * @param[in] var 目标变量名(用于限定化简范围)
197 * @return 化简后的节点;若无可化简的模式则返回原节点
198 * @internal
199 */
200static std::shared_ptr<const SymbolicNode> tf_simplify_pythagorean_node(
201 const std::shared_ptr<const SymbolicNode>& node,
202 const std::string& var) {
203
204 if (!node) {
205 return node;
206 }
207
208 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
209 return simplify_pythagorean_sum(*add, var);
210 }
211
212 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
213 std::vector<std::shared_ptr<const SymbolicNode>> new_ops;
214 new_ops.reserve(mul->operands().size());
215 for (const auto& op : mul->operands()) {
216 new_ops.push_back(tf_simplify_pythagorean_node(op, var));
217 }
218 return LMCAS::detail::make_node<MultiplyNode>(std::move(new_ops));
219 }
220
221 if (auto pow = std::dynamic_pointer_cast<const PowerNode>(node)) {
222 auto new_base = tf_simplify_pythagorean_node(pow->base(), var);
223 auto new_exp = tf_simplify_pythagorean_node(pow->exponent(), var);
224 return LMCAS::detail::make_node<PowerNode>(std::move(new_base), std::move(new_exp));
225 }
226
227 if (auto func = std::dynamic_pointer_cast<const FunctionNode>(node)) {
228 std::vector<std::shared_ptr<const SymbolicNode>> new_args;
229 new_args.reserve(func->arguments().size());
230 for (const auto& arg : func->arguments()) {
231 new_args.push_back(tf_simplify_pythagorean_node(arg, var));
232 }
233 return LMCAS::detail::make_node<FunctionNode>(func->type(), std::move(new_args));
234 }
235
236 return node;
237}

241 *
242 * @param[in] expr 待化简的符号表达式
243 * @param[in] var 目标变量名
244 * @return 化简后的表达式;若无可化简模式则返回原表达式
245 * @internal
246 */
247std::shared_ptr<SymbolicExpr> tf_simplify_pythagorean(
248 const std::shared_ptr<SymbolicExpr>& expr,
249 const std::string& var) {
250
251 if (!expr || !LMCAS::detail::node(expr)) {
252 return expr;
253 }
254
255 auto simplified_root = tf_simplify_pythagorean_node(LMCAS::detail::node(expr), var);
256
257 if (!simplified_root) {
258 return expr;
259 }
260 if (simplified_root->equals(*LMCAS::detail::node(expr))) {
261 return expr;
262 }
263
264 return LMCAS::detail::make_expression_ptr(simplified_root);
265}
