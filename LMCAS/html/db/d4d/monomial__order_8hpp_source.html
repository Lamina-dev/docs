36 * @param a 第一个单项式
37 * @param b 第二个单项式
38 * @return 各分量取最大值构成的单项式

40inline Monomial lcm_monomial(const Monomial& a, const Monomial& b) {
41 Monomial result(std::max(a.size(), b.size()), 0);
42 for (size_t i = 0; i < a.size(); ++i)
43 result[i] = std::max(result[i], a[i]);
44 for (size_t i = 0; i < b.size(); ++i)
45 result[i] = std::max(result[i], b[i]);
46 return result;
47}

49/**
50 * @brief 判断 divisor 是否整除 target
51 * @param divisor 除数单项式
52 * @param target 被除数单项式
53 * @return divisor 的每个分量都不超过 target 对应分量时返回 true
54 */
55inline bool divides_monomial(const Monomial& divisor, const Monomial& target) {
56 if (divisor.size() > target.size()) {
57 for (size_t i = target.size(); i < divisor.size(); ++i)
58 if (divisor[i] != 0) return false;
59 }
60 for (size_t i = 0; i < std::min(divisor.size(), target.size()); ++i) {
61 if (divisor[i] > target[i]) return false;
62 }
63 return true;
64}

66/** @brief 单项式序比较器，支持 Lex、GrevLex、DegLex、DegRevLex */
67class MonomialOrder {
68public:
69 /**
70 * @brief 构造指定类型的单项式序

72 */
73 explicit MonomialOrder(MonomialOrderType type) : type_(type) {}

75 /**
76 * @brief 比较两个单项式的大小
77 * @param a 左操作数
78 * @param b 右操作数
79 * @return a > b 时返回 true（按当前序）
80 */
81 inline bool operator()(const Monomial& a, const Monomial& b) const {
82 switch (type_) {
83 case MonomialOrderType::Lex:
84 return compare_lex(a, b);
85 case MonomialOrderType::GrevLex:
86 case MonomialOrderType::DegRevLex:
87 return compare_grevlex(a, b);
88 case MonomialOrderType::DegLex:
89 return compare_deglex(a, b);
90 }
91 return false;
92 }

112 static inline bool compare_lex(const Monomial& a, const Monomial& b) {
113 size_t n = std::max(a.size(), b.size());
114 for (size_t i = 0; i < n; ++i) {
115 int ai = (i < a.size()) ? a[i] : 0;
116 int bi = (i < b.size()) ? b[i] : 0;
117 if (ai != bi) return ai > bi;
118 }
119 return false;
120 }
121
122 static inline bool compare_grevlex(const Monomial& a, const Monomial& b) {
123 int deg_a = total_degree(a);
124 int deg_b = total_degree(b);
125 if (deg_a != deg_b) return deg_a > deg_b;
126
127 size_t n = std::max(a.size(), b.size());
128 for (size_t i = n; i > 0; --i) {
129 int ai = (i - 1 < a.size()) ? a[i - 1] : 0;
130 int bi = (i - 1 < b.size()) ? b[i - 1] : 0;
131 if (ai != bi) return ai < bi;
132 }
133 return false;
134 }
135
136 static inline bool compare_deglex(const Monomial& a, const Monomial& b) {
137 int deg_a = total_degree(a);
138 int deg_b = total_degree(b);
139 if (deg_a != deg_b) return deg_a > deg_b;
140 return compare_lex(a, b);
141 }
