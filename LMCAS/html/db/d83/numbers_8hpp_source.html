29 lmmc_real_t real = std::get<lmmc_real_t>(value_);
30 if (real == 0.0) real = 0.0; // canonicalize negative zero
31 std::size_t seed = std::hash<int>{}(2); // approximate-number domain
32 hash_combine(seed, std::hash<lmmc_real_t>{}(real));
33 return seed;
34 }

37 // reals intentionally occupy a different structural domain.
38 if (std::holds_alternative<Rational>(value_)) {
39 return std::get<Rational>(value_).hash();
40 }
41 return Rational(std::get<BigInt>(value_)).hash();

46
47 bool is_l_real = std::holds_alternative<lmmc_real_t>(value_);
48 bool is_r_real = std::holds_alternative<lmmc_real_t>(o.value_);
49
50 if (is_l_real) {
51 if (!is_r_real) {
52 const Rational lhs = Rational::from_double(std::get<lmmc_real_t>(value_));
53 const Rational rhs = std::holds_alternative<Rational>(o.value_)
54 ? std::get<Rational>(o.value_)
55 : Rational(std::get<BigInt>(o.value_));
56 if (lhs < rhs) return -1;
57 if (rhs < lhs) return 1;
58 return 1; // same numeric value, but approximate numbers remain structurally distinct
59 }
60 const lmmc_real_t lhs = std::get<lmmc_real_t>(value_);
61 const lmmc_real_t rhs = std::get<lmmc_real_t>(o.value_);
62 const bool lhs_nan = std::isnan(lhs);
63 const bool rhs_nan = std::isnan(rhs);
64 if (lhs_nan || rhs_nan) {
65 if (lhs_nan && rhs_nan) return 0;
66 return lhs_nan ? 1 : -1;
67 }
68 if (lhs < rhs) return -1;
69 if (lhs > rhs) return 1;
70 return 0;
71 }
72 if (is_r_real) {
73 const Rational lhs = std::holds_alternative<Rational>(value_)
74 ? std::get<Rational>(value_)
75 : Rational(std::get<BigInt>(value_));
76 const Rational rhs = Rational::from_double(std::get<lmmc_real_t>(o.value_));
77 if (lhs < rhs) return -1;
78 if (rhs < lhs) return 1;
79 return -1; // same numeric value, but exact numbers remain structurally distinct
80 }

82 bool is_l_rat = std::holds_alternative<Rational>(value_);
83 bool is_r_rat = std::holds_alternative<Rational>(o.value_);
84
85 if (is_l_rat || is_r_rat) {
86 Rational r1 = is_l_rat ? std::get<Rational>(value_) : Rational(std::get<BigInt>(value_));
87 Rational r2 = is_r_rat ? std::get<Rational>(o.value_) : Rational(std::get<BigInt>(o.value_));
88 if (r1 < r2) return -1;
89 if (r1 == r2) return 0;
90 return 1;
91 }

93 const auto& b1 = std::get<BigInt>(value_);
