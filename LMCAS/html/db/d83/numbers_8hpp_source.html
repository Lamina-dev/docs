44 std::size_t compute_hash() const override {
45 if (std::holds_alternative<lmmc_real_t>(value_)) {
46 lmmc_real_t real = std::get<lmmc_real_t>(value_);
47 if (real == 0.0) real = 0.0; // canonicalize negative zero
48 std::size_t seed = std::hash<int>{}(2); // approximate-number domain
49 hash_combine(seed, std::hash<lmmc_real_t>{}(real));
50 return seed;
51 }
52
53 // BigInt n and Rational n/1 are the same exact number. Approximate
54 // reals intentionally occupy a different structural domain.
55 if (std::holds_alternative<Rational>(value_)) {
56 return std::get<Rational>(value_).hash();
57 }
58 return Rational(std::get<BigInt>(value_)).hash();
59 }

61 int compare_same_type(const SymbolicNode& other) const override {
62 const auto& o = static_cast<const NumberNode&>(other);
63
64 bool is_l_real = std::holds_alternative<lmmc_real_t>(value_);
65 bool is_r_real = std::holds_alternative<lmmc_real_t>(o.value_);
66
67 if (is_l_real) {
68 if (!is_r_real) {
69 const Rational lhs = Rational::from_double(std::get<lmmc_real_t>(value_));
70 const Rational rhs = std::holds_alternative<Rational>(o.value_)
71 ? std::get<Rational>(o.value_)
72 : Rational(std::get<BigInt>(o.value_));
73 if (lhs < rhs) return -1;
74 if (rhs < lhs) return 1;
75 return 1; // same numeric value, but approximate numbers remain structurally distinct
76 }
77 const lmmc_real_t lhs = std::get<lmmc_real_t>(value_);
78 const lmmc_real_t rhs = std::get<lmmc_real_t>(o.value_);
79 if (lhs < rhs) return -1;
80 if (lhs > rhs) return 1;
81 return 0;
82 }
83 if (is_r_real) {
84 const Rational lhs = std::holds_alternative<Rational>(value_)
85 ? std::get<Rational>(value_)
86 : Rational(std::get<BigInt>(value_));
87 const Rational rhs = Rational::from_double(std::get<lmmc_real_t>(o.value_));
88 if (lhs < rhs) return -1;
89 if (rhs < lhs) return 1;
90 return -1; // same numeric value, but exact numbers remain structurally distinct
91 }
92
93 bool is_l_rat = std::holds_alternative<Rational>(value_);
94 bool is_r_rat = std::holds_alternative<Rational>(o.value_);
95
96 if (is_l_rat || is_r_rat) {
97 Rational r1 = is_l_rat ? std::get<Rational>(value_) : Rational(std::get<BigInt>(value_));
98 Rational r2 = is_r_rat ? std::get<Rational>(o.value_) : Rational(std::get<BigInt>(o.value_));
99 if (r1 < r2) return -1;
100 if (r1 == r2) return 0;
101 return 1;
102 }
103
104 const auto& b1 = std::get<BigInt>(value_);
105 const auto& b2 = std::get<BigInt>(o.value_);
106 if (b1 < b2) return -1;
107 if (b1 == b2) return 0;
108 return 1;
109 }

122

124 if (std::holds_alternative<Rational>(value_)) return std::get<Rational>(value_) == Rational(0);
125 return std::get<lmmc_real_t>(value_) == 0.0;
126 }
127 bool is_one() const override {
128 if (std::holds_alternative<BigInt>(value_)) return std::get<BigInt>(value_) == BigInt(1);
129 if (std::holds_alternative<Rational>(value_)) return std::get<Rational>(value_) == Rational(1);
130 return std::get<lmmc_real_t>(value_) == 1.0;
131 }
132 bool is_positive() const override {
133 if (std::holds_alternative<BigInt>(value_)) {
134 const auto& b = std::get<BigInt>(value_);
135 return !b.IsNegative() && !(b == BigInt(0));
136 }
137 if (std::holds_alternative<Rational>(value_)) {
138 const auto& r = std::get<Rational>(value_);
139 return r > Rational(0);
140 }
141 lmmc_real_t v = std::get<lmmc_real_t>(value_);
142 return std::isfinite(v) && v > 0.0;
143 }
