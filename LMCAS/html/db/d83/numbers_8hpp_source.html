9/**
10 * @brief 数值节点，存储 BigInt、Rational 或浮点数。
11 */
12class NumberNode : public SymbolicNode {
13private:
14 LMCAS_AST_NODE_FACTORY_FRIEND;
15
16 const std::variant<BigInt, Rational, lmmc_real_t> value_;
17 static lmmc_real_t validate_approximate(lmmc_real_t value) {
18 if (!std::isfinite(value)) {
19 throw std::invalid_argument("approximate number must be finite");
20 }
21 return value;
22 }
23
24 static std::variant<BigInt, Rational, lmmc_real_t> validate_number(
25 std::variant<BigInt, Rational, lmmc_real_t> value) {
26 if (const auto* approximate = std::get_if<lmmc_real_t>(&value)) {
27 validate_approximate(*approximate);
28 }
29 return value;
30 }
31
32 explicit NumberNode(const BigInt& v) : value_(v) {}
33 explicit NumberNode(const Rational& v) : value_(v) {}
34 explicit NumberNode(lmmc_real_t v) : value_(validate_approximate(v)) {}
35 explicit NumberNode(std::variant<BigInt, Rational, lmmc_real_t> v)
36 : value_(validate_number(std::move(v))) {}
37
38public:
39 const std::variant<BigInt, Rational, lmmc_real_t>& value() const noexcept {
40 return value_;
41 }
42
43 int type_priority() const override { return -10; }
44
45protected:
46 std::size_t compute_hash() const override {
47 if (std::holds_alternative<lmmc_real_t>(value_)) {
48 lmmc_real_t real = std::get<lmmc_real_t>(value_);
49 if (real == 0.0) real = 0.0; // canonicalize negative zero
50 std::size_t seed = std::hash<int>{}(2); // approximate-number domain
51 hash_combine(seed, std::hash<lmmc_real_t>{}(real));
52 return seed;
53 }
54
55 // BigInt n and Rational n/1 are the same exact number. Approximate
56 // reals intentionally occupy a different structural domain.
57 if (std::holds_alternative<Rational>(value_)) {
58 return std::get<Rational>(value_).hash();
59 }
60 return Rational(std::get<BigInt>(value_)).hash();
61 }
62
63 int compare_same_type(const SymbolicNode& other) const override {
64 const auto& o = static_cast<const NumberNode&>(other);
65
66 bool is_l_real = std::holds_alternative<lmmc_real_t>(value_);
67 bool is_r_real = std::holds_alternative<lmmc_real_t>(o.value_);
68
69 if (is_l_real) {
70 if (!is_r_real) {
71 const Rational lhs = Rational::from_double(std::get<lmmc_real_t>(value_));
72 const Rational rhs = std::holds_alternative<Rational>(o.value_)
73 ? std::get<Rational>(o.value_)
74 : Rational(std::get<BigInt>(o.value_));
75 if (lhs < rhs) return -1;
76 if (rhs < lhs) return 1;
77 return 1; // same numeric value, but approximate numbers remain structurally distinct
78 }
79 const lmmc_real_t lhs = std::get<lmmc_real_t>(value_);
80 const lmmc_real_t rhs = std::get<lmmc_real_t>(o.value_);
81 if (lhs < rhs) return -1;
82 if (lhs > rhs) return 1;
83 return 0;
84 }
85 if (is_r_real) {
86 const Rational lhs = std::holds_alternative<Rational>(value_)
87 ? std::get<Rational>(value_)
88 : Rational(std::get<BigInt>(value_));
89 const Rational rhs = Rational::from_double(std::get<lmmc_real_t>(o.value_));
90 if (lhs < rhs) return -1;
91 if (rhs < lhs) return 1;
92 return -1; // same numeric value, but exact numbers remain structurally distinct
93 }
94
95 bool is_l_rat = std::holds_alternative<Rational>(value_);
96 bool is_r_rat = std::holds_alternative<Rational>(o.value_);
97
98 if (is_l_rat || is_r_rat) {
99 Rational r1 = is_l_rat ? std::get<Rational>(value_) : Rational(std::get<BigInt>(value_));
100 Rational r2 = is_r_rat ? std::get<Rational>(o.value_) : Rational(std::get<BigInt>(o.value_));
101 if (r1 < r2) return -1;
102 if (r1 == r2) return 0;
103 return 1;
104 }
105
106 const auto& b1 = std::get<BigInt>(value_);
107 const auto& b2 = std::get<BigInt>(o.value_);
108 if (b1 < b2) return -1;
109 if (b1 == b2) return 0;
110 return 1;
111 }
112
113public:
114 void accept(LMCAS::detail::SymbolicVisitor& visitor) const override { LMCAS::detail::SymbolicVisitor::DepthGuard guard(visitor); visitor.visit(*this); }
115 std::shared_ptr<const SymbolicNode> clone() const override {
116
117 if (std::holds_alternative<BigInt>(value_)) return LMCAS::detail::make_node<NumberNode>(std::get<BigInt>(value_));
118 if (std::holds_alternative<Rational>(value_)) return LMCAS::detail::make_node<NumberNode>(std::get<Rational>(value_));
119 return LMCAS::detail::make_node<NumberNode>(std::get<lmmc_real_t>(value_));
120 }
121
122 bool is_number() const override { return true; }
123 bool is_zero() const override {
124
125 if (std::holds_alternative<BigInt>(value_)) return std::get<BigInt>(value_) == BigInt(0);
126 if (std::holds_alternative<Rational>(value_)) return std::get<Rational>(value_) == Rational(0);
127 return std::get<lmmc_real_t>(value_) == 0.0;
128 }
129 bool is_one() const override {
130 if (std::holds_alternative<BigInt>(value_)) return std::get<BigInt>(value_) == BigInt(1);
131 if (std::holds_alternative<Rational>(value_)) return std::get<Rational>(value_) == Rational(1);
132 return std::get<lmmc_real_t>(value_) == 1.0;
133 }
134 bool is_positive() const override {
135 if (std::holds_alternative<BigInt>(value_)) {
136 const auto& b = std::get<BigInt>(value_);
137 return !b.IsNegative() && !(b == BigInt(0));
138 }
139 if (std::holds_alternative<Rational>(value_)) {
140 const auto& r = std::get<Rational>(value_);
141 return r > Rational(0);
142 }
143 lmmc_real_t v = std::get<lmmc_real_t>(value_);
144 return std::isfinite(v) && v > 0.0;
145 }

151class ComplexNode : public SymbolicNode {
152private:
153 LMCAS_AST_NODE_FACTORY_FRIEND;
154
155 const std::shared_ptr<const SymbolicNode> real_;
156 const std::shared_ptr<const SymbolicNode> imag_;
157
158 ComplexNode(std::shared_ptr<const SymbolicNode> r, std::shared_ptr<const SymbolicNode> i)
159 : real_(std::move(r)), imag_(std::move(i)) {
160 if (!real_ || !imag_) {
161 throw std::invalid_argument("ComplexNode real and imaginary parts cannot be null");
162 }
163 }
164
165public:
166 const std::shared_ptr<const SymbolicNode>& real() const noexcept { return real_; }
167 const std::shared_ptr<const SymbolicNode>& imag() const noexcept { return imag_; }
168
169 int type_priority() const override { return -5; } // 在 NumberNode 和 VariableNode 之间
170
171protected:
172 std::size_t compute_hash() const override {
173 std::size_t seed = 0;
174 hash_combine(seed, type_priority());
175 hash_combine(seed, real_->hash());
176 hash_combine(seed, imag_->hash());
177 return seed;
178 }
179
180 int compare_same_type(const SymbolicNode& other) const override {
181 const auto& o = static_cast<const ComplexNode&>(other);
182 int cmp_r = real_->compare(*o.real_);
183 if (cmp_r != 0) return cmp_r;
184 return imag_->compare(*o.imag_);
185 }
186
187public:
188 void accept(LMCAS::detail::SymbolicVisitor& visitor) const override { LMCAS::detail::SymbolicVisitor::DepthGuard guard(visitor); visitor.visit(*this); }
189 std::shared_ptr<const SymbolicNode> clone() const override {
190 return LMCAS::detail::make_node<ComplexNode>(real_->clone(), imag_->clone());
191 }
192
193 bool is_zero() const override {
194 return real_->is_zero() && imag_->is_zero();
195 }
196};
