10 auto num = std::dynamic_pointer_cast<const NumberNode>(node);

50static lmmc_real_t extract_linear_coefficient(
51 const std::shared_ptr<const SymbolicNode>& node,
52 const std::string& var)
53{
54 if (!node || !expression_depends_on_variable(node, var)) {
55 return 0.0;
56 }
57
58 if (auto v = std::dynamic_pointer_cast<const VariableNode>(node)) {
59 if (!v->is_constant() && v->name() == var) {
60 return 1.0;
61 }
62 return std::numeric_limits<lmmc_real_t>::quiet_NaN();
63 }
64
65 if (auto mul = std::dynamic_pointer_cast<const MultiplyNode>(node)) {
66 return product_linear_coefficient(*mul, var);
67 }
68
69 if (auto add = std::dynamic_pointer_cast<const AddNode>(node)) {
70 lmmc_real_t coeff = std::numeric_limits<lmmc_real_t>::quiet_NaN();
71 bool found_var_term = false;
72
73 for (const auto& op : add->operands()) {
74 if (!expression_depends_on_variable(op, var)) {
75 continue;
76 }
77 if (found_var_term) {
78 return std::numeric_limits<lmmc_real_t>::quiet_NaN();
79 }
80 found_var_term = true;
81 coeff = extract_linear_coefficient(op, var);
82 }
83
84 return found_var_term ? coeff : std::numeric_limits<lmmc_real_t>::quiet_NaN();
85 }
86
87 return std::numeric_limits<lmmc_real_t>::quiet_NaN();
88}

102// Sine and cosine use 2pi/|k|; tangent uses pi/|k|.
103struct PeriodicCollector : public LMCAS::detail::RecursiveSymbolicVisitor {
104 const std::string& target_var;
105 lmmc_real_t max_period = 0.0;
106 bool found_periodic = false;
107 bool has_nonlinear_periodic = false;
108
109 explicit PeriodicCollector(const std::string& v) : target_var(v) {}
110
111 void visit(const MatrixNode&) override {}
112
113 void visit(const TransformNode& node) override {
114 visit_child(node.body());
115 }

117 void collect_period(const FunctionNode& n) {
118 if (!is_periodic_func(n.type()) || n.arguments().empty()) {
119 return;
120 }
121 if (!expression_depends_on_variable(n.arguments()[0], target_var)) {
122 return;
123 }
124 found_periodic = true;
125 const auto k = extract_linear_coefficient(n.arguments()[0], target_var);
126 if (std::isnan(k) || k == 0.0) {
127 has_nonlinear_periodic = true;
128 return;
129 }
130 const auto period = n.type() == FunctionNode::FuncType::Tan
131 ? LMMC_CONST_PI / std::fabs(k)
132 : 2.0 * LMMC_CONST_PI / std::fabs(k);
133 if (period > max_period) {
134 max_period = period;
135 }
136 }
137
138 void visit(const FunctionNode& node) override {
139 collect_period(node);
140 LMCAS::detail::RecursiveSymbolicVisitor::visit(node);
141 }
142};

146std::optional<SearchInterval> determine_search_interval(
147 const std::shared_ptr<SymbolicExpr>& expr,
148 const std::string& var,
149 const SolveOptions& opts)
150{
151 if (opts.has_search_interval) {
152 if (opts.search_lo >= opts.search_hi) {
153 return std::nullopt;
154 }
155 if ((opts.search_hi - opts.search_lo) <= opts.tolerance) {
156 return std::nullopt;
157 }
158 return SearchInterval{opts.search_lo, opts.search_hi};
159 }
160
161 lmmc_real_t lo = -10.0;
162 lmmc_real_t hi = 10.0;
163
164 if (expr && LMCAS::detail::node(expr)) {
165 PeriodicCollector collector(var);
166 LMCAS::detail::node(expr)->accept(collector);
167
168 if (collector.found_periodic && !collector.has_nonlinear_periodic && collector.max_period > 0.0) {
169 // [-period, period] covers two complete periods around zero.
170 lmmc_real_t half_span = collector.max_period;
171 if (half_span > lo * -1.0 || half_span > hi) {
172 lo = -half_span;
173 hi = half_span;
174 }
175 // Cap automatic interval growth to keep numeric scanning bounded.
176 if (lo < -100.0) {
177 lo = -100.0;
178 }
179 if (hi > 100.0) {
180 hi = 100.0;
181 }
182 }
183 }
184
185 if (lo >= hi) {
186 return std::nullopt;
187 }
188 if ((hi - lo) <= opts.tolerance) {
189 return std::nullopt;
190 }
191
192 return SearchInterval{lo, hi};
193}
