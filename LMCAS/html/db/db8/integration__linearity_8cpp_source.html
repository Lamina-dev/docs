7std::shared_ptr<const SymbolicNode> integration_exponential_argument(
8 const std::shared_ptr<const SymbolicNode>& factor) {
9 const auto* power = dynamic_cast<const PowerNode*>(factor.get());
10 const auto& base = power ? power->base() : factor;
11 const auto* function = dynamic_cast<const FunctionNode*>(base.get());
12 if (!function || function->type() != FunctionNode::FuncType::Exp ||
13 function->arguments().size() != 1) {
14 return nullptr;
15 }
16 if (!power) { return function->arguments()[0]; }
17 BigInt integer;
18 if (!try_get_integer_value(
19 std::dynamic_pointer_cast<const NumberNode>(power->exponent()), integer) ||
20 integer.is_zero()) {
21 return nullptr;
22 }
23 return make_normalized_multiply_node(
24 {power->exponent(), function->arguments()[0]});
25}

27std::shared_ptr<const SymbolicNode> combine_integration_exponentials(
28 const std::vector<std::shared_ptr<const SymbolicNode>>& dependents) {
29 if (dependents.empty() || (dependents.size() == 1 &&
30 !dynamic_cast<const PowerNode*>(dependents[0].get()))) {
31 return nullptr;
32 }
33 std::vector<std::shared_ptr<const SymbolicNode>> exponents;
34 exponents.reserve(dependents.size());
35 for (const auto& dependent : dependents) {
36 auto argument = integration_exponential_argument(dependent);
37 if (!argument) { return nullptr; }
38 exponents.push_back(std::move(argument));
39 }
40 return detail::make_node<FunctionNode>(
41 FunctionNode::FuncType::Exp,
42 std::vector<std::shared_ptr<const SymbolicNode>>{
43 SymbolicFactory::create_add(std::move(exponents))});
44}
