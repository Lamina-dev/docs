41QuantityResult make_quantity(std::shared_ptr<const SymbolicNode> value,
42 DimensionSignature dimension,
43 Rational scale,
44 std::string display) {
45 try {
46 if (dimension.is_dimensionless()) {
47 return QuantityResult::success(detail::make_expression_ptr(
48 scale == Rational(1)
49 ? std::move(value)
50 : SymbolicFactory::create_multiply({std::move(value), rational_node(scale)})));
51 }
52 return QuantityResult::success(detail::make_expression_ptr(
53 detail::make_node<QuantityNode>(std::move(value), std::move(dimension),
54 std::move(scale), std::move(display))));
55 } catch (const std::bad_alloc&) {
56 return failure(CasErrc::ResourceLimit, "quantity allocation failed");
57 } catch (const std::exception& error) {
58 return failure(CasErrc::InvalidArgument, error.what());
59 }

62Result<Rational> exact_rational(const std::shared_ptr<const SymbolicNode>& node) {
63 auto number = std::dynamic_pointer_cast<const NumberNode>(node);
64 if (!number) return Result<Rational>::failure(
65 CasErrc::UnitInvalid, "quantity exponent must be an exact rational", kOperation);
66 if (std::holds_alternative<BigInt>(number->value())) {
67 return Result<Rational>::success(Rational(std::get<BigInt>(number->value())));
68 }
69 if (std::holds_alternative<Rational>(number->value())) {
70 return Result<Rational>::success(std::get<Rational>(number->value()));
71 }
72 return Result<Rational>::failure(
73 CasErrc::UnitInvalid, "approximate quantity exponents are not supported", kOperation);
74}

104 return make_quantity(detail::node(value), definition.value().dimension,
105 definition.value().scale_to_base, unit);

128QuantityResult convert_unit(const std::shared_ptr<SymbolicExpr>& expression,
129 const std::string& target_unit,
130 ComputationContext& context) {
131 auto step = context.consume_steps(1, kOperation);
132 if (!step) return QuantityResult::failure(step.error());
133 auto quantity = expression
134 ? std::dynamic_pointer_cast<const QuantityNode>(detail::node(expression))
135 : nullptr;
136 if (!quantity) {
137 return failure(CasErrc::UnitStripTypeMismatch,
138 "unit conversion requires a quantity expression");
139 }
140 auto target = context.units().resolve(target_unit);
141 if (!target) return QuantityResult::failure(target.error());
142 if (quantity->dimension() != target.value().dimension) {
143 return failure(CasErrc::DimensionMismatch,
144 "unit conversion requires equal dimensions");
145 }
146 const Rational factor = quantity->scale_to_base() / target.value().scale_to_base;
147 auto value = SymbolicFactory::create_multiply(
148 {quantity->value(), rational_node(factor)});
149 return make_quantity(std::move(value), target.value().dimension,
150 target.value().scale_to_base, target_unit);
151}

211QuantityResult quantity_add(const std::shared_ptr<SymbolicExpr>& lhs,
212 const std::shared_ptr<SymbolicExpr>& rhs,
213 ComputationContext& context) {
214 auto checked = checked_binary(lhs, rhs, context, "quantity.add");
215 if (!checked) return QuantityResult::failure(checked.error());
216 const auto left = view(lhs);
217 const auto right = view(rhs);
218 if (left.dimension != right.dimension) {
219 return failure(CasErrc::DimensionMismatch,
220 "addition requires operands with equal dimensions");
221 }
222 const Rational factor = right.scale / left.scale;
223 auto right_value = SymbolicFactory::create_multiply(
224 {right.value, rational_node(factor)});
225 return make_quantity(SymbolicFactory::create_add({left.value, right_value}),
226 left.dimension, left.scale, left.display);

248 const std::shared_ptr<SymbolicExpr>& rhs,

251 if (!checked) return QuantityResult::failure(checked.error());
252 const auto left = view(lhs);

262 const std::shared_ptr<SymbolicExpr>& rhs,
263 ComputationContext& context) {

273 left.dimension.divided_by(right.dimension),

277QuantityResult quantity_power(const std::shared_ptr<SymbolicExpr>& base,
278 const std::shared_ptr<SymbolicExpr>& exponent,
279 ComputationContext& context) {
280 auto checked = checked_binary(base, exponent, context, "quantity.power");
281 if (!checked) return QuantityResult::failure(checked.error());
282 const auto exponent_view = view(exponent);
283 if (!exponent_view.dimension.is_dimensionless()) {
284 return failure(CasErrc::DimensionMismatch,
285 "a quantity exponent must be dimensionless");
286 }
287 auto power = exact_rational(exponent_view.value);
288 if (!power) return QuantityResult::failure(power.error());
289 const auto base_view = view(base);
290 auto base_value = scaled_value(base_view);
291 return make_quantity(SymbolicFactory::create_power(base_value, exponent_view.value),
292 base_view.dimension.raised_to(power.value()),
293 Rational(1), base_view.dimension.raised_to(power.value()).to_string());
294}
