26SolveResult solve_inner_equation(
27 const ExprPtr& original, const ExprPtr& inner, const ExprPtr& value,
28 const std::string& var, ComputationContext& context,
29 const SolveOptions& options, int depth) {
30 auto variable = std::dynamic_pointer_cast<const VariableNode>(detail::node(inner));
31 if (variable && !variable->is_constant() && variable->name() == var) {
32 return SolutionSet{FiniteSolutions{{FiniteSolution{value, 1, {}}}}};
33 }
34 const auto product =
35 std::dynamic_pointer_cast<const MultiplyNode>(detail::node(inner));
36 if (product && product->operands().size() == 2) {
37 const auto& first = product->operands()[0];
38 const auto& second = product->operands()[1];
39 const auto coefficient =
40 std::dynamic_pointer_cast<const NumberNode>(first);
41 const auto product_variable =
42 std::dynamic_pointer_cast<const VariableNode>(second);
43 if (coefficient && coefficient->is_negative_one() &&
44 product_variable && !product_variable->is_constant() &&
45 product_variable->name() == var) {
46 auto inverted = SymbolicExpr::multiply(
47 SymbolicExpr::number(-1), value)->simplify();
48 return SolutionSet{FiniteSolutions{{
49 FiniteSolution{std::move(inverted), 1, {}}}}};
50 }
51 }
52 auto equation = SymbolicExpr::add(inner,
53 SymbolicExpr::multiply(value, SymbolicExpr::number(-1)));
54 auto polynomial = symbolic_to_poly<SymbolicPolyCoeff>(equation, var);
55 if (polynomial) return solve_equation(equation, var, context, options);
56 if (polynomial.error().code != CasErrc::UnsupportedExpression) {
57 return SolveResult::failure(polynomial.error());
58 }
59 return solve_transcendental_impl(original, equation, var, context, options, depth + 1);
60}

62void append_finite_union(FiniteSolutions& finite, FiniteSolutions& prior) {
63 for (auto& candidate : finite.values) {
64 auto duplicate = std::find_if(prior.values.begin(), prior.values.end(),
65 [&](const FiniteSolution& previous) {
66 return candidate.conditions.empty() && previous.conditions.empty() &&
67 detail::node(candidate.value)->equals(*detail::node(previous.value));
68 });
69 if (duplicate == prior.values.end()) { prior.values.push_back(std::move(candidate)); }
70 }
71}

73void append_parametric_union(FiniteSolutions* finite, ParametricSolutions* families,
74 FiniteSolutions* prior, SolutionSet& output) {
75 if (prior) {
76 ParametricSolutions promoted;

85 result.push_back({std::move(point.value), {}, std::move(point.conditions)});

91void append_solutions(SolutionSet solved, SolutionSet& output,
92 const ExprPtr& original, const std::string& var) {
93 if (std::holds_alternative<EmptySolutions>(solved) ||
94 std::holds_alternative<UniversalSolutions>(output)) { return; }
95 if (std::holds_alternative<EmptySolutions>(output) ||
96 std::holds_alternative<UniversalSolutions>(solved)) {
97 output = std::move(solved);
98 return;
99 }
100 auto* finite = std::get_if<FiniteSolutions>(&solved);
101 auto* prior = std::get_if<FiniteSolutions>(&output);
102 if (finite && prior) {
103 append_finite_union(*finite, *prior);
104 return;
105 }
106 auto* families = std::get_if<ParametricSolutions>(&solved);
107 if ((finite || families) && (prior || std::holds_alternative<ParametricSolutions>(output))) {
108 append_parametric_union(finite, families, prior, output);
109 return;
110 }
111 output = conditional_preimage(original, var);
112}

114Result<bool> zero_difference(const ExprPtr& left, const ExprPtr& right, ComputationContext& context) {
115 auto difference = detail::simplify_expression(SymbolicExpr::add(left,
116 SymbolicExpr::multiply(SymbolicExpr::number(-1), right)), context);
117 if (!difference) { return Result<bool>::failure(difference.error()); }
118 auto proof = check_zero_residual(difference.value(), context);
119 if (!proof) { return Result<bool>::failure(proof.error()); }
120 return std::holds_alternative<ProvedZeroResidual>(proof.value());
121}

123bool compatible_periodic_conditions(const ParametricSolution& left, const ParametricSolution& right) {
124 if (right.integer_parameters.size() != 1 || left.conditions.size() != right.conditions.size()) {
125 return false;
126 }
127 for (std::size_t c = 0; c < left.conditions.size(); ++c) {
128 if (expression_depends_on_variable(detail::node(left.conditions[c]),
129 left.integer_parameters.front()) ||
130 expression_depends_on_variable(detail::node(right.conditions[c]),
131 right.integer_parameters.front()) ||
132 !detail::node(left.conditions[c])->equals(*detail::node(right.conditions[c]))) {
133 return false;
134 }
135 }
136 return true;
137}

139Result<std::optional<ExprPtr>> merged_half_period_value(const detail::AffineForm& first,
140 const detail::AffineForm& second, const std::string& parameter, ComputationContext& context) {
141 using MergeResult = Result<std::optional<ExprPtr>>;
142 auto half = detail::simplify_expression(SymbolicExpr::multiply(
143 SymbolicExpr::number(Rational(1, 2)), first.slope), context);
144 if (!half) { return MergeResult::failure(half.error()); }
145 const auto& half_period = half.value();
146 auto next_offset = SymbolicExpr::add(first.offset, half_period);
147 auto prior_offset = SymbolicExpr::add(second.offset, half_period);
148 ExprPtr offset;
149 auto forward = zero_difference(next_offset, second.offset, context);
150 if (!forward) { return MergeResult::failure(forward.error()); }
151 auto backward = zero_difference(prior_offset, first.offset, context);
152 if (!backward) { return MergeResult::failure(backward.error()); }
153 if (forward.value()) { offset = second.offset; }
154 else if (backward.value()) { offset = first.offset; }
155 else { return std::optional<ExprPtr>{}; }
156 auto merged = detail::simplify_expression(SymbolicExpr::add(offset,
157 SymbolicExpr::multiply(half_period, SymbolicExpr::variable(parameter))), context);
158 if (!merged) { return MergeResult::failure(merged.error()); }
159 return std::optional<ExprPtr>{std::move(merged.value())};
160}

162Result<bool> coalesce_periodic_pair(ParametricSolution& left, const ParametricSolution& right,
163 ComputationContext& context) {
164 if (!compatible_periodic_conditions(left, right)) { return false; }
165 auto first = detail::recognize_affine(*left.value, left.integer_parameters.front(), context);
166 if (!first) { return Result<bool>::failure(first.error()); }
167 auto second = detail::recognize_affine(*right.value, right.integer_parameters.front(), context);
168 if (!second) { return Result<bool>::failure(second.error()); }
169 if (!first.value() || !second.value()) { return false; }
170 for (auto* coefficient : {&first.value()->slope, &first.value()->offset,
171 &second.value()->slope, &second.value()->offset}) {
172 auto normalized = detail::simplify_expression(*coefficient, context);
173 if (!normalized) { return Result<bool>::failure(normalized.error()); }
174 *coefficient = std::move(normalized.value());
175 }
176 auto equal_slopes = zero_difference(first.value()->slope, second.value()->slope, context);
177 if (!equal_slopes) { return Result<bool>::failure(equal_slopes.error()); }
178 if (!equal_slopes.value()) { return false; }
179 auto merged = merged_half_period_value(
180 *first.value(), *second.value(), left.integer_parameters.front(), context);
181 if (!merged) { return Result<bool>::failure(merged.error()); }
182 if (!merged.value()) { return false; }
183 left.value = std::move(*merged.value());
184 return true;
185}

187Result<void> coalesce_periodic_families(SolutionSet& solutions, ComputationContext& context) {
188 auto* parametric = std::get_if<ParametricSolutions>(&solutions);
189 if (!parametric) { return Result<void>::success(); }
190 auto& families = parametric->values;
191 for (std::size_t i = 0; i < families.size(); ++i) {
192 if (families[i].integer_parameters.size() != 1) { continue; }
193 for (std::size_t j = i + 1; j < families.size();) {
194 auto merged = coalesce_periodic_pair(families[i], families[j], context);
195 if (!merged) { return Result<void>::failure(merged.error()); }
196 if (!merged.value()) { ++j; continue; }
197 families.erase(families.begin() + j);
198 j = i + 1;
199 }
200 }
201 return Result<void>::success();
202}

220 ComputationContext& context, const SolveOptions& options, int depth) {
221 auto polynomial = symbolic_to_poly<SymbolicPolyCoeff>(

223 if (!polynomial) {

236 if (!value.conditions.empty()) return conditional_preimage(original, var);

239 if (!inverted) return inverted;

275SolveResult solve_transcendental_impl(
276 const ExprPtr& original, const ExprPtr& expr, const std::string& var,
277 ComputationContext& context, const SolveOptions& options, int depth) {
278 auto step = context.consume_steps(1, "solve.transcendental");
279 if (!step) { return SolveResult::failure(step.error()); }
280 if (depth > maximum_transcendental_depth) { return inconclusive("Transcendental inversion depth exceeded"); }
281 if (!expression_depends_on_variable(detail::node(expr), var)) {
282 return solve_transcendental_constant(original, expr, var, context);
283 }
284 if (auto pattern = decompose_lambert_w_pattern(expr, var)) {
285 return solve_inverted_values(original, pattern->inner, invert_lambert_w(pattern->rhs),
286 var, context, options, depth);
287 }
288 if (auto pattern = decompose_exp_base_pattern(expr, var)) {
289 return solve_inverted_values(original, pattern->inner,
290 invert_exp_base(pattern->base, pattern->rhs), var, context, options, depth);
291 }
292 if (auto pattern = decompose_trig_exp_pattern(expr, var)) {
293 return solve_function_pattern(*pattern, original, var, context, options, depth);
294 }
295 auto substitution = detect_substitution(expr, var);
296 if (!substitution) {
297 return SolveResult::failure(substitution.error());
298 }
299 if (substitution.value()) {
300 return solve_substitution(original, *substitution.value(), var, context, options, depth);
301 }
302 return inconclusive("No complete transcendental inversion matched");
303}

307SolveResult solve_transcendental(
308 const ExprPtr& expr, const std::string& var,
309 ComputationContext& context, const SolveOptions& options) {
310 if (!expr || !detail::node(expr) || var.empty()) {
311 return SolveResult::failure(CasErrc::InvalidArgument,
312 "Expression and variable must not be empty", "solve.transcendental");
313 }
314 auto equation = expr;
315 if (auto relation = std::dynamic_pointer_cast<const RelationalNode>(detail::node(expr))) {
316 if (relation->op() != RelationalNode::Op::EQ) {
317 return SolveResult::failure(CasErrc::InvalidArgument,
318 "Expected an equality", "solve.transcendental");
319 }
320 equation = SymbolicExpr::add(detail::make_expression_ptr(relation->left()),
321 SymbolicExpr::multiply(SymbolicExpr::number(-1),
322 detail::make_expression_ptr(relation->right())));
323 }
324 equation = equation->simplify();
325 auto solved = std::dynamic_pointer_cast<const MultiplyNode>(detail::node(equation)) ?
326 solve_equation(equation, var, context, options) :
327 solve_transcendental_impl(expr, equation, var, context, options, 0);
328 if (!solved) return solved;
329 auto coalesced = coalesce_periodic_families(solved.value(), context);
330 if (!coalesced) return SolveResult::failure(coalesced.error());
331 return detail::finalize_solution_set(expr, var, std::move(solved).value(), context, Domain::Real, options);
332}
