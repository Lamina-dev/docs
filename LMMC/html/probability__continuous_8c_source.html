40 -3.066479806614716e+01, 2.506628277459239e+00

43 -5.447609879822406e+01, 1.615858368580409e+02,

62 /* Lower tail */
63 q = sqrt(-2.0 * log(p));
64 return (((((c[0]*q+c[1])*q+c[2])*q+c[3])*q+c[4])*q+c[5]) /

90lmmc_status_t lmmc_dist_t_pdf(lmmc_real_t x, lmmc_real_t df, lmmc_real_t* out) {
91 lmmc_real_t coeff, base;
92 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
93 if (df <= 0.0) return LMMC_STATUS_INVALID_ARGUMENT;
94
95 coeff = exp(lgamma((df + 1.0) / 2.0) - lgamma(df / 2.0)) /
96 sqrt(df * LMMC_PI);
97 base = 1.0 + x * x / df;
98 *out = coeff * pow(base, -(df + 1.0) / 2.0);
99 return LMMC_STATUS_OK;

118lmmc_status_t lmmc_dist_t_quantile(lmmc_real_t p, lmmc_real_t df, lmmc_real_t* out) {
119 /* Newton's method using t CDF and PDF */
120 lmmc_real_t x, cdf_val, pdf_val;
121 lmmc_status_t status;
122 int iter;
123 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
124 if (df <= 0.0) return LMMC_STATUS_INVALID_ARGUMENT;
125 if (p <= 0.0 || p >= 1.0) return LMMC_STATUS_INVALID_ARGUMENT;
126
127 /* Initial guess from normal quantile */
128 x = normal_quantile_rational(p);
129
130 for (iter = 0; iter < 100; ++iter) {
131 lmmc_real_t dx;
132 status = lmmc_dist_t_cdf(x, df, &cdf_val);
133 if (status != LMMC_STATUS_OK) return status;
134 status = lmmc_dist_t_pdf(x, df, &pdf_val);
135 if (status != LMMC_STATUS_OK) return status;
136 if (pdf_val < 1e-300) break;
137 dx = (cdf_val - p) / pdf_val;
138 x -= dx;
139 if (fabs(dx) < 1e-12 * (1.0 + fabs(x))) break;
140 }
141
142 *out = x;
143 return LMMC_STATUS_OK;

147 lmmc_real_t k2;
148 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
149 if (df <= 0.0) return LMMC_STATUS_INVALID_ARGUMENT;

171lmmc_status_t lmmc_dist_chi2_quantile(lmmc_real_t p, lmmc_real_t df, lmmc_real_t* out) {
172 /* Newton's method on chi2 CDF */
173 lmmc_real_t x, cdf_val, pdf_val, dx;
174 lmmc_status_t status;
175 int iter;
176 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
177 if (df <= 0.0) return LMMC_STATUS_INVALID_ARGUMENT;
178 if (p <= 0.0 || p >= 1.0) return LMMC_STATUS_INVALID_ARGUMENT;
179
180 /* Initial guess: Wilson-Hilferty approximation */
181 {
182 lmmc_real_t z = normal_quantile_rational(p);
183 lmmc_real_t tmp = 1.0 - 2.0 / (9.0 * df) + z * sqrt(2.0 / (9.0 * df));
184 x = df * tmp * tmp * tmp;
185 if (x <= 0.0) x = 0.01;
186 }
187
188 for (iter = 0; iter < 100; ++iter) {
189 status = lmmc_dist_chi2_cdf(x, df, &cdf_val);
190 if (status != LMMC_STATUS_OK) return status;
191 status = lmmc_dist_chi2_pdf(x, df, &pdf_val);
192 if (status != LMMC_STATUS_OK) return status;
193 if (pdf_val < 1e-300) break;
194 dx = (cdf_val - p) / pdf_val;
195 x -= dx;
196 if (x <= 0.0) x = 1e-10;
197 if (fabs(dx) < 1e-12 * (1.0 + fabs(x))) break;
198 }
199
200 *out = x;
201 return LMMC_STATUS_OK;

204lmmc_status_t lmmc_dist_f_pdf(lmmc_real_t x, lmmc_real_t df1, lmmc_real_t df2,
205 lmmc_real_t* out) {
206 lmmc_real_t num, den;
207 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
208 if (df1 <= 0.0 || df2 <= 0.0) return LMMC_STATUS_INVALID_ARGUMENT;
209 if (x < 0.0) { *out = 0.0; return LMMC_STATUS_OK; }
210 if (x == 0.0) {
211 if (df1 < 2.0) { *out = 1.0 / 0.0; return LMMC_STATUS_OK; }
212 if (df1 == 2.0) { *out = 1.0; return LMMC_STATUS_OK; }
213 *out = 0.0; return LMMC_STATUS_OK;
214 }
215
216 num = (df1 / 2.0) * log(df1 / df2) + (df1 / 2.0 - 1.0) * log(x);
217 den = ((df1 + df2) / 2.0) * log(1.0 + df1 * x / df2);
218 *out = exp(num - den - lgamma(df1 / 2.0) - lgamma(df2 / 2.0) +
219 lgamma((df1 + df2) / 2.0));
220 return LMMC_STATUS_OK;
221}
