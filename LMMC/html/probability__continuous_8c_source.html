48lmmc_status_t lmmc_dist_normal_pdf(lmmc_real_t x, lmmc_real_t mu,
49 lmmc_real_t sigma, lmmc_real_t* out) {

53 return LMMC_STATUS_INVALID_ARGUMENT;

76static lmmc_real_t normal_quantile_rational(lmmc_real_t p) {
77 /* Coefficients for rational approximation */
78 static const double a[] = {
79 -3.969683028665376e+01, 2.209460984245205e+02,
80 -2.759285104469687e+02, 1.383577518672690e+02,
81 -3.066479806614716e+01, 2.506628277459239e+00
82 };
83 static const double b[] = {
84 -5.447609879822406e+01, 1.615858368580409e+02,
85 -1.556989798598866e+02, 6.680131188771972e+01,
86 -1.328068155288572e+01
87 };
88 static const double c[] = {
89 -7.784894002430293e-03, -3.223964580411365e-01,
90 -2.400758277161838e+00, -2.549732539343734e+00,
91 4.374664141464968e+00, 2.938163982698783e+00
92 };
93 static const double d[] = {
94 7.784695709041462e-03, 3.224671290700398e-01,
95 2.445134137142996e+00, 3.754408661907416e+00
96 };
97
98 double q, r;
99 double plow = 0.02425;
100 double phigh = 1.0 - plow;
101
102 if (p < plow) {
103 /* Lower tail */
104 q = sqrt(-2.0 * log(p));
105 return (((((c[0]*q+c[1])*q+c[2])*q+c[3])*q+c[4])*q+c[5]) /
106 ((((d[0]*q+d[1])*q+d[2])*q+d[3])*q+1.0);
107 } else if (p <= phigh) {
108 /* Central region */
109 q = p - 0.5;
110 r = q * q;
111 return (((((a[0]*r+a[1])*r+a[2])*r+a[3])*r+a[4])*r+a[5])*q /
112 (((((b[0]*r+b[1])*r+b[2])*r+b[3])*r+b[4])*r+1.0);
113 } else {
114 /* Upper tail */
115 q = sqrt(-2.0 * log(1.0 - p));
116 return -(((((c[0]*q+c[1])*q+c[2])*q+c[3])*q+c[4])*q+c[5]) /
117 ((((d[0]*q+d[1])*q+d[2])*q+d[3])*q+1.0);
118 }
119}

134static lmmc_real_t student_t_log_kernel(
135 lmmc_real_t x, lmmc_real_t df) {
136 const lmmc_real_t abs_x = fabs(x);
137 const lmmc_real_t sqrt_df = sqrt(df);
138 if (abs_x <= sqrt_df) {
139 const lmmc_real_t ratio = abs_x / sqrt_df;
140 return log1p(ratio * ratio);
141 }
142 {
143 const lmmc_real_t ratio = sqrt_df / abs_x;
144 return 2.0 * log(abs_x) - log(df) +
145 log1p(ratio * ratio);
146 }
147}

149static lmmc_real_t student_t_beta_argument(
150 lmmc_real_t x, lmmc_real_t df) {
151 const lmmc_real_t abs_x = fabs(x);
152 const lmmc_real_t sqrt_df = sqrt(df);
153 if (abs_x <= sqrt_df) {
154 const lmmc_real_t ratio = abs_x / sqrt_df;
155 const lmmc_real_t ratio_sq = ratio * ratio;
156 return 1.0 / (1.0 + ratio_sq);
157 }
158 {
159 const lmmc_real_t ratio = sqrt_df / abs_x;
160 const lmmc_real_t ratio_sq = ratio * ratio;
161 return ratio_sq / (1.0 + ratio_sq);
162 }
163}

165lmmc_status_t lmmc_dist_t_pdf(lmmc_real_t x, lmmc_real_t df, lmmc_real_t* out) {
166 lmmc_real_t log_density;
167 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
168 if (!isfinite(x) || !valid_positive(df)) return LMMC_STATUS_INVALID_ARGUMENT;
169 if (student_t_uses_normal_limit(df)) {
170 return store_density_from_log(
171 -0.5 * x * x - LMMC_LOG_SQRT_2PI, out);
172 }
173
174 log_density = lgamma((df + 1.0) / 2.0) - lgamma(df / 2.0) -
175 0.5 * (log(df) + log(LMMC_PI)) -
176 ((df + 1.0) / 2.0) * student_t_log_kernel(x, df);
177 return store_density_from_log(log_density, out);
178}

180lmmc_status_t lmmc_dist_t_cdf(lmmc_real_t x, lmmc_real_t df, lmmc_real_t* out) {
181 lmmc_real_t t_val, ib, result;
182 lmmc_status_t status;
183 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
184 if (!isfinite(x) || !valid_positive(df)) return LMMC_STATUS_INVALID_ARGUMENT;
185 if (student_t_uses_normal_limit(df)) {
186 return store_probability_result(
187 0.5 * erfc(-x / sqrt(2.0)), out);
188 }
189
190 t_val = student_t_beta_argument(x, df);
191 if (t_val > 0.0) {
192 status = regularized_beta(t_val, df / 2.0, 0.5, &ib);
193 if (status != LMMC_STATUS_OK) return status;
194 } else {
195 const lmmc_real_t a = df / 2.0;
196 const lmmc_real_t log_t =
197 log(df) - 2.0 * log(fabs(x));
198 const lmmc_real_t log_beta =
199 lgamma(a) + lgamma(0.5) - lgamma(a + 0.5);
200 ib = exp(a * log_t - log(a) - log_beta);
201 }
202 result = x >= 0.0 ? 1.0 - 0.5 * ib : 0.5 * ib;
203 return store_probability_result(result, out);
204}

307 lmmc_real_t probability,
308 lmmc_real_t lower,
309 lmmc_real_t lower_cdf,

311 lmmc_real_t upper_cdf,
312 lmmc_real_t* out) {

324static lmmc_status_t bracketed_quantile(
325 lmmc_real_t probability,
326 lmmc_real_t initial,
327 lmmc_real_t lower,
328 lmmc_real_t upper,
329 int expand_lower,
330 int expand_upper,
331 distribution_eval_fn cdf,
332 distribution_eval_fn pdf,
333 const void* context,
334 lmmc_real_t* out) {
335 lmmc_real_t lower_cdf, upper_cdf, x;
336 lmmc_status_t status;
337 int iteration;
338
339 status = evaluate_distribution(cdf, lower, context, &lower_cdf);
340 if (status != LMMC_STATUS_OK) return status;
341 status = evaluate_distribution(cdf, upper, context, &upper_cdf);
342 if (status != LMMC_STATUS_OK) return status;
343
344 for (iteration = 0; lower_cdf > probability && expand_lower;
345 ++iteration) {
346 if (iteration >= 1024 || lower <= -DBL_MAX) {
347 return LMMC_STATUS_OUT_OF_RANGE;
348 }
349 lower = lower <= -DBL_MAX / 2.0 ? -DBL_MAX : lower * 2.0;
350 status = evaluate_distribution(cdf, lower, context, &lower_cdf);
351 if (status != LMMC_STATUS_OK) return status;
352 }
353 for (iteration = 0; upper_cdf < probability && expand_upper;
354 ++iteration) {
355 if (iteration >= 1024 || upper >= DBL_MAX) {
356 return LMMC_STATUS_OUT_OF_RANGE;
357 }
358 upper = upper >= DBL_MAX / 2.0 ? DBL_MAX : upper * 2.0;
359 status = evaluate_distribution(cdf, upper, context, &upper_cdf);
360 if (status != LMMC_STATUS_OK) return status;
361 }
362 if (lower_cdf > probability || upper_cdf < probability) {
363 return LMMC_STATUS_OUT_OF_RANGE;
364 }
365
366 x = initial;
367 if (!isfinite(x) || x <= lower || x >= upper) {
368 x = lower * 0.5 + upper * 0.5;
369 }
370 for (iteration = 0; iteration < 2048; ++iteration) {
371 lmmc_real_t cdf_value, pdf_value, candidate;
372 status = evaluate_distribution(cdf, x, context, &cdf_value);
373 if (status != LMMC_STATUS_OK) return status;
374 if (quantile_probability_close(cdf_value, probability)) {
375 *out = x;
376 return LMMC_STATUS_OK;
377 }
378
379 if (cdf_value < probability) {
380 lower = x;
381 lower_cdf = cdf_value;
382 } else {
383 upper = x;
384 upper_cdf = cdf_value;
385 }
386 if (nextafter(lower, upper) == upper) {
387 return finish_quantile_bracket(
388 probability, lower, lower_cdf, upper, upper_cdf, out);
389 }
390
391 status = pdf(x, context, &pdf_value);
392 candidate = NAN;
393 if (status == LMMC_STATUS_OK && isfinite(pdf_value) &&
394 pdf_value > DBL_MIN) {
395 candidate = x - (cdf_value - probability) / pdf_value;
396 }
397 if (!isfinite(candidate) || candidate <= lower ||
398 candidate >= upper) {
399 candidate = lower * 0.5 + upper * 0.5;
400 }
401 if (candidate == lower || candidate == upper) {
402 return finish_quantile_bracket(
403 probability, lower, lower_cdf, upper, upper_cdf, out);
404 }
405 x = candidate;
406 }
407 return LMMC_STATUS_CONVERGENCE_FAILED;
408}

430lmmc_status_t lmmc_dist_chi2_pdf(lmmc_real_t x, lmmc_real_t df, lmmc_real_t* out) {
431 lmmc_real_t k2;
432 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
433 if (!isfinite(x) || !valid_positive(df)) return LMMC_STATUS_INVALID_ARGUMENT;
434 if (x < 0.0) { *out = 0.0; return LMMC_STATUS_OK; }
435 if (x == 0.0) {

460 distribution_parameters_t parameters;
461 lmmc_real_t z, term, initial;
462 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
463 if (!valid_positive(df) || !valid_probability(p))
464 return LMMC_STATUS_INVALID_ARGUMENT;
465
466 z = normal_quantile_rational(p);
467 term = 1.0 - 2.0 / (9.0 * df) + z * sqrt(2.0 / (9.0 * df));
468 initial = df * term * term * term;
469 if (!isfinite(initial) || initial <= 0.0) initial = 1.0;
470 parameters.first = df;
471 parameters.second = 0.0;
472 return bracketed_quantile(
473 p, initial, 0.0, fmax(initial, 1.0), 0, 1,
474 eval_chi2_cdf, eval_chi2_pdf, &parameters, out);

477lmmc_status_t lmmc_dist_f_pdf(lmmc_real_t x, lmmc_real_t df1, lmmc_real_t df2,
478 lmmc_real_t* out) {
479 lmmc_real_t first_shape, second_shape, total_shape;
480 lmmc_real_t num, den;
481 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
482 if (!isfinite(x) || !valid_positive(df1) || !valid_positive(df2))
483 return LMMC_STATUS_INVALID_ARGUMENT;
484 if (x < 0.0) { *out = 0.0; return LMMC_STATUS_OK; }
485 if (x == 0.0) {
486 if (df1 < 2.0) return LMMC_STATUS_NUMERICAL_FAILURE;
487 if (df1 == 2.0) { *out = 1.0; return LMMC_STATUS_OK; }
488 *out = 0.0; return LMMC_STATUS_OK;
489 }
490
491 first_shape = df1 / 2.0;
492 second_shape = df2 / 2.0;
493 total_shape = first_shape + second_shape;
494 if (first_shape >= 16.0 && second_shape >= 16.0 &&
495 isfinite(total_shape)) {
496 const lmmc_real_t scale = df2 / df1;
497 if (isfinite(scale) && scale > 0.0) {
498 lmmc_real_t beta_x;
499 lmmc_real_t beta_complement;
500 if (x > scale) {
501 const lmmc_real_t ratio = scale / x;
502 beta_x = 1.0 / (1.0 + ratio);
503 beta_complement = ratio / (1.0 + ratio);
504 } else {
505 const lmmc_real_t ratio = (df1 / df2) * x;
506 beta_x = ratio / (1.0 + ratio);
507 beta_complement = 1.0 / (1.0 + ratio);
508 }
509 if (beta_x > 0.0 && beta_complement > 0.0) {
510 const lmmc_real_t center = first_shape / total_shape;
511 const lmmc_real_t complement_center =
512 second_shape / total_shape;
513 const lmmc_real_t correction =
514 lmmc_log_gamma_stirling_error(total_shape) -
515 lmmc_log_gamma_stirling_error(first_shape) -
516 lmmc_log_gamma_stirling_error(second_shape);
517 const lmmc_real_t deviance =
518 lmmc_deviance_part(
519 first_shape, total_shape * beta_x) +
520 lmmc_deviance_part(
521 second_shape, total_shape * beta_complement);
522 const lmmc_real_t log_density =
523 correction - deviance +
524 0.5 * (log(total_shape) + log(center) +
525 log(complement_center) -
526 2.0 * LMMC_LOG_SQRT_2PI) -
527 log(x);
528 return store_density_from_log(log_density, out);
529 }
530 }
531 }
532
533 num = first_shape * log(df1 / df2) +
534 (first_shape - 1.0) * log(x);
535 den = total_shape * log1p(df1 * x / df2);
536 return store_density_from_log(
537 num - den - lgamma(first_shape) - lgamma(second_shape) +
538 lgamma(total_shape),
539 out);
540}

542lmmc_status_t lmmc_dist_f_cdf(lmmc_real_t x, lmmc_real_t df1, lmmc_real_t df2,
543 lmmc_real_t* out) {
544 lmmc_real_t scale, result, beta_value;
545 lmmc_status_t status;
546 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
547 if (!isfinite(x) || !valid_positive(df1) || !valid_positive(df2))
548 return LMMC_STATUS_INVALID_ARGUMENT;
549 if (x <= 0.0) { *out = 0.0; return LMMC_STATUS_OK; }
550
551 scale = df2 / df1;
552 if (x > scale) {
553 lmmc_real_t ratio = scale / x;
554 lmmc_real_t tail_x = ratio / (1.0 + ratio);
555 status = regularized_beta(
556 tail_x, df2 / 2.0, df1 / 2.0, &beta_value);
557 if (status != LMMC_STATUS_OK) return status;
558 result = 1.0 - beta_value;
559 } else {
560 lmmc_real_t ratio = (df1 / df2) * x;
561 lmmc_real_t beta_x = ratio / (1.0 + ratio);
562 status = regularized_beta(
563 beta_x, df1 / 2.0, df2 / 2.0, &result);
564 if (status != LMMC_STATUS_OK) return status;
565 }
566 return store_probability_result(result, out);

584 lmmc_real_t scale, lmmc_real_t* out) {
585 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
586 if (!isfinite(x) || !valid_positive(shape) || !valid_positive(scale))
587 return LMMC_STATUS_INVALID_ARGUMENT;
588 if (x < 0.0) { *out = 0.0; return LMMC_STATUS_OK; }
589 if (x == 0.0) {
590 if (shape < 1.0) return LMMC_STATUS_NUMERICAL_FAILURE;
591 if (shape == 1.0)
592 return store_density_from_log(-log(scale), out);
593 *out = 0.0; return LMMC_STATUS_OK;
594 }
595
596 if (shape >= 16.0) {
597 const lmmc_real_t standardized = x / scale;
598 if (isfinite(standardized) && standardized > 0.0) {
599 const lmmc_real_t log_density =
600 -lmmc_deviance_part(shape, standardized) -
601 lmmc_log_gamma_stirling_error(shape) -
602 log(standardized) + 0.5 * log(shape) -
603 LMMC_LOG_SQRT_2PI - log(scale);
604 return store_density_from_log(log_density, out);
605 }
606 }
607
608 return store_density_from_log(
609 (shape - 1.0) * log(x) - x / scale -
610 shape * log(scale) - lgamma(shape),
611 out);
612}

648lmmc_status_t lmmc_dist_beta_pdf(lmmc_real_t x, lmmc_real_t alpha,
649 lmmc_real_t beta_param, lmmc_real_t* out) {
650 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
651 if (!isfinite(x) || !valid_positive(alpha) || !valid_positive(beta_param))
652 return LMMC_STATUS_INVALID_ARGUMENT;
653 if (x < 0.0 || x > 1.0) { *out = 0.0; return LMMC_STATUS_OK; }
654 if (x == 0.0) {
655 if (alpha < 1.0) return LMMC_STATUS_NUMERICAL_FAILURE;
656 if (alpha == 1.0) { *out = beta_param; return LMMC_STATUS_OK; }
657 *out = 0.0; return LMMC_STATUS_OK;
658 }
659 if (x == 1.0) {
660 if (beta_param < 1.0) return LMMC_STATUS_NUMERICAL_FAILURE;
661 if (beta_param == 1.0) { *out = alpha; return LMMC_STATUS_OK; }
662 *out = 0.0; return LMMC_STATUS_OK;
663 }
664
665 if (x == 0.5 && alpha == beta_param &&
666 !isfinite(alpha + beta_param)) {
667 const lmmc_real_t log_density =
668 log(2.0) + 0.5 * (log(alpha) - log(LMMC_PI)) -
669 1.0 / (8.0 * alpha);
670 return store_density_from_log(log_density, out);
671 }
672
673 if (alpha >= 16.0 && beta_param >= 16.0 &&
674 isfinite(alpha + beta_param)) {
675 const lmmc_real_t total = alpha + beta_param;
676 const lmmc_real_t center = alpha / total;
677 const lmmc_real_t complement_center = beta_param / total;
678 const lmmc_real_t complement_x = 1.0 - x;
679 const lmmc_real_t correction =
680 lmmc_log_gamma_stirling_error(total) -
681 lmmc_log_gamma_stirling_error(alpha) -
682 lmmc_log_gamma_stirling_error(beta_param);
683 const lmmc_real_t deviance =
684 lmmc_deviance_part(alpha, total * x) +
685 lmmc_deviance_part(beta_param, total * complement_x);
686 const lmmc_real_t log_density =
687 correction - deviance +
688 0.5 * (log(total) + log(center) +
689 log(complement_center) -
690 2.0 * LMMC_LOG_SQRT_2PI) -
691 log(x) - log1p(-x);
692 return store_density_from_log(log_density, out);
693 }
694
695 return store_density_from_log(
696 lgamma(alpha + beta_param) - lgamma(alpha) - lgamma(beta_param) +
697 (alpha - 1.0) * log(x) +
698 (beta_param - 1.0) * log1p(-x),
699 out);
700}
