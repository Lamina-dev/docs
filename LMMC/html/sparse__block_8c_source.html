12lmmc_status_t lmmc_sparse_bsr_create(size_t rows, size_t cols, size_t block_size,
13 size_t nnz_blocks, lmmc_sparse_bsr_t* out) {
14 size_t row_ptr_bytes = 0;
15 size_t col_idx_bytes = 0;
16 size_t val_count = 0;
17 size_t val_bytes = 0;
18
19 if (out == NULL || rows == 0 || cols == 0 || block_size == 0) {
20 return LMMC_STATUS_INVALID_ARGUMENT;
21 }
22
23 /* 计算所需内存大小，检查溢出 */
24 if (!lmmc_safe_mul_size(rows + 1, sizeof(size_t), &row_ptr_bytes)) {
25 return LMMC_STATUS_INVALID_ARGUMENT;
26 }
27 if (!lmmc_safe_mul_size(nnz_blocks, sizeof(size_t), &col_idx_bytes)) {
28 return LMMC_STATUS_INVALID_ARGUMENT;
29 }
30 if (!lmmc_safe_mul_size(block_size, block_size, &val_count)) {
31 return LMMC_STATUS_INVALID_ARGUMENT;
32 }
33 if (!lmmc_safe_mul_size(val_count, nnz_blocks, &val_count)) {
34 return LMMC_STATUS_INVALID_ARGUMENT;
35 }
36 if (!lmmc_safe_mul_size(val_count, sizeof(lmmc_real_t), &val_bytes)) {
37 return LMMC_STATUS_INVALID_ARGUMENT;
38 }
39
40 out->row_ptr = (size_t*)lmmc_alloc(row_ptr_bytes);
41 if (out->row_ptr == NULL) {
42 return LMMC_STATUS_ALLOCATION_FAILED;
43 }
44 memset(out->row_ptr, 0, row_ptr_bytes);
45
46 out->col_idx = NULL;
47 out->values = NULL;
48
49 if (nnz_blocks > 0) {
50 out->col_idx = (size_t*)lmmc_alloc(col_idx_bytes);
51 if (out->col_idx == NULL) {
52 lmmc_free(out->row_ptr);
53 out->row_ptr = NULL;
54 return LMMC_STATUS_ALLOCATION_FAILED;
55 }
56 memset(out->col_idx, 0, col_idx_bytes);
57
58 out->values = (lmmc_real_t*)lmmc_alloc(val_bytes);
59 if (out->values == NULL) {
60 lmmc_free(out->col_idx);
61 lmmc_free(out->row_ptr);
62 out->col_idx = NULL;
63 out->row_ptr = NULL;
64 return LMMC_STATUS_ALLOCATION_FAILED;
65 }
66 memset(out->values, 0, val_bytes);
67 }
68
69 out->rows = rows;
70 out->cols = cols;
71 out->block_size = block_size;
72 out->nnz_blocks = nnz_blocks;
73 out->owns_data = 1;
74 return LMMC_STATUS_OK;
75}

77lmmc_status_t lmmc_sparse_bsr_to_dense(const lmmc_sparse_bsr_t* bsr, lmmc_mat_t* out) {
78 size_t bi, p, li, lj;
79 size_t dense_rows, dense_cols;
80 lmmc_real_t zero;
81 lmmc_status_t st;
82
83 if (bsr == NULL || out == NULL || out->data == NULL) {
84 return LMMC_STATUS_INVALID_ARGUMENT;
85 }
86 if (bsr->rows == 0 || bsr->cols == 0 || bsr->block_size == 0) {
87 return LMMC_STATUS_INVALID_ARGUMENT;
88 }
89 if (bsr->row_ptr == NULL) {
90 return LMMC_STATUS_INVALID_ARGUMENT;
91 }
92 if (bsr->nnz_blocks > 0 && (bsr->col_idx == NULL || bsr->values == NULL)) {
93 return LMMC_STATUS_INVALID_ARGUMENT;
94 }
95
96 dense_rows = bsr->rows * bsr->block_size;
97 dense_cols = bsr->cols * bsr->block_size;
98
99 if (out->rows != dense_rows || out->cols != dense_cols) {
100 return LMMC_STATUS_DIMENSION_MISMATCH;
101 }
102
103 /* 先清零输出矩阵 */
104 LMMC_REAL_INIT(&zero);
105 LMMC_REAL_SET_D(&zero, 0.0);
106 st = lmmc_mat_fill(out, zero);
107 LMMC_REAL_CLEAR(&zero);
108 if (st != LMMC_STATUS_OK) {
109 return st;
110 }
111
112 /* 遍历每个块行 */
113 for (bi = 0; bi < bsr->rows; ++bi) {
114 for (p = bsr->row_ptr[bi]; p < bsr->row_ptr[bi + 1]; ++p) {
115 size_t bj = bsr->col_idx[p];
116 size_t block_offset = p * bsr->block_size * bsr->block_size;
117 size_t row_base = bi * bsr->block_size;
118 size_t col_base = bj * bsr->block_size;
119
120 /* 将块内元素写入稠密矩阵 */
121 for (li = 0; li < bsr->block_size; ++li) {
122 for (lj = 0; lj < bsr->block_size; ++lj) {
123 size_t dense_idx = (row_base + li) * out->stride + (col_base + lj);
124 size_t block_idx = block_offset + li * bsr->block_size + lj;
125 LMMC_REAL_SET(&out->data[dense_idx], &bsr->values[block_idx]);
126 }
127 }
128 }
129 }
130
131 return LMMC_STATUS_OK;
132}

134lmmc_status_t lmmc_sparse_dense_to_bsr(const lmmc_mat_t* dense, size_t block_size,
135 lmmc_real_t eps, lmmc_sparse_bsr_t* out) {
136 size_t block_rows, block_cols;
137 size_t bi, bj, li, lj;
138 size_t nnz_blocks = 0;
139 size_t nz_idx = 0;
140 lmmc_status_t st;
141 lmmc_real_t abs_v;
142 lmmc_real_t zero;
143
144 if (dense == NULL || out == NULL || dense->data == NULL) {
145 return LMMC_STATUS_INVALID_ARGUMENT;
146 }
147 if (block_size == 0) {
148 return LMMC_STATUS_INVALID_ARGUMENT;
149 }
150 if (dense->rows % block_size != 0 || dense->cols % block_size != 0) {
151 return LMMC_STATUS_INVALID_ARGUMENT;
152 }
153
154 LMMC_REAL_INIT(&abs_v);
155 LMMC_REAL_INIT(&zero);
156 LMMC_REAL_SET_D(&zero, 0.0);
157
158 if (LMMC_REAL_CMP(&eps, &zero) < 0) {
159 LMMC_REAL_CLEAR(&abs_v);
160 LMMC_REAL_CLEAR(&zero);
161 return LMMC_STATUS_INVALID_ARGUMENT;
162 }
163
164 block_rows = dense->rows / block_size;
165 block_cols = dense->cols / block_size;
166
167 /* 第一遍：计算非零块数量 */
168 for (bi = 0; bi < block_rows; ++bi) {
169 for (bj = 0; bj < block_cols; ++bj) {
170 int block_nonzero = 0;
171 for (li = 0; li < block_size && !block_nonzero; ++li) {
172 for (lj = 0; lj < block_size && !block_nonzero; ++lj) {
173 size_t row = bi * block_size + li;
174 size_t col = bj * block_size + lj;
175 LMMC_REAL_ABS(&abs_v, &dense->data[row * dense->stride + col]);
176 if (LMMC_REAL_CMP(&abs_v, &eps) > 0) {
177 block_nonzero = 1;
178 }
179 }
180 }
181 if (block_nonzero) {
182 ++nnz_blocks;
183 }
184 }
185 }
186
187 /* 分配 BSR 结构 */
188 st = lmmc_sparse_bsr_create(block_rows, block_cols, block_size, nnz_blocks, out);
189 if (st != LMMC_STATUS_OK) {
190 LMMC_REAL_CLEAR(&abs_v);
191 LMMC_REAL_CLEAR(&zero);
192 return st;
193 }
194 if (nnz_blocks == 0) {
195 LMMC_REAL_CLEAR(&abs_v);
196 LMMC_REAL_CLEAR(&zero);
197 return LMMC_STATUS_OK;
198 }
199
200 /* 第二遍：填充数据 */
201 nz_idx = 0;
202 out->row_ptr[0] = 0;
203 for (bi = 0; bi < block_rows; ++bi) {
204 for (bj = 0; bj < block_cols; ++bj) {
205 int block_nonzero = 0;
206 for (li = 0; li < block_size && !block_nonzero; ++li) {
207 for (lj = 0; lj < block_size && !block_nonzero; ++lj) {
208 size_t row = bi * block_size + li;
209 size_t col = bj * block_size + lj;
210 LMMC_REAL_ABS(&abs_v, &dense->data[row * dense->stride + col]);
211 if (LMMC_REAL_CMP(&abs_v, &eps) > 0) {
212 block_nonzero = 1;
213 }
214 }
215 }
216 if (block_nonzero) {
217 size_t block_offset = nz_idx * block_size * block_size;
218 out->col_idx[nz_idx] = bj;
219
220 /* 复制块内所有元素 */
221 for (li = 0; li < block_size; ++li) {
222 for (lj = 0; lj < block_size; ++lj) {
223 size_t row = bi * block_size + li;
224 size_t col = bj * block_size + lj;
225 size_t blk_idx = block_offset + li * block_size + lj;
226 LMMC_REAL_SET(&out->values[blk_idx],
227 &dense->data[row * dense->stride + col]);
228 }
229 }
230 ++nz_idx;
231 }
232 }
233 out->row_ptr[bi + 1] = nz_idx;
234 }
235
236 LMMC_REAL_CLEAR(&abs_v);
237 LMMC_REAL_CLEAR(&zero);
238 return LMMC_STATUS_OK;
239}
