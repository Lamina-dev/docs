83lmmc_status_t lmmc_lsr_num_set_make(const lmmc_real_t* values,
84 size_t count,
85 lmmc_lsr_num_set_t* out)
86{
87 lmmc_status_t status;
88 if (!out || (count > 0 && !values)) return LMMC_STATUS_INVALID_ARGUMENT;
89 if (!lmmc_lsr_real_array_is_finite(values, count)) {
90 return LMMC_STATUS_NUMERICAL_FAILURE;
91 }
92 status = lmmc_lsr_num_set_alloc(count, out);
93 if (status != LMMC_STATUS_OK) return status;
94 for (size_t i = 0; i < count; ++i) {
95 status = lmmc_lsr_num_set_append_unique(out, values[i]);
96 if (status != LMMC_STATUS_OK) {
97 lmmc_lsr_num_set_destroy(out);
98 return status;
99 }

154 lmmc_status_t status;
