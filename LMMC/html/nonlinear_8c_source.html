70lmmc_status_t lmmc_nonlinear_x_tolerance(
71 lmmc_real_t x,
72 const lmmc_nonlinear_config_t* cfg,
73 lmmc_real_t* out_x_tol
74) {
75 lmmc_real_t x_scale = 0.0;
76 lmmc_real_t x_tol = 0.0;
77 lmmc_real_t abs_x = 0.0;
78 lmmc_real_t one = 1.0;
79 lmmc_real_t tmp = 0.0;
80
81 if (cfg == NULL || out_x_tol == NULL || !lmmc_is_finite(&x)) {
82 return LMMC_STATUS_INVALID_ARGUMENT;
83 }
84
85 abs_x = lmmc_abs(x);
86 x_scale = lmmc_max(abs_x, one);
87 LMMC_REAL_MUL(&tmp, &cfg->rel_tol, &x_scale);
88 LMMC_REAL_ADD(&x_tol, &cfg->abs_tol, &tmp);
89
90 if (!lmmc_is_finite(&x_tol)) {
91 return LMMC_STATUS_NUMERICAL_FAILURE;
92 }
93
94 *out_x_tol = x_tol;
95 return LMMC_STATUS_OK;
96}
