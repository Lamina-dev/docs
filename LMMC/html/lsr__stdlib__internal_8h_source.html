136static inline int lmmc_lsr_compare_eval(lmmc_real_t lhs,
137 lmmc_lsr_compare_op_t op,
138 lmmc_real_t rhs,
139 uint8_t* out)
140{
141 if (!out) return 0;
142 switch (op) {
143 case LMMC_LSR_COMPARE_EQ:
144 *out = (uint8_t)(lhs == rhs);
145 return 1;
146 case LMMC_LSR_COMPARE_NE:
147 *out = (uint8_t)(lhs != rhs);
148 return 1;
149 case LMMC_LSR_COMPARE_LT:
150 *out = (uint8_t)(lhs < rhs);
151 return 1;
152 case LMMC_LSR_COMPARE_LE:
153 *out = (uint8_t)(lhs <= rhs);
154 return 1;
155 case LMMC_LSR_COMPARE_GT:
156 *out = (uint8_t)(lhs > rhs);
157 return 1;
158 case LMMC_LSR_COMPARE_GE:
159 *out = (uint8_t)(lhs >= rhs);
160 return 1;
161 }
162 return 0;
163}

280 *out = (lmmc_real_t)value;
