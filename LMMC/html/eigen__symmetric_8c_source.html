16static lmmc_status_t householder_tridiag(lmmc_mat_t *a, lmmc_real_t *diag, lmmc_real_t *offdiag, lmmc_real_t *taus) {
17 size_t n = a->rows;
18 size_t k, i, j;
19 if (taus) { for (i = 0; i + 1 < n; i++) taus[i] = 0.0; }
20 if (n == 1) { diag[0] = MAT_ELEM(a, 0, 0); return LMMC_STATUS_OK; }
21 for (k = 0; k < n - 1; k++) {
22 lmmc_real_t sigma = 0.0;
23 for (i = k + 2; i < n; i++) sigma += MAT_ELEM(a, i, k) * MAT_ELEM(a, i, k);
24 lmmc_real_t alpha_val = MAT_ELEM(a, k + 1, k);
25 if (sigma == 0.0) {
26 offdiag[k] = alpha_val; diag[k] = MAT_ELEM(a, k, k);
27 if (taus) taus[k] = 0.0;
28
29 for (i = k + 2; i < n; i++) MAT_ELEM(a, i, k) = 0.0;
30 continue;
31 }
32 lmmc_real_t norm_x = sqrt(alpha_val * alpha_val + sigma);
33 lmmc_real_t beta = (alpha_val >= 0.0) ? -norm_x : norm_x;
34 lmmc_real_t v_first = alpha_val - beta;
35 lmmc_real_t tau = -v_first / beta;
36 lmmc_real_t inv_v_first = 1.0 / v_first;
37 for (i = k + 2; i < n; i++) MAT_ELEM(a, i, k) *= inv_v_first;
38 offdiag[k] = beta;
39 if (taus) taus[k] = tau;
40 lmmc_real_t *w = (lmmc_real_t *)lmmc_alloc((n - k - 1) * sizeof(lmmc_real_t));
41 if (!w) return LMMC_STATUS_ALLOCATION_FAILED;
42 for (i = 0; i < n - k - 1; i++) {
43 lmmc_real_t sum = 0.0;
44 for (j = 0; j < n - k - 1; j++) {
45 lmmc_real_t vj = (j == 0) ? 1.0 : MAT_ELEM(a, k + 1 + j, k);
46 sum += MAT_ELEM(a, k + 1 + i, k + 1 + j) * vj;
47 }
48 w[i] = tau * sum;
49 }
50 lmmc_real_t gamma = 0.0;
51 for (i = 0; i < n - k - 1; i++) {
52 lmmc_real_t vi = (i == 0) ? 1.0 : MAT_ELEM(a, k + 1 + i, k);
53 gamma += w[i] * vi;
54 }
55 gamma *= 0.5 * tau;
56 for (i = 0; i < n - k - 1; i++) {
57 lmmc_real_t vi = (i == 0) ? 1.0 : MAT_ELEM(a, k + 1 + i, k);
58 w[i] -= gamma * vi;
59 }
60 for (i = 0; i < n - k - 1; i++) {
61 lmmc_real_t vi = (i == 0) ? 1.0 : MAT_ELEM(a, k + 1 + i, k);
62 for (j = i; j < n - k - 1; j++) {
63 lmmc_real_t vj = (j == 0) ? 1.0 : MAT_ELEM(a, k + 1 + j, k);
64 lmmc_real_t update = vi * w[j] + w[i] * vj;
65 MAT_ELEM(a, k + 1 + i, k + 1 + j) -= update;
66 if (i != j) MAT_ELEM(a, k + 1 + j, k + 1 + i) -= update;
67 }
68 }
69 lmmc_free(w);
70 }
71 for (i = 0; i < n; i++) diag[i] = MAT_ELEM(a, i, i);
72 return LMMC_STATUS_OK;
73}

75static lmmc_status_t accumulate_Q(const lmmc_mat_t *a, const lmmc_real_t *taus, lmmc_mat_t *Q) {
76 size_t n = a->rows; size_t i, j, k;
77 for (i = 0; i < n; i++) for (j = 0; j < n; j++) MAT_ELEM(Q, i, j) = (i == j) ? 1.0 : 0.0;
78 if (n <= 2) return LMMC_STATUS_OK;
79
80 for (k = n - 3; ; k--) {
81 lmmc_real_t tau = taus ? taus[k] : 0.0;
82 if (tau != 0.0) {
83
84 for (j = 0; j < n; j++) {
85 lmmc_real_t s = MAT_ELEM(Q, k + 1, j);
86 for (i = k + 2; i < n; i++) s += MAT_ELEM(a, i, k) * MAT_ELEM(Q, i, j);
87 s *= tau;
88 MAT_ELEM(Q, k + 1, j) -= s;
89 for (i = k + 2; i < n; i++) MAT_ELEM(Q, i, j) -= s * MAT_ELEM(a, i, k);
90 }
91 }
92 if (k == 0) break;
93 }
94 return LMMC_STATUS_OK;
95}

97static lmmc_status_t tridiag_ql(lmmc_real_t *diag, lmmc_real_t *offdiag, size_t n, lmmc_mat_t *Q) {
98 size_t i, k, l, m, iter;
99 lmmc_real_t c, g, p, r, s, f, b, tst1;
100 lmmc_real_t eps = 2.2204460492503131e-16;
101 if (n <= 1) return LMMC_STATUS_OK;
102 for (l = 0; l < n; l++) {
103 iter = 0;
104 while (1) {
105 for (m = l; m < n - 1; m++) {
106 tst1 = lmmc_abs(diag[m]) + lmmc_abs(diag[m + 1]);
107 if (tst1 == 0.0) tst1 = 1.0;
108 if (lmmc_abs(offdiag[m]) <= eps * tst1) break;
109 }
110 if (m == l) break;
111 if (iter >= (size_t)EIGEN_MAX_ITER) return LMMC_STATUS_CONVERGENCE_FAILED;
112 iter++;
113 g = (diag[l + 1] - diag[l]) / (2.0 * offdiag[l]);
114 r = sqrt(g * g + 1.0);
115 if (g >= 0.0) g = diag[m] - diag[l] + offdiag[l] / (g + r);
116 else g = diag[m] - diag[l] + offdiag[l] / (g - r);
117 s = 1.0; c = 1.0; p = 0.0;
118 for (i = m; i > l; i--) {
119 f = s * offdiag[i - 1]; b = c * offdiag[i - 1];
120 if (lmmc_abs(f) >= lmmc_abs(g)) {
121 c = g / f; r = sqrt(c * c + 1.0); offdiag[i] = f * r; s = 1.0 / r; c *= s;
122 } else {
123 s = f / g; r = sqrt(s * s + 1.0); offdiag[i] = g * r; c = 1.0 / r; s *= c;
124 }
125 g = diag[i] - p; r = (diag[i - 1] - g) * s + 2.0 * c * b;
126 p = s * r; diag[i] = g + p; g = c * r - b;
127 if (Q) {
128 for (k = 0; k < n; k++) {
129 lmmc_real_t fv = MAT_ELEM(Q, k, i);
130 MAT_ELEM(Q, k, i) = s * MAT_ELEM(Q, k, i - 1) + c * fv;
131 MAT_ELEM(Q, k, i - 1) = c * MAT_ELEM(Q, k, i - 1) - s * fv;
132 }
133 }
134 }
135 diag[l] -= p; offdiag[l] = g; offdiag[m] = 0.0;
136 }
137 }
138 return LMMC_STATUS_OK;
139}
