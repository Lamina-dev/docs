202lmmc_status_t lmmc_sparse_mat_mat_mul_sparse(const lmmc_sparse_mat_t* a, const lmmc_sparse_mat_t* b, lmmc_sparse_mat_t* c) {
203 lmmc_sparse_mat_t a_csr = {0};
204 lmmc_sparse_mat_t b_csr = {0};
205 const lmmc_sparse_mat_t *pa = a;
206 const lmmc_sparse_mat_t *pb = b;
207 lmmc_status_t st = LMMC_STATUS_OK;
208 size_t *marker = NULL;
209 lmmc_real_t *accumulator = NULL;
210 size_t *c_row_ptr = NULL;
211 size_t *c_col_idx = NULL;
212 lmmc_real_t *c_values = NULL;
213 size_t nnz_est = 0;
214 size_t i, j, k, p1, p2;
215
216 if (a == NULL || b == NULL || c == NULL) return LMMC_STATUS_INVALID_ARGUMENT;
217 if (a->cols != b->rows) return LMMC_STATUS_DIMENSION_MISMATCH;
218
219 if (a->format == LMMC_SPARSE_CSC) {
220 st = lmmc_sparse_to_csr(a, &a_csr);
221 if (st != LMMC_STATUS_OK) return st;
222 pa = &a_csr;
223 }
224 if (b->format == LMMC_SPARSE_CSC) {
225 st = lmmc_sparse_to_csr(b, &b_csr);
226 if (st != LMMC_STATUS_OK) {
227 lmmc_sparse_destroy(&a_csr);
228 return st;
229 }
230 pb = &b_csr;
231 }
232
233 marker = (size_t*)lmmc_alloc(pb->cols * sizeof(size_t));
234 c_row_ptr = (size_t*)lmmc_alloc((pa->rows + 1) * sizeof(size_t));
235 if (marker == NULL || c_row_ptr == NULL) {
236 st = LMMC_STATUS_ALLOCATION_FAILED;
237 goto cleanup;
238 }
239 memset(marker, 0xFF, pb->cols * sizeof(size_t));
240 memset(c_row_ptr, 0, (pa->rows + 1) * sizeof(size_t));
241
242 nnz_est = 0;
243 for (i = 0; i < pa->rows; ++i) {
244 for (p1 = pa->row_ptr[i]; p1 < pa->row_ptr[i + 1]; ++p1) {
245 k = pa->col_idx[p1];
246 for (p2 = pb->row_ptr[k]; p2 < pb->row_ptr[k + 1]; ++p2) {
247 j = pb->col_idx[p2];
248 if (marker[j] != i) {
249 marker[j] = i;
250 nnz_est++;
251 }
252 }
253 }
254 c_row_ptr[i + 1] = nnz_est;
255 }
256
257 c_col_idx = (size_t*)lmmc_alloc(nnz_est * sizeof(size_t));
258 c_values = (lmmc_real_t*)lmmc_alloc(nnz_est * sizeof(lmmc_real_t));
259 accumulator = (lmmc_real_t*)lmmc_alloc(pb->cols * sizeof(lmmc_real_t));
260 if ((nnz_est > 0 && (c_col_idx == NULL || c_values == NULL)) || accumulator == NULL) {
261 st = LMMC_STATUS_ALLOCATION_FAILED;
262 goto cleanup;
263 }
264 memset(marker, 0xFF, pb->cols * sizeof(size_t));
265 for (size_t act_i = 0; act_i < pb->cols; act_i++) {
266 LMMC_REAL_INIT(&accumulator[act_i]);
267 LMMC_REAL_SET_D(&accumulator[act_i], 0.0);
268 }
269 for (size_t act_i = 0; act_i < nnz_est; act_i++) {
270 LMMC_REAL_INIT(&c_values[act_i]);
271 }
272
273 lmmc_real_t tmp_mul; LMMC_REAL_INIT(&tmp_mul);
274 lmmc_real_t tmp_sum; LMMC_REAL_INIT(&tmp_sum);
275
276 size_t current_nnz = 0;
277 for (i = 0; i < pa->rows; ++i) {
278 size_t row_start = current_nnz;
279 for (p1 = pa->row_ptr[i]; p1 < pa->row_ptr[i + 1]; ++p1) {
280 k = pa->col_idx[p1];
281 for (p2 = pb->row_ptr[k]; p2 < pb->row_ptr[k + 1]; ++p2) {
282 j = pb->col_idx[p2];
283 if (marker[j] != i) {
284 marker[j] = i;
285 c_col_idx[current_nnz++] = j;
286 }
287 LMMC_REAL_MUL(&tmp_mul, &pa->values[p1], &pb->values[p2]);
288 LMMC_REAL_ADD(&tmp_sum, &accumulator[j], &tmp_mul);
289 LMMC_REAL_SET(&accumulator[j], &tmp_sum);
290 }
291 }
292 for (p1 = row_start; p1 < current_nnz; ++p1) {
293 j = c_col_idx[p1];
294 LMMC_REAL_SET(&c_values[p1], &accumulator[j]);
295 LMMC_REAL_SET_D(&accumulator[j], 0.0);
296 }
297 }
298
299 LMMC_REAL_CLEAR(&tmp_mul);
300 LMMC_REAL_CLEAR(&tmp_sum);
301
302 c->rows = pa->rows;
303 c->cols = pb->cols;
304 c->nnz = nnz_est;
305 c->row_ptr = c_row_ptr;
306 c->col_idx = c_col_idx;
307 c->values = c_values;
308 c->format = LMMC_SPARSE_CSR;
309 c->owns_data = 1;
310
311 c_row_ptr = NULL;
312 c_col_idx = NULL;
313 c_values = NULL;
314
315cleanup:
316 if (marker) lmmc_free(marker);
317 if (accumulator) {
318 for (size_t act_i = 0; act_i < pb->cols; act_i++) {
319 LMMC_REAL_CLEAR(&accumulator[act_i]);
320 }
321 lmmc_free(accumulator);
322 }
323 if (c_row_ptr) lmmc_free(c_row_ptr);
324 if (c_col_idx) lmmc_free(c_col_idx);
325 if (c_values) {
326 for (size_t act_i = 0; act_i < nnz_est; act_i++) {
327 LMMC_REAL_CLEAR(&c_values[act_i]);
328 }
329 lmmc_free(c_values);
330 }
331 lmmc_sparse_destroy(&a_csr);
332 lmmc_sparse_destroy(&b_csr);
333 return st;
