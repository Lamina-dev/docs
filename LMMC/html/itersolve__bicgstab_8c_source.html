22static lmmc_status_t lmmc_bicgstab_direction(lmmc_bicgstab_state_t* s, size_t iter) {
23 lmmc_status_t st = lmmc_vec_dot_checked(&s->r_hat, &s->r, &s->rho_hat);
24 if (st != LMMC_STATUS_OK) {
25 return st;
26 }
27 if (fabs(s->rho_hat) <= 1e-30) {
28 return LMMC_STATUS_NUMERICAL_FAILURE;
29 }
30 if (iter == 0) {
31 return lmmc_vec_copy(&s->r, &s->p);
32 }
33 if (fabs(s->omega) <= 1e-30) {
34 return LMMC_STATUS_NUMERICAL_FAILURE;
35 }
36 lmmc_real_t beta = (s->rho_hat / s->rho) * (s->alpha / s->omega);
37 if (!isfinite(beta)) {
38 return LMMC_STATUS_NUMERICAL_FAILURE;
39 }
40 for (size_t i = 0; i < s->p.size; ++i) {
41 lmmc_real_t product = s->omega * s->v.data[i];
42 lmmc_real_t difference = s->p.data[i] - product;
43 product = beta * difference;
44 s->p.data[i] = s->r.data[i] + product;
45 if (!isfinite(s->p.data[i])) {
46 return LMMC_STATUS_NUMERICAL_FAILURE;
47 }
48 }
49 return LMMC_STATUS_OK;
50}

53 const lmmc_precond_t* precond, lmmc_bicgstab_state_t* s) {

65 return st;

111static lmmc_status_t lmmc_bicgstab_update(lmmc_vec_t* x,
112 lmmc_bicgstab_state_t* s, int early) {
113 for (size_t i = 0; i < x->size; ++i) {
114 lmmc_real_t delta = s->alpha * s->y.data[i];
115 if (!early) {
116 lmmc_real_t second = s->omega * s->z.data[i];
117 delta = delta + second;
118 }
119 x->data[i] = x->data[i] + delta;
120 if (!isfinite(x->data[i])) {
121 return LMMC_STATUS_NUMERICAL_FAILURE;
122 }
123 }
124 if (early) {
125 s->norm = s->norm_s;
126 return LMMC_STATUS_OK;
127 }
128 lmmc_status_t st = lmmc_vec_copy(&s->s, &s->r);
129 if (st != LMMC_STATUS_OK) {
130 return st;
131 }
132 st = lmmc_vec_axpy(-s->omega, &s->t, &s->r);
133 if (st != LMMC_STATUS_OK) {
134 return st;
135 }
