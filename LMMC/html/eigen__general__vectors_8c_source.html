62 for (size_t i = 0; i < n; i++) {

64 MAT_ELEM(&out_result->vectors_imag, i, j) = 0.0;
65 }
66 }
67 status = lmmc_eigen_inverse_vectors(&scaled_a, out_result);
68 if (status == LMMC_STATUS_OK) {
69 status = lmmc_eigen_restore_scale(&out_result->real_parts,

71 }
72cleanup:

77lmmc_status_t lmmc_eigen_general_full(const lmmc_mat_t* a,
78 lmmc_eigen_gen_full_result_t* out_result)
79{
80 lmmc_real_t matrix_scale;
81 if (!lmmc_mat_descriptor_is_valid(a) || !out_result) {
82 return LMMC_STATUS_INVALID_ARGUMENT;
83 }
84 if (a->rows != a->cols) {
85 return LMMC_STATUS_INVALID_ARGUMENT;
86 }
87 lmmc_status_t status = lmmc_eigen_matrix_scale(a, &matrix_scale);
88 if (status != LMMC_STATUS_OK) {
89 return status;
90 }
91 size_t n = a->rows;
92 status = create_full_result(n, out_result);
93 if (status != LMMC_STATUS_OK) {
94 goto fail;
95 }
96 if (n == 1) {
97 out_result->real_parts.data[0] = MAT_ELEM(a, 0, 0);
98 out_result->imag_parts.data[0] = 0.0;
99 MAT_ELEM(&out_result->vectors_real, 0, 0) = 1.0;
100 MAT_ELEM(&out_result->vectors_imag, 0, 0) = 0.0;
101 return LMMC_STATUS_OK;
102 }
103if (n == 2) {
104 lmmc_eigen_vectors_2x2(a, matrix_scale, out_result);
105 return LMMC_STATUS_OK;
106 }
107 status = eigenvectors_large(a, matrix_scale, out_result);
108 if (status == LMMC_STATUS_OK) {
109 return status;
110 }
111fail:
112 lmmc_eigen_gen_full_result_destroy(out_result);
113 memset(out_result, 0, sizeof(*out_result));
114 return status;
