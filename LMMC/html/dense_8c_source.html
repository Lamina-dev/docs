15static int lmmc_mul_overflow_size(size_t a, size_t b, size_t* out) {
16 if (a == 0 || b == 0) {
17 *out = 0;
18 return 0;
19 }
20 if (a > ((size_t)-1) / b) {
21 return 1;

134lmmc_status_t lmmc_mat_gemm(lmmc_real_t alpha, const lmmc_mat_t* A, int transA,
135 const lmmc_mat_t* B, int transB, lmmc_real_t beta, lmmc_mat_t* C) {
136 if (A == NULL || B == NULL || C == NULL ||
137 A->data == NULL || B->data == NULL || C->data == NULL) {
138 return LMMC_STATUS_INVALID_ARGUMENT;
139 }
140
141 /* Determine effective dimensions of op(A) and op(B) */
142 size_t M = transA ? A->cols : A->rows;
143 size_t K_A = transA ? A->rows : A->cols;
144 size_t K_B = transB ? B->cols : B->rows;
145 size_t N = transB ? B->rows : B->cols;
146
147 /* Dimension validation */
148 if (K_A != K_B) {
149 return LMMC_STATUS_DIMENSION_MISMATCH;
150 }
151 if (C->rows != M || C->cols != N) {
152 return LMMC_STATUS_DIMENSION_MISMATCH;
153 }
154
155 size_t K = K_A;
156
157 const lmmc_real_t* restrict a_data = A->data;
158 const lmmc_real_t* restrict b_data = B->data;
159 lmmc_real_t* restrict c_data = C->data;
160
161 size_t a_stride = A->stride;
162 size_t b_stride = B->stride;
163 size_t c_stride = C->stride;
164
165#ifdef LMMC_USE_BLAS
166 {
167 /* Route through BLAS dgemm with transpose support */
168 lmmc_blas_dgemm_ex(transA, transB,
169 M, N, K,
170 alpha,
171 a_data, a_stride,
172 b_data, b_stride,
173 beta,
174 c_data, c_stride);
175 return LMMC_STATUS_OK;
176 }
177#endif
178
179 /* Scale C by beta */
180 if (beta == 0.0) {
181 for (size_t i = 0; i < M; ++i) {
182 for (size_t j = 0; j < N; ++j) {
183 LMMC_REAL_SET_D(&c_data[i * c_stride + j], 0.0);
184 }
185 }
186 } else if (beta != 1.0) {
187 lmmc_real_t tmp_mul; LMMC_REAL_INIT(&tmp_mul);
188 for (size_t i = 0; i < M; ++i) {
189 for (size_t j = 0; j < N; ++j) {
190 LMMC_REAL_MUL(&tmp_mul, &beta, &c_data[i * c_stride + j]);
191 LMMC_REAL_SET(&c_data[i * c_stride + j], &tmp_mul);
192 }
193 }
194 LMMC_REAL_CLEAR(&tmp_mul);
195 }
196
197 /* If alpha is zero, we're done */
198 if (alpha == 0.0) {
199 return LMMC_STATUS_OK;
200 }
201
202 /* Native triple-loop: C += alpha * op(A) * op(B) */
203 lmmc_real_t tmp_mul; LMMC_REAL_INIT(&tmp_mul);
204 lmmc_real_t tmp_sum; LMMC_REAL_INIT(&tmp_sum);
205 lmmc_real_t alpha_a_ik; LMMC_REAL_INIT(&alpha_a_ik);
206
207 size_t bs = 64;
208 for (size_t ii = 0; ii < M; ii += bs) {
209 size_t i_end = (ii + bs > M) ? M : ii + bs;
210 for (size_t kk = 0; kk < K; kk += bs) {
211 size_t k_end = (kk + bs > K) ? K : kk + bs;
212 for (size_t jj = 0; jj < N; jj += bs) {
213 size_t j_end = (jj + bs > N) ? N : jj + bs;
214 for (size_t i = ii; i < i_end; ++i) {
215 for (size_t k = kk; k < k_end; ++k) {
216 /* Get op(A)[i,k] */
217 lmmc_real_t a_ik;
218 if (transA) {
219 a_ik = a_data[k * a_stride + i];
220 } else {
221 a_ik = a_data[i * a_stride + k];
222 }
223 /* alpha * op(A)[i,k] */
224 LMMC_REAL_MUL(&alpha_a_ik, &alpha, &a_ik);
225
226 for (size_t j = jj; j < j_end; ++j) {
227 /* Get op(B)[k,j] */
228 lmmc_real_t b_kj;
229 if (transB) {
230 b_kj = b_data[j * b_stride + k];
231 } else {
232 b_kj = b_data[k * b_stride + j];
233 }
234 LMMC_REAL_MUL(&tmp_mul, &alpha_a_ik, &b_kj);
235 LMMC_REAL_ADD(&tmp_sum, &c_data[i * c_stride + j], &tmp_mul);
236 LMMC_REAL_SET(&c_data[i * c_stride + j], &tmp_sum);
237 }
238 }
239 }
240 }
241 }
242 }
243
244 LMMC_REAL_CLEAR(&tmp_mul);
245 LMMC_REAL_CLEAR(&tmp_sum);
246 LMMC_REAL_CLEAR(&alpha_a_ik);
247 return LMMC_STATUS_OK;
248}

373lmmc_status_t lmmc_mat_gemv(lmmc_real_t alpha, const lmmc_mat_t* A, int transA,
374 const lmmc_vec_t* x, lmmc_real_t beta, lmmc_vec_t* y) {
375 if (A == NULL || x == NULL || y == NULL ||
376 A->data == NULL || x->data == NULL || y->data == NULL) {
377 return LMMC_STATUS_INVALID_ARGUMENT;
378 }
379
380 /* Determine effective dimensions of op(A) */
381 size_t M = transA ? A->cols : A->rows;
382 size_t N = transA ? A->rows : A->cols;
383
384 /* Dimension validation */
385 if (N != x->size || M != y->size) {
386 return LMMC_STATUS_DIMENSION_MISMATCH;
387 }
388
389 const lmmc_real_t* restrict a_data = A->data;
390 size_t a_stride = A->stride;
391
392#ifdef LMMC_USE_BLAS
393 {
394 char trans_char = transA ? 'T' : 'N';
395 lmmc_blas_dgemv(trans_char, A->rows, A->cols,
396 alpha,
397 a_data, a_stride,
398 x->data, 1,
399 beta,
400 y->data, 1);
401 return LMMC_STATUS_OK;
402 }
403#endif
404
405 /* Scale y by beta */
406 if (beta == 0.0) {
407 for (size_t i = 0; i < M; ++i) {
408 LMMC_REAL_SET_D(&y->data[i], 0.0);
409 }
410 } else if (beta != 1.0) {
411 lmmc_real_t tmp_mul; LMMC_REAL_INIT(&tmp_mul);
412 for (size_t i = 0; i < M; ++i) {
413 LMMC_REAL_MUL(&tmp_mul, &beta, &y->data[i]);
414 LMMC_REAL_SET(&y->data[i], &tmp_mul);
415 }
416 LMMC_REAL_CLEAR(&tmp_mul);
417 }
418
419 /* If alpha is zero, we're done */
420 if (alpha == 0.0) {
421 return LMMC_STATUS_OK;
422 }
423
424 /* Native loop: y += alpha * op(A) * x */
425 lmmc_real_t sum; LMMC_REAL_INIT(&sum);
426 lmmc_real_t tmp_mul; LMMC_REAL_INIT(&tmp_mul);
427 lmmc_real_t tmp_sum; LMMC_REAL_INIT(&tmp_sum);
428 lmmc_real_t alpha_sum; LMMC_REAL_INIT(&alpha_sum);
429
430 for (size_t i = 0; i < M; ++i) {
431 LMMC_REAL_SET_D(&sum, 0.0);
432 for (size_t j = 0; j < N; ++j) {
433 lmmc_real_t a_ij;
434 if (transA) {
435 a_ij = a_data[j * a_stride + i];
436 } else {
437 a_ij = a_data[i * a_stride + j];
438 }
439 LMMC_REAL_MUL(&tmp_mul, &a_ij, &x->data[j]);
440 LMMC_REAL_ADD(&tmp_sum, &sum, &tmp_mul);
441 LMMC_REAL_SET(&sum, &tmp_sum);
442 }
443 /* y[i] += alpha * sum */
444 LMMC_REAL_MUL(&alpha_sum, &alpha, &sum);
445 LMMC_REAL_ADD(&tmp_sum, &y->data[i], &alpha_sum);
446 LMMC_REAL_SET(&y->data[i], &tmp_sum);
447 }
448
449 LMMC_REAL_CLEAR(&sum);
450 LMMC_REAL_CLEAR(&tmp_mul);
451 LMMC_REAL_CLEAR(&tmp_sum);
452 LMMC_REAL_CLEAR(&alpha_sum);
453 return LMMC_STATUS_OK;
454}

767lmmc_status_t lmmc_mat_det(const lmmc_mat_t* a, lmmc_real_t* out_det) {
768 if (a == NULL || out_det == NULL || a->data == NULL) {
769 return LMMC_STATUS_INVALID_ARGUMENT;
770 }
771 if (a->rows != a->cols) {
772 return LMMC_STATUS_INVALID_ARGUMENT;
773 }
774
775 size_t n = a->rows;
776
777
778 if (n == 1) {
779 LMMC_REAL_SET(out_det, &a->data[0]);
780 return LMMC_STATUS_OK;
781 }
782
783
784 if (n == 2) {
785 lmmc_real_t ad; LMMC_REAL_INIT(&ad);
786 lmmc_real_t bc; LMMC_REAL_INIT(&bc);
787 lmmc_real_t result; LMMC_REAL_INIT(&result);
788
789 LMMC_REAL_MUL(&ad, &a->data[0 * a->stride + 0], &a->data[1 * a->stride + 1]);
790 LMMC_REAL_MUL(&bc, &a->data[0 * a->stride + 1], &a->data[1 * a->stride + 0]);
791 LMMC_REAL_SUB(&result, &ad, &bc);
792 LMMC_REAL_SET(out_det, &result);
793
794 LMMC_REAL_CLEAR(&ad);
795 LMMC_REAL_CLEAR(&bc);
796 LMMC_REAL_CLEAR(&result);
797 return LMMC_STATUS_OK;
798 }
799
800
801 size_t n_elem = n * n;
802 size_t n_bytes = n_elem * sizeof(lmmc_real_t);
803 lmmc_real_t* lu = (lmmc_real_t*)lmmc_alloc(n_bytes);
804 if (lu == NULL) {
805 return LMMC_STATUS_ALLOCATION_FAILED;
806 }
807
808
809 for (size_t i = 0; i < n; ++i) {
810 for (size_t j = 0; j < n; ++j) {
811 lu[i * n + j] = a->data[i * a->stride + j];
812 }
813 }
814
815 int sign = 1;
816
817 lmmc_real_t abs_val; LMMC_REAL_INIT(&abs_val);
818 lmmc_real_t max_val; LMMC_REAL_INIT(&max_val);
819 lmmc_real_t factor; LMMC_REAL_INIT(&factor);
820 lmmc_real_t tmp_mul; LMMC_REAL_INIT(&tmp_mul);
821 lmmc_real_t tmp_sub; LMMC_REAL_INIT(&tmp_sub);
822
823 for (size_t k = 0; k < n; ++k) {
824
825 size_t pivot_row = k;
826 LMMC_REAL_ABS(&max_val, &lu[k * n + k]);
827
828 for (size_t i = k + 1; i < n; ++i) {
829 LMMC_REAL_ABS(&abs_val, &lu[i * n + k]);
830 if (LMMC_REAL_CMP(&abs_val, &max_val) > 0) {
831 LMMC_REAL_SET(&max_val, &abs_val);
832 pivot_row = i;
833 }
834 }
835
836
837 lmmc_real_t zero; LMMC_REAL_INIT(&zero);
838 LMMC_REAL_SET_D(&zero, 0.0);
839 if (LMMC_REAL_CMP(&max_val, &zero) == 0) {
840 LMMC_REAL_SET_D(out_det, 0.0);
841 LMMC_REAL_CLEAR(&zero);
842 LMMC_REAL_CLEAR(&abs_val);
843 LMMC_REAL_CLEAR(&max_val);
844 LMMC_REAL_CLEAR(&factor);
845 LMMC_REAL_CLEAR(&tmp_mul);
846 LMMC_REAL_CLEAR(&tmp_sub);
847 lmmc_free(lu);
848 return LMMC_STATUS_OK;
849 }
850 LMMC_REAL_CLEAR(&zero);
851
852
853 if (pivot_row != k) {
854 for (size_t j = 0; j < n; ++j) {
855 lmmc_real_t tmp = lu[k * n + j];
856 lu[k * n + j] = lu[pivot_row * n + j];
857 lu[pivot_row * n + j] = tmp;
858 }
859 sign = -sign;
860 }
861
862
863 for (size_t i = k + 1; i < n; ++i) {
864 LMMC_REAL_DIV(&factor, &lu[i * n + k], &lu[k * n + k]);
865 LMMC_REAL_SET(&lu[i * n + k], &factor);
866
867 for (size_t j = k + 1; j < n; ++j) {
868 LMMC_REAL_MUL(&tmp_mul, &factor, &lu[k * n + j]);
869 LMMC_REAL_SUB(&tmp_sub, &lu[i * n + j], &tmp_mul);
870 LMMC_REAL_SET(&lu[i * n + j], &tmp_sub);
871 }
872 }
873 }
874
875
876 lmmc_real_t det; LMMC_REAL_INIT(&det);
877 LMMC_REAL_SET_D(&det, (double)sign);
878
879 lmmc_real_t tmp_prod; LMMC_REAL_INIT(&tmp_prod);
880 for (size_t i = 0; i < n; ++i) {
881 LMMC_REAL_MUL(&tmp_prod, &det, &lu[i * n + i]);
882 LMMC_REAL_SET(&det, &tmp_prod);
883 }
884 LMMC_REAL_SET(out_det, &det);
885
886 LMMC_REAL_CLEAR(&det);
887 LMMC_REAL_CLEAR(&tmp_prod);
888 LMMC_REAL_CLEAR(&abs_val);
889 LMMC_REAL_CLEAR(&max_val);
890 LMMC_REAL_CLEAR(&factor);
891 LMMC_REAL_CLEAR(&tmp_mul);
892 LMMC_REAL_CLEAR(&tmp_sub);
893 lmmc_free(lu);
894
895 return LMMC_STATUS_OK;
896}

898lmmc_status_t lmmc_mat_inv(const lmmc_mat_t* A, lmmc_mat_t* A_inv) {
899 lmmc_mat_t lu_mat;
900 size_t* pivots = NULL;
901 lmmc_vec_t ei;
902 lmmc_vec_t col;
903 size_t n, i, j;
904 lmmc_status_t status;
905
906 if (A == NULL || A_inv == NULL || A->data == NULL || A_inv->data == NULL) {
907 return LMMC_STATUS_INVALID_ARGUMENT;
908 }
909 if (A->rows != A->cols) {
910 return LMMC_STATUS_INVALID_ARGUMENT;
911 }
912 if (A_inv->rows != A->rows || A_inv->cols != A->cols) {
913 return LMMC_STATUS_DIMENSION_MISMATCH;
914 }
915
916 n = A->rows;
917
918 /* Copy A into workspace for LU decomposition */
919 status = lmmc_mat_create(n, n, &lu_mat);
920 if (status != LMMC_STATUS_OK) {
921 return status;
922 }
923 status = lmmc_mat_copy(A, &lu_mat);
924 if (status != LMMC_STATUS_OK) {
925 lmmc_mat_destroy(&lu_mat);
926 return status;
927 }
928
929 /* Allocate pivot array */
930 pivots = (size_t*)lmmc_alloc(n * sizeof(size_t));
931 if (pivots == NULL) {
932 lmmc_mat_destroy(&lu_mat);
933 return LMMC_STATUS_ALLOCATION_FAILED;
934 }
935
936 /* LU decompose with partial pivoting */
937 status = lmmc_lu_decompose_inplace(&lu_mat, pivots, NULL);
938 if (status != LMMC_STATUS_OK) {
939 /* Singular or other failure — leave A_inv unmodified */
940 lmmc_free(pivots);
941 lmmc_mat_destroy(&lu_mat);
942 return status;
943 }
944
945 status = lmmc_vec_create(n, &ei);
946 if (status != LMMC_STATUS_OK) {
947 lmmc_free(pivots);
948 lmmc_mat_destroy(&lu_mat);
949 return status;
950 }
951 status = lmmc_vec_create(n, &col);
952 if (status != LMMC_STATUS_OK) {
953 lmmc_vec_destroy(&ei);
954 lmmc_free(pivots);
955 lmmc_mat_destroy(&lu_mat);
956 return status;
957 }
958
959 /* Solve A * col_i = e_i for each column of the identity */
960 for (i = 0; i < n; ++i) {
961 /* Set up e_i (standard basis vector) */
962 for (j = 0; j < n; ++j) {
963 LMMC_REAL_SET_D(&ei.data[j], 0.0);
964 }
965 LMMC_REAL_SET_D(&ei.data[i], 1.0);
966
967 /* Solve using LU factors */
968 status = lmmc_lu_solve(&lu_mat, pivots, &ei, &col);
969 if (status != LMMC_STATUS_OK) {
970 lmmc_vec_destroy(&col);
971 lmmc_vec_destroy(&ei);
972 lmmc_free(pivots);
973 lmmc_mat_destroy(&lu_mat);
974 return status;
975 }
976
977 /* Copy solution into column i of A_inv */
978 for (j = 0; j < n; ++j) {
979 LMMC_REAL_SET(&A_inv->data[j * A_inv->stride + i], &col.data[j]);
980 }
981 }
982
983 lmmc_vec_destroy(&col);
984 lmmc_vec_destroy(&ei);
985 lmmc_free(pivots);
986 lmmc_mat_destroy(&lu_mat);
987
988 return LMMC_STATUS_OK;
989}

991lmmc_status_t lmmc_solve_triangular(const lmmc_mat_t* T, int upper, int diag_unit,
992 const lmmc_vec_t* b, lmmc_vec_t* x) {
993 size_t n, i, j;
994 lmmc_real_t sum, tmp, diag_val, abs_diag, eps;
995
996 if (T == NULL || b == NULL || x == NULL || T->data == NULL || b->data == NULL || x->data == NULL) {
997 return LMMC_STATUS_INVALID_ARGUMENT;
998 }
999 if (T->rows != T->cols) {
1000 return LMMC_STATUS_INVALID_ARGUMENT;
1001 }
1002 if (b->size != T->rows || x->size != T->rows) {
1003 return LMMC_STATUS_DIMENSION_MISMATCH;
1004 }
1005
1006 n = T->rows;
1007 LMMC_REAL_INIT(&sum);
1008 LMMC_REAL_INIT(&tmp);
1009 LMMC_REAL_INIT(&diag_val);
1010 LMMC_REAL_INIT(&abs_diag);
1011 LMMC_REAL_INIT(&eps);
1012 LMMC_REAL_SET_D(&eps, 1e-15);
1013
1014 if (upper) {
1015 /* Back substitution for upper triangular */
1016 for (i = n; i-- > 0;) {
1017 LMMC_REAL_SET(&sum, &b->data[i]);
1018 for (j = i + 1; j < n; ++j) {
1019 LMMC_REAL_MUL(&tmp, &T->data[i * T->stride + j], &x->data[j]);
1020 LMMC_REAL_SUB(&sum, &sum, &tmp);
1021 }
1022 if (diag_unit) {
1023 LMMC_REAL_SET(&x->data[i], &sum);
1024 } else {
1025 LMMC_REAL_SET(&diag_val, &T->data[i * T->stride + i]);
1026 LMMC_REAL_ABS(&abs_diag, &diag_val);
1027 if (LMMC_REAL_CMP(&abs_diag, &eps) <= 0) {
1028 LMMC_REAL_CLEAR(&sum);
1029 LMMC_REAL_CLEAR(&tmp);
1030 LMMC_REAL_CLEAR(&diag_val);
1031 LMMC_REAL_CLEAR(&abs_diag);
1032 LMMC_REAL_CLEAR(&eps);
1033 return LMMC_STATUS_SINGULAR_MATRIX;
1034 }
1035 LMMC_REAL_DIV(&x->data[i], &sum, &diag_val);
1036 }
1037 }
1038 } else {
1039 /* Forward substitution for lower triangular */
1040 for (i = 0; i < n; ++i) {
1041 LMMC_REAL_SET(&sum, &b->data[i]);
1042 for (j = 0; j < i; ++j) {
1043 LMMC_REAL_MUL(&tmp, &T->data[i * T->stride + j], &x->data[j]);
1044 LMMC_REAL_SUB(&sum, &sum, &tmp);
1045 }
1046 if (diag_unit) {
1047 LMMC_REAL_SET(&x->data[i], &sum);
1048 } else {
1049 LMMC_REAL_SET(&diag_val, &T->data[i * T->stride + i]);
1050 LMMC_REAL_ABS(&abs_diag, &diag_val);
1051 if (LMMC_REAL_CMP(&abs_diag, &eps) <= 0) {
1052 LMMC_REAL_CLEAR(&sum);
1053 LMMC_REAL_CLEAR(&tmp);
1054 LMMC_REAL_CLEAR(&diag_val);
1055 LMMC_REAL_CLEAR(&abs_diag);
1056 LMMC_REAL_CLEAR(&eps);
1057 return LMMC_STATUS_SINGULAR_MATRIX;
1058 }
1059 LMMC_REAL_DIV(&x->data[i], &sum, &diag_val);
1060 }
1061 }
1062 }
1063
1064 LMMC_REAL_CLEAR(&sum);
1065 LMMC_REAL_CLEAR(&tmp);
1066 LMMC_REAL_CLEAR(&diag_val);
1067 LMMC_REAL_CLEAR(&abs_diag);
1068 LMMC_REAL_CLEAR(&eps);
1069
1070 return LMMC_STATUS_OK;
1071}

1417lmmc_status_t lmmc_vec_cross(const lmmc_vec_t* a, const lmmc_vec_t* b, lmmc_vec_t* c)
1418{

1421
1422 if (a == NULL || b == NULL || c == NULL ||
1423 a->data == NULL || b->data == NULL || c->data == NULL) {
1424 return LMMC_STATUS_INVALID_ARGUMENT;
1425 }
1426 if (a->size != 3 || b->size != 3 || c->size != 3) {
1427 return LMMC_STATUS_DIMENSION_MISMATCH;
1428 }
1429
1430 LMMC_REAL_INIT(&t0);
1431 LMMC_REAL_INIT(&t1);
1432 LMMC_REAL_INIT(&t2);
1433 LMMC_REAL_INIT(&mul1);
1434 LMMC_REAL_INIT(&mul2);
1435
1436 /* c[0] = a[1]*b[2] - a[2]*b[1] */
1437 LMMC_REAL_MUL(&mul1, &a->data[1], &b->data[2]);
1438 LMMC_REAL_MUL(&mul2, &a->data[2], &b->data[1]);
1439 LMMC_REAL_SUB(&t0, &mul1, &mul2);
1440
1441 /* c[1] = a[2]*b[0] - a[0]*b[2] */
1442 LMMC_REAL_MUL(&mul1, &a->data[2], &b->data[0]);
1443 LMMC_REAL_MUL(&mul2, &a->data[0], &b->data[2]);
1444 LMMC_REAL_SUB(&t1, &mul1, &mul2);
1445
1446 /* c[2] = a[0]*b[1] - a[1]*b[0] */
1447 LMMC_REAL_MUL(&mul1, &a->data[0], &b->data[1]);
1448 LMMC_REAL_MUL(&mul2, &a->data[1], &b->data[0]);
1449 LMMC_REAL_SUB(&t2, &mul1, &mul2);
1450
1451 /* Write results (safe for aliasing since we used temporaries) */
1452 LMMC_REAL_SET(&c->data[0], &t0);
1453 LMMC_REAL_SET(&c->data[1], &t1);
1454 LMMC_REAL_SET(&c->data[2], &t2);
1455
1456 LMMC_REAL_CLEAR(&t0);
1457 LMMC_REAL_CLEAR(&t1);
1458 LMMC_REAL_CLEAR(&t2);
1459 LMMC_REAL_CLEAR(&mul1);

1464lmmc_status_t lmmc_mat_rank(const lmmc_mat_t* a, lmmc_real_t tol, size_t* out_rank)
1465{
1466 lmmc_svd_result_t svd;
1467 lmmc_status_t status;
1468 size_t i, count;
1469 size_t max_dim;
1470 lmmc_real_t effective_tol;
1471 lmmc_real_t eps_val;
1472 lmmc_real_t dim_val;
1473 lmmc_real_t tmp;
1474
1475 if (a == NULL || out_rank == NULL || a->data == NULL) {
1476 return LMMC_STATUS_INVALID_ARGUMENT;
1477 }
1478
1479 /* Compute SVD */
1480 status = lmmc_svd(a, &svd);
1481 if (status != LMMC_STATUS_OK) {
1482 return status;
1483 }
1484
1485 /* Determine effective tolerance */
1486 LMMC_REAL_INIT(&effective_tol);
1487 if (tol <= 0.0) {
1488 /* Default: max(rows, cols) * LMMC_REAL_EPSILON * sigma[0] */
1489 max_dim = (a->rows > a->cols) ? a->rows : a->cols;
1490
1491 LMMC_REAL_INIT(&eps_val);
1492 LMMC_REAL_SET_D(&eps_val, LMMC_REAL_EPSILON);
1493
1494 LMMC_REAL_INIT(&dim_val);
1495 LMMC_REAL_SET_D(&dim_val, (double)max_dim);
1496
1497 LMMC_REAL_INIT(&tmp);
1498 LMMC_REAL_MUL(&tmp, &dim_val, &eps_val);
1499 LMMC_REAL_MUL(&effective_tol, &tmp, &svd.sigma.data[0]);
1500
1501 LMMC_REAL_CLEAR(&eps_val);
1502 LMMC_REAL_CLEAR(&dim_val);
1503 LMMC_REAL_CLEAR(&tmp);
1504 } else {
1505 LMMC_REAL_SET_D(&effective_tol, tol);
1506 }
1507
1508 /* Count singular values > tol */
1509 count = 0;
1510 for (i = 0; i < svd.sigma.size; ++i) {
1511 if (LMMC_REAL_CMP(&svd.sigma.data[i], &effective_tol) > 0) {
1512 count++;
1513 }
1514 }
1515
1516 *out_rank = count;
1517
1518 LMMC_REAL_CLEAR(&effective_tol);
1519 lmmc_svd_result_destroy(&svd);
1520 return LMMC_STATUS_OK;
1521}

1523lmmc_status_t lmmc_mat_pow(const lmmc_mat_t* a, int n, lmmc_mat_t* out)
1524{
1525 lmmc_mat_t base;
1526 lmmc_mat_t temp;
1527 lmmc_mat_t inv_mat;
1528 lmmc_status_t status;
1529 size_t dim;
1530 size_t total;
1531 unsigned int exp;
1532
1533 if (a == NULL || out == NULL || a->data == NULL || out->data == NULL) {
1534 return LMMC_STATUS_INVALID_ARGUMENT;
1535 }
1536 if (a->rows != a->cols) {
1537 return LMMC_STATUS_INVALID_ARGUMENT;
1538 }
1539 dim = a->rows;
1540 if (out->rows != dim || out->cols != dim) {
1541 return LMMC_STATUS_INVALID_ARGUMENT;
1542 }
1543
1544 /* n == 0 → identity */
1545 if (n == 0) {
1546 lmmc_real_t zero_val;
1547 lmmc_real_t one_val;
1548 size_t i, j;
1549 LMMC_REAL_SET_D(&zero_val, 0.0);
1550 LMMC_REAL_SET_D(&one_val, 1.0);
1551 for (i = 0; i < dim; ++i) {
1552 for (j = 0; j < dim; ++j) {
1553 if (i == j) {
1554 LMMC_REAL_SET(&out->data[i * out->stride + j], &one_val);
1555 } else {
1556 LMMC_REAL_SET(&out->data[i * out->stride + j], &zero_val);
1557 }
1558 }
1559 }
1560 return LMMC_STATUS_OK;
1561 }
1562
1563 /* n == 1 → copy */
1564 if (n == 1) {
1565 return lmmc_mat_copy(a, out);
1566 }
1567
1568 /* Check size overflow for temporaries */
1569 if (!lmmc_safe_mul_size(dim, dim, &total)) {
1570 return LMMC_STATUS_ALLOCATION_FAILED;
1571 }
1572
1573 /* n < 0 → compute inverse first, then raise to |n| */
1574 if (n < 0) {
1575 status = lmmc_mat_create(dim, dim, &inv_mat);
1576 if (status != LMMC_STATUS_OK) {
1577 return status;
1578 }
1579 status = lmmc_mat_inv(a, &inv_mat);
1580 if (status != LMMC_STATUS_OK) {
1581 lmmc_mat_destroy(&inv_mat);
1582 return status;
1583 }
1584 /* Use unsigned to handle INT_MIN safely */
1585 exp = (n == -1) ? 1u : (unsigned int)(-(n + 1)) + 1u;
1586 } else {
1587 exp = (unsigned int)n;
1588 }
1589
1590 /* Allocate two temporary matrices: base and temp */
1591 base.data = NULL;
1592 temp.data = NULL;
1593
1594 status = lmmc_mat_create(dim, dim, &base);
1595 if (status != LMMC_STATUS_OK) {
1596 if (n < 0) { lmmc_mat_destroy(&inv_mat); }
1597 return status;
1598 }
1599 status = lmmc_mat_create(dim, dim, &temp);
1600 if (status != LMMC_STATUS_OK) {
1601 lmmc_mat_destroy(&base);
1602 if (n < 0) { lmmc_mat_destroy(&inv_mat); }
1603 return status;
1604 }
1605
1606 /* Initialize base = A (or A^{-1} if n < 0) */
1607 if (n < 0) {
1608 status = lmmc_mat_copy(&inv_mat, &base);
1609 lmmc_mat_destroy(&inv_mat);
1610 } else {
1611 status = lmmc_mat_copy(a, &base);
1612 }
1613 if (status != LMMC_STATUS_OK) {
1614 lmmc_mat_destroy(&temp);
1615 lmmc_mat_destroy(&base);
1616 return status;
1617 }
1618
1619 /* Initialize result (out) = identity */
1620 {
1621 lmmc_real_t zero_val;
1622 lmmc_real_t one_val;
1623 size_t i, j;
1624 LMMC_REAL_SET_D(&zero_val, 0.0);
1625 LMMC_REAL_SET_D(&one_val, 1.0);
1626 for (i = 0; i < dim; ++i) {
1627 for (j = 0; j < dim; ++j) {
1628 if (i == j) {
1629 LMMC_REAL_SET(&out->data[i * out->stride + j], &one_val);
1630 } else {
1631 LMMC_REAL_SET(&out->data[i * out->stride + j], &zero_val);
1632 }
1633 }
1634 }
1635 }
1636
1637 /* Binary exponentiation */
1638 while (exp > 0) {
1639 if (exp & 1u) {
1640 /* temp = out * base */
1641 status = lmmc_mat_mul(out, &base, &temp);
1642 if (status != LMMC_STATUS_OK) {
1643 lmmc_mat_destroy(&temp);
1644 lmmc_mat_destroy(&base);
1645 return status;
1646 }
1647 /* out = temp */
1648 status = lmmc_mat_copy(&temp, out);
1649 if (status != LMMC_STATUS_OK) {
1650 lmmc_mat_destroy(&temp);
1651 lmmc_mat_destroy(&base);
1652 return status;
1653 }
1654 }
1655 exp >>= 1u;
1656 if (exp > 0) {
1657 /* temp = base * base */
1658 status = lmmc_mat_mul(&base, &base, &temp);
1659 if (status != LMMC_STATUS_OK) {
1660 lmmc_mat_destroy(&temp);
1661 lmmc_mat_destroy(&base);
1662 return status;
1663 }
1664 /* base = temp */
1665 status = lmmc_mat_copy(&temp, &base);
1666 if (status != LMMC_STATUS_OK) {
1667 lmmc_mat_destroy(&temp);
1668 lmmc_mat_destroy(&base);
1669 return status;
1670 }
1671 }
1672 }
1673
1674 lmmc_mat_destroy(&temp);
1675 lmmc_mat_destroy(&base);
1676 return LMMC_STATUS_OK;
1677}

1679lmmc_status_t lmmc_mat_rdiv(const lmmc_mat_t* B, const lmmc_mat_t* A, lmmc_mat_t* X)
1680{
1681 lmmc_mat_t at_mat;
1682 size_t* pivots = NULL;
1683 lmmc_vec_t rhs_vec;
1684 lmmc_vec_t sol_vec;
1685 lmmc_status_t status;
1686 size_t nn, m, i, j;
1687 size_t pivot_bytes;
1688
1689 if (B == NULL || A == NULL || X == NULL ||
1690 B->data == NULL || A->data == NULL || X->data == NULL) {
1691 return LMMC_STATUS_INVALID_ARGUMENT;
1692 }
1693 /* A must be square */
1694 if (A->rows != A->cols) {
1695 return LMMC_STATUS_INVALID_ARGUMENT;
1696 }
1697 nn = A->rows;
1698 m = B->rows;
1699
1700 /* B must have nn columns */
1701 if (B->cols != nn) {
1702 return LMMC_STATUS_INVALID_ARGUMENT;
1703 }
1704 /* X must be m × nn */
1705 if (X->rows != m || X->cols != nn) {
1706 return LMMC_STATUS_INVALID_ARGUMENT;
1707 }
1708
1709 /* Compute A^T into at_mat */
1710 status = lmmc_mat_create(nn, nn, &at_mat);
1711 if (status != LMMC_STATUS_OK) {
1712 return status;
1713 }
1714 status = lmmc_mat_transpose_to(A, &at_mat);
1715 if (status != LMMC_STATUS_OK) {
1716 lmmc_mat_destroy(&at_mat);
1717 return status;
1718 }
1719
1720 /* Allocate pivot array */
1721 if (!lmmc_safe_mul_size(nn, sizeof(size_t), &pivot_bytes)) {
1722 lmmc_mat_destroy(&at_mat);
1723 return LMMC_STATUS_ALLOCATION_FAILED;
1724 }
1725 pivots = (size_t*)lmmc_alloc(pivot_bytes);
1726 if (pivots == NULL) {
1727 lmmc_mat_destroy(&at_mat);
1728 return LMMC_STATUS_ALLOCATION_FAILED;
1729 }
1730
1731 /* LU decompose A^T in-place */
1732 status = lmmc_lu_decompose_inplace(&at_mat, pivots, NULL);
1733 if (status != LMMC_STATUS_OK) {
1734 lmmc_free(pivots);
1735 lmmc_mat_destroy(&at_mat);
1736 return status;
1737 }
1738
1739 /* Create temporary vectors for solving */
1740 rhs_vec.data = NULL;
1741 sol_vec.data = NULL;
1742
1743 status = lmmc_vec_create(nn, &rhs_vec);
1744 if (status != LMMC_STATUS_OK) {
1745 lmmc_free(pivots);
1746 lmmc_mat_destroy(&at_mat);
1747 return status;
1748 }
1749 status = lmmc_vec_create(nn, &sol_vec);
1750 if (status != LMMC_STATUS_OK) {
1751 lmmc_vec_destroy(&rhs_vec);
1752 lmmc_free(pivots);
1753 lmmc_mat_destroy(&at_mat);
1754 return status;
1755 }
1756
1757 /*
1758 * Solve A^T * x_col = b_col for each row i of B.
1759 * Row i of B is column i of B^T, which becomes the RHS.
1760 * The solution x_col becomes row i of X.
1761 */
1762 for (i = 0; i < m; ++i) {
1763 /* Extract row i of B as the RHS vector */
1764 for (j = 0; j < nn; ++j) {
1765 LMMC_REAL_SET(&rhs_vec.data[j], &B->data[i * B->stride + j]);
1766 }
1767
1768 /* Solve A^T * sol = rhs */
1769 status = lmmc_lu_solve(&at_mat, pivots, &rhs_vec, &sol_vec);
1770 if (status != LMMC_STATUS_OK) {
1771 lmmc_vec_destroy(&sol_vec);
1772 lmmc_vec_destroy(&rhs_vec);
1773 lmmc_free(pivots);
1774 lmmc_mat_destroy(&at_mat);
1775 return status;
1776 }
1777
1778 /* Write solution as row i of X */
1779 for (j = 0; j < nn; ++j) {
1780 LMMC_REAL_SET(&X->data[i * X->stride + j], &sol_vec.data[j]);
1781 }
1782 }
1783
1784 lmmc_vec_destroy(&sol_vec);
1785 lmmc_vec_destroy(&rhs_vec);
1786 lmmc_free(pivots);
1787 lmmc_mat_destroy(&at_mat);
1788 return LMMC_STATUS_OK;
1789}

1791lmmc_status_t lmmc_vec_apply(const lmmc_vec_t* in, lmmc_real_t (*func)(lmmc_real_t), lmmc_vec_t* out)
