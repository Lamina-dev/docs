52static lmmc_real_t inverse_matrix_norm(const lmmc_mat_t* A)
53{
54 lmmc_real_t norm_A = 0.0;
55 for (size_t i = 0; i < A->rows; i++) {
56 for (size_t j = 0; j < A->rows; j++) {
57 norm_A += MAT_ELEM(A, i, j) * MAT_ELEM(A, i, j);
58 }
59 }
60 norm_A = sqrt(norm_A);
61 if (norm_A < 1.0) {
62 norm_A = 1.0;
63 }
64 return norm_A;
65}

76static lmmc_status_t factor_real_shift(const lmmc_mat_t* A, lmmc_real_t mu,
77 lmmc_eigen_inverse_workspace_t* work)
78{
79 const lmmc_real_t eps = 2.2204460492503131e-16;
80 lmmc_real_t norm_A = inverse_matrix_norm(A);
81 lmmc_real_t perturb = eps * norm_A * (lmmc_real_t)A->rows;
82 fill_real_shift(A, mu + perturb, &work->shifted);
83 lmmc_status_t status = lmmc_lu_decompose_inplace(&work->shifted, work->pivots, NULL);
84 if (status == LMMC_STATUS_SINGULAR_MATRIX) {
85 perturb = sqrt(eps) * norm_A;
86 fill_real_shift(A, mu + perturb, &work->shifted);
87 status = lmmc_lu_decompose_inplace(&work->shifted, work->pivots, NULL);
88 }
89 return status;

129static void retry_complex_shift(const lmmc_mat_t* A, lmmc_real_t alpha_p,
130 lmmc_real_t beta, lmmc_mat_t* big)
131{
132 size_t n = A->rows;

135 for (j = 0; j < n; j++) {

137 MAT_ELEM(big, i, j) = aij;
138 MAT_ELEM(big, n + i, n + j) = aij;

147static lmmc_status_t factor_complex_shift(const lmmc_mat_t* A, lmmc_real_t alpha,
148 lmmc_real_t beta, lmmc_eigen_inverse_workspace_t* work)
149{
150 const lmmc_real_t eps = 2.2204460492503131e-16;
151 lmmc_real_t norm_A = inverse_matrix_norm(A);
152 lmmc_real_t perturb = eps * norm_A * (lmmc_real_t)A->rows;
153 fill_complex_shift(A, alpha + perturb, beta, &work->shifted);
154 lmmc_status_t status = lmmc_lu_decompose_inplace(&work->shifted, work->pivots, NULL);
155 if (status == LMMC_STATUS_SINGULAR_MATRIX) {
156 retry_complex_shift(A, alpha + sqrt(eps) * norm_A, beta, &work->shifted);
157 status = lmmc_lu_decompose_inplace(&work->shifted, work->pivots, NULL);
158 }
159 return status;

169static void inverse_start_vector(size_t n, lmmc_eigen_inverse_workspace_t* work)
170{
171 for (size_t i = 0; i < n; i++) {
172 work->rhs.data[i] = ((i % 2 == 0) ? 1.0 : -1.0) / (lmmc_real_t)(i + 1);
173 }
174 if (work->rhs.size != n) {
175 for (size_t i = 0; i < n; i++) {
176 work->rhs.data[n + i] = 1.0 / (lmmc_real_t)(n + i + 1);
177 }
178 }
179 lmmc_real_t nrm = inverse_vector_norm(&work->rhs);
180 if (nrm > 0.0) {
181 for (size_t i = 0; i < work->rhs.size; i++) work->rhs.data[i] /= nrm;
182 }

185static lmmc_status_t inverse_iterate(lmmc_eigen_inverse_workspace_t* work)
186{
187 const size_t max_iter = 20;
188 for (size_t iter = 0; iter < max_iter; iter++) {
189 lmmc_status_t status = lmmc_lu_solve(&work->shifted, work->pivots,
190 &work->rhs, &work->solution);
191 if (status != LMMC_STATUS_OK) {
192 return status;
193 }
194 lmmc_real_t nrm = inverse_vector_norm(&work->solution);
195 if (nrm == 0.0) {
196 nrm = 1.0;
197 }
198 for (size_t i = 0; i < work->solution.size; i++) work->solution.data[i] /= nrm;
199 for (size_t i = 0; i < work->solution.size; i++) {
200 work->rhs.data[i] = work->solution.data[i];
201 }
202 }
203 return LMMC_STATUS_OK;
204}

227lmmc_status_t lmmc_eigen_inverse_vectors(const lmmc_mat_t* a,
228 lmmc_eigen_gen_full_result_t* result)
229{
230 size_t n = a->rows;
231 lmmc_eigen_inverse_workspace_t work = {0};
232 lmmc_status_t status = LMMC_STATUS_OK;
233 for (size_t i = 0; i < n;) {
234 int is_real = result->imag_parts.data[i] == 0.0;
235 status = inverse_workspace_resize(&work, is_real ? n : 2 * n);
236 if (status != LMMC_STATUS_OK) {
237 break;
238 }
239 if (is_real) {
240 status = factor_real_shift(a, result->real_parts.data[i], &work);
241 } else {
242 status = factor_complex_shift(a, result->real_parts.data[i],
243 result->imag_parts.data[i], &work);
244 }
245 if (status != LMMC_STATUS_OK) {
246 break;
247 }
248 inverse_start_vector(n, &work);
249 status = inverse_iterate(&work);
250 if (status != LMMC_STATUS_OK) {
251 break;
252 }
253 if (is_real) {
254 store_real_vector(result, i, &work);
255 ++i;
256 } else {
257 store_complex_vectors(result, i, &work);
258 i += 2;
259 }
260 }
261 inverse_workspace_destroy(&work);
262 return status;
263}
