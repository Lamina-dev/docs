14static lmmc_status_t lmmc_jacobi_diagonals(const lmmc_sparse_mat_t* a,
15 lmmc_real_t* inverse) {
16 for (size_t i = 0; i < a->rows; ++i) {
17 size_t pos;
18 if (!lmmc_find_col_pos(a->col_idx, a->row_ptr[i], a->row_ptr[i + 1], i, &pos)) {
19 return LMMC_STATUS_SINGULAR_MATRIX;
20 }
21 lmmc_real_t diag = a->values[pos];
22 if (!isfinite(diag) || fabs(diag) <= 1e-15) {
23 return LMMC_STATUS_SINGULAR_MATRIX;
24 }
25 inverse[i] = 1.0 / diag;
26 if (!isfinite(inverse[i])) {
27 return LMMC_STATUS_NUMERICAL_FAILURE;
28 }
29 }
30 return LMMC_STATUS_OK;
31}

33lmmc_status_t lmmc_precond_create_jacobi(const lmmc_sparse_mat_t* a,
34 lmmc_precond_t* out_precond) {
35 lmmc_status_t st = lmmc_sparse_validate(a);
36 if (out_precond == NULL || st != LMMC_STATUS_OK || a->format != LMMC_SPARSE_CSR) {
37 return LMMC_STATUS_INVALID_ARGUMENT;
38 }
39 if (a->rows != a->cols) {
40 return LMMC_STATUS_DIMENSION_MISMATCH;
41 }
42 if (a->rows > (size_t)-1 / sizeof(lmmc_real_t)) {
43 return LMMC_STATUS_INVALID_ARGUMENT;
44 }
45 lmmc_real_t* inverse = lmmc_memory_alloc(a->rows * sizeof(lmmc_real_t));
46 if (inverse == NULL) {
47 return LMMC_STATUS_ALLOCATION_FAILED;
48 }
49 st = lmmc_jacobi_diagonals(a, inverse);
50 if (st != LMMC_STATUS_OK) {
51 lmmc_memory_free(inverse);
52 return st;
53 }
54 out_precond->type = LMMC_PRECOND_JACOBI;
55 out_precond->size = a->rows;
56 out_precond->impl = inverse;
57 out_precond->owns_data = 1;
58 return LMMC_STATUS_OK;
59}

79static lmmc_status_t lmmc_jacobi_apply(const lmmc_precond_t* precond,
80 const lmmc_vec_t* rhs, lmmc_vec_t* out) {
81 if (precond->impl == NULL) {
82 return LMMC_STATUS_INVALID_ARGUMENT;
83 }
84 const lmmc_real_t* inverse = precond->impl;
85 for (size_t i = 0; i < rhs->size; ++i) {
86 lmmc_real_t value = rhs->data[i] * inverse[i];
87 if (!isfinite(rhs->data[i]) || !isfinite(value)) {
88 return LMMC_STATUS_NUMERICAL_FAILURE;
89 }
90 out->data[i] = value;
91 }
92 return LMMC_STATUS_OK;
93}

95lmmc_status_t lmmc_precond_apply(const lmmc_precond_t* precond,
96 const lmmc_vec_t* rhs, lmmc_vec_t* out) {
97 if (precond == NULL || rhs == NULL || out == NULL || rhs->data == NULL || out->data == NULL) {
98 return LMMC_STATUS_INVALID_ARGUMENT;
99 }
100 if (rhs->size != precond->size || out->size != precond->size) {
101 return LMMC_STATUS_DIMENSION_MISMATCH;
102 }
103 switch (precond->type) {
104 case LMMC_PRECOND_NONE:
105 return lmmc_vec_copy(rhs, out);
106 case LMMC_PRECOND_JACOBI:
107 return lmmc_jacobi_apply(precond, rhs, out);
108 case LMMC_PRECOND_ILU0:
109 case LMMC_PRECOND_ILUT:
110 return lmmc_ilu_apply(precond, rhs, out);
111 default:
112 return LMMC_STATUS_NOT_IMPLEMENTED;
113 }
