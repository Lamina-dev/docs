308int lmmc_lapack_geqrf(int m, int n, double* A, int lda, double* tau)
309{
310 int info = 0;
311 double work_query;
312 int lwork = -1;
313 double* work = NULL;
314
315 /* Validate dimensions */
316 if (m < 0) return -1;
317 if (n < 0) return -2;
318 if (!A && (m > 0 && n > 0)) return -3;
319 if (lda < 1 || (m > 0 && lda < m)) return -4;
320 if (!tau && (m > 0 && n > 0)) return -5;
321 if (m == 0 || n == 0) return 0;
322
323 /* Workspace query */
324 dgeqrf_(&m, &n, A, &lda, tau, &work_query, &lwork, &info);
325 if (info != 0) return info;
326
327 lwork = (int)work_query;
328 work = (double*)malloc((size_t)lwork * sizeof(double));
329 if (!work) return -100; /* allocation failure */
330
331 /* Actual computation */
332 dgeqrf_(&m, &n, A, &lda, tau, work, &lwork, &info);
333 free(work);
334 return info;

338int lmmc_lapack_gesvd(char jobu, char jobvt, int m, int n, double* A, int lda,
339 double* S, double* U, int ldu, double* VT, int ldvt)
340{
341 int info = 0;
342 double work_query;
343 int lwork = -1;
344 double* work = NULL;
345 int minmn = m < n ? m : n;
346
347 /* Validate dimensions */
348 if (jobu != 'A' && jobu != 'a' && jobu != 'S' && jobu != 's' &&
349 jobu != 'O' && jobu != 'o' && jobu != 'N' && jobu != 'n') return -1;
350 if (jobvt != 'A' && jobvt != 'a' && jobvt != 'S' && jobvt != 's' &&
351 jobvt != 'O' && jobvt != 'o' && jobvt != 'N' && jobvt != 'n') return -2;
352 if (m < 0) return -3;
353 if (n < 0) return -4;
354 if (!A && (m > 0 && n > 0)) return -5;
355 if (lda < 1 || (m > 0 && lda < m)) return -6;
356 if (!S && minmn > 0) return -7;
357 /* U validation */
358 if ((jobu == 'A' || jobu == 'a' || jobu == 'S' || jobu == 's') && !U && m > 0)
359 return -8;
360 if (ldu < 1) {
361 if (jobu == 'A' || jobu == 'a' || jobu == 'S' || jobu == 's')
362 return -9;
363 } else if ((jobu == 'A' || jobu == 'a') && ldu < m) {
364 return -9;
365 } else if ((jobu == 'S' || jobu == 's') && ldu < m) {
366 return -9;
367 }
368 /* VT validation */
369 if ((jobvt == 'A' || jobvt == 'a' || jobvt == 'S' || jobvt == 's') && !VT && n > 0)
370 return -10;
371 if (ldvt < 1) {
372 if (jobvt == 'A' || jobvt == 'a' || jobvt == 'S' || jobvt == 's')
373 return -11;
374 } else if ((jobvt == 'A' || jobvt == 'a') && ldvt < n) {
375 return -11;
376 } else if ((jobvt == 'S' || jobvt == 's') && ldvt < minmn) {
377 return -11;
378 }
379 if (m == 0 || n == 0) return 0;
380
381 /* Workspace query */
382 dgesvd_(&jobu, &jobvt, &m, &n, A, &lda, S, U, &ldu, VT, &ldvt,
383 &work_query, &lwork, &info);
384 if (info != 0) return info;
385
386 lwork = (int)work_query;
387 work = (double*)malloc((size_t)lwork * sizeof(double));
388 if (!work) return -100; /* allocation failure */
389
390 /* Actual computation */
391 dgesvd_(&jobu, &jobvt, &m, &n, A, &lda, S, U, &ldu, VT, &ldvt,
392 work, &lwork, &info);
393 free(work);
394 return info;
395}
