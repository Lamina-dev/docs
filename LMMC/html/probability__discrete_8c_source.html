9lmmc_status_t lmmc_dist_binomial_pmf(size_t k, size_t n, lmmc_real_t p,
10 lmmc_real_t* out) {
11 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
12 if (p < 0.0 || p > 1.0) return LMMC_STATUS_INVALID_ARGUMENT;
13 if (k > n) { *out = 0.0; return LMMC_STATUS_OK; }
14
15 /* Use log to avoid overflow: log(C(n,k)) + k*log(p) + (n-k)*log(1-p) */
16 if (p == 0.0) {
17 *out = (k == 0) ? 1.0 : 0.0;
18 return LMMC_STATUS_OK;
19 }
20 if (p == 1.0) {
21 *out = (k == n) ? 1.0 : 0.0;
22 return LMMC_STATUS_OK;
23 }
24
25 {
26 lmmc_real_t log_pmf = lgamma((double)(n + 1)) - lgamma((double)(k + 1)) -
27 lgamma((double)(n - k + 1)) +
28 (double)k * log(p) + (double)(n - k) * log(1.0 - p);
29 *out = exp(log_pmf);
30 }
31 return LMMC_STATUS_OK;
32}

63 lmmc_real_t* out) {
64 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;

68 /* CDF = Q(k+1, lambda) = 1 - P(k+1, lambda) = upper regularized gamma */
