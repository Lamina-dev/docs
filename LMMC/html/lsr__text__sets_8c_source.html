54static lmmc_status_t lmmc_lsr_text_set_alloc(size_t capacity,
55 lmmc_lsr_text_set_t* out)
56{
57 size_t bytes = 0;
58 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
59 out->size = 0;
60 out->data = NULL;
61 out->owns_data = 0;
62 if (capacity == 0) return LMMC_STATUS_OK;
63 if (lmmc_lsr_mul_overflow_size(capacity, sizeof(char*), &bytes)) {
64 return LMMC_STATUS_INVALID_ARGUMENT;
65 }
66 out->data = (char**)lmmc_alloc(bytes);
67 if (!out->data) return LMMC_STATUS_ALLOCATION_FAILED;
68 memset(out->data, 0, bytes);
69 out->owns_data = 1;
70 return LMMC_STATUS_OK;
