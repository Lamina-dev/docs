43static lmmc_real_t lmmc_ilu0_entry_sum(const lmmc_precond_ilu_impl_t* impl,
44 size_t i, size_t p, size_t limit) {
45 const size_t j = impl->col_idx[p];
46 lmmc_real_t sum = impl->lu_values[p];
47 for (size_t q = impl->row_ptr[i]; q < impl->row_ptr[i + 1]; ++q) {
48 size_t k = impl->col_idx[q];
49 size_t pos;
50 if (k >= limit) {
51 continue;
52 }
53 if (lmmc_find_col_pos(impl->col_idx, impl->row_ptr[k],
54 impl->row_ptr[k + 1], j, &pos)) {
55 lmmc_real_t product = impl->lu_values[q] * impl->lu_values[pos];
56 sum = sum - product;
57 }

62static lmmc_status_t lmmc_ilu0_lower(lmmc_precond_ilu_impl_t* impl, size_t i) {
63 for (size_t p = impl->row_ptr[i]; p < impl->row_ptr[i + 1]; ++p) {
64 size_t j = impl->col_idx[p];
65 if (j >= i) {
66 continue;
67 }
68 lmmc_real_t sum = lmmc_ilu0_entry_sum(impl, i, p, j);
69 lmmc_real_t diag = impl->lu_values[impl->diag_pos[j]];
70 if (!isfinite(sum) || !isfinite(diag) || fabs(diag) <= 1e-15) {
71 return LMMC_STATUS_SINGULAR_MATRIX;
72 }
73 impl->lu_values[p] = sum / diag;
74 if (!isfinite(impl->lu_values[p])) {
75 return LMMC_STATUS_NUMERICAL_FAILURE;
76 }
77 }
78 return LMMC_STATUS_OK;
79}
