103{

132static void chol_update_column(const lmmc_sparse_chol_t* chol, size_t j,
133 lmmc_real_t* col_dense)
134{
136 for (size_t k = 0; k < j; k++) {
137 size_t L_start = chol->L_col_ptr[k];
138 size_t L_end = chol->L_col_ptr[k + 1];
139 lmmc_real_t Ljk = chol_column_entry(chol, k, j);
140 if (Ljk == 0.0) {
141 continue;
142 }
143
145 for (size_t p = L_start; p < L_end; p++) {
146 size_t i = chol->L_row_idx[p];
147 if (i >= j) {

154static lmmc_status_t chol_store_column(lmmc_sparse_chol_t* chol, size_t j,
155 const lmmc_real_t* col_dense)
156{
157 size_t n = chol->n;
158 lmmc_status_t status;
159 lmmc_real_t diag = col_dense[j];
160 lmmc_real_t sqrt_diag;
161
162 if (diag <= 0.0) {
163 status = LMMC_STATUS_NOT_POSITIVE_DEFINITE;
164 return status;
165 }
166
167 sqrt_diag = sqrt(diag);
168
169 status = lmmc_sparse_factor_capacity(&chol->L_row_idx, &chol->L_values,
170 &chol->L_capacity, chol->L_nnz);
171 if (status != LMMC_STATUS_OK) {
172 return status;
173 }
174 chol->L_row_idx[chol->L_nnz] = j;
175 chol->L_values[chol->L_nnz] = sqrt_diag;
176 chol->L_nnz++;
177
178 for (size_t i = j + 1; i < n; i++) {
179 if (col_dense[i] != 0.0) {
180 status = lmmc_sparse_factor_capacity(&chol->L_row_idx, &chol->L_values,
181 &chol->L_capacity, chol->L_nnz);
182 if (status != LMMC_STATUS_OK) {
183 return status;
184 }
185 chol->L_row_idx[chol->L_nnz] = i;
186 chol->L_values[chol->L_nnz] = col_dense[i] / sqrt_diag;
187 chol->L_nnz++;
188 }
189 }
190 return LMMC_STATUS_OK;
191}

252static lmmc_status_t chol_forward_solve(const lmmc_sparse_chol_t* chol,
253 lmmc_real_t* work)
254{
255 size_t n = chol->n;
257 for (size_t j = 0; j < n; j++) {
258 size_t L_start = chol->L_col_ptr[j];
259 size_t L_end = chol->L_col_ptr[j + 1];
260 lmmc_real_t diag = 0.0;
261 size_t diag_idx = (size_t)-1;
262
263 for (size_t p = L_start; p < L_end; p++) {
264 if (chol->L_row_idx[p] == j) {
265 diag = chol->L_values[p];
266 diag_idx = p;
267 break;
268 }
269 }
270 if (diag == 0.0) {
271 return LMMC_STATUS_NOT_POSITIVE_DEFINITE;
272 }
273
274 work[j] = work[j] / diag;
275
276 for (size_t p = L_start; p < L_end; p++) {
277 if (p == diag_idx) {
278 continue;
279 }
280 size_t i = chol->L_row_idx[p];
281 work[i] -= chol->L_values[p] * work[j];
282 }
283 }
284 return LMMC_STATUS_OK;
285}

287static lmmc_status_t chol_backward_solve(const lmmc_sparse_chol_t* chol,
288 lmmc_real_t* work)
289{
290 size_t n = chol->n;
292 for (size_t jj = 0; jj < n; jj++) {
293 size_t j = n - 1 - jj;
294 size_t L_start = chol->L_col_ptr[j];
295 size_t L_end = chol->L_col_ptr[j + 1];
296 lmmc_real_t diag = 0.0;
297 size_t diag_idx = (size_t)-1;
298
299 for (size_t p = L_start; p < L_end; p++) {
300 if (chol->L_row_idx[p] == j) {
301 diag = chol->L_values[p];
302 diag_idx = p;
303 break;
304 }
305 }
306 if (diag == 0.0) {
307 return LMMC_STATUS_NOT_POSITIVE_DEFINITE;
308 }
309
310 for (size_t p = L_start; p < L_end; p++) {
311 if (p == diag_idx) {
312 continue;
313 }
314 size_t i = chol->L_row_idx[p];
315 work[j] -= chol->L_values[p] * work[i];
316 }
317 work[j] = work[j] / diag;
318 }
319 return LMMC_STATUS_OK;

353 for (size_t i = 0; i < n; i++) {
354 work[i] = b->data[chol->perm[i]];
355 }
356
357lmmc_status_t status = chol_forward_solve(chol, work);
358if (status == LMMC_STATUS_OK) {
359 status = chol_backward_solve(chol, work);
360}
361if (status != LMMC_STATUS_OK) {
362 lmmc_memory_free(work);
363 return status;
364}
367 for (size_t k = 0; k < n; k++) {
368 x->data[chol->perm[k]] = work[k];
369 }
370
371 lmmc_memory_free(work);
372 return LMMC_STATUS_OK;
