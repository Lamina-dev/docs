56static void fft_bit_reverse(lmmc_real_t* real, lmmc_real_t* imag, size_t n) {
57 size_t i, j, bit;
58 lmmc_real_t tmp;
59
60 j = 0;
61 for (i = 1; i < n; ++i) {
62 bit = n >> 1;
63 while (j & bit) {
64 j ^= bit;
65 bit >>= 1;
66 }
67 j ^= bit;
68
69 if (i < j) {
70 tmp = real[i]; real[i] = real[j]; real[j] = tmp;
71 tmp = imag[i]; imag[i] = imag[j]; imag[j] = tmp;
72 }
73 }
74}

80static lmmc_status_t fft_radix2(lmmc_real_t* real, lmmc_real_t* imag, size_t n, int inverse) {
81 size_t len, i, j;
82 double angle_sign;
83
84 if (n <= 1) return LMMC_STATUS_OK;
85
86 fft_bit_reverse(real, imag, n);
87
88 angle_sign = inverse ? 1.0 : -1.0;
89
90 for (len = 2; len <= n; len <<= 1) {
91 double angle = angle_sign * 2.0 * LMMC_PI / (double)len;
92 double wlen_r = cos(angle);
93 double wlen_i = sin(angle);
94
95 for (i = 0; i < n; i += len) {
96 double w_r = 1.0, w_i = 0.0;
97 size_t half = len >> 1;
98
99 for (j = 0; j < half; ++j) {
100 size_t u_idx = i + j;
101 size_t v_idx = i + j + half;
102
104 double t_r = w_r * real[v_idx] - w_i * imag[v_idx];
105 double t_i = w_r * imag[v_idx] + w_i * real[v_idx];
106
108 real[v_idx] = real[u_idx] - t_r;
109 imag[v_idx] = imag[u_idx] - t_i;
110 real[u_idx] = real[u_idx] + t_r;
111 imag[u_idx] = imag[u_idx] + t_i;
112
114 {
115 double new_w_r = w_r * wlen_r - w_i * wlen_i;
116 double new_w_i = w_r * wlen_i + w_i * wlen_r;
117 w_r = new_w_r;
118 w_i = new_w_i;
119 }
120 }
121 }
122 }
123
124 if (inverse) {
125 double scale = 1.0 / (double)n;
126 for (i = 0; i < n; ++i) {
127 real[i] *= scale;
128 imag[i] *= scale;
129 }
130 }
131
132 return LMMC_STATUS_OK;
133}

145static lmmc_status_t fft_bluestein(lmmc_real_t* real, lmmc_real_t* imag, size_t n, int inverse) {
146 size_t m; /* padded power-of-2 length */
147 size_t k;
148 double sign;
149 lmmc_real_t *a_r = NULL, *a_i = NULL;
150 lmmc_real_t *b_r = NULL, *b_i = NULL;
151 lmmc_status_t st = LMMC_STATUS_OK;
152
153 if (n <= 1) {
154 return LMMC_STATUS_OK;
155 }
156
158 m = fft_next_power_of_two(2 * n - 1);
159
161 a_r = (lmmc_real_t*)lmmc_alloc(m * sizeof(lmmc_real_t));
162 a_i = (lmmc_real_t*)lmmc_alloc(m * sizeof(lmmc_real_t));
163 b_r = (lmmc_real_t*)lmmc_alloc(m * sizeof(lmmc_real_t));
164 b_i = (lmmc_real_t*)lmmc_alloc(m * sizeof(lmmc_real_t));
165
166 if (a_r == NULL || a_i == NULL || b_r == NULL || b_i == NULL) {
167 st = LMMC_STATUS_ALLOCATION_FAILED;
168 goto cleanup;
169 }
170
172 memset(a_r, 0, m * sizeof(lmmc_real_t));
173 memset(a_i, 0, m * sizeof(lmmc_real_t));
174 memset(b_r, 0, m * sizeof(lmmc_real_t));
175 memset(b_i, 0, m * sizeof(lmmc_real_t));
176
180 sign = inverse ? 1.0 : -1.0;
181
183 for (k = 0; k < n; ++k) {
184 double phase = sign * LMMC_PI * (double)(k * k) / (double)n;
185 double c = cos(phase);
186 double s = sin(phase);
188 a_r[k] = real[k] * c - imag[k] * s;
189 a_i[k] = imag[k] * c + real[k] * s;
190 }
191
193 for (k = 0; k < n; ++k) {
194 double phase = -sign * LMMC_PI * (double)(k * k) / (double)n;
195 double c = cos(phase);
196 double s = sin(phase);
197 b_r[k] = c;
198 b_i[k] = s;
199 }
201 for (k = 1; k < n; ++k) {
202 b_r[m - k] = b_r[k];
203 b_i[m - k] = b_i[k];
204 }
205
207 st = fft_radix2(a_r, a_i, m, 0);
208 if (st != LMMC_STATUS_OK) goto cleanup;
209
210 st = fft_radix2(b_r, b_i, m, 0);
211 if (st != LMMC_STATUS_OK) goto cleanup;
212
214 for (k = 0; k < m; ++k) {
215 double tr = a_r[k] * b_r[k] - a_i[k] * b_i[k];
216 double ti = a_r[k] * b_i[k] + a_i[k] * b_r[k];
217 a_r[k] = tr;
218 a_i[k] = ti;
219 }
220
221 /* Inverse FFT of the product */
222 st = fft_radix2(a_r, a_i, m, 1);
223 if (st != LMMC_STATUS_OK) goto cleanup;
224
225 /* Extract result: X[k] = a[k] * exp(sign * i * pi * k^2 / N) */
226 for (k = 0; k < n; ++k) {
227 double phase = sign * LMMC_PI * (double)(k * k) / (double)n;
228 double c = cos(phase);
229 double s = sin(phase);
230 /* result[k] = a[k] * exp(sign*i*pi*k^2/N) = a[k] * (c + i*s) */
231 real[k] = a_r[k] * c - a_i[k] * s;
232 imag[k] = a_i[k] * c + a_r[k] * s;
233 }
234
235 /* For inverse FFT, normalize by 1/N */
236 if (inverse) {
237 double scale = 1.0 / (double)n;
238 for (k = 0; k < n; ++k) {
239 real[k] *= scale;
240 imag[k] *= scale;
241 }
242 }
243
244cleanup:
245 if (a_r) lmmc_free(a_r);
246 if (a_i) lmmc_free(a_i);
247 if (b_r) lmmc_free(b_r);
248 if (b_i) lmmc_free(b_i);
249 return st;
250}

252lmmc_status_t lmmc_fft(lmmc_real_t* real, lmmc_real_t* imag, size_t n, int inverse) {
253 if (real == NULL || imag == NULL || n == 0) {
254 return LMMC_STATUS_INVALID_ARGUMENT;
255 }
256 if (n == 1) {
257 /* DFT of length 1 is identity */
258 return LMMC_STATUS_OK;
259 }
260
261 /* Dispatch: power-of-4 -> radix-4, power-of-2 -> radix-2, else -> Bluestein */
262 if (fft_is_power_of_four(n)) {
263 return lmmc_fft_radix4(real, imag, n, inverse ? 1 : 0);
264 } else if (fft_is_power_of_two(n)) {
265 return fft_radix2(real, imag, n, inverse ? 1 : 0);
266 } else {
267 return fft_bluestein(real, imag, n, inverse ? 1 : 0);
268 }
269}
