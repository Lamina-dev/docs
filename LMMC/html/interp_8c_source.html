14static size_t interp_find_interval(const lmmc_real_t* xs, size_t n,
15 lmmc_real_t query_x)
16{
17 size_t lo = 0, hi = n - 1, mid;
18 while (hi - lo > 1) {
19 mid = lo + (hi - lo) / 2;
20 if (xs[mid] <= query_x) lo = mid;
21 else hi = mid;
22 }
23 return lo;
24}

26static int interp_check_strictly_increasing(const lmmc_real_t* xs, size_t n)
27{
28 size_t i;
29 for (i = 0; i < n - 1; i++) {
30 if (xs[i + 1] <= xs[i]) return 0;
31 }
32 return 1;
33}

35lmmc_status_t lmmc_interp_linear(
36 const lmmc_real_t* xs, const lmmc_real_t* ys, size_t n,
37 lmmc_real_t query_x, lmmc_real_t* out_y)
38{
39 size_t lo;
40 lmmc_real_t t;
41 if (!xs || !ys || !out_y) return LMMC_STATUS_INVALID_ARGUMENT;
42 if (n < 2) return LMMC_STATUS_INVALID_ARGUMENT;
43 if (query_x < xs[0] || query_x > xs[n - 1]) return LMMC_STATUS_OUT_OF_RANGE;
44 if (query_x == xs[n - 1]) { *out_y = ys[n - 1]; return LMMC_STATUS_OK; }
45 lo = interp_find_interval(xs, n, query_x);
46 t = (query_x - xs[lo]) / (xs[lo + 1] - xs[lo]);
47 *out_y = ys[lo] + (ys[lo + 1] - ys[lo]) * t;
48 return LMMC_STATUS_OK;
49}

67lmmc_status_t lmmc_interp_cspline_eval(
68 const lmmc_interp_cspline_t* spline,
69 lmmc_real_t query_x, lmmc_real_t* out_y)
70{
71 size_t seg;
72 lmmc_real_t dx, a, b, c, d;
73 if (!spline || !out_y) return LMMC_STATUS_INVALID_ARGUMENT;
74 if (query_x < spline->xs[0] || query_x > spline->xs[spline->n - 1])
75 return LMMC_STATUS_OUT_OF_RANGE;
76 if (query_x == spline->xs[spline->n - 1]) {
77 *out_y = spline->ys[spline->n - 1];
78 return LMMC_STATUS_OK;
79 }
80 seg = interp_find_interval(spline->xs, spline->n, query_x);
81 dx = query_x - spline->xs[seg];
82 a = spline->coeffs[4 * seg + 0];
83 b = spline->coeffs[4 * seg + 1];
84 c = spline->coeffs[4 * seg + 2];
85 d = spline->coeffs[4 * seg + 3];
86 *out_y = a + dx * (b + dx * (c + dx * d));
87 return LMMC_STATUS_OK;
88}

101static void tridiag_solve(size_t n, lmmc_real_t* sub, lmmc_real_t* dia,
102 lmmc_real_t* sup, lmmc_real_t* rhs)
103{
104 size_t i;
105 lmmc_real_t w;
106 for (i = 1; i < n; i++) {
107 w = sub[i] / dia[i - 1];
108 dia[i] -= w * sup[i - 1];
109 rhs[i] -= w * rhs[i - 1];
110 }
111 rhs[n - 1] /= dia[n - 1];
112 for (i = n - 1; i > 0; i--) {
113 rhs[i - 1] = (rhs[i - 1] - sup[i - 1] * rhs[i]) / dia[i - 1];
114 }
115}

121static void cspline_compute_coeffs(lmmc_interp_cspline_t* spline,
122 const lmmc_real_t* h,
123 const lmmc_real_t* M)
124{
125 size_t i, nm1 = spline->n - 1;
126 for (i = 0; i < nm1; i++) {
127 lmmc_real_t ai = spline->ys[i];
128 lmmc_real_t ci = M[i] / 2.0;
129 lmmc_real_t di = (M[i + 1] - M[i]) / (6.0 * h[i]);
130 lmmc_real_t bi = (spline->ys[i + 1] - spline->ys[i]) / h[i]
131 - h[i] * (2.0 * M[i] + M[i + 1]) / 6.0;
132 spline->coeffs[4 * i + 0] = ai;
133 spline->coeffs[4 * i + 1] = bi;
134 spline->coeffs[4 * i + 2] = ci;
135 spline->coeffs[4 * i + 3] = di;
136 }
137}

139lmmc_status_t lmmc_interp_cspline_create_ex(
140 const lmmc_real_t* xs, const lmmc_real_t* ys, size_t n,
141 lmmc_spline_bc_t bc, lmmc_real_t deriv_left, lmmc_real_t deriv_right,
142 lmmc_interp_cspline_t** out_spline)
143{
144 lmmc_interp_cspline_t* spline = NULL;
145 lmmc_real_t* h = NULL;
146 lmmc_real_t* M = NULL; /* second derivatives (moments) */
147 lmmc_real_t* sub = NULL;
148 lmmc_real_t* dia = NULL;
149 lmmc_real_t* sup = NULL;
150 lmmc_real_t* rhs_arr = NULL;
151 size_t i, nm1, sys_n;
152
153 if (!xs || !ys || !out_spline) return LMMC_STATUS_INVALID_ARGUMENT;
154 if (n < 3) return LMMC_STATUS_INVALID_ARGUMENT;
155 if (!interp_check_strictly_increasing(xs, n))
156 return LMMC_STATUS_INVALID_ARGUMENT;
157
158 /* Periodic BC validation */
159 if (bc == LMMC_SPLINE_PERIODIC) {
160 if (fabs(ys[0] - ys[n - 1]) > 1e-12)
161 return LMMC_STATUS_INVALID_ARGUMENT;
162 }
163
164 nm1 = n - 1;
165
166 /* Allocate spline */
167 spline = (lmmc_interp_cspline_t*)lmmc_alloc(sizeof(*spline));
168 if (!spline) return LMMC_STATUS_ALLOCATION_FAILED;
169 spline->n = n;
170 spline->xs = NULL; spline->ys = NULL; spline->coeffs = NULL;
171
172 spline->xs = (lmmc_real_t*)lmmc_alloc(n * sizeof(lmmc_real_t));
173 spline->ys = (lmmc_real_t*)lmmc_alloc(n * sizeof(lmmc_real_t));
174 spline->coeffs = (lmmc_real_t*)lmmc_alloc(4 * nm1 * sizeof(lmmc_real_t));
175 if (!spline->xs || !spline->ys || !spline->coeffs) {
176 lmmc_interp_cspline_destroy(spline);
177 return LMMC_STATUS_ALLOCATION_FAILED;
178 }
179 memcpy(spline->xs, xs, n * sizeof(lmmc_real_t));
180 memcpy(spline->ys, ys, n * sizeof(lmmc_real_t));
181
182 /* Compute h[i] = xs[i+1] - xs[i] */
183 h = (lmmc_real_t*)lmmc_alloc(nm1 * sizeof(lmmc_real_t));
184 M = (lmmc_real_t*)lmmc_alloc(n * sizeof(lmmc_real_t));
185 if (!h || !M) {
186 lmmc_free(h); lmmc_free(M);
187 lmmc_interp_cspline_destroy(spline);
188 return LMMC_STATUS_ALLOCATION_FAILED;
189 }
190 for (i = 0; i < nm1; i++) h[i] = xs[i + 1] - xs[i];
191 for (i = 0; i < n; i++) M[i] = 0.0;
192
193 if (bc == LMMC_SPLINE_NATURAL || bc == LMMC_SPLINE_CLAMPED) {
194 /* System size = n. Natural: M[0]=M[n-1]=0. Clamped: boundary eqs. */
195 sys_n = n;
196 sub = (lmmc_real_t*)lmmc_alloc(sys_n * sizeof(lmmc_real_t));
197 dia = (lmmc_real_t*)lmmc_alloc(sys_n * sizeof(lmmc_real_t));
198 sup = (lmmc_real_t*)lmmc_alloc(sys_n * sizeof(lmmc_real_t));
199 rhs_arr = (lmmc_real_t*)lmmc_alloc(sys_n * sizeof(lmmc_real_t));
200 if (!sub || !dia || !sup || !rhs_arr) {
201 lmmc_free(sub); lmmc_free(dia); lmmc_free(sup); lmmc_free(rhs_arr);
202 lmmc_free(h); lmmc_free(M);
203 lmmc_interp_cspline_destroy(spline);
204 return LMMC_STATUS_ALLOCATION_FAILED;
205 }
206 for (i = 0; i < sys_n; i++) {
207 sub[i] = dia[i] = sup[i] = rhs_arr[i] = 0.0;
208 }
209
210 /* Interior rows */
211 for (i = 1; i < nm1; i++) {
212 sub[i] = h[i - 1];
213 dia[i] = 2.0 * (h[i - 1] + h[i]);
214 sup[i] = h[i];
215 rhs_arr[i] = 6.0 * ((ys[i + 1] - ys[i]) / h[i]
216 - (ys[i] - ys[i - 1]) / h[i - 1]);
217 }
218
219 if (bc == LMMC_SPLINE_NATURAL) {
220 dia[0] = 1.0; rhs_arr[0] = 0.0;
221 dia[nm1] = 1.0; rhs_arr[nm1] = 0.0;
222 } else { /* CLAMPED */
223 dia[0] = 2.0 * h[0];
224 sup[0] = h[0];
225 rhs_arr[0] = 6.0 * ((ys[1] - ys[0]) / h[0] - deriv_left);
226 sub[nm1] = h[nm1 - 1];
227 dia[nm1] = 2.0 * h[nm1 - 1];
228 rhs_arr[nm1] = 6.0 * (deriv_right - (ys[nm1] - ys[nm1 - 1]) / h[nm1 - 1]);
229 }
230
231 tridiag_solve(sys_n, sub, dia, sup, rhs_arr);
232 for (i = 0; i < n; i++) M[i] = rhs_arr[i];
233
234 lmmc_free(sub); lmmc_free(dia); lmmc_free(sup); lmmc_free(rhs_arr);
235 }
236 else if (bc == LMMC_SPLINE_NOT_A_KNOT) {
237 /* Not-a-knot: third derivative continuous at x[1] and x[n-2].
238 * This means d[0] = d[1] and d[n-3] = d[n-2] where d[i] = (M[i+1]-M[i])/(6*h[i]).
239 * Left: M[0]/h[0] - M[1]*(1/h[0]+1/h[1]) + M[2]/h[1] = 0
240 * => h[1]*M[0] - (h[0]+h[1])*M[1] + h[0]*M[2] = 0
241 * Right: h[n-2]*M[n-3] - (h[n-3]+h[n-2])*M[n-2] + h[n-3]*M[n-1] = 0
242 *
243 * We solve for M[0..n-1] with n equations:
244 * Row 0: not-a-knot left condition
245 * Rows 1..n-2: interior equations
246 * Row n-1: not-a-knot right condition
247 */
248 sys_n = n;
249 sub = (lmmc_real_t*)lmmc_alloc(sys_n * sizeof(lmmc_real_t));
250 dia = (lmmc_real_t*)lmmc_alloc(sys_n * sizeof(lmmc_real_t));
251 sup = (lmmc_real_t*)lmmc_alloc(sys_n * sizeof(lmmc_real_t));
252 rhs_arr = (lmmc_real_t*)lmmc_alloc(sys_n * sizeof(lmmc_real_t));
253 if (!sub || !dia || !sup || !rhs_arr) {
254 lmmc_free(sub); lmmc_free(dia); lmmc_free(sup); lmmc_free(rhs_arr);
255 lmmc_free(h); lmmc_free(M);
256 lmmc_interp_cspline_destroy(spline);
257 return LMMC_STATUS_ALLOCATION_FAILED;
258 }
259 for (i = 0; i < sys_n; i++) {
260 sub[i] = dia[i] = sup[i] = rhs_arr[i] = 0.0;
261 }
262
263 /* Row 0: h[1]*M[0] - (h[0]+h[1])*M[1] + h[0]*M[2] = 0
264 * This is a 3-term equation involving M[0], M[1], M[2].
265 * For tridiagonal, row 0 only has dia[0] and sup[0].
266 * We need to eliminate M[2] from row 0 using row 1.
267 * Alternative: use the condition to express M[0] in terms of M[1],M[2]
268 * and substitute into row 1.
269 *
270 * Better approach: build the full n x n system and solve with
271 * a modified tridiagonal that handles the extra coupling.
272 * Actually, the not-a-knot condition IS tridiagonal if we note:
273 * Row 0: h[1]*M[0] - (h[0]+h[1])*M[1] + h[0]*M[2] = 0
274 * This couples M[0], M[1], M[2] -- not tridiagonal.
275 *
276 * Standard approach: eliminate M[0] and M[n-1] using the not-a-knot
277 * conditions, solve the (n-2) interior system, then recover endpoints.
278 */
279 lmmc_free(sub); lmmc_free(dia); lmmc_free(sup); lmmc_free(rhs_arr);
280
281 /* Solve for M[1..n-2] with modified first and last equations */
282 if (n == 3) {
283 /* Special case: single interior unknown M[1].
284 * Not-a-knot left: h[1]*M[0] - (h[0]+h[1])*M[1] + h[0]*M[2] = 0
285 * Not-a-knot right: same condition for n=3.
286 * Interior: h[0]*M[0] + 2*(h[0]+h[1])*M[1] + h[1]*M[2] = rhs1
287 * From not-a-knot: M[0] = M[2] = ((h[0]+h[1])*M[1]) / h[1]
288 * Wait: h[1]*M[0] - (h[0]+h[1])*M[1] + h[0]*M[2] = 0
289 * For n=3, both not-a-knot conditions are the same.
290 * We have M[0] = ((h[0]+h[1])*M[1] - h[0]*M[2]) / h[1]
291 * and M[2] = ((h[0]+h[1])*M[1] - h[1]*M[0]) / h[0]
292 * These together imply M[0] = M[1] = M[2].
293 */
294 lmmc_real_t r1 = 6.0 * ((ys[2] - ys[1]) / h[1] - (ys[1] - ys[0]) / h[0]);
295 /* M[0]=M[1]=M[2]=val: h[0]*val + 2*(h[0]+h[1])*val + h[1]*val = r1
296 * => val*(3*h[0] + 3*h[1]) = r1 => val = r1/(3*(h[0]+h[1])) */
297 lmmc_real_t val = r1 / (3.0 * (h[0] + h[1]));
298 M[0] = M[1] = M[2] = val;
299 } else {
300 /* n >= 4. Solve (n-2) system for M[1]..M[n-2].
301 * From not-a-knot left: M[0] = ((h[0]+h[1])*M[1] - h[0]*M[2]) / h[1]
302 * From not-a-knot right: M[n-1] = ((h[n-3]+h[n-2])*M[n-2] - h[n-2]*M[n-3]) / h[n-3]
303 */
304 size_t sn = n - 2;
305 sub = (lmmc_real_t*)lmmc_alloc(sn * sizeof(lmmc_real_t));
306 dia = (lmmc_real_t*)lmmc_alloc(sn * sizeof(lmmc_real_t));
307 sup = (lmmc_real_t*)lmmc_alloc(sn * sizeof(lmmc_real_t));
308 rhs_arr = (lmmc_real_t*)lmmc_alloc(sn * sizeof(lmmc_real_t));
309 if (!sub || !dia || !sup || !rhs_arr) {
310 lmmc_free(sub); lmmc_free(dia); lmmc_free(sup); lmmc_free(rhs_arr);
311 lmmc_free(h); lmmc_free(M);
312 lmmc_interp_cspline_destroy(spline);
313 return LMMC_STATUS_ALLOCATION_FAILED;
314 }
315
316 /* Fill interior equations. Local index j maps to global M[j+1]. */
317 for (i = 0; i < sn; i++) {
318 size_t gi = i + 1;
319 sub[i] = h[gi - 1];
320 dia[i] = 2.0 * (h[gi - 1] + h[gi]);
321 sup[i] = h[gi];
322 rhs_arr[i] = 6.0 * ((ys[gi + 1] - ys[gi]) / h[gi]
323 - (ys[gi] - ys[gi - 1]) / h[gi - 1]);
324 }
325
326 /* Modify first equation (i=0, gi=1):
327 * Original: h[0]*M[0] + 2*(h[0]+h[1])*M[1] + h[1]*M[2] = rhs
328 * Substitute M[0] = ((h[0]+h[1])*M[1] - h[0]*M[2]) / h[1]:
329 * h[0]*((h[0]+h[1])*M[1] - h[0]*M[2])/h[1] + 2*(h[0]+h[1])*M[1] + h[1]*M[2] = rhs
330 * => M[1]*(h[0]*(h[0]+h[1])/h[1] + 2*(h[0]+h[1]))
331 * + M[2]*(h[1] - h[0]*h[0]/h[1]) = rhs
332 */
333 dia[0] = h[0] * (h[0] + h[1]) / h[1] + 2.0 * (h[0] + h[1]);
334 sup[0] = h[1] - h[0] * h[0] / h[1];
335 sub[0] = 0.0; /* no sub-diagonal for first row */
336
337 /* Modify last equation (i=sn-1, gi=n-2):
338 * Original: h[n-3]*M[n-3] + 2*(h[n-3]+h[n-2])*M[n-2] + h[n-2]*M[n-1] = rhs
339 * Substitute M[n-1] = ((h[n-3]+h[n-2])*M[n-2] - h[n-2]*M[n-3]) / h[n-3]:
340 * h[n-3]*M[n-3] + 2*(h[n-3]+h[n-2])*M[n-2]
341 * + h[n-2]*((h[n-3]+h[n-2])*M[n-2] - h[n-2]*M[n-3])/h[n-3] = rhs
342 * => M[n-3]*(h[n-3] - h[n-2]*h[n-2]/h[n-3])
343 * + M[n-2]*(2*(h[n-3]+h[n-2]) + h[n-2]*(h[n-3]+h[n-2])/h[n-3]) = rhs
344 */
345 {
346 size_t last = sn - 1;
347 size_t gi_last = last + 1; /* n-2 */
348 lmmc_real_t ha = h[gi_last - 1]; /* h[n-3] */
349 lmmc_real_t hb = h[gi_last]; /* h[n-2] */
350 sub[last] = ha - hb * hb / ha;
351 dia[last] = 2.0 * (ha + hb) + hb * (ha + hb) / ha;
352 sup[last] = 0.0; /* no super-diagonal for last row */
353 }
354
355 tridiag_solve(sn, sub, dia, sup, rhs_arr);
356 for (i = 0; i < sn; i++) M[i + 1] = rhs_arr[i];
357
358 /* Recover M[0] and M[n-1] */
359 M[0] = ((h[0] + h[1]) * M[1] - h[0] * M[2]) / h[1];
360 M[nm1] = ((h[nm1 - 2] + h[nm1 - 1]) * M[nm1 - 1]
361 - h[nm1 - 1] * M[nm1 - 2]) / h[nm1 - 2];
362
363 lmmc_free(sub); lmmc_free(dia); lmmc_free(sup); lmmc_free(rhs_arr);
364 }
365 }
366 else if (bc == LMMC_SPLINE_PERIODIC) {
367 /* Periodic: M[0] = M[n-1]. Solve cyclic tridiagonal for M[0..n-2].
368 * Use Sherman-Morrison formula.
369 * System rows for i=0..n-2:
370 * Row i (1<=i<=n-3): h[i-1]*M[i-1] + 2*(h[i-1]+h[i])*M[i] + h[i]*M[i+1] = rhs_i
371 * Row 0: h[n-2]*M[n-2] + 2*(h[n-2]+h[0])*M[0] + h[0]*M[1] = rhs_0
372 * where rhs_0 = 6*((y1-y0)/h0 - (y0-y[n-2])/h[n-2])
373 * (using y[n-1]=y[0])
374 * Row n-2: h[n-3]*M[n-3] + 2*(h[n-3]+h[n-2])*M[n-2] + h[n-2]*M[0] = rhs_{n-2}
375 */
376 size_t sn = nm1; /* n-1 unknowns: M[0]..M[n-2] */
377 lmmc_real_t* a_sub = NULL;
378 lmmc_real_t* a_dia = NULL;
379 lmmc_real_t* a_sup = NULL;
380 lmmc_real_t* b_vec = NULL;
381 lmmc_real_t* u_vec = NULL;
382 lmmc_real_t* y_sol = NULL;
383 lmmc_real_t* z_sol = NULL;
384 lmmc_real_t gamma_val, alpha_corner, beta_corner;
385 lmmc_real_t vy, vz;
386 size_t j;
387
388 a_sub = (lmmc_real_t*)lmmc_alloc(sn * sizeof(lmmc_real_t));
389 a_dia = (lmmc_real_t*)lmmc_alloc(sn * sizeof(lmmc_real_t));
390 a_sup = (lmmc_real_t*)lmmc_alloc(sn * sizeof(lmmc_real_t));
391 b_vec = (lmmc_real_t*)lmmc_alloc(sn * sizeof(lmmc_real_t));
392 u_vec = (lmmc_real_t*)lmmc_alloc(sn * sizeof(lmmc_real_t));
393 y_sol = (lmmc_real_t*)lmmc_alloc(sn * sizeof(lmmc_real_t));
394 z_sol = (lmmc_real_t*)lmmc_alloc(sn * sizeof(lmmc_real_t));
395 if (!a_sub || !a_dia || !a_sup || !b_vec || !u_vec || !y_sol || !z_sol) {
396 lmmc_free(a_sub); lmmc_free(a_dia); lmmc_free(a_sup);
397 lmmc_free(b_vec); lmmc_free(u_vec); lmmc_free(y_sol); lmmc_free(z_sol);
398 lmmc_free(h); lmmc_free(M);
399 lmmc_interp_cspline_destroy(spline);
400 return LMMC_STATUS_ALLOCATION_FAILED;
401 }
402
403 /* Build the cyclic system */
404 /* Row 0 */
405 a_dia[0] = 2.0 * (h[nm1 - 1] + h[0]);
406 a_sup[0] = h[0];
407 a_sub[0] = h[nm1 - 1]; /* cyclic corner: connects to M[n-2] */
408 b_vec[0] = 6.0 * ((ys[1] - ys[0]) / h[0]
409 - (ys[nm1] - ys[nm1 - 1]) / h[nm1 - 1]);
410 /* Note: ys[nm1] = ys[n-1] = ys[0] for periodic */
411
412 /* Interior rows 1..sn-2 */
413 for (j = 1; j < sn - 1; j++) {
414 a_sub[j] = h[j - 1];
415 a_dia[j] = 2.0 * (h[j - 1] + h[j]);
416 a_sup[j] = h[j];
417 b_vec[j] = 6.0 * ((ys[j + 1] - ys[j]) / h[j]
418 - (ys[j] - ys[j - 1]) / h[j - 1]);
419 }
420
421 /* Last row (j = sn-1 = n-2) */
422 j = sn - 1;
423 a_sub[j] = h[j - 1];
424 a_dia[j] = 2.0 * (h[j - 1] + h[j]);
425 a_sup[j] = h[j]; /* cyclic corner: connects to M[0] */
426 b_vec[j] = 6.0 * ((ys[j + 1] - ys[j]) / h[j]
427 - (ys[j] - ys[j - 1]) / h[j - 1]);
428
429 /* Sherman-Morrison: A = T + u*v^T where
430 * alpha_corner = a_sub[0] (top-right corner)
431 * beta_corner = a_sup[sn-1] (bottom-left corner)
432 * gamma = -a_dia[0]
433 * u = [gamma, 0, ..., 0, beta_corner]^T
434 * v = [1, 0, ..., 0, alpha_corner/gamma]^T
435 * Modified T: T[0][0] -= gamma, T[sn-1][sn-1] -= beta_corner*alpha_corner/gamma
436 */
437 alpha_corner = a_sub[0]; /* h[n-2] */
438 beta_corner = a_sup[sn - 1]; /* h[n-2] */
439 gamma_val = -a_dia[0];
440
441 /* Modify diagonal to form T */
442 a_dia[0] -= gamma_val; /* a_dia[0] = 2*original */
443 a_dia[sn - 1] -= beta_corner * alpha_corner / gamma_val;
444
445 /* Zero out corners in T */
446 a_sub[0] = 0.0;
447 a_sup[sn - 1] = 0.0;
448
449 /* u vector */
450 for (j = 0; j < sn; j++) u_vec[j] = 0.0;
451 u_vec[0] = gamma_val;
452 u_vec[sn - 1] = beta_corner;
453
454 /* Solve T*y = b */
455 memcpy(y_sol, b_vec, sn * sizeof(lmmc_real_t));
456 {
457 /* Need copies of a_sub, a_dia, a_sup for Thomas algorithm */
458 lmmc_real_t* ts = (lmmc_real_t*)lmmc_alloc(sn * sizeof(lmmc_real_t));
459 lmmc_real_t* td = (lmmc_real_t*)lmmc_alloc(sn * sizeof(lmmc_real_t));
460 lmmc_real_t* tu = (lmmc_real_t*)lmmc_alloc(sn * sizeof(lmmc_real_t));
461 if (!ts || !td || !tu) {
462 lmmc_free(ts); lmmc_free(td); lmmc_free(tu);
463 lmmc_free(a_sub); lmmc_free(a_dia); lmmc_free(a_sup);
464 lmmc_free(b_vec); lmmc_free(u_vec); lmmc_free(y_sol); lmmc_free(z_sol);
465 lmmc_free(h); lmmc_free(M);
466 lmmc_interp_cspline_destroy(spline);
467 return LMMC_STATUS_ALLOCATION_FAILED;
468 }
469 memcpy(ts, a_sub, sn * sizeof(lmmc_real_t));
470 memcpy(td, a_dia, sn * sizeof(lmmc_real_t));
471 memcpy(tu, a_sup, sn * sizeof(lmmc_real_t));
472 tridiag_solve(sn, ts, td, tu, y_sol);
473
474 /* Solve T*z = u */
475 memcpy(ts, a_sub, sn * sizeof(lmmc_real_t));
476 memcpy(td, a_dia, sn * sizeof(lmmc_real_t));
477 memcpy(tu, a_sup, sn * sizeof(lmmc_real_t));
478 memcpy(z_sol, u_vec, sn * sizeof(lmmc_real_t));
479 tridiag_solve(sn, ts, td, tu, z_sol);
480
481 lmmc_free(ts); lmmc_free(td); lmmc_free(tu);
482 }
483
484 /* x = y - z * (v^T * y) / (1 + v^T * z)
485 * v = [1, 0, ..., 0, alpha_corner/gamma]^T
486 */
487 vy = y_sol[0] + (alpha_corner / gamma_val) * y_sol[sn - 1];
488 vz = z_sol[0] + (alpha_corner / gamma_val) * z_sol[sn - 1];
489
490 for (j = 0; j < sn; j++) {
491 M[j] = y_sol[j] - z_sol[j] * vy / (1.0 + vz);
492 }
493 M[nm1] = M[0]; /* periodic: M[n-1] = M[0] */
494
495 lmmc_free(a_sub); lmmc_free(a_dia); lmmc_free(a_sup);
496 lmmc_free(b_vec); lmmc_free(u_vec); lmmc_free(y_sol); lmmc_free(z_sol);
497 }
498
499 /* Compute polynomial coefficients from moments */
500 cspline_compute_coeffs(spline, h, M);
501
502 lmmc_free(h);
503 lmmc_free(M);
504
505 *out_spline = spline;
506 return LMMC_STATUS_OK;
507}

520 lmmc_real_t* ys;

533lmmc_status_t lmmc_interp_pchip_create(
534 const lmmc_real_t* xs, const lmmc_real_t* ys, size_t n,
535 lmmc_interp_pchip_t** out)
536{
537 lmmc_interp_pchip_t* p = NULL;
538 lmmc_real_t* delta = NULL; /* slopes of each segment */
539 size_t i;
540
541 if (!xs || !ys || !out) return LMMC_STATUS_INVALID_ARGUMENT;
542 if (n < 2) return LMMC_STATUS_INVALID_ARGUMENT;
543 if (!interp_check_strictly_increasing(xs, n))
544 return LMMC_STATUS_INVALID_ARGUMENT;
545
546 p = (lmmc_interp_pchip_t*)lmmc_alloc(sizeof(*p));
547 if (!p) return LMMC_STATUS_ALLOCATION_FAILED;
548 p->n = n; p->xs = NULL; p->ys = NULL; p->d = NULL;
549
550 p->xs = (lmmc_real_t*)lmmc_alloc(n * sizeof(lmmc_real_t));
551 p->ys = (lmmc_real_t*)lmmc_alloc(n * sizeof(lmmc_real_t));
552 p->d = (lmmc_real_t*)lmmc_alloc(n * sizeof(lmmc_real_t));
553 delta = (lmmc_real_t*)lmmc_alloc((n - 1) * sizeof(lmmc_real_t));
554 if (!p->xs || !p->ys || !p->d || !delta) {
555 lmmc_free(delta);
556 lmmc_interp_pchip_destroy(p);
557 return LMMC_STATUS_ALLOCATION_FAILED;
558 }
559 memcpy(p->xs, xs, n * sizeof(lmmc_real_t));
560 memcpy(p->ys, ys, n * sizeof(lmmc_real_t));
561
562 /* Compute segment slopes */
563 for (i = 0; i < n - 1; i++) {
564 delta[i] = (ys[i + 1] - ys[i]) / (xs[i + 1] - xs[i]);
565 }
566
567 if (n == 2) {
568 /* Only one segment: use the segment slope */
569 p->d[0] = delta[0];
570 p->d[1] = delta[0];
571 } else {
572 /* Fritsch-Carlson method for interior points */
573 for (i = 1; i < n - 1; i++) {
574 if (delta[i - 1] * delta[i] <= 0.0) {
575 /* Sign change or zero: set derivative to zero for monotonicity */
576 p->d[i] = 0.0;
577 } else {
578 /* Harmonic mean weighted by interval widths */
579 lmmc_real_t h1 = xs[i] - xs[i - 1];
580 lmmc_real_t h2 = xs[i + 1] - xs[i];
581 lmmc_real_t w1 = 2.0 * h2 + h1;
582 lmmc_real_t w2 = h2 + 2.0 * h1;
583 p->d[i] = (w1 + w2) / (w1 / delta[i - 1] + w2 / delta[i]);
584 }
585 }
586
587 /* Endpoint derivatives: one-sided shape-preserving */
588 /* Left endpoint */
589 p->d[0] = ((2.0 * (xs[1] - xs[0]) + (xs[2] - xs[1])) * delta[0]
590 - (xs[1] - xs[0]) * delta[1]) / (xs[2] - xs[0]);
591 if (p->d[0] * delta[0] <= 0.0) {
592 p->d[0] = 0.0;
593 } else if (delta[0] * delta[1] <= 0.0 &&
594 fabs(p->d[0]) > 3.0 * fabs(delta[0])) {
595 p->d[0] = 3.0 * delta[0];
596 }
597
598 /* Right endpoint */
599 p->d[n - 1] = ((2.0 * (xs[n - 1] - xs[n - 2]) + (xs[n - 2] - xs[n - 3])) * delta[n - 2]
600 - (xs[n - 1] - xs[n - 2]) * delta[n - 3]) / (xs[n - 1] - xs[n - 3]);
601 if (p->d[n - 1] * delta[n - 2] <= 0.0) {
602 p->d[n - 1] = 0.0;
603 } else if (delta[n - 2] * delta[n - 3] <= 0.0 &&
604 fabs(p->d[n - 1]) > 3.0 * fabs(delta[n - 2])) {
605 p->d[n - 1] = 3.0 * delta[n - 2];
606 }
607 }
608
609 lmmc_free(delta);
610 *out = p;
611 return LMMC_STATUS_OK;
612}
613
614lmmc_status_t lmmc_interp_pchip_eval(
615 const lmmc_interp_pchip_t* p, lmmc_real_t x, lmmc_real_t* out_y)
616{
617 size_t seg;
618 lmmc_real_t h, t, a, b;
619 if (!p || !out_y) return LMMC_STATUS_INVALID_ARGUMENT;
620 if (x < p->xs[0] || x > p->xs[p->n - 1]) return LMMC_STATUS_OUT_OF_RANGE;
621 if (x == p->xs[p->n - 1]) { *out_y = p->ys[p->n - 1]; return LMMC_STATUS_OK; }
622
623 seg = interp_find_interval(p->xs, p->n, x);
624 h = p->xs[seg + 1] - p->xs[seg];
625 t = (x - p->xs[seg]) / h;
626
627 /* Hermite basis evaluation:
628 * p(t) = (1-t)^2*(1+2t)*y0 + t^2*(3-2t)*y1
629 * + t*(1-t)^2*h*d0 - t^2*(1-t)*h*d1
630 */
631 a = (1.0 - t);
632 b = t;
633 *out_y = a * a * (1.0 + 2.0 * t) * p->ys[seg]
634 + b * b * (3.0 - 2.0 * t) * p->ys[seg + 1]
635 + t * a * a * h * p->d[seg]
636 - b * b * a * h * p->d[seg + 1];
637 return LMMC_STATUS_OK;
638}

656lmmc_status_t lmmc_interp_akima_create(
657 const lmmc_real_t* xs, const lmmc_real_t* ys, size_t n,
658 lmmc_interp_akima_t** out)
659{
660 lmmc_interp_akima_t* a = NULL;
661 lmmc_real_t* m = NULL; /* extended slopes: m[-2]..m[n] stored as m[0..n+3] */
662 size_t i, nm1;
663
664 if (!xs || !ys || !out) return LMMC_STATUS_INVALID_ARGUMENT;
665 if (n < 5) return LMMC_STATUS_INVALID_ARGUMENT;
666 if (!interp_check_strictly_increasing(xs, n))
667 return LMMC_STATUS_INVALID_ARGUMENT;
668
669 nm1 = n - 1;
670
671 a = (lmmc_interp_akima_t*)lmmc_alloc(sizeof(*a));
672 if (!a) return LMMC_STATUS_ALLOCATION_FAILED;
673 a->n = n; a->xs = NULL; a->ys = NULL; a->d = NULL;
674
675 a->xs = (lmmc_real_t*)lmmc_alloc(n * sizeof(lmmc_real_t));
676 a->ys = (lmmc_real_t*)lmmc_alloc(n * sizeof(lmmc_real_t));
677 a->d = (lmmc_real_t*)lmmc_alloc(n * sizeof(lmmc_real_t));
678 /* Extended slopes: indices 0..n+2 map to m[-2]..m[n] */
679 m = (lmmc_real_t*)lmmc_alloc((n + 3) * sizeof(lmmc_real_t));
680 if (!a->xs || !a->ys || !a->d || !m) {
681 lmmc_free(m);
682 lmmc_interp_akima_destroy(a);
683 return LMMC_STATUS_ALLOCATION_FAILED;
684 }
685 memcpy(a->xs, xs, n * sizeof(lmmc_real_t));
686 memcpy(a->ys, ys, n * sizeof(lmmc_real_t));
687
688 /* Compute interior slopes m[i] = (y[i+1]-y[i])/(x[i+1]-x[i]) for i=0..n-2
689 * Stored at index i+2 in the extended array (offset by 2). */
690 for (i = 0; i < nm1; i++) {
691 m[i + 2] = (ys[i + 1] - ys[i]) / (xs[i + 1] - xs[i]);
692 }
693
694 /* Extend slopes at boundaries using Akima's extrapolation:
695 * m[-1] = 2*m[0] - m[1]
696 * m[-2] = 2*m[-1] - m[0]
697 * m[n-1] = 2*m[n-2] - m[n-3]
698 * m[n] = 2*m[n-1] - m[n-2]
699 */
700 m[1] = 2.0 * m[2] - m[3]; /* m[-1] at index 1 */
701 m[0] = 2.0 * m[1] - m[2]; /* m[-2] at index 0 */
702 m[nm1 + 2] = 2.0 * m[nm1 + 1] - m[nm1]; /* m[n-1] at index n+1 */
703 m[nm1 + 3] = 2.0 * m[nm1 + 2] - m[nm1 + 1]; /* m[n] at index n+2 */
704
705 /* Compute Akima derivatives:
706 * d[i] = (w1*m[i] + w2*m[i-1]) / (w1 + w2)
707 * where w1 = |m[i+1] - m[i]|, w2 = |m[i-1] - m[i-2]|
708 * Using extended array: m_ext[i+2] = m[i]
709 * So for node i: use m_ext[i], m_ext[i+1], m_ext[i+2], m_ext[i+3]
710 * w1 = |m_ext[i+3] - m_ext[i+2]|
711 * w2 = |m_ext[i+1] - m_ext[i]|
712 * d[i] = (w1*m_ext[i+1] + w2*m_ext[i+2]) / (w1+w2)
713 */
714 for (i = 0; i < n; i++) {
715 lmmc_real_t w1 = fabs(m[i + 3] - m[i + 2]);
716 lmmc_real_t w2 = fabs(m[i + 1] - m[i]);
717 if (w1 + w2 < 1e-30) {
718 /* All slopes equal: use average */
719 a->d[i] = 0.5 * (m[i + 1] + m[i + 2]);
720 } else {
721 a->d[i] = (w1 * m[i + 1] + w2 * m[i + 2]) / (w1 + w2);
722 }
723 }
724
725 lmmc_free(m);
726 *out = a;
727 return LMMC_STATUS_OK;
728}
729
730lmmc_status_t lmmc_interp_akima_eval(
731 const lmmc_interp_akima_t* a, lmmc_real_t x, lmmc_real_t* out_y)
732{
733 size_t seg;
734 lmmc_real_t h, t, omt;
735 if (!a || !out_y) return LMMC_STATUS_INVALID_ARGUMENT;
736 if (x < a->xs[0] || x > a->xs[a->n - 1]) return LMMC_STATUS_OUT_OF_RANGE;
737 if (x == a->xs[a->n - 1]) { *out_y = a->ys[a->n - 1]; return LMMC_STATUS_OK; }
738
739 seg = interp_find_interval(a->xs, a->n, x);
740 h = a->xs[seg + 1] - a->xs[seg];
741 t = (x - a->xs[seg]) / h;
742 omt = 1.0 - t;
743
744 /* Hermite basis */
745 *out_y = omt * omt * (1.0 + 2.0 * t) * a->ys[seg]
746 + t * t * (3.0 - 2.0 * t) * a->ys[seg + 1]
747 + t * omt * omt * h * a->d[seg]
748 - t * t * omt * h * a->d[seg + 1];
749 return LMMC_STATUS_OK;
750}

752struct lmmc_interp_lagrange_t {
753 size_t n;
754 lmmc_real_t* xs;
755 lmmc_real_t* ys;
756 lmmc_real_t* weights;
757};
758
759lmmc_status_t lmmc_interp_lagrange_create(
760 const lmmc_real_t* xs, const lmmc_real_t* ys, size_t n,
761 lmmc_interp_lagrange_t** out_lagrange)
762{
763 lmmc_interp_lagrange_t* lag = NULL;
764 size_t i, j, alloc_size;
765
766 if (!xs || !ys || !out_lagrange) return LMMC_STATUS_INVALID_ARGUMENT;
767 if (n < 1) return LMMC_STATUS_INVALID_ARGUMENT;
768
769 lag = (lmmc_interp_lagrange_t*)lmmc_alloc(sizeof(*lag));
770 if (!lag) return LMMC_STATUS_ALLOCATION_FAILED;
771 lag->n = n; lag->xs = NULL; lag->ys = NULL; lag->weights = NULL;
772
773 if (!lmmc_safe_mul_size(n, sizeof(lmmc_real_t), &alloc_size)) {
774 lmmc_free(lag);
775 return LMMC_STATUS_ALLOCATION_FAILED;
776 }
777
778 lag->xs = (lmmc_real_t*)lmmc_alloc(alloc_size);
779 lag->ys = (lmmc_real_t*)lmmc_alloc(alloc_size);
780 lag->weights = (lmmc_real_t*)lmmc_alloc(alloc_size);
781 if (!lag->xs || !lag->ys || !lag->weights) {
782 lmmc_interp_lagrange_destroy(lag);
783 return LMMC_STATUS_ALLOCATION_FAILED;
784 }
785 memcpy(lag->xs, xs, alloc_size);
786 memcpy(lag->ys, ys, alloc_size);
787
788 /* Barycentric weights */
789 for (j = 0; j < n; j++) {
790 lmmc_real_t prod = 1.0;
791 for (i = 0; i < n; i++) {
792 if (i != j) prod *= (xs[j] - xs[i]);
793 }
794 lag->weights[j] = 1.0 / prod;
795 }
796
797 *out_lagrange = lag;
798 return LMMC_STATUS_OK;
799}

801lmmc_status_t lmmc_interp_lagrange_eval(
802 const lmmc_interp_lagrange_t* lagrange,
803 lmmc_real_t query_x, lmmc_real_t* out_y)
804{
805 size_t j;
806 lmmc_real_t numer, denom, diff, term;
807 if (!lagrange || !out_y) return LMMC_STATUS_INVALID_ARGUMENT;
808
809 /* Check if query_x is exactly a node */
810 for (j = 0; j < lagrange->n; j++) {
811 if (query_x == lagrange->xs[j]) {
812 *out_y = lagrange->ys[j];
813 return LMMC_STATUS_OK;
814 }
815 }
816
817 numer = 0.0;
818 denom = 0.0;
819 for (j = 0; j < lagrange->n; j++) {
820 diff = query_x - lagrange->xs[j];
821 term = lagrange->weights[j] / diff;
822 numer += term * lagrange->ys[j];
823 denom += term;
824 }
825 *out_y = numer / denom;
826 return LMMC_STATUS_OK;
827}

832 lmmc_free(lagrange->weights);
833 lmmc_free(lagrange->ys);
834 lmmc_free(lagrange->xs);
835 lmmc_free(lagrange);
836}

838lmmc_status_t lmmc_interp_bilinear(
839 const lmmc_real_t* xs, size_t nx,
840 const lmmc_real_t* ys, size_t ny,
841 const lmmc_real_t* zs,
842 lmmc_real_t qx, lmmc_real_t qy,
843 lmmc_real_t* out_z)
844{
845 size_t ix, iy;
846 lmmc_real_t tx, ty;
847 lmmc_real_t z00, z01, z10, z11;
848
849 if (!xs || !ys || !zs || !out_z) return LMMC_STATUS_INVALID_ARGUMENT;
850 if (nx < 2 || ny < 2) return LMMC_STATUS_INVALID_ARGUMENT;
851 if (!interp_check_strictly_increasing(xs, nx))
852 return LMMC_STATUS_INVALID_ARGUMENT;
853 if (!interp_check_strictly_increasing(ys, ny))
854 return LMMC_STATUS_INVALID_ARGUMENT;
855
856 if (qx < xs[0] || qx > xs[nx - 1]) return LMMC_STATUS_OUT_OF_RANGE;
857 if (qy < ys[0] || qy > ys[ny - 1]) return LMMC_STATUS_OUT_OF_RANGE;
858
859 /* Find intervals */
860 ix = interp_find_interval(xs, nx, qx);
861 iy = interp_find_interval(ys, ny, qy);
862
863 /* Clamp to last valid interval */
864 if (ix >= nx - 1) ix = nx - 2;
865 if (iy >= ny - 1) iy = ny - 2;
866
867 tx = (qx - xs[ix]) / (xs[ix + 1] - xs[ix]);
868 ty = (qy - ys[iy]) / (ys[iy + 1] - ys[iy]);
869
870 /* Grid values: zs[i*ny + j] = f(xs[i], ys[j]) */
871 z00 = zs[ix * ny + iy];
872 z01 = zs[ix * ny + (iy + 1)];
873 z10 = zs[(ix + 1) * ny + iy];
874 z11 = zs[(ix + 1) * ny + (iy + 1)];
875
876 *out_z = (1.0 - tx) * (1.0 - ty) * z00
877 + (1.0 - tx) * ty * z01
878 + tx * (1.0 - ty) * z10
879 + tx * ty * z11;
880 return LMMC_STATUS_OK;
881}

889static lmmc_real_t cubic_interp_1d(lmmc_real_t p0, lmmc_real_t p1,
890 lmmc_real_t p2, lmmc_real_t p3,
891 lmmc_real_t t)
892{
893 /* Catmull-Rom spline */
894 lmmc_real_t t2 = t * t;
895 lmmc_real_t t3 = t2 * t;
896 return 0.5 * ((-p0 + 3.0*p1 - 3.0*p2 + p3) * t3
897 + (2.0*p0 - 5.0*p1 + 4.0*p2 - p3) * t2
898 + (-p0 + p2) * t
899 + 2.0*p1);
900}

902lmmc_status_t lmmc_interp_bicubic(
903 const lmmc_real_t* xs, size_t nx,
904 const lmmc_real_t* ys, size_t ny,
905 const lmmc_real_t* zs,
906 lmmc_real_t qx, lmmc_real_t qy,
907 lmmc_real_t* out_z)
908{
909 size_t ix, iy;
910 lmmc_real_t tx, ty;
911 int i, j;
912 size_t xi, yj;
913 lmmc_real_t col_vals[4];
914 lmmc_real_t row_vals[4];
915
916 if (!xs || !ys || !zs || !out_z) return LMMC_STATUS_INVALID_ARGUMENT;
917 if (nx < 4 || ny < 4) return LMMC_STATUS_INVALID_ARGUMENT;
918 if (!interp_check_strictly_increasing(xs, nx))
919 return LMMC_STATUS_INVALID_ARGUMENT;
920 if (!interp_check_strictly_increasing(ys, ny))
921 return LMMC_STATUS_INVALID_ARGUMENT;
922
923 if (qx < xs[0] || qx > xs[nx - 1]) return LMMC_STATUS_OUT_OF_RANGE;
924 if (qy < ys[0] || qy > ys[ny - 1]) return LMMC_STATUS_OUT_OF_RANGE;
925
926 /* Find intervals */
927 ix = interp_find_interval(xs, nx, qx);
928 iy = interp_find_interval(ys, ny, qy);
929 if (ix >= nx - 1) ix = nx - 2;
930 if (iy >= ny - 1) iy = ny - 2;
931
932 tx = (qx - xs[ix]) / (xs[ix + 1] - xs[ix]);
933 ty = (qy - ys[iy]) / (ys[iy + 1] - ys[iy]);
934
935 /* For bicubic, we need a 4x4 neighborhood.
936 * Center the stencil: use indices ix-1, ix, ix+1, ix+2 (clamped).
937 * Similarly for iy.
938 */
939 /* Interpolate along y for each of the 4 x-rows */
940 for (i = -1; i <= 2; i++) {
941 /* Clamp x index */
942 if (i == -1) {
943 xi = (ix > 0) ? ix - 1 : 0;
944 } else {
945 xi = ix + (size_t)i;
946 if (xi >= nx) xi = nx - 1;
947 }
948
949 /* Get 4 y-values for this x-row */
950 for (j = -1; j <= 2; j++) {
951 if (j == -1) {
952 yj = (iy > 0) ? iy - 1 : 0;
953 } else {
954 yj = iy + (size_t)j;
955 if (yj >= ny) yj = ny - 1;
956 }

963 }

967 return LMMC_STATUS_OK;
