44 lmmc_real_t tenth = 0.1;
45
46 if (out_cfg == NULL || problem_dim == 0 || !lmmc_is_finite(&t_start) ||
47 !lmmc_is_finite(&t_end) || !(t_end > t_start)) {
48 return LMMC_STATUS_INVALID_ARGUMENT;
49 }
50
51 LMMC_REAL_SUB(&span, &t_end, &t_start);
52 LMMC_REAL_DIV(&initial_step, &span, &hundred);
53 if (!lmmc_is_finite(&initial_step) || initial_step <= 0.0) {
54 LMMC_REAL_SET_D(&initial_step, 1e-3);
55 }
56
57 LMMC_REAL_MUL(&min_step, &span, &span_factor);
58 if (!lmmc_is_finite(&min_step) || min_step <= 0.0) {
59 LMMC_REAL_MUL(&min_step, &initial_step, &step_factor);
60 }
61 if (min_step > initial_step) {
62 LMMC_REAL_MUL(&min_step, &initial_step, &tenth);
63 }
64
65 LMMC_REAL_DIV(&max_step, &span, &ten);
66 if (!lmmc_is_finite(&max_step) || max_step <= 0.0) {
67 LMMC_REAL_SET(&max_step, &initial_step);
68 }
69 if (max_step < initial_step) {
70 LMMC_REAL_SET(&max_step, &initial_step);
71 }
72
73 if (problem_dim > ((size_t)-1) / 10000) {
74 max_steps = 100000;
75 } else {
76 max_steps = problem_dim * 10000;
77 if (max_steps < 1000) {
78 max_steps = 1000;
79 }
80 }
81
82 out_cfg->initial_step = initial_step;
83 out_cfg->min_step = min_step;
84 out_cfg->max_step = max_step;
85 out_cfg->abs_tol = LMMC_DEFAULT_ABS_TOL;
86 out_cfg->rel_tol = LMMC_DEFAULT_REL_TOL;
87 out_cfg->max_steps = max_steps;
88 out_cfg->adaptive_step_beta = 0.9;
89 out_cfg->diagnostics = (lmmc_diagnostic_sink_t){0};
90 out_cfg->jacobian = NULL;
91 out_cfg->time_derivative = NULL;
92 return LMMC_STATUS_OK;
93}
