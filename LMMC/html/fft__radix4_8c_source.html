37static void fft_radix4_butterfly(
38 lmmc_real_t* real, lmmc_real_t* imag, size_t quarter,
39 double angle, int inverse)
40{
41 lmmc_real_t ar[4], ai[4];
42 size_t k;
43 ar[0] = real[0];
44 ai[0] = imag[0];
45 for (k = 1; k < 4; ++k) {
46 const double phase = angle * (double)k;
47 const double c = cos(phase);
48 const double s = sin(phase);
49 ar[k] = real[k * quarter] * c - imag[k * quarter] * s;
50 ai[k] = real[k * quarter] * s + imag[k * quarter] * c;
51 }
52
53 real[0] = ((ar[0] + ar[1]) + ar[2]) + ar[3];
54 imag[0] = ((ai[0] + ai[1]) + ai[2]) + ai[3];
55 real[2 * quarter] = ((ar[0] - ar[1]) + ar[2]) - ar[3];
56 imag[2 * quarter] = ((ai[0] - ai[1]) + ai[2]) - ai[3];
57 if (!inverse) {
58 real[quarter] = ((ar[0] + ai[1]) - ar[2]) - ai[3];
59 imag[quarter] = ((ai[0] - ar[1]) - ai[2]) + ar[3];
60 real[3 * quarter] = ((ar[0] - ai[1]) - ar[2]) + ai[3];
61 imag[3 * quarter] = ((ai[0] + ar[1]) - ai[2]) - ar[3];
62 } else {
63 real[quarter] = ((ar[0] - ai[1]) - ar[2]) + ai[3];
64 imag[quarter] = ((ai[0] + ar[1]) - ai[2]) - ar[3];
65 real[3 * quarter] = ((ar[0] + ai[1]) - ar[2]) - ai[3];
66 imag[3 * quarter] = ((ai[0] - ar[1]) - ai[2]) + ar[3];
67 }

71 lmmc_real_t* real, lmmc_real_t* imag, size_t n,
72 size_t group, int inverse)
73{

85static lmmc_status_t lmmc_fft_radix4_core(
86 lmmc_real_t* real, lmmc_real_t* imag, size_t n, int inverse)
87{
88 unsigned digits = 0;
89 size_t i, len;
90 if (!lmmc_is_power_of_four(n, &digits)) {
91 return LMMC_STATUS_INVALID_ARGUMENT;
92 }
93 if (n <= 1) { return LMMC_STATUS_OK; }
94 for (i = 0; i < n; ++i) {
95 const size_t r = lmmc_reverse_base4(i, digits);
96 if (r > i) {
97 lmmc_swap(&real[i], &real[r]);
98 lmmc_swap(&imag[i], &imag[r]);
99 }
100 }
101 for (len = 4; len <= n; len *= 4) {
102 fft_radix4_stage(real, imag, n, len, inverse);
103 if (len == n) { break; }
104 }
105 if (inverse) {
106 const double scale = 1.0 / (double)n;
107 for (i = 0; i < n; ++i) {
108 real[i] *= scale;
109 imag[i] *= scale;
110 }
111 }
112 return LMMC_STATUS_OK;
113}
