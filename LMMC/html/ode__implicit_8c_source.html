31static lmmc_status_t implicit_euler_F(
32 const lmmc_vec_t* x, lmmc_vec_t* F, void* ud
33) {
34 ode_implicit_euler_ctx_t* ctx = (ode_implicit_euler_ctx_t*)ud;
35 size_t i;
36 lmmc_status_t st = lmmc_ode_rhs_eval(
37 ctx->rhs, ctx->t_next, x->data, F->data, ctx->dim, ctx->user_data,
38 ctx->rhs_eval_count, &ctx->callback_failed);
39 if (st != LMMC_STATUS_OK) return st;
40 /* G(z) = z - y_n - h*f(t_{n+1}, z) */
41 for (i = 0; i < ctx->dim; ++i) {
42 F->data[i] = x->data[i] - ctx->y_n[i] - ctx->h * F->data[i];
43 }
44 return LMMC_STATUS_OK;
45}

47static lmmc_status_t implicit_euler_J(
48 const lmmc_vec_t* x, lmmc_mat_t* J, void* ud
49) {
50 ode_implicit_euler_ctx_t* ctx = (ode_implicit_euler_ctx_t*)ud;
51 size_t i, j, n = ctx->dim;
52 lmmc_real_t* jac_data = J->data;
53
54 {
55 lmmc_status_t st = lmmc_ode_jacobian_eval(
56 ctx->rhs, ctx->jac_cb, ctx->user_data, ctx->t_next, x->data,
57 NULL, n, jac_data, ctx->jac_base_rhs, ctx->jac_y_perturbed,
58 ctx->jac_rhs_perturbed, ctx->rhs_eval_count,
59 &ctx->callback_failed);
60 if (st != LMMC_STATUS_OK) return st;
61 }
62 /* J_G = I - h * J_f */
63 for (i = 0; i < n; ++i) {
64 for (j = 0; j < n; ++j) {
65 jac_data[i * J->stride + j] = -ctx->h * jac_data[i * J->stride + j];
66 }
67 jac_data[i * J->stride + i] += 1.0;
68 }
69 return LMMC_STATUS_OK;
70}

72lmmc_status_t lmmc_ode_implicit_euler_solve(
73 lmmc_ode_rhs_t rhs,
74 void* user_data,
75 size_t dim,
76 lmmc_real_t t_start,
77 lmmc_real_t t_end,
78 lmmc_real_t* y,
79 const lmmc_ode_config_t* cfg,
80 lmmc_ode_result_t* out_result
81) {
82 lmmc_ode_config_t local_cfg = {0};
83 size_t work_bytes = 0;
84 size_t total_work_bytes = 0;
85 lmmc_real_t t = t_start;
86 lmmc_real_t h = 0.0;
87 lmmc_status_t init_st;
88 lmmc_vec_t x_vec = {0};
89 lmmc_optimize_config_t opt_cfg;
90 lmmc_optimize_result_t opt_res;
91 lmmc_real_t* work = NULL;
92 ode_implicit_euler_ctx_t ctx;
93
94 init_st = validate_and_init_ode_config(rhs, dim, t_start, t_end, y,
95 cfg, &local_cfg, out_result, &work_bytes);
96 if (init_st != LMMC_STATUS_OK) return init_st;
97
98 h = lmmc_clamp(local_cfg.initial_step, local_cfg.min_step, local_cfg.max_step);
99 { lmmc_real_t span = t_end - t_start; if (h > span) h = span; }
100
101 /* Setup Newton config */
102 lmmc_optimize_default_config(&opt_cfg);
103 opt_cfg.abs_tol = local_cfg.abs_tol;
104 opt_cfg.rel_tol = local_cfg.rel_tol;
105 opt_cfg.max_iter = 50;
106
107 if (!lmmc_safe_mul_size(work_bytes, 4, &total_work_bytes)) {
108 out_result->failure_reason = LMMC_ODE_FAILURE_INVALID_DIMENSION;
109 return LMMC_STATUS_INVALID_ARGUMENT;
110 }
111 work = (lmmc_real_t*)lmmc_alloc(total_work_bytes);
112 if (work == NULL) {
113 out_result->failure_reason = LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
114 return LMMC_STATUS_ALLOCATION_FAILED;
115 }
116 x_vec.size = dim;
117 x_vec.data = work;
118 x_vec.owns_data = 0;
119
120 ctx.rhs = rhs;
121 ctx.user_data = user_data;
122 ctx.jac_cb = local_cfg.jacobian;
123 ctx.dim = dim;
124
125 ctx.jac_base_rhs = work + dim;
126 ctx.jac_y_perturbed = work + 2 * dim;
127 ctx.jac_rhs_perturbed = work + 3 * dim;
128 ctx.rhs_eval_count = &out_result->num_rhs_evals;
129 ctx.callback_failed = 0;
130 lmmc_ode_do_log(&local_cfg, 0, t, y, dim);
131
132 while (t < t_end && out_result->num_steps < local_cfg.max_steps) {
133 lmmc_real_t rem = t_end - t;
134 lmmc_status_t st;
135 if (rem <= 0.0) break;
136 if (h > rem) h = rem;
137
138 ctx.t_next = t + h;
139 ctx.h = h;
140 ctx.y_n = y;
141
142 /* Initial guess: explicit Euler prediction */
143 memcpy(x_vec.data, y, dim * sizeof(lmmc_real_t));
144
145 st = lmmc_nleq_newton(implicit_euler_F, implicit_euler_J,
146 &ctx, &x_vec, &opt_cfg, &opt_res);
147 if (st != LMMC_STATUS_OK || !opt_res.converged) {
148 out_result->failure_reason =
149 ctx.callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
150 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
151 lmmc_free(work);
152 return st != LMMC_STATUS_OK ? st : LMMC_STATUS_NUMERICAL_FAILURE;
153 }
154
155 memcpy(y, x_vec.data, dim * sizeof(lmmc_real_t));
156
157 if (!lmmc_ode_state_is_finite(y, dim)) {
158 out_result->failure_reason = LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
159 lmmc_free(work);
160 return LMMC_STATUS_NUMERICAL_FAILURE;
161 }
162
163 t += h;
164 out_result->num_steps += 1;
165 out_result->final_t = t;
166 lmmc_ode_do_log(&local_cfg, out_result->num_steps, t, y, dim);
167 }
168
169 if (t >= t_end) {
170 out_result->converged = 1;
171 out_result->failure_reason = LMMC_ODE_FAILURE_NONE;
172 } else {
173 out_result->converged = 0;
174 out_result->failure_reason = LMMC_ODE_FAILURE_MAX_STEPS;
175 }
176
177 lmmc_free(work);
178 return out_result->converged ? LMMC_STATUS_OK : LMMC_STATUS_CONVERGENCE_FAILED;
179}

197static lmmc_status_t trapezoidal_F(
198 const lmmc_vec_t* x, lmmc_vec_t* F, void* ud
199) {
200 ode_trapezoidal_ctx_t* ctx = (ode_trapezoidal_ctx_t*)ud;
201 size_t i;
202 lmmc_status_t st = lmmc_ode_rhs_eval(
203 ctx->rhs, ctx->t_next, x->data, F->data, ctx->dim, ctx->user_data,
204 ctx->rhs_eval_count, &ctx->callback_failed);
205 if (st != LMMC_STATUS_OK) return st;
206 /* G(z) = z - y_n - h/2*(f_n + f(t_{n+1}, z)) */
207 for (i = 0; i < ctx->dim; ++i) {
208 F->data[i] = x->data[i] - ctx->y_n[i]
209 - 0.5 * ctx->h * (ctx->f_n[i] + F->data[i]);
210 }
211 return LMMC_STATUS_OK;
212}

214static lmmc_status_t trapezoidal_J(
215 const lmmc_vec_t* x, lmmc_mat_t* J, void* ud
216) {
217 ode_trapezoidal_ctx_t* ctx = (ode_trapezoidal_ctx_t*)ud;
218 size_t i, j, n = ctx->dim;
219 lmmc_real_t* jac_data = J->data;
220
221 {
222 lmmc_status_t st = lmmc_ode_jacobian_eval(
223 ctx->rhs, ctx->jac_cb, ctx->user_data, ctx->t_next, x->data,
224 NULL, n, jac_data, ctx->jac_base_rhs, ctx->jac_y_perturbed,
225 ctx->jac_rhs_perturbed, ctx->rhs_eval_count,
226 &ctx->callback_failed);
227 if (st != LMMC_STATUS_OK) return st;
228 }
229 /* J_G = I - (h/2) * J_f */
230 for (i = 0; i < n; ++i) {
231 for (j = 0; j < n; ++j) {
232 jac_data[i * J->stride + j] *= -0.5 * ctx->h;
233 }
234 jac_data[i * J->stride + i] += 1.0;
235 }
236 return LMMC_STATUS_OK;
237}
