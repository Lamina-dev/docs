22static const lmmc_real_t sdirk_gamma = 0.25;
23static const lmmc_real_t sdirk_c[SDIRK_STAGES] = {0.25, 0.75, 0.55, 0.5, 1.0};

39static lmmc_status_t sdirk_linear_solve(
40 lmmc_mat_t* matrix,
41 size_t* pivots,
42 lmmc_real_t* rhs,
43 lmmc_real_t* solution,
44 size_t dim
45) {
46 size_t swaps = 0;
47 const lmmc_vec_t rhs_vec = {dim, rhs, 0};
48 lmmc_vec_t solution_vec = {dim, solution, 0};
49 lmmc_status_t st = lmmc_lu_decompose_inplace(matrix, pivots, &swaps);
50 if (st != LMMC_STATUS_OK) return st;
51 st = lmmc_lu_solve(matrix, pivots, &rhs_vec, &solution_vec);
52 if (st != LMMC_STATUS_OK) return st;
53 return lmmc_ode_values_are_finite(solution, dim);

56lmmc_status_t lmmc_ode_sdirk4_solve(
57 lmmc_ode_rhs_t rhs,
58 void* user_data,
59 size_t dim,
60 lmmc_real_t t_start,
61 lmmc_real_t t_end,
62 lmmc_real_t* y,
63 const lmmc_ode_config_t* cfg,
64 lmmc_ode_result_t* out_result
65) {
66 lmmc_ode_config_t local_cfg = {0};
67 size_t work_bytes = 0, vectors_bytes = 0, matrix_count = 0, matrix_bytes = 0, pivot_bytes = 0;
68 lmmc_real_t* work = NULL;
69 lmmc_real_t* matrix_data = NULL;
70 size_t* pivots = NULL;
71 lmmc_real_t* k[SDIRK_STAGES];
72 lmmc_real_t *rhs_sum, *y_stage, *f_stage, *y_pert, *f_pert;
73 lmmc_real_t *residual, *delta, *y_new, *y_hat, *error, *smoothed;
74 lmmc_mat_t matrix = {0};
75 lmmc_real_t t = t_start, h;
76 size_t attempts = 0, i, j, stage;
77 lmmc_status_t st;
78
79 st = validate_and_init_ode_config(rhs, dim, t_start, t_end, y, cfg,
80 &local_cfg, out_result, &work_bytes);
81 if (st != LMMC_STATUS_OK) return st;
82 if (!lmmc_safe_mul_size(work_bytes, SDIRK_VECTORS, &vectors_bytes) ||
83 !lmmc_safe_mul_size(dim, dim, &matrix_count) ||
84 !lmmc_safe_mul_size(matrix_count, sizeof(lmmc_real_t), &matrix_bytes) ||
85 !lmmc_safe_mul_size(dim, sizeof(size_t), &pivot_bytes)) {
86 out_result->failure_reason = LMMC_ODE_FAILURE_INVALID_DIMENSION;
87 return LMMC_STATUS_INVALID_ARGUMENT;
88 }
89
90 work = (lmmc_real_t*)lmmc_alloc(vectors_bytes);
91 matrix_data = (lmmc_real_t*)lmmc_alloc(matrix_bytes);
92 pivots = (size_t*)lmmc_alloc(pivot_bytes);
93 if (work == NULL || matrix_data == NULL || pivots == NULL) {
94 st = LMMC_STATUS_ALLOCATION_FAILED;
95 goto cleanup;
96 }
97 for (stage = 0; stage < SDIRK_STAGES; ++stage) k[stage] = work + stage * dim;
98 rhs_sum = work + 5 * dim;
99 y_stage = work + 6 * dim;
100 f_stage = work + 7 * dim;
101 y_pert = work + 8 * dim;
102 f_pert = work + 9 * dim;
103 residual = work + 10 * dim;
104 delta = work + 11 * dim;
105 y_new = work + 12 * dim;
106 y_hat = work + 13 * dim;
107 error = work + 14 * dim;
108 smoothed = work + 15 * dim;
109 matrix = (lmmc_mat_t){dim, dim, dim, matrix_data, 0};
110
111 h = lmmc_clamp(local_cfg.initial_step, local_cfg.min_step, local_cfg.max_step);
112 if (h > t_end - t_start) h = t_end - t_start;
113 lmmc_ode_do_log(&local_cfg, 0, t, y, dim);
114
115 while (t < t_end && attempts < local_cfg.max_steps) {
116 lmmc_real_t error_norm = 0.0;
117 int callback_failed = 0;
118 ++attempts;
119 if (h > t_end - t) h = t_end - t;
120
121 for (stage = 0; stage < SDIRK_STAGES; ++stage) {
122 size_t iteration;
123 int converged = 0;
124 memset(rhs_sum, 0, work_bytes);
125 for (j = 0; j < stage; ++j) {
126 for (i = 0; i < dim; ++i) {
127 rhs_sum[i] += sdirk_a[stage][j] * k[j][i];
128 }
129 }
130 if (stage == 0) memset(k[stage], 0, work_bytes);
131 else memcpy(k[stage], k[stage - 1], work_bytes);
132
133 for (iteration = 0; iteration < SDIRK_NEWTON_MAX; ++iteration) {
134 lmmc_real_t residual_max = 0.0;
135 lmmc_real_t state_scale = 0.0;
136 for (i = 0; i < dim; ++i) {
137 y_stage[i] = y[i] + h * (rhs_sum[i] + sdirk_gamma * k[stage][i]);
138 state_scale = fmax(state_scale, fabs(k[stage][i]));
139 }
140 st = lmmc_ode_rhs_eval(rhs, t + sdirk_c[stage] * h, y_stage, f_stage,
141 dim, user_data, &out_result->num_rhs_evals,
142 &callback_failed);
143 if (st != LMMC_STATUS_OK) {
144 out_result->failure_reason = callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
145 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
146 goto cleanup;
147 }
148 for (i = 0; i < dim; ++i) {
149 residual[i] = k[stage][i] - f_stage[i];
150 residual_max = fmax(residual_max, fabs(residual[i]));
151 }
152 if (residual_max <= 1.0e-12 * fmax(1.0, state_scale)) {
153 converged = 1;
154 break;
155 }
156
157 st = lmmc_ode_jacobian_eval(
158 rhs, local_cfg.jacobian, user_data,
159 t + sdirk_c[stage] * h, y_stage, f_stage, dim,
160 matrix_data, rhs_sum, y_pert, f_pert,
161 &out_result->num_rhs_evals, &callback_failed);
162 if (st != LMMC_STATUS_OK) {
163 out_result->failure_reason =
164 callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
165 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
166 goto cleanup;
167 }
168 for (i = 0; i < dim; ++i) {
169 for (j = 0; j < dim; ++j) {
170 matrix.data[i * dim + j] = -h * sdirk_gamma * matrix_data[i * dim + j];
171 }
172 matrix.data[i * dim + i] += 1.0;
173 residual[i] = -residual[i];
174 }
175 st = sdirk_linear_solve(&matrix, pivots, residual, delta, dim);
176 if (st != LMMC_STATUS_OK) goto cleanup;
177 for (i = 0; i < dim; ++i) k[stage][i] += delta[i];
178 }
179 if (!converged) {
180 st = LMMC_STATUS_CONVERGENCE_FAILED;
181 out_result->failure_reason = LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
182 goto cleanup;
183 }
184 }
185
186 for (i = 0; i < dim; ++i) {
187 y_new[i] = y[i];
188 y_hat[i] = y[i];
189 for (stage = 0; stage < SDIRK_STAGES; ++stage) {
190 y_new[i] += h * sdirk_b[stage] * k[stage][i];
191 y_hat[i] += h * sdirk_bhat[stage] * k[stage][i];
192 }
193 error[i] = y_new[i] - y_hat[i];
194 }
195 if (lmmc_ode_values_are_finite(y_new, dim) != LMMC_STATUS_OK) {
196 st = LMMC_STATUS_NUMERICAL_FAILURE;
197 goto cleanup;
198 }
199
200 st = lmmc_ode_rhs_eval(rhs, t + h, y_new, f_stage, dim, user_data,
201 &out_result->num_rhs_evals, &callback_failed);
202 if (st != LMMC_STATUS_OK) {
203 out_result->failure_reason = callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
204 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
205 goto cleanup;
206 }
207 st = lmmc_ode_jacobian_eval(
208 rhs, local_cfg.jacobian, user_data, t + h, y_new, f_stage, dim,
209 matrix_data, rhs_sum, y_pert, f_pert,
210 &out_result->num_rhs_evals, &callback_failed);
211 if (st != LMMC_STATUS_OK) {
212 out_result->failure_reason =
213 callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
214 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
215 goto cleanup;
216 }
217 for (i = 0; i < dim; ++i) {
218 for (j = 0; j < dim; ++j) {
219 matrix.data[i * dim + j] = -h * sdirk_gamma * matrix_data[i * dim + j];
220 }
221 matrix.data[i * dim + i] += 1.0;
222 }
223 st = sdirk_linear_solve(&matrix, pivots, error, smoothed, dim);
224 if (st != LMMC_STATUS_OK) goto cleanup;
225 st = lmmc_ode_weighted_rms(smoothed, y, y_new, dim, local_cfg.abs_tol,
226 local_cfg.rel_tol, &error_norm);
227 if (st != LMMC_STATUS_OK) goto cleanup;
228
229 if (error_norm <= 1.0) {
230 memcpy(y, y_new, work_bytes);
231 t += h;
232 out_result->num_steps += 1;
233 out_result->final_t = t;
234 lmmc_ode_do_log(&local_cfg, out_result->num_steps, t, y, dim);
235 } else if (h <= local_cfg.min_step) {
236 st = LMMC_STATUS_CONVERGENCE_FAILED;
237 out_result->failure_reason = LMMC_ODE_FAILURE_INVALID_STEP;
238 goto cleanup;
239 }
240 h = lmmc_ode_next_step(h, error_norm, &local_cfg);
241 }
242
243 if (t >= t_end) {
244 out_result->converged = 1;
245 out_result->failure_reason = LMMC_ODE_FAILURE_NONE;
246 st = LMMC_STATUS_OK;
247 } else {
248 out_result->failure_reason = LMMC_ODE_FAILURE_MAX_STEPS;
249 st = LMMC_STATUS_CONVERGENCE_FAILED;
250 }
251
252cleanup:
253 if (st != LMMC_STATUS_OK && out_result != NULL &&
254 out_result->failure_reason == LMMC_ODE_FAILURE_NONE) {
255 out_result->failure_reason = LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
256 }
257 lmmc_free(pivots);
258 lmmc_free(matrix_data);
259 lmmc_free(work);
260 return st;
261}
