73static lmmc_status_t sdirk_newton_correction(ode_sdirk_state_t* s, size_t stage) {
74 size_t i, j;
75 int callback_failed = 0;
76 lmmc_status_t st;
77 st = lmmc_ode_jacobian_eval(&(lmmc_ode_jacobian_request_t){
78 s->rhs, s->local_cfg.jacobian, s->user_data,
79 s->t + sdirk_c[stage] * s->h, s->y_stage, s->f_stage, s->dim,
80 s->matrix_data, s->rhs_sum, s->y_pert, s->f_pert,
81 &s->out_result->num_rhs_evals, &callback_failed});
82 if (st != LMMC_STATUS_OK) {
83 s->out_result->failure_reason =
84 callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
85 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
86 return st;
87 }
88 for (i = 0; i < s->dim; ++i) {
89 for (j = 0; j < s->dim; ++j) {
90 s->matrix.data[i * s->dim + j] = -s->h * sdirk_gamma * s->matrix_data[i * s->dim + j];
91 }
92 s->matrix.data[i * s->dim + i] += 1.0;
93 s->residual[i] = -s->residual[i];
94 }
95 st = sdirk_linear_solve(&s->matrix, s->pivots, s->residual, s->delta, s->dim);
96 if (st != LMMC_STATUS_OK) { return st; }
97 for (i = 0; i < s->dim; ++i) s->k[stage][i] += s->delta[i];
98 return LMMC_STATUS_OK;
99}

101static lmmc_status_t sdirk_stage_norm(
102 ode_sdirk_state_t* s, const lmmc_real_t* derivative, lmmc_real_t* out_norm)
103{
104 size_t i;
105 int zero_scale = 0;
106 lmmc_status_t st;
107 for (i = 0; i < s->dim; ++i) s->error[i] = s->h * derivative[i];
108 st = lmmc_ode_weighted_rms(s->error, s->y, s->y_stage, s->dim,
109 s->local_cfg.abs_tol, s->local_cfg.rel_tol, out_norm);
110 if (st != LMMC_STATUS_NUMERICAL_FAILURE) return st;
111 for (i = 0; i < s->dim; ++i) {
112 const lmmc_real_t scale = s->local_cfg.abs_tol +
113 s->local_cfg.rel_tol * fmax(fabs(s->y[i]), fabs(s->y_stage[i]));
114 if (!isfinite(scale) || !isfinite(s->error[i])) {
115 return LMMC_STATUS_NUMERICAL_FAILURE;
116 }
117 if (scale == 0.0) zero_scale = 1;
118 }
120 if (zero_scale) {
121 *out_norm = INFINITY;
122 return LMMC_STATUS_OK;
123 }
124 return st;
125}

127static lmmc_status_t sdirk_stage_residual(
128 ode_sdirk_state_t* s, size_t stage, int* residual_zero)
129{
130 size_t i;
131 int callback_failed = 0;
132 lmmc_status_t st;
133 for (i = 0; i < s->dim; ++i) {
134 s->y_stage[i] = s->y[i] + s->h * (s->rhs_sum[i] + sdirk_gamma * s->k[stage][i]);
135 if (!isfinite(s->y_stage[i])) { return LMMC_STATUS_NUMERICAL_FAILURE; }
136 }
137 st = lmmc_ode_rhs_eval(s->rhs, s->t + sdirk_c[stage] * s->h, s->y_stage, s->f_stage,
138 s->dim, s->user_data, &s->out_result->num_rhs_evals, &callback_failed);
139 if (st != LMMC_STATUS_OK) {
140 s->out_result->failure_reason = callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
141 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
142 return st;
143 }
144 *residual_zero = 1;
145 for (i = 0; i < s->dim; ++i) {
146 s->residual[i] = s->k[stage][i] - s->f_stage[i];
147 if (!isfinite(s->residual[i])) { return LMMC_STATUS_NUMERICAL_FAILURE; }
148 if (s->residual[i] != 0.0) { *residual_zero = 0; }
149 }
150 return LMMC_STATUS_OK;
151}

153static lmmc_status_t sdirk_solve_stage(ode_sdirk_state_t* s, size_t stage) {
154 const lmmc_real_t eta = 0.001;
155 size_t i, j, corrections = 0;
156 lmmc_status_t st;
157 memset(s->rhs_sum, 0, s->work_bytes);
158 for (j = 0; j < stage; ++j) {
159 for (i = 0; i < s->dim; ++i) {
160 s->rhs_sum[i] += sdirk_a[stage][j] * s->k[j][i];
161 }
162 }
163 if (stage == 0) { memset(s->k[stage], 0, s->work_bytes); }
164 else { memcpy(s->k[stage], s->k[stage - 1], s->work_bytes); }
165
166 for (;;) {
167 int residual_zero;
168 st = sdirk_stage_residual(s, stage, &residual_zero);
169 if (st != LMMC_STATUS_OK) { return st; }
170 if (residual_zero) { return LMMC_STATUS_OK; }
172 if (corrections != 0) {
173 lmmc_real_t residual_norm, correction_norm;
174 st = sdirk_stage_norm(s, s->residual, &residual_norm);
175 if (st != LMMC_STATUS_OK) { return st; }
176 st = sdirk_stage_norm(s, s->delta, &correction_norm);
177 if (st != LMMC_STATUS_OK) { return st; }
178 if (residual_norm <= eta && correction_norm <= eta) { return LMMC_STATUS_OK; }
179 }
180 if (corrections == SDIRK_NEWTON_MAX) {
181 s->out_result->failure_reason = LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
182 return LMMC_STATUS_CONVERGENCE_FAILED;
183 }
184 st = sdirk_newton_correction(s, stage);
185 if (st != LMMC_STATUS_OK) { return st; }
186 ++corrections;
187 }
188}

190static lmmc_status_t sdirk_estimate_error(ode_sdirk_state_t* s, lmmc_real_t* error_norm) {
191 size_t i, j, stage;
192 int callback_failed = 0;
193 lmmc_status_t st;
194 for (i = 0; i < s->dim; ++i) {
195 s->y_new[i] = s->y[i];
196 s->y_hat[i] = s->y[i];
197 for (stage = 0; stage < SDIRK_STAGES; ++stage) {
198 s->y_new[i] += s->h * sdirk_b[stage] * s->k[stage][i];
199 s->y_hat[i] += s->h * sdirk_bhat[stage] * s->k[stage][i];
200 }
201 s->error[i] = s->y_new[i] - s->y_hat[i];
202 }
203 if (lmmc_ode_values_are_finite(s->y_new, s->dim) != LMMC_STATUS_OK) {
204 st = LMMC_STATUS_NUMERICAL_FAILURE;
205 return st;
206 }
207
208 st = lmmc_ode_rhs_eval(s->rhs, s->t + s->h, s->y_new, s->f_stage, s->dim, s->user_data,
209 &s->out_result->num_rhs_evals, &callback_failed);
210 if (st != LMMC_STATUS_OK) {
211 s->out_result->failure_reason = callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
212 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
213 return st;
214 }
215 st = lmmc_ode_jacobian_eval(&(lmmc_ode_jacobian_request_t){
216 s->rhs, s->local_cfg.jacobian, s->user_data, s->t + s->h, s->y_new, s->f_stage, s->dim,
217 s->matrix_data, s->rhs_sum, s->y_pert, s->f_pert,
218 &s->out_result->num_rhs_evals, &callback_failed});
219 if (st != LMMC_STATUS_OK) {
220 s->out_result->failure_reason =
221 callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
222 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
223 return st;
224 }
225 for (i = 0; i < s->dim; ++i) {
226 for (j = 0; j < s->dim; ++j) {
227 s->matrix.data[i * s->dim + j] = -s->h * sdirk_gamma * s->matrix_data[i * s->dim + j];
228 }
229 s->matrix.data[i * s->dim + i] += 1.0;
230 }
231 st = sdirk_linear_solve(&s->matrix, s->pivots, s->error, s->smoothed, s->dim);
232 if (st != LMMC_STATUS_OK) { return st; }
233 st = lmmc_ode_weighted_rms(s->smoothed, s->y, s->y_new, s->dim, s->local_cfg.abs_tol,
234 s->local_cfg.rel_tol, error_norm);
235 if (st != LMMC_STATUS_OK) { return st; }
236 return LMMC_STATUS_OK;
237}

239static lmmc_status_t sdirk_integrate(ode_sdirk_state_t* s, lmmc_real_t t_end) {
240 size_t attempts = 0;
241 lmmc_status_t st;
242 while (s->t < t_end && attempts < s->local_cfg.max_steps) {
243 lmmc_real_t error_norm = 0.0;
244 ++attempts;
245 if (s->h > t_end - s->t) s->h = t_end - s->t;
246 for (size_t stage = 0; stage < SDIRK_STAGES; ++stage) {
247 st = sdirk_solve_stage(s, stage);
248 if (st != LMMC_STATUS_OK) { return st; }
249 }
250 st = sdirk_estimate_error(s, &error_norm);
251 if (st != LMMC_STATUS_OK) { return st; }
252 if (error_norm <= 1.0) {
253 memcpy(s->y, s->y_new, s->work_bytes);
254 s->t += s->h;
255 s->out_result->num_steps += 1;
256 s->out_result->final_t = s->t;
257 lmmc_ode_do_log(&s->local_cfg, s->out_result->num_steps, s->t, s->y, s->dim);
258 } else if (s->h <= s->local_cfg.min_step) {
259 st = LMMC_STATUS_CONVERGENCE_FAILED;
260 s->out_result->failure_reason = LMMC_ODE_FAILURE_INVALID_STEP;
261 return st;
262 }
263 s->h = lmmc_ode_next_step(s->h, error_norm, &s->local_cfg);
264 }
265
266 if (s->t >= t_end) {
267 s->out_result->converged = 1;
268 s->out_result->failure_reason = LMMC_ODE_FAILURE_NONE;
269 st = LMMC_STATUS_OK;
270 } else {
271 s->out_result->failure_reason = LMMC_ODE_FAILURE_MAX_STEPS;
272 st = LMMC_STATUS_CONVERGENCE_FAILED;
273 }
274 return st;
275}

312 if (!lmmc_safe_mul_size(s->work_bytes, SDIRK_VECTORS, &vectors_bytes) ||

314 !lmmc_safe_mul_size(matrix_count, sizeof(lmmc_real_t), &matrix_bytes) ||
315 !lmmc_safe_mul_size(s->dim, sizeof(size_t), &pivot_bytes)) {
316 s->out_result->failure_reason = LMMC_ODE_FAILURE_INVALID_DIMENSION;
317 return LMMC_STATUS_INVALID_ARGUMENT;
318 }
319
320 s->work = (lmmc_real_t*)lmmc_memory_alloc(vectors_bytes);
321 s->matrix_data = (lmmc_real_t*)lmmc_memory_alloc(matrix_bytes);
322 s->pivots = (size_t*)lmmc_memory_alloc(pivot_bytes);
323 if (s->work == NULL || s->matrix_data == NULL || s->pivots == NULL) {
324 st = LMMC_STATUS_ALLOCATION_FAILED;
325 return sdirk_cleanup(s, st);
326 }
327 for (stage = 0; stage < SDIRK_STAGES; ++stage) s->k[stage] = s->work + stage * s->dim;
328 s->rhs_sum = s->work + 5 * s->dim;
329 s->y_stage = s->work + 6 * s->dim;
330 s->f_stage = s->work + 7 * s->dim;
331 s->y_pert = s->work + 8 * s->dim;
332 s->f_pert = s->work + 9 * s->dim;
333 s->residual = s->work + 10 * s->dim;
334 s->delta = s->work + 11 * s->dim;
335 s->y_new = s->work + 12 * s->dim;
336 s->y_hat = s->work + 13 * s->dim;
337 s->error = s->work + 14 * s->dim;
338 s->smoothed = s->work + 15 * s->dim;
339 s->matrix = (lmmc_mat_t){s->dim, s->dim, s->dim, s->matrix_data, 0};
340
341 s->h = lmmc_clamp(s->local_cfg.initial_step, s->local_cfg.min_step, s->local_cfg.max_step);
342 if (s->h > t_end - t_start) s->h = t_end - t_start;
343 lmmc_ode_do_log(&s->local_cfg, 0, s->t, s->y, s->dim);
344 st = sdirk_integrate(s, t_end);
345 return sdirk_cleanup(s, st);
