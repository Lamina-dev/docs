23static inline double lmmc_two_sum(double a, double b, double* residual)
24{
25 const double sum = a + b;
26 const double virtual_b = sum - a;
27 *residual = (a - (sum - virtual_b)) + (b - virtual_b);

29}

59static inline int lmmc_is_power_of_four(
60 size_t value, unsigned* out_digits)
61{
62 size_t remaining = value;
63 unsigned digits = 0;
64 if (value == 0) {
65 return 0;
66 }
67 while ((remaining & (size_t)3) == 0) {
68 remaining >>= 2;
69 ++digits;
70 }
71 if (remaining != (size_t)1) {
72 return 0;
73 }
74 if (out_digits != NULL) {
75 *out_digits = digits;
76 }
77 return 1;
78}

125static inline int lmmc_storage_envelope_checked(
126 const void *data,
127 size_t rows,
128 size_t cols,
129 size_t stride,
130 size_t element_size,
131 lmmc_storage_envelope_t *out)
132{
133 size_t last_row;
134 size_t elements;
135 size_t bytes;
136 uintptr_t begin;
137
138 if (data == NULL || out == NULL) {
139 return 0;
140 }
141 begin = (uintptr_t)data;
142 out->begin = begin;
143 out->end = begin;
144 if (rows == 0 || cols == 0) {
145 return 1;
146 }
147 if (!lmmc_safe_mul_size(rows - 1, stride, &last_row)) {
148 return 0;
149 }
150 if (!lmmc_safe_add_size(last_row, cols, &elements)) {
151 return 0;
152 }
153 if (!lmmc_safe_mul_size(elements, element_size, &bytes)) {
154 return 0;
155 }
156 if ((uintptr_t)bytes > UINTPTR_MAX - begin) {
157 return 0;
158 }
159 out->end = begin + (uintptr_t)bytes;
160 return 1;
161}

262static inline void lmmc_scaled_sumsq_add(
263 lmmc_scaled_sumsq_t *acc, lmmc_real_t value)
264{
265 lmmc_real_t magnitude = fabs(value);
266 if (isnan(acc->scale)) {
267 return;
268 }
269 if (isnan(magnitude)) {
270 acc->scale = magnitude;
271 acc->sumsq = 1.0;
272 return;
273 }
274 if (magnitude == 0.0 || isinf(acc->scale)) {
275 return;
276 }
277 if (isinf(magnitude)) {
278 acc->scale = magnitude;
279 acc->sumsq = 1.0;
280 return;
281 }
282 if (acc->scale < magnitude) {
283 lmmc_real_t ratio = acc->scale / magnitude;
284 acc->sumsq = 1.0 + acc->sumsq * ratio * ratio;
285 acc->scale = magnitude;
286 } else {
287 lmmc_real_t ratio = magnitude / acc->scale;
288 acc->sumsq += ratio * ratio;
289 }
290}
