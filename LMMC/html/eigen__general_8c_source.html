33void lmmc_eigen_values_2x2(
34 lmmc_real_t a00, lmmc_real_t a01,
35 lmmc_real_t a10, lmmc_real_t a11,
36 lmmc_real_t *real0, lmmc_real_t *imag0,
37 lmmc_real_t *real1, lmmc_real_t *imag1)
38{
39 const lmmc_real_t trace = a00 + a11;
40 const lmmc_real_t determinant = a00 * a11 - a01 * a10;
41 const lmmc_real_t discriminant =
42 fma(trace, trace, -4.0 * determinant);
43 if (discriminant >= 0.0) {
44 const lmmc_real_t root = sqrt(discriminant);
45 const lmmc_real_t q =
46 0.5 * (trace + copysign(root, trace));
47 *real0 = q;
48 *real1 = q == 0.0 ? 0.0 : determinant / q;
49 *imag0 = 0.0;
50 *imag1 = 0.0;
51 } else {
52 const lmmc_real_t imaginary = sqrt(-discriminant) * 0.5;
53 *real0 = trace * 0.5;
54 *real1 = trace * 0.5;
55 *imag0 = imaginary;
56 *imag1 = -imaginary;
57 }
58}

66 lmmc_real_t *re, lmmc_real_t *im) {
67 size_t i = 0;
68 while (i < n) {

71 re[i] = MAT_ELEM(H, i, i);
72 im[i] = 0.0;
73 i++;
74 } else {

77 lmmc_real_t a12 = MAT_ELEM(H, i, i + 1);
78 lmmc_real_t a21 = MAT_ELEM(H, i + 1, i);
79 lmmc_real_t a22 = MAT_ELEM(H, i + 1, i + 1);
80 lmmc_eigen_values_2x2(

82 &re[i], &im[i], &re[i + 1], &im[i + 1]);
83 i += 2;
84 }

87static void eigenvalues_small(const lmmc_mat_t* a, lmmc_real_t matrix_scale,
88 lmmc_eigen_gen_result_t* out_result)
89{

138lmmc_status_t lmmc_eigen_general(const lmmc_mat_t* a,

140{
141 lmmc_real_t matrix_scale;

145 if (a->rows != a->cols) {

151 }
152 size_t n = a->rows;
153 status = lmmc_vec_create(n, &out_result->real_parts);
154 if (status != LMMC_STATUS_OK) {
155 return status;
156 }
157 status = lmmc_vec_create(n, &out_result->imag_parts);
158 if (status != LMMC_STATUS_OK) {
159 lmmc_vec_destroy(&out_result->real_parts);
160 return status;
161 }
162if (n <= 2) {
163 eigenvalues_small(a, matrix_scale, out_result);
164 return LMMC_STATUS_OK;
165 }
166 status = eigenvalues_large(a, matrix_scale, out_result);
167 if (status != LMMC_STATUS_OK) {
168 lmmc_eigen_gen_result_destroy(out_result);
169 }
170 return status;
