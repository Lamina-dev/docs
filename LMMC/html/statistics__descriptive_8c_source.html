11static lmmc_status_t lmmc_vec_mean_m2(const lmmc_vec_t* x, lmmc_real_t* out_mean, lmmc_real_t* out_m2) {
12 size_t i = 0;
13 lmmc_real_t mean, m2;
14 lmmc_status_t st = lmmc_validate_vec(x);
15
16 if (out_mean == NULL || out_m2 == NULL) {
17 return LMMC_STATUS_INVALID_ARGUMENT;
18 }

20 return st;

24 LMMC_REAL_INIT(&m2);
25
26 LMMC_REAL_SET_D(&mean, 0.0);
27 LMMC_REAL_SET_D(&m2, 0.0);
28

31
32 LMMC_REAL_INIT(&v);
33 LMMC_REAL_INIT(&n);
34 LMMC_REAL_INIT(&delta);
35 LMMC_REAL_INIT(&delta2);
36 LMMC_REAL_INIT(&tmp);
37

39 LMMC_REAL_SET_D(&n, (double)(i + 1));
40
41 if (!lmmc_is_finite(&v)) {
42 LMMC_REAL_CLEAR(&tmp);
43 LMMC_REAL_CLEAR(&delta2);
44 LMMC_REAL_CLEAR(&delta);

171 LMMC_REAL_CLEAR(&vy);
172 LMMC_REAL_CLEAR(&vx);
173 }

221 }
222
223 if (sample) {
224 if (x->size < 2) {
225 LMMC_REAL_CLEAR(&variance);
226 LMMC_REAL_CLEAR(&denom);
227 LMMC_REAL_CLEAR(&m2);
228 LMMC_REAL_CLEAR(&mean);

315static lmmc_status_t lmmc_vec_correlation_common(
316 const lmmc_vec_t* x,
317 const lmmc_vec_t* y,
318 int sample,
319 lmmc_real_t* out_correlation
320) {
321 lmmc_real_t c, m2x, m2y, corr, tmp, zero, one, min_one, tol, upper, lower;
322 lmmc_status_t st = LMMC_STATUS_OK;
323
324 if (out_correlation == NULL) {
325 return LMMC_STATUS_INVALID_ARGUMENT;
326 }
327
328 LMMC_REAL_INIT(&c);
329 LMMC_REAL_INIT(&m2x);
330 LMMC_REAL_INIT(&m2y);
331 LMMC_REAL_INIT(&corr);
332 LMMC_REAL_INIT(&tmp);
333 LMMC_REAL_INIT(&zero);
334 LMMC_REAL_INIT(&one);
335 LMMC_REAL_INIT(&min_one);
336 LMMC_REAL_INIT(&tol);
337 LMMC_REAL_INIT(&upper);
338 LMMC_REAL_INIT(&lower);
339
340 LMMC_REAL_SET_D(&c, 0.0);
341 LMMC_REAL_SET_D(&m2x, 0.0);
342 LMMC_REAL_SET_D(&m2y, 0.0);
343 LMMC_REAL_SET_D(&corr, 0.0);
344 LMMC_REAL_SET_D(&zero, 0.0);
345 LMMC_REAL_SET_D(&one, 1.0);
346 LMMC_REAL_SET_D(&min_one, -1.0);
347 LMMC_REAL_SET_D(&tol, 1e-12);
348
349 st = lmmc_vec_cov_accumulate(x, y, &c, &m2x, &m2y);
350 if (st != LMMC_STATUS_OK) {
351 goto cleanup;
352 }
353
354 if (sample && x->size < 2) {
355 st = LMMC_STATUS_INVALID_ARGUMENT;
356 goto cleanup;
357 }
358
359 if (LMMC_REAL_CMP(&m2x, &zero) <= 0 || LMMC_REAL_CMP(&m2y, &zero) <= 0) {
360 st = LMMC_STATUS_NUMERICAL_FAILURE;
361 goto cleanup;
362 }
363
364 LMMC_REAL_MUL(&tmp, &m2x, &m2y);
365 LMMC_REAL_SQRT(&tmp, &tmp);
366 LMMC_REAL_DIV(&corr, &c, &tmp);
367
368 if (!lmmc_is_finite(&corr)) {
369 st = LMMC_STATUS_NUMERICAL_FAILURE;
370 goto cleanup;
371 }
372
373 LMMC_REAL_ADD(&upper, &one, &tol);
374 if (LMMC_REAL_CMP(&corr, &one) > 0 && LMMC_REAL_CMP(&corr, &upper) < 0) {
375 LMMC_REAL_SET(&corr, &one);
376 }
377
378 LMMC_REAL_SUB(&lower, &min_one, &tol);
379 if (LMMC_REAL_CMP(&corr, &min_one) < 0 && LMMC_REAL_CMP(&corr, &lower) > 0) {
380 LMMC_REAL_SET(&corr, &min_one);
381 }
382
383 if (LMMC_REAL_CMP(&corr, &min_one) < 0 || LMMC_REAL_CMP(&corr, &one) > 0) {
384 st = LMMC_STATUS_NUMERICAL_FAILURE;
385 goto cleanup;
386 }
387
388 LMMC_REAL_SET(out_correlation, &corr);
389
390cleanup:
391 LMMC_REAL_CLEAR(&lower);
392 LMMC_REAL_CLEAR(&upper);
393 LMMC_REAL_CLEAR(&tol);
394 LMMC_REAL_CLEAR(&min_one);
395 LMMC_REAL_CLEAR(&one);
396 LMMC_REAL_CLEAR(&zero);
397 LMMC_REAL_CLEAR(&tmp);
398 LMMC_REAL_CLEAR(&corr);
399 LMMC_REAL_CLEAR(&m2y);
400 LMMC_REAL_CLEAR(&m2x);
401 LMMC_REAL_CLEAR(&c);
402
403 return st;
404}

406static lmmc_status_t lmmc_mat_column_means_to_buffer(const lmmc_mat_t* x, lmmc_real_t* means) {
407 size_t col = 0;
408 lmmc_status_t st = lmmc_validate_mat(x);
409
410 if (st != LMMC_STATUS_OK || means == NULL) {
411 return LMMC_STATUS_INVALID_ARGUMENT;
412 }
413
414 for (col = 0; col < x->cols; ++col) {
415 size_t row = 0;
416 lmmc_real_t mean;
417 LMMC_REAL_INIT(&mean);
418 LMMC_REAL_SET_D(&mean, 0.0);
419
420 for (row = 0; row < x->rows; ++row) {
421 lmmc_real_t v, n, delta, tmp;
422 LMMC_REAL_INIT(&v);
423 LMMC_REAL_INIT(&n);
424 LMMC_REAL_INIT(&delta);
425 LMMC_REAL_INIT(&tmp);

429

431 LMMC_REAL_CLEAR(&tmp);
432 LMMC_REAL_CLEAR(&delta);
433 LMMC_REAL_CLEAR(&n);
434 LMMC_REAL_CLEAR(&v);
435 LMMC_REAL_CLEAR(&mean);
436 return LMMC_STATUS_NUMERICAL_FAILURE;
437 }
438
439 LMMC_REAL_SUB(&delta, &v, &mean);
440 LMMC_REAL_DIV(&tmp, &delta, &n);
441 LMMC_REAL_ADD(&mean, &mean, &tmp);

444 LMMC_REAL_CLEAR(&delta);
445 LMMC_REAL_CLEAR(&n);
446 LMMC_REAL_CLEAR(&v);

452 }
453
454 LMMC_REAL_SET(&means[col], &mean);
455 LMMC_REAL_CLEAR(&mean);
456 }
457
458 return LMMC_STATUS_OK;
459}

461static lmmc_status_t lmmc_mat_covariance_or_correlation(
462 const lmmc_mat_t* x,
463 int sample,
464 int correlation,
465 lmmc_mat_t* out_matrix
466) {
467 size_t bytes = 0;
468 size_t col_i = 0;
469 size_t col_j = 0;
470 lmmc_real_t denom;
471 lmmc_real_t* means = NULL;
472 lmmc_real_t* stddev = NULL;
473 lmmc_status_t st = lmmc_validate_mat(x);
474
475 if (st != LMMC_STATUS_OK || out_matrix == NULL || out_matrix->data == NULL) {
476 return LMMC_STATUS_INVALID_ARGUMENT;
477 }
478 if (out_matrix->rows != x->cols || out_matrix->cols != x->cols) {
479 return LMMC_STATUS_DIMENSION_MISMATCH;
480 }
481 if (sample && x->rows < 2) {
482 return LMMC_STATUS_INVALID_ARGUMENT;
483 }
484
485 LMMC_REAL_INIT(&denom);
486 if (sample) {
487 LMMC_REAL_SET_D(&denom, (double)(x->rows - 1));
488 } else {
489 LMMC_REAL_SET_D(&denom, (double)x->rows);
490 }
491
492 if (!lmmc_safe_mul_size(x->cols, sizeof(lmmc_real_t), &bytes)) {
493 LMMC_REAL_CLEAR(&denom);
494 return LMMC_STATUS_INVALID_ARGUMENT;
495 }
496
497 means = (lmmc_real_t*)lmmc_alloc(bytes);
498 if (means == NULL) {
499 LMMC_REAL_CLEAR(&denom);
500 return LMMC_STATUS_ALLOCATION_FAILED;
501 }
502 for (size_t k = 0; k < x->cols; ++k) LMMC_REAL_INIT(&means[k]);
503
504 if (correlation) {
505 stddev = (lmmc_real_t*)lmmc_alloc(bytes);
506 if (stddev == NULL) {
507 for (size_t k = 0; k < x->cols; ++k) LMMC_REAL_CLEAR(&means[k]);
508 lmmc_free(means);
509 LMMC_REAL_CLEAR(&denom);
510 return LMMC_STATUS_ALLOCATION_FAILED;
511 }
512 for (size_t k = 0; k < x->cols; ++k) LMMC_REAL_INIT(&stddev[k]);
513 }
514
515 st = lmmc_mat_column_means_to_buffer(x, means);
516 if (st != LMMC_STATUS_OK) {
517 if (correlation) {
518 for (size_t k = 0; k < x->cols; ++k) LMMC_REAL_CLEAR(&stddev[k]);
519 lmmc_free(stddev);
520 }
521 for (size_t k = 0; k < x->cols; ++k) LMMC_REAL_CLEAR(&means[k]);
522 lmmc_free(means);
523 LMMC_REAL_CLEAR(&denom);
524 return st;
525 }
526
527 if (correlation) {
528 for (col_i = 0; col_i < x->cols; ++col_i) {
529 size_t row = 0;
530 lmmc_real_t sumsq, variance, d, tmp;
531 LMMC_REAL_INIT(&sumsq);
532 LMMC_REAL_INIT(&variance);
533 LMMC_REAL_INIT(&d);
534 LMMC_REAL_INIT(&tmp);
535
536 LMMC_REAL_SET_D(&sumsq, 0.0);
537
538 for (row = 0; row < x->rows; ++row) {
539 LMMC_REAL_SUB(&d, &x->data[row * x->stride + col_i], &means[col_i]);
540 LMMC_REAL_MUL(&tmp, &d, &d);
541 LMMC_REAL_ADD(&sumsq, &sumsq, &tmp);
542 }
543
544 LMMC_REAL_DIV(&variance, &sumsq, &denom);
545 st = lmmc_finalize_nonnegative(variance, &variance);
546
547 lmmc_real_t zero;
548 LMMC_REAL_INIT(&zero);
549 LMMC_REAL_SET_D(&zero, 0.0);
550
551 if (st != LMMC_STATUS_OK || LMMC_REAL_CMP(&variance, &zero) <= 0) {
552 LMMC_REAL_CLEAR(&zero);
553 LMMC_REAL_CLEAR(&tmp);
554 LMMC_REAL_CLEAR(&d);
555 LMMC_REAL_CLEAR(&variance);
556 LMMC_REAL_CLEAR(&sumsq);
557 if (correlation) {
558 for (size_t k = 0; k < x->cols; ++k) LMMC_REAL_CLEAR(&stddev[k]);
559 lmmc_free(stddev);
560 }
561 for (size_t k = 0; k < x->cols; ++k) LMMC_REAL_CLEAR(&means[k]);
562 lmmc_free(means);
563 LMMC_REAL_CLEAR(&denom);
564 return LMMC_STATUS_NUMERICAL_FAILURE;
565 }
566 LMMC_REAL_CLEAR(&zero);
567
568 LMMC_REAL_SQRT(&stddev[col_i], &variance);
569
570 if (!lmmc_is_finite(&stddev[col_i])) {
571 LMMC_REAL_CLEAR(&tmp);
572 LMMC_REAL_CLEAR(&d);
573 LMMC_REAL_CLEAR(&variance);
574 LMMC_REAL_CLEAR(&sumsq);
575 for (size_t k = 0; k < x->cols; ++k) LMMC_REAL_CLEAR(&stddev[k]);
576 lmmc_free(stddev);
577 for (size_t k = 0; k < x->cols; ++k) LMMC_REAL_CLEAR(&means[k]);
578 lmmc_free(means);
579 LMMC_REAL_CLEAR(&denom);
580 return LMMC_STATUS_NUMERICAL_FAILURE;
581 }
582 LMMC_REAL_CLEAR(&tmp);
583 LMMC_REAL_CLEAR(&d);
584 LMMC_REAL_CLEAR(&variance);
585 LMMC_REAL_CLEAR(&sumsq);
586 }
587 }
588
589 for (col_i = 0; col_i < x->cols; ++col_i) {
590 for (col_j = col_i; col_j < x->cols; ++col_j) {
591 size_t row = 0;
592 lmmc_real_t sum, cov, out_value, di, dj, tmp;
593 LMMC_REAL_INIT(&sum);
594 LMMC_REAL_INIT(&cov);
595 LMMC_REAL_INIT(&out_value);
596 LMMC_REAL_INIT(&di);
597 LMMC_REAL_INIT(&dj);
598 LMMC_REAL_INIT(&tmp);
599
600 LMMC_REAL_SET_D(&sum, 0.0);
601
602 for (row = 0; row < x->rows; ++row) {
603 LMMC_REAL_SUB(&di, &x->data[row * x->stride + col_i], &means[col_i]);

632 LMMC_REAL_DIV(&out_value, &cov, &tmp);

635 LMMC_REAL_CLEAR(&dj);

661 LMMC_REAL_SET(&out_value, &one);

670 LMMC_REAL_CLEAR(&lower);

693 LMMC_REAL_CLEAR(&min_one);
694 LMMC_REAL_CLEAR(&one);
695 }
696 } else {
697 LMMC_REAL_SET(&out_value, &cov);
698 }

874lmmc_status_t lmmc_vec_quantile(const lmmc_vec_t* x, lmmc_real_t p, lmmc_real_t* out) {
875 lmmc_real_t* sorted = NULL;
876 size_t n;
877 double h, frac;
878 size_t lo, hi;
879
880 if (out == NULL) return LMMC_STATUS_INVALID_ARGUMENT;
881 if (x == NULL || x->data == NULL || x->size == 0) return LMMC_STATUS_INVALID_ARGUMENT;
882 if (p < 0.0 || p > 1.0) return LMMC_STATUS_INVALID_ARGUMENT;
883
884 n = x->size;
885 sorted = (lmmc_real_t*)lmmc_alloc(n * sizeof(lmmc_real_t));
886 if (sorted == NULL) return LMMC_STATUS_ALLOCATION_FAILED;
887
888 for (size_t i = 0; i < n; ++i) {
889 sorted[i] = x->data[i];
890 }
891 qsort(sorted, n, sizeof(lmmc_real_t), lmmc_real_compare);
892
893 /* Linear interpolation method (same as numpy default) */
894 h = p * (double)(n - 1);
895 lo = (size_t)h;
896 hi = lo + 1;
897 frac = h - (double)lo;
898
899 if (hi >= n) {
900 *out = sorted[n - 1];
901 } else {
902 *out = sorted[lo] * (1.0 - frac) + sorted[hi] * frac;
903 }
904
905 lmmc_free(sorted);
906 return LMMC_STATUS_OK;
907}

909lmmc_status_t lmmc_vec_histogram(const lmmc_vec_t* x, size_t nbins, lmmc_real_t* edges, size_t* counts) {
910 size_t n, i;
911 lmmc_real_t xmin, xmax, width;
912
913 if (x == NULL || x->data == NULL || x->size == 0) return LMMC_STATUS_INVALID_ARGUMENT;
914 if (nbins == 0 || edges == NULL || counts == NULL) return LMMC_STATUS_INVALID_ARGUMENT;
915
916 n = x->size;
917
918 /* Find min and max */
919 xmin = x->data[0];
920 xmax = x->data[0];
921 for (i = 1; i < n; ++i) {
922 if (x->data[i] < xmin) xmin = x->data[i];
923 if (x->data[i] > xmax) xmax = x->data[i];
924 }
925
926 /* Handle constant data */
927 if (xmax == xmin) {
928 xmin -= 0.5;
929 xmax += 0.5;
930 }
931
932 width = (xmax - xmin) / (lmmc_real_t)nbins;
933
934 /* Compute edges */
935 for (i = 0; i <= nbins; ++i) {
936 edges[i] = xmin + width * (lmmc_real_t)i;
937 }
938
939 /* Initialize counts */
940 for (i = 0; i < nbins; ++i) {
941 counts[i] = 0;
942 }
943
944 /* Bin the data */
945 for (i = 0; i < n; ++i) {
946 size_t bin;
947 if (x->data[i] >= xmax) {
948 bin = nbins - 1; /* last bin includes the right edge */
949 } else {
950 bin = (size_t)((x->data[i] - xmin) / width);
951 if (bin >= nbins) bin = nbins - 1;
952 }
953 counts[bin]++;
954 }
955
956 return LMMC_STATUS_OK;
957}
