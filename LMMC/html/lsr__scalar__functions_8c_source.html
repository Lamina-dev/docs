74lmmc_status_t lmmc_lsr_math_pow(lmmc_real_t x, lmmc_real_t y,
75 lmmc_real_t* out)
76{
77 double value;
78 double integral_part;
79 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
80 if (!lmmc_lsr_real_is_finite(x) || !lmmc_lsr_real_is_finite(y)) {
81 return LMMC_STATUS_NUMERICAL_FAILURE;
82 }
83 if ((double)x == 0.0 && (double)y < 0.0) {
84 return LMMC_STATUS_OUT_OF_RANGE;
85 }
86 if ((double)x < 0.0 && modf((double)y, &integral_part) != 0.0) {
87 return LMMC_STATUS_OUT_OF_RANGE;
88 }
89 value = pow((double)x, (double)y);
90 if (!isfinite(value)) return LMMC_STATUS_NUMERICAL_FAILURE;
91 *out = (lmmc_real_t)value;
92 return LMMC_STATUS_OK;
93}

191lmmc_status_t lmmc_lsr_math_clamp(lmmc_real_t x, lmmc_real_t lo,
192 lmmc_real_t hi, lmmc_real_t* out)
193{
194 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
195 if (!lmmc_lsr_real_is_finite(x) || !lmmc_lsr_real_is_finite(lo) ||
196 !lmmc_lsr_real_is_finite(hi)) {
197 return LMMC_STATUS_NUMERICAL_FAILURE;
198 }
199 if (lo > hi) return LMMC_STATUS_INVALID_ARGUMENT;
200 if (x < lo) *out = lo;
201 else if (x > hi) *out = hi;
202 else *out = x;
203 return lmmc_lsr_store_finite_real(*out, out);
