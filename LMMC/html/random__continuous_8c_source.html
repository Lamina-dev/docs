43 v = 1.0 + c * x;
44 } while (v <= 0.0);

47 u = u64_to_double01(xoshiro256ss_next(rng->state));

51 *out = scale * d * v;
52 return LMMC_STATUS_OK;
53 }
54
56 if (log(u) < 0.5 * x * x + d * (1.0 - v + log(v))) {
