47static lmmc_status_t ros_time_derivative(
48 lmmc_ode_rhs_t rhs,
49 lmmc_ode_dfdt_t time_derivative,
50 void* user_data,
51 lmmc_real_t t,
52 const lmmc_real_t* y,
53 const lmmc_real_t* f0,
54 size_t dim,
55 lmmc_real_t* dfdt,
56 lmmc_real_t* f_pert,
57 lmmc_ode_result_t* result
58) {
59 size_t i;
60 lmmc_status_t st;
61 int callback_failed = 0;
62
63 if (time_derivative != NULL) {
64 st = time_derivative(t, y, dfdt, dim, user_data);
65 if (st != LMMC_STATUS_OK) {
66 return st;
67 }
68 return lmmc_ode_values_are_finite(dfdt, dim);
69 }
70
71 {
72 lmmc_real_t delta = sqrt(2.2204460492503131e-16) * fmax(1.0, fabs(t));
73 st = lmmc_ode_rhs_eval(rhs, t + delta, y, f_pert, dim, user_data,
74 &result->num_rhs_evals, &callback_failed);
75 if (st != LMMC_STATUS_OK) {
76 return st;
77 }
78 for (i = 0; i < dim; ++i) {
79 dfdt[i] = (f_pert[i] - f0[i]) / delta;
80 }
81 }
82 return lmmc_ode_values_are_finite(dfdt, dim);
83}

101lmmc_status_t lmmc_ode_rosenbrock_grk4t_solve(
102 lmmc_ode_rhs_t rhs,
103 void* user_data,
104 size_t dim,
105 lmmc_real_t t_start,
106 lmmc_real_t t_end,
107 lmmc_real_t* y,
108 const lmmc_ode_config_t* cfg,
109 lmmc_ode_result_t* out_result
110) {
111 lmmc_ode_config_t local_cfg = {0};
112 size_t work_bytes = 0, vectors_bytes = 0, matrix_count = 0, matrix_bytes = 0;
113 lmmc_real_t* work = NULL;
114 lmmc_real_t* matrix_data = NULL;
115 size_t* pivots = NULL;
116 lmmc_real_t* k[ROS_STAGES];
117 lmmc_real_t *f0, *f_stage, *y_stage, *y_pert, *f_pert;
118 lmmc_real_t *rhs_vec, *dfdt, *y_new, *error;
119 lmmc_mat_t W = {0};
120 lmmc_real_t t = t_start, h;
121 size_t attempts = 0, i, j;
122 lmmc_status_t st;
123
124 st = validate_and_init_ode_config(rhs, dim, t_start, t_end, y, cfg,
125 &local_cfg, out_result, &work_bytes);
126 if (st != LMMC_STATUS_OK) {
127 return st;
128 }
129 if (!lmmc_safe_mul_size(work_bytes, ROS_VECTORS, &vectors_bytes) ||
130 !lmmc_safe_mul_size(dim, dim, &matrix_count) ||
131 !lmmc_safe_mul_size(matrix_count, sizeof(lmmc_real_t), &matrix_bytes) ||
132 !lmmc_safe_mul_size(dim, sizeof(size_t), &matrix_count)) {
133 out_result->failure_reason = LMMC_ODE_FAILURE_INVALID_DIMENSION;
134 return LMMC_STATUS_INVALID_ARGUMENT;
135 }
136
137 work = (lmmc_real_t*)lmmc_alloc(vectors_bytes);
138 matrix_data = (lmmc_real_t*)lmmc_alloc(matrix_bytes);
139 pivots = (size_t*)lmmc_alloc(matrix_count);
140 if (work == NULL || matrix_data == NULL || pivots == NULL) {
141 st = LMMC_STATUS_ALLOCATION_FAILED;
142 goto cleanup;
143 }
144
145 for (i = 0; i < ROS_STAGES; ++i) {
146 k[i] = work + i * dim;
147 }
148 f0 = work + 4 * dim;
149 f_stage = work + 5 * dim;
150 y_stage = work + 6 * dim;
151 y_pert = work + 7 * dim;
152 f_pert = work + 8 * dim;
153 rhs_vec = work + 9 * dim;
154 dfdt = work + 10 * dim;
155 y_new = work + 11 * dim;
156 error = work + 12 * dim;
157 W = (lmmc_mat_t){dim, dim, dim, matrix_data, 0};
158
159 h = lmmc_clamp(local_cfg.initial_step, local_cfg.min_step, local_cfg.max_step);
160 if (h > t_end - t_start) {
161 h = t_end - t_start;
162 }
163 lmmc_ode_do_log(&local_cfg, 0, t, y, dim);
164
165 while (t < t_end && attempts < local_cfg.max_steps) {
166 lmmc_real_t error_norm = 0.0;
167 size_t swap_count = 0;
168 int callback_failed = 0;
169 ++attempts;
170 if (h > t_end - t) {
171 h = t_end - t;
172 }
173
174 st = lmmc_ode_rhs_eval(rhs, t, y, f0, dim, user_data,
175 &out_result->num_rhs_evals, &callback_failed);
176 if (st != LMMC_STATUS_OK) {
177 out_result->failure_reason = callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
178 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
179 goto cleanup;
180 }
181 st = lmmc_ode_jacobian_eval(
182 rhs, local_cfg.jacobian, user_data, t, y, f0, dim, matrix_data,
183 f_stage, y_pert, f_pert, &out_result->num_rhs_evals,
184 &callback_failed);
185 if (st != LMMC_STATUS_OK) {
186 out_result->failure_reason =
187 callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
188 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
189 goto cleanup;
190 }
191 st = ros_time_derivative(rhs, local_cfg.time_derivative, user_data, t, y,
192 f0, dim, dfdt, f_pert, out_result);
193 if (st != LMMC_STATUS_OK) {
194 out_result->failure_reason = LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
195 goto cleanup;
196 }
197
198 for (i = 0; i < dim; ++i) {
199 for (j = 0; j < dim; ++j) {
200 W.data[i * dim + j] = -matrix_data[i * dim + j];
201 }
202 W.data[i * dim + i] += 1.0 / (h * ros_gamma);
203 }
204 st = lmmc_lu_decompose_inplace(&W, pivots, &swap_count);
205 if (st != LMMC_STATUS_OK) {
206 goto cleanup;
207 }
208
209 for (i = 0; i < dim; ++i) {
210 rhs_vec[i] = f0[i] + h * ros_d[0] * dfdt[i];
211 }
212 st = ros_solve_stage(&W, pivots, rhs_vec, k[0], dim);
213 if (st != LMMC_STATUS_OK) goto cleanup;
214
215 for (i = 0; i < dim; ++i) {
216 y_stage[i] = y[i] + ros_a21 * k[0][i];
217 }
218 st = lmmc_ode_rhs_eval(rhs, t + ros_c2 * h, y_stage, f_stage, dim, user_data,
219 &out_result->num_rhs_evals, &callback_failed);
220 if (st != LMMC_STATUS_OK) {
221 out_result->failure_reason = callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
222 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
223 goto cleanup;
224 }
225 for (i = 0; i < dim; ++i) {
226 rhs_vec[i] = f_stage[i] + h * ros_d[1] * dfdt[i] + (ros_c21 / h) * k[0][i];
227 }
228 st = ros_solve_stage(&W, pivots, rhs_vec, k[1], dim);
229 if (st != LMMC_STATUS_OK) goto cleanup;
230
231 for (i = 0; i < dim; ++i) {
232 y_stage[i] = y[i] + ros_a31 * k[0][i] + ros_a32 * k[1][i];
233 }
234 st = lmmc_ode_rhs_eval(rhs, t + ros_c3 * h, y_stage, f_stage, dim, user_data,
235 &out_result->num_rhs_evals, &callback_failed);
236 if (st != LMMC_STATUS_OK) {
237 out_result->failure_reason = callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
238 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
239 goto cleanup;
240 }
241 for (i = 0; i < dim; ++i) {
242 rhs_vec[i] = f_stage[i] + h * ros_d[2] * dfdt[i]
243 + (ros_c31 / h) * k[0][i] + (ros_c32 / h) * k[1][i];
244 }
245 st = ros_solve_stage(&W, pivots, rhs_vec, k[2], dim);
246 if (st != LMMC_STATUS_OK) goto cleanup;
247
248 for (i = 0; i < dim; ++i) {
249 rhs_vec[i] = f_stage[i] + h * ros_d[3] * dfdt[i]
250 + (ros_c41 / h) * k[0][i] + (ros_c42 / h) * k[1][i]
251 + (ros_c43 / h) * k[2][i];
252 }
253 st = ros_solve_stage(&W, pivots, rhs_vec, k[3], dim);
254 if (st != LMMC_STATUS_OK) goto cleanup;
255
256 for (i = 0; i < dim; ++i) {
257 y_new[i] = y[i] + ros_b[0] * k[0][i] + ros_b[1] * k[1][i]
258 + ros_b[2] * k[2][i] + ros_b[3] * k[3][i];
259 error[i] = ros_e[0] * k[0][i] + ros_e[1] * k[1][i]
260 + ros_e[2] * k[2][i] + ros_e[3] * k[3][i];
261 }
262 st = lmmc_ode_values_are_finite(y_new, dim);
263 if (st != LMMC_STATUS_OK) goto cleanup;
264 st = lmmc_ode_weighted_rms(error, y, y_new, dim, local_cfg.abs_tol,
265 local_cfg.rel_tol, &error_norm);
266 if (st != LMMC_STATUS_OK) goto cleanup;
267
268 if (error_norm <= 1.0) {
269 memcpy(y, y_new, work_bytes);
270 t += h;
271 out_result->num_steps += 1;
272 out_result->final_t = t;
273 lmmc_ode_do_log(&local_cfg, out_result->num_steps, t, y, dim);
274 } else if (h <= local_cfg.min_step) {
275 st = LMMC_STATUS_CONVERGENCE_FAILED;
276 out_result->failure_reason = LMMC_ODE_FAILURE_INVALID_STEP;
277 goto cleanup;
278 }
279 h = lmmc_ode_next_step(h, error_norm, &local_cfg);
280 }
281
282 if (t >= t_end) {
283 out_result->converged = 1;
284 out_result->failure_reason = LMMC_ODE_FAILURE_NONE;
285 st = LMMC_STATUS_OK;
286 } else {
287 out_result->failure_reason = LMMC_ODE_FAILURE_MAX_STEPS;
288 st = LMMC_STATUS_CONVERGENCE_FAILED;
289 }
290
291cleanup:
292 if (st != LMMC_STATUS_OK && out_result != NULL &&
293 out_result->failure_reason == LMMC_ODE_FAILURE_NONE) {
294 out_result->failure_reason = LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
295 }
296 lmmc_free(pivots);
297 lmmc_free(matrix_data);
298 lmmc_free(work);
299 return st;
300}
