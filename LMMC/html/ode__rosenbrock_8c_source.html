47static lmmc_status_t ros_time_derivative(
48 lmmc_ode_rhs_t rhs,
49 lmmc_ode_dfdt_t time_derivative,
50 void* user_data,
51 lmmc_real_t t,
52 const lmmc_real_t* y,
53 const lmmc_real_t* f0,
54 size_t dim,
55 lmmc_real_t* dfdt,
56 lmmc_real_t* f_pert,
57 lmmc_ode_result_t* result
58) {
59 size_t i;
60 lmmc_status_t st;
61 int callback_failed = 0;
62
63 if (time_derivative != NULL) {
64 st = time_derivative(t, y, dfdt, dim, user_data);
65 if (st != LMMC_STATUS_OK) {
66 return st;
67 }
68 return lmmc_ode_values_are_finite(dfdt, dim);
69 }
70
71 {
72 lmmc_real_t delta = sqrt(2.2204460492503131e-16) * fmax(1.0, fabs(t));
73 st = lmmc_ode_rhs_eval(rhs, t + delta, y, f_pert, dim, user_data,
74 &result->num_rhs_evals, &callback_failed);
75 if (st != LMMC_STATUS_OK) {
76 return st;
77 }
78 for (i = 0; i < dim; ++i) {
79 dfdt[i] = (f_pert[i] - f0[i]) / delta;
80 }
81 }
82 return lmmc_ode_values_are_finite(dfdt, dim);
83}

118static lmmc_status_t ros_prepare_system(ode_rosenbrock_state_t* s) {
119 size_t i, j, swap_count = 0;
120 int callback_failed = 0;
121 lmmc_status_t st;
122 st = lmmc_ode_rhs_eval(s->rhs, s->t, s->y, s->f0, s->dim, s->user_data,
123 &s->out_result->num_rhs_evals, &callback_failed);
124 if (st != LMMC_STATUS_OK) {
125 s->out_result->failure_reason = callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
126 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
127 return st;
128 }
129 st = lmmc_ode_jacobian_eval(&(lmmc_ode_jacobian_request_t){
130 s->rhs, s->local_cfg.jacobian, s->user_data, s->t, s->y, s->f0, s->dim, s->matrix_data,
131 s->f_stage, s->y_pert, s->f_pert, &s->out_result->num_rhs_evals,
132 &callback_failed});
133 if (st != LMMC_STATUS_OK) {
134 s->out_result->failure_reason =
135 callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
136 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
137 return st;
138 }
139 st = ros_time_derivative(s->rhs, s->local_cfg.time_derivative, s->user_data, s->t, s->y,
140 s->f0, s->dim, s->dfdt, s->f_pert, s->out_result);
141 if (st != LMMC_STATUS_OK) {
142 s->out_result->failure_reason = LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
143 return st;
144 }
145
146 for (i = 0; i < s->dim; ++i) {
147 for (j = 0; j < s->dim; ++j) {
148 s->W.data[i * s->dim + j] = -s->matrix_data[i * s->dim + j];
149 }
150 s->W.data[i * s->dim + i] += 1.0 / (s->h * ros_gamma);
151 }
152 st = lmmc_lu_decompose_inplace(&s->W, s->pivots, &swap_count);
153 if (st != LMMC_STATUS_OK) {
154 return st;
155 }
156 return LMMC_STATUS_OK;
157}

159static lmmc_status_t ros_evaluate_stages(ode_rosenbrock_state_t* s) {
160 size_t i;
161 int callback_failed = 0;
162 lmmc_status_t st;
163 for (i = 0; i < s->dim; ++i) {
164 s->rhs_vec[i] = s->f0[i] + s->h * ros_d[0] * s->dfdt[i];
165 }
166 st = ros_solve_stage(&s->W, s->pivots, s->rhs_vec, s->k[0], s->dim);
167 if (st != LMMC_STATUS_OK) { return st; }
168
169 for (i = 0; i < s->dim; ++i) {
170 s->y_stage[i] = s->y[i] + ros_a21 * s->k[0][i];
171 }
172 st = lmmc_ode_rhs_eval(s->rhs, s->t + ros_c2 * s->h, s->y_stage, s->f_stage, s->dim, s->user_data,
173 &s->out_result->num_rhs_evals, &callback_failed);
174 if (st != LMMC_STATUS_OK) {
175 s->out_result->failure_reason = callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
176 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
177 return st;
178 }
179 for (i = 0; i < s->dim; ++i) {
180 s->rhs_vec[i] = s->f_stage[i] + s->h * ros_d[1] * s->dfdt[i] + (ros_c21 / s->h) * s->k[0][i];
181 }
182 st = ros_solve_stage(&s->W, s->pivots, s->rhs_vec, s->k[1], s->dim);
183 if (st != LMMC_STATUS_OK) { return st; }
184
185 for (i = 0; i < s->dim; ++i) {
186 s->y_stage[i] = s->y[i] + ros_a31 * s->k[0][i] + ros_a32 * s->k[1][i];
187 }
188 st = lmmc_ode_rhs_eval(s->rhs, s->t + ros_c3 * s->h, s->y_stage, s->f_stage, s->dim, s->user_data,
189 &s->out_result->num_rhs_evals, &callback_failed);
190 if (st != LMMC_STATUS_OK) {
191 s->out_result->failure_reason = callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED
192 : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
193 return st;
194 }
195 for (i = 0; i < s->dim; ++i) {
196 s->rhs_vec[i] = s->f_stage[i] + s->h * ros_d[2] * s->dfdt[i]
197 + (ros_c31 / s->h) * s->k[0][i] + (ros_c32 / s->h) * s->k[1][i];
198 }
199 st = ros_solve_stage(&s->W, s->pivots, s->rhs_vec, s->k[2], s->dim);
200 if (st != LMMC_STATUS_OK) { return st; }
201
202 for (i = 0; i < s->dim; ++i) {
203 s->rhs_vec[i] = s->f_stage[i] + s->h * ros_d[3] * s->dfdt[i]
204 + (ros_c41 / s->h) * s->k[0][i] + (ros_c42 / s->h) * s->k[1][i]
205 + (ros_c43 / s->h) * s->k[2][i];
206 }
207 st = ros_solve_stage(&s->W, s->pivots, s->rhs_vec, s->k[3], s->dim);
208 if (st != LMMC_STATUS_OK) { return st; }
209 return LMMC_STATUS_OK;
210}

212static lmmc_status_t ros_estimate_error(ode_rosenbrock_state_t* s, lmmc_real_t* error_norm) {
213 size_t i;
214 lmmc_status_t st;
215 for (i = 0; i < s->dim; ++i) {
216 s->y_new[i] = s->y[i] + ros_b[0] * s->k[0][i] + ros_b[1] * s->k[1][i]
217 + ros_b[2] * s->k[2][i] + ros_b[3] * s->k[3][i];
218 s->error[i] = ros_e[0] * s->k[0][i] + ros_e[1] * s->k[1][i]
219 + ros_e[2] * s->k[2][i] + ros_e[3] * s->k[3][i];
220 }
221 st = lmmc_ode_values_are_finite(s->y_new, s->dim);
222 if (st != LMMC_STATUS_OK) { return st; }
223 st = lmmc_ode_weighted_rms(s->error, s->y, s->y_new, s->dim, s->local_cfg.abs_tol,
224 s->local_cfg.rel_tol, error_norm);
225 if (st != LMMC_STATUS_OK) { return st; }
226 return LMMC_STATUS_OK;
227}

229static lmmc_status_t ros_integrate(ode_rosenbrock_state_t* s, lmmc_real_t t_end) {
230 size_t attempts = 0;
231 lmmc_status_t st;
232 while (s->t < t_end && attempts < s->local_cfg.max_steps) {
233 lmmc_real_t error_norm = 0.0;
234 ++attempts;
235 if (s->h > t_end - s->t) {
236 s->h = t_end - s->t;
237 }
238 st = ros_prepare_system(s);
239 if (st != LMMC_STATUS_OK) { return st; }
240 st = ros_evaluate_stages(s);
241 if (st != LMMC_STATUS_OK) { return st; }
242 st = ros_estimate_error(s, &error_norm);
243 if (st != LMMC_STATUS_OK) { return st; }
244 if (error_norm <= 1.0) {
245 memcpy(s->y, s->y_new, s->work_bytes);
246 s->t += s->h;
247 s->out_result->num_steps += 1;
248 s->out_result->final_t = s->t;
249 lmmc_ode_do_log(&s->local_cfg, s->out_result->num_steps, s->t, s->y, s->dim);
250 } else if (s->h <= s->local_cfg.min_step) {
251 st = LMMC_STATUS_CONVERGENCE_FAILED;
252 s->out_result->failure_reason = LMMC_ODE_FAILURE_INVALID_STEP;
253 return st;
254 }
255 s->h = lmmc_ode_next_step(s->h, error_norm, &s->local_cfg);
256 }
257
258 if (s->t >= t_end) {
259 s->out_result->converged = 1;
260 s->out_result->failure_reason = LMMC_ODE_FAILURE_NONE;
261 st = LMMC_STATUS_OK;
262 } else {
263 s->out_result->failure_reason = LMMC_ODE_FAILURE_MAX_STEPS;
264 st = LMMC_STATUS_CONVERGENCE_FAILED;
265 }
266 return st;
267}

280lmmc_status_t lmmc_ode_rosenbrock_grk4t_solve(
281 lmmc_ode_rhs_t rhs,
282 void* user_data,
283 size_t dim,
284 lmmc_real_t t_start,
285 lmmc_real_t t_end,
286 lmmc_real_t* y,
287 const lmmc_ode_config_t* cfg,
288 lmmc_ode_result_t* out_result
289) {

291 ode_rosenbrock_state_t* s = &state;
292 size_t vectors_bytes = 0, matrix_count = 0, matrix_bytes = 0;
293 size_t i;
294 lmmc_status_t st;
295 s->rhs = rhs;
296 s->user_data = user_data;
297 s->dim = dim;
298 s->y = y;
299 s->out_result = out_result;
300 s->t = t_start;
301 st = validate_and_init_ode_config(s->rhs, s->dim, t_start, t_end, s->y, cfg,
302 &s->local_cfg, s->out_result, &s->work_bytes);
303 if (st != LMMC_STATUS_OK) {
304 return st;
305 }
306 if (!lmmc_safe_mul_size(s->work_bytes, ROS_VECTORS, &vectors_bytes) ||
307 !lmmc_safe_mul_size(s->dim, s->dim, &matrix_count) ||
308 !lmmc_safe_mul_size(matrix_count, sizeof(lmmc_real_t), &matrix_bytes) ||
309 !lmmc_safe_mul_size(s->dim, sizeof(size_t), &matrix_count)) {
310 s->out_result->failure_reason = LMMC_ODE_FAILURE_INVALID_DIMENSION;
311 return LMMC_STATUS_INVALID_ARGUMENT;
312 }
313
314 s->work = (lmmc_real_t*)lmmc_memory_alloc(vectors_bytes);
315 s->matrix_data = (lmmc_real_t*)lmmc_memory_alloc(matrix_bytes);
316 s->pivots = (size_t*)lmmc_memory_alloc(matrix_count);
317 if (s->work == NULL || s->matrix_data == NULL || s->pivots == NULL) {
318 st = LMMC_STATUS_ALLOCATION_FAILED;
319 return ros_cleanup(s, st);
320 }
321
322 for (i = 0; i < ROS_STAGES; ++i) {
323 s->k[i] = s->work + i * s->dim;
324 }
325 s->f0 = s->work + 4 * s->dim;
326 s->f_stage = s->work + 5 * s->dim;
327 s->y_stage = s->work + 6 * s->dim;
328 s->y_pert = s->work + 7 * s->dim;
329 s->f_pert = s->work + 8 * s->dim;
330 s->rhs_vec = s->work + 9 * s->dim;
331 s->dfdt = s->work + 10 * s->dim;
332 s->y_new = s->work + 11 * s->dim;
333 s->error = s->work + 12 * s->dim;
334 s->W = (lmmc_mat_t){s->dim, s->dim, s->dim, s->matrix_data, 0};
335
336 s->h = lmmc_clamp(s->local_cfg.initial_step, s->local_cfg.min_step, s->local_cfg.max_step);
337 if (s->h > t_end - t_start) {
338 s->h = t_end - t_start;
339 }
340 lmmc_ode_do_log(&s->local_cfg, 0, s->t, s->y, s->dim);
341 st = ros_integrate(s, t_end);
342 return ros_cleanup(s, st);
343}
