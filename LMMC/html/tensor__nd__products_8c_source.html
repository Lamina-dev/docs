87static lmmc_status_t lmmc_tensor_nd_contraction_shape(
88 const lmmc_tensor_nd_t* a, const lmmc_tensor_nd_t* b,
89 lmmc_tensor_nd_contraction_t* contraction, size_t* out_ndim, size_t* out_dims) {
90 int a_contracted[LMMC_TENSOR_MAX_NDIM] = {0};
91 int b_contracted[LMMC_TENSOR_MAX_NDIM] = {0};
92 contraction->contract_size = 1;
93 for (size_t i = 0; i < contraction->naxes; ++i) {
94 const size_t axis_a = contraction->axes_a[i];
95 const size_t axis_b = contraction->axes_b[i];
96 if (axis_a >= a->ndim || axis_b >= b->ndim) {
97 return LMMC_STATUS_INVALID_ARGUMENT;
98 }
99 if (a_contracted[axis_a] || b_contracted[axis_b]) {
100 return LMMC_STATUS_INVALID_ARGUMENT;
101 }
102 if (a->dims[axis_a] != b->dims[axis_b] ||
103 !lmmc_safe_mul_size(
104 contraction->contract_size, a->dims[axis_a],
105 &contraction->contract_size)) {
106 return LMMC_STATUS_INVALID_ARGUMENT;
107 }
108 a_contracted[axis_a] = 1;
109 b_contracted[axis_b] = 1;
110 }
111 contraction->n_a_free = lmmc_tensor_nd_free_axes(a->ndim, a_contracted, contraction->a_free_axes);
112 contraction->n_b_free = lmmc_tensor_nd_free_axes(b->ndim, b_contracted, contraction->b_free_axes);
113 const size_t free_count = contraction->n_a_free + contraction->n_b_free;
115 *out_ndim = free_count == 0 ? 1 : free_count;
116 if (*out_ndim > LMMC_TENSOR_MAX_NDIM) {
117 return LMMC_STATUS_INVALID_ARGUMENT;
118 }
119
120 size_t j = 0;
121 for (size_t i = 0; i < contraction->n_a_free; ++i) {
122 out_dims[j++] = a->dims[contraction->a_free_axes[i]];
123 }
124 for (size_t i = 0; i < contraction->n_b_free; ++i) {
125 out_dims[j++] = b->dims[contraction->b_free_axes[i]];
126 }
127 if (free_count == 0) {
128 out_dims[0] = 1;
129 }
130 return LMMC_STATUS_OK;
131}

133static size_t lmmc_tensor_nd_contracted_offset(
134 const lmmc_tensor_nd_t* tensor, const size_t* free_axes, size_t n_free,
135 const size_t* out_idx, const size_t* contracted_axes, size_t naxes,
136 const size_t* contract_idx) {
137 size_t full_idx[LMMC_TENSOR_MAX_NDIM] = {0};
138 for (size_t i = 0; i < n_free; ++i) {
139 full_idx[free_axes[i]] = out_idx[i];
140 }
141 for (size_t i = 0; i < naxes; ++i) {
142 full_idx[contracted_axes[i]] = contract_idx[i];
143 }
144 return lmmc_tensor_nd_offset(tensor->ndim, tensor->strides, full_idx);
145}

147static void lmmc_tensor_nd_next_contracted_index(
148 const lmmc_tensor_nd_t* a, const lmmc_tensor_nd_contraction_t* contraction,
149 size_t* contract_idx) {

162static void lmmc_tensor_nd_contract_element(
163 const lmmc_tensor_nd_t* a, const lmmc_tensor_nd_t* b,
164 const lmmc_tensor_nd_contraction_t* contraction, size_t contract_size,
165 const size_t* out_idx, lmmc_real_t* sum) {
166 size_t contract_idx[LMMC_TENSOR_MAX_NDIM] = {0};
167 LMMC_REAL_SET_D(sum, 0.0);
168 for (size_t k = 0; k < contract_size; ++k) {
169 const size_t a_offset = lmmc_tensor_nd_contracted_offset(
170 a, contraction->a_free_axes, contraction->n_a_free, out_idx,
171 contraction->axes_a, contraction->naxes, contract_idx);
172 const size_t b_offset = lmmc_tensor_nd_contracted_offset(
173 b, contraction->b_free_axes, contraction->n_b_free,
174 out_idx + contraction->n_a_free, contraction->axes_b,
175 contraction->naxes, contract_idx);
176 lmmc_tensor_nd_add_product(sum, &a->data[a_offset], &b->data[b_offset]);
177 lmmc_tensor_nd_next_contracted_index(a, contraction, contract_idx);
178 }
179}

181static void lmmc_tensor_nd_execute_contraction(
182 const lmmc_tensor_nd_t* a, const lmmc_tensor_nd_t* b,
183 const lmmc_tensor_nd_contraction_t* contraction, lmmc_tensor_nd_t* out) {
184 size_t out_idx[LMMC_TENSOR_MAX_NDIM] = {0};
185 do {
186 lmmc_real_t sum;
187 LMMC_REAL_INIT(&sum);
188 lmmc_tensor_nd_contract_element(
189 a, b, contraction, contraction->contract_size, out_idx, &sum);
190 const size_t offset = lmmc_tensor_nd_offset(out->ndim, out->strides, out_idx);
191 LMMC_REAL_SET(&out->data[offset], &sum);
192 LMMC_REAL_CLEAR(&sum);
193 } while (lmmc_tensor_nd_next_index(out->ndim, out->dims, out_idx));
194}

196lmmc_status_t lmmc_tensor_nd_contract(const lmmc_tensor_nd_t* a, const lmmc_tensor_nd_t* b,
197 const size_t* axes_a, const size_t* axes_b, size_t naxes, lmmc_tensor_nd_t* out) {
198 lmmc_tensor_nd_contraction_t contraction;
199 lmmc_tensor_nd_t a_snapshot;
200 lmmc_tensor_nd_t b_snapshot;
201 lmmc_tensor_nd_t result = {0};
202 size_t out_ndim;
203 size_t out_dims[LMMC_TENSOR_MAX_NDIM];
204 contraction.naxes = naxes;
205 contraction.axes_a = axes_a;
206 contraction.axes_b = axes_b;
207 lmmc_status_t st = lmmc_tensor_nd_validate_contraction(a, b, &contraction, out);
208 if (st == LMMC_STATUS_OK) {
209 st = lmmc_tensor_nd_contraction_shape(a, b, &contraction, &out_ndim, out_dims);
210 }
211 if (st != LMMC_STATUS_OK) {
212 return st;
213 }
214 if (out == a) {
215 a_snapshot = *a;
216 a = &a_snapshot;
217 }
218 if (out == b) {
219 b_snapshot = *b;
220 b = &b_snapshot;
221 }
222 st = lmmc_tensor_nd_allocate_result(out_ndim, out_dims, &result);
223 if (st != LMMC_STATUS_OK) {
224 return st;
225 }
226 lmmc_tensor_nd_execute_contraction(a, b, &contraction, &result);

247static void lmmc_tensor_nd_mode_product_element(
248 const lmmc_tensor_nd_t* t, const lmmc_mat_t* mat, size_t mode,
249 const size_t* out_idx, lmmc_real_t* sum) {
250 const size_t j_dim = t->dims[mode];
251 LMMC_REAL_SET_D(sum, 0.0);
252 for (size_t i = 0; i < j_dim; ++i) {
254 size_t src_offset = 0;
255 for (size_t d = 0; d < t->ndim; ++d) {
256 if (d == mode) {
257 src_offset += i * t->strides[d];
258 } else {
259 src_offset += out_idx[d] * t->strides[d];
260 }
261 }
262 lmmc_tensor_nd_add_product(
263 sum, &mat->data[out_idx[mode] * mat->stride + i], &t->data[src_offset]);
264 }
265}
