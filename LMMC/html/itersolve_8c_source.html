103lmmc_status_t lmmc_itersolve_validate(
104 const lmmc_sparse_mat_t* a,
105 const lmmc_vec_t* b,
106 const lmmc_precond_t* precond,
107 const lmmc_itersolve_config_t* cfg,
108 lmmc_vec_t* x,
109 lmmc_itersolve_config_t* out_cfg,
110 lmmc_itersolve_result_t* out_result
111) {
112 lmmc_status_t st = lmmc_itersolve_square_dimensions(a, b, x);
113 if (st != LMMC_STATUS_OK) {
114 return st;
115 }
116
117 if (cfg != NULL) {
118 *out_cfg = *cfg;
119 } else {
120 st = lmmc_itersolve_default_config(b->size, out_cfg);
121 if (st != LMMC_STATUS_OK) {
122 goto cleanup;
123 }
124 }
125
126 if (!isfinite(out_cfg->abs_tol) || !isfinite(out_cfg->rel_tol) ||
127 out_cfg->abs_tol < 0.0 || out_cfg->rel_tol < 0.0 || out_cfg->max_iter == 0) {
128 return LMMC_STATUS_INVALID_ARGUMENT;
129 }
130
131 if (precond != NULL && precond->size != b->size) {
132 st = LMMC_STATUS_DIMENSION_MISMATCH;
133 goto cleanup;
134 }
135
136 if (out_result != NULL) {
137 out_result->converged = 0;
138 out_result->num_iter = 0;
139 LMMC_REAL_INIT(&out_result->initial_residual_norm);
140 LMMC_REAL_INIT(&out_result->final_residual_norm);
141 LMMC_REAL_SET_D(&out_result->initial_residual_norm, 0.0);
142 LMMC_REAL_SET_D(&out_result->final_residual_norm, 0.0);
143 }
144
145cleanup:
146 return st;
147}

149lmmc_status_t lmmc_itersolve_workspace(lmmc_vec_t* const* vectors,
150 const size_t* sizes, size_t count, lmmc_real_t** storage) {
151 size_t total = 0;
152 for (size_t i = 0; i < count; ++i) {
153 if (!lmmc_safe_add_size(total, sizes[i], &total)) {
154 return LMMC_STATUS_INVALID_ARGUMENT;
155 }
156 }
157 size_t bytes;
158 if (!lmmc_safe_mul_size(total, sizeof(lmmc_real_t), &bytes)) {
159 return LMMC_STATUS_INVALID_ARGUMENT;
160 }
161 *storage = lmmc_memory_alloc(bytes);
162 if (*storage == NULL) {
163 return LMMC_STATUS_ALLOCATION_FAILED;
164 }
165 memset(*storage, 0, bytes);
166 lmmc_real_t* next = *storage;
167 for (size_t i = 0; i < count; ++i) {
168 *vectors[i] = (lmmc_vec_t){sizes[i], next, 0};
169 next += sizes[i];
170 }
171 return LMMC_STATUS_OK;
172}

267static lmmc_status_t lmmc_matvec_transpose_csr(
268 const lmmc_sparse_mat_t* a,
269 const lmmc_vec_t* x_in,
270 lmmc_vec_t* y_out
271) {
272 size_t i, j;
273 for (i = 0; i < y_out->size; ++i) {
274 y_out->data[i] = 0.0;
275 }
276 for (i = 0; i < a->rows; ++i) {
277 size_t start = a->row_ptr[i];
278 size_t end = a->row_ptr[i + 1];
279 for (j = start; j < end; ++j) {
280 size_t col = a->col_idx[j];
281 y_out->data[col] += a->values[j] * x_in->data[i];
282 }
283 }
284 return LMMC_STATUS_OK;
