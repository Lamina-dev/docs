16void householder_make(lmmc_real_t *x, size_t len, lmmc_real_t *tau_out,
17 lmmc_real_t *beta_out) {
18 size_t i;
19 if (len == 0) { *tau_out = 0.0; *beta_out = 0.0; return; }
20 if (len == 1) { *tau_out = 0.0; *beta_out = x[0]; return; }
21 lmmc_real_t sigma = 0.0;
22 for (i = 1; i < len; i++) sigma += x[i] * x[i];
23 lmmc_real_t alpha = x[0];
24 if (sigma == 0.0) { *tau_out = 0.0; *beta_out = alpha; return; }
25 lmmc_real_t mu = sqrt(alpha * alpha + sigma);
26
27 lmmc_real_t beta = (alpha >= 0.0) ? -mu : mu;
28 lmmc_real_t v0 = alpha - beta;
29 *beta_out = beta;
30
31 lmmc_real_t v0_sq = v0 * v0;
32 *tau_out = 2.0 * v0_sq / (sigma + v0_sq);
33 lmmc_real_t inv = 1.0 / v0;
34 for (i = 1; i < len; i++) x[i] *= inv;
35 x[0] = 1.0;
36}

38void householder_apply_left(lmmc_mat_t *M, size_t i0, size_t len,
39 size_t j0, size_t j1,
40 const lmmc_real_t *v, lmmc_real_t tau) {
41 size_t i, j;
42 if (tau == 0.0) return;
43 for (j = j0; j < j1; j++) {
44 lmmc_real_t s = MAT_ELEM(M, i0, j);
45 for (i = 1; i < len; i++) s += v[i] * MAT_ELEM(M, i0 + i, j);
46 s *= tau;
47 MAT_ELEM(M, i0, j) -= s;
48 for (i = 1; i < len; i++) MAT_ELEM(M, i0 + i, j) -= s * v[i];
49 }
50}

52void householder_apply_right(lmmc_mat_t *M, size_t i0, size_t i1,
53 size_t j0, size_t len,
54 const lmmc_real_t *v, lmmc_real_t tau) {
55 size_t i, j;
56 if (tau == 0.0) return;
57 for (i = i0; i < i1; i++) {
58 lmmc_real_t s = MAT_ELEM(M, i, j0);
59 for (j = 1; j < len; j++) s += v[j] * MAT_ELEM(M, i, j0 + j);
60 s *= tau;
61 MAT_ELEM(M, i, j0) -= s;
62 for (j = 1; j < len; j++) MAT_ELEM(M, i, j0 + j) -= s * v[j];
63 }

66void givens_right(lmmc_mat_t *M, size_t rows, size_t p, size_t q,
67 lmmc_real_t c, lmmc_real_t s) {
68 size_t i;
69 for (i = 0; i < rows; i++) {
70 lmmc_real_t mp = MAT_ELEM(M, i, p);
71 lmmc_real_t mq = MAT_ELEM(M, i, q);
72 MAT_ELEM(M, i, p) = c * mp + s * mq;
73 MAT_ELEM(M, i, q) = -s * mp + c * mq;
74 }
75}

77void givens_compute(lmmc_real_t a, lmmc_real_t b,
78 lmmc_real_t *c, lmmc_real_t *s, lmmc_real_t *r) {
79 if (b == 0.0) { *c = (a >= 0.0) ? 1.0 : -1.0; *s = 0.0; *r = lmmc_abs(a); }
80 else if (a == 0.0) { *c = 0.0; *s = (b >= 0.0) ? 1.0 : -1.0; *r = lmmc_abs(b); }
81 else {
82 lmmc_real_t h = sqrt(a * a + b * b);
83 *r = h;
84 *c = a / h;
85 *s = b / h;
86 }
