30static lmmc_status_t lmmc_lsqr_start(const lmmc_sparse_mat_t* a,
31 const lmmc_vec_t* b, const lmmc_itersolve_config_t* cfg,
32 lmmc_vec_t* x, lmmc_lsqr_state_t* s) {
33 const size_t m = b->size;
34 lmmc_status_t st;
36 st = lmmc_itersolve_matvec(a, cfg, x, &s->av_tmp);
37 if (st != LMMC_STATUS_OK) {
38 return st;
39 }
40 {
41 size_t i;
42 for (i = 0; i < m; ++i) {
43 s->u.data[i] = b->data[i] - s->av_tmp.data[i];
44 }
45 }
46
48 st = lmmc_vec_norm2_checked(&s->u, &s->beta_l);
49 if (st != LMMC_STATUS_OK) {
50 return st;
51 }
52
53 st = lmmc_vec_norm2_checked(b, &s->norm_b_val);
54 if (st != LMMC_STATUS_OK) {
55 return st;
56 }
57
58 s->norm_r = s->beta_l;
59 s->threshold = cfg->abs_tol + cfg->rel_tol * s->norm_b_val;
60 return LMMC_STATUS_OK;

63static lmmc_status_t lmmc_lsqr_basis(const lmmc_sparse_mat_t* a,
64 const lmmc_itersolve_config_t* cfg, lmmc_lsqr_state_t* s, int* converged) {
65 const size_t m = s->u.size;
66 const size_t n_cols = s->v.size;
67 lmmc_status_t st;
68 {
69 size_t i;
70 for (i = 0; i < m; ++i) {
71 s->u.data[i] /= s->beta_l;
72 }
73 }
74

77 if (st != LMMC_STATUS_OK) {
78 return st;
79 }

83 if (st != LMMC_STATUS_OK) {
84 return st;

88 *converged = 1;
89 return st;
90 }
91
92 {
93 size_t i;
94 for (i = 0; i < n_cols; ++i) {
95 s->v.data[i] /= s->alpha_l;
96 }
97 }
98
99 st = lmmc_vec_copy(&s->v, &s->w);
100 if (st != LMMC_STATUS_OK) {
101 return st;
102 }
103
104 s->rho_bar = s->alpha_l;
105 s->phi_bar_l = s->beta_l;
106 return LMMC_STATUS_OK;

166static lmmc_status_t lmmc_lsqr_rotate(lmmc_vec_t* x, lmmc_lsqr_state_t* s) {
167 const size_t n_cols = x->size;
168 size_t i;
169 double scale;
170 lmmc_status_t st;
172 s->rho_val = sqrt(s->rho_bar * s->rho_bar + s->beta_l * s->beta_l);
173 if (s->rho_val <= 1e-30) {
174 st = LMMC_STATUS_NUMERICAL_FAILURE;
175 return st;
176 }
177
178 s->cs_l = s->rho_bar / s->rho_val;
179 s->sn_l = s->beta_l / s->rho_val;
180 s->theta_l = s->sn_l * s->alpha_l;
181 s->rho_bar = -s->cs_l * s->alpha_l;
182 s->phi_l = s->cs_l * s->phi_bar_l;
183 s->phi_bar_l = s->sn_l * s->phi_bar_l;
184
186 scale = s->phi_l / s->rho_val;
187 for (i = 0; i < n_cols; ++i) {
188 x->data[i] += scale * s->w.data[i];
189 if (!isfinite(x->data[i])) {
190 st = LMMC_STATUS_NUMERICAL_FAILURE;
191 return st;
192 }
193 }
194
196 scale = s->theta_l / s->rho_val;
197 for (i = 0; i < n_cols; ++i) {
198 s->w.data[i] = s->v.data[i] - scale * s->w.data[i];
199 }
200 return LMMC_STATUS_OK;
201}

203static lmmc_status_t lmmc_lsqr_iterate(const lmmc_sparse_mat_t* a,
204 const lmmc_itersolve_config_t* cfg, lmmc_vec_t* x,
205 lmmc_lsqr_state_t* s, lmmc_itersolve_result_t* result,
206 size_t* iterations, int* converged) {
207 lmmc_status_t st = lmmc_lsqr_basis(a, cfg, s, converged);
208 if (st != LMMC_STATUS_OK || *converged) {
209 return st;
210 }
211 for (size_t iter = 0; iter < cfg->max_iter; ++iter) {
212 st = lmmc_lsqr_bidiagonalize(a, cfg, s, converged);
213 if (st != LMMC_STATUS_OK) {
214 return st;
215 }
216 if (*converged) {
217 *iterations = iter + 1;
218 break;
219 }
220 st = lmmc_lsqr_rotate(x, s);
221 if (st != LMMC_STATUS_OK) {
222 return st;
223 }
224 s->norm_r = fabs(s->phi_bar_l);
225 *iterations = iter + 1;
226 if (result != NULL) {
227 result->final_residual_norm = s->norm_r;
228 }
229 lmmc_itersolve_do_log(cfg, *iterations, s->norm_r);
230 if (s->norm_r <= s->threshold) {
231 *converged = 1;
232 break;
233 }
234 }
235 return LMMC_STATUS_OK;
236}
