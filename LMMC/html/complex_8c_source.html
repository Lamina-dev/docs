65lmmc_status_t lmmc_complex_mul(const lmmc_complex_t* a, const lmmc_complex_t* b, lmmc_complex_t* out)
66{
67 lmmc_real_t ac, bd, ad, bc;
68
69 if (a == NULL || b == NULL || out == NULL) {
70 return LMMC_STATUS_INVALID_ARGUMENT;
71 }
72
73 /* (a.real * b.real - a.imag * b.imag) + i*(a.real * b.imag + a.imag * b.real) */
74 LMMC_REAL_MUL(&ac, &a->real, &b->real);
75 LMMC_REAL_MUL(&bd, &a->imag, &b->imag);
76 LMMC_REAL_MUL(&ad, &a->real, &b->imag);
77 LMMC_REAL_MUL(&bc, &a->imag, &b->real);
78
79 LMMC_REAL_SUB(&out->real, &ac, &bd);
80 LMMC_REAL_ADD(&out->imag, &ad, &bc);
81 return LMMC_STATUS_OK;
82}

84lmmc_status_t lmmc_complex_div(const lmmc_complex_t* a, const lmmc_complex_t* b, lmmc_complex_t* out)
85{
86 lmmc_real_t zero;
87 lmmc_real_t abs_c, abs_d;
88 lmmc_real_t r, denom, tmp1, tmp2;
89
90 if (a == NULL || b == NULL || out == NULL) {
91 return LMMC_STATUS_INVALID_ARGUMENT;
92 }
93
95 LMMC_REAL_SET_D(&zero, 0.0);
96 if (LMMC_REAL_CMP(&b->real, &zero) == 0 && LMMC_REAL_CMP(&b->imag, &zero) == 0) {
97 return LMMC_STATUS_NUMERICAL_FAILURE;
98 }
99
101 LMMC_REAL_ABS(&abs_c, &b->real);
102 LMMC_REAL_ABS(&abs_d, &b->imag);
103
104 if (LMMC_REAL_CMP(&abs_d, &abs_c) <= 0) {
106 LMMC_REAL_DIV(&r, &b->imag, &b->real);
107 LMMC_REAL_MUL(&tmp1, &b->imag, &r);
108 LMMC_REAL_ADD(&denom, &b->real, &tmp1);
109
111 LMMC_REAL_MUL(&tmp1, &a->imag, &r);
112 LMMC_REAL_ADD(&tmp2, &a->real, &tmp1);
113 LMMC_REAL_DIV(&out->real, &tmp2, &denom);
114
116 LMMC_REAL_MUL(&tmp1, &a->real, &r);
117 LMMC_REAL_SUB(&tmp2, &a->imag, &tmp1);
118 LMMC_REAL_DIV(&out->imag, &tmp2, &denom);
119 } else {
121 LMMC_REAL_DIV(&r, &b->real, &b->imag);
122 LMMC_REAL_MUL(&tmp1, &b->real, &r);
123 LMMC_REAL_ADD(&denom, &b->imag, &tmp1);
124
126 LMMC_REAL_MUL(&tmp1, &a->real, &r);
127 LMMC_REAL_ADD(&tmp2, &tmp1, &a->imag);
128 LMMC_REAL_DIV(&out->real, &tmp2, &denom);
129
131 LMMC_REAL_MUL(&tmp1, &a->imag, &r);
132 LMMC_REAL_SUB(&tmp2, &tmp1, &a->real);
133 LMMC_REAL_DIV(&out->imag, &tmp2, &denom);
134 }
135
136 return LMMC_STATUS_OK;
137}

293lmmc_status_t lmmc_complex_pow(const lmmc_complex_t* base, const lmmc_complex_t* exp, lmmc_complex_t* out)
294{
295 lmmc_real_t zero;
296 lmmc_complex_t log_base, exp_times_log;
297 lmmc_status_t st;
298
299 if (base == NULL || exp == NULL || out == NULL) {
300 return LMMC_STATUS_INVALID_ARGUMENT;
301 }
302
303 /* Check if base == 0+0i and exp.real < 0 */
304 LMMC_REAL_SET_D(&zero, 0.0);
305 if (LMMC_REAL_CMP(&base->real, &zero) == 0 && LMMC_REAL_CMP(&base->imag, &zero) == 0) {
306 if (LMMC_REAL_CMP(&exp->real, &zero) < 0) {
307 return LMMC_STATUS_OUT_OF_RANGE;
308 }
309 if (LMMC_REAL_CMP(&exp->real, &zero) == 0 && LMMC_REAL_CMP(&exp->imag, &zero) == 0) {
310 /* 0^0 = 1+0i (cpow convention) */
311 LMMC_REAL_SET_D(&out->real, 1.0);
312 LMMC_REAL_SET_D(&out->imag, 0.0);
313 return LMMC_STATUS_OK;
314 }
315 /* 0^(positive) = 0+0i */
316 LMMC_REAL_SET_D(&out->real, 0.0);
317 LMMC_REAL_SET_D(&out->imag, 0.0);
318 return LMMC_STATUS_OK;
319 }
320
321 /* pow(base, exp) = exp(exp * log(base)) */
322 st = lmmc_complex_log(base, &log_base);
323 if (st != LMMC_STATUS_OK) {
324 return st;
325 }
326
327 st = lmmc_complex_mul(exp, &log_base, &exp_times_log);
328 if (st != LMMC_STATUS_OK) {
329 return st;
330 }
331
332 return lmmc_complex_exp(&exp_times_log, out);
333}
