31static lmmc_real_t lm_gradient(lmmc_lm_workspace_t* work)
32{
33 const size_t n = work->gradient.size;
34 size_t i, k;
35 for (i = 0; i < n; ++i) {
36 work->gradient.data[i] = 0.0;
37 for (k = 0; k < n; ++k) {
38 work->gradient.data[i] += work->jacobian.data[k * work->jacobian.stride + i] *
39 work->residual.data[k];
40 }
41 }
42 return lmmc_optimize_norm(&work->gradient);

45static void lm_gram(lmmc_lm_workspace_t* work)
46{
47 const size_t n = work->gradient.size;
48 size_t i, j, k;
49 for (i = 0; i < n; ++i) {
50 for (j = 0; j < n; ++j) {
51 lmmc_real_t sum = 0.0;
52 for (k = 0; k < n; ++k) {
53 sum += work->jacobian.data[k * work->jacobian.stride + i] *
54 work->jacobian.data[k * work->jacobian.stride + j];
55 }
56 work->gram.data[i * work->gram.stride + j] = sum;
57 }
58 }

61static lmmc_lm_trial_t lm_trial_direction(
62 lmmc_lm_workspace_t* work, lmmc_real_t lambda)
63{
64 const size_t n = work->gradient.size;
65 lmmc_status_t status;
66 size_t i;
67 memcpy(work->augmented.data, work->gram.data, n * n * sizeof(lmmc_real_t));
68 for (i = 0; i < n; ++i) {
69 work->augmented.data[i * work->augmented.stride + i] += lambda;
70 }
71 status = lmmc_lu_decompose_inplace(&work->augmented, work->arena.pivots, NULL);
72 if (status == LMMC_STATUS_SINGULAR_MATRIX) { return LMMC_LM_TRIAL_RETRY; }
73 if (status != LMMC_STATUS_OK) { return LMMC_LM_TRIAL_FAILURE; }
74 for (i = 0; i < n; ++i) {
75 work->rhs.data[i] = -work->gradient.data[i];
76 }
77 status = lmmc_lu_solve(&work->augmented, work->arena.pivots,
78 &work->rhs, &work->delta);
79 return status == LMMC_STATUS_OK ? LMMC_LM_TRIAL_READY : LMMC_LM_TRIAL_RETRY;
80}

82static int lm_step(
83 lmmc_lm_workspace_t* work, lmmc_opt_func_t residual, void* user_data,
84 lmmc_vec_t* x, lmmc_real_t* lambda, lmmc_real_t* norm)
85{
86 int attempt;
87 size_t i;
88 lm_gram(work);
89 for (attempt = 0; attempt < 20; ++attempt) {
90 lmmc_real_t next_norm;
91 lmmc_lm_trial_t trial = lm_trial_direction(work, *lambda);
92 if (trial == LMMC_LM_TRIAL_FAILURE) { return 0; }
93 if (trial == LMMC_LM_TRIAL_RETRY) {
94 *lambda *= 10.0;
95 continue;
96 }
97 for (i = 0; i < x->size; ++i) {
98 work->trial.data[i] = x->data[i] + work->delta.data[i];
99 }
100 if (residual(&work->trial, &work->next_residual, user_data) != LMMC_STATUS_OK) {
101 *lambda *= 10.0;
102 continue;
103 }
104 next_norm = lmmc_optimize_norm(&work->next_residual);
105 if (next_norm < *norm) {
106 memcpy(x->data, work->trial.data, x->size * sizeof(lmmc_real_t));
107 memcpy(work->residual.data, work->next_residual.data,
108 x->size * sizeof(lmmc_real_t));
109 *norm = next_norm;
110 *lambda *= 0.1;
111 if (*lambda < 1e-15) *lambda = 1e-15;
112 return 1;
113 }
114 *lambda *= 10.0;
115 if (*lambda > 1e15) { return 0; }
116 }
117 return 0;
118}

120static int lm_stop(lmmc_real_t norm, lmmc_real_t grad_norm,
121 lmmc_real_t initial_residual, lmmc_real_t* initial_gradient,
122 size_t iter, const lmmc_optimize_config_t* cfg, lmmc_optimize_result_t* out)
123{
124 if (!isfinite(grad_norm)) {
125 out->failure_reason = LMMC_OPT_FAILURE_NUMERICAL_ISSUE;
126 return 1;
127 }
128 if (iter == 0) *initial_gradient = grad_norm;
129 if (lmmc_optimize_has_converged(norm, initial_residual, cfg) ||
130 lmmc_optimize_has_converged(grad_norm, *initial_gradient, cfg)) {
131 out->converged = 1;
132 return 1;
133 }
134 if (iter == cfg->max_iter) {
135 out->failure_reason = LMMC_OPT_FAILURE_MAX_ITER;
136 return 1;
137 }
138 return 0;
139}

149 size_t iter;

159 goto cleanup;
160 }
161 norm = lmmc_optimize_norm(&work.residual);
162 initial_residual = norm;

164 lmmc_real_t grad_norm;
165 out->final_residual = norm;
166 if (!isfinite(norm)) {
167 out->failure_reason = LMMC_OPT_FAILURE_NUMERICAL_ISSUE;
168 break;
169 }
171 status = lmmc_optimize_jacobian(residual, J, user_data, x,
172 &work.residual, &work.jacobian, &work.trial, &work.next_residual);
173 if (status != LMMC_STATUS_OK) {
174 out->failure_reason = LMMC_OPT_FAILURE_NUMERICAL_ISSUE;

184 out->num_iter = iter + 1;

187 lmmc_optimize_emit(cfg, "levenberg_marquardt", iter + 1, values, 2);
