113 size_t max_idx = 0;
114
115 LMMC_REAL_ABS(&max_val, &x->data[0]);
116
117 for (size_t i = 1; i < x->size; ++i) {
118 LMMC_REAL_ABS(&abs_val, &x->data[i]);
119 if (LMMC_REAL_CMP(&abs_val, &max_val) > 0) {
120 LMMC_REAL_SET(&max_val, &abs_val);

125
126 LMMC_REAL_CLEAR(&max_val);
