53static void descending_order(const lmmc_real_t *d, size_t n, size_t *order)
54{
55 for (size_t i = 0; i < n; ++i) order[i] = i;
56 for (size_t i = 0; i + 1 < n; ++i) {
57 size_t maximum = i;
58 for (size_t j = i + 1; j < n; ++j) {
59 if (d[order[j]] > d[order[maximum]]) maximum = j;
60 }
61 if (maximum != i) {
62 size_t tmp = order[i];
63 order[i] = order[maximum];
64 order[maximum] = tmp;
65 }
66 }
67}

69static lmmc_status_t reorder_left_vectors(lmmc_mat_t *u, size_t n,

89static lmmc_status_t assemble_factors(svd_workspace_t *work,
90 lmmc_svd_result_t *result)
91{
92 const size_t n = result->sigma.size;
93 normalize_signs(work->d, &work->v);
94 size_t *order = lmmc_memory_alloc_array(n, sizeof(*order));
95 if (!order) return LMMC_STATUS_ALLOCATION_FAILED;
96 descending_order(work->d, n, order);
97 for (size_t i = 0; i < n; ++i) {
98 result->sigma.data[i] = work->d[order[i]];
99 for (size_t k = 0; k < n; ++k) {
100 MAT_ELEM(&result->Vt, i, k) = MAT_ELEM(&work->v, k, order[i]);
101 }
102 }
103 lmmc_status_t status = reorder_left_vectors(&result->U, n, order);
104 lmmc_memory_free(order);
105 return status;
106}

108lmmc_status_t lmmc_svd_tall(const lmmc_mat_t *a, lmmc_real_t input_scale,
109 lmmc_svd_result_t *result)
110{
111 svd_workspace_t work = {0};
112 lmmc_svd_result_t factors = {0};
113 lmmc_status_t status = create_factors(a->rows, a->cols, &factors);
114 if (status != LMMC_STATUS_OK) goto cleanup;
115 status = create_workspace(a->cols, &work);
116 if (status != LMMC_STATUS_OK) goto cleanup;
117 status = lmmc_svd_bidiagonalize(a, input_scale, work.d, work.e,
118 &factors.U, &work.v);
119 if (status != LMMC_STATUS_OK) goto cleanup;
120 status = lmmc_svd_diagonalize(work.d, work.e, a->cols, &factors.U, &work.v);
121 if (status != LMMC_STATUS_OK) goto cleanup;
122 status = assemble_factors(&work, &factors);
123cleanup:
124 destroy_workspace(&work);
125 if (status == LMMC_STATUS_OK) {
126 *result = factors;
127 } else {
128 lmmc_svd_result_destroy(&factors);
129 }
