220lmmc_status_t lmmc_lsr_linalg_rank(const lmmc_mat_t* a,
221 size_t* out_rank)
222{
223 size_t rows, cols, count, rank, col;
224 lmmc_real_t* work;
225 lmmc_real_t scale = (lmmc_real_t)0;
226 lmmc_real_t tol;
227
228 if (!out_rank) return LMMC_STATUS_INVALID_ARGUMENT;
229 lmmc_status_t input_status = lmmc_lsr_require_finite_mat(a);
230 if (input_status != LMMC_STATUS_OK) return input_status;
231 if (a->rows != 0 && a->cols > ((size_t)-1) / a->rows) {
232 return LMMC_STATUS_INVALID_ARGUMENT;
233 }
234
235 rows = a->rows;
236 cols = a->cols;
237 count = rows * cols;
238 work = (lmmc_real_t*)malloc(count * sizeof(lmmc_real_t));
239 if (!work) return LMMC_STATUS_ALLOCATION_FAILED;
240
241 for (size_t i = 0; i < rows; ++i) {
242 for (size_t j = 0; j < cols; ++j) {
243 lmmc_real_t value = a->data[i * a->stride + j];
244 work[i * cols + j] = value;
245 if (fabs((double)value) > fabs((double)scale)) scale = value;
246 }
247 }
248
249 if (scale == (lmmc_real_t)0) {
250 free(work);
251 *out_rank = 0;
252 return LMMC_STATUS_OK;
253 }
254 tol = (lmmc_real_t)(1e-12 * fabs((double)scale) *
255 (double)(rows > cols ? rows : cols));
256
257 rank = 0;
258 for (col = 0; col < cols && rank < rows; ++col) {
259 size_t pivot = rank;
260 lmmc_real_t pivot_abs =
261 (lmmc_real_t)fabs((double)work[pivot * cols + col]);
262 for (size_t row = rank + 1; row < rows; ++row) {
263 lmmc_real_t candidate =
264 (lmmc_real_t)fabs((double)work[row * cols + col]);
265 if (candidate > pivot_abs) {
266 pivot = row;
267 pivot_abs = candidate;
268 }
269 }
270 if (pivot_abs <= tol) continue;
271
272 if (pivot != rank) {
273 for (size_t j = col; j < cols; ++j) {
274 lmmc_real_t tmp = work[rank * cols + j];
275 work[rank * cols + j] = work[pivot * cols + j];
276 work[pivot * cols + j] = tmp;
277 }
278 }
279
280 for (size_t row = rank + 1; row < rows; ++row) {
281 lmmc_real_t factor = work[row * cols + col] /
282 work[rank * cols + col];
283 work[row * cols + col] = (lmmc_real_t)0;
284 for (size_t j = col + 1; j < cols; ++j) {
285 work[row * cols + j] -= factor * work[rank * cols + j];
286 }
287 }
288 ++rank;
289 }
290
291 free(work);
292 *out_rank = rank;
293 return LMMC_STATUS_OK;
294}

309lmmc_status_t lmmc_lsr_linalg_solve_left(const lmmc_mat_t* a,
310 const lmmc_mat_t* b,
311 lmmc_mat_t* out)
312{
313 lmmc_mat_t lu = {0};
314 lmmc_vec_t rhs = {0};
315 lmmc_vec_t sol = {0};
316 size_t* pivots = NULL;
317 lmmc_status_t status;
318
319 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
320 status = lmmc_lsr_require_finite_mat(a);
321 if (status != LMMC_STATUS_OK) return status;
322 status = lmmc_lsr_require_finite_mat(b);
323 if (status != LMMC_STATUS_OK) return status;
324 if (a->rows != a->cols || b->rows != a->rows) {
325 return LMMC_STATUS_DIMENSION_MISMATCH;
326 }
327
328 status = lmmc_mat_create(a->rows, a->cols, &lu);
329 if (status != LMMC_STATUS_OK) return status;
330 status = lmmc_mat_copy(a, &lu);
331 if (status != LMMC_STATUS_OK) goto cleanup;
332
333 pivots = (size_t*)malloc(a->rows * sizeof(size_t));
334 if (!pivots) {
335 status = LMMC_STATUS_ALLOCATION_FAILED;
336 goto cleanup;
337 }
338 status = lmmc_lu_decompose_inplace(&lu, pivots, NULL);
339 if (status != LMMC_STATUS_OK) goto cleanup;
340
341 status = lmmc_mat_create(a->cols, b->cols, out);
342 if (status != LMMC_STATUS_OK) goto cleanup;
343 status = lmmc_vec_create(b->rows, &rhs);
344 if (status != LMMC_STATUS_OK) {
345 lmmc_mat_destroy(out);
346 goto cleanup;
347 }
348 status = lmmc_vec_create(a->cols, &sol);
349 if (status != LMMC_STATUS_OK) {
350 lmmc_mat_destroy(out);
351 goto cleanup;
352 }
353
354 for (size_t col = 0; col < b->cols; ++col) {
355 for (size_t row = 0; row < b->rows; ++row) {
356 rhs.data[row] = b->data[row * b->stride + col];
357 }
358 status = lmmc_lu_solve(&lu, pivots, &rhs, &sol);
359 if (status != LMMC_STATUS_OK) {
360 lmmc_mat_destroy(out);
361 goto cleanup;
362 }
363 for (size_t row = 0; row < a->cols; ++row) {
364 out->data[row * out->stride + col] = sol.data[row];
365 }
366 }
367
368cleanup:
369 lmmc_vec_destroy(&sol);
370 lmmc_vec_destroy(&rhs);
371 if (pivots) free(pivots);
372 lmmc_mat_destroy(&lu);
373 return status;
374}
