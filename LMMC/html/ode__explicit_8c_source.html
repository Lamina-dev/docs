13 lmmc_ode_rhs_t rhs,
14 void* user_data,
15 size_t dim,

20 lmmc_ode_result_t* out_result
21) {

23 size_t work_bytes = 0;

26 lmmc_real_t h = 0.0;

29 init_st = validate_and_init_ode_config(rhs, dim, t_start, t_end, y, cfg, &local_cfg, out_result, &work_bytes);

40 h = lmmc_clamp(local_cfg.initial_step, local_cfg.min_step, local_cfg.max_step);

88 lmmc_free(y_prime);

95 }
96
97 if (t >= t_end) {
98 out_result->converged = 1;
99 out_result->failure_reason = LMMC_ODE_FAILURE_NONE;
100 } else {
101 out_result->converged = 0;
102 out_result->failure_reason = LMMC_ODE_FAILURE_MAX_STEPS;
103 }
104
105 lmmc_free(y_prime);
106 return out_result->converged ? LMMC_STATUS_OK : LMMC_STATUS_CONVERGENCE_FAILED;

109/*
110 * Cash-Karp embedded Runge-Kutta 4(5) pair.
111 * 6 stages, 4th-order solution for stepping, 5th-order for error estimation.
112 *
113 * Butcher tableau nodes (c_i):
114 */
115static const lmmc_real_t ck_c[6] = {
116 0.0,
117 1.0 / 5.0,
118 3.0 / 10.0,
119 3.0 / 5.0,
120 1.0,
121 7.0 / 8.0
122};
123
124/*
125 * Butcher tableau matrix (a_ij), stored row-major.
126 * a[i][j] for j < i. Only non-zero entries listed per row.
127 */
128static const lmmc_real_t ck_a21 = 1.0 / 5.0;
129
130static const lmmc_real_t ck_a31 = 3.0 / 40.0;
131static const lmmc_real_t ck_a32 = 9.0 / 40.0;
132
133static const lmmc_real_t ck_a41 = 3.0 / 10.0;
134static const lmmc_real_t ck_a42 = -9.0 / 10.0;
135static const lmmc_real_t ck_a43 = 6.0 / 5.0;
136
137static const lmmc_real_t ck_a51 = -11.0 / 54.0;
138static const lmmc_real_t ck_a52 = 5.0 / 2.0;
139static const lmmc_real_t ck_a53 = -70.0 / 27.0;
140static const lmmc_real_t ck_a54 = 35.0 / 27.0;
141
142static const lmmc_real_t ck_a61 = 1631.0 / 55296.0;
143static const lmmc_real_t ck_a62 = 175.0 / 512.0;
144static const lmmc_real_t ck_a63 = 575.0 / 13824.0;
145static const lmmc_real_t ck_a64 = 44275.0 / 110592.0;
146static const lmmc_real_t ck_a65 = 253.0 / 4096.0;
147
148
149/* 5th-order weights (for the error estimation solution) */
150static const lmmc_real_t ck_b5[6] = {
151 37.0 / 378.0,
152 0.0,
153 250.0 / 621.0,
154 125.0 / 594.0,
155 0.0,
156 512.0 / 1771.0
157};
158
159/* Error coefficients: e_i = b5_i - b4_i */
160static const lmmc_real_t ck_e[6] = {
161 37.0 / 378.0 - 2825.0 / 27648.0,
162 0.0,
163 250.0 / 621.0 - 18575.0 / 48384.0,
164 125.0 / 594.0 - 13525.0 / 55296.0,
165 0.0 - 277.0 / 14336.0,
166 512.0 / 1771.0 - 1.0 / 4.0
167};

169lmmc_status_t lmmc_ode_rk45_solve(
170 lmmc_ode_rhs_t rhs,
171 void* user_data,
172 size_t dim,
173 lmmc_real_t t_start,
174 lmmc_real_t t_end,
175 lmmc_real_t* y,
176 const lmmc_ode_config_t* cfg,
177 lmmc_ode_result_t* out_result
178) {
179 lmmc_ode_config_t local_cfg = {0};
180 size_t work_bytes = 0;
181 lmmc_real_t* k1 = NULL;
182 lmmc_real_t* k2 = NULL;
183 lmmc_real_t* k3 = NULL;
184 lmmc_real_t* k4 = NULL;
185 lmmc_real_t* k5 = NULL;
186 lmmc_real_t* k6 = NULL;
187 lmmc_real_t* y_tmp = NULL;
188 lmmc_real_t t = t_start;
189 lmmc_real_t h = 0.0;
190 lmmc_status_t init_st;
191 size_t attempts = 0;
192 lmmc_status_t failure_status = LMMC_STATUS_NUMERICAL_FAILURE;
193
194 /* Pre-flight validation */
195 init_st = validate_and_init_ode_config(rhs, dim, t_start, t_end, y, cfg, &local_cfg, out_result, &work_bytes);
196 if (init_st != LMMC_STATUS_OK) {
197 return init_st;
198 }
199
200 k1 = (lmmc_real_t*)lmmc_alloc(work_bytes);
201 k2 = (lmmc_real_t*)lmmc_alloc(work_bytes);
202 k3 = (lmmc_real_t*)lmmc_alloc(work_bytes);
203 k4 = (lmmc_real_t*)lmmc_alloc(work_bytes);
204 k5 = (lmmc_real_t*)lmmc_alloc(work_bytes);
205 k6 = (lmmc_real_t*)lmmc_alloc(work_bytes);
206 y_tmp = (lmmc_real_t*)lmmc_alloc(work_bytes);
207
208 if (k1 == NULL || k2 == NULL || k3 == NULL || k4 == NULL ||
209 k5 == NULL || k6 == NULL || y_tmp == NULL) {
210 lmmc_free(y_tmp);
211 lmmc_free(k6);
212 lmmc_free(k5);
213 lmmc_free(k4);
214 lmmc_free(k3);
215 lmmc_free(k2);
216 lmmc_free(k1);
217 out_result->failure_reason = LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
218 return LMMC_STATUS_ALLOCATION_FAILED;
219 }
220
221 /* Initial step size */
222 h = lmmc_clamp(local_cfg.initial_step, local_cfg.min_step, local_cfg.max_step);
223 {
224 lmmc_real_t span;
225 LMMC_REAL_SUB(&span, &t_end, &t_start);
226 if (h > span) {
227 h = span;
228 }
229 }
230
231 /* Log initial state */
232 lmmc_ode_do_log(&local_cfg, 0, t, y, dim);
233
234 /* Main integration loop */
235 while (t < t_end && attempts < local_cfg.max_steps) {
236 size_t i = 0;
237 lmmc_real_t rem;
238 int callback_failed = 0;
239 lmmc_status_t st = LMMC_STATUS_OK;
240 lmmc_real_t err_norm = 0.0;
241 lmmc_real_t h_new = 0.0;
242 int step_accepted = 0;
243 ++attempts;
244
245 LMMC_REAL_SUB(&rem, &t_end, &t);
246 if (rem <= 0.0 || !lmmc_is_finite(&rem)) {
247 break;
248 }
249
250 if (h > rem) {
251 h = rem;
252 }
253
254 /* --- Stage 1: k1 = f(t, y) --- */
255 st = lmmc_ode_rhs_eval(rhs, t, y, k1, dim, user_data,
256 &out_result->num_rhs_evals, &callback_failed);
257 if (st != LMMC_STATUS_OK) {
258 out_result->failure_reason = callback_failed ?
259 LMMC_ODE_FAILURE_RHS_EVAL_FAILED : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
260 failure_status = st;
261 goto rk45_fail;
262 }
263
264 /* --- Stage 2: k2 = f(t + c2*h, y + h*a21*k1) --- */
265 for (i = 0; i < dim; ++i) {
266 y_tmp[i] = y[i] + h * ck_a21 * k1[i];
267 }
268 st = lmmc_ode_rhs_eval(rhs, t + ck_c[1] * h, y_tmp, k2, dim, user_data,
269 &out_result->num_rhs_evals, &callback_failed);
270 if (st != LMMC_STATUS_OK) {
271 out_result->failure_reason = callback_failed ?
272 LMMC_ODE_FAILURE_RHS_EVAL_FAILED : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
273 failure_status = st;
274 goto rk45_fail;
275 }
276
277 /* --- Stage 3: k3 = f(t + c3*h, y + h*(a31*k1 + a32*k2)) --- */
278 for (i = 0; i < dim; ++i) {
279 y_tmp[i] = y[i] + h * (ck_a31 * k1[i] + ck_a32 * k2[i]);
280 }
281 st = lmmc_ode_rhs_eval(rhs, t + ck_c[2] * h, y_tmp, k3, dim, user_data,
282 &out_result->num_rhs_evals, &callback_failed);
283 if (st != LMMC_STATUS_OK) {
284 out_result->failure_reason = callback_failed ?
285 LMMC_ODE_FAILURE_RHS_EVAL_FAILED : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
286 failure_status = st;
287 goto rk45_fail;
288 }
289
290 /* --- Stage 4: k4 = f(t + c4*h, y + h*(a41*k1 + a42*k2 + a43*k3)) --- */
291 for (i = 0; i < dim; ++i) {
292 y_tmp[i] = y[i] + h * (ck_a41 * k1[i] + ck_a42 * k2[i] + ck_a43 * k3[i]);
293 }
294 st = lmmc_ode_rhs_eval(rhs, t + ck_c[3] * h, y_tmp, k4, dim, user_data,
295 &out_result->num_rhs_evals, &callback_failed);
296 if (st != LMMC_STATUS_OK) {
297 out_result->failure_reason = callback_failed ?
298 LMMC_ODE_FAILURE_RHS_EVAL_FAILED : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
299 failure_status = st;
300 goto rk45_fail;
301 }
302
303 /* --- Stage 5: k5 = f(t + c5*h, y + h*(a51*k1 + ... + a54*k4)) --- */
304 for (i = 0; i < dim; ++i) {
305 y_tmp[i] = y[i] + h * (ck_a51 * k1[i] + ck_a52 * k2[i] +
306 ck_a53 * k3[i] + ck_a54 * k4[i]);
307 }
308 st = lmmc_ode_rhs_eval(rhs, t + ck_c[4] * h, y_tmp, k5, dim, user_data,
309 &out_result->num_rhs_evals, &callback_failed);
310 if (st != LMMC_STATUS_OK) {
311 out_result->failure_reason = callback_failed ?
312 LMMC_ODE_FAILURE_RHS_EVAL_FAILED : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
313 failure_status = st;
314 goto rk45_fail;
315 }
316
317 /* --- Stage 6: k6 = f(t + c6*h, y + h*(a61*k1 + ... + a65*k5)) --- */
318 for (i = 0; i < dim; ++i) {
319 y_tmp[i] = y[i] + h * (ck_a61 * k1[i] + ck_a62 * k2[i] +
320 ck_a63 * k3[i] + ck_a64 * k4[i] +
321 ck_a65 * k5[i]);
322 }
323 st = lmmc_ode_rhs_eval(rhs, t + ck_c[5] * h, y_tmp, k6, dim, user_data,
324 &out_result->num_rhs_evals, &callback_failed);
325 if (st != LMMC_STATUS_OK) {
326 out_result->failure_reason = callback_failed ?
327 LMMC_ODE_FAILURE_RHS_EVAL_FAILED : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
328 failure_status = st;
329 goto rk45_fail;
330 }
331
332 /* 先构造候选解, 再用新旧状态共同缩放嵌入误差. */
333 err_norm = 0.0;
334 for (i = 0; i < dim; ++i) {
335 lmmc_real_t err_i = h * (ck_e[0] * k1[i] + ck_e[2] * k3[i] +
336 ck_e[3] * k4[i] + ck_e[4] * k5[i] +
337 ck_e[5] * k6[i]);
338 lmmc_real_t scale;
339 lmmc_real_t ratio;
340 y_tmp[i] = y[i] + h * (ck_b5[0] * k1[i] + ck_b5[2] * k3[i] +
341 ck_b5[3] * k4[i] + ck_b5[5] * k6[i]);
342 scale = local_cfg.abs_tol +
343 local_cfg.rel_tol * fmax(fabs(y[i]), fabs(y_tmp[i]));
344 ratio = err_i / scale;
345 err_norm += ratio * ratio;
346 }
347 err_norm = sqrt(err_norm / (lmmc_real_t)dim);
348
349 /* --- Step acceptance / rejection --- */
350 if (err_norm <= 1.0) {
351 step_accepted = 1;
352 memcpy(y, y_tmp, work_bytes);
353
354 /* Check for non-finite state */
355 if (!lmmc_ode_state_is_finite(y, dim)) {
356 out_result->failure_reason = LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
357 goto rk45_fail;
358 }
359
360 /* Advance time */
361 t += h;
362 if (!lmmc_is_finite(&t)) {
363 out_result->failure_reason = LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
364 goto rk45_fail;
365 }
366
367 out_result->num_steps += 1;
368 out_result->final_t = t;
369 lmmc_ode_do_log(&local_cfg, out_result->num_steps, t, y, dim);
370 }
371
372 if (err_norm > 0.0 && isfinite(err_norm)) {
373 lmmc_real_t factor = local_cfg.adaptive_step_beta * pow(err_norm, -0.2);
374 factor = lmmc_clamp(factor, 0.2, 5.0);
375 h_new = lmmc_clamp(h * factor, local_cfg.min_step, local_cfg.max_step);
376 } else {
377 h_new = lmmc_clamp(h * 5.0, local_cfg.min_step, local_cfg.max_step);
378 }
379
380 if (!step_accepted) {
382 if (h_new <= local_cfg.min_step && err_norm > 1.0) {
383 if (h <= local_cfg.min_step) {
384 out_result->failure_reason = LMMC_ODE_FAILURE_INVALID_STEP;
385 failure_status = LMMC_STATUS_CONVERGENCE_FAILED;
386 goto rk45_fail;
387 }
388 }
389 }
390
391 h = h_new;
392 }
393
394 if (t >= t_end) {
395 out_result->converged = 1;
396 out_result->failure_reason = LMMC_ODE_FAILURE_NONE;
397 } else {
398 out_result->converged = 0;
399 out_result->failure_reason = LMMC_ODE_FAILURE_MAX_STEPS;
400 lmmc_free(y_tmp);
401 lmmc_free(k6);
402 lmmc_free(k5);
403 lmmc_free(k4);
404 lmmc_free(k3);
405 lmmc_free(k2);
406 lmmc_free(k1);
407 return LMMC_STATUS_CONVERGENCE_FAILED;
408 }
409
410 lmmc_free(y_tmp);
411 lmmc_free(k6);
412 lmmc_free(k5);
413 lmmc_free(k4);
414 lmmc_free(k3);
415 lmmc_free(k2);
416 lmmc_free(k1);
417 return LMMC_STATUS_OK;
418
419rk45_fail:
420 lmmc_free(y_tmp);
421 lmmc_free(k6);
422 lmmc_free(k5);
423 lmmc_free(k4);
424 lmmc_free(k3);
425 lmmc_free(k2);
426 lmmc_free(k1);
427 return failure_status;
428}

443 lmmc_real_t* k2 = NULL;
444 lmmc_real_t* k3 = NULL;

485 lmmc_real_t rem;

488 lmmc_real_t half_h;

513 LMMC_REAL_MUL(&half_h, &half, &h);
514 for (i = 0; i < dim; ++i) {
515 lmmc_real_t tmp;
516 LMMC_REAL_MUL(&tmp, &half_h, &k1[i]);
517 LMMC_REAL_ADD(&y_tmp[i], &y[i], &tmp);
518 }
519
520
521 LMMC_REAL_ADD(&t_mid, &t, &half_h);
522 st = lmmc_ode_rhs_eval(rhs, t_mid, y_tmp, k2, dim, user_data, &out_result->num_rhs_evals, &callback_failed);
523 if (st != LMMC_STATUS_OK) {
524 out_result->failure_reason = callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
525 goto rk4_fail;
526 }
527
528
529 for (i = 0; i < dim; ++i) {
530 lmmc_real_t tmp;
531 LMMC_REAL_MUL(&tmp, &half_h, &k2[i]);
532 LMMC_REAL_ADD(&y_tmp[i], &y[i], &tmp);
533 }
534
535
536 st = lmmc_ode_rhs_eval(rhs, t_mid, y_tmp, k3, dim, user_data, &out_result->num_rhs_evals, &callback_failed);
537 if (st != LMMC_STATUS_OK) {
538 out_result->failure_reason = callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
539 goto rk4_fail;
540 }
541
542
543 for (i = 0; i < dim; ++i) {
544 lmmc_real_t tmp;
545 LMMC_REAL_MUL(&tmp, &h, &k3[i]);
546 LMMC_REAL_ADD(&y_tmp[i], &y[i], &tmp);
547 }
548
549
550 LMMC_REAL_ADD(&t_next, &t, &h);
551 st = lmmc_ode_rhs_eval(rhs, t_next, y_tmp, k4, dim, user_data, &out_result->num_rhs_evals, &callback_failed);
552 if (st != LMMC_STATUS_OK) {
553 out_result->failure_reason = callback_failed ? LMMC_ODE_FAILURE_RHS_EVAL_FAILED : LMMC_ODE_FAILURE_NUMERICAL_ISSUE;
554 goto rk4_fail;
555 }
556
557
558 LMMC_REAL_DIV(&h_over_6, &h, &six);
559 for (i = 0; i < dim; ++i) {
560 lmmc_real_t t2k2, t2k3, sum1, sum2, sum3, weighted;
561 LMMC_REAL_MUL(&t2k2, &two, &k2[i]);

568 }

573 }

576 if (!lmmc_is_finite(&t)) {

578 goto rk4_fail;
579 }
580

582 out_result->final_t = t;
583 lmmc_ode_do_log(&local_cfg, out_result->num_steps, t, y, dim);
584 }
585
586 if (t >= t_end) {
587 out_result->converged = 1;
588 out_result->failure_reason = LMMC_ODE_FAILURE_NONE;
589 } else {
590 out_result->converged = 0;
591 out_result->failure_reason = LMMC_ODE_FAILURE_MAX_STEPS;
592 }
593
594 lmmc_free(y_tmp);
595 lmmc_free(k4);
596 lmmc_free(k3);
597 lmmc_free(k2);
598 lmmc_free(k1);
599 return out_result->converged ? LMMC_STATUS_OK : LMMC_STATUS_CONVERGENCE_FAILED;
600
601rk4_fail:

606 lmmc_free(k1);
