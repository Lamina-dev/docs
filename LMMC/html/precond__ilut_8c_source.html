3static lmmc_status_t lmmc_ilut_load_row(const lmmc_sparse_mat_t* a,
4 size_t i, lmmc_ilu_row_t* row) {
5 row->count = 0;
6 for (size_t p = a->row_ptr[i]; p < a->row_ptr[i + 1]; ++p) {
7 size_t col = a->col_idx[p];
8 lmmc_real_t value = a->values[p];
9 if (!isfinite(value)) {
10 return LMMC_STATUS_NUMERICAL_FAILURE;
11 }
12 if (!row->present[col]) {
13 row->present[col] = 1;
14 row->processed[col] = 0;
15 row->active[row->count++] = col;
16 row->values[col] = value;
17 } else {
18 row->values[col] = row->values[col] + value;
19 if (!isfinite(row->values[col])) {
20 return LMMC_STATUS_NUMERICAL_FAILURE;
21 }
22 }
23 }
24 return LMMC_STATUS_OK;
25}

41static lmmc_status_t lmmc_ilut_subtract_upper(const lmmc_precond_ilu_impl_t* impl,
42 lmmc_ilu_row_t* row, size_t k, lmmc_real_t multiplier) {
43 for (size_t p = impl->diag_pos[k] + 1; p < impl->row_ptr[k + 1]; ++p) {
44 size_t j = impl->col_idx[p];
45 lmmc_real_t delta = multiplier * impl->lu_values[p];
46 if (j <= k) {
47 continue;
48 }
49 if (!isfinite(delta)) {
50 return LMMC_STATUS_NUMERICAL_FAILURE;
51 }
52 if (!row->present[j]) {
53 row->present[j] = 1;
54 row->processed[j] = 0;
55 row->active[row->count++] = j;
56 row->values[j] = -delta;
57 } else {
58 row->values[j] = row->values[j] - delta;
59 }
60 if (!isfinite(row->values[j])) {
61 return LMMC_STATUS_NUMERICAL_FAILURE;

67static lmmc_status_t lmmc_ilut_eliminate_entry(const lmmc_precond_ilu_impl_t* impl,
68 lmmc_ilu_row_t* row, size_t k, lmmc_real_t drop_tol) {
69 row->processed[k] = 1;
70 lmmc_real_t aik = row->values[k];
71 if (!isfinite(aik)) {
72 return LMMC_STATUS_NUMERICAL_FAILURE;
73 }
74 if (fabs(aik) <= drop_tol) {
75 row->present[k] = 0;
76 row->values[k] = 0.0;
77 return LMMC_STATUS_OK;
78 }
79 if (impl->diag_pos[k] == (size_t)-1 || impl->diag_pos[k] >= impl->nnz) {
80 return LMMC_STATUS_NUMERICAL_FAILURE;
81 }
82 lmmc_real_t diag = impl->lu_values[impl->diag_pos[k]];
83 if (!isfinite(diag) || fabs(diag) <= 1e-15) {
84 return LMMC_STATUS_SINGULAR_MATRIX;
85 }
86 lmmc_real_t multiplier = aik / diag;
87 if (!isfinite(multiplier)) {
88 return LMMC_STATUS_NUMERICAL_FAILURE;
89 }
90 row->values[k] = multiplier;
91 return lmmc_ilut_subtract_upper(impl, row, k, multiplier);
92}

123static lmmc_status_t lmmc_ilut_store_row(lmmc_precond_ilu_impl_t* impl,
124 lmmc_ilu_row_t* row, size_t i, lmmc_real_t drop_tol, size_t fill) {
125 if (!row->present[i]) {
126 row->present[i] = 1;
127 row->processed[i] = 0;
128 row->active[row->count++] = i;
129 row->values[i] = 0.0;
130 }
131 lmmc_real_t diag = row->values[i];
132 if (!isfinite(diag) || fabs(diag) <= 1e-15) {
133 return LMMC_STATUS_SINGULAR_MATRIX;
134 }
135 size_t lower_count, upper_count;
136 lmmc_ilu_select(row, i, 1, drop_tol, fill, row->lower, &lower_count);
137 lmmc_ilu_select(row, i, 0, drop_tol, fill, row->upper, &upper_count);
138 size_t needed = lower_count + 1 + upper_count;
139 if (impl->nnz > (size_t)-1 - needed) {
140 return LMMC_STATUS_INVALID_ARGUMENT;
141 }
142 lmmc_status_t st = lmmc_ilu_impl_reserve(impl, impl->nnz + needed);
143 if (st != LMMC_STATUS_OK) {
144 return st;
145 }
146 st = lmmc_ilut_append_entries(impl, row, row->lower, lower_count);
147 if (st != LMMC_STATUS_OK) {
148 return st;
149 }
150 impl->diag_pos[i] = impl->nnz;
151 impl->col_idx[impl->nnz] = i;
152 impl->lu_values[impl->nnz] = diag;
153 impl->nnz += 1;
154 st = lmmc_ilut_append_entries(impl, row, row->upper, upper_count);
155 if (st != LMMC_STATUS_OK) {
156 return st;
157 }
158 impl->row_ptr[i + 1] = impl->nnz;
159 return LMMC_STATUS_OK;

162static lmmc_status_t lmmc_ilut_factor(const lmmc_sparse_mat_t* a,
163 lmmc_precond_ilu_impl_t* impl, lmmc_ilu_row_t* row,
164 lmmc_real_t drop_tol, size_t fill) {
165 size_t capacity = a->nnz;
166 if (capacity < 16) {
167 capacity = 16;
168 }
169 if (capacity < a->rows && a->rows <= (size_t)-1 - capacity) {
170 capacity += a->rows;
171 }
172 lmmc_status_t st = lmmc_ilu_impl_reserve(impl, capacity);
173 if (st != LMMC_STATUS_OK) {
174 return st;
175 }
176 impl->row_ptr[0] = 0;
177 for (size_t i = 0; i < a->rows; ++i) {
178 st = lmmc_ilut_load_row(a, i, row);
179 if (st != LMMC_STATUS_OK) {
180 return st;
181 }
182 st = lmmc_ilut_eliminate_row(impl, row, i, drop_tol);
183 if (st != LMMC_STATUS_OK) {
184 return st;
185 }
186 st = lmmc_ilut_store_row(impl, row, i, drop_tol, fill);
187 if (st != LMMC_STATUS_OK) {
188 return st;
189 }
190 lmmc_ilu_row_reset(row);
191 }
192 return LMMC_STATUS_OK;
