22static inline lmmc_status_t lmmc_sparse_validate(const lmmc_sparse_mat_t* sparse) {
23 size_t i = 0;
24 size_t outer_size = 0;

26

31 return LMMC_STATUS_INVALID_ARGUMENT;

41 outer_size = sparse->cols;

46 return LMMC_STATUS_INVALID_ARGUMENT;

56 if (sparse->col_idx[i] >= inner_limit) {
57 return LMMC_STATUS_INVALID_ARGUMENT;
58 }

69static inline lmmc_status_t lmmc_sparse_create(size_t rows, size_t cols, size_t nnz, lmmc_sparse_format_t format, lmmc_sparse_mat_t* out_sparse) {
70 size_t outer_size = (format == LMMC_SPARSE_CSR) ? rows : cols;
71 size_t outer_ptr_bytes = 0;
72 size_t idx_bytes = 0;
73 size_t val_bytes = 0;
74 size_t* outer_ptr = NULL;
75 size_t* inner_idx = NULL;
76 lmmc_real_t* values = NULL;
77
78 if (out_sparse == NULL || rows == 0 || cols == 0) {
79 return LMMC_STATUS_INVALID_ARGUMENT;
80 }
81
82 if (!lmmc_safe_mul_size(outer_size + 1, sizeof(size_t), &outer_ptr_bytes) ||
83 !lmmc_safe_mul_size(nnz, sizeof(size_t), &idx_bytes) ||
84 !lmmc_safe_mul_size(nnz, sizeof(lmmc_real_t), &val_bytes)) {
85 return LMMC_STATUS_INVALID_ARGUMENT;
86 }
87
88 outer_ptr = (size_t*)lmmc_alloc(outer_ptr_bytes);
89 if (outer_ptr == NULL) {
90 return LMMC_STATUS_ALLOCATION_FAILED;
91 }
92 memset(outer_ptr, 0, outer_ptr_bytes);
93
94 if (nnz > 0) {
95 inner_idx = (size_t*)lmmc_alloc(idx_bytes);
96 values = (lmmc_real_t*)lmmc_alloc(val_bytes);
97 if (inner_idx == NULL || values == NULL) {
98 if (inner_idx != NULL) lmmc_free(inner_idx);
99 if (values != NULL) lmmc_free(values);
100 lmmc_free(outer_ptr);
101 return LMMC_STATUS_ALLOCATION_FAILED;
102 }
103 memset(inner_idx, 0, idx_bytes);
104 memset(values, 0, val_bytes);
105 for (size_t i = 0; i < nnz; ++i) {
106 LMMC_REAL_INIT(&values[i]);
107 }
108 }
109
110 out_sparse->rows = rows;
111 out_sparse->cols = cols;
112 out_sparse->nnz = nnz;
113 out_sparse->row_ptr = outer_ptr;
114 out_sparse->col_idx = inner_idx;
115 out_sparse->values = values;
116 out_sparse->format = format;
117 out_sparse->owns_data = 1;
118 return LMMC_STATUS_OK;
119}
