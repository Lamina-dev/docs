25static lmmc_optimize_failure_t broyden_step(
26 lmmc_broyden_workspace_t* work, lmmc_vec_t* x)
27{
28 const size_t n = x->size;
29 lmmc_status_t status;
30 size_t i;
31 memcpy(work->lu.data, work->B.data, n * n * sizeof(lmmc_real_t));
32 status = lmmc_lu_decompose_inplace(&work->lu, work->arena.pivots, NULL);
33 if (status == LMMC_STATUS_SINGULAR_MATRIX) {
34 return LMMC_OPT_FAILURE_SINGULAR_JACOBIAN;
35 }
36 if (status != LMMC_STATUS_OK) { return LMMC_OPT_FAILURE_NUMERICAL_ISSUE; }
37 for (i = 0; i < n; ++i) {
38 work->rhs.data[i] = -work->value.data[i];
39 }
40 status = lmmc_lu_solve(&work->lu, work->arena.pivots,
41 &work->rhs, &work->delta_x);
42 if (status != LMMC_STATUS_OK) { return LMMC_OPT_FAILURE_SINGULAR_JACOBIAN; }
43 for (i = 0; i < n; ++i) {
44 x->data[i] += work->delta_x.data[i];
45 }
46 return LMMC_OPT_FAILURE_NONE;
47}

49static void broyden_rank_one_update(lmmc_broyden_workspace_t* work)
50{
51 const size_t n = work->value.size;
52 lmmc_real_t dxTdx = 0.0;
53 size_t i, j;
54 for (i = 0; i < n; ++i) {
55 work->delta_F.data[i] = work->next_value.data[i] - work->value.data[i];
56 }
57 for (i = 0; i < n; ++i) {
58 work->Bdx.data[i] = 0.0;
59 for (j = 0; j < n; ++j) {
60 work->Bdx.data[i] += work->B.data[i * work->B.stride + j] *
61 work->delta_x.data[j];
62 }
63 }
64 for (i = 0; i < n; ++i) {
65 dxTdx += work->delta_x.data[i] * work->delta_x.data[i];
66 }
67 if (dxTdx > 1e-300) {
68 for (i = 0; i < n; ++i) {
69 const lmmc_real_t num_i = work->delta_F.data[i] - work->Bdx.data[i];
70 for (j = 0; j < n; ++j) {
71 work->B.data[i * work->B.stride + j] +=
72 num_i * work->delta_x.data[j] / dxTdx;
73 }
74 }
75 }
76 memcpy(work->value.data, work->next_value.data, n * sizeof(lmmc_real_t));
77}
