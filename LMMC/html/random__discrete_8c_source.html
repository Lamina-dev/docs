40 double a = -0.059 + 0.02483 * b;

121static uint64_t binomial_inversion(uint64_t* state, uint64_t n, double p) {
122 const double q = 1.0 - p;
123 const double qn = exp((double)n * log1p(-p));
124 const double np = (double)n * p;
125 const uint64_t bound = (uint64_t)fmin((double)n, np + 10.0 * sqrt(np * q + 1.0));
126 uint64_t x = 0;
127 double px = qn;
128 double u = u64_to_double01(xoshiro256ss_next(state));
129
130 while (u > px) {
131 ++x;
132 if (x > bound) {
133 x = 0;
134 px = qn;
135 u = u64_to_double01(xoshiro256ss_next(state));
136 } else {
137 u -= px;
138 px = (((double)(n - x + 1) * p) * px) / ((double)x * q);
139 }
140 }
141 return x;
142}

144static int binomial_btpe_accept(
145 uint64_t n, double r, double q, int64_t m, double nrq,
146 double xm, int64_t y, double v) {
147 const uint64_t k = y >= m ? (uint64_t)y - (uint64_t)m : (uint64_t)m - (uint64_t)y;

149 const double s = r / q;
150 const double aa = s * ((double)n + 1.0);
151 double f = 1.0;
152 int64_t i;
153 if (m < y) {
154 for (i = m + 1; i <= y; ++i) {
155 f *= aa / (double)i - s;
156 }
157 } else if (m > y) {
158 for (i = y + 1; i <= m; ++i) {
159 f /= aa / (double)i - s;
160 }
161 }
162 if (v > f) {
163 return 0;
164 }
165 } else {

167 const double rho = (kd / nrq) *
168 ((kd * (kd / 3.0 + 0.625) + 1.0 / 6.0) / nrq + 0.5);
169 const double t = -kd * kd / (2.0 * nrq);
170 const double logv = log(v);
171 if (logv > t - rho) {
172 const double x1 = (double)y + 1.0;
173 const double f1 = (double)m + 1.0;
174 const double z = (double)n + 1.0 - (double)m;
175 const double w = (double)n - (double)y + 1.0;
176 const double x2 = x1 * x1, f2 = f1 * f1, z2 = z * z, w2 = w * w;
177 double accept;
178 if (logv > t + rho) {
179 return 0;
180 }
181 accept = xm * log(f1 / x1) + ((double)n - (double)m + 0.5) * log(z / w)
182 + ((double)y - (double)m) * log(w * r / (x1 * q))
183 + (13860.0 - (462.0 - (132.0 - (99.0 - 140.0 / f2) / f2) / f2) / f2) / f1 / 166320.0

195static uint64_t binomial_btpe(uint64_t* state, uint64_t n, double p) {
196 const double r = fmin(p, 1.0 - p);
197 const double q = 1.0 - r;
198 const double fm = (double)n * r + r;
199 const int64_t m = (int64_t)floor(fm);
200 const double p1 = floor(2.195 * sqrt((double)n * r * q) - 4.6 * q) + 0.5;
201 const double xm = (double)m + 0.5;
202 const double xl = xm - p1;
203 const double xr = xm + p1;
204 const double c = 0.134 + 20.5 / (15.3 + (double)m);
205 double a = (fm - xl) / (fm - xl * r);
206 const double laml = a * (1.0 + a / 2.0);
207 double lamr;
208 const double p2 = p1 * (1.0 + 2.0 * c);
209 const double p3 = p2 + c / laml;
210 double p4;
211 const double nrq = (double)n * r * q;
212
213 a = (xr - fm) / (xr * q);
214 lamr = a * (1.0 + a / 2.0);
215 p4 = p3 + c / lamr;
216
217 for (;;) {
218 double u = u64_to_double01(xoshiro256ss_next(state)) * p4;
219 double v = u64_to_double01(xoshiro256ss_next(state));
220 double x;
221 int64_t y;
222
223 if (u <= p1) {
224 uint64_t result;
225 y = (int64_t)floor(xm - p1 * v + u);
226 result = (uint64_t)y;
227 return p > 0.5 ? n - result : result;
228 } else if (u <= p2) {
229 x = xl + (u - p1) / c;
230 v = v * c + 1.0 - fabs((double)m - x + 0.5) / p1;
231 if (v > 1.0) {
232 continue;
233 }
234 y = (int64_t)floor(x);
235 } else if (u <= p3) {
236 if (v == 0.0) {
237 continue;
238 }
239 y = (int64_t)floor(xl + log(v) / laml);
240 if (y < 0) {
241 continue;
242 }
243 v = v * (u - p2) * laml;
244 } else {
245 if (v == 0.0) {
246 continue;
247 }
248 y = (int64_t)floor(xr - log(v) / lamr);
249 if (y < 0 || (uint64_t)y > n) {
250 continue;
251 }
252 v = v * (u - p3) * lamr;
253 }
254
255 if (!binomial_btpe_accept(n, r, q, m, nrq, xm, y, v)) {
256 continue;
257 }
258 {
259 uint64_t result = (uint64_t)y;
260 return p > 0.5 ? n - result : result;
261 }
262 }
263}
