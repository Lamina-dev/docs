153 }

178 if (c_col_idx) lmmc_free(c_col_idx);
179 if (c_values) {

181 LMMC_REAL_CLEAR(&c_values[k]);
182 }
183 lmmc_free(c_values);
184 }

236 LMMC_REAL_MUL(&tmp, &a->values[i], &a->values[i]);
237 LMMC_REAL_ADD(&tmp_add, &sum, &tmp);
