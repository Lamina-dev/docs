113static int lmmc_lsr_parse_int(const char** cursor, int* out)
114{
115 int sign = 1;
116 int value = 0;
117 int have_digit = 0;
118 if (**cursor == '+') {
119 ++(*cursor);
120 } else if (**cursor == '-') {
121 sign = -1;
122 ++(*cursor);
123 }
124 while (isdigit((unsigned char)**cursor)) {
125 int digit = **cursor - '0';
126 have_digit = 1;
127 if (value > (LMMC_LSR_MAX_UNIT_EXPONENT - digit) / 10) {
128 return 0;
129 }
130 value = value * 10 + digit;
131 ++(*cursor);
132 }
133 if (!have_digit) return 0;
134 *out = sign * value;
135 if (*out < -LMMC_LSR_MAX_UNIT_EXPONENT ||
136 *out > LMMC_LSR_MAX_UNIT_EXPONENT) {
137 return 0;
138 }
139 return 1;
140}

142static int lmmc_lsr_parse_unit_expr(const char* text,
143 lmmc_lsr_unit_sig_t* out)
144{
145 const char* cursor = text;
146 int op_sign = 1;
147 lmmc_lsr_unit_identity(out);
148 if (!text || !*text) return 0;
149 if (strcmp(text, "1") == 0) return 1;
150 while (*cursor) {
151 const char* name_begin = cursor;
152 int exponent = 1;
153 while (isalpha((unsigned char)*cursor) || *cursor == '_') ++cursor;
154 if (cursor == name_begin) return 0;
155 const lmmc_lsr_unit_entry_t* unit =
156 lmmc_lsr_find_unit(name_begin, (size_t)(cursor - name_begin));
157 if (!unit) return 0;
158 if (*cursor == '^') {
159 ++cursor;
160 if (!lmmc_lsr_parse_int(&cursor, &exponent)) return 0;
161 }
162 exponent *= op_sign;
163 out->scale *= pow(unit->scale, (double)exponent);
164 if (!isfinite(out->scale)) return 0;
165 for (size_t i = 0; i < LMMC_LSR_UNIT_DIM_COUNT; ++i) {
166 out->dims[i] += unit->dims[i] * exponent;
167 if (out->dims[i] < -LMMC_LSR_MAX_UNIT_EXPONENT ||
168 out->dims[i] > LMMC_LSR_MAX_UNIT_EXPONENT) {
169 return 0;
170 }
171 }
172 if (*cursor == '\0') break;
173 if (*cursor == '*') {
174 op_sign = 1;
175 } else if (*cursor == '/') {
176 op_sign = -1;
177 } else {
178 return 0;
179 }
180 ++cursor;
181 if (*cursor == '\0') return 0;
182 }
183 return 1;
184}

263lmmc_status_t lmmc_lsr_units_convert(lmmc_real_t x,
264 const char* from_unit,
265 const char* to_unit,
266 lmmc_real_t* out)
267{
268 lmmc_lsr_unit_sig_t from_sig;
269 lmmc_lsr_unit_sig_t to_sig;
270 double converted;
271 if (!out || !from_unit || !to_unit) return LMMC_STATUS_INVALID_ARGUMENT;
272 if (!lmmc_lsr_real_is_finite(x)) return LMMC_STATUS_NUMERICAL_FAILURE;
273 if (!lmmc_lsr_parse_unit_expr(from_unit, &from_sig) ||
274 !lmmc_lsr_parse_unit_expr(to_unit, &to_sig)) {
275 return LMMC_STATUS_INVALID_ARGUMENT;
276 }
277 if (!lmmc_lsr_same_dimension(&from_sig, &to_sig)) {
278 return LMMC_STATUS_DIMENSION_MISMATCH;
279 }
280 converted = (double)x * from_sig.scale / to_sig.scale;
281 return lmmc_lsr_store_finite_real((lmmc_real_t)converted, out);
282}
