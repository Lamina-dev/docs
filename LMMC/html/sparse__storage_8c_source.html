90void lmmc_sparse_destroy(lmmc_sparse_mat_t* sparse) {
91 if (sparse == NULL) {
92 return;
93 }
94 if (sparse->owns_data) {
95 if (sparse->row_ptr != NULL) lmmc_free(sparse->row_ptr);
96 if (sparse->col_idx != NULL) lmmc_free(sparse->col_idx);
97 if (sparse->values != NULL) {
98 for (size_t i = 0; i < sparse->nnz; ++i) {
99 LMMC_REAL_CLEAR(&sparse->values[i]);
100 }
101 lmmc_free(sparse->values);
102 }
103 }
104 sparse->rows = 0;
105 sparse->cols = 0;
106 sparse->nnz = 0;
107 sparse->row_ptr = NULL;
108 sparse->col_idx = NULL;
109 sparse->values = NULL;
