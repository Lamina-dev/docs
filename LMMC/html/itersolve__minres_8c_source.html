30static lmmc_status_t lmmc_minres_beta(const lmmc_vec_t* r,
31 const lmmc_vec_t* y, double* beta) {
32 int nonzero = 0;
33 for (size_t i = 0; i < r->size; ++i) {
34 if (!isfinite(r->data[i]) || !isfinite(y->data[i])) {
35 return LMMC_STATUS_NUMERICAL_FAILURE;
36 }
37 nonzero |= r->data[i] != 0.0;
38 }
39 double dot;
40 lmmc_status_t st = lmmc_vec_dot_checked(r, y, &dot);
41 if (st != LMMC_STATUS_OK) {
42 return st;
43 }
44 if (dot < 0.0) {
45 return LMMC_STATUS_NOT_POSITIVE_DEFINITE;
46 }
47 if (dot == 0.0 && nonzero) {
48 return LMMC_STATUS_NUMERICAL_FAILURE;
49 }
50 *beta = sqrt(dot);
51 return LMMC_STATUS_OK;
52}

73static lmmc_status_t lmmc_minres_lanczos(const lmmc_sparse_mat_t* a,
74 const lmmc_precond_t* precond, const lmmc_itersolve_config_t* cfg,
75 lmmc_minres_state_t* s) {
76 for (size_t i = 0; i < s->v.size; ++i) {
77 s->v.data[i] = s->y.data[i] / s->beta;
78 if (!isfinite(s->v.data[i])) {
79 return LMMC_STATUS_NUMERICAL_FAILURE;
80 }
81 }
82 lmmc_status_t st = lmmc_itersolve_matvec(a, cfg, &s->v, &s->lanczos_tmp);
83 if (st != LMMC_STATUS_OK) {
84 return st;
85 }
86 if (s->old_beta != 0.0) {
87 const double scale = s->beta / s->old_beta;
88 if (!isfinite(scale)) {
89 return LMMC_STATUS_NUMERICAL_FAILURE;
90 }
91 for (size_t i = 0; i < s->v.size; ++i) {
92 s->lanczos_tmp.data[i] -= scale * s->r_prev.data[i];
93 }
94 }
95 for (size_t i = 0; i < s->v.size; ++i) {
96 if (!isfinite(s->lanczos_tmp.data[i])) {
97 return LMMC_STATUS_NUMERICAL_FAILURE;
98 }
99 }
100 st = lmmc_vec_dot_checked(&s->v, &s->lanczos_tmp, &s->alpha);
101 if (st != LMMC_STATUS_OK) {
102 return st;
103 }
104 const double scale = s->alpha / s->beta;
105 if (!isfinite(scale)) {
106 return LMMC_STATUS_NUMERICAL_FAILURE;
107 }
108 for (size_t i = 0; i < s->v.size; ++i) {
109 s->lanczos_tmp.data[i] -= scale * s->r_curr.data[i];
110 if (!isfinite(s->lanczos_tmp.data[i])) {
111 return LMMC_STATUS_NUMERICAL_FAILURE;
112 }
113 }
114 lmmc_vec_t tmp = s->r_prev;
115 s->r_prev = s->r_curr;
116 s->r_curr = s->lanczos_tmp;
117 s->lanczos_tmp = tmp;
118 st = lmmc_apply_precond_or_identity(precond, &s->r_curr, &s->y);
119 if (st != LMMC_STATUS_OK) {
120 return st;
121 }
122 return lmmc_minres_beta(&s->r_curr, &s->y, &s->beta_next);
123}

125static lmmc_status_t lmmc_minres_rotate(lmmc_vec_t* x,
126 lmmc_minres_state_t* s, int* updated) {
127 const double oldeps = s->epsln;
128 const double delta = s->cs * s->dbar + s->sn * s->alpha;
129 const double gbar = s->sn * s->dbar - s->cs * s->alpha;
130 s->epsln = s->sn * s->beta_next;
131 s->dbar = -s->cs * s->beta_next;
132 const double gamma = hypot(gbar, s->beta_next);
133 if (!isfinite(delta) || !isfinite(gbar) || !isfinite(s->epsln) ||
134 !isfinite(s->dbar) || !isfinite(gamma)) {
135 return LMMC_STATUS_NUMERICAL_FAILURE;
136 }
137 *updated = 0;
138 if (gamma == 0.0) {
139 return LMMC_STATUS_OK;
140 }
141 s->cs = gbar / gamma;
142 s->sn = s->beta_next / gamma;
143 const double phi = s->cs * s->phibar;
144 s->phibar = s->sn * s->phibar;
145 if (!isfinite(s->cs) || !isfinite(s->sn) || !isfinite(phi) || !isfinite(s->phibar)) {
146 return LMMC_STATUS_NUMERICAL_FAILURE;
147 }
148 for (size_t i = 0; i < x->size; ++i) {
149 s->w_next.data[i] = (s->v.data[i] - oldeps * s->w_prev2.data[i] -
150 delta * s->w_prev.data[i]) / gamma;
151 if (!isfinite(s->w_next.data[i])) {
152 return LMMC_STATUS_NUMERICAL_FAILURE;
153 }
154 }
155 for (size_t i = 0; i < x->size; ++i) {
156 x->data[i] += phi * s->w_next.data[i];
157 if (!isfinite(x->data[i])) {
158 return LMMC_STATUS_NUMERICAL_FAILURE;
159 }
160 }
161 lmmc_vec_t tmp = s->w_prev2;
162 s->w_prev2 = s->w_prev;
163 s->w_prev = s->w_next;
164 s->w_next = tmp;
165 *updated = 1;
166 return LMMC_STATUS_OK;
167}

169static lmmc_status_t lmmc_minres_iterate(const lmmc_sparse_mat_t* a,
170 const lmmc_vec_t* b, const lmmc_precond_t* precond,
171 const lmmc_itersolve_config_t* cfg, lmmc_vec_t* x,
172 lmmc_minres_state_t* s, size_t* iterations, int* converged) {
173 lmmc_status_t st = lmmc_apply_precond_or_identity(precond, &s->r_curr, &s->y);
174 if (st != LMMC_STATUS_OK) {
175 return st;
176 }
177 st = lmmc_minres_beta(&s->r_curr, &s->y, &s->beta);
178 if (st != LMMC_STATUS_OK) {
179 return st;
180 }
181 s->cs = -1.0;
182 s->phibar = s->beta;
183 for (size_t iter = 0; iter < cfg->max_iter; ++iter) {
184 st = lmmc_minres_lanczos(a, precond, cfg, s);
185 if (st != LMMC_STATUS_OK) {
186 return st;
187 }
188 int updated;
189 st = lmmc_minres_rotate(x, s, &updated);
190 if (st != LMMC_STATUS_OK) {
191 return st;
192 }
193 if (!updated) {
194 break;
195 }
196 *iterations = iter + 1;
197 st = lmmc_itersolve_true_residual(a, cfg, b, x, &s->true_r, &s->norm_r);
198 if (st != LMMC_STATUS_OK) {
199 return st;
200 }
201 lmmc_itersolve_do_log(cfg, *iterations, s->norm_r);
202 if (s->norm_r <= s->threshold) {
203 *converged = 1;
204 break;
205 }
206 if (s->beta_next == 0.0) {
207 break;
208 }
209 s->old_beta = s->beta;
210 s->beta = s->beta_next;
211 }
212 return LMMC_STATUS_OK;
213}
