37static lmmc_status_t lmmc_minres_start(const lmmc_sparse_mat_t* a,
38 const lmmc_vec_t* b, const lmmc_precond_t* precond,
39 const lmmc_itersolve_config_t* cfg, lmmc_vec_t* x,
40 lmmc_minres_state_t* s) {
41 size_t n = b->size;
42 lmmc_status_t st;
44 st = lmmc_itersolve_matvec(a, cfg, x, &s->av);
45 if (st != LMMC_STATUS_OK) {
46 return st;
47 }
48 {
49 size_t i;
50 for (i = 0; i < n; ++i) {
51 s->v_curr.data[i] = b->data[i] - s->av.data[i];
52 }
53 }
54
55 st = lmmc_apply_precond_or_identity(precond, &s->v_curr, &s->z_vec);
56 if (st != LMMC_STATUS_OK) {
57 return st;
58 }
59
61 {
62 double dot_rz;
63 st = lmmc_vec_dot_checked(&s->v_curr, &s->z_vec, &dot_rz);
64 if (st != LMMC_STATUS_OK) {
65 return st;
66 }
67 if (dot_rz < 0.0) {
68 dot_rz = -dot_rz;
69 }
70 s->beta_k = sqrt(dot_rz);
71 }
72
73 st = lmmc_vec_norm2_checked(b, &s->norm_b_val);
74 if (st != LMMC_STATUS_OK) {
75 return st;
76 }
77
78 s->norm_r = s->beta_k;
79 s->phi_bar = s->beta_k;
80 s->threshold = cfg->abs_tol + cfg->rel_tol * s->norm_b_val;
81 return LMMC_STATUS_OK;
82}

104 const lmmc_precond_t* precond, const lmmc_itersolve_config_t* cfg,
105 lmmc_minres_state_t* s) {
106 size_t n = s->v_curr.size;
107 size_t i;
108 lmmc_status_t st;
110 st = lmmc_itersolve_matvec(a, cfg, &s->v_curr, &s->av);
111 if (st != LMMC_STATUS_OK) {
112 return st;
113 }
114
115 st = lmmc_apply_precond_or_identity(precond, &s->av, &s->z_vec);
116 if (st != LMMC_STATUS_OK) {
117 return st;
118 }

122 if (st != LMMC_STATUS_OK) {
123 return st;
124 }
125
127 for (i = 0; i < n; ++i) {
128 s->v_next.data[i] = s->z_vec.data[i] - s->alpha_k * s->v_curr.data[i] - s->beta_k * s->v_prev.data[i];
129 }

135 if (st != LMMC_STATUS_OK) {
136 return st;

138 if (dot_vv < 0.0) {
139 dot_vv = -dot_vv;

142 }

146static lmmc_status_t lmmc_minres_rotate(lmmc_vec_t* x, lmmc_minres_state_t* s) {
147 size_t n = x->size;
148 size_t i;
149 double cs_new, sn_new;
150 lmmc_status_t st;
152 s->epsilon_curr = s->sn_prev * s->beta_k;
153 s->delta_bar = s->cs_curr * s->alpha_k - s->sn_curr * s->epsilon_curr;
154
156 s->gamma_val = sqrt(s->delta_bar * s->delta_bar + s->beta_kp1 * s->beta_kp1);
157
158 if (s->gamma_val <= 1e-30) {
159 st = LMMC_STATUS_NUMERICAL_FAILURE;
160 return st;
161 }
162
163 cs_new = s->delta_bar / s->gamma_val;
164 sn_new = s->beta_kp1 / s->gamma_val;
165
167 s->phi_val = cs_new * s->phi_bar;
169 s->phi_bar = -sn_new * s->phi_bar;
170
172 for (i = 0; i < n; ++i) {
173 s->z_vec.data[i] = (s->v_curr.data[i] - s->epsilon_curr * s->w_prev.data[i] - s->delta_bar * s->w_curr.data[i]) / s->gamma_val;
174 }
175
177 for (i = 0; i < n; ++i) {
178 x->data[i] += s->phi_val * s->z_vec.data[i];
179 if (!isfinite(x->data[i])) {
180 st = LMMC_STATUS_NUMERICAL_FAILURE;
181 return st;
182 }
183 }
184
185 for (i = 0; i < n; ++i) {
186 s->w_prev.data[i] = s->w_curr.data[i];
187 s->w_curr.data[i] = s->z_vec.data[i];
188 }
189
190 s->sn_prev = s->sn_curr;
191 s->sn_curr = sn_new;
192 s->cs_curr = cs_new;
193 return LMMC_STATUS_OK;
194}

196static lmmc_status_t lmmc_minres_iterate(const lmmc_sparse_mat_t* a,
197 const lmmc_precond_t* precond, const lmmc_itersolve_config_t* cfg,
198 lmmc_vec_t* x, lmmc_minres_state_t* s, lmmc_itersolve_result_t* result,
199 size_t* iterations, int* converged) {
200 lmmc_minres_normalize(s);
201 for (size_t iter = 0; iter < cfg->max_iter; ++iter) {
202 lmmc_status_t st = lmmc_minres_lanczos(a, precond, cfg, s);
203 if (st != LMMC_STATUS_OK) {
204 return st;
205 }
206 st = lmmc_minres_rotate(x, s);
207 if (st != LMMC_STATUS_OK) {
208 return st;
209 }
210 s->norm_r = fabs(s->phi_bar);
211 *iterations = iter + 1;
212 if (result != NULL) {
213 result->final_residual_norm = s->norm_r;
214 }
215 lmmc_itersolve_do_log(cfg, *iterations, s->norm_r);
216 if (s->norm_r <= s->threshold) {
217 *converged = 1;
218 break;
219 }
220 if (s->beta_kp1 <= 1e-30) {
221 break;
222 }
223 for (size_t i = 0; i < x->size; ++i) {
224 s->v_prev.data[i] = s->v_curr.data[i];
225 s->v_curr.data[i] = s->v_next.data[i] / s->beta_kp1;
226 }
227 s->beta_k = s->beta_kp1;
228 }
229 return LMMC_STATUS_OK;
230}
