30static int lmmc_quad_sum_add(lmmc_quad_sum_t* acc, lmmc_real_t value)
31{
32 const lmmc_real_t sum = acc->sum + value;
33 lmmc_real_t residual;
34 if (!isfinite(sum)) {
35 return 0;
36 }
37 if (fabs(acc->sum) >= fabs(value)) {
38 residual = (acc->sum - sum) + value;
39 } else {
40 residual = (value - sum) + acc->sum;
41 }
42 acc->correction += residual;
43 acc->sum = sum;
44 return isfinite(acc->correction);
45}

58static lmmc_real_t lmmc_quad_simpson_panel(
59 const lmmc_quad_leaf_t* leaf, size_t first, size_t step)
60{
61 lmmc_real_t center, half_length;
62 lmmc_quad_center_half_length(
63 leaf->x[first], leaf->x[first + 2 * step], &center, &half_length);
64 return (half_length / 3.0) *
65 (leaf->f[first] + 4.0 * leaf->f[first + step] +
66 leaf->f[first + 2 * step]);

69static int lmmc_quad_leaf_estimate(lmmc_quad_leaf_t* leaf)
70{
71 const lmmc_real_t whole = lmmc_quad_simpson_panel(leaf, 0, 2);
72 const lmmc_real_t left = lmmc_quad_simpson_panel(leaf, 0, 1);
73 const lmmc_real_t right = lmmc_quad_simpson_panel(leaf, 2, 1);
74 const lmmc_real_t combined = left + right;
75 const lmmc_real_t delta = combined - whole;
76 leaf->value = combined + delta / 15.0;
77 leaf->error = fabs(delta) / 15.0;
78 return isfinite(whole) && isfinite(left) && isfinite(right) &&
79 isfinite(combined) && isfinite(leaf->value) && isfinite(leaf->error);
80}

165static int lmmc_quad_heap_reserve(lmmc_quad_workspace_t* work, size_t extra)
166{
167 size_t needed, capacity, bytes;
168 lmmc_quad_leaf_t* heap;
169 if (!lmmc_safe_add_size(work->size, extra, &needed)) {
170 return 0;
171 }
172 if (needed <= work->capacity) {
173 return 1;
174 }
175 capacity = work->capacity == 0 ? 16 : work->capacity;
176 while (capacity < needed) {
177 if (capacity > SIZE_MAX / 2) {
178 capacity = needed;
179 break;
180 }
181 capacity *= 2;
182 }
183 if (!lmmc_safe_mul_size(capacity, sizeof(*heap), &bytes)) {
184 return 0;
185 }
186 heap = (lmmc_quad_leaf_t*)lmmc_memory_realloc(work->heap, bytes);
187 if (heap == NULL) {
188 return 0;
189 }
190 work->heap = heap;
191 work->capacity = capacity;
192 return 1;
193}

195static void lmmc_quad_heap_push(
196 lmmc_quad_workspace_t* work, const lmmc_quad_leaf_t* leaf)
197{
198 size_t pos = work->size++;
199 while (pos != 0) {
200 const size_t parent = (pos - 1) / 2;
201 if (!lmmc_quad_leaf_precedes(leaf, &work->heap[parent])) break;
202 work->heap[pos] = work->heap[parent];
203 pos = parent;
204 }
205 work->heap[pos] = *leaf;
206}

208static lmmc_quad_leaf_t lmmc_quad_heap_pop(lmmc_quad_workspace_t* work)
209{
210 const lmmc_quad_leaf_t result = work->heap[0];
211 const lmmc_quad_leaf_t last = work->heap[--work->size];
212 size_t pos = 0;
214 while (pos < work->size / 2) {
215 size_t child = 2 * pos + 1;
216 if (child + 1 < work->size &&
217 lmmc_quad_leaf_precedes(&work->heap[child + 1], &work->heap[child])) {
218 ++child;
219 }
220 if (!lmmc_quad_leaf_precedes(&work->heap[child], &last)) break;
221 work->heap[pos] = work->heap[child];
222 pos = child;
223 }
224 if (work->size != 0) work->heap[pos] = last;
225 return result;
226}

236static lmmc_status_t lmmc_quad_refine(lmmc_quad_workspace_t* work)
237{
238 lmmc_quad_leaf_t parent = lmmc_quad_heap_pop(work);
239 lmmc_quad_leaf_t left, right;
240 int split_left, split_right;
241 if (work->num_evals > SIZE_MAX - 4) {
242 return LMMC_STATUS_NUMERICAL_FAILURE;
243 }
244 if (!lmmc_quad_leaf_child(work, &parent, 0, &left) ||
245 !lmmc_quad_leaf_child(work, &parent, 2, &right)) {
246 return LMMC_STATUS_NUMERICAL_FAILURE;
247 }
248 split_left = lmmc_quad_leaf_can_split(&left, work->max_depth);
249 split_right = lmmc_quad_leaf_can_split(&right, work->max_depth);
250 if (!lmmc_quad_heap_reserve(work, (size_t)split_left + (size_t)split_right)) {
251 return LMMC_STATUS_ALLOCATION_FAILED;
252 }
253 if (!lmmc_quad_totals_add(work, &parent, -1.0) ||
254 !lmmc_quad_totals_add(work, &left, 1.0) ||
255 !lmmc_quad_totals_add(work, &right, 1.0)) {
256 return LMMC_STATUS_NUMERICAL_FAILURE;
257 }
259 if (split_left) {
260 lmmc_quad_heap_push(work, &left);
261 }
262 if (split_right) {
263 lmmc_quad_heap_push(work, &right);
264 }
265 return LMMC_STATUS_OK;
266}

268static lmmc_status_t lmmc_quad_global_integral(
269 lmmc_quad_workspace_t* work, const lmmc_quad_leaf_t* root,
270 lmmc_real_t abs_tol, lmmc_real_t rel_tol,
271 lmmc_quad_result_t* result)
272{
273 for (;;) {
274 lmmc_real_t tolerance;
275 lmmc_status_t status;
276 result->value = work->value.sum + work->value.correction;
277 result->error = work->error.sum + work->error.correction;
278 result->num_evals = work->num_evals;
279 tolerance = lmmc_max(abs_tol, rel_tol * fabs(result->value));
280 if (!isfinite(result->value) || !isfinite(result->error) ||
281 result->error < 0.0 || !isfinite(tolerance)) {
282 return LMMC_STATUS_NUMERICAL_FAILURE;
283 }
284 if (result->error <= tolerance) {
285 return LMMC_STATUS_OK;
286 }
287 if (root != NULL) {
288 if (lmmc_quad_leaf_can_split(root, work->max_depth)) {
289 if (!lmmc_quad_heap_reserve(work, 1)) {
290 return LMMC_STATUS_ALLOCATION_FAILED;
291 }
292 lmmc_quad_heap_push(work, root);
293 }
294 root = NULL;
295 }
296 if (work->size == 0) {
297 return LMMC_STATUS_WARNING_MAX_DEPTH;
298 }
299 status = lmmc_quad_refine(work);
300 if (status != LMMC_STATUS_OK) {
301 return status;
302 }
303 }
304}

318lmmc_status_t lmmc_quad_adaptive(
319 lmmc_quad_func_t func, void* user_data,
320 lmmc_real_t a, lmmc_real_t b,
321 lmmc_real_t abs_tol, lmmc_real_t rel_tol,
322 size_t max_depth, lmmc_quad_result_t* out_result)
323{
324 lmmc_quad_workspace_t work = {0};
325 lmmc_quad_leaf_t root;
326 lmmc_quad_result_t result;
327 lmmc_status_t status;
328 if (func == NULL || out_result == NULL ||
329 !lmmc_quad_adaptive_arguments(a, b, abs_tol, rel_tol)) {
330 return LMMC_STATUS_INVALID_ARGUMENT;
331 }
332 work.func = func;
333 work.user_data = user_data;
334 work.max_depth = max_depth;
335 if (!lmmc_quad_leaf_initial(&work, &root, a, b) ||
336 !lmmc_quad_totals_add(&work, &root, 1.0)) {
337 return LMMC_STATUS_NUMERICAL_FAILURE;
338 }
339 status = lmmc_quad_global_integral(&work, &root, abs_tol, rel_tol, &result);
340 lmmc_memory_free(work.heap);
341 if (status == LMMC_STATUS_OK || status == LMMC_STATUS_WARNING_MAX_DEPTH) {
342 *out_result = result;
343 }
344 return status;
