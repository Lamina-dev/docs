50static lmmc_status_t svd_wide(const lmmc_mat_t *a, lmmc_real_t scale,
51 lmmc_svd_result_t *result)
52{
53 lmmc_mat_t transpose;
54 lmmc_svd_result_t temporary = {0};

56 if (status != LMMC_STATUS_OK) {
57 return status;
58 }
59 status = lmmc_mat_transpose_to(a, &transpose);
60 if (status == LMMC_STATUS_OK) {

76{
77 lmmc_real_t scale;
78 if (!lmmc_mat_descriptor_is_valid(a) || !out_result) {
79 return LMMC_STATUS_INVALID_ARGUMENT;
80 }
81 lmmc_status_t status = lmmc_eigen_matrix_scale(a, &scale);
82 if (status != LMMC_STATUS_OK) {
83 return status;
84 }
85 lmmc_svd_result_t result = {0};
86 if (a->rows >= a->cols) {
87 status = lmmc_svd_tall(a, scale, &result);
88 if (status == LMMC_STATUS_OK) {
89 status = restore_singular_value_scale(&result.sigma, scale);

91 } else {
92 status = svd_wide(a, scale, &result);
93 }
94 if (status == LMMC_STATUS_OK) {
95 *out_result = result;
96 } else {
97 lmmc_svd_result_destroy(&result);
98 }
99 return status;

115static lmmc_status_t prepare_reciprocals(lmmc_svd_result_t *svd,
116 lmmc_real_t tol)
117{
118 if (!matrix_is_finite(&svd->U) || !matrix_is_finite(&svd->Vt)) {
119 return LMMC_STATUS_NUMERICAL_FAILURE;
120 }
121 const size_t m = svd->U.rows;
122 const size_t n = svd->Vt.rows;
123 if (tol <= 0.0) {
124 lmmc_real_t dimension = (lmmc_real_t)((m > n) ? m : n);
125 tol = LMMC_REAL_EPSILON * dimension * svd->sigma.data[0];
126 }
127 if (!isfinite(tol)) {
128 return LMMC_STATUS_NUMERICAL_FAILURE;
129 }
130 for (size_t k = 0; k < svd->sigma.size; ++k) {
131 lmmc_real_t singular = svd->sigma.data[k];
132 if (!isfinite(singular) || singular < 0.0) {
133 return LMMC_STATUS_NUMERICAL_FAILURE;
134 }
135 lmmc_real_t reciprocal = (singular > tol) ? 1.0 / singular : 0.0;
136 if (!isfinite(reciprocal)) {
137 return LMMC_STATUS_NUMERICAL_FAILURE;
138 }
139 svd->sigma.data[k] = reciprocal;
140 }
141 return LMMC_STATUS_OK;
142}

144static lmmc_status_t reconstruct_entry(const lmmc_svd_result_t *svd,
145 size_t i, size_t j, lmmc_real_t *value)
146{
147 lmmc_real_t sum = 0.0;
148 for (size_t k = 0; k < svd->sigma.size; ++k) {
149 const lmmc_real_t reciprocal = svd->sigma.data[k];
150 if (reciprocal == 0.0) {
151 continue;
152 }
153 lmmc_real_t product = MAT_ELEM(&svd->Vt, k, i) * MAT_ELEM(&svd->U, j, k);
154 if (!isfinite(product)) {
155 return LMMC_STATUS_NUMERICAL_FAILURE;
156 }
157 product *= reciprocal;
158 if (!isfinite(product)) {
159 return LMMC_STATUS_NUMERICAL_FAILURE;
160 }
161 sum += product;
162 if (!isfinite(sum)) {
163 return LMMC_STATUS_NUMERICAL_FAILURE;
164 }
165 }
166 *value = sum;
167 return LMMC_STATUS_OK;
168}

179 }
180 }
181 }
182 return LMMC_STATUS_OK;

185static lmmc_status_t validate_pinv(const lmmc_mat_t *a, lmmc_real_t tol,
186 const lmmc_mat_t *out_pinv)

192 return LMMC_STATUS_INVALID_ARGUMENT;

199 }

203lmmc_status_t lmmc_pinv(const lmmc_mat_t *a, lmmc_real_t tol, lmmc_mat_t *out_pinv)
204{
205 lmmc_status_t status = validate_pinv(a, tol, out_pinv);
206 if (status != LMMC_STATUS_OK) {
207 return status;
208 }
209 lmmc_svd_result_t svd;
210 status = lmmc_svd(a, &svd);
211 if (status != LMMC_STATUS_OK) {
212 return status;
213 }
214 lmmc_mat_t candidate = {0};
215 status = prepare_reciprocals(&svd, tol);
216 if (status != LMMC_STATUS_OK) {
217 goto cleanup;
218 }
219 status = lmmc_mat_create(a->cols, a->rows, &candidate);
220 if (status != LMMC_STATUS_OK) {
221 goto cleanup;
222 }
223 status = reconstruct_pinv(&svd, &candidate);
224 if (status == LMMC_STATUS_OK) {
225 for (size_t i = 0; i < candidate.rows; ++i) {
226 memcpy(out_pinv->data + i * out_pinv->stride,
227 candidate.data + i * candidate.stride,
228 candidate.cols * sizeof(*candidate.data));
229 }
230 }
231cleanup:
232 lmmc_mat_destroy(&candidate);
233 lmmc_svd_result_destroy(&svd);
234 return status;
235}

237lmmc_status_t lmmc_cond(const lmmc_mat_t *a, lmmc_real_t *out_cond)
238{
239 if (!a || !out_cond || !a->data) {
240 return LMMC_STATUS_INVALID_ARGUMENT;
241 }
242 if (a->rows == 0 || a->cols == 0) {
243 return LMMC_STATUS_INVALID_ARGUMENT;
244 }
245 lmmc_svd_result_t svd;
246 lmmc_status_t status = lmmc_svd(a, &svd);
247 if (status != LMMC_STATUS_OK) {
248 return status;
249 }
250 size_t p = svd.sigma.size;
251 if (p == 0) {
252 lmmc_svd_result_destroy(&svd);
253 return LMMC_STATUS_INVALID_ARGUMENT;
254 }
255 lmmc_real_t s_max = svd.sigma.data[0];
256 lmmc_real_t s_min = svd.sigma.data[p - 1];
257 *out_cond = (s_min <= 0.0) ? INFINITY : s_max / s_min;
258 lmmc_svd_result_destroy(&svd);
259 return LMMC_STATUS_OK;
260}
