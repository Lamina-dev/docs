36static int ode_steps_valid(const lmmc_ode_config_t* cfg) {
37 if (!lmmc_is_finite(&cfg->initial_step) || !lmmc_is_finite(&cfg->min_step)) {
38 return 0;

145lmmc_status_t lmmc_ode_weighted_rms(
146 const lmmc_real_t* error,
147 const lmmc_real_t* y_old,
148 const lmmc_real_t* y_new,
149 size_t dim,
150 lmmc_real_t abs_tol,
151 lmmc_real_t rel_tol,
152 lmmc_real_t* out_norm
153) {
154 lmmc_real_t sum = 0.0;
155 size_t i;
156 if (error == NULL || y_old == NULL || y_new == NULL || out_norm == NULL || dim == 0) {
157 return LMMC_STATUS_INVALID_ARGUMENT;
158 }
159 for (i = 0; i < dim; ++i) {
160 lmmc_real_t scale = abs_tol + rel_tol * fmax(fabs(y_old[i]), fabs(y_new[i]));
161 lmmc_real_t ratio;
162 if (!isfinite(scale) || scale <= 0.0 || !isfinite(error[i])) {
163 return LMMC_STATUS_NUMERICAL_FAILURE;
164 }
165 ratio = error[i] / scale;
166 sum += ratio * ratio;
167 if (!isfinite(sum)) {
168 return LMMC_STATUS_NUMERICAL_FAILURE;
169 }
170 }
171 *out_norm = sqrt(sum / (lmmc_real_t)dim);
172 return isfinite(*out_norm) ? LMMC_STATUS_OK : LMMC_STATUS_NUMERICAL_FAILURE;
173}

175lmmc_real_t lmmc_ode_next_step(
176 lmmc_real_t h,
177 lmmc_real_t error_norm,
178 const lmmc_ode_config_t* cfg
179) {
180 lmmc_real_t factor = 5.0;
181 if (error_norm > 0.0 && isfinite(error_norm)) {
182 factor = cfg->adaptive_step_beta * pow(error_norm, -0.25);
183 }
184 factor = lmmc_clamp(factor, 0.2, 5.0);
185 return lmmc_clamp(h * factor, cfg->min_step, cfg->max_step);
186}
