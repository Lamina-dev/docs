40static inline lmmc_status_t lmmc_ode_load_and_validate_config(
41 size_t dim,
42 lmmc_real_t t_start,
43 lmmc_real_t t_end,
44 const lmmc_ode_config_t* cfg,
45 lmmc_ode_config_t* out_cfg,
46 lmmc_ode_result_t* out_result
47) {
48 if (dim == 0) {
49 if (out_result != NULL) {
50 out_result->failure_reason = LMMC_ODE_FAILURE_INVALID_DIMENSION;
51 }
52 return LMMC_STATUS_INVALID_ARGUMENT;
53 }
54
55 if (!lmmc_is_finite(&t_start) || !lmmc_is_finite(&t_end) || !(t_end > t_start)) {
56 if (out_result != NULL) {
57 out_result->failure_reason = LMMC_ODE_FAILURE_INVALID_STEP;
58 }
59 return LMMC_STATUS_INVALID_ARGUMENT;
60 }
61
62 if (out_cfg == NULL) {
63 if (out_result != NULL) {
64 out_result->failure_reason = LMMC_ODE_FAILURE_INVALID_STEP;
65 }
66 return LMMC_STATUS_INVALID_ARGUMENT;
67 }
68
69 if (cfg != NULL) {
70 *out_cfg = *cfg;
71 } else {
72 lmmc_status_t st = lmmc_ode_default_config(t_start, t_end, dim, out_cfg);
73 if (st != LMMC_STATUS_OK) {
74 if (out_result != NULL) {
75 out_result->failure_reason = LMMC_ODE_FAILURE_INVALID_STEP;
76 }
77 return st;
78 }
79 }
80
81 if (!lmmc_is_finite(&out_cfg->abs_tol) || !lmmc_is_finite(&out_cfg->rel_tol) ||
82 out_cfg->abs_tol < 0.0 || out_cfg->rel_tol < 0.0 ||
83 (out_cfg->abs_tol == 0.0 && out_cfg->rel_tol == 0.0)) {
84 if (out_result != NULL) {
85 out_result->failure_reason = LMMC_ODE_FAILURE_TOLERANCE_INCONSISTENT;
86 }
87 return LMMC_STATUS_INVALID_ARGUMENT;
88 }
89
90 if (!lmmc_is_finite(&out_cfg->initial_step) || !lmmc_is_finite(&out_cfg->min_step) ||
91 !lmmc_is_finite(&out_cfg->max_step) || out_cfg->initial_step <= 0.0 ||
92 out_cfg->min_step <= 0.0 || out_cfg->max_step <= 0.0 || out_cfg->min_step > out_cfg->max_step ||
93 out_cfg->max_steps == 0 || !lmmc_is_finite(&out_cfg->adaptive_step_beta) ||
94 out_cfg->adaptive_step_beta <= 0.0 || out_cfg->adaptive_step_beta > 1.0) {
95 if (out_result != NULL) {
96 out_result->failure_reason = LMMC_ODE_FAILURE_INVALID_STEP;
97 }
98 return LMMC_STATUS_INVALID_ARGUMENT;
99 }
100
101 return LMMC_STATUS_OK;
102}

185static inline lmmc_status_t lmmc_ode_weighted_rms(
186 const lmmc_real_t* error,
187 const lmmc_real_t* y_old,
188 const lmmc_real_t* y_new,
189 size_t dim,
190 lmmc_real_t abs_tol,
191 lmmc_real_t rel_tol,
192 lmmc_real_t* out_norm
193) {
194 lmmc_real_t sum = 0.0;
195 size_t i;
196 if (error == NULL || y_old == NULL || y_new == NULL || out_norm == NULL || dim == 0) {
197 return LMMC_STATUS_INVALID_ARGUMENT;
198 }
199 for (i = 0; i < dim; ++i) {
200 lmmc_real_t scale = abs_tol + rel_tol * fmax(fabs(y_old[i]), fabs(y_new[i]));
201 lmmc_real_t ratio;
202 if (!isfinite(scale) || scale <= 0.0 || !isfinite(error[i])) {
203 return LMMC_STATUS_NUMERICAL_FAILURE;
204 }
205 ratio = error[i] / scale;
206 sum += ratio * ratio;
207 if (!isfinite(sum)) {
208 return LMMC_STATUS_NUMERICAL_FAILURE;
209 }
210 }
211 *out_norm = sqrt(sum / (lmmc_real_t)dim);
212 return isfinite(*out_norm) ? LMMC_STATUS_OK : LMMC_STATUS_NUMERICAL_FAILURE;
213}

215static inline lmmc_real_t lmmc_ode_next_step(
216 lmmc_real_t h,
217 lmmc_real_t error_norm,
218 const lmmc_ode_config_t* cfg
219) {
220 lmmc_real_t factor = 5.0;
221 if (error_norm > 0.0 && isfinite(error_norm)) {
222 factor = cfg->adaptive_step_beta * pow(error_norm, -0.25);
223 }
224 factor = lmmc_clamp(factor, 0.2, 5.0);
225 return lmmc_clamp(h * factor, cfg->min_step, cfg->max_step);
226}

249static inline lmmc_status_t lmmc_ode_jacobian_eval(
250 lmmc_ode_rhs_t rhs,
251 lmmc_ode_jac_t jacobian,
252 void* user_data,
253 lmmc_real_t t,
254 const lmmc_real_t* y,
255 const lmmc_real_t* base_rhs,
256 size_t dim,
257 lmmc_real_t* jacobian_data,
258 lmmc_real_t* base_rhs_work,
259 lmmc_real_t* y_perturbed,
260 lmmc_real_t* rhs_perturbed,
261 size_t* io_rhs_evals,
262 int* out_callback_failed
263) {
264 const lmmc_real_t* base = base_rhs;
265 size_t i, j;
266 size_t jacobian_count;
267 lmmc_status_t st;
268
269 if (rhs == NULL || y == NULL || jacobian_data == NULL ||
270 base_rhs_work == NULL || y_perturbed == NULL || rhs_perturbed == NULL ||
271 io_rhs_evals == NULL || out_callback_failed == NULL || dim == 0 ||
272 !lmmc_safe_mul_size(dim, dim, &jacobian_count)) {
273 return LMMC_STATUS_INVALID_ARGUMENT;
274 }
275 *out_callback_failed = 0;
276 if (jacobian != NULL) {
277 st = jacobian(t, y, jacobian_data, dim, user_data);
278 if (st != LMMC_STATUS_OK) return st;
279 return lmmc_ode_values_are_finite(jacobian_data, jacobian_count);
280 }
281
282 if (base == NULL) {
283 st = lmmc_ode_rhs_eval(rhs, t, y, base_rhs_work, dim, user_data,
284 io_rhs_evals, out_callback_failed);
285 if (st != LMMC_STATUS_OK) return st;
286 base = base_rhs_work;
287 }
288
289 for (j = 0; j < dim; ++j) {
290 const lmmc_real_t delta =
291 sqrt(LMMC_REAL_EPSILON) * fmax(1.0, fabs(y[j]));
292 memcpy(y_perturbed, y, dim * sizeof(*y));
293 y_perturbed[j] += delta;
294 st = lmmc_ode_rhs_eval(rhs, t, y_perturbed, rhs_perturbed, dim,
295 user_data, io_rhs_evals, out_callback_failed);
296 if (st != LMMC_STATUS_OK) return st;
297 for (i = 0; i < dim; ++i) {
298 jacobian_data[i * dim + j] =
299 (rhs_perturbed[i] - base[i]) / delta;
300 }
301 }
302 return lmmc_ode_values_are_finite(jacobian_data, jacobian_count);
303}
