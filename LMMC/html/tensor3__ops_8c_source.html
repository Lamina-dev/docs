343static lmmc_status_t lmmc_tensor3_sum_line(
344 const lmmc_real_t* data, size_t count, size_t stride,
345 lmmc_real_t* sum, lmmc_real_t* tmp_sum) {
346 LMMC_REAL_SET_D(sum, 0.0);
347 for (size_t i = 0; i < count; ++i) {
348 const lmmc_real_t* v = &data[i * stride];
349 if (!lmmc_is_finite(v)) {
350 return LMMC_STATUS_NUMERICAL_FAILURE;
351 }
352 LMMC_REAL_ADD(tmp_sum, sum, v);
353 LMMC_REAL_SET(sum, tmp_sum);
354 if (!lmmc_is_finite(sum)) {
355 return LMMC_STATUS_NUMERICAL_FAILURE;
356 }
357 }
358 return LMMC_STATUS_OK;
359}

361lmmc_status_t lmmc_tensor3_sum_axis(
362 const lmmc_tensor3_t* tensor, size_t axis, lmmc_mat_t* out_matrix) {
363 lmmc_real_t sum, tmp_sum;
364 LMMC_REAL_INIT(&sum);
365 LMMC_REAL_INIT(&tmp_sum);
366 lmmc_status_t st = lmmc_tensor3_validate_axis_sum(tensor, axis, out_matrix);
367 if (st == LMMC_STATUS_OK) {
368 const size_t dims[3] = {tensor->dim0, tensor->dim1, tensor->dim2};
369 const size_t strides[3] = {tensor->stride0, tensor->stride1, tensor->stride2};
370 const size_t row_stride = axis == 0 ? tensor->stride1 : tensor->stride0;
371 const size_t col_stride = axis == 2 ? tensor->stride1 : tensor->stride2;
372 for (size_t row = 0; row < out_matrix->rows; ++row) {
373 for (size_t col = 0; col < out_matrix->cols; ++col) {
374 const size_t offset = row * row_stride + col * col_stride;
375 st = lmmc_tensor3_sum_line(
376 &tensor->data[offset], dims[axis], strides[axis], &sum, &tmp_sum);
377 if (st != LMMC_STATUS_OK) {
378 goto done;
379 }
380 LMMC_REAL_SET(&out_matrix->data[row * out_matrix->stride + col], &sum);
381 }
382 }
383 }
384done:
385 LMMC_REAL_CLEAR(&sum);
386 LMMC_REAL_CLEAR(&tmp_sum);
387 return st;
