50static int lmmc_quad_romberg_sample_row(
51 lmmc_quad_romberg_state_t* state, size_t row)
52{
53 const size_t n = (size_t)1 << row;
54 const lmmc_real_t normalized_h = 2.0 / (lmmc_real_t)n;
55 lmmc_real_t sum_new = 0.0;
56 for (size_t k = 1; k <= n / 2; ++k) {
57 const lmmc_real_t normalized_x =
58 -1.0 + (2.0 * (lmmc_real_t)k - 1.0) * normalized_h;
59 const lmmc_real_t x = fma(state->half_length, normalized_x, state->center);
60 const lmmc_real_t value = state->func(x, state->user_data);
61 ++state->out->num_evals;
62 if (!isfinite(value)) {
63 return 0;
64 }
65 sum_new += value;
66 if (!isfinite(sum_new)) {
67 return 0;
68 }
69 }
70 state->curr[0] = state->prev[0] / 2.0 +
71 state->half_length * normalized_h * sum_new;
72 return isfinite(state->curr[0]);
73}

75static int lmmc_quad_romberg_extrapolate(
76 lmmc_quad_romberg_state_t* state, size_t row)
77{
78 lmmc_real_t pow4 = 1.0;
79 for (size_t j = 1; j <= row; ++j) {
80 pow4 *= 4.0;
81 state->curr[j] = state->curr[j - 1] +
82 (state->curr[j - 1] - state->prev[j - 1]) / (pow4 - 1.0);
83 if (!isfinite(state->curr[j])) {
84 return 0;
85 }
86 }
87 state->out->value = state->curr[row];
88 state->out->error = lmmc_abs(state->curr[row] - state->prev[row - 1]);
89 return isfinite(state->out->error);
90}

95{

100 out->value = 0.0;
101 out->error = INFINITY;
102 out->num_evals = 0;

106 state.func = f;
107 state.user_data = ud;
108 state.out = out;
109 if (!lmmc_quad_romberg_initial(&state, a, b)) {
110 return LMMC_STATUS_NUMERICAL_FAILURE;

112 for (size_t i = 1; i < max_iter; ++i) {

115 return LMMC_STATUS_NUMERICAL_FAILURE;

118 return LMMC_STATUS_OK;
119 }
120 for (size_t j = 0; j <= i; ++j) state.prev[j] = state.curr[j];
121 }
122 return LMMC_STATUS_CONVERGENCE_FAILED;
123}
