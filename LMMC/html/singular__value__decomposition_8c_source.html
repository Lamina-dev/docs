16static void bidiag_qr_step(lmmc_real_t *d, lmmc_real_t *e,
17 size_t lo, size_t hi,
18 lmmc_mat_t *U, lmmc_mat_t *V) {
19
20 lmmc_real_t f, g;
21 {
22 lmmc_real_t emm = (hi >= lo + 2) ? e[hi - 2] : 0.0;
23 lmmc_real_t tnm = d[hi - 1] * d[hi - 1] + emm * emm;
24 lmmc_real_t tnn = d[hi] * d[hi] + e[hi - 1] * e[hi - 1];
25 lmmc_real_t tmn = d[hi - 1] * e[hi - 1];
26 lmmc_real_t bb = (tnm - tnn) * 0.5;
27 lmmc_real_t cc = tmn * tmn;
28 lmmc_real_t shift;
29 if (bb == 0.0 && cc == 0.0) shift = tnn;
30 else {
31 lmmc_real_t disc = sqrt(bb * bb + cc);
32 lmmc_real_t denom = (bb >= 0.0) ? (bb + disc) : (bb - disc);
33 shift = tnn - cc / denom;
34 }
35 f = d[lo] * d[lo] - shift;
36 g = d[lo] * e[lo];
37 }
38
39 size_t i;
40 for (i = lo; i < hi; i++) {
41 lmmc_real_t c, s, r;
42
43 givens_compute(f, g, &c, &s, &r);
44 if (i > lo) e[i - 1] = r;
45 f = c * d[i] + s * e[i];
46 e[i] = c * e[i] - s * d[i];
47 g = s * d[i + 1];
48 d[i + 1] = c * d[i + 1];
49 if (V) givens_right(V, V->rows, i, i + 1, c, s);
50
51
52 givens_compute(f, g, &c, &s, &r);
53 d[i] = r;
54 f = c * e[i] + s * d[i + 1];
55 d[i + 1] = c * d[i + 1] - s * e[i];
56 if (i + 1 < hi) {
57 g = s * e[i + 1];
58 e[i + 1] = c * e[i + 1];
59 }
60 if (U) givens_right(U, U->rows, i, i + 1, c, s);
61 }
62 e[hi - 1] = f;
63}

65static lmmc_status_t bidiagonalize(const lmmc_mat_t *a,
66 lmmc_real_t *d, lmmc_real_t *e,
67 lmmc_mat_t *U, lmmc_mat_t *V) {
68 size_t m = a->rows;
69 size_t n = a->cols;
70 size_t i, j, k;
71 lmmc_status_t status;
72
73
74 lmmc_mat_t W;
75 status = lmmc_mat_create(m, n, &W);
76 if (status != LMMC_STATUS_OK) return status;
77 for (i = 0; i < m; i++)
78 for (j = 0; j < n; j++)
79 MAT_ELEM(&W, i, j) = MAT_ELEM(a, i, j);
80
81
82 for (i = 0; i < m; i++)
83 for (j = 0; j < m; j++) MAT_ELEM(U, i, j) = (i == j) ? 1.0 : 0.0;
84 for (i = 0; i < n; i++)
85 for (j = 0; j < n; j++) MAT_ELEM(V, i, j) = (i == j) ? 1.0 : 0.0;
86
87 lmmc_real_t *vbuf = (lmmc_real_t *)lmmc_alloc(((m > n) ? m : n) * sizeof(lmmc_real_t));
88 if (!vbuf) { lmmc_mat_destroy(&W); return LMMC_STATUS_ALLOCATION_FAILED; }
89
90 for (k = 0; k < n; k++) {
91
92 size_t len = m - k;
93 for (i = 0; i < len; i++) vbuf[i] = MAT_ELEM(&W, k + i, k);
94 lmmc_real_t tau, beta;
95 householder_make(vbuf, len, &tau, &beta);
96 d[k] = beta;
97
98 if (k + 1 < n) {
99 householder_apply_left(&W, k, len, k + 1, n, vbuf, tau);
100 }
101
102 if (tau != 0.0) {
103 size_t r;
104 for (r = 0; r < m; r++) {
105 lmmc_real_t s = MAT_ELEM(U, r, k);
106 for (i = 1; i < len; i++) s += vbuf[i] * MAT_ELEM(U, r, k + i);
107 s *= tau;
108 MAT_ELEM(U, r, k) -= s;
109 for (i = 1; i < len; i++) MAT_ELEM(U, r, k + i) -= s * vbuf[i];
110 }
111 }
112
113
114 if (k + 1 < n) {
115 size_t rlen = n - k - 1;
116 for (j = 0; j < rlen; j++) vbuf[j] = MAT_ELEM(&W, k, k + 1 + j);
117 householder_make(vbuf, rlen, &tau, &beta);
118 e[k] = beta;
119
120 if (k + 1 < m) {
121 householder_apply_right(&W, k + 1, m, k + 1, rlen, vbuf, tau);
122 }
123
124 if (tau != 0.0) {
125 size_t r;
126 for (r = 0; r < n; r++) {
127 lmmc_real_t s = MAT_ELEM(V, r, k + 1);
128 for (j = 1; j < rlen; j++) s += vbuf[j] * MAT_ELEM(V, r, k + 1 + j);
129 s *= tau;
130 MAT_ELEM(V, r, k + 1) -= s;
131 for (j = 1; j < rlen; j++) MAT_ELEM(V, r, k + 1 + j) -= s * vbuf[j];
132 }
133 }
134 }
135 }
136
137 lmmc_free(vbuf);
138 lmmc_mat_destroy(&W);
139 return LMMC_STATUS_OK;
140}

142static lmmc_status_t svd_tall(const lmmc_mat_t *a,
143 lmmc_mat_t *out_U,
144 lmmc_vec_t *out_sigma,
145 lmmc_mat_t *out_Vt) {
146 size_t m = a->rows;
147 size_t n = a->cols;
148 size_t i, j, k;
149 lmmc_status_t status;
150 int U_init = 0, sigma_init = 0, Vt_init = 0;
151 lmmc_mat_t V; int V_init = 0;
152 lmmc_real_t *d = NULL;
153 lmmc_real_t *e = NULL;
154 size_t *order = NULL;
155
156 status = lmmc_mat_create(m, m, out_U);
157 if (status != LMMC_STATUS_OK) goto fail;
158 U_init = 1;
159 status = lmmc_vec_create(n, out_sigma);
160 if (status != LMMC_STATUS_OK) goto fail;
161 sigma_init = 1;
162 status = lmmc_mat_create(n, n, out_Vt);
163 if (status != LMMC_STATUS_OK) goto fail;
164 Vt_init = 1;
165 status = lmmc_mat_create(n, n, &V);
166 if (status != LMMC_STATUS_OK) goto fail;
167 V_init = 1;
168
169 d = (lmmc_real_t *)lmmc_alloc(n * sizeof(lmmc_real_t));
170 e = (lmmc_real_t *)lmmc_alloc(((n > 0) ? n : 1) * sizeof(lmmc_real_t));
171 if (!d || !e) { status = LMMC_STATUS_ALLOCATION_FAILED; goto fail; }
172 for (i = 0; i < n; i++) { d[i] = 0.0; e[i] = 0.0; }
173
174
175 status = bidiagonalize(a, d, e, out_U, &V);
176 if (status != LMMC_STATUS_OK) goto fail;
177
178 if (n >= 1) e[n - 1] = 0.0;
179
180
181 {
182 const size_t max_iter_total = 30 * n + 30;
183 size_t iter = 0;
184 size_t hi = (n == 0) ? 0 : n - 1;
185 while (n >= 1) {
186
187 while (hi > 0) {
188 lmmc_real_t thr = LMMC_REAL_EPSILON *
189 (lmmc_abs(d[hi - 1]) + lmmc_abs(d[hi]));
190 if (lmmc_abs(e[hi - 1]) <= thr) {
191 e[hi - 1] = 0.0;
192 hi--;
193 } else {
194 break;
195 }
196 }
197 if (hi == 0) break;
198
199
200 size_t lo = hi;
201 while (lo > 0) {
202 lmmc_real_t thr = LMMC_REAL_EPSILON *
203 (lmmc_abs(d[lo - 1]) + lmmc_abs(d[lo]));
204 if (lmmc_abs(e[lo - 1]) <= thr) {
205 e[lo - 1] = 0.0;
206 break;
207 }
208 lo--;
209 }
210
211 int handled = 0;
212 for (k = lo; k < hi; k++) {
213 if (d[k] == 0.0) {
214
215 lmmc_real_t f = e[k];
216 e[k] = 0.0;
217 for (i = k + 1; i <= hi; i++) {
218 lmmc_real_t c, s, r;
219 givens_compute(d[i], f, &c, &s, &r);
220 d[i] = r;
221 if (i < hi) {
222 f = -s * e[i];
223 e[i] = c * e[i];
224 }
225 givens_right(out_U, out_U->rows, k, i, c, s);
226 }
227 handled = 1;
228 break;
229 }
230 }
231 if (handled) continue;
232
233 if (iter++ > max_iter_total) {
234 status = LMMC_STATUS_CONVERGENCE_FAILED;
235 goto fail;
236 }
237 bidiag_qr_step(d, e, lo, hi, out_U, &V);
238 }
239 }
240
241
242 for (i = 0; i < n; i++) {
243 if (d[i] < 0.0) {
244 d[i] = -d[i];
245 for (k = 0; k < n; k++) MAT_ELEM(&V, k, i) = -MAT_ELEM(&V, k, i);
246 }
247 }
248
249
250 order = (size_t *)lmmc_alloc(n * sizeof(size_t));
251 if (!order) { status = LMMC_STATUS_ALLOCATION_FAILED; goto fail; }
252 for (i = 0; i < n; i++) order[i] = i;
253
254 for (i = 0; i + 1 < n; i++) {
255 size_t mx = i;
256 for (j = i + 1; j < n; j++) {
257 if (d[order[j]] > d[order[mx]]) mx = j;
258 }
259 if (mx != i) { size_t tmp = order[i]; order[i] = order[mx]; order[mx] = tmp; }
260 }
261
262
263 for (i = 0; i < n; i++) out_sigma->data[i] = d[order[i]];
264
265
266 for (i = 0; i < n; i++)
267 for (k = 0; k < n; k++)
268 MAT_ELEM(out_Vt, i, k) = MAT_ELEM(&V, k, order[i]);
269
270
271 {
272 lmmc_real_t *col_buf = (lmmc_real_t *)lmmc_alloc(m * n * sizeof(lmmc_real_t));
273 if (!col_buf) { status = LMMC_STATUS_ALLOCATION_FAILED; goto fail; }
274 for (j = 0; j < n; j++)
275 for (i = 0; i < m; i++)
276 col_buf[j * m + i] = MAT_ELEM(out_U, i, j);
277 for (j = 0; j < n; j++) {
278 size_t src = order[j];
279 for (i = 0; i < m; i++)
280 MAT_ELEM(out_U, i, j) = col_buf[src * m + i];
281 }
282 lmmc_free(col_buf);
283 }
284
285 lmmc_free(d);
286 lmmc_free(e);
287 lmmc_free(order);
288 lmmc_mat_destroy(&V);
289 return LMMC_STATUS_OK;
290
291fail:
292 if (order) lmmc_free(order);
293 if (d) lmmc_free(d);
294 if (e) lmmc_free(e);
295 if (V_init) lmmc_mat_destroy(&V);
296 if (Vt_init) lmmc_mat_destroy(out_Vt);
297 if (sigma_init) lmmc_vec_destroy(out_sigma);
298 if (U_init) lmmc_mat_destroy(out_U);
299 return status;
300}

302lmmc_status_t lmmc_svd(const lmmc_mat_t *a, lmmc_svd_result_t *out_result) {
303 if (!a || !out_result || !a->data) return LMMC_STATUS_INVALID_ARGUMENT;
304 if (a->rows == 0 || a->cols == 0) return LMMC_STATUS_INVALID_ARGUMENT;
305
306 size_t m = a->rows;
307 size_t n = a->cols;

368 lmmc_mat_destroy(&Vtp);

372lmmc_status_t lmmc_pinv(const lmmc_mat_t *a, lmmc_real_t tol, lmmc_mat_t *out_pinv) {
373 if (!a || !out_pinv || !a->data || !out_pinv->data)
374 return LMMC_STATUS_INVALID_ARGUMENT;
375 size_t m = a->rows;
376 size_t n = a->cols;
377 if (m == 0 || n == 0) return LMMC_STATUS_INVALID_ARGUMENT;
378 if (out_pinv->rows != n || out_pinv->cols != m)
379 return LMMC_STATUS_DIMENSION_MISMATCH;
380
381 lmmc_svd_result_t svd;
382 lmmc_status_t status = lmmc_svd(a, &svd);
383 if (status != LMMC_STATUS_OK) return status;
384
385 size_t p = (m < n) ? m : n;
386
387 lmmc_real_t sigma_max = (p > 0) ? svd.sigma.data[0] : 0.0;
388 lmmc_real_t use_tol = tol;
389 if (use_tol <= 0.0) {
390 lmmc_real_t mn = (m > n) ? (lmmc_real_t)m : (lmmc_real_t)n;
391 use_tol = LMMC_REAL_EPSILON * mn * sigma_max;
392 }
393
394
395 size_t i, j, k;
396 for (i = 0; i < n; i++) {
397 for (j = 0; j < m; j++) {
398 lmmc_real_t sum = 0.0;
399 for (k = 0; k < p; k++) {
400 lmmc_real_t s = svd.sigma.data[k];
401 if (s > use_tol) {
402 sum += MAT_ELEM(&svd.Vt, k, i) * MAT_ELEM(&svd.U, j, k) / s;
403 }
404 }
405 MAT_ELEM(out_pinv, i, j) = sum;
406 }
407 }
408
409 lmmc_svd_result_destroy(&svd);
410 return LMMC_STATUS_OK;
411}

413lmmc_status_t lmmc_cond(const lmmc_mat_t *a, lmmc_real_t *out_cond) {
414 if (!a || !out_cond || !a->data) return LMMC_STATUS_INVALID_ARGUMENT;
415 if (a->rows == 0 || a->cols == 0) return LMMC_STATUS_INVALID_ARGUMENT;
416
417 lmmc_svd_result_t svd;
418 lmmc_status_t status = lmmc_svd(a, &svd);
419 if (status != LMMC_STATUS_OK) return status;
420
421 size_t p = svd.sigma.size;
422 if (p == 0) {
423 lmmc_svd_result_destroy(&svd);
424 return LMMC_STATUS_INVALID_ARGUMENT;
425 }
426 lmmc_real_t s_max = svd.sigma.data[0];
427 lmmc_real_t s_min = svd.sigma.data[p - 1];
428 if (s_min <= 0.0) {
429 *out_cond = INFINITY;
430 } else {
431 *out_cond = s_max / s_min;
432 }
433 lmmc_svd_result_destroy(&svd);
434 return LMMC_STATUS_OK;
435}
