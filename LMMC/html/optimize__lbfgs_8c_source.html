37static void lbfgs_scale_direction(lmmc_lbfgs_workspace_t* work)
38{
39 const size_t n = work->g.size;
40 size_t i, newest;
41 lmmc_real_t sy = 0.0, yy = 0.0;
42 if (work->history_count == 0) return;
43 newest = (work->oldest + work->history_count - 1) % work->s.rows;
44 for (i = 0; i < n; ++i) {
45 sy += work->s.data[newest * n + i] * work->y.data[newest * n + i];
46 yy += work->y.data[newest * n + i] * work->y.data[newest * n + i];
47 }
48 if (yy > 1e-300) {
49 const lmmc_real_t gamma = sy / yy;
50 for (i = 0; i < n; ++i) {
51 work->q.data[i] *= gamma;
52 }
53 }

57static void lbfgs_direction(lmmc_lbfgs_workspace_t* work)
58{
59 const size_t n = work->g.size;
60 const size_t m = work->s.rows;
61 const size_t bound = work->history_count;
62 size_t k, i;
63 memcpy(work->q.data, work->g.data, n * sizeof(lmmc_real_t));
64 for (k = 0; k < bound; ++k) {
65 const size_t idx = (work->oldest + work->history_count - 1 - k) % m;
66 lmmc_real_t dot = 0.0;
67 for (i = 0; i < n; ++i) {
68 dot += work->s.data[idx * n + i] * work->q.data[i];
69 }
70 work->alpha.data[k] = work->rho.data[idx] * dot;
71 for (i = 0; i < n; ++i) {
72 work->q.data[i] -= work->alpha.data[k] * work->y.data[idx * n + i];
73 }
74 }
75 lbfgs_scale_direction(work);
76 for (k = bound; k-- > 0;) {
77 const size_t idx = (work->oldest + work->history_count - 1 - k) % m;
78 lmmc_real_t dot = 0.0;
79 lmmc_real_t beta;
80 for (i = 0; i < n; ++i) {
81 dot += work->y.data[idx * n + i] * work->q.data[i];
82 }
83 beta = work->rho.data[idx] * dot;
84 for (i = 0; i < n; ++i) {
85 work->q.data[i] += (work->alpha.data[k] - beta) * work->s.data[idx * n + i];
86 }
87 }
88 for (i = 0; i < n; ++i) {
89 work->q.data[i] = -work->q.data[i];
90 }
91}

93static void lbfgs_store_history(
94 lmmc_lbfgs_workspace_t* work, const lmmc_vec_t* x)
95{
96 const size_t n = x->size;
97 const size_t m = work->s.rows;
98 size_t store_idx, i;
99 lmmc_real_t sy = 0.0;
100 if (work->history_count < m) {
101 store_idx = (work->oldest + work->history_count) % m;
102 work->history_count++;
103 } else {
104 store_idx = work->oldest;
105 work->oldest = (work->oldest + 1) % m;
106 }
107 for (i = 0; i < n; ++i) {
108 work->s.data[store_idx * n + i] = work->x_new.data[i] - x->data[i];
109 work->y.data[store_idx * n + i] = work->g_new.data[i] - work->g.data[i];
110 sy += work->s.data[store_idx * n + i] * work->y.data[store_idx * n + i];
111 }
112 if (fabs(sy) > 1e-300) {
113 work->rho.data[store_idx] = 1.0 / sy;
114 } else {
115 work->rho.data[store_idx] = 0.0;
116 }
117}

119static lmmc_real_t lbfgs_descent_derivative(
120 lmmc_lbfgs_workspace_t* work, lmmc_real_t norm)
121{
122 size_t i;
123 lmmc_real_t dg = 0.0;
124 for (i = 0; i < work->g.size; ++i) {
125 dg += work->g.data[i] * work->q.data[i];
126 }
127 if (dg >= 0.0) {
128 for (i = 0; i < work->g.size; ++i) {
129 work->q.data[i] = -work->g.data[i];
130 }
131 dg = -norm * norm;
132 }
133 return dg;
134}

136static lmmc_optimize_failure_t lbfgs_step(
137 lmmc_lbfgs_workspace_t* work, lmmc_opt_obj_t obj,
138 lmmc_opt_grad_t grad, void* user_data, lmmc_vec_t* x,
139 lmmc_real_t* value, lmmc_real_t norm)
140{
141 lmmc_real_t dg;
142 lbfgs_direction(work);
143 dg = lbfgs_descent_derivative(work, norm);
144 if (!lmmc_optimize_armijo(obj, user_data, x, &work->q,
145 &work->x_new, value, dg, 40, 0)) {
146 return LMMC_OPT_FAILURE_LINE_SEARCH_FAILED;
147 }
148 if (grad(&work->x_new, &work->g_new, user_data) != LMMC_STATUS_OK) {
149 return LMMC_OPT_FAILURE_NUMERICAL_ISSUE;
150 }
151 lbfgs_store_history(work, x);
152 memcpy(x->data, work->x_new.data, x->size * sizeof(lmmc_real_t));
153 memcpy(work->g.data, work->g_new.data, x->size * sizeof(lmmc_real_t));
154 return LMMC_OPT_FAILURE_NONE;
155}
