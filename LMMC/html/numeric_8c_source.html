34static int lmmc_is_power_of_four(size_t n, unsigned* out_digits) {
35 size_t t = n;
36 unsigned digits = 0;
37
38 if (n == 0) {
39 return 0;
40 }
41 while ((t & (size_t)3) == 0) {
42 t >>= 2;
43 ++digits;
44 }
45 if (t != (size_t)1) {
46 return 0;
47 }
48
49 if (out_digits != NULL) {
50 *out_digits = digits;
51 }
52 return 1;
53}

55static size_t lmmc_reverse_base4(size_t value, unsigned digits) {
56 size_t reversed = 0;
57 unsigned i = 0;
58
59 for (i = 0; i < digits; ++i) {
60 reversed = (reversed << 2) | (value & (size_t)3);
61 value >>= 2;
62 }
63 return reversed;
64}

209lmmc_status_t lmmc_approx_eq(lmmc_real_t a, lmmc_real_t b, lmmc_real_t epsilon, int* out_equal) {
210 lmmc_real_t diff, zero;
211 if (out_equal == NULL) return LMMC_STATUS_INVALID_ARGUMENT;
212 if (isnan(a) || isnan(b)) {
213 *out_equal = 0;
214 return LMMC_STATUS_OK;
215 }
216 if (isinf(a) || isinf(b)) {
217 *out_equal = (LMMC_REAL_CMP(&a, &b) == 0) ? 1 : 0;
218 return LMMC_STATUS_OK;
219 }
220
221 LMMC_REAL_INIT(&diff);
222 LMMC_REAL_INIT(&zero);
223
224 LMMC_REAL_SET_D(&zero, 0.0);
225 LMMC_REAL_SUB(&diff, &a, &b);
226
227 if (LMMC_REAL_CMP(&diff, &zero) < 0) {
228 lmmc_real_t tmp;
229 LMMC_REAL_INIT(&tmp);
230 LMMC_REAL_SET(&tmp, &diff);
231 LMMC_REAL_SUB(&diff, &zero, &tmp);
232 LMMC_REAL_CLEAR(&tmp);
233 }
234 *out_equal = (LMMC_REAL_CMP(&diff, &epsilon) <= 0) ? 1 : 0;
235
236 LMMC_REAL_CLEAR(&diff);
237 LMMC_REAL_CLEAR(&zero);
238 return LMMC_STATUS_OK;
239}

259static lmmc_status_t lmmc_fft_radix4_core(lmmc_real_t* real, lmmc_real_t* imag, size_t n, int inverse) {
260 unsigned digits = 0;
261 size_t i = 0;
262 size_t len = 0;
263
264 if (!lmmc_is_power_of_four(n, &digits)) {
265 return LMMC_STATUS_INVALID_ARGUMENT;
266 }
267 if (n <= 1) {
268 return LMMC_STATUS_OK;
269 }
270
271 for (i = 0; i < n; ++i) {
272 size_t r = lmmc_reverse_base4(i, digits);
273 if (r > i) {
274 lmmc_swapd(&real[i], &real[r]);
275 lmmc_swapd(&imag[i], &imag[r]);
276 }
277 }
278
279 len = 4;
280 while (len <= n) {
281 size_t group = len;
282 size_t quarter = group / 4;
283
284 lmmc_real_t angle_step, tmp1, tmp2;
285 LMMC_REAL_INIT(&angle_step);
286 LMMC_REAL_INIT(&tmp1);
287 LMMC_REAL_INIT(&tmp2);
288
289 LMMC_REAL_SET_D(&tmp1, (inverse ? 2.0 : -2.0) * LMMC_PI);
290 LMMC_REAL_SET_D(&tmp2, (double)group);
291 LMMC_REAL_DIV(&angle_step, &tmp1, &tmp2);
292
293 for (i = 0; i < n; i += group) {
294 size_t j = 0;
295 for (j = 0; j < quarter; ++j) {
296 size_t i0 = i + j;
297 size_t i1 = i0 + quarter;
298 size_t i2 = i1 + quarter;
299 size_t i3 = i2 + quarter;
300
301 lmmc_real_t a0r, a0i, a1r, a1i, a2r, a2i, a3r, a3i;
302 lmmc_real_t ang1, ang2, ang3;
303 lmmc_real_t c1, s1, c2, s2, c3, s3;
304 double d_ang1, d_ang2, d_ang3;
305
306 LMMC_REAL_INIT(&a0r); LMMC_REAL_INIT(&a0i);
307 LMMC_REAL_INIT(&a1r); LMMC_REAL_INIT(&a1i);
308 LMMC_REAL_INIT(&a2r); LMMC_REAL_INIT(&a2i);
309 LMMC_REAL_INIT(&a3r); LMMC_REAL_INIT(&a3i);
310 LMMC_REAL_INIT(&ang1); LMMC_REAL_INIT(&ang2); LMMC_REAL_INIT(&ang3);
311 LMMC_REAL_INIT(&c1); LMMC_REAL_INIT(&s1);
312 LMMC_REAL_INIT(&c2); LMMC_REAL_INIT(&s2);
313 LMMC_REAL_INIT(&c3); LMMC_REAL_INIT(&s3);
314
315 LMMC_REAL_SET(&a0r, &real[i0]);
316 LMMC_REAL_SET(&a0i, &imag[i0]);
317 LMMC_REAL_SET_D(&a1r, 0.0); LMMC_REAL_SET_D(&a1i, 0.0);
318 LMMC_REAL_SET_D(&a2r, 0.0); LMMC_REAL_SET_D(&a2i, 0.0);
319 LMMC_REAL_SET_D(&a3r, 0.0); LMMC_REAL_SET_D(&a3i, 0.0);
320
321 LMMC_REAL_SET_D(&tmp1, (double)j);
322 LMMC_REAL_MUL(&ang1, &angle_step, &tmp1);
323 LMMC_REAL_SET_D(&tmp1, 2.0);
324 LMMC_REAL_MUL(&ang2, &ang1, &tmp1);
325 LMMC_REAL_SET_D(&tmp1, 3.0);
326 LMMC_REAL_MUL(&ang3, &ang1, &tmp1);
327
328 d_ang1 = ((inverse ? 2.0 : -2.0) * LMMC_PI / (double)group) * (double)j;
329 d_ang2 = d_ang1 * 2.0;
330 d_ang3 = d_ang1 * 3.0;
331
332 LMMC_REAL_SET_D(&c1, cos(d_ang1)); LMMC_REAL_SET_D(&s1, sin(d_ang1));
333 LMMC_REAL_SET_D(&c2, cos(d_ang2)); LMMC_REAL_SET_D(&s2, sin(d_ang2));
334 LMMC_REAL_SET_D(&c3, cos(d_ang3)); LMMC_REAL_SET_D(&s3, sin(d_ang3));
335
336 lmmc_complex_mul(&real[i1], &imag[i1], &c1, &s1, &a1r, &a1i);
337 lmmc_complex_mul(&real[i2], &imag[i2], &c2, &s2, &a2r, &a2i);
338 lmmc_complex_mul(&real[i3], &imag[i3], &c3, &s3, &a3r, &a3i);
339
340 LMMC_REAL_ADD(&tmp1, &a0r, &a1r); LMMC_REAL_ADD(&tmp1, &tmp1, &a2r); LMMC_REAL_ADD(&real[i0], &tmp1, &a3r);
341 LMMC_REAL_ADD(&tmp1, &a0i, &a1i); LMMC_REAL_ADD(&tmp1, &tmp1, &a2i); LMMC_REAL_ADD(&imag[i0], &tmp1, &a3i);
342
343 LMMC_REAL_SUB(&tmp1, &a0r, &a1r); LMMC_REAL_ADD(&tmp1, &tmp1, &a2r); LMMC_REAL_SUB(&real[i2], &tmp1, &a3r);
344 LMMC_REAL_SUB(&tmp1, &a0i, &a1i); LMMC_REAL_ADD(&tmp1, &tmp1, &a2i); LMMC_REAL_SUB(&imag[i2], &tmp1, &a3i);
345
346 if (!inverse) {
347 LMMC_REAL_ADD(&tmp1, &a0r, &a1i); LMMC_REAL_SUB(&tmp1, &tmp1, &a2r); LMMC_REAL_SUB(&real[i1], &tmp1, &a3i);
348 LMMC_REAL_SUB(&tmp1, &a0i, &a1r); LMMC_REAL_SUB(&tmp1, &tmp1, &a2i); LMMC_REAL_ADD(&imag[i1], &tmp1, &a3r);
349 LMMC_REAL_SUB(&tmp1, &a0r, &a1i); LMMC_REAL_SUB(&tmp1, &tmp1, &a2r); LMMC_REAL_ADD(&real[i3], &tmp1, &a3i);
350 LMMC_REAL_ADD(&tmp1, &a0i, &a1r); LMMC_REAL_SUB(&tmp1, &tmp1, &a2i); LMMC_REAL_SUB(&imag[i3], &tmp1, &a3r);
351 } else {
352 LMMC_REAL_SUB(&tmp1, &a0r, &a1i); LMMC_REAL_SUB(&tmp1, &tmp1, &a2r); LMMC_REAL_ADD(&real[i1], &tmp1, &a3i);
353 LMMC_REAL_ADD(&tmp1, &a0i, &a1r); LMMC_REAL_SUB(&tmp1, &tmp1, &a2i); LMMC_REAL_SUB(&imag[i1], &tmp1, &a3r);
354 LMMC_REAL_ADD(&tmp1, &a0r, &a1i); LMMC_REAL_SUB(&tmp1, &tmp1, &a2r); LMMC_REAL_SUB(&real[i3], &tmp1, &a3i);
355 LMMC_REAL_SUB(&tmp1, &a0i, &a1r); LMMC_REAL_SUB(&tmp1, &tmp1, &a2i); LMMC_REAL_ADD(&imag[i3], &tmp1, &a3r);
356 }
357
358 LMMC_REAL_CLEAR(&a0r); LMMC_REAL_CLEAR(&a0i);
359 LMMC_REAL_CLEAR(&a1r); LMMC_REAL_CLEAR(&a1i);
360 LMMC_REAL_CLEAR(&a2r); LMMC_REAL_CLEAR(&a2i);
361 LMMC_REAL_CLEAR(&a3r); LMMC_REAL_CLEAR(&a3i);
362 LMMC_REAL_CLEAR(&ang1); LMMC_REAL_CLEAR(&ang2); LMMC_REAL_CLEAR(&ang3);
363 LMMC_REAL_CLEAR(&c1); LMMC_REAL_CLEAR(&s1);
364 LMMC_REAL_CLEAR(&c2); LMMC_REAL_CLEAR(&s2);
365 LMMC_REAL_CLEAR(&c3); LMMC_REAL_CLEAR(&s3);
366 }
367 }
368 LMMC_REAL_CLEAR(&angle_step);
369 LMMC_REAL_CLEAR(&tmp1);
370 LMMC_REAL_CLEAR(&tmp2);
371
372 if (len == n) {
373 break;
374 }
375 len *= 4;
376 }
377
378 if (inverse) {
379 lmmc_real_t scale, tmp1, tmp2;
380 LMMC_REAL_INIT(&scale);
381 LMMC_REAL_INIT(&tmp1);
382 LMMC_REAL_INIT(&tmp2);
383
384 LMMC_REAL_SET_D(&tmp1, 1.0);
385 LMMC_REAL_SET_D(&tmp2, (double)n);
386 LMMC_REAL_DIV(&scale, &tmp1, &tmp2);
387 for (i = 0; i < n; ++i) {
388 LMMC_REAL_MUL(&real[i], &real[i], &scale);
389 LMMC_REAL_MUL(&imag[i], &imag[i], &scale);
390 }
391
392 LMMC_REAL_CLEAR(&scale);
393 LMMC_REAL_CLEAR(&tmp1);
394 LMMC_REAL_CLEAR(&tmp2);
395 }
396
397 return LMMC_STATUS_OK;
398}

453lmmc_status_t lmmc_lgamma(lmmc_real_t x, lmmc_real_t* out) {
454 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
455 if (x <= 0.0) return LMMC_STATUS_INVALID_ARGUMENT;
456
457 /* Lanczos 系数 (g=7, n=9) - 来自 Numerical Recipes / Cephes */
458 static const double coeff[9] = {
459 0.99999999999980993,
460 676.5203681218851,
461 -1259.1392167224028,
462 771.32342877765313,
463 -176.61502916214059,
464 12.507343278686905,
465 -0.13857109526572012,
466 9.9843695780195716e-6,
467 1.5056327351493116e-7
468 };
469
470 double g = 7.0;
471 double tmp, ser;
472 int i;
473
474 if (x < 0.5) {
475 /* 使用反射公式: Gamma(x) * Gamma(1-x) = pi / sin(pi*x) */
476 /* lgamma(x) = ln(pi) - ln(sin(pi*x)) - lgamma(1-x) */
477 double sinpx = sin(LMMC_PI * x);
478 if (fabs(sinpx) < 1e-300) return LMMC_STATUS_NUMERICAL_FAILURE;
479 lmmc_real_t lg1mx;
480 lmmc_status_t st = lmmc_lgamma(1.0 - x, &lg1mx);
481 if (st != LMMC_STATUS_OK) return st;
482 *out = log(LMMC_PI) - log(fabs(sinpx)) - lg1mx;
483 return LMMC_STATUS_OK;
484 }
485
486 x -= 1.0;
487 tmp = x + g + 0.5;
488 ser = coeff[0];
489 for (i = 1; i < 9; i++) {
490 ser += coeff[i] / (x + (double)i);
491 }
492
493 *out = 0.5 * log(2.0 * LMMC_PI) + (x + 0.5) * log(tmp) - tmp + log(ser);
494 return LMMC_STATUS_OK;
495}

522lmmc_status_t lmmc_beta(lmmc_real_t a, lmmc_real_t b, lmmc_real_t* out) {
523 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
524 if (a <= 0.0 || b <= 0.0) return LMMC_STATUS_INVALID_ARGUMENT;
525
526 lmmc_real_t lga, lgb, lgab;
527 lmmc_status_t st;
528
529 st = lmmc_lgamma(a, &lga);
530 if (st != LMMC_STATUS_OK) return st;
531
532 st = lmmc_lgamma(b, &lgb);
533 if (st != LMMC_STATUS_OK) return st;
534
535 st = lmmc_lgamma(a + b, &lgab);
536 if (st != LMMC_STATUS_OK) return st;
537
538 double result = lga + lgb - lgab;
539 if (result > 709.0) {
540 return LMMC_STATUS_NUMERICAL_FAILURE;
541 }
542
543 *out = exp(result);

556lmmc_status_t lmmc_digamma(lmmc_real_t x, lmmc_real_t* out) {
557 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
558 if (x <= 0.0) return LMMC_STATUS_INVALID_ARGUMENT;
559
560 double result = 0.0;
561
562 /* 递推:将 x 提升到 >= 7 以确保渐近展开精度 */
563 while (x < 7.0) {
564 result -= 1.0 / x;
565 x += 1.0;
566 }
567
568 /* 渐近展开 psi(x) ~ ln(x) - 1/(2x) - 1/(12x^2) + 1/(120x^4) - 1/(252x^6) + ... */
569 /* Bernoulli 数: B2=1/6, B4=-1/30, B6=1/42, B8=-1/30, B10=5/66, B12=-691/2730 */
570 {
571 double ix = 1.0 / x;
572 double ix2 = ix * ix;
573
574 /* 系数 = B_{2k} / (2k) */
575 static const double bernoulli_coeff[] = {
576 1.0 / 12.0, /* B2/(2*1) = (1/6)/2 */
577 -1.0 / 120.0, /* B4/(2*2) = (-1/30)/4 */
578 1.0 / 252.0, /* B6/(2*3) = (1/42)/6 */
579 -1.0 / 240.0, /* B8/(2*4) = (-1/30)/8 */
580 5.0 / 660.0, /* B10/(2*5) = (5/66)/10 */
581 -691.0 / 32760.0, /* B12/(2*6) = (-691/2730)/12 */
582 1.0 / 12.0 /* B14/(2*7) = (7/6)/14 = 1/12 */
583 };
584
585 double sum = 0.0;
586 double ix2k = ix2; /* ix^(2k) starting at ix^2 */
587 int k;
588 for (k = 0; k < 7; k++) {
589 sum -= bernoulli_coeff[k] * ix2k;
590 ix2k *= ix2;
591 }
592
593 result += log(x) - 0.5 * ix + sum;
594 }
595
596 *out = result;
597 return LMMC_STATUS_OK;
598}

600lmmc_status_t lmmc_lambertw(lmmc_real_t z, lmmc_real_t* out_res) {
601 const double tol = LMMC_DEFAULT_REL_TOL;
602 const int max_iter = 100;
603 double w, expw, w_expw, diff, abs_diff, denom, f_prime, f_double_prime;
604 double threshold;
605 int i;
606
607 if (!out_res) return LMMC_STATUS_INVALID_ARGUMENT;
608 if (z < -LMMC_INV_E) return LMMC_STATUS_INVALID_ARGUMENT;
609 if (z == 0.0) { *out_res = 0.0; return LMMC_STATUS_OK; }
610
611 /* Initial approximation */
612 if (z > 2.0) {
613 double lnz = log(z);
614 w = lnz - log(lnz);
615 } else if (z > -LMMC_INV_E && z < -0.3) {
616 /* Near the branch point: use a series expansion around -1/e */
617 double p = sqrt(2.0 * (exp(1.0) * z + 1.0));
618 w = -1.0 + p - p * p / 3.0;
619 } else if (z == -LMMC_INV_E) {
620 *out_res = -1.0;
621 return LMMC_STATUS_OK;
622 } else {
623 w = z;
624 }
625
626 /* Halley iteration */
627 for (i = 0; i < max_iter; ++i) {
628 expw = exp(w);
629 w_expw = w * expw;
630 diff = w_expw - z;
631 abs_diff = fabs(diff);
632 threshold = tol * (1.0 + fabs(z));
633 if (abs_diff <= threshold) {
634 *out_res = w;
635 return LMMC_STATUS_OK;
636 }
637 f_prime = expw * (w + 1.0);
638 f_double_prime = expw * (w + 2.0);
639 /* Halley step: w -= f / (f' - f * f'' / (2 * f')) */
640 denom = f_prime - diff * f_double_prime / (2.0 * f_prime);
641 if (fabs(denom) < 1e-300) {
642 /* Fallback to Newton step if Halley denominator is too small */
643 w -= diff / f_prime;
644 } else {
645 w -= diff / denom;
646 }
647 }
648 *out_res = w;
649 return LMMC_STATUS_OK;
650}

652lmmc_status_t lmmc_lambertw_wm1(lmmc_real_t z, lmmc_real_t* out_res) {
653 const double tol = LMMC_DEFAULT_REL_TOL;
654 const int max_iter = 100;
655 double w, expw, w_expw, diff, abs_diff, denom, f_prime, f_double_prime;
656 double threshold;
657 int i;
658
659 if (!out_res) return LMMC_STATUS_INVALID_ARGUMENT;
660 /* Domain: z in [-1/e, 0) */
661 if (z < -LMMC_INV_E || z >= 0.0) return LMMC_STATUS_INVALID_ARGUMENT;
662
663 /* Special case: z == -1/e => W_-_1(-1/e) = -1 */
664 if (fabs(z + LMMC_INV_E) < 1e-300) {
665 *out_res = -1.0;
666 return LMMC_STATUS_OK;
667 }
668
669 /* Initial guess from asymptotic: log(-z) - log(-log(-z)) */
670 {
671 double lnmz = log(-z);
672 double lnlnmz = log(-lnmz);
673 w = lnmz - lnlnmz;
674 }
675
676 /* Halley iteration */
677 for (i = 0; i < max_iter; ++i) {
678 expw = exp(w);
679 w_expw = w * expw;
680 diff = w_expw - z;
681 abs_diff = fabs(diff);
682 threshold = tol * (1.0 + fabs(z));
683 if (abs_diff <= threshold) {
684 *out_res = w;
685 return LMMC_STATUS_OK;
686 }
687 f_prime = expw * (w + 1.0);
688 f_double_prime = expw * (w + 2.0);
689 /* Halley step */
690 denom = f_prime - diff * f_double_prime / (2.0 * f_prime);
691 if (fabs(denom) < 1e-300) {
692 w -= diff / f_prime;
693 } else {
694 w -= diff / denom;
695 }
696 }
697 *out_res = w;
698 return LMMC_STATUS_OK;
699}

701lmmc_status_t lmmc_double_nearly_equal_tol(
702 lmmc_real_t a,
703 lmmc_real_t b,
704 lmmc_real_t abs_tol,
705 lmmc_real_t rel_tol,
706 int* out_equal
707) {
708 lmmc_real_t diff, scale, threshold, zero, tmp1, tmp2;
709
710 if (out_equal == NULL) {
711 return LMMC_STATUS_INVALID_ARGUMENT;
712 }
713
714 LMMC_REAL_INIT(&diff);
715 LMMC_REAL_INIT(&scale);
716 LMMC_REAL_INIT(&threshold);
717 LMMC_REAL_INIT(&zero);
718 LMMC_REAL_INIT(&tmp1);
719 LMMC_REAL_INIT(&tmp2);
720
721 LMMC_REAL_SET_D(&zero, 0.0);
722
723 if (LMMC_REAL_CMP(&abs_tol, &zero) < 0 || LMMC_REAL_CMP(&rel_tol, &zero) < 0 || !LMMC_REAL_IS_FINITE(&abs_tol) || !LMMC_REAL_IS_FINITE(&rel_tol)) {
724 LMMC_REAL_CLEAR(&diff); LMMC_REAL_CLEAR(&scale); LMMC_REAL_CLEAR(&threshold);
725 LMMC_REAL_CLEAR(&zero); LMMC_REAL_CLEAR(&tmp1); LMMC_REAL_CLEAR(&tmp2);
726 return LMMC_STATUS_INVALID_ARGUMENT;
727 }
728
729 if (isnan(a) || isnan(b)) {
730 LMMC_REAL_CLEAR(&diff); LMMC_REAL_CLEAR(&scale); LMMC_REAL_CLEAR(&threshold);
731 LMMC_REAL_CLEAR(&zero); LMMC_REAL_CLEAR(&tmp1); LMMC_REAL_CLEAR(&tmp2);
732 return LMMC_STATUS_NUMERICAL_FAILURE;
733 }
734
735 if (isinf(a) || isinf(b)) {
736 *out_equal = (LMMC_REAL_CMP(&a, &b) == 0) ? 1 : 0;
737 LMMC_REAL_CLEAR(&diff); LMMC_REAL_CLEAR(&scale); LMMC_REAL_CLEAR(&threshold);
738 LMMC_REAL_CLEAR(&zero); LMMC_REAL_CLEAR(&tmp1); LMMC_REAL_CLEAR(&tmp2);
739 return LMMC_STATUS_OK;
740 }
741
742 LMMC_REAL_SUB(&tmp1, &a, &b);
743 lmmc_abs_inplace(&diff, &tmp1);
744
745 lmmc_abs_inplace(&tmp1, &a);
746 lmmc_abs_inplace(&tmp2, &b);
747 lmmc_max_inplace(&scale, &tmp1, &tmp2);
748
749 LMMC_REAL_MUL(&tmp1, &rel_tol, &scale);
750 LMMC_REAL_ADD(&threshold, &abs_tol, &tmp1);
751
752 if (isnan(diff) || isnan(threshold)) {
753 LMMC_REAL_CLEAR(&diff); LMMC_REAL_CLEAR(&scale); LMMC_REAL_CLEAR(&threshold);
754 LMMC_REAL_CLEAR(&zero); LMMC_REAL_CLEAR(&tmp1); LMMC_REAL_CLEAR(&tmp2);
755 return LMMC_STATUS_NUMERICAL_FAILURE;
756 }
757
758 if (isinf(threshold)) {
759 *out_equal = 1;
760 LMMC_REAL_CLEAR(&diff); LMMC_REAL_CLEAR(&scale); LMMC_REAL_CLEAR(&threshold);
761 LMMC_REAL_CLEAR(&zero); LMMC_REAL_CLEAR(&tmp1); LMMC_REAL_CLEAR(&tmp2);
762 return LMMC_STATUS_OK;
763 }
764
765 *out_equal = (LMMC_REAL_CMP(&diff, &threshold) <= 0) ? 1 : 0;
766
767 LMMC_REAL_CLEAR(&diff);
768 LMMC_REAL_CLEAR(&scale);
769 LMMC_REAL_CLEAR(&threshold);
770 LMMC_REAL_CLEAR(&zero);
771 LMMC_REAL_CLEAR(&tmp1);
772 LMMC_REAL_CLEAR(&tmp2);
773 return LMMC_STATUS_OK;
774}
