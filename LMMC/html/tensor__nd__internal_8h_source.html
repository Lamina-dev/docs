14static inline lmmc_status_t lmmc_tensor_nd_layout(
15 size_t ndim, const size_t* dims, size_t* strides, size_t* total) {
16 size_t count = 1;
17 if (dims == NULL || total == NULL ||
18 ndim == 0 || ndim > LMMC_TENSOR_MAX_NDIM) {
19 return LMMC_STATUS_INVALID_ARGUMENT;
20 }
21 for (size_t i = ndim; i > 0; --i) {
22 const size_t axis = i - 1;
23 if (dims[axis] == 0) {
24 return LMMC_STATUS_INVALID_ARGUMENT;
25 }
26 if (strides != NULL) {
27 strides[axis] = count;
28 }
29 if (!lmmc_safe_mul_size(count, dims[axis], &count)) {
30 return LMMC_STATUS_INVALID_ARGUMENT;
31 }
32 }
33 *total = count;
34 return LMMC_STATUS_OK;
35}

37static inline int lmmc_tensor_nd_descriptor_is_valid(const lmmc_tensor_nd_t* tensor) {
38 size_t total;
39 size_t logical_bytes;
40 size_t max_offset = 0;
41 size_t elements;
42 size_t bytes;
43 if (tensor == NULL || tensor->data == NULL ||
44 lmmc_tensor_nd_layout(tensor->ndim, tensor->dims, NULL, &total) != LMMC_STATUS_OK ||
45 !lmmc_safe_mul_size(total, sizeof(lmmc_real_t), &logical_bytes)) {
46 return 0;
47 }
48 for (size_t i = 0; i < tensor->ndim; ++i) {
49 size_t axis_offset;
50 if (tensor->strides[i] == 0 ||
51 !lmmc_safe_mul_size(tensor->dims[i] - 1, tensor->strides[i], &axis_offset) ||
52 !lmmc_safe_add_size(max_offset, axis_offset, &max_offset)) {
53 return 0;
54 }
55 }
56 return lmmc_safe_add_size(max_offset, 1, &elements) &&
57 lmmc_safe_mul_size(elements, sizeof(lmmc_real_t), &bytes) &&
58 (uintptr_t)bytes <= UINTPTR_MAX - (uintptr_t)tensor->data;
59}

62static inline lmmc_status_t lmmc_tensor_nd_allocate_result(
63 size_t ndim, const size_t* dims, lmmc_tensor_nd_t* out) {
64 size_t total;
65 size_t bytes;
66 lmmc_tensor_nd_t result = {0};
67 if (out == NULL ||
68 lmmc_tensor_nd_layout(ndim, dims, result.strides, &total) != LMMC_STATUS_OK ||
69 !lmmc_safe_mul_size(total, sizeof(lmmc_real_t), &bytes)) {
70 return LMMC_STATUS_INVALID_ARGUMENT;
71 }
72 for (size_t i = 0; i < ndim; ++i) {
73 result.dims[i] = dims[i];
74 }
75 lmmc_real_t* data = (lmmc_real_t*)lmmc_memory_alloc(bytes);
76 if (data == NULL) {
77 return LMMC_STATUS_ALLOCATION_FAILED;
78 }
79 memset(data, 0, bytes);
80 for (size_t i = 0; i < total; ++i) {
81 LMMC_REAL_INIT(&data[i]);
82 }
83
84 result.ndim = ndim;
85 result.data = data;
86 result.owns_data = 1;
87 *out = result;
88 return LMMC_STATUS_OK;
89}
