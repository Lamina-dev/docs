47 lmmc_gmres_state_t* s) {
48 size_t scalar_count, bytes, metadata;
49 s->restart = restart == 0 ? (n < 30 ? n : 30) : restart;
50 if (s->restart > n) {
51 s->restart = n;

57 if (st != LMMC_STATUS_OK) {
58 return st;

71 s->r = lmmc_gmres_view(n, &next);
72 s->z = lmmc_gmres_view(n, &next);
73 s->w = lmmc_gmres_view(n, &next);
74 s->ax = lmmc_gmres_view(n, &next);
75 s->x_base = lmmc_gmres_view(n, &next);
76 s->x_trial = lmmc_gmres_view(n, &next);
77 for (size_t i = 0; i < basis_count; ++i) {
78 s->basis[i] = lmmc_gmres_view(n, &next);
79 }
80 s->h = next;
81 next += s->h_count;
82 s->cs = next;
83 next += s->restart;
84 s->sn = next;
85 next += s->restart;
86 s->g = next;

88 s->y = next;

122static void lmmc_gmres_previous_rotations(lmmc_gmres_state_t* s, size_t j) {
123 for (size_t i = 0; i < j; ++i) {
124 lmmc_real_t h0 = s->h[i * s->restart + j];
125 lmmc_real_t h1 = s->h[(i + 1) * s->restart + j];

135lmmc_status_t lmmc_gmres_rotate(lmmc_gmres_state_t* s, size_t j, int* breakdown) {
136 *breakdown = 0;
137 lmmc_gmres_previous_rotations(s, j);
138 lmmc_real_t hj = s->h[j * s->restart + j];
139 lmmc_real_t hsub = s->h[(j + 1) * s->restart + j];
140 lmmc_real_t first = hj * hj;
141 lmmc_real_t second = hsub * hsub;
142 lmmc_real_t denom = sqrt(first + second);
143 if (!isfinite(denom)) {
144 return LMMC_STATUS_NUMERICAL_FAILURE;
145 }
146 if (denom <= 1e-30) {
147 *breakdown = 1;
148 return LMMC_STATUS_OK;
149 }
150 s->cs[j] = hj / denom;
151 s->sn[j] = hsub / denom;
152 if (!isfinite(s->cs[j]) || !isfinite(s->sn[j])) {
153 return LMMC_STATUS_NUMERICAL_FAILURE;
154 }
155 first = s->cs[j] * hj;
156 second = s->sn[j] * hsub;
157 s->h[j * s->restart + j] = first + second;
158 s->h[(j + 1) * s->restart + j] = 0.0;
159 lmmc_real_t gtmp = s->cs[j] * s->g[j];
160 s->g[j + 1] = -s->sn[j] * s->g[j];
161 s->g[j] = gtmp;
162 return LMMC_STATUS_OK;
163}

165lmmc_status_t lmmc_gmres_back_substitute(lmmc_gmres_state_t* s, size_t dim) {
166 for (size_t ii = dim; ii > 0; --ii) {
167 size_t i = ii - 1;
168 lmmc_real_t sum = s->g[i];
169 for (size_t j = i + 1; j < dim; ++j) {
170 lmmc_real_t product = s->h[i * s->restart + j] * s->y[j];
171 sum = sum - product;
172 }
173 lmmc_real_t diag = s->h[i * s->restart + i];
174 if (!isfinite(sum) || !isfinite(diag) || fabs(diag) <= 1e-30) {
175 return LMMC_STATUS_NUMERICAL_FAILURE;
176 }
177 s->y[i] = sum / diag;
178 if (!isfinite(s->y[i])) {
179 return LMMC_STATUS_NUMERICAL_FAILURE;
180 }
181 }
182 return LMMC_STATUS_OK;
183}
