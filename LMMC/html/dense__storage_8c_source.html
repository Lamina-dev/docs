105lmmc_status_t lmmc_mat_transpose_to(const lmmc_mat_t* src, lmmc_mat_t* dst) {
106 size_t i = 0;
107 size_t j = 0;
108 if (src == NULL || dst == NULL || src->data == NULL || dst->data == NULL) {
109 return LMMC_STATUS_INVALID_ARGUMENT;
110 }
111 if (dst->rows != src->cols || dst->cols != src->rows) {
112 return LMMC_STATUS_DIMENSION_MISMATCH;
113 }
114 {
115 lmmc_storage_envelope_t src_envelope;
116 lmmc_storage_envelope_t dst_envelope;
117 if (!lmmc_storage_envelope_checked(src->data, src->rows, src->cols,
118 src->stride, sizeof(lmmc_real_t),
119 &src_envelope) ||
120 !lmmc_storage_envelope_checked(dst->data, dst->rows, dst->cols,
121 dst->stride, sizeof(lmmc_real_t),
122 &dst_envelope)) {
123 return LMMC_STATUS_INVALID_ARGUMENT;
124 }
125 if (lmmc_storage_envelopes_overlap(&src_envelope, &dst_envelope)) {
126 return LMMC_STATUS_INVALID_ARGUMENT;
127 }
128 }
129 for (i = 0; i < src->rows; ++i) {
130 for (j = 0; j < src->cols; ++j) {
131 LMMC_REAL_SET(&dst->data[j * dst->stride + i], &src->data[i * src->stride + j]);
132 }
133 }
134 return LMMC_STATUS_OK;
