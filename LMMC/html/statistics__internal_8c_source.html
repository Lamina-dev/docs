47lmmc_real_t lmmc_deviance_part(
48 lmmc_real_t x, lmmc_real_t mean) {
49 if (x == mean) return 0.0;
50 if (x == 0.0) return mean;
51 if (mean == 0.0) return INFINITY;
52
53 if (fabs(x - mean) < 0.1 * (x + mean)) {
54 const lmmc_real_t difference = x - mean;
55 lmmc_real_t ratio = difference / (x + mean);
56 const lmmc_real_t ratio_squared = ratio * ratio;
57 lmmc_real_t sum = 0.5 * difference * ratio;
58 lmmc_real_t term = x * ratio;
59 size_t order;
60
61 for (order = 1; order < 100; ++order) {
62 const lmmc_real_t previous = sum;
63 term *= ratio_squared;
64 sum += term / (lmmc_real_t)(2 * order + 1);
65 if (sum == previous) break;
66 }
67 return 2.0 * sum;
68 }
69
70 {
71 const lmmc_real_t quotient = x / mean;
72 const lmmc_real_t log_ratio =
73 isfinite(quotient) ? log(quotient) :
74 log(x) - log(mean);
75 return x > mean ?
76 x * (log_ratio - 1.0) + mean :
77 x * log_ratio + mean - x;
78 }
79}

90/*
91 * DLMF 8.12.3: leading uniform large-a expansion for Q(a, x).
92 * The continued fraction and power series require O(sqrt(a)) work around
93 * x == a; this expansion is both bounded-cost and stable in that region.
94 */
95static int regularized_gamma_large_central_upper(
96 lmmc_real_t a, lmmc_real_t x, lmmc_real_t* upper) {
97 lmmc_real_t delta, eta, eta_squared, c0, correction, value;
98
99 if (a < 10000.0 || x <= 0.0) return 0;
100 delta = x / a - 1.0;
101 if (fabs(delta) > 0.1) return 0;
102
103 eta_squared = 2.0 * (delta - log1p(delta));
104 if (eta_squared < 0.0) {
105 if (eta_squared > -DBL_EPSILON) {
106 eta_squared = 0.0;
107 } else {
108 return 0;
109 }
110 }
111 eta = copysign(sqrt(eta_squared), delta);
112 if (fabs(eta) * sqrt(a) > 8.0) return 0;
113
114 if (fabs(eta) < 1e-3) {
115 c0 = -1.0 / 3.0 + eta / 12.0 -
116 2.0 * eta * eta / 135.0;
117 } else {
118 c0 = 1.0 / delta - 1.0 / eta;
119 }
120 correction =
121 exp(-0.5 * a * eta_squared) * c0 /
122 (LMMC_SQRT_2PI * sqrt(a));
123 value = 0.5 * erfc(eta * sqrt(0.5 * a)) + correction;
124 if (!isfinite(value) || value < 0.0 || value > 1.0) return 0;
125
126 *upper = value;
127 return 1;
128}

134lmmc_status_t regularized_gamma_lower(
135 lmmc_real_t a, lmmc_real_t x, lmmc_real_t* out) {
136 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
137 if (x <= 0.0) return store_regularized_probability(0.0, out);
138 {
139 lmmc_real_t upper;
140 if (regularized_gamma_large_central_upper(a, x, &upper))
141 return store_regularized_probability(1.0 - upper, out);
142 }
143 if (x < a + 1.0) {
144 lmmc_real_t sum = 1.0 / a;
145 lmmc_real_t term = sum;
146 lmmc_real_t ap = a;
147 int i;
148 for (i = 0; i < SPECIAL_FUNCTION_MAX_ITERATIONS; ++i) {
149 ap += 1.0;
150 term *= x / ap;
151 sum += term;
152 if (!isfinite(term) || !isfinite(sum))
153 return LMMC_STATUS_NUMERICAL_FAILURE;
154 if (fabs(term) < fabs(sum) * SPECIAL_FUNCTION_TOLERANCE)
155 break;
156 }
157 if (i == SPECIAL_FUNCTION_MAX_ITERATIONS)
158 return LMMC_STATUS_CONVERGENCE_FAILED;
159 return store_regularized_probability(
160 sum * exp(-x + a * log(x) - lgamma(a)), out);
161 } else {
162 lmmc_real_t upper;
163 lmmc_status_t status = regularized_gamma_upper_cf(a, x, &upper);
164 if (status != LMMC_STATUS_OK) return status;
165 return store_regularized_probability(1.0 - upper, out);
166 }
167}

173lmmc_status_t regularized_gamma_upper_cf(
174 lmmc_real_t a, lmmc_real_t x, lmmc_real_t* out) {
175 lmmc_real_t f, c, d, delta;
176 int i;
177
178 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
179 f = x - a + 1.0;
180 if (!isfinite(f)) return LMMC_STATUS_NUMERICAL_FAILURE;
181 if (fabs(f) < LENTZ_TINY) f = copysign(LENTZ_TINY, f == 0.0 ? 1.0 : f);
182 c = f;
183 d = 0.0;
184
185 for (i = 1; i <= SPECIAL_FUNCTION_MAX_ITERATIONS; ++i) {
186 lmmc_real_t an = (lmmc_real_t)i * (a - (lmmc_real_t)i);
187 lmmc_real_t bn = x - a + 2.0 * (lmmc_real_t)i + 1.0;
188 if (!isfinite(an) || !isfinite(bn))
189 return LMMC_STATUS_NUMERICAL_FAILURE;
190
191 d = bn + an * d;
192 if (fabs(d) < LENTZ_TINY)
193 d = copysign(LENTZ_TINY, d == 0.0 ? 1.0 : d);
194 d = 1.0 / d;
195 c = bn + an / c;
196 if (fabs(c) < LENTZ_TINY)
197 c = copysign(LENTZ_TINY, c == 0.0 ? 1.0 : c);
198 delta = c * d;
199 f *= delta;
200 if (!isfinite(c) || !isfinite(d) || !isfinite(delta) || !isfinite(f))
201 return LMMC_STATUS_NUMERICAL_FAILURE;
202 if (fabs(delta - 1.0) < SPECIAL_FUNCTION_TOLERANCE)
203 break;
204 }
205 if (i > SPECIAL_FUNCTION_MAX_ITERATIONS)
206 return LMMC_STATUS_CONVERGENCE_FAILED;
207 return store_regularized_probability(
208 exp(-x + a * log(x) - lgamma(a)) / f, out);
209}

235lmmc_status_t regularized_beta_cf(
236 lmmc_real_t x, lmmc_real_t a, lmmc_real_t b, lmmc_real_t* out) {
237 lmmc_real_t qab = a + b;
238 lmmc_real_t qap = a + 1.0;
239 lmmc_real_t qam = a - 1.0;
240 lmmc_real_t c = 1.0;
241 lmmc_real_t d, f;
242 int m;
243
244 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
245 if (!isfinite(qab) || !isfinite(qap) || !isfinite(qam))
246 return LMMC_STATUS_NUMERICAL_FAILURE;
247 d = 1.0 - qab * x / qap;
248 if (fabs(d) < LENTZ_TINY)
249 d = copysign(LENTZ_TINY, d == 0.0 ? 1.0 : d);
250 d = 1.0 / d;
251 f = d;
252
253 for (m = 1; m <= SPECIAL_FUNCTION_MAX_ITERATIONS; ++m) {
254 lmmc_real_t aa, del;
255 int m2 = 2 * m;
256 aa = (lmmc_real_t)m * (b - (lmmc_real_t)m) * x /
257 ((qam + (lmmc_real_t)m2) * (a + (lmmc_real_t)m2));
258 d = 1.0 + aa * d;
259 if (fabs(d) < LENTZ_TINY)
260 d = copysign(LENTZ_TINY, d == 0.0 ? 1.0 : d);
261 c = 1.0 + aa / c;
262 if (fabs(c) < LENTZ_TINY)
263 c = copysign(LENTZ_TINY, c == 0.0 ? 1.0 : c);
264 d = 1.0 / d;
265 f *= c * d;
266
267 aa = -(a + (lmmc_real_t)m) * (qab + (lmmc_real_t)m) * x /
268 ((a + (lmmc_real_t)m2) * (qap + (lmmc_real_t)m2));
269 d = 1.0 + aa * d;
270 if (fabs(d) < LENTZ_TINY)
271 d = copysign(LENTZ_TINY, d == 0.0 ? 1.0 : d);
272 c = 1.0 + aa / c;
273 if (fabs(c) < LENTZ_TINY)
274 c = copysign(LENTZ_TINY, c == 0.0 ? 1.0 : c);
275 d = 1.0 / d;
276 del = c * d;
277 f *= del;
278 if (!isfinite(aa) || !isfinite(c) || !isfinite(d) ||
279 !isfinite(del) || !isfinite(f))
280 return LMMC_STATUS_NUMERICAL_FAILURE;
281 if (fabs(del - 1.0) < SPECIAL_FUNCTION_TOLERANCE)
282 break;
283 }
284 if (m > SPECIAL_FUNCTION_MAX_ITERATIONS)
285 return LMMC_STATUS_CONVERGENCE_FAILED;
286 if (!isfinite(f)) return LMMC_STATUS_NUMERICAL_FAILURE;
287 *out = f;
288 return LMMC_STATUS_OK;
289}

294lmmc_status_t regularized_beta(
295 lmmc_real_t x, lmmc_real_t a, lmmc_real_t b, lmmc_real_t* out) {
296 lmmc_real_t log_bt, bt, fraction, result;
297 lmmc_status_t status;
298 if (!out) return LMMC_STATUS_INVALID_ARGUMENT;
299 if (x <= 0.0) return store_regularized_probability(0.0, out);
300 if (x >= 1.0) return store_regularized_probability(1.0, out);
301 if (x == 0.5 && a == b)
302 return store_regularized_probability(0.5, out);
303
304 log_bt = lgamma(a + b) - lgamma(a) - lgamma(b) +
305 a * log(x) + b * log1p(-x);
306 if (isnan(log_bt) || log_bt > log(DBL_MAX))
307 return LMMC_STATUS_NUMERICAL_FAILURE;
308 bt = exp(log_bt);
309 if (!isfinite(bt)) return LMMC_STATUS_NUMERICAL_FAILURE;
310
311 if (x < (a + 1.0) / (a + b + 2.0)) {
312 status = regularized_beta_cf(x, a, b, &fraction);
313 if (status != LMMC_STATUS_OK) return status;
314 result = bt * fraction / a;
315 } else {
316 status = regularized_beta_cf(1.0 - x, b, a, &fraction);
317 if (status != LMMC_STATUS_OK) return status;
318 result = 1.0 - bt * fraction / b;
319 }
320 return store_regularized_probability(result, out);
321}
