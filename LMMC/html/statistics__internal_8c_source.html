8 if (x <= 0.0) return 0.0;

32lmmc_real_t regularized_gamma_upper_cf(lmmc_real_t a, lmmc_real_t x) {
33 /* Lentz continued fraction for Q(a, x) */
34 lmmc_real_t f, c, d, delta;
35 int i;
36 lmmc_real_t tiny = 1e-30;
37
38
39 /* CF: b0 = 0, a1 = 1, b1 = x - a + 1, a_i = (i-1)*(a - (i-1)), b_i = x - a + 2*i - 1 */
40 /* Modified Lentz: start with b0 = x - a + 1 */
41 {
42 lmmc_real_t b0 = x - a + 1.0;
43 if (fabs(b0) < tiny) b0 = tiny;
44 f = b0;
45 c = b0;
46 d = 0.0;
47 }
48
49 for (i = 1; i <= 200; ++i) {
50 lmmc_real_t an = (lmmc_real_t)i * (a - (lmmc_real_t)i);
51 lmmc_real_t bn = x - a + 2.0 * (lmmc_real_t)i + 1.0;
52
53 d = bn + an * d;
54 if (fabs(d) < tiny) d = tiny;
55 d = 1.0 / d;
56
57 c = bn + an / c;
58 if (fabs(c) < tiny) c = tiny;
59
60 delta = c * d;
61 f *= delta;
62 if (fabs(delta - 1.0) < 1e-15) break;
63 }
64
65 return exp(-x + a * log(x) - lgamma(a)) / f;
66}
