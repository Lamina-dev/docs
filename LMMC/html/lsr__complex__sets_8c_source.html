68lmmc_status_t lmmc_lsr_math_complex_hash(const lmmc_complex_t* z,
69 uint64_t* out)
70{
71 uint64_t real_hash;
72 uint64_t imag_hash;
73 if (!z || !out) return LMMC_STATUS_INVALID_ARGUMENT;
74 if (!lmmc_lsr_complex_is_finite(z)) {
75 return LMMC_STATUS_NUMERICAL_FAILURE;
76 }
77 real_hash = lmmc_lsr_real_hash_bits(z->real);
78 imag_hash = lmmc_lsr_real_hash_bits(z->imag);
79 *out = lmmc_lsr_hash_mix64(real_hash ^
80 (imag_hash + UINT64_C(0x9e3779b97f4a7c15) +
81 (real_hash << 6) + (real_hash >> 2)));
82 return LMMC_STATUS_OK;
83}
