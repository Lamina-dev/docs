30static lmmc_real_t lmmc_quad_tanh_sinh_weight(lmmc_real_t t, lmmc_real_t u)
31{
32 const lmmc_real_t pi_half = LMMC_CONST_PI / 2.0;
33 if (u > 6.0) {
34 return pi_half * cosh(t) * 4.0 * exp(-2.0 * u);
35 }
36 const lmmc_real_t cosh_u = cosh(u);
37 return pi_half * cosh(t) / (cosh_u * cosh_u);

40static lmmc_status_t lmmc_quad_tanh_sinh_pair(
41 lmmc_quad_tanh_sinh_state_t* state, lmmc_real_t u,
42 lmmc_real_t weight, lmmc_real_t* sum)
43{
44 const lmmc_real_t tanh_u = u > 6.0 ? 1.0 - 2.0 * exp(-2.0 * u) : tanh(u);
45 const lmmc_real_t xp = fma(state->half_length, tanh_u, state->center);
46 const lmmc_real_t xn = fma(-state->half_length, tanh_u, state->center);
47 const size_t needed = (size_t)(xp != state->b) + (size_t)(xn != state->a);
48 if (state->max_nodes - state->num_evals < needed) {
49 return LMMC_STATUS_CONVERGENCE_FAILED;
50 }
51 lmmc_real_t fp = 0.0, fn = 0.0;
53 if (xp != state->b) {
54 fp = state->func(xp, state->user_data);
55 ++state->num_evals;
56 }
57 if (xn != state->a) {
58 fn = state->func(xn, state->user_data);
59 ++state->num_evals;
60 }
61 if (!isfinite(fp) || !isfinite(fn)) {
62 return LMMC_STATUS_NUMERICAL_FAILURE;
63 }
64 const lmmc_real_t next = *sum + (fp + fn) * weight;
65 if (!isfinite(next)) return LMMC_STATUS_NUMERICAL_FAILURE;
66 *sum = next;
67 return LMMC_STATUS_OK;
68}

70static lmmc_status_t lmmc_quad_tanh_sinh_level(
71 lmmc_quad_tanh_sinh_state_t* state, lmmc_real_t h, lmmc_real_t* integral)
72{
73 const lmmc_real_t pi_half = LMMC_CONST_PI / 2.0;
74 if (state->num_evals == state->max_nodes) {
75 return LMMC_STATUS_CONVERGENCE_FAILED;
76 }
77 const lmmc_real_t fmid = state->func(state->center, state->user_data);
78 lmmc_real_t sum = fmid * pi_half;
79 ++state->num_evals;
80 if (!isfinite(fmid) || !isfinite(sum)) {
81 return LMMC_STATUS_NUMERICAL_FAILURE;
82 }
83 for (size_t j = 1; ; ++j) {
84 const lmmc_real_t t = (lmmc_real_t)j * h;
85 const lmmc_real_t u = pi_half * sinh(t);
86 lmmc_real_t weight;
87 if (u > 20.0) break;
88 weight = lmmc_quad_tanh_sinh_weight(t, u);
89 if (!isfinite(weight)) {
90 return LMMC_STATUS_NUMERICAL_FAILURE;
91 }
92 if (weight < 1e-50) break;
93 const lmmc_status_t status = lmmc_quad_tanh_sinh_pair(state, u, weight, &sum);
94 if (status != LMMC_STATUS_OK) return status;
95 }
96 const lmmc_real_t completed = state->half_length * h * sum;
97 if (!isfinite(completed)) return LMMC_STATUS_NUMERICAL_FAILURE;
98 *integral = completed;
99 return LMMC_STATUS_OK;
100}

102static lmmc_status_t lmmc_quad_tanh_sinh_integrate(
103 lmmc_quad_tanh_sinh_state_t* state, lmmc_real_t abs_tol,
104 lmmc_quad_result_t* out)
105{
106 int have_complete = 0;
107 lmmc_real_t last_complete = 0.0;
108 lmmc_real_t last_error = INFINITY;
109 lmmc_real_t h = 0.03125;
110 for (size_t level = 0; level < 10; ++level) {
111 lmmc_real_t integral;
112 const lmmc_status_t status = lmmc_quad_tanh_sinh_level(state, h, &integral);
113 out->num_evals = state->num_evals;
114 if (status != LMMC_STATUS_OK) {
115 out->value = last_complete;
116 out->error = INFINITY;
117 return status;
118 }
119 if (have_complete) {
120 last_error = lmmc_abs(integral - last_complete);
121 if (!isfinite(last_error)) {
122 out->error = INFINITY;
123 return LMMC_STATUS_NUMERICAL_FAILURE;
124 }
125 }
126 last_complete = integral;
127 out->value = last_complete;
128 out->error = last_error;
129 if (have_complete && last_error <= abs_tol) return LMMC_STATUS_OK;
130 have_complete = 1;
131 h /= 2.0;
132 }
133 return LMMC_STATUS_CONVERGENCE_FAILED;
134}
