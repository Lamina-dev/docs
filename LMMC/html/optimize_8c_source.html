48lmmc_real_t lmmc_optimize_norm(const lmmc_vec_t* v) {
49 lmmc_scaled_sumsq_t acc;
50 size_t i;
51 lmmc_scaled_sumsq_init(&acc);
52 for (i = 0; i < v->size; ++i) {
53 const lmmc_real_t value_abs = fabs(v->data[i]);
54 if (!isfinite(value_abs)) {
55 return value_abs;
56 }
57 lmmc_scaled_sumsq_add(&acc, value_abs);
58 }
59 return lmmc_scaled_sumsq_norm(&acc);
60}

90int lmmc_optimize_stop(lmmc_real_t norm, lmmc_real_t* initial,
91 size_t iteration, const lmmc_optimize_config_t* cfg,
92 lmmc_optimize_result_t* out)
93{
94 out->final_residual = norm;
95 if (!isfinite(norm)) {
96 out->failure_reason = LMMC_OPT_FAILURE_NUMERICAL_ISSUE;
97 return 1;
98 }
99 if (iteration == 0) {
100 *initial = norm;
101 }
102 if (lmmc_optimize_has_converged(norm, *initial, cfg)) {
103 out->converged = 1;
104 return 1;
105 }
106 if (iteration == cfg->max_iter) {
107 out->failure_reason = LMMC_OPT_FAILURE_MAX_ITER;
108 return 1;
109 }
110 return 0;
111}

113static int optimize_workspace_count(
114 size_t n, size_t vectors, size_t matrices, size_t extra, size_t* count)
115{
116 size_t vector_count, matrix_count = 0;
117 if (!lmmc_safe_mul_size(n, vectors, &vector_count)) {
118 return 0;
119 }
120 if (matrices != 0) {
121 if (!lmmc_safe_mul_size(n, n, &matrix_count) ||
122 !lmmc_safe_mul_size(matrix_count, matrices, &matrix_count)) {
123 return 0;
124 }
125 }
126 return lmmc_safe_add_size(vector_count, matrix_count, count) &&
127 lmmc_safe_add_size(*count, extra, count);
128}

130lmmc_status_t lmmc_opt_workspace_create(
131 size_t n, size_t vectors, size_t matrices, size_t extra,
132 int need_pivots, lmmc_opt_workspace_t* work)
133{
134 size_t count, scalar_bytes, pivot_bytes = 0, bytes;
135 const size_t alignment = _Alignof(lmmc_real_t);
136 if (!optimize_workspace_count(n, vectors, matrices, extra, &count) ||
137 !lmmc_safe_mul_size(count, sizeof(lmmc_real_t), &scalar_bytes)) {
138 return LMMC_STATUS_INVALID_ARGUMENT;
139 }
140 if (need_pivots) {
141 if (!lmmc_safe_mul_size(n, sizeof(size_t), &pivot_bytes) ||
142 !lmmc_safe_add_size(pivot_bytes, alignment - 1, &pivot_bytes)) {
143 return LMMC_STATUS_INVALID_ARGUMENT;
144 }
145 pivot_bytes -= pivot_bytes % alignment;
146 }
147 if (!lmmc_safe_add_size(pivot_bytes, scalar_bytes, &bytes)) {
148 return LMMC_STATUS_INVALID_ARGUMENT;
149 }
150 work->allocation = lmmc_memory_alloc(bytes);
151 if (work->allocation == NULL) {
152 return LMMC_STATUS_ALLOCATION_FAILED;
153 }
154 memset(work->allocation, 0, bytes);
155 work->pivots = need_pivots ? (size_t*)work->allocation : NULL;
156 work->next = (lmmc_real_t*)((unsigned char*)work->allocation + pivot_bytes);
157 return LMMC_STATUS_OK;
158}

174static lmmc_status_t optimize_perturb(
175 const lmmc_vec_t* x, size_t j, lmmc_vec_t* x_pert, lmmc_real_t* step)
176{
177 lmmc_real_t h = sqrt(LMMC_REAL_EPSILON) * fmax(fabs(x->data[j]), 1.0);
178 lmmc_real_t perturbed_x = x->data[j] + h;
179 memcpy(x_pert->data, x->data, x->size * sizeof(lmmc_real_t));
180 if (!isfinite(perturbed_x) || perturbed_x == x->data[j]) {
181 perturbed_x = x->data[j] - h;
182 }
183 if (!isfinite(perturbed_x) || perturbed_x == x->data[j]) {
184 perturbed_x = nextafter(x->data[j], 0.0);
185 }
186 h = perturbed_x - x->data[j];
187 if (!isfinite(perturbed_x) || !isfinite(h) || h == 0.0) {
188 return LMMC_STATUS_NUMERICAL_FAILURE;
189 }
190 x_pert->data[j] = perturbed_x;
191 *step = h;
192 return LMMC_STATUS_OK;
193}

195static lmmc_status_t optimize_difference_column(
196 const lmmc_vec_t* Fx, const lmmc_vec_t* F_pert,
197 lmmc_real_t h, size_t j, lmmc_mat_t* matrix)
198{
199 size_t i;
200 for (i = 0; i < Fx->size; ++i) {
201 lmmc_real_t derivative;
202 if (!isfinite(F_pert->data[i]) || !isfinite(Fx->data[i])) {
203 return LMMC_STATUS_NUMERICAL_FAILURE;
204 }
205 derivative = (F_pert->data[i] - Fx->data[i]) / h;
206 if (!isfinite(derivative)) {
207 return LMMC_STATUS_NUMERICAL_FAILURE;
208 }
209 matrix->data[i * matrix->stride + j] = derivative;
210 }
211 return LMMC_STATUS_OK;
212}

214lmmc_status_t lmmc_optimize_jacobian(
215 lmmc_opt_func_t F, lmmc_opt_jac_t J, void* user_data,
216 const lmmc_vec_t* x, const lmmc_vec_t* Fx, lmmc_mat_t* matrix,
217 lmmc_vec_t* x_pert, lmmc_vec_t* F_pert)
218{
219 size_t j;
220 if (J != NULL) {
221 return J(x, matrix, user_data);
222 }
223 for (j = 0; j < x->size; ++j) {
224 lmmc_real_t h;
225 lmmc_status_t status = optimize_perturb(x, j, x_pert, &h);
226 if (status != LMMC_STATUS_OK) {
227 return status;
228 }
229 status = F(x_pert, F_pert, user_data);
230 if (status != LMMC_STATUS_OK) {
231 return status;
232 }
233 status = optimize_difference_column(Fx, F_pert, h, j, matrix);
234 if (status != LMMC_STATUS_OK) {
235 return status;
236 }
237 }
238 return LMMC_STATUS_OK;
239}

241int lmmc_optimize_armijo(
242 lmmc_opt_obj_t obj, void* user_data, const lmmc_vec_t* x,
243 const lmmc_vec_t* direction, lmmc_vec_t* trial,
244 lmmc_real_t* value, lmmc_real_t derivative, int attempts, int subtract)
245{
246 lmmc_real_t step = 1.0;
247 const lmmc_real_t c1 = 1e-4;
248 int attempt;
249 size_t i;
250 for (attempt = 0; attempt < attempts; ++attempt) {
251 for (i = 0; i < x->size; ++i) {
252 if (subtract) {
253 trial->data[i] = x->data[i] - step * direction->data[i];
254 } else {
255 trial->data[i] = x->data[i] + step * direction->data[i];
256 }
257 }
258 const lmmc_real_t next = obj(trial, user_data);
259 if (next <= *value + c1 * step * derivative) {
260 *value = next;
261 return 1;
262 }
263 step *= 0.5;
264 }
265 return 0;
266}
