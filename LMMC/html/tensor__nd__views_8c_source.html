8static lmmc_status_t lmmc_tensor_nd_permutation_shape(
9 const lmmc_tensor_nd_t* in, const size_t* perm, const lmmc_tensor_nd_t* out,
10 size_t* new_dims) {
11 int seen[LMMC_TENSOR_MAX_NDIM] = {0};
12 if (!lmmc_tensor_nd_descriptor_is_valid(in) || perm == NULL || out == NULL) {
13 return LMMC_STATUS_INVALID_ARGUMENT;
14 }
15 if (in == out && in->owns_data) {
16 return LMMC_STATUS_INVALID_ARGUMENT;
17 }
18 for (size_t i = 0; i < in->ndim; ++i) {
19 if (perm[i] >= in->ndim || seen[perm[i]]) {
20 return LMMC_STATUS_INVALID_ARGUMENT;
21 }
22 seen[perm[i]] = 1;
23 new_dims[i] = in->dims[perm[i]];
24 }
25 return LMMC_STATUS_OK;
26}

65static int lmmc_tensor_nd_is_contiguous(const lmmc_tensor_nd_t* t) {
66 size_t i;
67 size_t expected_stride;
68 if (t == NULL || t->ndim == 0) return 0;
69
70 expected_stride = 1;
71 for (i = t->ndim; i > 0; --i) {
72 if (t->strides[i - 1] != expected_stride) return 0;
73 if (t->dims[i - 1] == 0 ||
74 !lmmc_safe_mul_size(expected_stride, t->dims[i - 1], &expected_stride)) return 0;
75 }
76 return 1;
77}

97lmmc_status_t lmmc_tensor_nd_reshape_view(const lmmc_tensor_nd_t* src,
98 size_t new_ndim, const size_t* new_dims, lmmc_tensor_nd_t* out_view) {
99 size_t old_total, new_total;
100 lmmc_tensor_nd_t view = {0};
101 if (lmmc_tensor_nd_validate_reshape(src, new_ndim, new_dims, out_view) != LMMC_STATUS_OK) {
102 return LMMC_STATUS_INVALID_ARGUMENT;
103 }
104 if (lmmc_tensor_nd_layout(new_ndim, new_dims, view.strides, &new_total) !=
105 LMMC_STATUS_OK ||
106 lmmc_tensor_nd_layout(src->ndim, src->dims, NULL, &old_total) !=
107 LMMC_STATUS_OK) {
108 return LMMC_STATUS_INVALID_ARGUMENT;
109 }
110 if (old_total != new_total) {
111 return LMMC_STATUS_INVALID_ARGUMENT;
112 }
113 if (!lmmc_tensor_nd_is_contiguous(src)) {
114 return LMMC_STATUS_INVALID_ARGUMENT;
115 }
116
118 for (size_t i = 0; i < new_ndim; ++i) {
119 view.dims[i] = new_dims[i];
120 }
121 view.ndim = new_ndim;
122 view.data = src->data;
123 view.owns_data = 0;
124 *out_view = view;
125 return LMMC_STATUS_OK;
126}
