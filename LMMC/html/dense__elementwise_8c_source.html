253lmmc_status_t lmmc_vec_cmp_eq(const lmmc_vec_t* a, const lmmc_vec_t* b, lmmc_real_t tol, int* out) {
254 size_t i;
255 lmmc_real_t diff;
256 lmmc_real_t abs_diff;
257
258 if (a == NULL || b == NULL || out == NULL ||
259 a->data == NULL || b->data == NULL) {
260 return LMMC_STATUS_INVALID_ARGUMENT;
261 }
262 if (a->size != b->size) {
263 return LMMC_STATUS_DIMENSION_MISMATCH;
264 }
265
266 LMMC_REAL_INIT(&diff);
267 LMMC_REAL_INIT(&abs_diff);
268
269 for (i = 0; i < a->size; ++i) {
270 LMMC_REAL_SUB(&diff, &a->data[i], &b->data[i]);
271 LMMC_REAL_ABS(&abs_diff, &diff);
272 out[i] = (LMMC_REAL_CMP(&abs_diff, &tol) <= 0) ? 1 : 0;
273 }
274
275 LMMC_REAL_CLEAR(&diff);
276 LMMC_REAL_CLEAR(&abs_diff);
277 return LMMC_STATUS_OK;
278}

280lmmc_status_t lmmc_vec_cross(const lmmc_vec_t* a, const lmmc_vec_t* b, lmmc_vec_t* c)
281{
282 lmmc_real_t t0, t1, t2;
283 lmmc_real_t mul1, mul2;
284
285 if (a == NULL || b == NULL || c == NULL ||
286 a->data == NULL || b->data == NULL || c->data == NULL) {
287 return LMMC_STATUS_INVALID_ARGUMENT;
288 }
289 if (a->size != 3 || b->size != 3 || c->size != 3) {
290 return LMMC_STATUS_DIMENSION_MISMATCH;
291 }
292
293 LMMC_REAL_INIT(&t0);
294 LMMC_REAL_INIT(&t1);
295 LMMC_REAL_INIT(&t2);
296 LMMC_REAL_INIT(&mul1);
297 LMMC_REAL_INIT(&mul2);
298
299 /* c[0] = a[1]*b[2] - a[2]*b[1] */
300 LMMC_REAL_MUL(&mul1, &a->data[1], &b->data[2]);
301 LMMC_REAL_MUL(&mul2, &a->data[2], &b->data[1]);
302 LMMC_REAL_SUB(&t0, &mul1, &mul2);
303
304 /* c[1] = a[2]*b[0] - a[0]*b[2] */
305 LMMC_REAL_MUL(&mul1, &a->data[2], &b->data[0]);
306 LMMC_REAL_MUL(&mul2, &a->data[0], &b->data[2]);
307 LMMC_REAL_SUB(&t1, &mul1, &mul2);
308
309 /* c[2] = a[0]*b[1] - a[1]*b[0] */
310 LMMC_REAL_MUL(&mul1, &a->data[0], &b->data[1]);
311 LMMC_REAL_MUL(&mul2, &a->data[1], &b->data[0]);
312 LMMC_REAL_SUB(&t2, &mul1, &mul2);
313
314 /* Write results (safe for aliasing since we used temporaries) */
315 LMMC_REAL_SET(&c->data[0], &t0);
316 LMMC_REAL_SET(&c->data[1], &t1);
317 LMMC_REAL_SET(&c->data[2], &t2);
318
319 LMMC_REAL_CLEAR(&t0);
320 LMMC_REAL_CLEAR(&t1);
321 LMMC_REAL_CLEAR(&t2);
322 LMMC_REAL_CLEAR(&mul1);
323 LMMC_REAL_CLEAR(&mul2);
324 return LMMC_STATUS_OK;
325}
