26static lmmc_real_t interp_pchip_endpoint(lmmc_real_t h1, lmmc_real_t h2,
27 lmmc_real_t slope1, lmmc_real_t slope2)
28{
29 lmmc_real_t derivative = ((2.0 * h1 + h2) * slope1 - h1 * slope2) / (h1 + h2);
30 if (derivative * slope1 <= 0.0) {
31 return 0.0;
32 }
33 if (slope1 * slope2 <= 0.0 && fabs(derivative) > 3.0 * fabs(slope1)) {
34 return 3.0 * slope1;
35 }
36 return derivative;
37}

39static void interp_pchip_derivatives(lmmc_interp_pchip_t* p,
40 const lmmc_real_t* delta)
41{
42 const lmmc_real_t* xs = p->xs;
43 size_t i, n = p->n;
44 if (n == 2) {
45 p->d[0] = p->d[1] = delta[0];
46 return;
47 }
48 for (i = 1; i < n - 1; ++i) {
49 if (delta[i - 1] * delta[i] <= 0.0) {
50 p->d[i] = 0.0;
51 } else {
52 lmmc_real_t h1 = xs[i] - xs[i - 1];
53 lmmc_real_t h2 = xs[i + 1] - xs[i];
54 lmmc_real_t w1 = 2.0 * h2 + h1;
55 lmmc_real_t w2 = h2 + 2.0 * h1;
56 p->d[i] = (w1 + w2) / (w1 / delta[i - 1] + w2 / delta[i]);
57 }
58 }
59 p->d[0] = interp_pchip_endpoint(xs[1] - xs[0], xs[2] - xs[1], delta[0], delta[1]);
60 p->d[n - 1] = interp_pchip_endpoint(xs[n - 1] - xs[n - 2],
61 xs[n - 2] - xs[n - 3], delta[n - 2], delta[n - 3]);
62}

115lmmc_status_t lmmc_interp_pchip_eval(
116 const lmmc_interp_pchip_t* p, lmmc_real_t x, lmmc_real_t* out_y)
117{
118 size_t seg;
119 lmmc_real_t h, t, a, b, result;
120 if (!p || !out_y || !isfinite(x)) {
121 return LMMC_STATUS_INVALID_ARGUMENT;
122 }
123 if (x < p->xs[0] || x > p->xs[p->n - 1]) {
124 return LMMC_STATUS_OUT_OF_RANGE;
125 }
126 if (x == p->xs[p->n - 1]) {
127 *out_y = p->ys[p->n - 1];
128 return LMMC_STATUS_OK;
129 }
130
131 seg = interp_find_interval(p->xs, p->n, x);
132 h = p->xs[seg + 1] - p->xs[seg];
133 t = (x - p->xs[seg]) / h;
134
135 a = 1.0 - t;
136 b = t;
137 result = a * a * (1.0 + 2.0 * t) * p->ys[seg]
138 + b * b * (3.0 - 2.0 * t) * p->ys[seg + 1]
139 + t * a * a * h * p->d[seg]
140 - b * b * a * h * p->d[seg + 1];
141 if (!isfinite(result)) {
142 return LMMC_STATUS_NUMERICAL_FAILURE;
143 }
144 *out_y = result;
145 return LMMC_STATUS_OK;
146}

166static void interp_akima_derivatives(lmmc_interp_akima_t* a, lmmc_real_t* m)
167{
168 size_t i, n = a->n, nm1 = n - 1;
169 const lmmc_real_t* xs = a->xs;
170 const lmmc_real_t* ys = a->ys;
172 for (i = 0; i < nm1; i++) {
173 m[i + 2] = (ys[i + 1] - ys[i]) / (xs[i + 1] - xs[i]);
174 }
175
181 m[1] = 2.0 * m[2] - m[3];
182 m[0] = 2.0 * m[1] - m[2];
183 m[nm1 + 2] = 2.0 * m[nm1 + 1] - m[nm1];
184 m[nm1 + 3] = 2.0 * m[nm1 + 2] - m[nm1 + 1];
185
191 for (i = 0; i < n; i++) {
192 lmmc_real_t w1 = fabs(m[i + 3] - m[i + 2]);
193 lmmc_real_t w2 = fabs(m[i + 1] - m[i]);
194 if (w1 + w2 < 1e-30) {
196 a->d[i] = 0.5 * (m[i + 1] + m[i + 2]);
197 } else {
198 a->d[i] = (w1 * m[i + 1] + w2 * m[i + 2]) / (w1 + w2);
199 }
200 }
201}
