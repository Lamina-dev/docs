14lmmc_status_t lmmc_mat_gemm(lmmc_real_t alpha, const lmmc_mat_t* A, int transA,
15 const lmmc_mat_t* B, int transB, lmmc_real_t beta, lmmc_mat_t* C) {
16 if (!lmmc_mat_descriptor_is_valid(A) ||
17 !lmmc_mat_descriptor_is_valid(B) ||
18 !lmmc_mat_descriptor_is_valid(C)) {
19 return LMMC_STATUS_INVALID_ARGUMENT;
20 }
21
22 /* Determine effective dimensions of op(A) and op(B) */
23 size_t M = transA ? A->cols : A->rows;
24 size_t K_A = transA ? A->rows : A->cols;
25 size_t K_B = transB ? B->cols : B->rows;
26 size_t N = transB ? B->rows : B->cols;
27
28 /* Dimension validation */
29 if (K_A != K_B) {
30 return LMMC_STATUS_DIMENSION_MISMATCH;
31 }
32 if (C->rows != M || C->cols != N) {
33 return LMMC_STATUS_DIMENSION_MISMATCH;
34 }
35 {
36 lmmc_storage_envelope_t a_envelope;
37 lmmc_storage_envelope_t b_envelope;
38 lmmc_storage_envelope_t c_envelope;
39 if (!lmmc_storage_envelope_checked(
40 A->data, A->rows, A->cols, A->stride,
41 sizeof(lmmc_real_t), &a_envelope) ||
42 !lmmc_storage_envelope_checked(
43 B->data, B->rows, B->cols, B->stride,
44 sizeof(lmmc_real_t), &b_envelope) ||
45 !lmmc_storage_envelope_checked(
46 C->data, C->rows, C->cols, C->stride,
47 sizeof(lmmc_real_t), &c_envelope)) {
48 return LMMC_STATUS_INVALID_ARGUMENT;
49 }
50 if (lmmc_storage_envelopes_overlap(&c_envelope, &a_envelope) ||
51 lmmc_storage_envelopes_overlap(&c_envelope, &b_envelope)) {
52 return LMMC_STATUS_INVALID_ARGUMENT;
53 }
54 }
55
56
57 size_t K = K_A;
58
59 const lmmc_real_t* restrict a_data = A->data;
60 const lmmc_real_t* restrict b_data = B->data;
61 lmmc_real_t* restrict c_data = C->data;
62
63 size_t a_stride = A->stride;
64 size_t b_stride = B->stride;
65 size_t c_stride = C->stride;
66
67
68 /* Scale C by beta */
69 if (beta == 0.0) {
70 for (size_t i = 0; i < M; ++i) {
71 for (size_t j = 0; j < N; ++j) {
72 LMMC_REAL_SET_D(&c_data[i * c_stride + j], 0.0);
73 }
74 }
75 } else if (beta != 1.0) {
76 lmmc_real_t tmp_mul; LMMC_REAL_INIT(&tmp_mul);
77 for (size_t i = 0; i < M; ++i) {
78 for (size_t j = 0; j < N; ++j) {
79 LMMC_REAL_MUL(&tmp_mul, &beta, &c_data[i * c_stride + j]);
80 LMMC_REAL_SET(&c_data[i * c_stride + j], &tmp_mul);
81 }
82 }
83 LMMC_REAL_CLEAR(&tmp_mul);
84 }
85
86 /* If alpha is zero, we're done */
87 if (alpha == 0.0) {
88 return LMMC_STATUS_OK;
89 }
90
91 /* Native triple-loop: C += alpha * op(A) * op(B) */
92 lmmc_real_t tmp_mul; LMMC_REAL_INIT(&tmp_mul);
93 lmmc_real_t tmp_sum; LMMC_REAL_INIT(&tmp_sum);
94 lmmc_real_t alpha_a_ik; LMMC_REAL_INIT(&alpha_a_ik);
95
96 size_t bs = 64;
97 for (size_t ii = 0; ii < M; ii += bs) {
98 size_t i_end = (ii + bs > M) ? M : ii + bs;
99 for (size_t kk = 0; kk < K; kk += bs) {
100 size_t k_end = (kk + bs > K) ? K : kk + bs;
101 for (size_t jj = 0; jj < N; jj += bs) {
102 size_t j_end = (jj + bs > N) ? N : jj + bs;
103 for (size_t i = ii; i < i_end; ++i) {
104 for (size_t k = kk; k < k_end; ++k) {
105 /* Get op(A)[i,k] */
106 lmmc_real_t a_ik;
107 if (transA) {
108 a_ik = a_data[k * a_stride + i];
109 } else {
110 a_ik = a_data[i * a_stride + k];
111 }
112 /* alpha * op(A)[i,k] */
113 LMMC_REAL_MUL(&alpha_a_ik, &alpha, &a_ik);
114
115 for (size_t j = jj; j < j_end; ++j) {
116 /* Get op(B)[k,j] */
117 lmmc_real_t b_kj;
118 if (transB) {
119 b_kj = b_data[j * b_stride + k];
120 } else {
121 b_kj = b_data[k * b_stride + j];
122 }
123 LMMC_REAL_MUL(&tmp_mul, &alpha_a_ik, &b_kj);
124 LMMC_REAL_ADD(&tmp_sum, &c_data[i * c_stride + j], &tmp_mul);
125 LMMC_REAL_SET(&c_data[i * c_stride + j], &tmp_sum);
126 }
127 }
128 }
129 }
130 }
131 }
132
133 LMMC_REAL_CLEAR(&tmp_mul);
134 LMMC_REAL_CLEAR(&tmp_sum);
135 LMMC_REAL_CLEAR(&alpha_a_ik);
136 return LMMC_STATUS_OK;
137}

143lmmc_status_t lmmc_vec_dot(const lmmc_vec_t* a, const lmmc_vec_t* b, lmmc_real_t* out_dot) {
144 size_t i = 0;
145 if (!lmmc_vec_descriptor_is_valid(a) ||
146 !lmmc_vec_descriptor_is_valid(b) || out_dot == NULL) {
147 return LMMC_STATUS_INVALID_ARGUMENT;
148 }
149 if (a->size != b->size) {
150 return LMMC_STATUS_DIMENSION_MISMATCH;
151 }
152
153 lmmc_real_t sum; LMMC_REAL_INIT(&sum);
154 lmmc_real_t tmp_mul; LMMC_REAL_INIT(&tmp_mul);
155 lmmc_real_t tmp_sum; LMMC_REAL_INIT(&tmp_sum);
156 LMMC_REAL_SET_D(&sum, 0.0);
157
158 for (i = 0; i < a->size; ++i) {
159 LMMC_REAL_MUL(&tmp_mul, &a->data[i], &b->data[i]);
160 LMMC_REAL_ADD(&tmp_sum, &sum, &tmp_mul);
161 LMMC_REAL_SET(&sum, &tmp_sum);
162 }
163 LMMC_REAL_SET(out_dot, &sum);
164
165 LMMC_REAL_CLEAR(&sum);
166 LMMC_REAL_CLEAR(&tmp_mul);
167 LMMC_REAL_CLEAR(&tmp_sum);
168 return LMMC_STATUS_OK;

171lmmc_status_t lmmc_mat_gemv(lmmc_real_t alpha, const lmmc_mat_t* A, int transA,
172 const lmmc_vec_t* x, lmmc_real_t beta, lmmc_vec_t* y) {
173 if (!lmmc_mat_descriptor_is_valid(A) ||
174 !lmmc_vec_descriptor_is_valid(x) ||
175 !lmmc_vec_descriptor_is_valid(y)) {
176 return LMMC_STATUS_INVALID_ARGUMENT;
177 }
178
179 /* Determine effective dimensions of op(A) */
180 size_t M = transA ? A->cols : A->rows;
181 size_t N = transA ? A->rows : A->cols;
182
183 /* Dimension validation */
184 if (N != x->size || M != y->size) {
185 return LMMC_STATUS_DIMENSION_MISMATCH;
186 }
187
188 {
189 lmmc_storage_envelope_t a_envelope;
190 lmmc_storage_envelope_t x_envelope;
191 lmmc_storage_envelope_t y_envelope;
192 if (!lmmc_storage_envelope_checked(A->data, A->rows, A->cols,
193 A->stride, sizeof(lmmc_real_t),
194 &a_envelope) ||
195 !lmmc_storage_envelope_checked(x->data, 1, x->size, x->size,
196 sizeof(lmmc_real_t), &x_envelope) ||
197 !lmmc_storage_envelope_checked(y->data, 1, y->size, y->size,
198 sizeof(lmmc_real_t), &y_envelope)) {
199 return LMMC_STATUS_INVALID_ARGUMENT;
200 }
201 if (lmmc_storage_envelopes_overlap(&y_envelope, &a_envelope) ||
202 lmmc_storage_envelopes_overlap(&y_envelope, &x_envelope)) {
203 return LMMC_STATUS_INVALID_ARGUMENT;
204 }
205 }
206
207 const lmmc_real_t* restrict a_data = A->data;
208 size_t a_stride = A->stride;
209
210
211 /* Scale y by beta */
212 if (beta == 0.0) {
213 for (size_t i = 0; i < M; ++i) {
214 LMMC_REAL_SET_D(&y->data[i], 0.0);
215 }
216 } else if (beta != 1.0) {
217 lmmc_real_t tmp_mul; LMMC_REAL_INIT(&tmp_mul);
218 for (size_t i = 0; i < M; ++i) {
219 LMMC_REAL_MUL(&tmp_mul, &beta, &y->data[i]);
220 LMMC_REAL_SET(&y->data[i], &tmp_mul);
221 }
222 LMMC_REAL_CLEAR(&tmp_mul);
223 }
224
225 /* If alpha is zero, we're done */
226 if (alpha == 0.0) {
227 return LMMC_STATUS_OK;
228 }
229
230 /* Native loop: y += alpha * op(A) * x */
231 lmmc_real_t sum; LMMC_REAL_INIT(&sum);
232 lmmc_real_t tmp_mul; LMMC_REAL_INIT(&tmp_mul);
233 lmmc_real_t tmp_sum; LMMC_REAL_INIT(&tmp_sum);
234 lmmc_real_t alpha_sum; LMMC_REAL_INIT(&alpha_sum);
235
236 for (size_t i = 0; i < M; ++i) {
237 LMMC_REAL_SET_D(&sum, 0.0);
238 for (size_t j = 0; j < N; ++j) {
239 lmmc_real_t a_ij;
240 if (transA) {
241 a_ij = a_data[j * a_stride + i];
242 } else {
243 a_ij = a_data[i * a_stride + j];
244 }
245 LMMC_REAL_MUL(&tmp_mul, &a_ij, &x->data[j]);
246 LMMC_REAL_ADD(&tmp_sum, &sum, &tmp_mul);
247 LMMC_REAL_SET(&sum, &tmp_sum);
248 }
249 /* y[i] += alpha * sum */
250 LMMC_REAL_MUL(&alpha_sum, &alpha, &sum);
251 LMMC_REAL_ADD(&tmp_sum, &y->data[i], &alpha_sum);
252 LMMC_REAL_SET(&y->data[i], &tmp_sum);
253 }
254
255 LMMC_REAL_CLEAR(&sum);
256 LMMC_REAL_CLEAR(&tmp_mul);
257 LMMC_REAL_CLEAR(&tmp_sum);
258 LMMC_REAL_CLEAR(&alpha_sum);
259 return LMMC_STATUS_OK;
260}
