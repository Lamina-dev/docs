14lmmc_status_t lmmc_mat_gemm(lmmc_real_t alpha, const lmmc_mat_t* A, int transA,
15 const lmmc_mat_t* B, int transB, lmmc_real_t beta, lmmc_mat_t* C) {
16 if (A == NULL || B == NULL || C == NULL ||
17 A->data == NULL || B->data == NULL || C->data == NULL) {
18 return LMMC_STATUS_INVALID_ARGUMENT;
19 }
20
21 /* Determine effective dimensions of op(A) and op(B) */
22 size_t M = transA ? A->cols : A->rows;
23 size_t K_A = transA ? A->rows : A->cols;
24 size_t K_B = transB ? B->cols : B->rows;
25 size_t N = transB ? B->rows : B->cols;
26
27 /* Dimension validation */
28 if (K_A != K_B) {
29 return LMMC_STATUS_DIMENSION_MISMATCH;
30 }
31 if (C->rows != M || C->cols != N) {
32 return LMMC_STATUS_DIMENSION_MISMATCH;
33 }
34
35 size_t K = K_A;
36
37 const lmmc_real_t* restrict a_data = A->data;
38 const lmmc_real_t* restrict b_data = B->data;
39 lmmc_real_t* restrict c_data = C->data;
40
41 size_t a_stride = A->stride;
42 size_t b_stride = B->stride;
43 size_t c_stride = C->stride;
44
45
46 /* Scale C by beta */
47 if (beta == 0.0) {
48 for (size_t i = 0; i < M; ++i) {
49 for (size_t j = 0; j < N; ++j) {
50 LMMC_REAL_SET_D(&c_data[i * c_stride + j], 0.0);
51 }
52 }
53 } else if (beta != 1.0) {
54 lmmc_real_t tmp_mul; LMMC_REAL_INIT(&tmp_mul);
55 for (size_t i = 0; i < M; ++i) {
56 for (size_t j = 0; j < N; ++j) {
57 LMMC_REAL_MUL(&tmp_mul, &beta, &c_data[i * c_stride + j]);
58 LMMC_REAL_SET(&c_data[i * c_stride + j], &tmp_mul);
59 }
60 }
61 LMMC_REAL_CLEAR(&tmp_mul);
62 }
63
64 /* If alpha is zero, we're done */
65 if (alpha == 0.0) {
66 return LMMC_STATUS_OK;
67 }
68
69 /* Native triple-loop: C += alpha * op(A) * op(B) */
70 lmmc_real_t tmp_mul; LMMC_REAL_INIT(&tmp_mul);
71 lmmc_real_t tmp_sum; LMMC_REAL_INIT(&tmp_sum);
72 lmmc_real_t alpha_a_ik; LMMC_REAL_INIT(&alpha_a_ik);
73
74 size_t bs = 64;
75 for (size_t ii = 0; ii < M; ii += bs) {
76 size_t i_end = (ii + bs > M) ? M : ii + bs;
77 for (size_t kk = 0; kk < K; kk += bs) {
78 size_t k_end = (kk + bs > K) ? K : kk + bs;
79 for (size_t jj = 0; jj < N; jj += bs) {
80 size_t j_end = (jj + bs > N) ? N : jj + bs;
81 for (size_t i = ii; i < i_end; ++i) {
82 for (size_t k = kk; k < k_end; ++k) {
83 /* Get op(A)[i,k] */
84 lmmc_real_t a_ik;
85 if (transA) {
86 a_ik = a_data[k * a_stride + i];
87 } else {
88 a_ik = a_data[i * a_stride + k];
89 }
90 /* alpha * op(A)[i,k] */
91 LMMC_REAL_MUL(&alpha_a_ik, &alpha, &a_ik);
92
93 for (size_t j = jj; j < j_end; ++j) {
94 /* Get op(B)[k,j] */
95 lmmc_real_t b_kj;
96 if (transB) {
97 b_kj = b_data[j * b_stride + k];
98 } else {
99 b_kj = b_data[k * b_stride + j];
100 }
101 LMMC_REAL_MUL(&tmp_mul, &alpha_a_ik, &b_kj);
102 LMMC_REAL_ADD(&tmp_sum, &c_data[i * c_stride + j], &tmp_mul);
103 LMMC_REAL_SET(&c_data[i * c_stride + j], &tmp_sum);
104 }
105 }
106 }
107 }
108 }
109 }
110
111 LMMC_REAL_CLEAR(&tmp_mul);
112 LMMC_REAL_CLEAR(&tmp_sum);
113 LMMC_REAL_CLEAR(&alpha_a_ik);
114 return LMMC_STATUS_OK;
115}

121lmmc_status_t lmmc_vec_dot(const lmmc_vec_t* a, const lmmc_vec_t* b, lmmc_real_t* out_dot) {
122 size_t i = 0;
123 if (a == NULL || b == NULL || out_dot == NULL || a->data == NULL || b->data == NULL) {
124 return LMMC_STATUS_INVALID_ARGUMENT;
125 }
126 if (a->size != b->size) {
127 return LMMC_STATUS_DIMENSION_MISMATCH;
128 }
129
130 lmmc_real_t sum; LMMC_REAL_INIT(&sum);
131 lmmc_real_t tmp_mul; LMMC_REAL_INIT(&tmp_mul);
132 lmmc_real_t tmp_sum; LMMC_REAL_INIT(&tmp_sum);
133 LMMC_REAL_SET_D(&sum, 0.0);
134
135 for (i = 0; i < a->size; ++i) {
136 LMMC_REAL_MUL(&tmp_mul, &a->data[i], &b->data[i]);
137 LMMC_REAL_ADD(&tmp_sum, &sum, &tmp_mul);
138 LMMC_REAL_SET(&sum, &tmp_sum);
139 }
140 LMMC_REAL_SET(out_dot, &sum);
141
142 LMMC_REAL_CLEAR(&sum);
143 LMMC_REAL_CLEAR(&tmp_mul);
144 LMMC_REAL_CLEAR(&tmp_sum);
145 return LMMC_STATUS_OK;
146}

148lmmc_status_t lmmc_mat_gemv(lmmc_real_t alpha, const lmmc_mat_t* A, int transA,
149 const lmmc_vec_t* x, lmmc_real_t beta, lmmc_vec_t* y) {
150 if (A == NULL || x == NULL || y == NULL ||
151 A->data == NULL || x->data == NULL || y->data == NULL) {
152 return LMMC_STATUS_INVALID_ARGUMENT;
153 }
154
155 /* Determine effective dimensions of op(A) */
156 size_t M = transA ? A->cols : A->rows;
157 size_t N = transA ? A->rows : A->cols;
158
159 /* Dimension validation */
160 if (N != x->size || M != y->size) {
161 return LMMC_STATUS_DIMENSION_MISMATCH;
162 }
163
164 {
165 lmmc_storage_envelope_t a_envelope;
166 lmmc_storage_envelope_t x_envelope;
167 lmmc_storage_envelope_t y_envelope;
168 if (!lmmc_storage_envelope_checked(A->data, A->rows, A->cols,
169 A->stride, sizeof(lmmc_real_t),
170 &a_envelope) ||
171 !lmmc_storage_envelope_checked(x->data, 1, x->size, x->size,
172 sizeof(lmmc_real_t), &x_envelope) ||
173 !lmmc_storage_envelope_checked(y->data, 1, y->size, y->size,
174 sizeof(lmmc_real_t), &y_envelope)) {
175 return LMMC_STATUS_INVALID_ARGUMENT;
176 }
177 if (lmmc_storage_envelopes_overlap(&y_envelope, &a_envelope) ||
178 lmmc_storage_envelopes_overlap(&y_envelope, &x_envelope)) {
179 return LMMC_STATUS_INVALID_ARGUMENT;
180 }
181 }
182
183 const lmmc_real_t* restrict a_data = A->data;
184 size_t a_stride = A->stride;
185
186
187 /* Scale y by beta */
188 if (beta == 0.0) {
189 for (size_t i = 0; i < M; ++i) {
190 LMMC_REAL_SET_D(&y->data[i], 0.0);
191 }
192 } else if (beta != 1.0) {
193 lmmc_real_t tmp_mul; LMMC_REAL_INIT(&tmp_mul);
194 for (size_t i = 0; i < M; ++i) {
195 LMMC_REAL_MUL(&tmp_mul, &beta, &y->data[i]);
196 LMMC_REAL_SET(&y->data[i], &tmp_mul);
197 }
198 LMMC_REAL_CLEAR(&tmp_mul);
199 }
200
201 /* If alpha is zero, we're done */
202 if (alpha == 0.0) {
203 return LMMC_STATUS_OK;
204 }
205
206 /* Native loop: y += alpha * op(A) * x */
207 lmmc_real_t sum; LMMC_REAL_INIT(&sum);
208 lmmc_real_t tmp_mul; LMMC_REAL_INIT(&tmp_mul);
209 lmmc_real_t tmp_sum; LMMC_REAL_INIT(&tmp_sum);
210 lmmc_real_t alpha_sum; LMMC_REAL_INIT(&alpha_sum);
211
212 for (size_t i = 0; i < M; ++i) {
213 LMMC_REAL_SET_D(&sum, 0.0);
214 for (size_t j = 0; j < N; ++j) {
215 lmmc_real_t a_ij;
216 if (transA) {
217 a_ij = a_data[j * a_stride + i];
218 } else {
219 a_ij = a_data[i * a_stride + j];
220 }
221 LMMC_REAL_MUL(&tmp_mul, &a_ij, &x->data[j]);
222 LMMC_REAL_ADD(&tmp_sum, &sum, &tmp_mul);
223 LMMC_REAL_SET(&sum, &tmp_sum);
224 }
225 /* y[i] += alpha * sum */
226 LMMC_REAL_MUL(&alpha_sum, &alpha, &sum);
227 LMMC_REAL_ADD(&tmp_sum, &y->data[i], &alpha_sum);
228 LMMC_REAL_SET(&y->data[i], &tmp_sum);
229 }
230
231 LMMC_REAL_CLEAR(&sum);
232 LMMC_REAL_CLEAR(&tmp_mul);
233 LMMC_REAL_CLEAR(&tmp_sum);
234 LMMC_REAL_CLEAR(&alpha_sum);
235 return LMMC_STATUS_OK;
236}
