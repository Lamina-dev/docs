25static inline uint64_t splitmix64_next(uint64_t* state) {
26 uint64_t z = (*state += UINT64_C(0x9e3779b97f4a7c15));
27 z = (z ^ (z >> 30)) * UINT64_C(0xbf58476d1ce4e5b9);
28 z = (z ^ (z >> 27)) * UINT64_C(0x94d049bb133111eb);
29 return z ^ (z >> 31);
30}

37static uint64_t generate_default_seed(void) {
38 static _Atomic uint64_t seed_counter = 0;
39
40 uint64_t t = (uint64_t)time(NULL);
41 /* Use address of a local variable for ASLR entropy */
42 volatile int stack_var = 0;
43 uint64_t addr = (uint64_t)(uintptr_t)&stack_var;
44 uint64_t counter = atomic_fetch_add(&seed_counter, 1);
45
46 /* Mix all sources via SplitMix64 finalizer */
47 uint64_t mixed = t ^ addr ^ counter;
48 return splitmix64_next(&mixed);
49}

65static uint64_t rng_bounded_u64(uint64_t* state, uint64_t bound) {
66 const uint64_t threshold = (uint64_t)(-bound) % bound;
67 uint64_t value;
68 do {
69 value = xoshiro256ss_next(state);
70 } while (value < threshold);
71 return value % bound;
72}

167static void rng_apply_jump_poly(uint64_t* s, const uint64_t* poly) {
168 uint64_t s0 = 0, s1 = 0, s2 = 0, s3 = 0;
169 int i, b;
170
171 for (i = 0; i < 4; i++) {
172 for (b = 0; b < 64; b++) {
173 if (poly[i] & (UINT64_C(1) << b)) {
174 s0 ^= s[0];
175 s1 ^= s[1];
176 s2 ^= s[2];
177 s3 ^= s[3];
178 }
179 xoshiro256ss_next(s);
180 }
181 }
182
183 s[0] = s0;
184 s[1] = s1;
185 s[2] = s2;
186 s[3] = s3;
187}

222 lmmc_real_t a,
223 lmmc_real_t b,
224 lmmc_real_t* out_value)
225{
226 double u;
227

236 *out_value = fma(b, u, a * (1.0 - u));

240lmmc_status_t lmmc_rng_exponential(
241 lmmc_rng_t* rng,
242 lmmc_real_t rate,
243 lmmc_real_t* out_value)
244{
245 double u;
246
247 if (rng == NULL || out_value == NULL) {
248 return LMMC_STATUS_INVALID_ARGUMENT;
249 }
250 if (!isfinite(rate) || rate <= 0.0) {
251 return LMMC_STATUS_INVALID_ARGUMENT;
252 }
253
254 do {
255 u = u64_to_double01(xoshiro256ss_next(rng->state));
256 } while (u == 1.0);
257
258 *out_value = -log(1.0 - u) / rate;
259 return LMMC_STATUS_OK;
260}

288lmmc_status_t lmmc_rng_shuffle(
289 lmmc_rng_t* rng,
290 void* array,
291 size_t count,
292 size_t elem_size)
293{
294 size_t i, j;
295 unsigned char* arr;
296 unsigned char* tmp;
297
298 if (rng == NULL || array == NULL) {
299 return LMMC_STATUS_INVALID_ARGUMENT;
300 }
301 if (elem_size == 0) {
302 return LMMC_STATUS_INVALID_ARGUMENT;
303 }
304
305 if (count <= 1) {
306 return LMMC_STATUS_OK;
307 }
308
309 arr = (unsigned char*)array;
310 tmp = (unsigned char*)lmmc_memory_alloc(elem_size);
311 if (tmp == NULL) {
312 return LMMC_STATUS_ALLOCATION_FAILED;
313 }
314
315 for (i = count - 1; i > 0; i--) {
316 j = (size_t)rng_bounded_u64(rng->state, (uint64_t)i + UINT64_C(1));
317
318 if (i != j) {
319 memcpy(tmp, arr + i * elem_size, elem_size);
320 memcpy(arr + i * elem_size, arr + j * elem_size, elem_size);
321 memcpy(arr + j * elem_size, tmp, elem_size);
322 }
323 }
324
325 lmmc_memory_free(tmp);
326 return LMMC_STATUS_OK;
327}

329lmmc_status_t lmmc_rng_int_uniform(
330 lmmc_rng_t* rng,
331 int64_t lo,
332 int64_t hi,
333 int64_t* out)
334{
335 uint64_t span, offset, bits;
336
337 if (rng == NULL || out == NULL) {
338 return LMMC_STATUS_INVALID_ARGUMENT;
339 }
340 if (lo > hi) {
341 return LMMC_STATUS_INVALID_ARGUMENT;
342 }
343 if (lo == hi) {
344 *out = lo;
345 return LMMC_STATUS_OK;
346 }
347
348 span = (uint64_t)hi - (uint64_t)lo + UINT64_C(1);
349 offset = span == 0 ? xoshiro256ss_next(rng->state)
350 : rng_bounded_u64(rng->state, span);
351 bits = (uint64_t)lo + offset;
352 if (bits <= (uint64_t)INT64_MAX) {
353 *out = (int64_t)bits;
354 } else {
355 *out = INT64_MIN + (int64_t)(bits - (UINT64_C(1) << 63));
356 }
357 return LMMC_STATUS_OK;
