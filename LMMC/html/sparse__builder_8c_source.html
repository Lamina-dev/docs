133lmmc_status_t lmmc_sparse_builder_build(lmmc_sparse_builder_t* b, lmmc_sparse_format_t format, lmmc_sparse_mat_t* out) {
134 lmmc_status_t st;
135 size_t i, outer_dim;
136 if (b == NULL || out == NULL) return LMMC_STATUS_INVALID_ARGUMENT;
137
138 st = (format == LMMC_SPARSE_CSR) ? lmmc_sparse_create_csr(b->rows, b->cols, b->nnz, out)
139 : lmmc_sparse_create_csc(b->rows, b->cols, b->nnz, out);
140 if (st != LMMC_STATUS_OK) return st;
141
142 outer_dim = (format == LMMC_SPARSE_CSR) ? b->rows : b->cols;
143 size_t* major = (format == LMMC_SPARSE_CSR) ? b->r_idx : b->c_idx;
144 size_t* minor = (format == LMMC_SPARSE_CSR) ? b->c_idx : b->r_idx;
145
146
147 for (i = 0; i < b->nnz; ++i) {
148 out->row_ptr[major[i] + 1]++;
149 }
150
151 for (i = 0; i < outer_dim; ++i) {
152 out->row_ptr[i+1] += out->row_ptr[i];
153 }
154
155
156 size_t* next = (size_t*)lmmc_alloc(outer_dim * sizeof(size_t));
157 if (next == NULL) {
158 lmmc_sparse_destroy(out);
159 return LMMC_STATUS_ALLOCATION_FAILED;
160 }
161 memcpy(next, out->row_ptr, outer_dim * sizeof(size_t));
162
163 for (i = 0; i < b->nnz; ++i) {
164 size_t idx = next[major[i]]++;
165 out->col_idx[idx] = minor[i];
166 LMMC_REAL_SET(&out->values[idx], &b->vals[i]);
167 }
168
169 lmmc_free(next);
170 return LMMC_STATUS_OK;
171}
