22static lmmc_status_t lmmc_sparse_entries_grow(
23 size_t nnz, size_t* capacity, size_t** rows, size_t** cols,
24 lmmc_real_t** values)
25{
26 size_t new_cap = 0;
27 size_t sz_idx = 0;
28 size_t sz_vals = 0;
29 size_t* nr = NULL;
30 size_t* nc = NULL;
31 lmmc_real_t* nv = NULL;
32
33 if (!lmmc_safe_mul_size(*capacity, 2, &new_cap) ||
34 !lmmc_safe_mul_size(new_cap, sizeof(size_t), &sz_idx) ||
35 !lmmc_safe_mul_size(new_cap, sizeof(lmmc_real_t), &sz_vals)) {
36 return LMMC_STATUS_ALLOCATION_FAILED;
37 }
38
39 nr = (size_t*)lmmc_memory_alloc(sz_idx);
40 nc = (size_t*)lmmc_memory_alloc(sz_idx);
41 nv = (lmmc_real_t*)lmmc_memory_alloc(sz_vals);
42
43 if (nr == NULL || nc == NULL || nv == NULL) {
44 if (nr) {
45 lmmc_memory_free(nr);
46 }
47 if (nc) {
48 lmmc_memory_free(nc);
49 }
50 if (nv) {
51 lmmc_memory_free(nv);
52 }
53 return LMMC_STATUS_ALLOCATION_FAILED;
54 }
55
56 for (size_t k = 0; k < new_cap; ++k) {
57 LMMC_REAL_INIT(&nv[k]);
58 }
59
60 if (nnz > 0) {
61 memcpy(nr, *rows, nnz * sizeof(size_t));
62 memcpy(nc, *cols, nnz * sizeof(size_t));
63 for (size_t k = 0; k < nnz; ++k) {
64 LMMC_REAL_SET(&nv[k], &(*values)[k]);
65 }
66 }
67
68 for (size_t k = 0; k < *capacity; ++k) {
69 LMMC_REAL_CLEAR(&(*values)[k]);
70 }
71 lmmc_memory_free(*rows);
72 lmmc_memory_free(*cols);
73 lmmc_memory_free(*values);
74
75 *rows = nr;
76 *cols = nc;
77 *values = nv;
78 *capacity = new_cap;
79 return LMMC_STATUS_OK;
80}

83lmmc_status_t lmmc_sparse_builder_create(size_t rows, size_t cols, size_t initial_capacity, lmmc_sparse_builder_t** out_builder) {
84 lmmc_sparse_builder_t* b = NULL;

89 b = (lmmc_sparse_builder_t*)lmmc_memory_alloc(sizeof(lmmc_sparse_builder_t));

94 b->rows = rows;
95 b->cols = cols;
96 b->nnz = 0;
97 b->capacity = initial_capacity > 0 ? initial_capacity : 16;

99 size_t sz_idx = 0;
100 size_t sz_vals = 0;

107 b->r_idx = (size_t*)lmmc_memory_alloc(sz_idx);
108 b->c_idx = (size_t*)lmmc_memory_alloc(sz_idx);
109 b->vals = (lmmc_real_t*)lmmc_memory_alloc(sz_vals);
110
111 if (b->r_idx == NULL || b->c_idx == NULL || b->vals == NULL) {
112 lmmc_sparse_builder_destroy(b);
113 return LMMC_STATUS_ALLOCATION_FAILED;
114 }
115
116 for (size_t k = 0; k < b->capacity; ++k) {
117 LMMC_REAL_INIT(&b->vals[k]);
118 }
119
120 *out_builder = b;
121 return LMMC_STATUS_OK;
122}

124lmmc_status_t lmmc_sparse_builder_add(lmmc_sparse_builder_t* b, size_t row, size_t col, lmmc_real_t val) {
125 if (b == NULL || row >= b->rows || col >= b->cols) {
126 return LMMC_STATUS_INVALID_ARGUMENT;
127 }
128
129 if (b->nnz >= b->capacity) {
130 lmmc_status_t st = lmmc_sparse_entries_grow(
131 b->nnz, &b->capacity, &b->r_idx, &b->c_idx, &b->vals);
132 if (st != LMMC_STATUS_OK) {
133 return st;
134 }
135 }
136
137 b->r_idx[b->nnz] = row;
138 b->c_idx[b->nnz] = col;
139 LMMC_REAL_SET(&b->vals[b->nnz], &val);
140 b->nnz++;
141 return LMMC_STATUS_OK;
142}

162lmmc_status_t lmmc_sparse_builder_build(lmmc_sparse_builder_t* b, lmmc_sparse_format_t format, lmmc_sparse_mat_t* out) {
163 if (b == NULL || out == NULL ||
164 (format != LMMC_SPARSE_CSR && format != LMMC_SPARSE_CSC)) {
165 return LMMC_STATUS_INVALID_ARGUMENT;
166 }
167 lmmc_sparse_coo_t coo = {0};
168 coo.rows = b->rows;
169 coo.cols = b->cols;
170 coo.nnz = b->nnz;
171 coo.capacity = b->capacity;
172 coo.row_idx = b->r_idx;
173 coo.col_idx = b->c_idx;
174 coo.values = b->vals;
175 return format == LMMC_SPARSE_CSR
176 ? lmmc_sparse_coo_to_csr(&coo, out)
177 : lmmc_sparse_coo_to_csc(&coo, out);
178}
